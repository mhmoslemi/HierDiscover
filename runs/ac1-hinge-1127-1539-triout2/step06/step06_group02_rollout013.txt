def _adam_gradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    beta_temp = 1.0
    beta_max = 1e5
    best_a = a.copy()
    best_score = _score(a)
    while time.time() < deadline:
        # compute gradient via softmax approximate max
        g, max_b = _softmax_gradient(a, beta_temp)
        t_iter += 1
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        lr = 0.1 * target_sum  # base
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project to simplex
        a = _project_to_simplex(a, target_sum)
        # maybe add jitter occasionally
        if t_iter % 500 == 0:
            jitter_scale = target_sum * 0.001 / (beta_temp+1)
            a += np.random.default_rng().normal(scale=jitter_scale, size=n)
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)
        # evaluate
        val = _score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam beta={beta_temp:.1f} score={best_score:.6f}")
        # increase beta
        if t_iter % 2000 == 0 and beta_temp < beta_max:
            beta_temp = min(beta_temp * 1.5, beta_max)
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0
    best_score = float('inf')
    best_seq = None
    sizes = [2000, 4000, 8000]
    for n in sizes:
        if time.time() >= deadline - 5:
            break
        target = np.sqrt(2.0 * n)
        # initialization
        mode = rng.choice(['global','triangular','power','random'])
        if mode == 'global' and 'height_sequence_1' in globals():
            seq = np.array(globals()['height_sequence_1'], dtype=float)
            if seq.size > n:
                seq = seq[:n]
            else:
                seq = np.pad(seq, (0, n - seq.size), 'constant')
        elif mode == 'triangular':
            i = np.arange(n)
            seq = (i+1) * (n - i)
            seq = seq.astype(float)
            seq = seq / seq.sum() * target
        elif mode == 'power':
            seq = _power_law_sequence(n, 0.8, target)
        else:  # random
            seq = rng.random(n)
            seq = seq / seq.sum() * target
        seq = _project_to_simplex(seq, target)

        # gradient descent (Adam)
        remaining = deadline - time.time()
        if remaining > 20:
            gd_deadline = time.time() + min(30.0, remaining * 0.5)
            seq, gd_score = _adam_gradient_descent(seq, gd_deadline)
        else:
            gd_score = _score(seq)

        # SA
        cur_time = time.time()
        if cur_time < deadline:
            seq, sa_score = _simulated_annealing(seq, deadline)
            cur_score = min(gd_score, sa_score)
        else:
            cur_score = gd_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} mode={mode} new best={best_score:.6f}")

        # maybe break early if score near target
        if best_score <= 1.2:
            break
    # final refinement

"""
Propose a step‑function sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages:

1️⃣  Multi‑scale initialisation (global, power‑law, triangular or random).
2️⃣  Projected Adam optimisation on a soft‑max approximation of the max‑convolution.
    The temperature β of the soft‑max is increased gradually up to 1e5,
    while Adam’s adaptive steps keep the updates stable.
3️⃣  A short simulated‑annealing (SA) refinement that moves mass between
    indices that contribute most to the current worst convolution entry.

All iterates are projected onto the ℓ¹‑simplex Σa = √(2·n) after each update,
so the evaluation function `evaluate_sequence` (scale‑invariant) always sees a
properly normalised candidate.  The best sequence found before the time budget
expires is returned (trailing zeros are stripped).

The implementation is completely self‑contained, uses only NumPy and the
provided `evaluate_sequence`, and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective for a normalised sum (Σa = √(2·n))."""
    n = a.size
    max_b = _max_convolution_fft(a)
    # after normalisation Σa² = 2·n, thus the score equals max_b
    return max_b


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.  Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (gradient, raw convolution array).  The gradient is
    ∑ₖ pₖ ∂bₖ/∂a where pₖ = softmax(β·bₖ).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max weights, numerically stable
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected Adam optimiser on the soft‑max surrogate of the max convolution.
    The temperature β is increased up to 1e5.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # temperature schedule
    beta_temp = 1.0
    beta_max = 1e5

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    rng = np.random.default_rng()
    # a small base learning rate, later scaled by Adam’s adaptive denominator
    base_lr = 0.05 * target

    while time.time() < deadline:
        # gradient of the smoothed objective
        g, _ = _gradient_softmax(a, beta_temp)

        it += 1
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_to_simplex(a, target)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta_temp + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # evaluate
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam β={beta_temp:.1f} iter={it} score={best_val:.6f}")

        # increase temperature slowly
        if it % 2000 == 0 and beta_temp < beta_max:
            beta_temp = min(beta_temp * 1.5, beta_max)

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.005,
                         max_window: int = 35):
    """
    A lightweight simulated‑annealing phase that moves mass between indices.
    Moves are:
      * donor → receiver mass transfer (≈ 70 %),
      * short‑window smoothing (≈ 20 %),
      * random swap (≈ 10 %).
    The temperature and the allowed transfer fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)          # current convolution

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate the worst convolution entry
        k_max = int(np.argmax(conv))

        # compute the contribution weight of each coordinate to that entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]            # a_j where i+j = k_max
        weight = a[idx] * sens           # a_i * a_{k_max-i}

        move_type = None
        # ---- donor → receiver mass transfer ---------------------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # pick a receiver that contributes little to the worst entry
            low_thr = np.percentile(weight, 5)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ---- short‑window smoothing -----------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        # ---- random swap ----------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ---- evaluate new state (single FFT) ----------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ---- Metropolis acceptance -------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter (helps escape deep minima)
        if it % 6000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _run_one_instance(n: int, init_mode: str, deadline: float):
    """
    Optimise a single problem size.
    Returns the best sequence and its score.
    """
    target = np.sqrt(2.0 * n)

    # ---- initialisation -------------------------------------------------
    if init_mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif init_mode == "triangular":
        i = np.arange(n, dtype=np.float64)
        seq = (i + 1) * (n - i)          # simple triangular bump
        seq = seq / seq.sum() * target
    elif init_mode == "power":
        seq = _power_law_sequence(n, 0.75, target)
    else:  # random
        rng = np.random.default_rng()
        seq = rng.random(n)
        seq = seq / seq.sum() * target

    seq = _project_to_simplex(seq, target)

    # ---- Adam gradient descent (up to 30 s or 40 % of remaining) -----
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + min(30.0, remaining * 0.40)
        seq, gd_score = _adam_gradient_descent(seq, gd_deadline)
    else:
        gd_score = _score(seq)

    # ---- Simulated annealing --------------------------------------------
    cur_time = time.time()
    if cur_time < deadline:
        seq, sa_score = _simulated_annealing(seq, deadline)
        final_score = min(gd_score, sa_score)
    else:
        final_score = gd_score

    return seq, final_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Runs a few independent optimisation instances of different lengths,
    each consisting of a projected‑Adam phase followed by a simulated‑annealing
    refinement.  The best sequence found before the time budget expires is
    returned (trailing zeros are stripped).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes – larger sizes give more freedom; we keep the list short
    # to stay within the time budget.
    sizes = [2000, 4000, 8000]

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break
        # allocate a slice of the remaining time to this size
        remaining = deadline - time.time()
        # keep at least 15 s for the final refinement
        time_for_this = min(remaining * 0.45, 250.0)

        init_modes = ["global", "triangular", "power", "random"]
        init_mode = init_modes[rng.integers(len(init_modes))]

        run_deadline = time.time() + time_for_this
        seq, score = _run_one_instance(n, init_mode, run_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} init={init_mode} → new best score={best_score:.6f}")

        # early exit if we already beat the target bound
        if best_score <= 1.20:
            break

    # ------------------------------------------------------------------
    # Final short Adam refinement on the best candidate (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _adam_gradient_descent(best_seq, extra_deadline)

    # Strip trailing zeros (they do not influence the score)
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
