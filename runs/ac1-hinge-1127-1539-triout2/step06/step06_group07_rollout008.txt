def _max_feasible_step(conv, D, cur_max, a_i):

    # D: length 2n-1

def _refine_with_mass_moves(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    sum_target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(a)
        cur_max = np.max(conv)
        # indices where within tolerance of max
        # Use absolute tolerance relative maybe 1e-12.
        max_idxs = np.flatnonzero(conv >= cur_max - 1e-12)
        # Randomly choose one if many
        k = rng.choice(max_idxs)

        # Determine range for i indices that contribute to k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idxs = np.arange(i_low, i_high + 1)
        # contributions from a at positions idx_contrib = k - i
        contrib_vals = a[k - idxs]  # because a index = k - i
        i_max = idxs[np.argmax(contrib_vals)]

        # pick j as argmin a (lowest value)
        j_candidates = np.where(a < np.percentile(a, 10))[0]  # low values
        if j_candidates.size == 0:
            j = rng.integers(0, n)
        else:
            # choose random low
            j = rng.choice(j_candidates)

        # Ensure j != i_max
        if j == i_max:
            # choose next min
            j = (j + 1) % n

        # Compute D = 2*(shift_j - shift_i)
        shift_i = np.concatenate([np.zeros(i_max), a, np.zeros(n - 1 - i_max)])
        shift_j = np.concatenate([np.zeros(j), a, np.zeros(n - 1 - j)])
        D = 2.0 * (shift_j - shift_i)

        eps_max = _max_feasible_step(conv, D, cur_max, a[i_max])

        if eps_max <= 1e-12:
            # fallback random mass swap small
            i = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if i != j2 and a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j2] += delta
                a = _project_simplex(a, sum_target)
            continue

        epsilon = min(eps_max * 0.6, a[i_max])  # safety factor
        a[i_max] -= epsilon
        a[j] += epsilon

        # enforce simplex (may be minor inaccuracies)
        a = _project_simplex(a, sum_target)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # early stop condition: if no improvement over many iterations
        # not needed

    return best_a, best_score

    external = evaluate_sequence(final_seq)

def _full_convolution(a):
   n = a.shape[0]
   conv_len = 2 * n - 1
   fft_len = 1 << (conv_len - 1).bit_length()
   Fa = np.fft.rfft(a, n=fft_len)
   conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
   return conv

"""
Propose a non‑negative step function that (approximately) minimizes the
maximum entry of its self‑convolution.

Algorithm (≈1000 s budget)
---------------------------
1. Choose a problem size n (4000 if time allows) and initialise a vector
   on the ℓ¹‑simplex Σ a_i = √(2 n).  If a previous construction is
   available (global ``height_sequence_1``) it is up‑sampled, otherwise a
   random Dirichlet point is used.

2. **Adam phase** – minimise a smooth soft‑max approximation of the
   max‑convolution using Adam, projecting onto the simplex after each
   update.

3. **Sub‑gradient phase** – finish the Adam run with a true
   sub‑gradient descent on the max‑entry.

4. **Mass‑move refinement** – repeatedly move a small amount of mass
   from a coefficient that contributes most to the current worst
   convolution entry to a coefficient that is currently smallest.
   The move size is chosen analytically so that no other entry exceeds
   the current maximum, guaranteeing a monotone improvement.

5. **Lightweight GA** – a steady‑state genetic algorithm is applied for a
   few seconds to escape possible local minima.

6. The best vector found is returned as a plain Python list.  The
   external ``evaluate_sequence`` function is called once at the end
   for a sanity check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max helpers (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j) (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear up‑sampling
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on soft‑max approximation
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth soft‑max of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5               # start smooth
    alpha_max = 1500.0        # final sharpness

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a, S)

        # occasional random mass swap for exploration
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        alpha = min(alpha * 1.05, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent directly on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined Adam → sub‑gradient optimiser
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.7) -> tuple[np.ndarray, float]:
    """Run Adam for a fraction of the time, then Sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Simple steady‑state genetic optimiser
# ----------------------------------------------------------------------


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very light GA that respects the simplex."""
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 12 if n > 2000 else 20

    pop = []
    scores = []
    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # choose two distinct parents
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)

            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            if rng.random() < 0.35:
                noise = rng.normal(scale=child.mean() * 0.05 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # diversity injection
        if rng.random() < 0.12:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # break a little early if time is almost up
        if time.time() + 0.1 >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Mass‑move refinement (monotone reduction of max‑convolution)
# ----------------------------------------------------------------------


def _max_feasible_step(conv: np.ndarray,
                       D: np.ndarray,
                       cur_max: float,
                       a_i: float) -> float:
    """Largest epsilon ≥ 0 such that conv + eps·D never exceeds cur_max,
       and a_i – eps stays non‑negative."""
    pos = D > 1e-12
    if np.any(pos):
        eps_candidate = (cur_max - conv[pos]) / D[pos]
        eps_max = np.min(eps_candidate)
    else:
        eps_max = np.inf
    eps_max = min(eps_max, a_i)
    return eps_max


def _refine_with_mass_moves(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from a coefficient that
    contributes most to the current worst convolution entry to a
    coefficient that is (currently) smallest.  The step size is chosen
    analytically to keep the max unchanged or reduced, guaranteeing a
    monotone improvement.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        conv = _full_convolution(a)
        cur_max = np.max(conv)

        # indices attaining the max (may be a few due to ties)
        max_idxs = np.flatnonzero(conv >= cur_max - 1e-12)
        k = rng.choice(max_idxs)

        # range of i that actually participates in this entry
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idxs = np.arange(i_low, i_high + 1)

        # contribution of a[i] to entry k is a[k - i]
        contribs = a[k - idxs]
        i_max = idxs[np.argmax(contribs)]

        # target index j = index with minimal coefficient
        j = int(np.argmin(a))
        if j == i_max:
            # pick the second smallest if the absolute smallest coincides
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != i_max:
                    j = cand
                    break

        # build the two shifted vectors needed for the directional derivative
        shift_i = np.concatenate([np.zeros(i_max), a, np.zeros(n - 1 - i_max)])
        shift_j = np.concatenate([np.zeros(j), a, np.zeros(n - 1 - j)])
        D = 2.0 * (shift_j - shift_i)

        eps_max = _max_feasible_step(conv, D, cur_max, a[i_max])

        if eps_max <= 1e-12:
            # tiny random perturbation to avoid being stuck
            i = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if i != j2 and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j2] += delta
                a = _project_simplex(a, target_sum)
            continue

        # safety factor to stay comfortably inside the feasible region
        epsilon = 0.6 * eps_max
        a[i_max] -= epsilon
        a[j] += epsilon
        a = _project_simplex(a, target_sum)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine runs a large‑scale Adam/sub‑gradient
    phase, followed by a deterministic mass‑move refinement and a brief
    genetic algorithm.  It works primarily on n = 4000 (if time permits);
    otherwise it falls back to n = 2000.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the eventual evaluation call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered sequence if available
    # ------------------------------------------------------------------
    best_seq = None
    if 'height_sequence_1' in globals():
        try:
            best_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            best_seq = None

    best_score = np.inf
    last_improve = start_time

    # ------------------------------------------------------------------
    # Choose problem size (biggest we can afford)
    # ------------------------------------------------------------------
    # we try 4000 first; if time runs out we will automatically shrink to
    # the next size because the inner loops respect the deadline.
    for n in (4000, 2000, 800):
        if time.time() >= deadline:
            break

        # initialise
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)
        # small jitter to break exact symmetry
        jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
        a0 = np.clip(a0 + jitter, 0.0, None)

        # project onto the ℓ¹‑simplex Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        a0 = a0 / a0.sum() * S

        # --------------------------------------------------------------
        # Phase 1 – Adam + Sub‑gradient (majority of the remaining time)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        phase1_deadline = time.time() + min(remaining * 0.65, remaining - 5.0)
        a_opt, sc_opt = _optimize_combined_adam(a0, phase1_deadline, rng,
                                                fraction_adam=0.78)

        # --------------------------------------------------------------
        # Phase 2 – deterministic mass‑move refinement
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            refine_deadline = time.time() + min(10.0, deadline - time.time())
            a_ref, sc_ref = _refine_with_mass_moves(a_opt, refine_deadline, rng)
            if sc_ref < sc_opt:
                a_opt, sc_opt = a_ref, sc_ref

        # --------------------------------------------------------------
        # Phase 3 – brief GA to escape local minima
        # --------------------------------------------------------------
        if time.time() < deadline - 0.5:
            ga_deadline = time.time() + min(5.0, deadline - time.time())
            a_ga, sc_ga = _run_genetic_optimizer(a_opt, ga_deadline, rng)
            if sc_ga < sc_opt:
                a_opt, sc_opt = a_ga, sc_ga

        # update global best
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[stage] n={n} new global best score={best_score:.6f}")

        # shrink the total remaining budget for the next (smaller) stage
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
