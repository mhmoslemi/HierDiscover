def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)  # which assumes sum normalized
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution
        conv = _conv_fft(a)  # length 2n-1
        max_val = conv.max()
        # find maximal indices (list)
        max_indices = np.where(conv == max_val)[0]
        # compute gradient as sum of contributions
        # g_i = sum_{k ∈ max_indices} 2 * a_{k-i} if 0 <= k-i < n else 0
        # We can compute using convolution: create indicator vector p of length 2n-1
        p = np.zeros_like(conv)
        p[max_indices] = 1.0/len(max_indices)  # average contributions
        # gradient: g = 2 * conv_rev(p) where conv_rev is convolution of reversed a with p
        a_rev = a[::-1]
        size = 1 << ((2 * n - 1).bit_length())
        fa_rev = np.fft.rfft(a_rev, n=size)
        fp = np.fft.rfft(p, n=size)
        grad = 2.0 * np.fft.irfft(fa_rev * fp, n=size)[:n]
        # step size
        norm_g = np.linalg.norm(grad)
        if norm_g == 0:
            break
        # diminishing step size
        # Could use step = step0 / (1 + it * decay)
        step0 = target_sum * 0.5 # typical magnitude
        step = step0 / (norm_g * (1 + it * 0.001))
        a = a - step * grad
        # ensure nonnegative

"""
A fast two‑phase optimizer for the max‑convolution objective.

Phase 1 – Projected sub‑gradient descent on the true max (no soft‑max
approximation).  The gradient of max (a*a) is easy to compute:
for each index k attaining the maximum we have
∂b_k/∂a_i = 2·a_{k‑i} (if the index is in range).  The algorithm moves
a in the negative gradient direction, projects back onto the ℓ¹‑simplex
Σa = √(2·n) and repeats with a diminishing step size.

Phase 2 – A short simulated‑annealing refinement (the same moves as the
previous state‑of‑the‑art code) to escape any remaining plateaus.

Multiple problem sizes (800, 1500, 3000, 5000) are tried with different
initialisations (global warm‑start, random power‑law, random Dirichlet).
The best sequence found before the 1000 s budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / convolution helpers
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective (identical to evaluate_sequence
    when Σa = √(2·n)).  The function is cheap because it avoids the
    safety checks of evaluate_sequence.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0,
    Σx_i = s}.  Implements the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation utilities
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(‑p) and renormalise to the given ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _init_sequence(n: int, mode: str, rng: np.random.Generator) -> np.ndarray:
    """
    Create an initial sequence of length n.
    mode ∈ {'global','power','random'}.
    """
    target = np.sqrt(2.0 * n)

    if mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    elif mode == "power":
        # random exponent in a sensible interval
        p = rng.uniform(0.35, 0.75)
        seq = _power_law_sequence(n, p, target)
    else:   # 'random'
        seq = rng.random(n)
        seq = seq / seq.sum() * target

    # ensure feasibility
    seq = _project_to_simplex(seq, target)
    return seq


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _subgradient_of_max(a: np.ndarray):
    """
    Compute the sub‑gradient of max(a*a) and the current maximal value.
    Returns (grad, max_val) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    max_idxs = np.where(conv == max_val)[0]

    grad = np.zeros_like(a)

    # each k contributes 2*a_{k-i} to coordinate i (if in range)
    for k in max_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        grad[i] += 2.0 * a[j]

    return grad, max_val


def _subgradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on max(a*a).  A simple diminishing
    step‑size is used together with occasional jitter to avoid stagnation.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    # base step proportional to the size of the simplex
    step_base = target_sum * 0.6

    while time.time() < deadline:
        grad, max_val = _subgradient_of_max(a)

        norm_g = np.linalg.norm(grad)
        if norm_g == 0:
            break

        # diminishing step – 1/√(iteration) schedule
        step = step_base / (norm_g * (1.0 + np.sqrt(it)))
        a -= step * grad

        # projection back to the feasible simplex
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it} score={best_val:.6f}")

        # tiny random jitter every few hundred steps
        if it % 400 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (same moves as the previous best program)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-5,
    max_frac_start: float = 0.40,
    max_frac_end: float = 0.005,
    max_window: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Local search that moves mass between indices that contribute most to
    the current worst convolution entry, performs short‑window smoothing
    and occasional swaps.  Temperature decays exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # worst convolution entry and contributions
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to the worst entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        weight = a[idx] * a[k_max - idx]          # a_i * a_{k_max-i}

        # ------------------------------------------------------------------
        # propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:      # mass transfer
            total_w = weight.sum()
            donor = rng.choice(idx, p=weight / total_w) if total_w > 0 else rng.integers(n)

            # choose a receiver that contributes little to the worst entry
            low_thr = np.percentile(weight, 10)
            low_cand = idx[weight <= low_thr]
            receiver = rng.choice(low_cand) if low_cand.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:    # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved = a[start:end].copy()
                avg = saved.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        else:                        # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # evaluate new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    evaluate_sequence.  The routine tries several problem sizes,
    multiple initialisations, a projected sub‑gradient phase and a
    simulated‑annealing refinement.  The best sequence found before the
    time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes – larger gives more freedom but costs more time.
    sizes = [800, 1500, 3000, 5000]

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break

        # allocate a slice of the remaining budget for this size
        remaining = deadline - time.time()
        time_for_n = min(remaining * 0.30, 200.0)   # max 200 s per size

        # try a few different initialisation modes
        init_modes = ["global", "power", "random"]
        for mode in init_modes:
            if time.time() >= deadline - 5.0:
                break

            a = _init_sequence(n, mode, rng)

            # -------- sub‑gradient phase (≈40 % of the allocated time) ----------
            sub_deadline = time.time() + min(time_for_n * 0.40,
                                            (deadline - time.time()) * 0.30)
            a, sub_score = _subgradient_descent(a, sub_deadline)

            # -------- simulated annealing phase (use the rest) -----------------
            if time.time() < deadline:
                a, sa_score = _simulated_annealing(a, deadline)
                final_score = min(sub_score, sa_score)
            else:
                final_score = sub_score

            if final_score < best_score:
                best_score = final_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] size={n} init={mode} → new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tuning if time permits
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        extra_deadline = time.time() + min((deadline - time.time()) * 0.20, 30.0)
        best_seq, _ = _subgradient_descent(best_seq, extra_deadline)

    # Strip trailing zeros – they do not affect the score.
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
