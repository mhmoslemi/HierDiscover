Algorithm:

5. w = np.maximum(v - theta, 0).

"""
adversarial step‑function optimizer – multi‑phase explorer

The optimizer searches a non‑negative vector ``a`` of length ``n`` with
ℓ₁‑norm ``S = sqrt(2·n)`` that minimises the peak of the self‑convolution
``c = a*a``.  Three phases are used:

1. **Exploration** – many short Adam runs with diverse initial shapes
   (uniform, Dirichlet, edge‑heavy, monotone power‑law, exponential,
   Gaussian).  Lengths are sampled from a wide interval and the best
   candidate is kept.

2. **Refinement** – the current best sequence is optionally scaled to
   larger lengths and refined with a longer Adam run.

3. **Polishing** – a projected‑gradient descent with the exact
   sub‑gradient of the max‑convolution, followed by a light simulated‑
   annealing pass, further reduces the peak.

All steps respect the non‑negativity and ℓ₁‑norm constraint; the final
candidate is stored in the global ``height_sequence_1`` variable."

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Projection of ``v`` onto the simplex {x≥0, sum(x)=S}."""
    # Duchi et al. (2008) – O(n) algorithm
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - S
    rho_arr = u - css / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) and the corresponding maximum."""
    b = _full_convolution_fft(a)
    max_b = b.max()
    # tolerance to collect all active indices (helps when ties are floating‑point)
    mask = (np.abs(b - max_b) < 1e-12).astype(float)
    k = mask.sum()
    if k == 0:
        mask[:] = 1.0 / mask.size
    else:
        mask /= k
    a_rev = a[::-1]
    conv_len = mask.size + a.size - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[a.size - 1:a.size - 1 + a.size]
    return max_b, grad

# Adam optimiser on the simplex (log‑softmax parametrisation)

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a τ‑soft‑max approximation of ``max(a*a)``.
    Returns the best encountered vector and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible start vector
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.35                      # a slightly larger initial learning rate
    # initial temperature for soft‑max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & τ‑soft‑max gradient
        # (re‑use the same routine as in the reference implementation)
        b = _full_convolution_fft(a)
        max_b = b.max()

        # mask for soft‑max (τ‑smoothed)
        if tau < 1e-12:
            mask = (b == max_b).astype(float)
        else:
            diff = b - max_b
            exps = np.exp(diff / tau)
            w = exps / exps.sum()
            mask = w

        # gradient w.r.t. a
        a_rev = a[::-1]
        conv_len = mask.size + a.size - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[a.size - 1:a.size - 1 + a.size]

        # keep best hard peak
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # gradient clipping (helps stability)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9995          # tighter soft‑max
        lr *= 0.9999           # slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak

def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law shape (symmetric w.r.t. both ends)."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 1.0)         # exponent
    offset = rng.uniform(0.0, 5.0)    # denominator shift
    mix = rng.uniform(0.0, 1.0)       # mix proportion
    edge_part = (d + offset) ** (-p)
    raw = mix * edge_part + (1.0 - mix) * np.ones(n)
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _mono_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone decreasing power‑law (left‑heavy)."""
    S = np.sqrt(2.0 * n)
    p = rng.uniform(0.2, 2.0)
    offset = rng.uniform(0.0, 5.0)
    i = np.arange(n)
    raw = (i + offset) ** (-p)
    raw = raw / raw.sum() * S
    return raw


def _exp_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay shape."""
    S = np.sqrt(2.0 * n)
    beta = rng.uniform(0.001, 0.02) * n
    i = np.arange(n)
    raw = np.exp(-beta * i)
    raw = raw / raw.sum() * S
    return raw


def _gauss_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian bump (centered optionally off‑centre)."""
    S = np.sqrt(2.0 * n)
    mu = rng.uniform(0, n - 1)
    sigma = rng.uniform(0.1, 0.5) * n
    i = np.arange(n)
    raw = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    raw = raw / raw.sum() * S
    return raw


def _uniform_initial(n: int) -> np.ndarray:
    """Flat uniform distribution (feasible but usually sub‑optimal)."""
    S = np.sqrt(2.0 * n)
    return np.full(n, S / n)


def _dirichlet_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet‑like start."""
    S = np.sqrt(2.0 * n)
    raw = rng.random(n)
    raw = raw / raw.sum() * S
    return raw


def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick one of the families uniformly at random."""
    style = rng.choice(['uniform', 'dirichlet', 'edge', 'mono', 'exp', 'gauss'])
    if style == 'uniform':
        return _uniform_initial(n)
    if style == 'dirichlet':
        return _dirichlet_initial(n, rng)
    if style == 'edge':
        return _edge_initial(n, rng)
    if style == 'mono':
        return _mono_initial(n, rng)
    if style == 'exp':
        return _exp_initial(n, rng)
    # gauss
    return _gauss_initial(n, rng)

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs three phases (exploration, refinement, polishing)
    within the given time budget.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Reuse a previously found construction if present
    # --------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # --------------------------------------------------------------
    # Phase 1 – many short Adam restarts (exploration)
    # --------------------------------------------------------------
    short_budget = 6.0   # seconds per short run
    while time.time() + short_budget < end_time - 120.0:
        attempt += 1

        # choose a length (mix of moderate and larger sizes)
        if rng.random() < 0.7:
            n = rng.integers(1500, 8000)
        else:
            n = rng.integers(8000, 40000)

        # occasionally warm‑start from a previously found shape
        if base_seq is not None and rng.random() < 0.12:
            a_start = _scale_sequence(base_seq, n)
            S = np.sqrt(2.0 * n)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum() * S
        else:
            a_start = _random_initial(n, rng)

        slice_deadline = min(time.time() + short_budget, end_time - 120.0)
        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng)

        score = peak  # because sum = sqrt(2n) ⇒ evaluate = max_b
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart] attempt={attempt} n={n}  new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – refinement on the current best (including optional scaling)
    # --------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        # try a couple of scalings (including staying at the original size)
        for factor in [1.0, 1.2, 1.5]:
            if remaining < 20.0:
                break
            n_target = int(best_n * factor)
            if n_target < 2:
                continue
            seq_scaled = _scale_sequence(best_seq, n_target)
            S = np.sqrt(2.0 * n_target)
            seq_scaled = np.maximum(seq_scaled, 0.0)
            seq_scaled = seq_scaled / seq_scaled.sum() * S
            # allocate a modest share of the remaining budget
            portion = min(remaining * 0.25, 120.0)
            deadline = time.time() + portion
            a_opt, peak = _adam_optimize(seq_scaled, deadline, rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_opt.copy()
                best_n = n_target
                print(f"[refine] scale={factor:.2f}  n={n_target}  new best={best_score:.6f}")
            remaining = end_time - time.time()

    # --------------------------------------------------------------
    # Phase 3 – projected gradient descent with exact sub‑gradient,
    #          followed by a brief simulated‑annealing polish.
    # --------------------------------------------------------------
    if best_seq is not None:
        # Ensure exact feasibility before the final phases
        S = np.sqrt(2.0 * len(best_seq))
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum() * S

        # ----- projected gradient descent (sub‑gradient) -----
        pgd_time = min(0.45 * (end_time - time.time()), 180.0)
        if pgd_time > 10.0:
            pgd_deadline = time.time() + pgd_time
            a = best_seq.copy()
            n = len(a)
            vel = np.zeros_like(a)
            beta = 0.9
            step = 0.25 * S            # initial step size (scaled)
            while time.time() < pgd_deadline:
                cur_max, grad = _subgradient_max(a)
                # momentum update
                vel = beta * vel + (1 - beta) * grad
                a_candidate = a - step * vel
                a_candidate = _project_onto_simplex(a_candidate, S)
                new_max = _full_convolution_fft(a_candidate).max()
                if new_max < cur_max:
                    a = a_candidate
                    cur_max = new_max
                    step *= 1.02          # slight acceleration on improvement
                else:
                    step *= 0.70          # otherwise shrink step
                # global decay of step size
                step *= 0.9995
                # occasional jitter to shake out plateaus
                if rng.random() < 0.01:
                    jitter = rng.normal(scale=0.01 * S, size=n)
                    a = _project_onto_simplex(a + jitter, S)
            best_seq = a
            best_score = cur_max
            print(f"[pgd] after PGD best={best_score:.6f}")

        # ----- simulated annealing polish -----
        sa_time = end_time - time.time()
        if sa_time > 5.0:
            T0 = 0.08 * best_score
            T = T0
            decay = 0.9995
            a = best_seq.copy()
            n = len(a)
            cur_max = best_score
            while time.time() < end_time:
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                # propose moving a tiny amount of mass from i to j
                delta = rng.uniform(-0.012, 0.012) * S / n
                if a[i] - delta < 0 or a[j] + delta < 0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                new_max = _full_convolution_fft(a_new).max()
                dE = new_max - cur_max
                if dE < 0 or rng.random() < np.exp(-dE / T):
                    a = a_new
                    cur_max = new_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_seq = a.copy()
                        print(f"[SA] new best={best_score:.6f}  time={time.time():.1f}")
                T *= decay
                if rng.random() < 0.001:   # occasional reheating
                    T = T0

    # --------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # --------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external={external:.6f} internal={best_score:.6f} length={len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
