def project_simplex(v, S):
    # ensure sum(v) = S and v_i >=0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
adversarial step‑function optimizer – aggressive cooling version

The optimizer performs many short Adam restarts with a very diverse set of
initial shapes (uniform, Dirichlet, edge‑heavy power‑laws, Gaussians and
random mixtures).  Each run optimises the τ‑soft‑max approximation of the
peak of the self‑convolution on the simplex using a log‑softmax
parameterisation (so the ℓ₁‑norm = √(2·n) is automatically enforced).
Compared with the reference code we use

* a larger learning‑rate (0.5) and a much faster τ‑cooling (×0.99 per
  iteration) to reach the exact sub‑gradient regime quickly,
* stronger gradient clipping (10·S) and larger jitter,
* a few extra initialisation styles,
* a second “refinement” phase where the current best sequence is scaled
  to several larger lengths and optimised again with the remaining time.

The routine runs for the allotted budget (≈ 1000 s), always keeping the
best candidate seen and finally storing it in the global variable
``height_sequence_1``.  The external ``evaluate_sequence`` function is
called only once at the very end to report the true score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Soft‑max gradient of the convolution peak
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Returns (grad, max_b) where ``grad`` is ∂ τ‑soft‑max(b) / ∂ a
    and ``max_b`` is the exact maximum b_k (b = a*a).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        # exact sub‑gradient: average over all active entries
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights
        mask = w

    # gradient w.r.t. a:
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax representation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr0: float = 0.5,
                   lr_decay: float = 0.9999,
                   tau_decay: float = 0.99,
                   jitter_int: int = 500,
                   jitter_sigma: float = 0.02,
                   grad_clip_factor: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Minimises the τ‑soft‑max of the self‑convolution peak.
    Returns the best vector encountered and its *hard* peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits so that softmax(p)·S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr0

    # initial temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft‑max) -----
        g, peak = _softmax_gradient(a, tau)

        # update best (hard peak)
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # gradient clipping (helps stability for large n)
        clip_val = grad_clip_factor * S
        norm = np.max(np.abs(grad_p))
        if norm > clip_val:
            grad_p = grad_p * (clip_val / norm)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= tau_decay          # aggressive cooling
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if it % jitter_int == 0:
            p += rng.normal(scale=jitter_sigma, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parameterised edge‑heavy initial shapes (and a few more families)
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Single power‑law edge‑heavy shape."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 1.0)          # exponent
    offset = rng.uniform(0.0, 5.0)     # shift of denominator
    mix = rng.uniform(0.0, 1.0)        # mix with a uniform background
    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian bump."""
    S = np.sqrt(2.0 * n)
    sigma = rng.uniform(0.08, 0.35) * n
    x = np.arange(n) - (n - 1) / 2.0
    raw = np.exp(-(x ** 2) / (2 * sigma ** 2))
    raw = raw / raw.sum() * S
    return raw


def _combo_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of several random edge‑heavy components (plus optional Gaussian)."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    comp = np.zeros(n)
    k = rng.integers(1, 4)            # number of components
    for _ in range(k):
        p = rng.uniform(0.2, 1.5)
        offset = rng.uniform(0.0, 5.0)
        weight = rng.uniform(0.3, 1.0)
        comp += weight * (d + offset) ** (-p)
    if rng.random() < 0.5:            # possibly add a Gaussian term
        sigma = rng.uniform(0.1, 0.3) * n
        x = np.arange(n) - (n - 1) / 2.0
        comp += rng.uniform(0.2, 0.8) * np.exp(-(x ** 2) / (2 * sigma ** 2))
    comp = np.clip(comp, 0.0, None)
    if comp.sum() == 0:
        comp = np.ones(n)
    comp = comp / comp.sum() * S
    return comp

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs many short Adam restarts with
    diverse initial shapes, keeps the best candidate and finally refines
    it on several enlarged lengths before a long final Adam run.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # reuse a previously found construction if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Helper to build an initial feasible vector for a given length
    # ------------------------------------------------------------------
    def _initial_vector(n):
        # occasional warm‑start from the global best (scaled)
        if base_seq is not None and rng.random() < 0.15:
            a = _scale_sequence(base_seq, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * np.sqrt(2.0 * n)
            return a

        # pick a construction style
        style = rng.choice(['uniform', 'dirichlet', 'edge', 'gaussian', 'combo'])
        S = np.sqrt(2.0 * n)

        if style == 'uniform':
            return np.full(n, S / n)
        elif style == 'dirichlet':
            a = rng.random(n)
            return a / a.sum() * S
        elif style == 'edge':
            return _edge_initial(n, rng)
        elif style == 'gaussian':
            return _gaussian_initial(n, rng)
        else:  # combo
            return _combo_initial(n, rng)

    # ------------------------------------------------------------------
    # Phase 1 – many short restarts (exploration)
    # ------------------------------------------------------------------
    short_budget = 10.0  # seconds per short run
    while time.time() + short_budget < end_time - 30.0:
        attempt += 1

        # choose a length (mix of arbitrary and power‑of‑two values)
        if rng.random() < 0.25:

            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart #{attempt}] n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refinement on several enlarged lengths
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        # candidate lengths to try (up to a hard cap)

        for factor in length_factors:
            if remaining < 8.0:

            n_new = max(500, min(n_new, 50000))
            a_scaled = _scale_sequence(best_seq, n_new)
            S = np.sqrt(2.0 * n_new)
            a_scaled = np.clip(a_scaled, 0.0, None)

                a_scaled,

                print(f"[refine len={n_new}] improved score={best_score:.6f}")

            remaining = end_time - time.time()

    # ------------------------------------------------------------------
    # Phase 3 – final long optimisation on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and (end_time - time.time()) > 5.0:
