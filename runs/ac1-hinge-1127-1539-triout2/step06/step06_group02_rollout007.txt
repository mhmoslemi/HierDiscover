def _lp_improvement(a, target_sum, t_mix, rng):
    """
    Perform one LP improvement step: solve for g that maximizes L1 norm
    under convolution bound with current a, then mix.
    Returns updated a and the improvement flag.
    """
    n = a.size
    max_conv = _max_convolution_fft(a)
    M = max_conv

    # Build Toeplitz matrix A for constraints A g <= M
    # Using scipy.linalg.toeplitz (dense)
    first_col = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])
    first_row = np.concatenate([np.array([a[0]], dtype=np.float64), np.zeros(n-1, dtype=np.float64)])
    A = scipy.linalg.toeplitz(first_col, first_row)  # shape (2n-1, n)

    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum => minimize -sum

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=[(0, None)] * n,
                                method='highs',
                                options={'presolve': True})
    if not res.success:
        # fallback: no change
        return a, False

    g_raw = res.x
    sum_g = g_raw.sum()
    if sum_g <= target_sum * (1 + 1e-12):
        # No benefit
        return a, False

    # Scale to target sum
    g = g_raw * (target_sum / sum_g)

    # Mix with current a
    a_new = (1 - t_mix) * a + t_mix * g

    # small re‑projection to keep sum exactly target (numerical)
    # we can just renormalise to be safe.
    a_new = a_new * (target_sum / a_new.sum())

    return a_new, True

"""
Propose a step‑function (non‑negative sequence) that minimizes the
provided `evaluate_sequence`.  The algorithm mixes two ideas:

1. **LP improvement step** –  given a current sequence `f`,
   solve a linear program that maximises the ℓ¹‑norm of a new
   sequence `g` under the constraints
   `conv(f, g) ≤ max(conv(f, f))`.  Scaling `g` back to the prescribed
   ℓ¹‑sum and mixing it with `f` provably reduces the worst convolution
   entry whenever the LP succeeds.

2. **Projected gradient descent** – a short soft‑max gradient‑descent
   phase (the same routine used in the previous state‑of‑the‑art
   implementation) finishes the optimisation.

The routine works on several problem sizes (1500 and 2000 are tried
by default), keeps the best sequence found before the 1000 s budget
expires and finally strips trailing zeros.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolution, projection, etc.)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the (linear) convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv) / (sum(a)²) for a normalised sequence."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.  Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient descent (identical to the previous implementation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (gradient, raw convolution array) where gradient has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a (see analysis in the description)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # the gradient lives on the original index range
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of the max.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def _lp_improvement(a: np.ndarray, target_sum: float, mix: float, rng: np.random.Generator):
    """
    One LP improvement:
    maximise Σ g_i  subject to  g ≥ 0  and  (a * g) ≤ M,
    where M = max(a * a).  The optimum g is scaled to the target ℓ¹‑sum
    and mixed with a using the factor `mix`.
    Returns the updated sequence and a boolean indicating whether the LP
    produced a genuinely larger ℓ¹‑norm (i.e. an improvement).
    """
    import scipy.linalg
    import scipy.optimize

    n = a.size
    M = _max_convolution_fft(a)

    # Build the Toeplitz matrix for the convolution constraints
    # A[k, j] = a[k-j]  if 0 ≤ k-j < n  else 0
    first_col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
    first_row = np.concatenate([np.array([a[0]], dtype=np.float64), np.zeros(n - 1, dtype=np.float64)])
    A = scipy.linalg.toeplitz(first_col, first_row)  # shape (2n‑1, n)

    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g_i  ≡ maximise Σ g_i

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # LP failed – fall back to the original vector
        return a, False

    g_raw = res.x
    sum_g = g_raw.sum()
    if sum_g <= target_sum * (1.0 + 1e-12):
        return a, False        # no ℓ¹‑norm increase → no improvement

    # Scale to the prescribed ℓ¹‑sum and mix
    g = g_raw * (target_sum / sum_g)
    a_new = (1.0 - mix) * a + mix * g

    # Re‑project onto the simplex to kill accumulated floating errors
    a_new = _project_to_simplex(a_new, target_sum)

    return a_new, True

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a good step‑function within the given time budget.
    Returns the best sequence (as a plain Python list) found.
    """
    import scipy.optimize  # make sure SciPy is available
    import scipy.linalg

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # Sizes we will explore – 1500 is the size of the previous best,
    # 2000 gives a few extra degrees of freedom without exploding memory.
    sizes = [1500, 2000]

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break

        # ------------------------------------------------------------------
        # Initialisation
        # ------------------------------------------------------------------
        target_sum = math.sqrt(2.0 * n)

        if "height_sequence_1" in globals():
            seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size > n:
                seq = seq[:n]
            else:
                seq = np.pad(seq, (0, n - seq.size), "constant")
        else:
            # power‑law initialisation (exponent tuned experimentally)
            seq = _power_law_sequence(n, 0.55, target_sum)

        seq = _project_to_simplex(seq, target_sum)

        # ------------------------------------------------------------------
        # LP‑based improvement phase
        # ------------------------------------------------------------------
        it_deadline = min(deadline, time.time() + 250.0)   # allocate ≤250 s for LPs
        mix = 0.30
        lp_successes = 0

        while time.time() < it_deadline:
            seq, improved = _lp_improvement(seq, target_sum, mix, rng)
            if improved:
                lp_successes += 1
                cur_score = _score(seq)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.copy()
                    print(f"[{time.time():.1f}] n={n} LP step → new best {best_score:.6f}")
                # slowly damp the mixing factor
                mix = max(0.02, mix * 0.97)

            # occasional jitter to avoid stagnation
            if rng.random() < 0.05:
                jitter = rng.normal(scale=target_sum * 0.001, size=n)
                seq = np.maximum(seq + jitter, 0.0)
                seq = _project_to_simplex(seq, target_sum)

            # stop early if we have made many successful LP updates
            if lp_successes >= 12:
                break

        # ------------------------------------------------------------------
        # Short projected‑gradient refinement (soft‑max)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 20.0:
            gd_deadline = time.time() + min(remaining * 0.4, 30.0)
            seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)
            if gd_score < best_score:
                best_score = gd_score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] n={n} GD refinement → new best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final tidy‑up
    # ----------------------------------------------------------------------
    if best_seq is not None:
        # Remove trailing near‑zero entries (they do not affect the score)
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
