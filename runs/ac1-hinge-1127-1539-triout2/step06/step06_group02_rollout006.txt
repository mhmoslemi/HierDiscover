def _adam_optimize(a0, deadline, beta_start=1.0, beta_end=16384.0):
    a = a0.copy()
    n = a.shape[0]; target = sqrt(2*n)
    m = np.zeros_like(a); v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t = 0
    best_a = a.copy()
    best_score = _score(a)  # compute final objective max conv (prev)
    while time.time() < deadline:
        t += 1
        # schedule beta: exponential maybe
        beta = beta_start * (beta_end / beta_start) ** min(1.0, (t * 1.0) / 100000.0)  # half-life
        # compute gradient of smoothmax conv w.r.t a
        g, b = _gradient_softmax(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # base learning rate adapt: maybe step = target * 0.3 / (np.sqrt(v_hat)+eps)
        step = target * 0.3 / (np.sqrt(v_hat) + eps)  # elementwise, but we want step vector?
        # Actually step may be scalar: step size scaled by m_hat/v_hat: typical Adam: param = param - lr * m_hat/(sqrt(v_hat)+eps)
        lr = target * 0.1 / (np.linalg.norm(g) + eps)  # adapt step magnitude
        # Let's compute param update: a = a - lr * m_hat/(np.sqrt(v_hat)+eps)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project to simplex
        a = _project_to_simplex(a, target)
        # check score
        if (t % 100) == 0:
            val = _score(a)
            if val < best_score:
                best_score = val
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam iter {t} beta={beta:.1f} score={best_score:.6f}")
        # maybe jitter occasionally
        if t % 5000 == 0:
            jitter = np.random.default_rng().normal(scale=target * 1e-3, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
    return best_a, best_score

"""
Multi‑scale Adam‑gradient + simulated annealing optimizer.

The search proceeds in three stages of increasing problem size.
Each stage starts from the best sequence of the previous stage
(resized by linear interpolation) and performs

1.  Adam‑style projected gradient descent on a soft‑max approximation of
    max{a∗a}.  The temperature β is slowly raised (up to 16384) while a
    momentum term stabilises the updates.  After each step the vector is
    projected onto the ℓ¹‑simplex Σa = √(2·n) to keep the evaluation
    function scale‑invariant.

2.  A long simulated‑annealing phase that moves mass between indices
    that contribute most to the current worst convolution entry,
    smoothes short windows and occasionally swaps entries.
    The temperature and the allowed move size decay exponentially.

Finally the best sequence found (trailing near‑zeros stripped) is
returned.  The routine respects the supplied time budget (default 1000 s)
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / score utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective for a simplex‑normalised sequence."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence, then normalise to a prescribed sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize a sequence to a new length by linear interpolation.
    Works both for up‑ and down‑scaling.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0, old_n - 1, old_n)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a∗a).
    Returns (gradient, full convolution array).  The gradient has the same
    shape as `a`.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights over the convolution entries
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a (see derivation in the description)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Projected Adam‑style gradient descent on the soft‑max of the max‑convolution.
    Returns the best vector found before `deadline` and its score.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    # temperature schedule
    beta = 1.0
    max_beta = 16384.0
    step_base = target * 0.12      # base step size (tuned)

    best_a = a.copy()
    best_val = _score(a)

    while time.time() < deadline:
        t += 1

        # gradient of the smooth‑max
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g) + eps

        # Adam update (using the *direction* rather than raw gradient)
        direction = g / norm_g
        m = beta1 * m + (1.0 - beta1) * direction
        v = beta2 * v + (1.0 - beta2) * (direction * direction)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)

        # learning rate scaled by temperature
        lr = step_base / (beta + 1.0)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_to_simplex(a, target)

        # temperature increase
        if t % 800 == 0 and beta < max_beta:
            beta = min(beta * 1.4, max_beta)

        # occasional jitter to escape plateaus
        if t % 2500 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # keep the best solution seen so far
        if t % 200 == 0:
            cur = _score(a)
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam iter={t} β={beta:.1f} score={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Simulated‑annealing phase that moves mass between indices, smoothes short
    windows and performs occasional swaps.
    Returns the best vector found before `deadline` and its score.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(a)          # current convolution (full length)

    # annealing schedule parameters
    start_temp = 0.9
    end_temp = 1e-6
    max_frac_start = 0.30
    max_frac_end = 0.004
    max_window = 40

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time          # 0 → 1
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # worst convolution entry and its index
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # contribution of each coefficient to that worst entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]                 # a_j where i+j = k_max
        weight = a * sens                          # actual contribution a_i * a_{k_max-i}

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.78:            # mass transfer (dominant)
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # look for a receiver that contributes little
            low_thr = np.percentile(weight, 12)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.94:          # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        else:                               # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # evaluate new candidate (single FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter (helps escape deep minima)
        if it % 6000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimisation driver for a single problem size
# ----------------------------------------------------------------------


def _optimise_sequence(n: int, init_seq: np.ndarray,
                       time_budget: float, rng: np.random.Generator):
    """
    Optimises a sequence of length `n` starting from `init_seq`.
    The total time spent is limited by `time_budget` seconds.
    Returns the best sequence found and its (soft‑max) score.
    """
    target = np.sqrt(2.0 * n)

    # ensure proper length
    a = _resize_sequence(init_seq, n)
    a = _project_to_simplex(a, target)

    start = time.time()
    # 30 % of the budget → Adam gradient, 60 % → SA, 10 % → final tidy‑up
    grad_deadline = start + 0.30 * time_budget
    a, best_g = _adam_optimize(a, grad_deadline, rng)

    now = time.time()
    if now < start + time_budget:
        sa_deadline = start + time_budget
        a, best_s = _simulated_annealing(a, sa_deadline, rng)
    else:
        best_s = best_g

    # final tiny Adam run (if any time left)
    if time.time() < start + time_budget - 2.0:
        final_deadline = time.time() + 2.0
        a, _ = _adam_optimize(a, final_deadline, rng)

    final_score = min(best_g, best_s)
    return a, final_score

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Entry point used by the evaluation harness.
    Executes a multi‑scale optimisation (sizes 2000 → 4000 → 8000)
    and returns the best non‑negative step function found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # optional warm‑start from a previously discovered construction
    global_seq = None
    if "height_sequence_1" in globals():
        try:
            global_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        except Exception:
            global_seq = None

    # problem sizes to explore (larger sizes give more degrees of freedom)
    sizes = [2000, 4000, 8000]

    for n in sizes:
        now = time.time()
        if now >= deadline - 5.0:
            break

        remaining = deadline - now
        # allocate up to 40 % of the remaining budget (capped at 400 s)
        time_for_this = min(remaining * 0.40, 400.0)

        # initialise the sequence for this size
        if global_seq is not None:
            init_seq = global_seq
        else:
            # power‑law start – exponent tuned empirically
            init_seq = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

        seq, score = _optimise_sequence(n, init_seq, time_for_this, rng)

        # evaluate with the provided security‑checked function (more reliable)
        try:
            eval_score = evaluate_sequence(seq.tolist())
        except Exception:
            eval_score = float('inf')

        if eval_score < best_score:
            best_score = eval_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} → new global best score={best_score:.6f}")

        # use the current best as warm‑start for larger sizes
        global_seq = seq

    # final short Adam refinement on the overall best (time permitting)
    if best_seq is not None and time.time() < deadline - 2.0:
        extra_deadline = time.time() + (deadline - time.time()) * 0.20
        best_seq, _ = _adam_optimize(best_seq, extra_deadline, rng)

    # strip trailing zeros – they do not affect the score
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
