def _hill_climb(a0, deadline):
    a = a0.copy()
    n = len(a)
    S = sqrt(2n)
    best_score = np.max(_full_convolution(a))
    best_a = a.copy()
    while time.time() < deadline:

        if i==j: continue
        # Transfer some mass from i to j
        delta = a[i] * rng.random() * 0.01
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a,best_score

"""
Step‑function optimisation for the “max self‑convolution” objective.

The algorithm proceeds in three stages of increasing problem size.
For each size it runs:

1. Adam optimisation on a smooth soft‑max approximation of the max‑norm,
   with projection onto the ℓ₁‑simplex Σaᵢ = √(2·n).
2. A short projected sub‑gradient phase that attacks the true worst entry.
3. A lightweight steady‑state genetic algorithm for diversification.
4. (if time remains) a linear‑programming refinement that solves
   maximise Σbⱼ  s.t.  a∗b ≤ M   (M = max(a∗a)),
   then scales the LP solution back to the simplex and repeats a few times.
5. A final random hill‑climbing phase.

All phases respect the overall time budget (≈1000 s).  The best vector
found is returned as a plain Python list; by construction the external
score equals the maximal entry of its self‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on smooth max‑norm
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5
    alpha_max = 1500.0

    while time.time() < deadline:
        t += 1
        # gradient of smooth max‑norm
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass swap (exploration)
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule
        alpha = min(alpha * 1.05, alpha_max)

        # cheap early‑stop test (purely for logging)
        if t % 5000 == 0 and cur_score >= best_score * (1 + 1e-9):
            pass
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient w.r.t. the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 8000 == 0 and cur_score >= best_score * (1 + 1e-9):
            pass
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient) with configurable split
# ----------------------------------------------------------------------
def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Run Adam for a fraction of the allocated time, then finish with sub‑gradient.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_s, sc_s = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Genetic algorithm optimiser (convex recombination + mutation)
# ----------------------------------------------------------------------
def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple steady‑state GA that works on the simplex.
    Returns the best vector found before ``deadline``.
    """
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 20 if n <= 2000 else 12
    pop = []
    scores = []
    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            if rng.random() < 0.35:
                noise = rng.normal(scale=child.mean() * 0.05 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        if rng.random() < 0.12:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        if time.time() + 0.1 >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximise total mass under the current bound)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iters: int = 5) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a linear programme:
        maximise Σ b_j  subject to b ≥ 0, a∗b ≤ M,
    where M = max(a∗a).  The LP solution is scaled back to the simplex
    (Σ = √(2n)) and used as a new candidate.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # Import lazily – SciPy may be unavailable.
    try:
        from scipy.sparse import diags
        from scipy.optimize import linprog
    except Exception as e:
        print("[LP] SciPy not available:", e)
        return best_a, best_score

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _full_convolution(a)
        M = np.max(conv)

        # Build the constraint matrix A where (A @ b)_k = (a∗b)_k.
        # Each diagonal d (0 ≤ d < n) holds the constant a[d].
        diag_data = [np.full(n, a[d]) for d in range(n)]
        A = diags(diag_data, offsets=range(n), shape=(2 * n - 1, n), format='csc')

        c = -np.ones(n)                     # maximise Σb  ⇔  minimise −Σb
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as e:
            print("[LP] linprog exception:", e)
            break

        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        b = res.x
        sum_b = b.sum()
        # The LP always has a feasible point a, so sum_b ≥ S.
        if sum_b <= S * (1 + 1e-9):
            # No room for improvement.
            break

        g = b * (S / sum_b)                 # scale back to the simplex
        new_score = np.max(_full_convolution(g))

        if new_score < best_score - 1e-6:
            best_score = new_score
            best_a = g.copy()
            a = g
            print(f"[LP] iter {it} improved score to {best_score:.6f}")
        else:
            # No further progress.
            break
    return best_a, best_score

# Final random hill‑climbing refinement (tiny swaps)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap coordinate‑swap hill climbing.
    Moves a random small amount of mass from one coordinate to another.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_score = np.max(_full_convolution(a))
    best_a = a.copy()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            print(f"[hill] improved score to {best_score:.6f}")
    return best_a, best_score

        # ---- initialise starting vector for this stage ----

            best_score = sc_opt
            best_seq = a_opt.copy()

    # ---- final hill‑climbing refinement (use any remaining time) ----
    if best_seq is not None and time.time() < deadline:
        a_final, sc_final = _hill_climb(best_seq, deadline, rng)
