def propose_candidate(...):
    import time, numpy as np
    seed default 123, budget_s default 1000.

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # If global variable height_sequence_1 exists, use it as initial best
    try:
        init_seq = globals().get('height_sequence_1')
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            a0 = np.asarray(init_seq, dtype=float)
            if a0.ndim == 1:
                # Ensure non-negative
                a0 = np.maximum(a0, 0.0)
                # Compute score via evaluation to be consistent
                cur_score = evaluate_sequence(a0.tolist())
                best_seq = a0
                best_score = cur_score
                print(f"[{time.time():.1f}] Seeded from height_sequence_1 score={cur_score:.6f}")
    except Exception as e: pass

    # Choose length schedule based on remaining time: start with moderate lengths like 2000, 5000, 10000, 20000, 30000.
    schedule = [2000, 5000, 10000, 20000, 30000]

    for n in schedule:
        if time.time() >= deadline:
            break

        # if best_seq exists and not same length, upscale using interpolation
        if best_seq is not None and best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = None

        # We'll use parameters: number of random shape samples per family
        samples_per_family = 20
        families = ['uniform','powerlaw','sympower','exp','revexp','piecewise']
        # Define data collection per candidate
        for fam in families:
            # break if time
            if time.time() >= deadline:
                break
            for s in range(samples_per_family):
                # generate shape vector
                if fam == 'uniform':
                    a = np.full(n, 1.0, dtype=float)
                elif fam == 'powerlaw':
                    p = rng.uniform(0.01, 2.0)
                    a = (np.arange(1, n+1, dtype=float) ** (-p))
                elif fam == 'sympower':
                    p = rng.uniform(0.01, 2.0)
                    idx = np.arange(1, n+1, dtype=float)
                    a = (idx ** (-p) + (n+1-idx) ** (-p))
                elif fam == 'exp':
                    lam = rng.exponential(scale=0.02) + 1e-8
                    a = np.exp(-lam * np.arange(n, dtype=float))
                elif fam == 'revexp':
                    lam = rng.exponential(scale=0.02) + 1e-8
                    a = np.exp(-lam * np.arange(n-1, -1, -1, dtype=float))
                elif fam == 'piecewise':
                    # random fraction of indices high, low values elsewhere
                    frac = rng.uniform(0.1, 0.9)
                    cutoff = int(frac * n)
                    high = rng.uniform(0.5, 2.0)
                    low = rng.uniform(0.0, 0.5)
                    a = np.empty(n, dtype=float)
                    a[:cutoff] = high
                    a[cutoff:] = low
                else:
                    a = np.random.rand(n)

                # Ensure non-negative
                a = np.maximum(a, 0.0)
                # No need to project onto simplex because evaluation is scale invariant but we can Normalizing to sum 1 for numeric stability
                # We'll just compute evaluation directly.

                # Evaluate using fast FFT
                maxb = _max_convolution_fft(a)
                sum_a = a.sum()
                # Avoid division by zero
                if sum_a < 1e-12:
                    continue
                score = 2.0 * n * maxb / (sum_a * sum_a)

                if score < best_score - 1e-9:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] New shape {fam} p={p if 'p' in locals() else ''} score={score:.6f} n={n}")

        # After scanning shapes, we may want to refine best_seq for this length using gradient descent
        if best_seq is not None and best_seq.size == n:
            # gradient descent restarts
            gd_restarts = 3
            for r in range(gd_restarts):
                if time.time() >= deadline:
                    break
                # allocate some time per restart
                gd_deadline = deadline - 0.1   # we can run within remaining time
                a0 = best_seq.copy()
                a_gd = _gradient_descent_opt(a0, gd_deadline)
                # compute score
                maxb = _max_convolution_fft(a_gd)
                sum_a = a_gd.sum()
                score = 2.0 * n * maxb / (sum_a * sum_a)
                if score < best_score - 1e-9:
                    best_score = score
                    best_seq = a_gd.copy()
                    print(f"[{time.time():.1f}] GD improved score={score:.6f} n={n} restart={r}")

        # hill climbing, similar to prior, with longer time if remaining.
        if best_seq is not None and best_seq.size == n and time.time() < deadline - 10.0:
            # compute current convolution full
            a_cur = best_seq.copy()
            b_full = _full_convolution(a_cur)
            hill_deadline = deadline - 2.0
            a_hc, b_hc = _hill_climbing(a_cur, b_full, hill_deadline)
            cur_score = b_hc.max()
            # evaluation uses same scaling: compute 2 * n * cur_score / (sum_a^2)
            sum_a = a_hc.sum()
            score = 2.0 * n * cur_score / (sum_a * sum_a)
            if score < best_score - 1e-9:
                best_score = score
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill climb improved to {score:.6f} n={n}")

    # final polishing with remaining time
    if best_seq is not None and time.time() < deadline -5.0:
        n = best_seq.size
        a = best_seq.copy()
        b = _full_convolution(a)
        a_final, b_final = _hill_climbing(a, b, deadline)
        sum_a = a_final.sum()
        final_score = 2.0 * n * b_final.max() / (sum_a * sum_a)
        if final_score < best_score - 1e-9:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # Clean-up: trimming trailing near-zero entries
    if best_seq is None:
        # fallback uniform
        n = 1000
        best_seq = np.full(n, 1.0)
    else:
        best_seq = np.asarray(best_seq, dtype=float)
        # Remove trailing near zero entries
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[:nz[-1]+1]
        else:
            best_seq = best_seq[:1]

    # Compute official evaluation and print
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:
        print(f"Error during final evaluation: {exc}")

    return best_seq.tolist()

"""
Propose a step‑function that minimises the official ``evaluate_sequence``
objective.  The strategy is:

*  If a previous construction (global ``height_sequence_1``) exists, start
   from it.
*  For a schedule of lengths (up to 30 000) generate many cheap
   parametric families (power‑law, symmetric power‑law, exponential,
   reversed exponential, simple piecewise) and keep the best based on a
   fast FFT based evaluation of max(self‑convolution).
*  Refine the current best with a projected gradient descent on a smooth
   log‑sum‑exp surrogate (moments, temperature annealing) – the same
   routine used in the reference implementation.
*  Finish the refinement with a mass‑transfer hill‑climb that updates the
   full convolution in O(n) per move and accepts occasional uphill
   steps (simulated annealing).
*  All work respects the 1000 s wall‑clock budget and the final sequence
   is trimmed of trailing zeros before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT, projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return max entry of the linear convolution a*a using an FFT.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (FFT based)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (full convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Momentum, annealed inverse temperature β and jitter are used.
    The loop stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers a random amount
    of mass from a source index to a destination index while keeping the
    full self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling

    while time.time() < deadline:
        # ----- pick a max‑convolution index -------------------------------------------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # ----- source index weighted by its contribution to entry M ------------------------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ----- destination index ------------------------------------------------------------
        # 30 % of the time bias towards the smallest entries
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # ----- amount of mass to move (up to 80 % of source) -------------------------------
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update the full convolution efficiently -------------------------------------
        b_new = b.copy()
        # linear terms
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- Metropolis acceptance --------------------------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

# Shape generators used for cheap exploration

def _shape_uniform(n):
    return np.full(n, 1.0, dtype=float)


def _shape_powerlaw(n, p):
    """Monotone decreasing power‑law: a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _shape_sym_powerlaw(n, p):
    """U‑shaped power‑law: large at both ends, small in the middle."""
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p) + (n + 1 - idx) ** (-p)


def _shape_exponential(n, lam):
    return np.exp(-lam * np.arange(n, dtype=float))


def _shape_rev_exponential(n, lam):
    return np.exp(-lam * np.arange(n - 1, -1, -1, dtype=float))


def _shape_piecewise(n, frac, high, low):
    """Fraction `frac` of the indices receive `high`, the rest `low`."""
    cut = int(frac * n)
    a = np.empty(n, dtype=float)
    a[:cut] = high
    a[cut:] = low
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

        except Exception:
            pass

    # ------------------------------------------------------------------
    # Length schedule – we gradually increase the dimension
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 30000]

        # --------------------------------------------------------------
        # 1) cheap parametric exploration
        # --------------------------------------------------------------
        families = [
