def _softmax_gradient(a, tau):
    b = _full_convolution_fft(a)  # returns array length 2n-1
    max_b = b.max()
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()
    # gradient: 2 * conv(mask * a) with reversed a
    a_rev = a[::-1]
    # convolution mask*a conv a_rev
    conv_len = len(mask) + len(a) - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    grad = 2.0 * conv[len(a)-1:len(a)-1+len(a)]
    return grad, max_b
```

def _initial_vector(n, rng, base_seq=None):
    S = np.sqrt(2.0 * n)
    if base_seq is not None and rng.random() < 0.2:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a
    style = rng.choice(['powerlaw','exponential','linear','gamma','uniform','dirichlet'])
    if style == 'powerlaw':
        a = _init_powerlaw(n, rng)
    elif style == 'exponential':
        a = _init_exponential(n, rng)
    elif style == 'linear':
        a = _init_linear(n, rng)
    elif style == 'gamma':
        a = _init_gamma(n, rng)
    elif style == 'uniform':
        a = np.full(n, S / n)    # but we will renormalize later anyway
    else: # dirichlet
        a = rng.random(n)            # uniform random
    # Normalize
    a = a.clip(min=0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0)
        total = n
    a = a * (S / total)
    return a

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    global height_sequence_1
    rng = np.random.default_rng(seed)

            base_seq = None

    best_seq = None

    attempt = 0

    # Exploration phase (up to 70% of time)

    while time.time() < explore_deadline:
        attempt += 1
        # choose length
        if rng.random() < 0.6:

            n = rng.integers(5000, 20000)  # larger

    # Refinement phase
    if best_seq is None:
        # fallback: uniform of length 2000
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # List of lengths to refine: original n and few scaled versions
    lengths_to_refine = [best_n]
    if best_n < 10000:
        lengths_to_refine.append(min(20000, best_n * 2))

        lengths_to_refine.append(int(best_n * 1.5))

        # allocate time chunk
        remaining = end_time - time.time()
        if remaining < 30.0:

   linear, sparse‑gamma, uniform).  Each candidate is normalised so that
   its ℓ₁‑norm equals √(2 n) and its self‑convolution peak is measured
   using an FFT.  The best candidate seen so far is stored globally.

2. **Scaling & long‑run refinement** – the best candidate is interpolated
   to several larger lengths (the larger the length the richer the
   search space) and each scaled vector is refined with a long Adam
   optimiser that works on the simplex via a log‑softmax
   parametrisation.  The optimiser uses the exact sub‑gradient of the
   max‑convolution (τ‑softmax with τ≈0) and a decaying learning‑rate.

3. **Final sanity check** – the best sequence found is evaluated with
   the black‑box ``evaluate_sequence`` function, stored in the global
   variable ``height_sequence_1`` and returned.

The whole routine respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(b) where b = a * a (full convolution)."""
    return _full_convolution_fft(a).max()


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b) with grad = ∂ τ‑soft‑max(b) / ∂a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / np.sum(exps)

    # gradient: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (log‑softmax parametrisation).
    Minimises the exact max‑convolution (τ‑soft‑max with τ≈0).
    Returns the best sequence found and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from the feasible a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4                       # a bit larger than the reference

    # initialise temperature (soft‑max smoothing)
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 0.5)       # start with a small amount of smoothing

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (hard max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S            # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)            # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995                # shrink smoothing quickly
        lr *= 0.9998                 # very slow decay

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parametric families for random initialisation
# ----------------------------------------------------------------------
def _init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law with optional uniform mixing."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    p = rng.uniform(0.2, 3.0)          # exponent
    offset = rng.uniform(0.0, 15.0)    # shift of the denominator
    mix = rng.uniform(0.0, 1.0)        # how much uniform background
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    return raw


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges with optional uniform background."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    lam = rng.uniform(0.0005, 0.02)   # decay rate
    mix = rng.uniform(0.0, 1.0)
    exp_part = np.exp(-lam * d)
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    return raw


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Triangular shape – linear decrease from the edges."""
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    max_h = rng.uniform(0.5, 5.0)                     # height at the edge
    slope = rng.uniform(0.0, max_h / (n / 2 + 1e-9))  # controls width of the plateau
    lin = np.maximum(0.0, max_h - slope * d)
    mix = rng.uniform(0.0, 1.0)
    base = np.ones(n) * rng.uniform(0.0, 0.2)
    raw = mix * lin + (1.0 - mix) * base
    return raw


def _init_gamma(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Gamma‑distributed entries (shape < 1)."""
    shape = rng.uniform(0.1, 0.8)
    raw = rng.gamma(shape, 1.0, size=n)
    return raw


def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Trivial uniform start."""
    return np.full(n, 1.0)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like start (raw random numbers)."""
    raw = rng.random(n)
    return raw


def _initial_vector(n: int,
                    rng: np.random.Generator,
                    warm_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best and otherwise samples
    from a random parametric family.
    """
    S = np.sqrt(2.0 * n)

    # Warm start with probability ~0.2
    if warm_seq is not None and rng.random() < 0.22:
        a = _scale_sequence(warm_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['powerlaw', 'exponential', 'linear',
                        'gamma', 'uniform', 'dirichlet'])
    if style == 'powerlaw':
        raw = _init_powerlaw(n, rng)
    elif style == 'exponential':
        raw = _init_exponential(n, rng)
    elif style == 'linear':
        raw = _init_linear(n, rng)
    elif style == 'gamma':
        raw = _init_gamma(n, rng)
    elif style == 'uniform':
        raw = _init_uniform(n, rng)
    else:
        raw = _init_dirichlet(n, rng)

    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    if total == 0.0:
        raw = np.full(n, 1.0)
        total = n
    a = raw * (S / total)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the time budget and stores the final construction in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Phase 1 – fast random exploration (≈ 70 % of the total budget)
    # ------------------------------------------------------------------
    explore_deadline = end_time - 250.0   # reserve at least 250 s for refinement
    while time.time() < explore_deadline:
        attempt += 1
        # length selection – a mix of moderate and larger vectors
        if rng.random() < 0.65:
            n = int(rng.integers(1500, 5000))
        else:
            n = int(rng.integers(5000, 24000))

        a0 = _initial_vector(n, rng, base_seq)
        peak = _max_convolution(a0)

        if peak < best_score:
            best_score = peak
            best_seq = a0.copy()
            best_n = n
            print(f"[explore] attempt={attempt:4d} n={n:5d}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – scaling & long Adam‑refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a simple uniform construction
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_n = n
        best_score = _max_convolution(best_seq)

    # lengths to which we will scale the current best candidate
    lengths = [best_n]
    if best_n < 12000:
        lengths.append(min(12000, best_n * 2))
    if best_n < 8000:
        lengths.append(int(best_n * 1.5))
    lengths = sorted(set(lengths))

    for target_n in lengths:
        remaining = end_time - time.time()
        if remaining < 30.0:
            break
        # allocate a fraction of the remaining budget (capped)
        time_slice = min(remaining * 0.45, 200.0)

        a_scaled = _scale_sequence(best_seq, target_n)
        S = np.sqrt(2.0 * target_n)
        a_scaled = np.clip(a_scaled, 0.0, None)
        a_scaled = a_scaled * (S / a_scaled.sum())

        a_opt, peak = _adam_optimize(a_scaled,
                                      time.time() + time_slice,
                                      rng)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = target_n
            print(f"[refine] target_n={target_n:5d}  improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
