"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Strategy
--------
1. **Initialisation** – start from the best known construction if the
   global variable ``height_sequence_1`` is present; otherwise generate a
   power‑law seed of length 8000 (the same family that gave the previous
   record).

2. **Sub‑gradient descent on the exact L∞‑norm** – we repeatedly
   evaluate the full self‑convolution, locate the shift(s) attaining the
   current maximum, compute the (piece‑wise) sub‑gradient of the
   max‑convolution and take a projected step on the simplex
   Σa_i=√(2 n).  The step size decays with √iteration and a tiny random
   jitter is added every few hundred steps to escape plateaux.

3. **Permutation‑annealing** – once the values have been refined we keep
   the multiset fixed and improve the ordering by a simulated‑annealing
   procedure that swaps two positions and accepts worsening moves with a
   slowly cooling temperature.

4. **Final sanitising** – trim trailing zeros, report the final
   ``evaluate_sequence`` value and return the best list found.

All heavy work respects the global time budget (default 1000 s).  The
implementation uses only NumPy and the provided ``evaluate_sequence``
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (only a fallback seed)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale vector to have ℓ¹‑sum equal to *target*."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient(a: np.ndarray, max_inds: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of   max_k (a*a)[k]   w.r.t. a.
    For each maximal shift k the contribution is
        g_i += 2 * a_{k-i}
    where the index k-i is inside the array.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_inds:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        counterpart = k - idx
        g[idx] += 2.0 * a[counterpart]
    return g


def _subgradient_minimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly minimises
    max(a*a).  Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # best‑ever bookkeeping
    b = _full_convolution(a)
    best_val = float(b.max())
    best_a = a.copy()

    rng = np.random.default_rng(42)
    it = 0
    # tolerance for “max” equality – a tiny absolute value is enough
    eps = 1e-12

    while time.time() < deadline:
        # full convolution
        conv = _full_convolution(a)
        M = float(conv.max())

        # record improvement
        if M < best_val:
            best_val = M
            best_a = a.copy()
            print(f"[{time.time():.1f}] Sub‑grad it={it} new best={M:.6f}")

        # indices attaining (or almost attaining) the maximum
        max_inds = np.where(conv >= M - eps)[0]

        # sub‑gradient of the max‑operator
        g = _subgradient(a, max_inds)
        norm = np.linalg.norm(g)
        if norm == 0.0:
            break

        # step‑size: 0.5*target_sum / (norm * sqrt(it+1))
        lr = 0.5 * target_sum / (norm * (np.sqrt(it + 1) + 1.0))

        # gradient step
        a = a - lr * g

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = a + jitter
            a = _project_to_simplex(a, target_sum)

        it += 1
        # safety guard – we never need more than a few hundred thousand iterations
        if it > 300000:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm stays unchanged; only the permutation changes.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initialise temperature as a small fraction of the current score
    T = best_score * 0.03 + 1e-9
    cooling = 0.9993
    rng = np.random.default_rng(123)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # swap
        a[i], a[j] = a[j], a[i]

        cur = _max_convolution_fft(a)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur < best_score:
                best_score = cur
                best = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # rare global reshuffle of a tiny fraction – helps jump out of local minima
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur = _max_convolution_fft(a)
                if cur < best_score:
                    best_score = cur
                    best = a.copy()

    return best
