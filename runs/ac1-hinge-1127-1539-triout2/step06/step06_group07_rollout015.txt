        cand[i] -= delta

def _run_genetic_optimizer(initial_a, deadline, rng):
    a = initial_a.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    pop_size = 16
    # init population
    pop = np.empty((pop_size, n), dtype=float)
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        if i == 0:
            pop[i] = a
        else:
            pop[i] = rng.random(n)
        pop[i] = _project_simplex(pop[i], S)
        scores[i] = np.max(_full_convolution(pop[i]))
    best_idx = np.argmin(scores)
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]
    while time.time() < deadline:
        # produce offspring
        for i in range(pop_size):
            # random parents
            p1, p2 = rng.choice(pop_size, size=2, replace=False)
            alpha = rng.random()
            child = alpha * pop[p1] + (1 - alpha) * pop[p2]
            if rng.random() < 0.3:
                noise = rng.normal(scale=child.mean()*0.05+1e-12, size=n)
                child += noise
            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)
            child_score = np.max(_full_convolution(child))
            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()
        # diversity injection
        if rng.random() < 0.1:
            worst = np.argmax(scores)
            new = rng.random(n)
            new = _project_simplex(new, S)
            new_score = np.max(_full_convolution(new))
            pop[worst] = new
            scores[worst] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new.copy()
        # early stop if time low
        if time.time() + 0.1 >= deadline:
            break
    return best_a, best_score

def _refine_smoothing(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    # schedule of lambda decreasing from 0.2 to 0
    start_time = time.time()
    while time.time() < deadline:
        # compute elapsed fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        if total <= 0:
            break
        lam = max(0.0, 0.3 * (1 - elapsed / total))  # linear decay
        # simple moving average filter of radius 1 (kernel [0.5,0.5])
        # we can use convolution via np.convolve
        smoothed = np.convolve(a, np.array([0.5, 0.5]), mode='same')
        a_new = (1 - lam) * a + lam * smoothed
        a_new = _project_simplex(a_new, S)
        cur_score = np.max(_full_convolution(a_new))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        # small random jitter occasionally
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
    return best_a, best_score

def _anneal_optimize(a0, deadline, rng):
    a = a0.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    cur_score = np.max(_full_convolution(a))
    best_a = a.copy()
    best_score = cur_score
    T0 = cur_score * 0.05
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        T = T0 * (1 - elapsed / total)  # linear cooling
        # propose transfer
        i = rng.choice(n, p=a / a.sum())
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.02
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # cand already sum-preserving and nonnegative (due to delta <= a[i]).
        cand_score = np.max(_full_convolution(cand))
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T,1e-12)):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

"""
Propose a non‑negative step function (vector) that minimises the maximal entry
of its self‑convolution.  The routine runs a long‑running multi‑stage search
inside a 1000 s budget:

* Simplex projection and FFT‑based convolution utilities.
* An Adam optimiser on a soft‑max (exponential) approximation of the max.
* A momentum sub‑gradient descent that works on the true max‑norm.
* A modest steady‑state genetic algorithm for extra exploration.
* A short smoothing / annealing refinement.
* Progressive optimisation on increasing problem sizes (800 → 2000 → 4000
  → 6000) with many independent restarts.  The best vector found is
  returned as a plain Python list.

All helpers are top‑level functions – no closures – and the code respects
the 1000 s external budget (leaving a 5 s safety margin).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a * a (full convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max: w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam_enhanced(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimisation on a soft‑max approximation of the max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters (tuned for this problem)
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    conv = _full_convolution(a)
    best_score = float(conv.max())

    alpha = 0.5          # temperature – starts smooth
    alpha_max = 5000.0   # final sharpness

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)            # O(n log n)
        w = _smooth_weights(conv, alpha)       # soft‑max weights
        grad = _conv_with_weights(a, w)         # gradient of smooth max

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass transfer (exploration)
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase the sharpness of the soft‑max
        alpha = min(alpha * 1.08, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent with momentum (operates on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad_momentum(a0: np.ndarray,

    n = a.shape[0]

    best_a = a.copy()
    best_score = float(_full_convolution(a).max())

        k_max = int(np.argmax(conv))

                    if cur_score < best_score:

# ----------------------------------------------------------------------
# Combined optimiser (Adam → momentum sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_improved(a0: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                fraction_adam: float = 0.65) -> tuple[np.ndarray, float]:
    """
    Run Adam for a fraction of the allocated time, then finish with the
    momentum sub‑gradient method.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, float(_full_convolution(a0).max())

    adam_deadline = now + fraction_adam * remaining
    a_adam, sc_adam = _optimize_adam_enhanced(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_sub, sc_sub = _optimize_subgrad_momentum(a_adam, deadline, rng)
        if sc_sub < sc_adam:
            return a_sub, sc_sub
    return a_adam, sc_adam


# ----------------------------------------------------------------------
# Steady‑state Genetic Algorithm optimiser
# ----------------------------------------------------------------------


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple GA that works on the simplex."""
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 16
    pop = np.empty((pop_size, n), dtype=float)
    scores = np.empty(pop_size, dtype=float)

    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        pop[i] = a
        scores[i] = float(_full_convolution(a).max())

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # two distinct parents
            p1, p2 = rng.choice(pop_size, size=2, replace=False)
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            if rng.random() < 0.3:
                noise = rng.normal(scale=child.mean() * 0.05 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = float(_full_convolution(child).max())
            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # occasional injection of fresh random individual
        if rng.random() < 0.1:
            worst = int(np.argmax(scores))
            fresh = rng.random(n)
            fresh = _project_simplex(fresh, S)
            fresh_score = float(_full_convolution(fresh).max())
            pop[worst] = fresh
            scores[worst] = fresh_score
            if fresh_score < best_score:
                best_score = fresh_score
                best_a = fresh.copy()

        if time.time() + 0.1 >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothing / local refinement
# ----------------------------------------------------------------------


def _refine_smoothing(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Apply a decreasing moving‑average smoothing while staying on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = float(_full_convolution(a).max())

    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        lam = max(0.0, 0.3 * (1.0 - elapsed / total))   # linear decay

        # 2‑point moving average (kernel [0.5, 0.5])
        smoothed = np.convolve(a, np.array([0.5, 0.5]), mode='same')
        a_new = (1.0 - lam) * a + lam * smoothed
        a_new = _project_simplex(a_new, S)

        cur = float(_full_convolution(a_new).max())
        if cur < best_score:
            best_score = cur
            best_a = a_new.copy()
        a = a_new

        # tiny random jitter every now and then
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Transfer a small amount of mass between two coordinates."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    cur_score = float(_full_convolution(a).max())
    best_a = a.copy()
    best_score = cur_score

    T0 = cur_score * 0.05                      # initial temperature
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        T = T0 * (1.0 - elapsed / total)      # linear cooling

        # pick donor proportional to its mass, random recipient
        i = rng.choice(n, p=a / a.sum())
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.02     # move at most 2 % of donor
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta                       # sum stays unchanged, non‑negativity kept
        cand_score = float(_full_convolution(cand).max())

        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# One full optimisation run for a given problem size
# ----------------------------------------------------------------------


def _run_one_instance(n: int,
                      init_a: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the whole pipeline (Adam → sub‑gradient → GA → smoothing → annealing)
    for a single start vector.
    """
    a_opt, sc_opt = _optimize_combined_improved(init_a, deadline, rng)

    # genetic refinement (if time permits)
    remaining = deadline - time.time()
    if remaining > 2.0:
        ga_deadline = time.time() + min(8.0, remaining)
        a_ga, sc_ga = _run_genetic_optimizer(a_opt, ga_deadline, rng)
        if sc_ga < sc_opt:
            a_opt, sc_opt = a_ga, sc_ga

    # smoothing refinement
    remaining = deadline - time.time()
    if remaining > 1.0:
        smooth_deadline = time.time() + min(4.0, remaining)
        a_smooth, sc_smooth = _refine_smoothing(a_opt, smooth_deadline, rng)
        if sc_smooth < sc_opt:
            a_opt, sc_opt = a_smooth, sc_smooth

    # annealing refinement
    remaining = deadline - time.time()
    if remaining > 1.0:
        anneal_deadline = time.time() + min(6.0, remaining)
        a_ann, sc_ann = _anneal_optimize(a_opt, anneal_deadline, rng)
        if sc_ann < sc_opt:
            a_opt, sc_opt = a_ann, sc_ann

    return a_opt, sc_opt


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The method runs several independent optimisation
    runs on increasing problem sizes, keeping the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    best_seq = None
    best_score = np.inf

    # If a previous construction is present, use it as a warm‑start.
    if 'height_sequence_1' in globals():
        try:
            best_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            best_seq = None

    # ------------------------------------------------------------------
    # Progressive optimisation over several sizes
    # ------------------------------------------------------------------
    sizes = [800, 2000, 4000, 6000]        # enlarge the search space gradually
    # rough time allocation per size (will be re‑scaled to the remaining budget)
    size_fractions = {800: 0.18, 2000: 0.28, 4000: 0.35, 6000: 0.15}

    for n in sizes:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining time for this size
        remaining_total = deadline - time.time()
        if remaining_total <= 0:
            break
        phase_frac = size_fractions.get(n, 0.1)
        phase_deadline = time.time() + remaining_total * phase_frac

        # run many short restarts inside the phase
        run_target = 22.0 if n <= 2000 else 30.0      # seconds per restart
        run_idx = 0
        while time.time() + run_target < phase_deadline:
            run_idx += 1
            # initialise either from the current global best (upsampled) or randomly
            if best_seq is not None and rng.random() < 0.7:
                a0 = _upsample(best_seq, n)
                jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
                a0 = np.clip(a0 + jitter, 0.0, None)
            else:
                a0 = rng.random(n)

            # project onto the L1‑simplex Σa = √(2·n)
            S = np.sqrt(2.0 * n)
            a0 = a0 / a0.sum() * S

            run_deadline = time.time() + run_target
            a_res, sc_res = _run_one_instance(n, a0, run_deadline, rng)

            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"[run] n={n:4d} run={run_idx:3d} new best score={best_score:.6f}")

        # one final longer run for the current size (if time permits)
        leftover = phase_deadline - time.time()
        if leftover > 5.0:
            if best_seq is not None:
                a0 = _upsample(best_seq, n)
                jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
                a0 = np.clip(a0 + jitter, 0.0, None)
            else:
                a0 = rng.random(n)
            a0 = a0 / a0.sum() * np.sqrt(2.0 * n)

            a_res, sc_res = _run_one_instance(n, a0, phase_deadline, rng)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"[phase] n={n:4d} final run new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
