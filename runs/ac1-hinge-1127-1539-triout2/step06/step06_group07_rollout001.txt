"""
Propose a non‑negative step function that minimizes the value returned by
``evaluate_sequence``.
Algorithm
---------
* Progressive refinement: start from a modest length (≈2000) and
  progressively double the length (up to 12000) while keeping the
  best‑so‑far solution as warm‑start.
* For each problem size we run two optimisation phases
    1. Adam on a smooth soft‑max approximation of the infinity‑norm of
       the auto‑convolution.  The gradient is obtained by a weighted
       convolution (FFT) and the iterate is projected onto the L¹‑simplex.
    2. Projected sub‑gradient descent that directly reduces the true
       worst convolution entry.
* After the gradient phase a short simulated‑annealing / random‑swap
  refinement is performed to escape possible local minima.
* The whole run respects the 1000 s budget (5 s safety margin) and
  returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i    (full convolution with flipped weights)
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   alpha_start: float = 0.5,
                   alpha_end: float = 4000.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth approximation of max(conv).
    Returns the best vector encountered and its (true) max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # temperature (soft‑max sharpness) schedule
    alpha = alpha_start

    while time.time() < deadline:
        t += 1

        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass‑swap for exploration
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase sharpness, but never exceed alpha_end
        alpha = min(alpha * 1.06, alpha_end)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that directly reduces the worst convolution entry.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the current worst entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it + 1.0)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation (helps escape plateaus)
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                # accept if better or with a small probability
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined Adam → Sub‑gradient optimiser
# ----------------------------------------------------------------------


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       fraction_adam: float = 0.75) -> tuple[np.ndarray, float]:
    """
    Run Adam for ``fraction_adam`` of the allocated time, then continue with sub‑gradient.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Simulated‑annealing / local random‑swap refinement
# ----------------------------------------------------------------------


def _refine_local(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple annealing: repeatedly move a random fraction of mass from one coordinate
    to another and accept if the max‑convolution improves or with a temperature‑based
    probability. Returns the best vector found.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # temperature schedule: start relatively hot, cool to near zero
    T0 = 0.02
    T_end = 0.0001
    total = deadline - time.time()
    while time.time() < deadline:
        # linear cooling based on remaining fraction
        remaining = deadline - time.time()
        tau = remaining / total
        T = T_end + (T0 - T_end) * tau

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # take a random fraction of the source mass
        frac = rng.random() * 0.05  # up to 5 %
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # we keep the L¹ sum exactly, but we must enforce non‑negativity
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand, S)

        cand_score = np.max(_full_convolution(cand))

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
        else:
            # metropolis acceptance
            if rng.random() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):
                a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.
    The routine works on several increasing problem sizes, applying
    Adam → sub‑gradient optimisation followed by a brief annealing step.
    The best sequence found before the deadline is returned as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a few seconds for the final external evaluation
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Progressive optimisation over increasing sizes
    # ------------------------------------------------------------------
    # sizes are chosen such that the later stages have enough time to
    # perform a reasonable number of iterations; we stop early if the
    # global deadline is reached.
    sizes = [2000, 4000, 8000, 12000]
    # time fractions sum to < 1; the remainder is safety margin.
    fractions = [0.15, 0.20, 0.35, 0.25]

    total_budget = deadline - start_time

    for stage, n in enumerate(sizes):
        if time.time() >= deadline:
            break

        # allocate time slice for this stage
        stage_time = total_budget * fractions[stage]
        stage_deadline = min(time.time() + stage_time, deadline)

        # initialise a0 -------------------------------------------------
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)

        # tiny jitter to avoid exact repetitions and to break possible symmetry
        jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
        a0 = np.clip(a0 + jitter, 0.0, None)

        S = np.sqrt(2.0 * n)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0)
        a0 = a0 / a0.sum() * S

        # optimisation phase ------------------------------------------------
        a_opt, sc_opt = _optimize_combined(a0, stage_deadline, rng, fraction_adam=0.75)

        # local refinement (simulated annealing) if time permits
        remaining = stage_deadline - time.time()
        if remaining > 1.5:
            a_ref, sc_ref = _refine_local(a_opt, time.time() + min(12.0, remaining), rng)
            if sc_ref < sc_opt:
                a_opt, sc_opt = a_ref, sc_ref

        # update global best
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[stage] n={n}  new global best score={best_score:.6f}")

        # recompute remaining total budget for later stages
        total_budget = deadline - time.time()

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: a single coefficient (the problem is scale‑invariant)
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
