    best_seq = None
    best_score = np.inf
    attempts = 0

    # Phase 1: Random search over families to generate initial candidates
    # We'll allocate maybe 0.3 of total budget for this.
    phase1_deadline = start + 0.3*total_budget

    while time.time() < phase1_deadline:
        attempts += 1
        # choose n
        if base_seq is not None and rng.random() < 0.5:
            scale = rng.uniform(0.8, 1.2)
            n = max(500, int(base_seq.shape[0] * scale))
        else:
            n = rng.integers(2000, 12001)
        # choose a family
        family = rng.integers(0, 6)
        if family == 0:
            # upsample base if possible
            if base_seq is not None:
                a = _upsample_seq(base_seq, n)
            else:
                a = _initial_sequence(None, n, rng)  # fallback uniform
        elif family == 1:
            # uniform Dirichlet (random)
            a = rng.random(n)
        elif family == 2:
            # geometric left and right
            rate = rng.uniform(0.95, 0.9999)  # moderate
            left = rate ** np.arange(n)
            right = left[::-1]
            a = (left + right) / 2.0
        elif family == 3:
            # power law symmetric
            gamma = rng.uniform(0.2, 5.0)
            idx = np.arange(n)
            a = ((idx+1)*(n-idx)) ** gamma
            a = a.astype(float)
            a = a / a.max()
        elif family == 4:

            p = rng.uniform(0.5, 4.0)

            a = np.sin(np.pi * (x+0.5) / n) ** p
        # clip negative
        a = np.clip(a, 0, None)
        # normalize to sqrt(2n)
        S = np.sqrt(2.0 * n)
        total = a.sum()
        if total == 0:
            a = np.full(n, 1.0)
            total = a.sum()
        a = a / total * S

        # optionally symmetrize
        if rng.random() < 0.4:
            a = _symmetrize(a)

        # quick evaluate
        score = eval_score(a)
        if score < best_score:

            best_seq = a.copy()

    # Phase 2: Intensive refinement on best_seq

def _optimize_adam(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0*n)
    # hyperparameters
    lr = 0.6  # slightly higher
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    # temperature schedule
    alpha = 0.1
    alpha_mul = 1.04
    max_alpha = 200.0
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # compute softmax weights
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)
        # jitter optional
        if rng.random() < 0.01:
            i,j = rng.integers(0,n, size=2)
            if i!=j and a[i] > 0:
                delta = rng.random() * 0.05 * a[i]
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
        # update hyperparams
        alpha = min(alpha*alpha_mul, max_alpha)
        lr *= 0.99997  # slower decay
    return best_a, best_score

"""
Proposes a step‑function (non‑negative vector) whose self‑convolution peak is
as low as possible.  The search consists of three stages:

1. **Exploratory sampling** – generate many analytic families (uniform,
   geometric, power‑law, beta‑like, sinusoidal) for a variety of lengths.
   Normalise each candidate to Σa = √(2·n) and keep the best one.

2. **Smooth Adam optimisation** – minimise a soft‑max approximation of the
   max‑convolution using projected Adam.  The temperature (α) is increased
   up to 200 so that the approximation becomes tight.  The optimiser runs
   for the majority of the time budget.

3. **Refinement** – a sub‑gradient descent on the true maximum followed by a
   cheap random‑swap hill‑climbing.

The routine is time‑aware, respects the 1000 s budget and stores the best
sequence in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helpers
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.1
    alpha_mul = 1.04
    max_alpha = 200.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.012:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        # schedule updates
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.99997

    return best_a, best_score


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient (single‑term version – a valid sub‑gradient)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1, dtype=int)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a -= eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = cand.copy()

        # early exit if stuck
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that swaps tiny masses between random positions.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous best if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0

    # ------------------------------------------------------------------
    # Phase 1 – many cheap random families
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    while time.time() < phase1_deadline:
        attempts += 1

        # choose a length
        if base_seq is not None and rng.random() < 0.55:
            scale = rng.uniform(0.9, 1.1)
            n = max(800, int(base_seq.shape[0] * scale))
        else:
            n = int(rng.integers(1800, 12001))

        # pick a family
        fam = rng.integers(0, 6)
        if fam == 0 and base_seq is not None:
            a = _upsample_seq(base_seq, n)
        elif fam == 1:
            a = rng.random(n)                     # uniform random
        elif fam == 2:
            rate = rng.uniform(0.95, 0.9999)       # geometric from both ends
            left = rate ** np.arange(n)
            a = (left + left[::-1]) / 2.0
        elif fam == 3:
            # power‑law symmetric: (i·(n‑i))^p
            p = rng.uniform(0.2, 4.0)
            idx = np.arange(n, dtype=float)
            a = ((idx + 1) * (n - idx)) ** p
        elif fam == 4:
            # skewed beta‑like shape
            p1 = rng.uniform(0.1, 5.0)
            p2 = rng.uniform(0.1, 5.0)
            x = np.linspace(0, 1, n)
            a = (x ** p1) * ((1 - x) ** p2)
        else:
            # sinusoidal power
            p = rng.uniform(0.5, 4.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

        a = np.clip(a, 0.0, None)

        # normalise to Σa = √(2·n)
        S = np.sqrt(2.0 * n)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0)
            s = a.sum()
        a = a / s * S

        # optional symmetrisation
        if rng.random() < 0.4:
            a = _symmetrize(a)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = a.copy()
            print(f"[init] attempt={attempts} n={n} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive refinement of the current best
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback: plain uniform of moderate size
        n = 5000
        a = rng.random(n)
        a = a / a.sum() * np.sqrt(2.0 * n)
        best_overall_seq = a
        best_overall_score = np.max(_full_convolution(a))

    # Adam optimisation (takes about 60 % of the remaining time)
    now = time.time()
    remaining = deadline - now
    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(best_overall_seq, t_adam, rng,
                                    enforce_sym=False)
    if sc_adam < best_overall_score:
        best_overall_score = sc_adam
        best_overall_seq = a_adam.copy()
        print(f"[adam] improved to {best_overall_score:.6f}")

    # Sub‑gradient descent (≈30 % of the rest)
    now = time.time()
    if now < deadline:
        t_sub = now + 0.30 * (deadline - now)
        a_sub, sc_sub = _optimize_subgrad(best_overall_seq, t_sub, rng,
                                          enforce_sym=False)
        if sc_sub < best_overall_score:
            best_overall_score = sc_sub
            best_overall_seq = a_sub.copy()
            print(f"[subgrad] improved to {best_overall_score:.6f}")

    # Final tiny random‑swap hill‑climb (≈10 % of the rest)
    now = time.time()
    if now < deadline:
        a_loc, sc_loc = _local_swap_search(best_overall_seq, deadline, rng,
                                           enforce_sym=False)
        if sc_loc < best_overall_score:
            best_overall_score = sc_loc
            best_overall_seq = a_loc.copy()
            print(f"[local] improved to {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (also expose it globally for future runs)
    # ------------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
