"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search proceeds in several stages:

* **Initialisation** – start from the globally stored sequence
  (if any) or from a random point on the ℓ¹‑simplex.
* **Length schedule** – gradually increase the dimension (up to
  80 000 entries).  Larger vectors give a lower theoretical bound,
  because the average entry of the self‑convolution is 2 n/(2 n‑1)
  → 1.
* **Projected gradient descent** – minimise a smooth log‑sum‑exp
  surrogate of the maximum convolution entry.  Momentum,
  β‑annealing and occasional jitter are used for robustness.
* **Mass‑transfer hill climbing** – exact O(n) updates of the full
  self‑convolution while performing a simulated‑annealing style
  random walk that moves mass from positions contributing heavily
  to the current worst convolution entry to positions that are
  under‑represented.
* **Final smoothing** – a few moving‑average passes (projected onto
  the simplex) often reduce the peak further.
* **Logging** – intermittent ``print`` statements report the current
  best value and the wall‑clock time.

All vectors stay non‑negative and are projected onto the simplex
{ a_i ≥ 0, Σ a_i = √(2 n) }.  The routine respects the 1000 s
wall‑clock budget and returns the best sequence found as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT for speed)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) – O(n) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # --- soft‑max probabilities over b ------------------------------------
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # --- gradient g = 2·(a ⋆ p) -----------------------------------------
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Uses momentum, β‑annealing and occasional jitter.
    Terminates when ``time.time()`` passes *deadline*.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σa = √(2 n).  Several families (power‑law, exponential,
    piecewise‑linear, Dirichlet, sinusoidal) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential (two ends)
        cut = rng.integers(int(0.2 * n), int(0.8 * n))
        lam1 = rng.exponential(scale=0.022)
        lam2 = rng.exponential(scale=0.022)
        left = np.exp(-lam1 * np.arange(cut))
        right = np.exp(-lam2 * np.arange(n - cut))
        a = np.concatenate([left, right])

    elif family == 5:                   # sinusoidal (zero at ends)
        a = np.sin(np.pi * idx / (n - 1))
        a = np.maximum(a, 0.0)

    elif family == 6:                   # linear ramp up then down (triangular)
        mid = n // 2
        ramp = np.arange(1, mid + 1, dtype=float)
        if n % 2 == 0:
            a = np.concatenate([ramp, ramp[::-1]])[:n]
        else:
            a = np.concatenate([ramp, ramp[::-1]])[:n]

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length,
    followed by projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers a random
    amount of mass from a source index to a destination index while
    keeping the full self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.08 + 1e-9          # initial temperature
    cooling = 0.9996                     # slower cooling for better exploration

    while time.time() < deadline:
        # ----- pick a max‑convolution index -----------------------------------
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # ----- source index (biased by its contribution to entry M) ----------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ----- destination index (bias towards small heights) ---------------
        if rng.random() < 0.4:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # ----- amount of mass to move ----------------------------------------
        max_delta = a[i] * 0.85
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (linear‑time slice updates) -----------------
        b_new = b.copy()
        # linear terms
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- Metropolis acceptance -----------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b


def _smooth_sequence(a: np.ndarray, steps: int = 8) -> np.ndarray:
    """
    Simple moving‑average smoothing (3‑point kernel) with re‑projection
    onto the simplex after each pass.
    """
    n = a.size
    target = np.sqrt(2.0 * n)
    for _ in range(steps):
        a = np.convolve(a, np.array([0.25, 0.5, 0.25]), mode="same")
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)
    return a

                local_best_a = a_gd.copy()

        # keep the best result of this length
        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill climbing with exact convolution updates
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 5.0:
            break
        hill_budget = min(remaining * 0.35, 200.0)
        hill_deadline = time.time() + hill_budget

        # make sure we have a vector of current length
        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)
        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing(a_current, b_full, hill_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Optional smoothing (cheap) – can further lower the peak
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            a_smooth = _smooth_sequence(best_seq, steps=6)
            smooth_score = _max_convolution_fft(a_smooth)
            if smooth_score < best_score:
                best_score = smooth_score
                best_seq = a_smooth.copy()
                print(f"[{time.time():.1f}] Smoothing improvement for n={n} score={smooth_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        b = _full_convolution(a)
        final_deadline = deadline
        a_final, b_final = _hill_climbing(a, b, final_deadline)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

        # one last smoothing pass
        a_final = _smooth_sequence(best_seq, steps=4)
        final_score2 = _max_convolution_fft(a_final)
        if final_score2 < best_score:
            best_score = final_score2
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Post‑smooth improvement: score={final_score2:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 2000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
