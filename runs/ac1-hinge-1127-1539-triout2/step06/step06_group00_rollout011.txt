def _adam_optimize(a0, deadline, rng, phase='max', tau=None):
    # common code
    n = len(a0)
    S = np.sqrt(2.0*n)
    # convert a0 to logits p (log softmax)
    eps = 1e-12
    soft0 = np.maximum(a0 / S, eps)
    soft0 = soft0 / soft0.sum()
    p = np.log(soft0)
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4 if phase=='l2' else 0.3
    if phase == 'max':
        # init tau using current max_b
        b0 = _full_convolution_fft(a0)
        tau = max(b0.max(), 1.0)
    else:
        tau = None
    best_a = a0.copy()
    best_peak = float('inf')
    it=0
    while time.time() < deadline:
        it+=1
        # recover a
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S
        # compute gradient
        if phase == 'max':
            g, peak = _softmax_gradient(a, tau)
        else:
            # l2 objective
            b = _full_convolution_fft(a)
            avg = (a.sum()**2) / (len(b))
            d = b - avg
            # gradient
            a_rev = a[::-1]
            conv_len = len(d) + n - 1
            f_len = _next_pow2(conv_len)
            Fd = np.fft.rfft(d, n=f_len)
            Fa_rev = np.fft.rfft(a_rev, n=f_len)
            conv = np.fft.irfft(Fd * Fa_rev, n=f_len)[:conv_len]
            g = 2.0 * conv[n-1:n-1+n]
            peak = b.max()
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
        # gradient w.r.t logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)
        # clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)
        # Adam update
        m = beta1*m + (1-beta1)*grad_p
        v = beta2*v + (1-beta2)*(grad_p**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # schedule updates
        if phase == 'max':
            tau *= 0.9995
            lr *= 0.9999
        else:
            lr *= 0.9998
        # occasional jitter
        if it % 1000 == 0:
            p += rng.normal(scale=0.01, size=n)
    return best_a, best_peak

"""
Adversarial step‑function optimizer – uniform‑ising version.

The optimizer searches for a non‑negative vector `a` of length `n`
with ℓ₁‑norm `S = sqrt(2·n)` that minimises the peak of the
self‑convolution `a*a`.  It consists of three stages:

1. **Diverse random restarts** – each restart chooses a random length
   (≈ 1–15 k) and an initial shape (uniform, Dirichlet, single‑edge,
   double‑edge).  The restart runs a short *L2‑flattening* phase that
   minimises the variance of the convolution, followed by a *max‑phase*
   that directly reduces the highest convolution entry via a τ‑soft‑max
   gradient (Adam on the simplex).

2. **Refinement** – the best candidate found is refined for the
   remaining time budget with the max‑phase optimizer.

3. **Export** – the final sequence is stored in the global variable
   `height_sequence_1` and returned as a plain Python list.

All optimisation is performed on the log‑soft‑max parametrisation,
so the ℓ₁‑norm constraint is satisfied automatically.
The code aims at beating the previous score 1.5113 and driving the
evaluation value closer to the theoretical limit ≈ 1.0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    τ‑soft‑max gradient of max(b) where b = a * a.
    Returns (grad, max_b).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _l2_gradient(a: np.ndarray):
    """
    Gradient of the L2‑flattening objective
        J = Σ_k (b_k - μ)² ,  μ = average(b) .
    Returns (grad, max_b).
    """
    n = a.shape[0]
    b = _full_convolution_fft(a)                 # length 2n‑1
    mu = (a.sum() ** 2) / float(2 * n - 1)       # average of b
    d = b - mu                                   # deviation vector

    # grad = 2 * (d * a_rev) , sliced to length n
    a_rev = a[::-1]
    conv_len = len(d) + n - 1
    f_len = _next_pow2(conv_len)
    Fd = np.fft.rfft(d, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fd * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g, b.max()


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   phase: str = 'max') -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex.
    `phase` can be:
        'max' – minimise the τ‑soft‑max approximation of the max(b)
        'l2'  – minimise the L2‑flattening objective (variance of b)

    Returns the best vector encountered and the corresponding hard peak
    (i.e. max(b) of the current best a).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits from a0 (log‑softmax)
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft /= soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.4 if phase == 'l2' else 0.3

    # τ for the max‑phase (ignored for L2)
    if phase == 'max':
        b0 = _full_convolution_fft(a0)
        tau = max(b0.max(), 1.0)
    else:
        tau = None

    best_a = a0.copy()
    best_peak = np.inf
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)               # numerical stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- gradient & hard peak -----
        if phase == 'max':
            g, peak = _softmax_gradient(a, tau)
            tau *= 0.9995                       # tighten soft‑max
        else:  # 'l2'
            g, peak = _l2_gradient(a)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- chain rule to logits -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- learning‑rate schedule -----
        lr *= 0.9999 if phase == 'max' else 0.9998

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            p += rng.normal(scale=0.01, size=n)

    return best_a, best_peak

# Parameterised initial shapes

def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Single edge‑heavy family: (d+offset)^(-p) optionally mixed with uniform.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    p = rng.uniform(0.2, 1.0)
    offset = rng.uniform(0.0, 5.0)
    mix = rng.uniform(0.0, 1.0)

    edge_part = (d + offset) ** (-p)
    raw = mix * edge_part + (1.0 - mix) * np.ones(n)
    raw = raw / raw.sum() * S
    return raw


def _double_edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Mix of two edge‑heavy components plus a uniform background.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    p1 = rng.uniform(0.2, 1.0)
    off1 = rng.uniform(0.0, 5.0)
    p2 = rng.uniform(0.2, 1.0)
    off2 = rng.uniform(0.0, 5.0)

    mix1 = rng.uniform(0.0, 1.0)
    mix2 = rng.uniform(0.0, 1.0 - mix1)
    mix0 = 1.0 - mix1 - mix2

    part1 = (d + off1) ** (-p1)
    part2 = (d + off2) ** (-p2)
    raw = mix1 * part1 + mix2 * part2 + mix0 * np.ones(n)
    raw = raw / raw.sum() * S
    return raw


def _initial_vector(n: int, rng: np.random.Generator, base_seq: np.ndarray | None) -> np.ndarray:
    """Build a feasible start vector for a given length."""
    # occasional warm‑start from a previously found global solution
    if base_seq is not None and rng.random() < 0.15:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * np.sqrt(2.0 * n)
        return a

    style = rng.choice(['uniform', 'dirichlet', 'edge', 'double'])
    S = np.sqrt(2.0 * n)

    if style == 'uniform':
        a = np.full(n, S / n)
    elif style == 'dirichlet':
        a = rng.random(n)
        a = a / a.sum() * S
    elif style == 'edge':
        a = _edge_initial(n, rng)
    else:  # double edge
        a = _double_edge_initial(n, rng)

    return a
