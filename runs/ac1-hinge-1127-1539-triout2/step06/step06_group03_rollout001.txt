"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Algorithm outline
-----------------
1. **Initialisation** – keep the best known construction (if the global
   variable ``height_sequence_1`` exists) and a few uniform baselines.

2. **Parametric random search** – search the two‑parameter family
   a_i = (i + s)^{-p} (power‑law) and a_i = exp(-r·(i+s)) (exponential).
   A simulated‑annealing style loop explores p∈[0.3,0.8] and shift s≥0,
   gradually decreasing the perturbation size.  For each candidate the
   exact objective 2·n·max(a*a)/‖a‖_1² is evaluated by an FFT‑based
   convolution.

3. **Refinement** – the current best sequence is refined by
   *projected gradient descent* on a smooth log‑sum‑exp surrogate of the
   max–convolution (the same routine used in the previous solution) and
   by a *sub‑gradient descent* that directly reduces the true max entry.
   Finally a permutation‑annealing stage swaps entries to improve the
   ordering.

4. **Return** – the best sequence found (as a plain Python list) is
   returned.  All steps respect the global time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (ℓ¹‑simplex Σx = s, x≥0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective of a sequence (exactly what evaluate_sequence computes)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a (no clipping needed)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    sum_a = float(a.sum())
    # evaluate_sequence returns np.inf for sum_a < 0.01 – we avoid that.
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, truncated linear)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have ℓ¹‑sum equal to *target*."""
    scale = target / float(v.sum())
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i + shift)^{-p}, projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    """a_i = exp(-rate·(i+shift)), projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    """a_i = max(0, intercept - slope·i), projected onto the ℓ¹‑simplex."""
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Gradient descent on the smooth log‑sum‑exp surrogate of max(b)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where
        b = a*a      (full convolution)
        grad = ∇_a ( (1/β)·log Σ exp(β·b) )
             = 2·(a ⋆ p) ,   p_k ∝ exp(β·b_k)
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # grad = 2 * (a ⋆ p)  (linear convolution, then take centre n entries)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * p_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp smooth approximation
    of max(a*a).  The routine runs until the supplied deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7          # larger than before
    momentum = np.zeros_like(a)
    rho = 0.9               # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 6e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent directly on the true max entry
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple sub‑gradient descent on max(a*a).  Each iteration moves a
    against the gradient of the (one) maximal convolution entry and
    projects back onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        # pick a random index among those attaining the max (helps when many ties)
        ks = np.flatnonzero(b >= max_b * (1 - 1e-12))
        k = ks[rng.integers(len(ks))]
        # gradient of b_k w.r.t a: (grad)_i = 2 * a_{k-i} if 0≤k-i<n else 0
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            # step size slowly decays with iteration count
            lr = 0.4 * target_sum / (norm * (it + 1.0) ** 0.6)
            a -= lr * grad
            a = _project_to_simplex(a, target_sum)
        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improve the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm stays unchanged – only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]           # swap
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[i], a[j] = a[j], a[i]       # revert

        T *= cooling

        # occasional small random reshuffle of a tiny fraction
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                tmp = a[mask]
                rng.shuffle(tmp)
                a[mask] = tmp
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
    return best

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    explores parametric families (power‑law, exponential, linear),
    refines promising candidates with projected gradient & sub‑gradient
    descent, and finally runs a permutation‑annealing pass.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf

    # --------------------------------------------------------------
    # 0.  Start from the historic construction if available
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        if init.size > 0 and np.all(init >= 0):
            obj = _objective(init)
            best_seq, best_obj = init.copy(), obj
            print(f"[{time.time():.1f}] Starting from height_sequence_1, obj={obj:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Uniform baselines for a few lengths (cheap sanity checks)
    # --------------------------------------------------------------
    for n in (8000, 10000, 12000):
        if time.time() >= deadline:
            break
        target = np.sqrt(2.0 * n)
        seq = np.full(n, target / n)
        obj = _objective(seq)
        if obj < best_obj:
            best_obj, best_seq = obj, seq.copy()
            print(f"[{time.time():.1f}] Uniform baseline n={n}, obj={obj:.6f}")

    # --------------------------------------------------------------
    # 2.  Parametric random search (power‑law + exponential)
    # --------------------------------------------------------------
    # allocate roughly 60 % of remaining time to this stage
    search_deadline = time.time() + 0.6 * (deadline - time.time())
    lengths = [8000, 10000, 12000]

    for n in lengths:
        if time.time() >= search_deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # Local search state (starts from the current global best if it has same length)
        if best_seq is not None and best_seq.size == n:
            p_curr = 0.5
            shift_curr = 0.6
        else:
            p_curr = rng.uniform(0.4, 0.6)
            shift_curr = rng.uniform(0.3, 1.0)

        step_p = 0.12
        step_shift = 0.8
        trials = 0

        while time.time() < search_deadline and trials < 25000:
            # after a few thousand random attempts we start local perturbations
            if trials < 4000:
                p = rng.uniform(0.3, 0.8)
                shift = rng.uniform(0.0, 2.0)
                use_exp = rng.random() < 0.15   # 15 % exponential
            else:
                p = p_curr + rng.normal() * step_p
                shift = shift_curr + rng.normal() * step_shift
                # keep parameters in a sensible range
                if p <= 0.0:

                if shift < 0.0:
                    shift = -shift
                use_exp = rng.random() < 0.1

            # generate candidate

                cand = _exponential_sequence(n, rate, exp_shift)
            else:
                cand = _powerlaw_sequence(n, p, shift)

            # evaluate objective
            obj = _objective(cand)

            # keep global best
            if obj < best_obj:

                print(f"[{time.time():.1f}] New global best (n={n}) obj={obj:.6f}")

            # keep local best for guided perturbations
            if obj < _objective(_powerlaw_sequence(n, p_curr, shift_curr)):
                p_curr, shift_curr = p, shift
                step_p *= 0.995
                step_shift *= 0.995

            trials += 1

    # --------------------------------------------------------------
    # 3.  Refinement of the best candidate found so far
    # --------------------------------------------------------------
    if best_seq is not None:
        # allocate the remaining time to refinement
        refine_deadline = deadline

        # 3a. Gradient descent on smoothed max
        a = np.array(best_seq, dtype=float)
        a = _gradient_descent_opt(a, refine_deadline)

        # 3b. Sub‑gradient descent on the true max
        a = _subgradient_descent_opt(a, refine_deadline)

        # 3c. Permutation annealing (order optimisation)
        a = _permutation_annealing(a, deadline)

        obj = _objective(a)
        if obj < best_obj:
            best_obj, best_seq = obj, a.copy()
            print(f"[{time.time():.1f}] Refined global best, obj={obj:.6f}")

    # --------------------------------------------------------------
    # 4.  Final sanity & return
    # --------------------------------------------------------------
    if best_seq is None:

        n = 8000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]

    # Verify with the supplied evaluate_sequence (optional)

        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception as e:
        print("evaluate_sequence call failed:", e)

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
