def simulated_annealing_max(a0, deadline, start_temp=1.0, end_temp=1e-4):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2*n)
    # Precompute size of FFT for each conv
    size_fft = 1 << ((2*n - 1).bit_length())
    # Compute initial conv and score
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A*A, n=size_fft)[:2*n-1]
    cur_score = float(conv.max())
    best_score = cur_score
    best_a = a.copy()
    start_time = time.time()
    total_time = deadline - start_time
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau

        # Choose move type
        move_type = rng.choice(['transfer','smooth','swap'])
        a_new = a.copy()
        # Transfer
        if move_type == 'transfer':
            # Compute sensitivities as before using conv max position
            k_max = int(np.argmax(conv))
            # Compute weight = a * sensitivity
            sens = np.zeros(n)
            lo = max(0, k_max - (n-1))
            hi = min(k_max, n-1)
            idx = np.arange(lo, hi+1)
            sens[idx] = a[k_max - idx]
            weight = a * sens
            total_w = weight.sum()
            if total_w == 0:
                donor = rng.integers(n)
            else:
                donor = rng.choice(n, p=weight/total_w)
            # receiver among low weight
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * 0.2  # max 20% transfer
                a_new[donor] -= delta
                a_new[receiver] += delta
        elif move_type == 'smooth':
            start = rng.integers(0, n-1)
            length = rng.integers(2, min(50, n - start))
            seg = a_new[start:start+length]
            avg = seg.mean()
            a_new[start:start+length] = avg
        else:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        # Ensure non-negative
        a_new = np.maximum(a_new, 0)
        # Normalize sum
        cur_sum = a_new.sum()
        if cur_sum > 0:
            a_new = a_new * (target_sum / cur_sum)

        # Compute new score
        A_new = np.fft.rfft(a_new, n=size_fft)
        conv_new = np.fft.irfft(A_new * A_new, n=size_fft)[:2*n-1]
        new_score = float(conv_new.max())

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new best={best_score:.6f}")

        # occasional jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum*0.001, size=n)
            a += jitter
            a = np.maximum(a, 0)
            a = a * (target_sum / a.sum())
            A = np.fft.rfft(a, n=size_fft)
            conv = np.fft.irfft(A*A, n=size_fft)[:2*n-1]
            cur_score = float(conv.max())

        it += 1

    return best_a, best_score

def _simulated_annealing_max(a0, deadline, start_temp=1.0, end_temp=1e-4, max_frac_start=0.3, max_frac_end=0.01, max_window=50):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0*n)
    size_fft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A * A, n=size_fft)[:2*n-1]
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / max(1e-12, deadline - start_time)
        temp = start_temp * (end_temp/start_temp)**tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start)**tau

        # identify index of current maximum
        k_max = int(np.argmax(conv))

        # weight for mass transfer:
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        # Choose move
        move_type = rng.choice(['transfer','smooth','swap'])
        a_new = a.copy()
        if move_type == 'transfer':
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight/total_w)
            else:
                donor = rng.integers(n)
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a_new[donor] -= delta
                a_new[receiver] += delta
        elif move_type == 'smooth':
            start_idx = rng.integers(0, n)
            length = rng.integers(2, min(max_window, n - start_idx) + 1)
            end_idx = start_idx + length
            avg = a_new[start_idx:end_idx].mean()
            a_new[start_idx:end_idx] = avg
        else:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        # ensure non-negative and sum normalization
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)

        # compute new convolution and score
        A_new = np.fft.rfft(a_new, n=size_fft)
        conv_new = np.fft.irfft(A_new * A_new, n=size_fft)[:2*n-1]
        new_score = float(conv_new.max())

        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:

            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new best={best_score:.6f}")

        # occasional jitter

            conv = np.fft.irfft(A * A, n=size_fft)[:2*n-1]
            cur_score = float(conv.max())

        it += 1

    return best_a, best_score

"""
Improved optimizer for the non‑negative step‑function benchmark.

The algorithm consists of three stages:

1. Multi‑scale warm‑starting (≈1500 → 3000 → 6000 points).  The best
   solution of the previous scale is linearly interpolated and used as
   a starting point for the next scale.

2. Projected sub‑gradient descent directly on the non‑smooth objective
   `max(a * a)`.  The gradient of a single maximal convolution entry
   is computed analytically (only a slice of the reversed vector) and
   the iterate is projected onto the ℓ¹‑simplex `Σ a = √(2 n)`.  A
   simple decaying step size and occasional jitter help to escape
   plateaus.

3. A lightweight simulated‑annealing refinement that works with the
   true objective (maximum convolution) and uses mass‑transfer,
   short‑window smoothing and swaps.  The acceptance probability follows
   a temperature schedule that decays exponentially.

All moves respect the non‑negativity and ℓ¹‑simplex constraints.
The routine runs for the full allotted budget (default 1000 s) and
returns the best sequence it found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true objective (max convolution)
# ----------------------------------------------------------------------


def _subgradient_opt(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent that minimises max(a*a) under
    non‑negativity and Σ a = √(2n).  The method works directly with the
    non‑smooth objective; the gradient is the slice of the reversed
    vector corresponding to the currently maximal convolution entry.
    """
    a = a0.astype(np.float64, copy=True)
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    size_fft = 1 << ((2 * n - 1).bit_length())
    # initial convolution and score
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    step0 = target_sum * 0.5          # initial step magnitude
    it = 0

    while time.time() < deadline:
        i_max = int(np.argmax(conv))

        # sub‑gradient of the selected maximal entry
        offset = n - 1 - i_max
        rev = a[::-1]
        grad = np.zeros_like(a)
        if offset >= 0:
            grad[: n - offset] = 2.0 * rev[offset:]
        else:
            grad[-offset:] = 2.0 * rev[: n + offset]

        # decaying step size
        step = step0 / (1.0 + it * 0.01)
        gnorm = np.linalg.norm(grad)
        if gnorm > 0.0:
            a_candidate = a - step * grad / gnorm
        else:
            a_candidate = a.copy()

        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        # evaluate candidate
        A = np.fft.rfft(a_candidate, n=size_fft)
        conv_new = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
        new_score = float(conv_new.max())

        # accept if we improve, otherwise keep the old point
        if new_score <= cur_score:
            a = a_candidate
            conv = conv_new
            cur_score = new_score
        else:
            # very occasional uphill move to escape shallow traps
            if rng.random() < 0.005:
                a = a_candidate
                conv = conv_new
                cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_score:.6f}")

        # jitter to avoid stagnation
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            A = np.fft.rfft(a, n=size_fft)
            conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
            cur_score = float(conv.max())

        it += 1

    return best_a, best_score

def _simulated_annealing_max(a0: np.ndarray, deadline: float,
                             start_temp: float = 1.0,
                             end_temp: float = 1e-4,
                             max_frac_start: float = 0.30,
                             max_frac_end: float = 0.01,
                             max_window: int = 50) -> tuple[np.ndarray, float]:
    """
    Light‑weight SA that works directly with the maximum convolution.
    Moves: mass transfer, short‑window smoothing, random swap.
    Temperature and the allowed fraction of transferred mass decay
    exponentially from `start_temp` / `max_frac_start` to `end_temp` /
    `max_frac_end`.
    """
    a = a0.astype(np.float64, copy=True)
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    size_fft = 1 << ((2 * n - 1).bit_length())

    # initial state
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
    cur_score = float(conv.max())

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-12)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current index of the maximal convolution entry
        k_max = int(np.argmax(conv))

        # ----- mass‑transfer move -----
        # sensitivity of each coefficient for the current maximal entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]
        weight = a * sens

        move_type = rng.choice(['transfer', 'smooth', 'swap'])

        a_new = a.copy()

        if move_type == 'transfer':
            total_w = weight.sum()
            if total_w > 0.0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a_new[donor] -= delta
                a_new[receiver] += delta

        elif move_type == 'smooth':
            start_idx = rng.integers(0, n)
            length = rng.integers(2, min(max_window, n - start_idx) + 1)
            end_idx = start_idx + length
            seg = a_new[start_idx:end_idx]
            avg = seg.mean()
            a_new[start_idx:end_idx] = avg

        else:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        # maintain feasibility
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new * (target_sum / s)

        # evaluate new state
        A_new = np.fft.rfft(a_new, n=size_fft)
        conv_new = np.fft.irfft(A_new * A_new, n=size_fft)[: 2 * n - 1]
        new_score = float(conv_new.max())

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            A = np.fft.rfft(a, n=size_fft)
            conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
            cur_score = float(conv.max())

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence that minimises the benchmark.
    Three scales (≈1500 → 3000 → 6000) are explored; each scale runs
    many short random restarts consisting of a sub‑gradient phase
    followed by a simulated‑annealing refinement.
    The best sequence found before the deadline is returned.
    """
    start = time.time()
    deadline = start + budget_s - 1.0  # small safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # (length, fraction of remaining time)
    scales = [(1500, 0.20), (3000, 0.30), (6000, 0.50)]

    for n, frac in scales:
        if time.time() >= deadline - 5.0:
            break
        # allocate time for this scale
        remaining = deadline - time.time()
        scale_time = remaining * frac
        scale_deadline = time.time() + scale_time

        # initialise a warm‑start for the current scale
        if cur_seq is not None:
            init_base = _resize_sequence(cur_seq, n)
        else:
            if "height_sequence_1" in globals():
                init_base = _resize_sequence(np.array(globals()["height_sequence_1"],
                                                    dtype=np.float64), n)
            else:
                init_base = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

        # best candidate for this scale
        best_seq_scale = None
        best_score_scale = float("inf")

        # run many short restarts while time permits
        while time.time() < scale_deadline - 5.0:
            time_left = scale_deadline - time.time()
            # ensure we have enough time for a meaningful run
            if time_left < 8.0:
                break

            # allocate a slice of the remaining budget to this restart
            restart_budget = min(30.0, time_left * 0.9)

            # random initialisation blended with the warm‑start
            rng = np.random.default_rng()
            random_simplex = rng.dirichlet(np.full(n, 0.3)) * np.sqrt(2.0 * n)
            mix = rng.random()
            init = mix * init_base + (1.0 - mix) * random_simplex

            # --- sub‑gradient phase (≈70 % of the restart budget) ---
            sub_deadline = time.time() + restart_budget * 0.7
            a_sub, sub_score = _subgradient_opt(init, sub_deadline)

            # --- simulated‑annealing refinement (remaining budget) ---
            sa_deadline = time.time() + restart_budget * 0.3
            a_sa, sa_score = _simulated_annealing_max(a_sub, sa_deadline)

            # pick the better of the two
            if sub_score <= sa_score:
                cand_seq = a_sub
                cand_score = sub_score
            else:
                cand_seq = a_sa
                cand_score = sa_score

            if cand_score < best_score_scale:
                best_score_scale = cand_score
                best_seq_scale = cand_seq.copy()
                print(f"[{time.time():.2f}] Scale n={n} new local best {best_score_scale:.6f}")

        # update global best
        if best_seq_scale is not None and best_score_scale < best_score:
            best_score = best_score_scale
            best_seq = best_seq_scale.copy()
            print(f"[{time.time():.2f}] Updated global best (n={n}) score={best_score:.6f}")

        # warm‑start the next scale – if we failed to improve, keep the previous base
        cur_seq = best_seq_scale if best_seq_scale is not None else init_base

    # ------------------------------------------------------------------
    # final post‑processing
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = []

    arr = np.array(best_seq, dtype=float)
    # trim trailing near‑zeros
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final benchmark evaluation (optional, for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq

Improved optimizer for the non‑negative step‑function benchmark.

The algorithm works in three stages:

1. **Multi‑scale warm‑start** – we optimise three lengths
   (≈1500 → 3000 → 6000).  The best solution of the previous scale is
   linearly interpolated and used as a starting point for the next
   scale.

2. **Projected sub‑gradient descent** directly on the true objective
   `max(a*a)`.  The gradient of a single maximal convolution entry is
   a slice of the reversed vector, so it can be computed in O(n) time.
   After each step the iterate is projected onto the ℓ¹‑simplex
   `Σ a = √(2 n)` and occasional jitter is added to escape plateaus.

3. **Simulated‑annealing refinement** – a lightweight SA that works
   with the exact maximum‑convolution objective.  Moves are
   mass‑transfer, short‑window smoothing and swaps; a temperature
   schedule controls the acceptance of uphill moves.

All moves respect non‑negativity and the simplex constraint.
The routine runs for the whole allotted budget (default 1000 s) and
returns the best sequence it found.

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence that minimises the benchmark.
    Three scales (≈1500 → 3000 → 6000) are explored; each scale runs
    many short random restarts consisting of a sub‑gradient phase
    followed by a simulated‑annealing refinement.
    The best sequence found before the deadline is returned.
    """
    start = time.time()
    deadline = start + budget_s - 1.0     # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # (length, fraction of remaining time)
    scales = [(1500, 0.20), (3000, 0.30), (6000, 0.50)]

    for n, frac in scales:
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        scale_time = remaining * frac
        scale_deadline = time.time() + scale_time

        # initialise a warm‑start for this scale
        if cur_seq is not None:
            init_base = _resize_sequence(cur_seq, n)
        else:
            if "height_sequence_1" in globals():
                init_base = _resize_sequence(
                    np.array(globals()["height_sequence_1"], dtype=np.float64), n)
            else:
                init_base = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

        best_seq_scale = None
        best_score_scale = float("inf")

        # run many short restarts while time permits
        while time.time() < scale_deadline - 5.0:
            time_left = scale_deadline - time.time()
            if time_left < 8.0:
                break

            # allocate a slice of the remaining budget to this restart
            restart_budget = min(30.0, time_left * 0.9)

            rng = np.random.default_rng()
            random_simplex = rng.dirichlet(np.full(n, 0.3)) * np.sqrt(2.0 * n)
            mix = rng.random()
            init = mix * init_base + (1.0 - mix) * random_simplex

            # --- sub‑gradient phase (≈70 % of the restart budget) ---
            sub_deadline = time.time() + restart_budget * 0.7
            a_sub, sub_score = _subgradient_opt(init, sub_deadline)

            # --- simulated‑annealing refinement (remaining budget) ---
            sa_deadline = time.time() + restart_budget * 0.3
            a_sa, sa_score = _simulated_annealing_max(a_sub, sa_deadline)

            # pick the better of the two
            if sub_score <= sa_score:
                cand_seq = a_sub
                cand_score = sub_score
            else:
                cand_seq = a_sa
                cand_score = sa_score

            if cand_score < best_score_scale:
                best_score_scale = cand_score
                best_seq_scale = cand_seq.copy()
                print(f"[{time.time():.2f}] Scale n={n} new local best {best_score_scale:.6f}")

        # update global best
        if best_seq_scale is not None and best_score_scale < best_score:
            best_score = best_score_scale
            best_seq = best_seq_scale.copy()
            print(f"[{time.time():.2f}] Updated global best (n={n}) score={best_score:.6f}")

        # warm‑start next scale
        cur_seq = best_seq_scale if best_seq_scale is not None else init_base

    # ------------------------------------------------------------------
    # final post‑processing
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = []

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (only for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
