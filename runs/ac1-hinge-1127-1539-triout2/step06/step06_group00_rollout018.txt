def flatten_convolution(a, deadline):
    while time.time()<deadline:
        b = convolve_self(a)
        max_b = b.max()
        active = np.where(np.isclose(b, max_b, atol=1e-12))[0]
        grad = subgradient_max_convolution(a, active)
        # Normalize grad magnitude to S scale
        grad_norm = np.linalg.norm(grad)
        if grad_norm==0: break
        # step size: maybe 0.1 * S / (iteration+1) ??? We'll use eta = step_factor * S / (iteration+1)
        # Introduce schedule.
        # We'll compute step = step_size * grad / grad_norm
        step = eta * grad / grad_norm
        a_new = a - step
        # ensure nonnegative
        a_new = np.maximum(a_new, 0.0)
        # renorm to sum S
        S = np.sqrt(2.0 * len(a))
        sum_new = a_new.sum()
        if sum_new == 0:
            break
        a_new = a_new * S / sum_new
        # compute new max
        new_max = convolve_self(a_new).max()
        if new_max <= max_b:
            a = a_new
            max_b = new_max

            eta *= 0.999
        else:
            # maybe reduce eta
            eta *= 0.5
        # break if eta too small
        if eta < 1e-6:
            break
    return a, max_b

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0

    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except:
            base_seq = None

    best_seq, best_score = None, np.inf
    attempt = 0

    short_budget = 10.0  # seconds per restart
    # create varying lengths distribution
    while time.time() + short_budget < end_time - 30.0:
        attempt += 1
        # choose length
        n = int(rng.integers(1500, 20000))
        a0 = _initial_vector(n)
        deadline = min(time.time() + short_budget, end_time - 30.0)
        a_opt, peak = _adam_optimize(a0, deadline, rng)
        # Using peak as evaluation (since sum normalized)
        score = peak
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart] attempt={attempt} n={n} new best score={best_score:.6f}")

    # Phase 2 refinement
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 10.0:
            # ensure sum constraint exactly
            S = np.sqrt(2.0 * len(best_seq))
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq * S / best_seq.sum()
            # longer Adam run
            a_refined, peak = _adam_optimize(best_seq, end_time - 5.0, rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_refined.copy()
                print(f"[refine] improved to {best_score:.6f}")
        # flattening phase
        if time.time() < end_time - 2.0:
            a_flat, max_flat = _flatten_convolution(best_seq, end_time - 2.0, rng)
            if max_flat < best_score:
                best_score = max_flat
                best_seq = a_flat.copy()
                print(f"[flatten] improved to {best_score:.6f}")

    # Final check
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}, internal max = {best_score:.6f}, length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq

def _flatten_convolution(a, deadline, rng):
    n = len(a)
    S = np.sqrt(2.0 * n)
    eta = S * 0.1  # starting step size relative to sum
    iteration = 0
    while time.time() < deadline:
        b = _full_convolution_fft(a)
        max_b = b.max()
        active = np.where(np.isclose(b, max_b, atol=1e-12))[0]
        # compute subgradient
        grad = _subgradient_max_convolution(a, active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break
        # step direction: move opposite to gradient (reduce max)
        step = eta * grad / grad_norm
        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, 0.0)
        # reproject to sum S
        sum_candidate = a_candidate.sum()
        if sum_candidate == 0:
            break
        a_candidate = a_candidate * S / sum_candidate
        new_max = _full_convolution_fft(a_candidate).max()
        # accept if improvement
        if new_max < max_b:
            a = a_candidate
            max_b = new_max
            eta *= 1.05  # maybe increase step if improvement
        else:
            eta *= 0.5
        iteration += 1
        if iteration % 100 == 0:
            print(f"[flatten] iter={iteration} max={max_b:.6f} eta={eta:.6f}")
        if eta < 1e-6 * S:
            break
    return a, max_b

    if style == 'uniform':
        a = np.full(n, S / n)
    elif style == 'dirichlet':
        a = rng.random(n)
        a = a / a.sum() * S

        a = raw / raw.sum() * S
    return a

"""
adversarial step‑function optimizer – extended version

This version adds a deterministic “arcsine” start shape (the
continuous arcsine density 1/√(x(1‑x)) discretised on the interval),
a simple projected sub‑gradient “flattening” phase that directly
reduces the peak of the self‑convolution, and a richer pool of
initialisers.  The pipeline is

1. many short Adam restarts with a mixture of random, edge‑heavy and
   arcsine initial vectors;
2. a long Adam refinement of the current best candidate;
3. a final deterministic flattening pass that moves mass away from
   indices that contribute most to the active maximal convolution
   entries;
4. a sanity check with the supplied ``evaluate_sequence`` routine.

The result is returned as a plain list of non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parameter‑ised edge‑heavy initial shapes
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy families (power law + optional uniform mix)."""
    S = np.sqrt(2.0 * n)

    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    p = rng.uniform(0.2, 1.0)          # exponent
    offset = rng.uniform(0.0, 5.0)     # denominator shift
    mix = rng.uniform(0.0, 1.0)        # uniform background proportion

    edge_part = (d + offset) ** (-p)
    raw = mix * edge_part + (1.0 - mix) * np.ones(n)

    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _arcsine_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Deterministic arcsine‑type start vector (≈ 1/√(x(1‑x)) )."""
    S = np.sqrt(2.0 * n)

    # mid‑point grid  x_i = (i+0.5)/n
    i = np.arange(n) + 0.5
    # denominator ≈ n·√(x(1‑x))
    denom = np.sqrt(i * (n + 0.5 - i))
    raw = 1.0 / (denom + 1e-8)          # 1/(n·√(x(1‑x))) up to a constant

    # mix a tiny uniform part to avoid pathological zero‑derivatives
    mix = rng.uniform(0.0, 0.4)
    raw = (1.0 - mix) * raw + mix * np.ones(n)

    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.ifft(Fmask * Fa_rev).real[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered and its hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)           # required ℓ₁‑norm

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3

    # initialise temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient (soft‑max)
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9997
        lr *= 0.9999

        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Sub‑gradient of the true max(b) (active‑set formulation)
# ----------------------------------------------------------------------
def _subgradient_max_convolution(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(b) where b = a*a.
    active_shifts – indices k where b_k attains the maximum.
    """
    n = len(a)
    g = np.zeros_like(a)

    for k in active_shifts:
        if k == 0:
            g += 2.0 * a
            continue
        # contributions for i + k = j  (i from 0..n-1-k)
        a_i = a[: n - k]
        a_j = a[k:]
        g[: n - k] += a_j          # derivative w.r.t. a_i
        g[k:] += a_i                # derivative w.r.t. a_j

    # average over the active set (any convex combination is valid)
    if len(active_shifts) > 0:
        g /= len(active_shifts)
    return g


def _flatten_convolution(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent that explicitly reduces
    the hard peak max(b).  The routine runs until time runs out or
    the step size becomes negligible.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    eta = S * 0.2                # initial step magnitude
    it = 0

    while time.time() < deadline:
        it += 1
        b = _full_convolution_fft(a)
        max_b = b.max()
        active = np.where(np.isclose(b, max_b, atol=1e-12))[0]

        grad = _subgradient_max_convolution(a, active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break

        # step opposite to the gradient
        step = eta * grad / grad_norm
        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            break
        a_candidate = a_candidate * (S / s)

        new_max = _full_convolution_fft(a_candidate).max()
        # accept if we managed to lower the peak
        if new_max < max_b:
            a = a_candidate
            max_b = new_max
            eta *= 1.03                     # slightly enlarge step after success
        else:
            eta *= 0.6                      # otherwise shrink aggressively

        # safety‑net for tiny steps
        if eta < 1e-8 * S:
            break

        if it % 200 == 0:
            print(f"[flatten] iter={it}  peak={max_b:.6f}  eta={eta:.6f}")

    return a, max_b
