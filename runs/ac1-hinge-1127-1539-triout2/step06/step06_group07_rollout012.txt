```
def full_convolution_fft(a):
    n = len(a)
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv
```

```
    best_a = a.copy()
    best_max = cur_max

    start = time.time()

    while time.time() < time_limit:

        # Plan move
        i = rng.integers(0, n)  # source

        while j == i:
            j = rng.integers(0, n)
        # delta fraction
        delta = a[i] * rng.random() * 0.05 # up to 5% of source
        # apply move
        a[i] -= delta
        a[j] += delta

        conv_new = full_convolution_fft(a)
        max_new = conv_new.max()
        # acceptance
        if max_new <= cur_max or rng.random() < np.exp(-(max_new - cur_max) / T):

            cur_max = max_new

                best_max = max_new
                best_a = a.copy()

            a[i] += delta
```

```python
def _shape_score(p, n, S):
    # p includes interior heights only (length K-2)
    K = p.shape[0] + 2
    h = np.concatenate(([0.0], p, [0.0]))
    # Interpolate linearly
    # Use normalized x coordinates from 0 to 1
    xi = np.linspace(0.0, 1.0, n)
    xh = np.linspace(0.0, 1.0, K)
    a = np.interp(xi, xh, h)
    # Scale to sum S
    sum_a = a.sum()
    if sum_a == 0:
        a = np.full(n, S / n)
    else:
        a = a * (S / sum_a)
    conv = _full_convolution(a)
    return conv.max(), a
```

def _optimize_shape(n, deadline, rng):
    S = np.sqrt(2.0 * n)
    K = 30  # number of control points
    # initial p: linear ramp to 1 then back
    # simple start: half-sine values sampled.
    t = np.linspace(0.0, 1.0, K)
    init_h = np.sin(np.pi * t)  # 0 at ends, 1 at center
    p = init_h[1:-1].copy()
    # Initialize best
    best_score, best_a = _shape_score(p, n, S)
    best_p = p.copy()

    start = time.time()
    # Estimate initial temperature as fraction of best_score
    T0 = best_score * 0.1  # adapt
    it = 0
    while time.time() < deadline:
        it += 1
        # Mutate p
        sigma = 0.04 * np.maximum(p, 0.1)  # relative scale
        p_new = p + rng.normal(scale=sigma)
        p_new = np.maximum(p_new, 0.0)  # enforce nonnegative
        # Optionally regularize to keep shape smooth
        # Score
        score_new, a_new = _shape_score(p_new, n, S)
        # Accept if better or probabilistically

            best_p = p_new.copy()

def _vector_sa(a, deadline, rng):

    S = np.sqrt(2.0 * n)   # not needed? scaling invariance
    # Ensure sum normalized to S
    a = a * (S / a.sum())
    cur_conv = _full_convolution(a)
    cur_max = cur_conv.max()
    best_a = a.copy()
    best_val = cur_max
    # parameters
    while time.time() < deadline:
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        # delta bound
        delta = a[i] * rng.random() * 0.02
        a[i] -= delta
        a[j] += delta
        # maintain sum (it will be unchanged)
        conv = _full_convolution(a)
        new_max = conv.max()
        # accept if improvement or small chance
        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
                print(f"[vecSA] new best {best_val:.6f}")
        else:
            # revert with some probability maybe 0.01
            if rng.random() < 0.005:
                cur_max = new_max
                if new_max < best_val:
                    best_val = new_max
                    best_a = a.copy()
                    print(f"[vecSA] new best (uphill) {best_val:.6f}")
            else:
                a[i] += delta
                a[j] -= delta
        # occasional random reshuffle or jitter to escape
        if rng.random() < 0.001:
            # random perturb whole vector a bit
            noise = rng.normal(scale=0.001 * a.mean(), size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            conv = _full_convolution(a)
            cur_max = conv.max()
    return best_a, best_val

"""
A two‑stage optimizer for the step‑function problem.

Stage 1 – low‐dimensional shape search
    The vector is represented by a small set of control points (K≈30)
    linearly interpolated to length n and finally scaled so that its
    ℓ¹‑norm equals √(2 n).  Simulated annealing on the control points
    (non‑negative, no other constraints) quickly discovers a smooth
    shape with a low peak in the self‑convolution.

Stage 2 – fine‑grained mass‑transfer SA
    Starting from the best shape found, a classic simulated‑annealing
    routine moves tiny amounts of mass between two entries while keeping
    the total sum fixed.  The convolution is recomputed with an FFT each
    step; uphill moves are occasionally accepted to escape shallow traps.

Both stages respect the global 1000 s budget and print intermediate
progress.  The final sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via FFT (real input → real output)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Compute the linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())  # next power of two
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Helper for the shape‑based search (interpolation + scaling)
# ----------------------------------------------------------------------


def _shape_score(p: np.ndarray, n: int, S: float):
    """
    Given control points p (length K), build a length‑n candidate,
    scale it to ℓ¹‑norm S, return (max(convolution), candidate vector).
    """
    K = p.shape[0]
    # interpolation points in [0,1]
    x_control = np.linspace(0.0, 1.0, K)
    x_target = np.linspace(0.0, 1.0, n)
    a = np.interp(x_target, x_control, p)

    # scale to the required ℓ¹‑norm (the evaluation metric is scale‑invariant)
    sum_a = a.sum()
    if sum_a == 0.0:
        a = np.full(n, S / n)
    else:
        a = a * (S / sum_a)

    conv = _full_convolution(a)
    return float(conv.max()), a


# ----------------------------------------------------------------------
# Stage 1 – simulated annealing on the low‑dimensional shape
# ----------------------------------------------------------------------


def _optimize_shape(n: int, deadline: float, rng: np.random.Generator):
    """
    Find a good shape for a vector of length n.
    Returns the best full‑length vector and its max‑convolution value.
    """
    S = np.sqrt(2.0 * n)            # target ℓ¹‑norm (any positive number works)
    K = 30                           # number of control points (feel free to tweak)
    # initialise control points with a half‑sine curve – zero at the ends,
    # a single bump in the middle.
    t = np.linspace(0.0, 1.0, K)
    p = np.sin(np.pi * t)           # shape is already non‑negative
    cur_score, cur_a = _shape_score(p, n, S)

    best_score = cur_score
    best_p = p.copy()
    best_a = cur_a.copy()

    # initial temperature (a fraction of the current objective)
    T0 = best_score * 0.4
    start = time.time()

    while time.time() < deadline:
        # ------------------- propose a mutation -------------------
        # relative Gaussian perturbation, scaled to the magnitude of each coordinate
        sigma = 0.04 * np.maximum(p, 0.1)
        p_new = p + rng.normal(scale=sigma)
        p_new = np.maximum(p_new, 0.0)    # enforce non‑negativity

        # ------------------- evaluate -------------------
        new_score, new_a = _shape_score(p_new, n, S)

        # ------------------- acceptance -------------------
        # linear cooling: temperature falls from T0 to ~0 at the deadline
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(T0 * (1.0 - frac), 1e-12)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            p = p_new
            cur_score = new_score
            cur_a = new_a

            if new_score < best_score:
                best_score = new_score
                best_p = p_new.copy()
                best_a = new_a.copy()
                print(f"[shape] time {elapsed:.1f}s new best {best_score:.6f}")

        # occasional large random restart to diversify the search
        if rng.random() < 0.001:
            p = rng.random(K)          # fresh random control points
            cur_score, cur_a = _shape_score(p, n, S)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – fine‑grained simulated annealing directly on the full vector
# ----------------------------------------------------------------------


def _refine_vector(a_start: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Local SA that moves tiny amounts of mass between two entries.
    The total ℓ¹‑norm stays constant, and we keep the best vector seen.
    """
    a = a_start.copy()
    n = a.shape[0]

    # current objective
    cur_conv = _full_convolution(a)
    cur_max = float(cur_conv.max())

    best_a = a.copy()
    best_val = cur_max

    # temperature schedule (start relatively high, cool to near‑zero)
    start = time.time()
    T0 = best_val * 0.2

    while time.time() < deadline:
        # pick source and destination indices
        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # move a small fraction of the source mass
        delta = a[i] * rng.random() * 0.03   # ≤ 3 % of source entry
        a[i] -= delta
        a[j] += delta

        # recompute the whole convolution (fast FFT)
        conv = _full_convolution(a)
        new_max = float(conv.max())

        # acceptance test
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(T0 * (1.0 - frac), 1e-12)

        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
                print(f"[refine] time {elapsed:.1f}s new best {best_val:.6f}")
        else:
            # reject – revert the move
            a[i] += delta
            a[j] -= delta

        # tiny global jitter every few thousand steps to avoid stagnation
        if rng.random() < 0.0005:
            noise = rng.normal(scale=0.001 * a.mean(), size=n)
            a += noise
            a = np.maximum(a, 0.0)
            # renormalise to keep the ℓ¹‑norm constant
            a = a * (np.sqrt(2.0 * n) / a.sum())
            cur_max = float(_full_convolution(a).max())
            if cur_max < best_val:
                best_val = cur_max
                best_a = a.copy()
                print(f"[refine] jitter improvement {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` (i.e. the peak of its self‑convolution).
    The routine runs a two‑stage optimiser (shape‑SA + vector‑SA)
    and returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final external call
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Decide on problem size – 4000 provides a good trade‑off between
    # expressiveness and speed under the 1000 s limit.
    # --------------------------------------------------------------
    n = 4000

    # -------- Stage 1: shape optimisation ---------------------------------
    # Allocate roughly 40 % of the available time to the low‑dimensional search.
    shape_deadline = start_time + 0.40 * (deadline - start_time)
    best_vec, best_score_shape = _optimize_shape(n, shape_deadline, rng)

    # -------- Stage 2: fine‑grained SA ------------------------------------
    # Use the remaining time to polish the full vector.
    best_vec, best_score = _refine_vector(best_vec, deadline, rng)

    # --------------------------------------------------------------
    # Return the result as a plain Python list.
    # --------------------------------------------------------------
    final_seq = best_vec.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external_score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
