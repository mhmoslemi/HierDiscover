"""
Multi‑stage optimisation for the step‑function problem.

The objective after the ℓ¹‑normalisation (Σa = √(2 n)) reduces to the
maximum entry of the self‑convolution a*a.  We therefore try to make
this convolution as flat as possible.

The optimiser proceeds in phases:

1. **Progressive up‑sampling** – start with a modest length, improve it,
   then interpolate to a larger length.  This gives a good warm‑start for
   big n (up to 40 000).

2. **Projected gradient descent on a smooth log‑sum‑exp surrogate of
   the max** – the surrogate approaches the true max when the inverse
   temperature β grows.  Momentum and occasional jitter help escape
   plateaus.

3. **Targeted hill‑climbing** – move a random amount of mass from a
   source index that contributes heavily to the current worst convolution
   entry to a destination that is “light”.  The full convolution is kept
   up‑to‑date in O(n) per move, allowing many moves in the given time.

4. **Optional symmetry & smoothing** – a symmetric distribution often
   lowers the peak; a light Gaussian smoothing followed by a projection
   onto the simplex can give a final polish.

All vectors stay non‑negative and are projected onto the simplex
{ a_i≥0, Σa_i = √(2 n) } after every stochastic modification,
so the evaluation function sees a valid step function.
The routine respects a wall‑clock budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)   (cross‑correlation)
    # cross‑correlation can be obtained via FFT:
    #   corr[t] = Σ_i a_i * p_{i+t} = ifft(conj(A) * P)[t]
    P_fft = np.fft.rfft(p, n=size1)
    g_full = np.fft.irfft(np.conj(a_fft) * P_fft, n=size1)[: 2 * n - 1]
    # we need only the part corresponding to shifts 0..n‑1
    g = 2.0 * g_full[:n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Momentum, annealed β and occasional jitter are used.
    The loop terminates when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 2 ** 14          # ≈ 16384
    beta_factor = 1.35

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σa = √(2 n).  Several families (power‑law, exponential, Dirichlet,
    piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:          # power‑law
        p = rng.uniform(0.15, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:        # exponential decay
        lam = rng.exponential(scale=0.02)
        a = np.exp(-lam * idx)

    elif family == 2:        # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.02)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:        # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:        # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    else:                    # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed
    by a projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers a random amount
    of mass from a source index to a destination index while keeping the
    full self‑convolution up‑to‑date in O(n) per move.  The source index is
    chosen proportionally to its contribution to the current worst
    convolution entry; the destination is biased towards small entries.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.12 + 1e-9          # initial temperature
    cooling = 0.9994                     # slower cooling for fine search

    while time.time() < deadline:
        # pick a maximal convolution index
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index – weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – mostly in low‑mass region
        if rng.random() < 0.35:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.78
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution efficiently (linear work)
        b_new = b.copy()
        # linear terms (pairwise contributions)
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        # quadratic correction
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b


# ----------------------------------------------------------------------
# Additional polishing helpers
# ----------------------------------------------------------------------


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (a_i = a_{n-1-i}) and project back."""
    a_sym = (a + a[::-1]) * 0.5
    target = np.sqrt(2.0 * a.size)
    return _project_to_simplex(a_sym, target)


def _smooth_and_project(a: np.ndarray, sigma: float) -> np.ndarray:
    """
    Apply a light Gaussian smoothing (via FFT) and project onto the simplex.
    sigma is the standard deviation as a fraction of the length.
    """
    n = a.size
    # build Gaussian kernel on the same grid
    x = np.arange(-n // 2, n // 2 + 1)
    kernel = np.exp(-0.5 * (x / (sigma * n)) ** 2)
    kernel /= kernel.sum()
    size = _next_pow2(n + kernel.size - 1)
    A = np.fft.rfft(a, n=size)
    K = np.fft.rfft(kernel, n=size)
    a_smooth = np.fft.irfft(A * K, n=size)[:n]
    a_smooth = np.maximum(a_smooth, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(a_smooth, target)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Length schedule – gradually increase the dimension.
    # More time is reserved for the larger lengths where improvements
    # are more significant.
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 30000, 40000]

    for n in schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – projected gradient descent with multiple restarts
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        # allocate roughly 30 % of the remaining budget to gradient descent
        gd_budget = min(remaining * 0.30, 180.0)   # cap to keep later phases alive
        gd_deadline = time.time() + gd_budget

        # a few restarts – the exact number is adaptive to remaining time
        while time.time() < gd_deadline:
            a_start = a_init.copy() if rng.random() < 0.7 else _random_initial_sequence(n, rng)
            a_gd = _gradient_descent_opt(a_start, gd_deadline)
            score = _max_convolution_fft(a_gd)
            if score < best_score:
                best_score = score
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] GD improvement for n={n} score={score:.6f}")

        # --------------------------------------------------------------
        # Phase B – hill climbing with exact convolution updates
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:
            break
        hc_budget = min(remaining * 0.30, 250.0)   # allocate a good chunk to HC
        hc_deadline = time.time() + hc_budget

        # ensure we are working with the current length
        if best_seq.size != n:
            a_cur = _upsample_sequence(best_seq, n)
        else:
            a_cur = best_seq.copy()
        a_cur = _project_to_simplex(a_cur, target_sum)
        b_cur = _full_convolution(a_cur)

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

        # --------------------------------------------------------------
        # Phase C – optional symmetry & light smoothing (if time permits)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 4.0:
            # try symmetry
            a_sym = _symmetrize(best_seq)
            score_sym = _max_convolution_fft(a_sym)
            if score_sym < best_score:
                best_score = score_sym
                best_seq = a_sym.copy()
                print(f"[{time.time():.1f}] Symmetrisation improvement for n={n} score={score_sym:.6f}")

            # light Gaussian smoothing (σ ≈ 0.005 of length)
            a_smooth = _smooth_and_project(best_seq, sigma=0.005)
            score_smooth = _max_convolution_fft(a_smooth)
            if score_smooth < best_score:
                best_score = score_smooth
                best_seq = a_smooth.copy()
                print(f"[{time.time():.1f}] Smoothing improvement for n={n} score={score_smooth:.6f}")

        # --------------------------------------------------------------
        # Small final gradient pass to polish after the above moves
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 3.0:
            a_post = _gradient_descent_opt(best_seq.copy(), time.time() + remaining)
            score_post = _max_convolution_fft(a_post)
            if score_post < best_score:
                best_score = score_post
                best_seq = a_post.copy()
                print(f"[{time.time():.1f}] Final GD polish for n={n} score={score_post:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best length we have
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        b = _full_convolution(a)
        a_fin, b_fin = _hill_climbing(a, b, deadline)
        final_score = b_fin.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_fin.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

        # one more symmetry/smoothing pass (cheap)
        a_sym = _symmetrize(best_seq)
        if _max_convolution_fft(a_sym) < best_score:
            best_seq = a_sym.copy()
            best_score = _max_convolution_fft(best_seq)
        a_smooth = _smooth_and_project(best_seq, sigma=0.004)
        if _max_convolution_fft(a_smooth) < best_score:
            best_seq = a_smooth.copy()
            best_score = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
