def _lp_optimize(f, target_sum, deadline):
    n = f.size
    # Build A matrix
    from scipy.linalg import toeplitz
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    A = toeplitz(col, row)  # shape (2n-1, n)
    m = 2 * n - 1
    # Assemble A_ub
    # add column -1 for t variable
    A_ub = np.hstack([A, -np.ones((m,1))])
    b_ub = np.zeros(m)
    # Equality constraints
    A_eq = np.hstack([np.ones((1,n)), np.zeros((1,1))])
    b_eq = np.array([target_sum])
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    # objective
    c = np.concatenate([np.zeros(n), np.array([1.0])])
    # solve
    from scipy.optimize import linprog
    # we may need to adjust method to 'highs' for speed.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'time_limit': max(10.0, deadline - time.time())})
    if res.success:
        g = res.x[:n]
        return g
    else:
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None

The goal is to minimise  

\[
\max_k (a*a)_k\qquad\text{with}\qquad a_i\ge 0,\; \sum_i a_i=\sqrt{2n},
\]

i.e. to make the peak of the self‑convolution as small as possible.
The programme below searches for a good vector `a` under the
1000 s wall‑clock limit.

Main ideas
----------

1. **Simplex projection** – after every stochastic move the vector is
   projected onto the ℓ¹‑simplex  
   \(\{a\mid a_i\ge 0,\;\sum_i a_i = \sqrt{2n}\}\) so the objective
   is always well‑defined.

2. **Projected gradient descent** on a **log‑sum‑exp** smooth approximation
   of the maximum.  
   Temperature (β) is annealed, momentum and occasional jitter are used.
   Several short random restarts give a diverse set of candidates.

3. **Linear‑programming refinement** (small dimensions only).  
   For the current candidate `f` we solve  

   \[
   \min_{g,t}\;t\qquad
   \text{s.t.}\;\;(f*g)_k \le t\;(k=0\ldots 2n-2),\;
   \sum_i g_i = \sqrt{2n},\; g_i\ge0 .
   \]

   This LP finds a vector `g` that reduces the worst cross‑convolution
   with `f`.  After scaling `g` to the correct ℓ¹‑norm we keep the
   better of `f` and `g` (or a convex blend of the two).

4. **Exact‑update mass‑transfer hill climbing** – a simulated‑annealing
   style local search that moves a random amount of mass from a source
   index to a destination index while keeping the full convolution
   up‑to‑date in O(n) per move.

5. **Length schedule** – start with modest dimensions (≤ 4096) where the
   LP is cheap, then progressively up‑sample to larger sizes (up to
   30 000) and finish with a long hill‑climb.  The best vector found at
   each stage is carried forward.

All building blocks are pure NumPy/ SciPy, run on a single core and

```python
"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence (i.e. the maximal entry of a*a under a fixed ℓ¹‑norm).

The search combines:
* projected gradient descent on a log‑sum‑exp surrogate,
* a tiny LP that minimises the worst cross‑convolution (small n only),
* a mass‑transfer hill‑climber with exact O(n) convolution updates,
* an increasing length schedule (up‑sampling) and a final polishing
  hill‑climb.

All vectors are projected onto the simplex {a_i≥0, Σa_i = √(2n)} after
every stochastic move, guaranteeing that evaluate_sequence sees a
valid step‑function.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x_i≥0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)               # soft‑max probabilities

    # gradient = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Momentum, annealed inverse temperature β and occasional jitter are used.
    The loop stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few iterations
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σa = √(2 n).  Several families are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )
    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a onto a new length, followed by projection.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray,
                   b0: np.ndarray,
                   deadline: float,
                   *, initial_temp: float = None,
                   cooling: float = 0.9995) -> tuple[np.ndarray, np.ndarray]:
    """
    Simulated‑annealing style hill climbing.
    Moves a random amount of mass from a source index to a destination index,
    keeping the full self‑convolution up‑to‑date in O(n) per move.
    Parameters can be tuned via *initial_temp* and *cooling*.
    Returns (best_a, best_b) where best_b is the final convolution.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    if initial_temp is None:
        T = cur_score * 0.1 + 1e-9
    else:
        T = initial_temp

    while time.time() < deadline:
        # pick one of the current max positions
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index (biased to small entries)
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of the source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution efficiently
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a          # remove contribution of source
        b_new[j:j + n] += 2.0 * delta * a          # add contribution of destination

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling
    return best_a, b


# ----------------------------------------------------------------------
# Linear‑programming refinement (small dimensions)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, target_sum: float, deadline: float) -> np.ndarray | None:
    """
    Solve a tiny LP that minimises the worst cross‑convolution
        min_t  t
        s.t.   (f * g)_k ≤ t   ∀k,
               Σ g_i = target_sum,
               g_i ≥ 0.
    Returns a new feasible vector g (already projected onto the simplex) or
    None if the LP fails or the time budget is exhausted.
    """
    n = f.size
    # Build convolution matrix A (2n‑1 × n) with Toeplitz structure
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    A = toeplitz(col, row)                      # shape (2n‑1, n)

    m = 2 * n - 1
    # variables: [g_0,…,g_{n‑1}, t]
    A_ub = np.hstack([A, -np.ones((m, 1))])    # A g – t ≤ 0
    b_ub = np.zeros(m)

    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([target_sum])

    bounds = [(0.0, None)] * n + [(0.0, None)]   # t ≥ 0 as well

    c = np.concatenate([np.zeros(n), np.array([1.0])])   # minimise t

    # tiny safety margin for the LP solver
    time_left = max(0.0, deadline - time.time())
    # SciPy's Highs accepts a 'time_limit' option (seconds) from 1.9.0 onward.
    # If unavailable it will just ignore the entry.
    options = {"presolve": True}
    if time_left > 0.0:
        options["time_limit"] = time_left

    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  A_eq=A_eq,
                  b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options=options)

    if not res.success:
        print(f"[{time.time():.1f}] LP failed: {res.message}")
        return None

    g = res.x[:n]
    # tiny numerical drift – enforce exact simplex
    g = _project_to_simplex(g, target_sum)
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget (default 1000 s) and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)
            best_seq = a0.copy()
            best_score = cur
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={cur:.6f}")

    # ------------------------------------------------------------------
    # Small‑dimension schedule (LP works well here)
    # ------------------------------------------------------------------
    small_schedule = [512, 1024, 2048, 4096]    # ≤ 4096 to keep LP memory reasonable

    for n in small_schedule:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # initialise
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)
        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase 1 – several short gradient‑descent restarts
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 5.0:
            break
        gd_budget = min(remaining * 0.30, 180.0)      # at most 180 s for this length
        n_restarts = 4
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            per_restart = min(gd_budget / (n_restarts - r), 45.0)
            start = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_gd = _gradient_descent_opt(start, time.time() + per_restart)
            score = _max_convolution_fft(a_gd)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_gd.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] GD improvement n={n} score={local_best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – LP refinement (only for modest n)
        # --------------------------------------------------------------
        if n <= 4096 and time.time() < deadline - 5.0:
            cur = best_seq if best_seq.size == n else _upsample_sequence(best_seq, n)
            cur = _project_to_simplex(cur, target_sum)
            lp_iters = 2
            for _ in range(lp_iters):
                if time.time() >= deadline:
                    break
                g = _lp_refine(cur, target_sum, deadline)
                if g is None:
                    break
                # evaluate the candidate g
                g_score = _max_convolution_fft(g)
                if g_score < best_score:
                    best_score = g_score
                    best_seq = g.copy()
                    cur = g.copy()
                    print(f"[{time.time():.1f}] LP improvement n={n} score={g_score:.6f}")
                else:
                    # try a convex blend of the two vectors
                    blend = 0.5
                    blended = _project_to_simplex(blend * g + (1 - blend) * cur, target_sum)
                    blended_score = _max_convolution_fft(blended)
                    if blended_score < best_score:
                        best_score = blended_score
                        best_seq = blended.copy()
                        cur = blended.copy()
                        print(f"[{time.time():.1f}] blended LP‑GD improvement n={n} score={blended_score:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – hill climbing with exact convolution updates
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            a_cur = best_seq if best_seq.size == n else _upsample_sequence(best_seq, n)
            a_cur = _project_to_simplex(a_cur, target_sum)
            b_full = _full_convolution(a_cur)
            a_hc, b_hc = _hill_climbing(a_cur, b_full, deadline)
            score_hc = b_hc.max()
            if score_hc < best_score:
                best_score = score_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement n={n} score={score_hc:.6f}")

    # ------------------------------------------------------------------
    # Upsample to large sizes and finish with a long hill‑climb
    # ------------------------------------------------------------------
    large_lengths = [10000, 20000, 30000]

    for n in large_lengths:
        if time.time() >= deadline - 5.0:
            break
        target_sum = np.sqrt(2.0 * n)
        a_large = _upsample_sequence(best_seq, n)
        a_large = _project_to_simplex(a_large, target_sum)

        # allocate most of the remaining budget to this final hill‑climb
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        hill_deadline = time.time() + remaining * 0.85
        b_full = _full_convolution(a_large)

        # use a slower cooling schedule for a thorough search
        a_hc, b_hc = _hill_climbing(a_large,
                                    b_full,
                                    hill_deadline,
                                    initial_temp=None,
                                    cooling=0.9998)
        score = b_hc.max()
        if score < best_score:
            best_score = score
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Final hill‑climb length={n} score={score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        b = _full_convolution(a)
        a_fin, b_fin = _hill_climbing(a, b, deadline)
        fin_score = b_fin.max()
        if fin_score < best_score:
            best_score = fin_score
            best_seq = a_fin.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {fin_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never be needed)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (should equal the maximal convolution entry)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```
