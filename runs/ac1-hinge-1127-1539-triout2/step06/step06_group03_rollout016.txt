def _bit_reverse_order(arr):
    n = arr.shape[0]
    bits = int(np.ceil(np.log2(n)))
    rev = np.empty(n, dtype=int)
    for i in range(n):
        x = i
        r = 0
        for b in range(bits):
            r = (r << 1) | (x & 1)
            x >>= 1
        rev[i] = r
    # sorting indices by rev modulo n to avoid >n
    order = np.argsort(rev % n)
    return arr[order]

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.
The algorithm consists of three stages:

1. **Broad sampling** – generate many candidate vectors from several
   parametric families (power‑law, exponential, truncated linear,
   beta‑shaped, mixture of two power‑laws).  For each base vector a small
   collection of permutations (ascending, descending, alternating,
   step‑permutations, bit‑reversal and a few random shuffles) is tried.
   The best few candidates (the “elite”) are kept.

2. **Refinement** – each elite candidate is refined by an Adam‑style
   projected gradient descent on a smooth log‑sum‑exp surrogate of the
   maximal convolution entry, followed by a permutation‑annealing phase
   (swap moves) and a cheap local‐value search that transfers a small
   amount of mass between two positions.

3. **Final polish** – a last local‐value search is performed on the
   globally best sequence.

All steps respect a 1000 s deadline and use only NumPy and the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σ x = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of a smoothed maximum (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam‑style optimisation of the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    beta = 1.0
    max_beta = 5e5
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size scaled to the current norm of the gradient
        step = 0.2 * target_sum / (np.sqrt(np.mean(v_hat)) + 1e-12)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

        # increase the smoothness parameter periodically
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return a

def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improves the ordering of a fixed multiset by random swaps.
    The L¹‑norm stays unchanged – only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny reshuffle
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best

def _local_value_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Repeatedly move a small amount of weight between two entries."""
    best = a0.copy()
    best_score = _max_convolution_fft(best)
    n = best.size
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] < 1e-12:
            continue
        delta = rng.uniform(0, best[i] * 0.5)   # up to half of a_i
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        score = _max_convolution_fft(cand)
        if score < best_score:
            best_score = score
            best = cand
            print(f"[{time.time():.1f}]   Local search improvement: {best_score:.6f}")

        # occasional full reshuffle to escape shallow local minima
        if rng.random() < 0.001:
            perm = np.arange(n)
            rng.shuffle(perm)
            cand2 = best[perm]
            score2 = _max_convolution_fft(cand2)
            if score2 < best_score:
                best_score = score2
                best = cand2

    return best

def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale an array to have ℓ¹‑norm = target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _beta_shape_sequence(n: int, a_exp: float, b_exp: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^a_exp * (n‑i)^b_exp."""
    i = np.arange(1, n + 1, dtype=float)
    raw = (i ** a_exp) * ((n - i + 1) ** b_exp)
    raw = np.maximum(raw, 0.0)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _mix_powerlaw_sequence(n: int, p1: float, shift1: float,
                          p2: float, shift2: float, w: float) -> np.ndarray:
    """Convex combination of two power‑law tails."""
    idx = np.arange(n, dtype=float)
    raw = w * (idx + shift1) ** (-p1) + (1.0 - w) * (idx + shift2) ** (-p2)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))

def _bit_reverse_permutation(arr: np.ndarray) -> np.ndarray:
    """Reorder ``arr`` according to a bit‑reversal permutation."""
    n = arr.size
    bits = int(math.ceil(math.log2(n)))
    rev = np.empty(n, dtype=int)
    for i in range(n):
        x = i
        r = 0
        for _ in range(bits):
            r = (r << 1) | (x & 1)
            x >>= 1
        rev[i] = r
    order = np.argsort(rev % n)
    return arr[order]


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a modest collection of orderings for a given multiset.
    The goal is to scatter large values so that they overlap as little as possible.
    """
    n = base.size
    asc = np.sort(base)                 # small → large
    desc = asc[::-1]                    # large → small
    out = [asc, desc]

    # alternating low–high (low first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # step permutations with coprime strides
    for step in (2, 3, 5, 7, 11, 13, 17, 19, 23):
        if math.gcd(step, n) != 1:
            continue
        perm = np.empty_like(base)
        pos = 0
        for i in range(n):
            perm[pos] = asc[i]
            pos = (pos + step) % n
        out.append(perm)

    # bit‑reversal ordering
    out.append(_bit_reverse_permutation(asc))

    # a few random shuffles
    for _ in range(3):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out

            p1 = rng.uniform(0.44, 0.58)
            p2 = rng.uniform(0.44, 0.58)
            shift1 = rng.uniform(0.0, 3.0)
            shift2 = rng.uniform(0.0, 3.0)
            w = rng.uniform(0.0, 1.0)
            base = _mix_powerlaw_sequence(n, p1, shift1, p2, shift2, w)
            for cand in _candidate_orderings(base, rng):
                score = _max_convolution_fft(cand)
                if score < best_obj:
                    best_obj = score
                    best_seq = cand.copy()
                    print(f"[{time.time():.1f}]   New global best (mix) n={n} obj={score:.6f}")
                if len(elite) < max_elite or score < elite[-1][0]:
                    elite.append((score, cand))
                    elite.sort(key=lambda x: x[0])
                    elite = elite[:max_elite]

        # --- short random sampling --------------------------------------
        rnd_end = min(deadline, time.time() + 1.0)
        while time.time() < rnd_end:
            rand = rng.random(n)
            rand = _normalize_to_sum(rand, target_sum)
            score = _max_convolution_fft(rand)
            if score < best_obj:
                best_obj = score
                best_seq = rand.copy()
                print(f"[{time.time():.1f}]   New global best (rand) n={n} obj={score:.6f}")
            if len(elite) < max_elite or score < elite[-1][0]:
                elite.append((score, rand))
                elite.sort(key=lambda x: x[0])
                elite = elite[:max_elite]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if available
    # ------------------------------------------------------------------
    try:
        seq1 = height_sequence_1          # type: ignore[name-defined]
        n1 = len(seq1)
        a1 = np.array(seq1, dtype=float)
        a1 = _project_to_simplex(a1, np.sqrt(2.0 * n1))
        score = _max_convolution_fft(a1)
        if score < best_obj:
            best_obj = score
            best_seq = a1.copy()
            print(f"[{time.time():.1f}]   New global best (prev) n={n1} obj={score:.6f}")
        elite.append((score, a1))
        elite.sort(key=lambda x: x[0])
        elite = elite[:max_elite]
    except NameError:
        pass  # no previous sequence provided

    # ------------------------------------------------------------------
    # 3. Refinement of the elite candidates
    # ------------------------------------------------------------------
    # Keep only the top few (up to 12) for a more intensive refinement
    elite_to_refine = elite[: min(12, len(elite))]

    for base_score, seed_seq in elite_to_refine:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        # allocate roughly 30 s per elite (or whatever remains)
        remaining = deadline - time.time()
        if remaining <= 5.0:
            break
        alloc = min(30.0, remaining * 0.85)
        sub_dead = time.time() + alloc

        # 3.1 Adam gradient descent on the smooth surrogate
        refined = _adam_optimize(seed_seq, sub_dead)

        # 3.2 Permutation annealing (uses whatever time is left)
        refined = _permutation_annealing(refined, deadline)

        # 3.3 Light local value search to polish the numbers
        refined = _local_value_search(refined, deadline)

        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj = obj
            best_seq = refined.copy()
            print(f"[{time.time():.1f}]   NEW GLOBAL BEST (refined) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Final local search around the global best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq = _local_value_search(best_seq, deadline)

    # Trim far‑right negligible entries (purely cosmetic)
    best_arr = np.array(best_seq, dtype=float)
    nz = np.where(best_arr > 1e-12)[0]
    if nz.size:
        best_arr = best_arr[: nz[-1] + 1]

    # Final verification using the provided evaluate_sequence
    try:
        val = evaluate_sequence(best_arr.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}   length = {best_arr.size}")
    return best_arr.tolist()
