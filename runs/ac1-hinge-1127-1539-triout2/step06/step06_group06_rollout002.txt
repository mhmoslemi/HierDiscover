    target_sum = np.sqrt(2*n)

    beta2 = 0.999
    eps = 1e-8

    beta_soft = 1.0

        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

"""
Proposes a non‑negative step‑function (height sequence) that minimises the
benchmark `evaluate_sequence`.  The optimiser works on three increasing
lengths (≈1500 → 4000 → 8000).  For each length it runs:

1. Adam‑based projected gradient descent on a soft‑max approximation of the
   convolution maximum (β starts at 1 and is increased up to 16384).  The
   gradient is obtained analytically via FFT, the iterate is projected onto
   the ℓ¹‑simplex Σa_i = √(2·n) after each step.

2. Simulated‑annealing refinement with focused “mass‑transfer’’ moves that
   shift mass from indices that contribute heavily to the current maximum
   convolution entry to indices that contribute little.  The temperature and
   allowed transfer fraction decay exponentially.

The best sequence seen across all scales (including a few cheap random
re‑starts) is returned.  All numbers are forced to be non‑negative; the
final sequence is trimmed of trailing near‑zeros.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  For sequences normalised to
    Σa = √(2·n) this is exactly max(a*a), i.e. the quantity we try to
    minimise.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Gradient (soft‑max) utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))          # stabilised soft‑max
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p, then slice the centre part
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected Adam optimisation on the soft‑max approximation of the
    max‑convolution.  β is increased up to 16384.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_size = target_sum * 0.2

    beta = 1.0
    max_beta = 16384.0
    beta_factor = 1.25

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        a -= step_size * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase β gradually
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing utilities
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.25,
                         max_frac_end: float = 0.02,
                         max_window: int = 200) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves:
        * mass transfer from a “sensitive’’ index (large contribution to the
          current maximum convolution entry) to a low‑sensitivity one,
        * short‑window smoothing,
        * random swap.
    Temperature and allowed transfer fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index of the current maximum convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities (contribution of each a_i to the max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        # Weighted donor choice (more weight → larger chance to give away mass)
        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.68:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # Pick receiver among indices with low weight
            low_thr = np.percentile(weight, 20)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- local smoothing -----------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to shake stagnation
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _run_one_instance(n: int, init_seq: np.ndarray, deadline: float):
    """
    Optimise a single length‑n instance.
    Performs an Adam‑based gradient phase (≈30 % of allotted time) and a
    simulated‑annealing phase for the remainder.
    Returns (best_sequence, best_score).
    """
    target_sum = np.sqrt(2.0 * n)
    a = init_seq.astype(np.float64, copy=True)
    if a.size != n:
        a = _resize_sequence(a, n)
    a = _project_to_simplex(a, target_sum)

    # Gradient‑descent phase (≈30 % of remaining budget)
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + min(30.0, remaining * 0.30)
        a, gd_score = _adam_gradient_descent(a, gd_deadline)
    else:
        gd_score = _score(a)

    # Simulated‑annealing phase for the rest of the budget
    if time.time() < deadline:
        a, sa_score = _simulated_annealing(a, deadline)
        final_score = min(gd_score, sa_score)
    else:
        final_score = gd_score

    return a, final_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with low benchmark score.
    Uses three increasing scales (≈1500 → 4000 → 8000) and a mixture of
    deterministic optimisation (Adam + SA) together with cheap random
    restarts.  The best sequence found before the deadline is returned.
    """
    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # (length, fraction of remaining time) for each scale
    scales = [(1500, 0.15), (4000, 0.35), (8000, 0.50)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # Warm‑start from the previous scale if available
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            # fall back to a reasonable power‑law initialisation
            init = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

        # Main optimisation run for this scale
        seq, score = _run_one_instance(n, init, stage_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Scale n={n} (main) new best score={best_score:.6f}")

        cur_seq = seq   # warm start for the next (larger) scale

        # ----------- cheap random restarts on the same scale -------------
        # Allocate the remaining part of the stage for quick restarts.
        while time.time() < stage_deadline - 2.0:
            # random power‑law exponent
            p = np.random.uniform(0.35, 0.80)
            init_rand = _power_law_sequence(n, p, np.sqrt(2.0 * n))

            # give the restart at most 10 % of the remaining time
            sub_deadline = time.time() + (stage_deadline - time.time()) * 0.10
            seq2, score2 = _run_one_instance(n, init_rand, sub_deadline)

            if score2 < best_score:
                best_score = score2
                best_seq = seq2.copy()
                print(f"[{time.time():.2f}] Scale n={n} (restart p={p:.3f}) new best score={best_score:.6f}")

            # keep the best of the restarts as warm start for possible next run
            if score2 < _score(cur_seq):
                cur_seq = seq2

    # Trim trailing near‑zeros for neatness
    if best_seq is None:
        best_seq = np.array([], dtype=float)
    else:
        best_seq = np.array(best_seq, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()

    # Optional final evaluation (useful for debugging / logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
