assistantfinal**Solution Overview**

The evaluation function only depends on the *peak of the self‑convolution* of the step‑function, because the sum of the coefficients is normalised to  
\(\sum a_i = \sqrt{2\,n}\) and the returned score is  

\[
\text{score}= \frac{2n\;\max_k (a*a)_k}{(\sum_i a_i)^2}= \max_k (a*a)_k .
\]

Hence the task is: *find a non‑negative vector with given ℓ¹‑norm whose auto‑convolution has the smallest possible maximum entry*.

The program works in two stages  

1. **Explicit parametric search** – a few analytically‑motivated families
   (`sum_pow`, `prod_pow`, `exp_decay` and convex mixtures of them) are
   generated for several lengths `n`.  For each family a small grid of
   parameters (mainly a decay exponent `p` and a mixing weight `w`) is
   explored.  The best vector found by this cheap exhaustive search is kept
   as a global candidate.

2. **Local refinement** – the best candidate is fed into a three‑phase
   optimiser (smooth Adam → sub‑gradient descent → tiny random swaps) that
   works on the *soft‑max* approximation of the max‑convolution and respects
   the ℓ¹‑simplex constraint.  The optimiser runs until the remaining time
   budget is exhausted.

Both stages respect the time limit (default 1000 s) and never write to the
filesystem.  The final vector is stored in the global variable
`height_sequence_1` so that subsequent runs can start from it.

```python
"""
search for a non‑negative step‑function that minimises the peak of its
self‑convolution.  The algorithm consists of:

1. a fast exhaustive scan over a few analytically‑motivated families
   (edge‑heavy power laws, exponential decay and mixtures);
2. a refined Adam / sub‑gradient optimisation that works on the
   smooth log‑sum‑exp approximation of the max‑convolution.

The whole routine is time‑aware and returns the best vector found
within the allotted budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based) – full linear convolution a * a
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_full_convolution(a).max())


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp) – used in the smooth Adam phase
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over the convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth objective (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Optional symmetry enforcement
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _shape_sum_pow(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p} + (n-i)^{-p}  (symmetric, edge‑heavy)."""
    idx = np.arange(1, n + 1, dtype=float)
    return idx**(-p) + (n - idx + 1)**(-p)


def _shape_prod_pow(n: int, p: float) -> np.ndarray:
    """a_i = ((i+1)*(n-i))^{-p}   (strongly edge‑heavy)."""
    idx = np.arange(1, n + 1, dtype=float)
    prod = idx * (n - idx + 1)
    return prod**(-p)


def _shape_exp_decay(n: int, p: float) -> np.ndarray:
    """a_i = exp(-p·i) + exp(-p·(n-1-i))  (exponential edge decay)."""
    i = np.arange(n, dtype=float)
    return np.exp(-p * i) + np.exp(-p * (n - 1 - i))


def _shape_mix(n: int, p: float, w: float) -> np.ndarray:
    """
    Convex mixture of the previous two families:
    w * sum‑pow  + (1‑w) * prod‑pow.
    """
    a1 = _shape_sum_pow(n, p)
    a2 = _shape_prod_pow(n, p)
    return w * a1 + (1.0 - w) * a2


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.5
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = _conv_max(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat plateaus
        if rng.random() < 0.018:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = _conv_max(cand)
                if cand_max < best_score or rng.random() < 0.12:
                    a = cand

        alpha = min(alpha * alpha_mul, 45.0)
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum convolution entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _conv_max(a)

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = _conv_max(a)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = _conv_max(cand)
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap hill‑climbing that swaps tiny masses between random positions
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = _conv_max(a)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = _conv_max(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local search)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Runs the three optimisation phases within the supplied time budget.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, _conv_max(a0)

    # split 60 % / 30 % / 10 %
    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    The function first scans a handful of analytic families (power‑law
    and exponential edge‑heavy shapes) on several lengths, keeps the best
    candidate and then refines it with a three‑phase optimiser
    (Adam → sub‑gradient → local swaps).
    The returned list is suitable for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin for external call

    # ------------------------------------------------------------------
    # 1) cheap exhaustive search over parametrised families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_n = None

    # candidate lengths (keep them moderate so that FFTs stay cheap)
    candidate_lengths = [2000, 4000, 8000, 12000, 16000]

    # parameter grids
    p_grid = np.linspace(0.01, 0.49, 30)          # safe region (p<0.5)
    w_grid = [0.0, 0.33, 0.66, 1.0]              # mixture weight

    # pure families
    pure_families = ['sum_pow', 'prod_pow', 'exp_decay']

    for n in candidate_lengths:
        if time.time() > deadline:
            break
        S = np.sqrt(2.0 * n)

        # --- pure families -------------------------------------------------
        for mode in pure_families:
            if time.time() > deadline:
                break
            if mode == 'exp_decay':
                p_vals = np.linspace(0.001, 0.15, 30)
            else:
                p_vals = p_grid

            for p in p_vals:
                if mode == 'sum_pow':
                    a = _shape_sum_pow(n, p)
                elif mode == 'prod_pow':
                    a = _shape_prod_pow(n, p)
                else:  # exp_decay
                    a = _shape_exp_decay(n, p)

                total = a.sum()
                if total == 0:
                    continue
                a = a * (S / total)

                if a.max() > 1000.0:          # safety – evaluate_sequence caps at 1000
                    continue

                cur_score = _conv_max(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    best_n = n
                    print(f"[grid] n={n} mode={mode} p={p:.4f} score={cur_score:.6f}")

        # --- mixed family (sum_pow vs prod_pow) ----------------------------
        for p in p_grid:
            if time.time() > deadline:
                break
            for w in w_grid:
                a = _shape_mix(n, p, w)
                total = a.sum()
                if total == 0:
                    continue
                a = a * (S / total)

                if a.max() > 1000.0:
                    continue

                cur_score = _conv_max(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    best_n = n
                    print(f"[grid] n={n} mode=mix p={p:.4f} w={w:.2f} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2) local refinement of the best candidate (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Use the combined optimiser for the remaining time budget
        enforce_sym = True   # all families we used are symmetric
        best_seq, refined_score = _optimize_combined(best_seq,
                                                    deadline,
                                                    rng,
                                                    enforce_sym)
        best_score = refined_score

    # ------------------------------------------------------------------
    # 3) final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} "
          f"internal_best={best_score:.6f} "
          f"length={len(final_seq)}")

    # expose the best sequence for possible up‑sampling by a later run
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
```
