def _adam_opt(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0*n)
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    step_alpha = target_sum * 0.2  # maybe
    t = 1

    rng = np.random.default_rng()
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        m = 0.9*m + 0.1*g
        v = 0.999*v + 0.001*(g**2)
        m_hat = m/(1 - 0.9**t)
        v_hat = v/(1 - 0.999**t)
        lr = step_alpha/ (np.sqrt(v_hat)+eps) # elementwise
        a -= lr * m_hat
        # projection
        a = _project_to_simplex(a, target_sum)

        # jitter occasional
        if t % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005/(beta+1), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # beta annealing
        if t % 300 == 0 and beta < beta_max:
            beta = min(beta*beta_factor, beta_max)

        t += 1

    return a
```

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

* Initialise from a previously found construction (global ``height_sequence_1``)
  if it exists, otherwise generate a random simplex point.
* Upsample the seed to a relatively large size (default 20000).  A larger
  domain gives more freedom to flatten the self‑convolution.
* Run several restarts of projected gradient descent on a smooth
  log‑sum‑exp surrogate of the maximum convolution entry.  The optimiser
  uses momentum, an annealing inverse temperature β and occasional jitter.
* With the best vector from the gradient phase, perform a long
  simulated‑annealing hill‑climb that updates the full convolution in O(n)
  per move.  The source index is chosen proportionally to its contribution
  to the current worst convolution entry, while the destination is biased
  towards small entries.
* A final polishing hill‑climb uses any remaining budget.
* The result is projected onto the ℓ¹‑simplex (∑aᵢ = √(2 n)) and trailing
  near‑zero entries are trimmed before returning.

The implementation respects the 1000 s wall‑clock budget and prints
intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full self‑convolution
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max (log‑sum‑exp).
    Uses momentum, temperature annealing and occasional jitter.
    Stops when wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σa = √(2 n).  Several families (power‑law, exponential, Dirichlet,
    piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing hill climbing that transfers a random amount of
    mass from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling

    while time.time() < deadline:
        # pick an index where the convolution attains its current maximum
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – 30 % of the time bias towards the smallest entries
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update the full convolution efficiently
        b_new = b.copy()
        # linear terms
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b
