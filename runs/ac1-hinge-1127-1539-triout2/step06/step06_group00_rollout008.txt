def local_refinement(a, best_score, deadline, rng):
    while time.time() < deadline:
        # choose random move
        i, j = rng.integers(0, len(a), size=2)
        if i == j: continue
        # epsilon proportion
        eps = rng.uniform(-0.05, 0.05) * a[i]
        if eps == 0: continue
        # propose move
        # Ensure not negative
        if a[i] - eps < 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # Scale not needed because sum unchanged.
        # Evaluate
        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            a = a_new
    return a, best_score

#!/usr/bin/env python
"""
adversarial step‑function search – diversified random shapes + Adam + hill‑climb.

The algorithm works in three phases:

1️⃣  Random exploration of parametrised “edge‑heavy” shapes (power‑law,
    exponential, and a mixture of two power‑laws) for many different
    lengths.  Each candidate is scaled to the required ℓ₁‑norm
    √(2·n) and evaluated by computing the peak of its self‑convolution.

2️⃣  The best candidate found so far is refined with a projected Adam
    optimiser on the simplex (log‑softmax parametrisation).  The Adam
    run runs until the remaining time budget expires.

3️⃣  A cheap gradient‑free hill‑climbing stage makes small mass‑transfer
    moves between coordinates to squeeze out any remaining improvement.

The whole search respects the 1000‑second budget, stores the best
construction in the global variable ``height_sequence_1`` and returns it
as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector using an FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _score(a: np.ndarray) -> float:
    """Peak of the self‑convolution (the quantity minimised)."""
    return float(_full_convolution_fft(a).max())


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence to length *new_n* by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parametric shape families (all return vectors with ℓ₁‑norm √(2·n))
# ----------------------------------------------------------------------
def _shape_power(n: int, p: float, offset: float, mix: float) -> np.ndarray:
    """
    Edge‑heavy power‑law shape.

    a_i ∝ mix·(d_i+offset)^{-p} + (1‑mix)·1,
    where d_i is the distance to the nearest boundary (1‑based).
    The result is scaled to the required ℓ₁‑norm.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    edge = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    raw = raw / raw.sum() * S
    return raw


def _shape_exp(n: int, lam: float, mix: float) -> np.ndarray:
    """
    Exponential decay shape.

    a_i ∝ mix·exp(-lam·(d_i‑1)) + (1‑mix)·1,
    with the same distance definition as above.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    edge = np.exp(-lam * (d - 1.0))
    base = np.ones(n)
    raw = mix * edge + (1.0 - mix) * base
    raw = raw / raw.sum() * S
    return raw


def _shape_two_power(n: int,
                     p1: float, offset1: float, w1: float,
                     p2: float, offset2: float, w2: float) -> np.ndarray:
    """
    Mixture of two power‑law components plus a uniform background.
    The three weights satisfy w1 + w2 ≤ 1.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0
    edge1 = (d + offset1) ** (-p1)
    edge2 = (d + offset2) ** (-p2)
    base = np.ones(n)
    raw = w1 * edge1 + w2 * edge2 + (1.0 - w1 - w2) * base
    raw = raw / raw.sum() * S
    return raw


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the τ‑soft‑max approximation of max(b) where b = a*a.
    Returns (grad, max_b).  The gradient is ∂/∂a τ‑soft‑max(b).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:                     # exact sub‑gradient
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)          # soft‑max weights
        mask = w

    # grad = 2 * (mask * a) convolved with reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex (log‑softmax) to minimise the
    τ‑soft‑max approximation of the convolution peak.

    Returns the best hard‑peak vector encountered and its peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # exact ℓ₁‑norm required

    # initialise logits so that softmax(p)·S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3

    # start temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient (soft‑max)
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9997
        lr *= 0.9999

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Generation of a feasible start vector (optionally warm‑starting)
# ----------------------------------------------------------------------
def _initial_vector(n: int,
                    rng: np.random.Generator,
                    base_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible vector of length *n*.
    With probability 0.15 we warm‑start from *base_seq* (linearly
    interpolated), otherwise a random parametric shape is returned.
    """
    S = np.sqrt(2.0 * n)

    # Warm‑start from the global best (if available)
    if base_seq is not None and rng.random() < 0.15:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    # Random shape choice
    shape = rng.choice(['power', 'exp', 'twopower'])
    if shape == 'power':
        p = rng.uniform(0.05, 3.0)
        offset = rng.uniform(0.0, 10.0)
        mix = rng.uniform(0.0, 1.0)
        a = _shape_power(n, p, offset, mix)
    elif shape == 'exp':
        lam = rng.uniform(0.0, 5.0)
        mix = rng.uniform(0.0, 1.0)
        a = _shape_exp(n, lam, mix)
    else:  # two‑power mixture
        p1 = rng.uniform(0.05, 3.0)
        off1 = rng.uniform(0.0, 10.0)
        p2 = rng.uniform(0.05, 3.0)
        off2 = rng.uniform(0.0, 10.0)
        w1 = rng.random()
        w2 = rng.random() * (1.0 - w1)    # enforce w1 + w2 ≤ 1
        a = _shape_two_power(n, p1, off1, w1, p2, off2, w2)

    return a


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                current_score: float,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy coordinate‑wise hill‑climbing: move a random small amount of
    mass from one index to another if it reduces the convolution peak.
    """
    n = a.shape[0]
    a_cur = a.copy()
    best = current_score

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # amount to move: up to 5 % of a[i]
        eps = rng.uniform(0.0, 0.05) * a_cur[i]
        if eps <= 0.0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # sum unchanged, non‑negativity preserved by construction
        new_score = _score(a_new)
        if new_score < best:
            best = new_score
            a_cur = a_new
    return a_cur, best


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function (equivalently the peak of its self‑convolution).

    The method consists of
      * many short random shape trials,
      * a long Adam refinement on the best candidate,
      * a final cheap hill‑climb.
    The result is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    end_time = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve previously found construction (if any) for warm‑starts
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    it = 0

    # ------------------------------------------------------------------
    # Phase 1 – random exploration (≈70 % of the time budget)
    # ------------------------------------------------------------------
    while time.time() + 4.0 < end_time:          # keep a few seconds for later phases
        it += 1
        # Choose a length – mix moderate and larger sizes
        if rng.random() < 0.7:
            n = rng.integers(1500, 20000)      # medium sizes
        else:
            n = rng.integers(20000, 60000)     # larger sizes

        a = _initial_vector(n, rng, base_seq)
        sc = _score(a)

        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = n
            print(f"[search] it={it:4d} n={n:5d} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        # make sure we honour the exact ℓ₁ constraint (important after rounding)
        S = np.sqrt(2.0 * len(best_seq))
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum() * S

        # run Adam until the remaining deadline
        a_refined, peak = _adam_optimize(best_seq, end_time, rng)
        if peak < best_score:
            best_score = peak
            best_seq = a_refined.copy()
            print(f"[adam]   refined peak={peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap hill‑climbing (use any time that is left)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining_deadline = end_time
        a_hc, hc_score = _hill_climb(best_seq, best_score, remaining_deadline, rng)
        if hc_score < best_score:
            best_score = hc_score
            best_seq = a_hc.copy()
            print(f"[hill]   hill‑climb score={hc_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the supplied black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
        external = evaluate_sequence(final_seq)
        print("[final]  fallback sequence, external score =", external)
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final]  external score={external:.6f}  internal max={best_score:.6f}  length={len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
