"""
Multi‑restart optimisation for the self‑convolution peak problem.

Strategy
--------
1. **Exploration** – many short attempts with random initial shapes
   (uniform, geometric, reciprocal, triangular, cosine and up‑sampled
   best).  All candidates are projected onto the simplex
   Σa = sqrt(2·n) and forced to be symmetric (a[i] = a[-1-i]) which
   halves the effective dimensionality.

2. **Combined optimiser** – for each attempt we run three phases
   * a soft‑max Adam optimiser (log‑sum‑exp of the convolution) that
     follows the true peak increasingly,
   * a sub‑gradient descent directly on the worst convolution entry,
   * a cheap random “mass‑swap’’ hill‑climb.

   Each phase respects the simplex constraint via Euclidean projection.

3. **Refinement** – after the main budget we up‑sample the current best
   sequence to a larger length (≈1.5×) and run the optimiser again for
   the remaining time.

All heavy work is performed with FFT‑based convolutions (O(n log n)).
The code stays within the 1000 s budget (a 5 s safety margin) and
returns the best sequence it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a * a (length 2·n−1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """2·∂/∂a ( (1/α)·log‑sum‑exp(α·(a*a)) ) for the given soft‑max weights."""
    n = a.size
    w_rev = w[::-1]                     # reverse for convolution
    conv_len = n + (2 * n - 1) - 1      # = 3·n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)      # pick the slice that yields ∂c_k/∂a_i = 2·a_{k‑i}
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Log‑sum‑exp soft‑max weights
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities proportional to exp(alpha·conv)."""
    scaled = alpha * conv
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation (exact mirror symmetry)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Upsample / linear interpolation
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Rich family of random initialisers
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    The generator mixes several analytic shapes to provide diversity.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 7)   # 0–6

    # 0 – up‑sample a known good sequence
    if typ == 0 and base_seq is not None and base_seq.size > 8:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    # 1 – pure uniform Dirichlet
    elif typ == 1:
        a = rng.random(n)

    # 2 – geometric decay from both ends
    elif typ == 2:
        rate = rng.uniform(0.90, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) / 2.0

    # 3 – edge‑heavy reciprocal shape
    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    # 4 – triangular (peak in centre)
    elif typ == 4:
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    # 5 – cosine bump (zero at edges, max at centre)
    elif typ == 5:
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** rng.uniform(0.5, 2.0)

    # 6 – random convex combination of a few bases
    else:
        cand = []
        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.90, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)          # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                       # reciprocal
        centre = (n - 1) / 2.0
        tri = 1.0 - np.abs(np.arange(n) - centre) / centre
        cand.append(tri)                                 # triangle
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalise to required ℓ¹‑norm (projection later will enforce exact sum)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.size
    S = np.sqrt(2.0 * n)

    # hyper‑parameters
    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for soft‑max
    alpha = 0.3
    alpha_mul = 1.025

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape plateaus
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.15:
                    a = cand

        # schedule updates
        alpha = min(alpha * alpha_mul, 48.0)
        lr *= 0.99993

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max entry + occasional mass‑swap
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.size
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.13
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g

        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random mass‑swap (simulated‑annealing flavour)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # occasional early stop if progress stalls
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap hill‑climbing that swaps tiny masses between random positions
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.size
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → Sub‑grad → Local)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = True) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # allocate ~60 % / 30 % / 10 % of the slice to the three phases
    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # pick the best among the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function whose self‑convolution peak
    is as small as possible.  The routine is time‑aware: it repeatedly
    restarts with different lengths, keeps the globally best sequence,
    and performs a final up‑sampling refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for external call

    best_seq = None
    best_score = np.inf
    best_n = None
    base_seq = None
    attempts = 0
    last_improve = start

    # --------------------------------------------------------------
    # Main timed loop – many short restarts
    # --------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Choose a length.
        if best_seq is not None and rng.random() < 0.55:
            # Perturb the current best length modestly.
            scale = rng.uniform(0.9, 1.1)
            n = max(500, int(best_n * scale))
        else:
            n = int(rng.integers(1500, 25000))

        enforce_sym = True                     # keep symmetry for all attempts

        a0 = _initial_sequence(base_seq, n, rng)
        a0 = _symmetrize(a0) if enforce_sym else a0

        # allocate a time slice (shorter for very large n)
        slice_len = rng.uniform(9.0, 15.0) if n < 10000 else rng.uniform(7.0, 12.0)
        slice_deadline = min(time.time() + slice_len, deadline)

        a_opt, sc = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            best_n = n
            base_seq = best_seq.copy()
            last_improve = time.time()
            print(f"[global] att={attempts} n={n} best={best_score:.6f}")

        # If we haven't improved for a while, drop the base_seq to regain diversity.
        if time.time() - last_improve > 45.0:
            base_seq = None

        # Early exit if we have already crossed a promising target.
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # Final up‑sampling refinement (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        larger_n = min(30000, int(best_n * 1.5))
        a_up = _upsample_seq(best_seq, larger_n)
        a_up = _symmetrize(a_up)
        a_refined, sc_ref = _optimize_combined(a_up, deadline, rng, enforce_sym=True)
        if sc_ref < best_score:
            best_seq, best_score = a_refined, sc_ref
            best_n = larger_n
            print(f"[refine] upsampled n={larger_n} new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found
    # --------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    # sanity check via the provided black‑box evaluator
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal_best={best_score:.6f} "
          f"len={len(final_seq)} attempts={attempts}")

    # expose the best sequence for possible future up‑sampling
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
