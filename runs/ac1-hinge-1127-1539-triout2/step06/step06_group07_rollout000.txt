   steady‑state GA tries to improve the solution further.
4. Return the best vector found (converted to a plain list).  

All vectors are kept on the L¹‑simplex Σaᵢ = √(2·n) so that the external
``evaluate_sequence`` score equals the convolution peak.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of the max‑norm.
    For a (len n) and weight vector w (len 2·n‑1):
        g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max: w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)."""
    scaled = alpha * conv
    m = np.max(scaled)
    exp_shifted = np.exp(scaled - m)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Adam optimiser on smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max approximation of the peak of a*a."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters (tuned for this problem)
    lr = 0.8
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # temperature schedule
    alpha = 0.5
    alpha_max = 2000.0

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass‑swap to escape plateaus
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase sharpness
        alpha = min(alpha * 1.07, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (true peak)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent directly on the worst convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the worst entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional perturbation
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.025
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.6) -> tuple[np.ndarray, float]:
    """Run Adam for a fraction of the allocated time, then sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Simple steady‑state Genetic Algorithm
# ----------------------------------------------------------------------


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Light GA working on the simplex; returns the best vector found."""
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 16 if n <= 2000 else 10
    # initialise population
    pop = []
    scores = []
    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # select two distinct parents different from individual i
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)

            # convex recombination
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            # mutation
            if rng.random() < 0.35:
                noise = rng.normal(scale=child.mean() * 0.06 + 1e-12, size=n)
                child += noise

            # enforce constraints
            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # occasional diversity injection
        if rng.random() < 0.12:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # break early if time is almost up
        if time.time() + 0.1 >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine consists of a parametric seeding phase, many short
    local optimiser restarts on n = 4000, and a final brief GA.
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the external call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    S = np.sqrt(2.0 * n)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1️⃣  Parametric seeding (cheap, exhaustive)
    # ------------------------------------------------------------------
    print("[phase] parametric seeding")
    i = np.arange(n, dtype=float)

    # sin^p family (positive on whole interval)
    p_vals = np.linspace(0.2, 5.0, 30)
    for p in p_vals:
        a = np.sin(np.pi * (i + 0.5) / n) ** p
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a = a / a.sum() * S
        sc = np.max(_full_convolution(a))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  [seed] sin^p p={p:.3f} → {sc:.6f}")

    # exponential decay family
    lam_vals = np.linspace(0.001, 0.020, 25)
    center = (n - 1) / 2.0
    for lam in lam_vals:
        a = np.exp(-lam * np.abs(i - center))
        a = np.maximum(a, 0.0)
        a = a / a.sum() * S
        sc = np.max(_full_convolution(a))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  [seed] exp lam={lam:.4f} → {sc:.6f}")

    # a few random starts just in case
    for _ in range(4):
        a = rng.random(n)
        a = _project_simplex(a, S)
        sc = np.max(_full_convolution(a))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  [seed] random → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Multi‑restart local optimisation (Adam → Sub‑grad)
    # ------------------------------------------------------------------
    print("[phase] local optimisation restarts")
    while time.time() < deadline - 2.0:
        time_left = deadline - time.time()
        # allocate at most 30 s per restart (or the remaining 60 % of time)
        restart_time = min(30.0, time_left * 0.6)
        restart_deadline = time.time() + restart_time

        # initialise start point: usually a jittered version of the current best
        if best_seq is not None and rng.random() < 0.75:
            a0 = best_seq.copy()
            jitter = rng.normal(scale=a0.mean() * 0.03 + 1e-12, size=n)
            a0 = np.maximum(a0 + jitter, 0.0)
            a0 = _project_simplex(a0, S)
        else:
            a0 = rng.random(n)
            a0 = _project_simplex(a0, S)

        a_opt, sc_opt = _optimize_combined_adam(a0, restart_deadline,
                                                rng,
                                                fraction_adam=0.6)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  [restart] new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Brief GA refinement after each restart (if time permits)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 6.0:
            ga_deadline = time.time() + min(5.0, remaining)
            a_ga, sc_ga = _run_genetic_optimizer(a_opt, ga_deadline, rng)
            if sc_ga < best_score:
                best_score = sc_ga
                best_seq = a_ga.copy()
                print(f"  [ga] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final sanity check & external evaluation
    # ------------------------------------------------------------------

        final_seq = [1.0]
    else:

    print(f"[final] external_score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
