"""
A fast heuristic for reducing the max‐entry of a self‑convolution.

The algorithm follows the successful “gradient + hill‑climb” pattern but
adds a few new ideas that help push the objective a bit further:

* **Symmetrisation** – after each gradient step the vector is mirrored and
  averaged (a = (a + a[::-1])/2).  The optimum of the max‑convolution problem
  is symmetric, and enforcing this reduces the dominant central peak.
* **Longer final length** – we upscale the previous best construction
  (``height_sequence_1``) to a larger size (24 000) and devote the bulk of the
  time budget there.  More degrees of freedom usually allow a lower maximum.
* **More aggressive step size** – the base learning‑rate is increased
  (0.2·‖a‖₁) and the β‑annealing schedule is kept aggressive.
* **Repeated alternating phases** – while time permits we run short
  gradient‑descent phases (≈30 s) followed by targeted hill‑climbing phases.
  The best vector found so far is always carried forward.
* **Random jitter restarts** – occasionally a small Gaussian jitter is added
  to the current best vector; this helps escape plateaus.
* **Final polishing** – after the time‑budget is almost exhausted a final
  hill‑climbing run is performed.

All steps respect the ℓ¹‑simplex constraint Σa = √(2 n) (the scaling that
makes ``evaluate_sequence`` equal to the max‑convolution value) and the
non‑negativity constraints required by the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT version)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform
        a = np.full(n, target / n)

    else:                                 # Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a via FFT
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt_sym(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max with explicit symmetrisation.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # annealing schedule for β
    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    # step‑size base (a bit larger than the original implementation)
    step_base = target_sum * 0.20
    momentum = np.zeros_like(a)
    rho = 0.92                       # Nesterov‑style momentum decay

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # enforce non‑negativity and symmetry
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Targeted hill‑climbing (as in the reference code)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that reduces the current maximal convolution entry.
    Returns the best sequence found and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # initial temperature
    cooling = 0.9995

    while time.time() < deadline:
        # index of the current peak
        k_max = int(np.argmax(b))

        # contributions of each index to that peak
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # analytic convolution update (same as reference)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the time budget (default 1000 s) and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Initialise from the previously found construction if available.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")
    else:
        # no seed – start with a modest random vector
        n0 = 2000
        a0 = _random_initial_sequence(n0, rng)
        best_seq = a0.copy()
        best_score = np.inf

    # ------------------------------------------------------------------
    # Upsample to a larger target length (more freedom to improve).
    # ------------------------------------------------------------------
    target_n = 24000
    a = _upsample_sequence(best_seq, target_n)
    target_sum = np.sqrt(2.0 * target_n)
    a = _project_to_simplex(a, target_sum)
    best_seq = a.copy()
    best_score = _max_convolution_fft(a)
    print(f"[{time.time():.1f}] Upsampled to n={target_n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop: repeated short GD + hill‑climbing phases.
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        time_left = deadline - time.time()
        # allocate at most 30 s for a GD phase
        phase_gd = min(30.0, time_left * 0.4)
        a_gd = _gradient_descent_opt_sym(a, time.time() + phase_gd)

        b_gd = _full_convolution(a_gd)
        max_gd = b_gd.max()
        if max_gd < best_score:
            best_score = max_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement, max={max_gd:.6f}")

        # allocate remaining budget (but leave a small margin for final polishing)
        if time.time() >= deadline - 5.0:
            break
        phase_hc = min(30.0, deadline - time.time() - 5.0)
        a_hc, max_hc = _hill_climb_targeted(best_seq, b_gd, time.time() + phase_hc)

        # symmetrise and re‑project before evaluating the new max
        a_hc = (a_hc + a_hc[::-1]) * 0.5
        a_hc = _project_to_simplex(a_hc, target_sum)
        b_hc = _full_convolution(a_hc)
        max_hc = b_hc.max()
        if max_hc < best_score:
            best_score = max_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb improvement, max={max_hc:.6f}")

        # prepare for the next outer iteration
        a = best_seq.copy()

        # occasional random jitter to avoid stagnation
        if rng.random() < 0.15:
            jitter = rng.normal(scale=target_sum * 0.001, size=target_n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a = _project_to_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Final polishing (as much time as remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_final, deadline)
        a_final = (a_final + a_final[::-1]) * 0.5
        a_final = _project_to_simplex(a_final, target_sum)
        b_final = _full_convolution(a_final)
        max_final = b_final.max()
        if max_final < best_score:
            best_score = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement, max={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries (clean output) and report final value.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
