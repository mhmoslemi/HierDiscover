"""
Optimise a non‑negative step function for the objective
    2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Initialisation** – reuse the best sequence found so far
   (`height_sequence_1`) or build a fresh edge‑heavy candidate.
2. **Iterated linear‑programming improvement** – given the current
   vector `f` with L1‑norm S=√(2 n) we solve the LP
       maximise Σ g_i  s.t.   f * g ≤ M   (component‑wise)
   where M = max(f*f).  The solution g₀ typically has Σg₀ > S;
   after scaling g = (S/Σg₀)·g₀ we mix f←(1−t)f+t g.
   The mix step provably reduces the convolution peak while
   preserving the norm.  The LP is solved with SciPy’s HiGHS
   optimiser; the full constraint matrix (2 n‑1 rows, n columns)
   is built as a sparse Toeplitz matrix.
3. **Final Adam refinement** – a short projected‐Adam run on the
   simplex (log‑softmax parametrisation) fine‑tunes the vector.

The routine respects the 1000 s time budget, prints progress
and stores the best sequence in the global variable
`height_sequence_1` for reuse by future runs.
"""

import time

# -------------------------------------------------------------
# Linear‑algebra utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution via real FFT (full length)."""
    n = a.shape[0]
    total = 2 * n - 1
    f_len = _next_pow2(total)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total]
    return conv


def _max_convolution(seq: np.ndarray):
    """Return (max value, full convolution array)."""
    conv = _full_convolution_fft(seq)
    return conv.max(), conv


# -------------------------------------------------------------
# Parameterised “edge‑heavy’’ starter
# -------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a feasible start vector of length n.
    Shape: (d+offset)^(-p) mixed with a uniform background.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0     # distance to nearest edge (1‑based)

    p = rng.uniform(0.2, 1.0)                 # power‑law exponent
    offset = rng.uniform(0.0, 5.0)            # denominator shift
    mix = rng.uniform(0.0, 1.0)               # mixing proportion

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


# -------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max of convolution peak)
# -------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Gradient of the τ‑soft‑max approximation of max(conv(a,a))."""
    b = _full_convolution_fft(a)
    max_b = b.max()
    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Adam on the simplex to reduce the hard convolution peak."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # required ℓ₁‑norm

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3

    # initial temperature (soft‑max smoothing)
    _, b0 = _softmax_gradient(a0, 1.0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # --- recover a from logits (log‑softmax) ---
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # --- convolution & gradient (soft‑max) ---
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # --- gradient w.r.t. logits ---
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9997
        lr *= 0.9999

        # occasional jitter
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak

def _build_constraint_matrix(f: np.ndarray):
    """Build the sparse matrix A (2n‑1 × n) for constraints f * g ≤ M."""
    n = f.shape[0]
    # rows = i + j  (i = index in f, j = column index)
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = np.repeat(i, n) + np.tile(j, n)          # shape n²
    cols = np.tile(j, n)                           # shape n²
    data = np.repeat(f, n)                         # shape n²
    from scipy import sparse
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_improve(f: np.ndarray,
                M: float,
                S: float,
                t: float,
                rng: np.random.Generator):
    """
    One LP‑based improvement step.
    Returns (new_f, improved_flag, new_M).
    """
    n = f.shape[0]

    # Build the full constraint matrix (sparse)
    A = _build_constraint_matrix(f)

    # Linear program: maximise Σ g_i  ⇔ minimise -Σ g_i
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    b_ub = np.full(2 * n - 1, M)

    from scipy.optimize import linprog
    res = linprog(c_obj,
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return f, False, M

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= S * (1.0 + 1e-12):
        # No gain in L1‑norm → cannot improve the bound
        return f, False, M

    # Scale g down so that its L1‑norm is exactly S (scale < 1)
    scale = S / sum_g0
    g = g0 * scale

    # Convex combination
    new_f = (1.0 - t) * f + t * g
    new_f = np.maximum(new_f, 0.0)
    # Renormalise (numerical safety)
    new_f = new_f / new_f.sum() * S

    new_M, _ = _max_convolution(new_f)
    improved = new_M < M - 1e-8
    return new_f, improved, new_M


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine proceeds with:
      * reuse or creation of an initial feasible vector,
      * a loop of LP‑based improvement steps,
      * a final Adam refinement.
    The best sequence is stored in the global ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Load a previously found construction if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    # ------------------------------------------------------------------
    # If nothing to reuse, create a fresh edge‑heavy candidate
    # ------------------------------------------------------------------
    if base_seq is None:
        n0 = rng.integers(1500, 2500)
        base_seq = _edge_initial(n0, rng)

    n = len(base_seq)
    S = np.sqrt(2.0 * n)

    # Normalise to the required ℓ₁‑norm
    base_seq = np.clip(base_seq, 0.0, None)
    if base_seq.sum() == 0.0:
        base_seq = np.full(n, S / n)
    else:
        base_seq = base_seq / base_seq.sum() * S

    best_seq = base_seq.copy()
    best_M, _ = _max_convolution(best_seq)
    print(f"[init] length={n}, initial max_b={best_M:.6f}")

    # ------------------------------------------------------------------
    # LP‑based improvement loop
    # ------------------------------------------------------------------
    t = 0.5                  # initial mixing weight
    min_t = 0.02
    iteration = 0
    while time.time() < end_time - 5.0:
        iteration += 1
        # one improvement step
        new_seq, improved, new_M = _lp_improve(best_seq, best_M, S, t, rng)

        if improved:
            best_seq = new_seq
            best_M = new_M
            print(f"[LP  {iteration:3d}] improved max_b={best_M:.6f} (t={t:.3f})")
            # slowly reduce the step size to fine‑tune later
            t = max(t * 0.92, min_t)
        else:
            # if no progress, shrink t and try again; give up after many failures
            t = max(t * 0.85, min_t)
            print(f"[LP  {iteration:3d}] no improvement, reducing t to {t:.3f}")
            if t <= min_t:
                break

    # ------------------------------------------------------------------
    # Final Adam refinement (optional – uses remaining time)
    # ------------------------------------------------------------------
    remaining = end_time - time.time()
    if remaining > 5.0:
        print(f"[final] starting Adam refinement, remaining time≈{remaining:.1f}s")
        refined_seq, refined_peak = _adam_optimize(best_seq,
                                                   time.time() + remaining,
                                                   rng)
        if refined_peak < best_M:
            best_seq = refined_seq
            best_M = refined_peak
            print(f"[final] Adam refinement improved max_b={best_M:.6f}")

    # ------------------------------------------------------------------
    # Sanity check via the external evaluator and store the result
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external score = {external:.6f}  internal max = {best_M:.6f}  length = {len(final_seq)}")
    globals()['height_sequence_1'] = final_seq
    return final_seq
