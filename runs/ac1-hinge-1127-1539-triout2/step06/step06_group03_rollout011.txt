def subgradient_optimize(a0, deadline):
    a = a0.copy()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)  # ensure sum

    iteration = 0
    rng = np.random.default_rng()
    best_obj = float('inf')
    best_a = a.copy()
    while time.time() < deadline:
        # compute convolution b and max
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2 * n - 1]

        max_b = b.max()
        # compute current objective
        cur_obj = max_b  # because sum= sqrt(2n)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{time.time():.1f}] New best subgrad n={n} obj={best_obj:.6f}")

        # gradient
        ks = np.where(b == max_b)[0]
        # compute gradient via loops (if small ks) for speed perhaps
        g = np.zeros_like(a)
        for k in ks:
            # compute contributions
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            # i indices start...end inclusive
            i_len = end - start + 1
            # array of indices i
            i_arr = np.arange(start, end + 1, dtype=int)
            j_arr = k - i_arr
            # g[i] += 2 * a[j]
            g[i_arr] += 2.0 * a[j_arr]
        # After loops, we have gradient.

        # step size schedule
        # we can compute gradient norm
        norm_g = np.linalg.norm(g)
        if norm_g == 0:
            break

        # step size = gamma / sqrt(iteration+1) maybe gamma = target_sum / (norm_g * 10)
        gamma = target_sum / (norm_g * 10.0)
        step = gamma / np.sqrt(iteration + 1)
        a_new = a - step * g

        # enforce nonnegative
        a_new = np.maximum(a_new, 0.0)

        # project onto simplex
        a_new = _project_to_simplex(a_new, target_sum)

            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        iteration += 1

    return best_a

        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2 * n - 1]
        max_b = b.max()
        # update best
        if max_b < best_obj:
            best_obj = max_b
            best_a = a.copy()

        # compute gradient via loops
        ks = np.where(b == max_b)[0]
        # do gradient compute using loops
        # We'll accumulate in g array
        g = np.zeros_like(a)
        for k in ks:

            end = min(n - 1, k)
            # i indices from start to end

            i_range = np.arange(start, end + 1)
            j_range = k - i_range

            g[i_range] += 2.0 * a[j_range]

        norm_g = np.linalg.norm(g)

        eta = step_factor / np.sqrt(iteration + 1)

"""
Improved search for a low‑value step function.

The algorithm consists of three stages:

1. **Initialisation** – start from the best known construction
   (`height_sequence_1` if available, otherwise a power‑law sequence)
   and project it onto the ℓ¹‑simplex Σa = √(2·n).

2. **Sub‑gradient descent** – minimise the true objective
   `max(a*a)` (the scaling cancels out).  At each iteration the exact
   sub‑gradient of the maximum convolution entry is computed (using
   simple slice arithmetic) and a diminishing step is taken, followed
   by a projection onto the simplex.  A small stochastic jitter is
   injected periodically to escape plateaus.

3. **Permutation annealing** – with the multiset of values fixed, the
   ordering is optimised by simulated‑annealing swaps (the classic
   “swap‑move” neighbourhood).  The remaining time budget is spent here.

Both phases keep a running record of the best sequence seen.  The final
sequence is trimmed of negligible trailing entries and returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the ℓ¹‑simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Duchi et al. (2008) projection onto {x ≥ 0, Σx = s}."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast maximum of the full linear convolution a*a (FFT)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the full convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform projected sub‑gradient descent on   max(a*a) .
    The ℓ¹‑norm is kept fixed to √(2·n); this makes the objective
    equal to the maximum convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # initialise feasibility
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    # initial objective
    best_obj = _max_convolution_fft(best_a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # ---- convolution & maximum ----
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_b = float(b.max())

        # keep best seen
        if max_b < best_obj:
            best_obj = max_b
            best_a = a.copy()
            print(f"[{time.time():.1f}] Sub‑grad new best obj={best_obj:.6f}  iter={it}")

        # ---- sub‑gradient of the max entry ----
        # treat all (numerically) active indices as active
        tol = 1e-12
        ks = np.where(b >= max_b - tol)[0]

        g = np.zeros_like(a)
        for k in ks:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            i_range = np.arange(start, end + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # ---- step length (diminishing) ----
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        # step factor chosen so that updates are moderate
        step_factor = 0.5 * target_sum / (norm_g + 1e-12)
        eta = step_factor / np.sqrt(it + 1.0)

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, target_sum)

        a = a_new

        # occasional stochastic jitter to shake the process
        if it % 200 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 7e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Permutation‑only simulated annealing (swap moves)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Optimise the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm (and hence the objective) stays unchanged.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # tiny “global” reshuffle with low probability
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Simple parametric families (fallback if no global seed is present)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^{-p}, normalised to √(2n)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _project_to_simplex(raw, target)

# Main entry point
# ----------------------------------------------------------------------
