def _lp_optimize(f, S):
    n = len(f)
    # build constraint matrix for conv f * g <= t
    rows = []
    cols = []
    data = []
    for i in range(2*n-1):
        # range of j valid: max(0, i-(n-1)) to min(i, n-1)
        j_start = max(0, i-(n-1))
        j_end = min(i, n-1)
        for j in range(j_start, j_end+1):
            coeff = f[i-j]   # f index (i-j)
            rows.append(i)
            cols.append(j)
            data.append(coeff)
        # add t variable:
        rows.append(i)
        cols.append(n)
        data.append(-1.0)
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)

    # equality constraint for sum g = S
    A_eq = np.zeros((1, n+1))
    A_eq[0,:n] = 1.0
    b_eq = np.array([S])

    c = np.zeros(n+1)
    c[n] = 1.0   # minimize t

    bounds = [(0, None)]*n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return original f maybe
        return f, None
    x = res.x
    g = x[:n]
    t_opt = x[n]
    return g, t_opt
```

"""
adversarial step‑function optimizer – hybrid Adam / linear‑programming + hill‑climb

The search works in three stages:

1. **Exploration** – many short Adam restarts with diverse
   initial shapes (edge‑heavy, Dirichlet, uniform).  The Adam
   optimiser works on the simplex via a log‑softmax (the τ‑soft‑max
   approximates the convolution peak).  The best candidate seen
   during this phase is kept.

2. **Refinement** – the current best vector is refined with a longer
   Adam run, followed by a small number of *linear‑programming*
   improvements.  For a fixed vector `f` we solve

        minimise   t
        subject to conv(f,g) ≤ t   (for all convolution positions)
                    sum(g) = √(2·n)
                    g ≥ 0

   which is a linear program (variables are the new heights `g` and
   the bound `t`).  The solution reduces the worst convolution value
   while preserving the ℓ₁‑norm.  After each LP step a short Adam run
   is performed to fine‑tune the result.

3. **Local hill‑climbing** – a cheap random‑swap neighbourhood is
   explored for the remaining time budget.  This often yields a
   final tiny improvement.

All phases respect the time budget (default 1000 s).  The final
sequence is stored in the global ``height_sequence_1`` variable and
returned as a plain ``list[float]``.
"""

import time
import numpy as np

# optional imports – linear programming is used only if SciPy is available
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_OK = True
except Exception:               # pragma: no cover
    _SCIPY_OK = False


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT (fast, accurate)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# τ‑soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a
    and b = a * a (full convolution).  For very small τ the exact
    sub‑gradient is used.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)      # exact sub‑gradient
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    # gradient w.r.t. a:
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   *,
                   lr_start: float = 0.3,
                   tau_decay: float = 0.9997,
                   lr_decay: float = 0.9999,
                   jitter_every: int = 800,
                   jitter_scale: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns (best_vector, best_hard_peak).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # initialise logits such that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr_start

    # initial temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & gradient (soft‑max) -----
        g, peak = _softmax_gradient(a, tau)

        # keep best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- gradient w.r.t. logits (chain rule) -----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional gradient clipping (helps stability for large n)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= tau_decay               # tighten soft‑max
        lr *= lr_decay                 # slowly decay learning rate

        # occasional jitter to escape plateaus
        if it % jitter_every == 0:
            p += rng.normal(scale=jitter_scale, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Edge‑heavy initial shapes (hand‑crafted family)
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a feasible start vector of length n.
    The shape is a power‑law decay from the edges possibly mixed with a uniform background.
    """
    S = np.sqrt(2.0 * n)

    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0            # distance to nearest edge, 1‑based

    p = rng.uniform(0.2, 1.0)          # exponent
    offset = rng.uniform(0.0, 5.0)     # shift
    mix = rng.uniform(0.0, 1.0)        # blend with uniform

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base

    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


# ----------------------------------------------------------------------
# Linear‑programming refinement (if SciPy is available)
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Solve the LP:
        minimise   t
        s.t.       conv(f, g) ≤ t
                   sum(g) = √(2·n)
                   g ≥ 0
    Returns (g_opt, t_opt).  If the LP fails or SciPy is unavailable,
    the original vector and its current max‑convolution are returned.
    """
    if not _SCIPY_OK:
        # fallback – return unchanged
        return f, _full_convolution_fft(f).max()

    n = len(f)
    S = np.sqrt(2.0 * n)

    # Build sparse constraint matrix A_ub * [g; t] ≤ 0
    rows = []
    cols = []
    data = []

    # For each convolution index i = 0 .. 2n‑2
    for i in range(2 * n - 1):
        j_start = max(0, i - (n - 1))
        j_end = min(i, n - 1)
        # coefficients for g_j
        for j in range(j_start, j_end + 1):
            coeff = f[i - j]    # f_{i-j}
            rows.append(i)
            cols.append(j)
            data.append(coeff)
        # coefficient for the scalar t (always -1)
        rows.append(i)
        cols.append(n)          # the last variable is t
        data.append(-1.0)

    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    # Equality constraint: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Objective: minimise t
    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(0, None)]   # g >=0, t >=0

    # Solve
    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  A_eq=A_eq,
                  b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30.0})
    if not res.success or res.x is None:
        # Infeasible / timeout – keep original
        return f, _full_convolution_fft(f).max()

    x = res.x
    g = x[:n]
    t_opt = x[n]

    # Numerical cleanup – ensure non‑negative and exact norm
    g = np.maximum(g, 0.0)
    if g.sum() == 0:
        g = f.copy()
    else:
        g = g / g.sum() * S

    return g, t_opt


# ----------------------------------------------------------------------
# Small random hill‑climbing – cheap local improvement
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                max_steps: int = 2000) -> tuple[np.ndarray, float]:
    """
    Very cheap neighbourhood search: repeatedly transfer a small amount
    of mass between two random positions while preserving the ℓ₁‑norm.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best = a.copy()
    best_val = _full_convolution_fft(best).max()

    steps = 0
    while time.time() < deadline and steps < max_steps:
        steps += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a transfer delta (relative to current magnitude)
        scale = 0.05 * best[i]   # 5 % of the source height
        delta = rng.normal(loc=0.0, scale=scale)
        if best[i] + delta < 0 or best[j] - delta < 0:
            continue
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta

        # keep exact ℓ₁‑norm (tiny numerical drift may appear)
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum() * S

        val = _full_convolution_fft(cand).max()
        if val < best_val:
            best_val = val
            best = cand
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine runs three phases:
    (1) many short Adam restarts, (2) a long Adam run + LP refinement,
    (3) a final stochastic hill‑climbing pass.
    The best sequence found is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load a previously stored construction, if any
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:          # pragma: no cover
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0

    # ------------------------------------------------------------------
    # Helper to build an initial feasible vector for a given length
    # ------------------------------------------------------------------
    def _initial_vector(n):
        """Return a feasible start vector of length n."""
        # Occasionally warm‑start from the global best (scaled)
        if base_seq is not None and rng.random() < 0.12:
            a = _scale_sequence(base_seq, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * np.sqrt(2.0 * n)
            return a

        style = rng.choice(['uniform', 'dirichlet', 'edge'])
        S = np.sqrt(2.0 * n)

        if style == 'uniform':
            a = np.full(n, S / n)
        elif style == 'dirichlet':
            a = rng.random(n)
            a = a / a.sum() * S
        else:
            a = _edge_initial(n, rng)

        return a

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam restarts (exploration)
    # ------------------------------------------------------------------
    short_budget = 8.0   # seconds per restart (feel free to tune)
    while time.time() + short_budget < end_time - 30.0:
        attempts += 1
        # Choose a length – a bias towards moderate sizes (1500‑4000)
        if rng.random() < 0.70:
            n = rng.integers(1500, 4000)
        else:
            n = rng.integers(4000, 8000)

        a_start = _initial_vector(n)

        slice_deadline = min(time.time() + short_budget, end_time - 30.0)
        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng)
        score = peak                     # sum is normalised → score = max_b

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart] attempt={attempts} n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long refinement (Adam + LP)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        # allocate half of the remaining time to a longer Adam run
        adam_budget = max(5.0, remaining * 0.5)
        adam_deadline = time.time() + adam_budget
        a_refined, peak = _adam_optimize(best_seq, adam_deadline, rng,
                                         lr_start=0.35,
                                         tau_decay=0.9995,
                                         lr_decay=0.9997)
        if peak < best_score:
            best_score = peak
            best_seq = a_refined.copy()
            print(f"[refine‑adam] improved score={best_score:.6f}")

        # After the long Adam, perform a few LP‑Adam cycles
        cycles = 5
        for c in range(cycles):
            if time.time() >= end_time - 5.0:
                break
            # LP step
            g_lp, t_lp = _lp_refine(best_seq, rng)
            # Mix the LP solution with the current vector (α = 0.5)
            alpha = 0.5
            mixed = (1.0 - alpha) * best_seq + alpha * g_lp
            mixed = np.maximum(mixed, 0.0)
            mixed = mixed / mixed.sum() * np.sqrt(2.0 * len(mixed))

            # Short Adam on the mixed vector (≈5 s)
            short_dead = time.time() + min(5.0, end_time - time.time() - 5.0)
            mixed_opt, peak_mix = _adam_optimize(mixed, short_dead, rng,
                                                 lr_start=0.30,
                                                 tau_decay=0.9996,
                                                 lr_decay=0.9998)
            if peak_mix < best_score:
                best_score = peak_mix
                best_seq = mixed_opt.copy()
                print(f"[lp‑adam] cycle={c+1} new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill‑climbing pass
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < end_time - 2.0:
        hill_deadline = end_time - 2.0
        best_seq, hill_score = _hill_climb(best_seq, hill_deadline, rng, max_steps=5000)
        if hill_score < best_score:
            best_score = hill_score
            print(f"[hill‑climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        # External evaluation (includes the security clamping)
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
