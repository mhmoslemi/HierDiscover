def _run_descent_and_transfer(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Ensure sum constraint via simplex projection:
    a = project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_score = max_convolution_and_index(a)[0]

    # compute initial conv and gradient
    while time.time() < deadline:
        # compute conv and index
        max_val, k_max, conv = max_convolution_and_index(a)
        if max_val < best_score:

            best_a = a.copy()

        # use small fraction

"""
A fast, deterministic optimiser for the benchmark ``evaluate_sequence``.
The method works directly with the sub‑gradient of the *maximum* convolution
entry (instead of the soft‑max used previously) and keeps the vector on the
ℓ¹‑simplex (∑a_i = √(2·n), a_i ≥ 0).  The optimiser consists of two stages:

1. **Projected sub‑gradient descent** – at each iteration the index *k* that
   attains the current maximum autocorrelation is found, the exact gradient
   of that entry is computed, a step is taken in the direction that reduces
   the maximum and the result is projected back onto the simplex.

2. **Targeted mass‑transfer moves** – after each gradient step a small amount
   of mass is transferred from the coordinate with the largest gradient
   (largest contribution to the current maximum) to the coordinate with the
   smallest gradient.  The move is always accepted when it improves the
   objective and otherwise is rejected.

The optimiser is run on several random initialisations (including the
previously discovered ``height_sequence_1`` when available) and the best
sequence found before the deadline is returned.  All numbers are forced to
be non‑negative; the final list is trimmed of trailing near‑zeros before it
is handed back to the scorer.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of size power‑of‑2."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]
    return conv


def _max_conv_and_index(a: np.ndarray):
    """Return (max_value, argmax_index, full_convolution) for a*a."""
    conv = _full_convolution(a)
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _gradient_maxconv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the autocorrelation entry at lag *k*.
    The derivative w.r.t. a[i] is a[i+k] (if i+k<n) + a[i-k] (if i-k≥0).
    """
    n = a.shape[0]
    if k == 0:
        # derivative of sum_i a_i^2  -> 2*a
        return 2.0 * a
    grad = np.zeros_like(a)
    # contribution from a[i+k]
    grad[: n - k] += a[k:]
    # contribution from a[i‑k]
    grad[k:] += a[: n - k]
    return grad


def _initial_power_law(n: int, p: float) -> np.ndarray:
    """Power‑law (i+1)^(-p) normalised to √(2·n)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= np.sqrt(2.0 * n) / a.sum()
    return a


def _resize_linear(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to change length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Core optimiser for one length
# ----------------------------------------------------------------------


def _optimise_one_length(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent + deterministic mass transfers.
    Returns the best sequence seen and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val, _, _ = _max_conv_and_index(a)

    rng = np.random.default_rng()
    it = 0
    # learning‑rate schedule parameters
    base_lr = target_sum * 0.25
    decay = 0.9995

    while time.time() < deadline:
        max_val, k_max, _ = _max_conv_and_index(a)

        # keep the globally best sequence
        if max_val < best_val:
            best_val = max_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] it={it:6d} new best max={best_val:.6f}")

        # ---- sub‑gradient step (exact) ---------------------------------
        grad = _gradient_maxconv(a, k_max)

        # project gradient onto the hyperplane Σδ = 0
        grad -= grad.mean()

        norm = np.linalg.norm(grad)
        if norm > 0:
            lr = base_lr / (norm + 1e-12)
            a -= lr * grad
            # keep the sum exactly √(2·n)
            a = _project_to_simplex(a, target_sum)

        # ---- targeted mass transfer ------------------------------------
        # move a small fraction from the coordinate with the largest
        # gradient (most harmful) to the smallest (most helpful).
        donor = int(np.argmax(grad))
        receiver = int(np.argmin(grad))
        if donor != receiver:
            delta = a[donor] * 0.05  # 5 % of the donor's mass
            if delta > 0:
                a[donor] -= delta
                a[receiver] += delta
                # projection is unnecessary because the transfer preserves
                # the total sum; we only need to enforce non‑negativity.
                a = np.maximum(a, 0.0)

        # ---- occasional jitter to escape flat regions ------------------
        if it % 800 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # decay the learning rate a little
        base_lr *= decay
        it += 1

    return best_a, best_val

# Public entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence with a low benchmark score.
    The routine performs a few independent runs (different random seeds
    and different initial shapes) and returns the best sequence obtained
    before the deadline.
    """
    start = time.time()
    # safety margin – we stop a little before the strict deadline
    deadline = start + budget_s - 2.0

    # Choose a length that balances quality and speed.
    # Empirically n≈6000 gives good results in the allotted time.
    n = 6000

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Prepare a few distinct initialisations
    # ------------------------------------------------------------------
    inits = []

    # 1) warm‑start from the previous best if available
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        seq = _resize_linear(seq, n)
        inits.append(seq)

    # 2) a simple power‑law (exponent drawn uniformly)
    rng = np.random.default_rng()
    for _ in range(2):
        p = rng.uniform(0.45, 0.85)
        inits.append(_initial_power_law(n, p))

    # 3) a completely random normalised vector
    rand_vec = rng.random(n)
    rand_vec = rand_vec * (np.sqrt(2.0 * n) / rand_vec.sum())
    inits.append(rand_vec)

    # ------------------------------------------------------------------
    # Run the optimiser on each initialisation, partitioning the remaining
    # time roughly equally among them.
    # ------------------------------------------------------------------
    now = time.time()
    for idx, init in enumerate(inits):
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        # allocate a fair share (last run gets whatever is left)
        if idx < len(inits) - 1:
            run_deadline = now + remaining / (len(inits) - idx)
        else:
            run_deadline = deadline

        seq, score = _optimise_one_length(init, run_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] run {idx+1}/{len(inits)} improved best max={best_score:.6f}")

        now = time.time()

    # ------------------------------------------------------------------
    # Trim trailing negligible entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        return []

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # One final call to the supplied scorer (optional – useful for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
