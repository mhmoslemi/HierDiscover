def _lp_step(f, M, target_sum):
    n = f.shape[0]
    # Build sparse matrix A_ub
    rows = []
    cols = []
    data = []
    for i in range(2*n - 1):
        lo = max(0, i - (n - 1))
        hi = min(n - 1, i)
        # j from lo to hi
        # coefficient for g_j is f[i - j]
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            data.append(f[i - j])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None, False
    g = res.x
    # Ensure non-negative
    g = np.maximum(g, 0.0)
    return g, True

"""
Multi‑scale optimisation of a non‑negative step function for the benchmark
`evaluate_sequence`.

The algorithm proceeds in four increasing lengths (≈1500 → 3000 → 6000 → 12000)
and combines several complementary optimisation phases:

1. **Initialisation** – use a power‑law shape or the globally supplied
   `height_sequence_1` and project it onto the ℓ¹‑simplex
   Σa_i = √(2·n).

2. **Projected gradient descent** – minimise a soft‑max approximation of the
   convolution maximum.  A momentum term, an increasing β‑parameter (up to
   8192) and occasional jitter provide robust descent.

3. **Linear‑programming “mass‑inflation” step** (only for n ≤ 3000) – maximise
   the total mass of a vector `g` under the linear constraints
   (f * g)_k ≤ M (where M = max(f*f)).  The optimiser then moves a fraction
   `t` towards `g` and re‑projects onto the simplex.  This step can strictly
   reduce the maximum convolution value.

4. **Simulated‑annealing refinement** – stochastic moves (mass transfer from
   indices that contribute most to the current maximal convolution entry,
   short‑window smoothing and random swaps) are accepted with a Metropolis
   rule.  Temperature and the allowed transfer fraction decay exponentially.

All updates maintain non‑negativity and the ℓ¹‑normalisation; because the
objective
`2·n·max(a*a) / (sum(a)²)` reduces to `max(a*a)` after normalisation, the
optimiser directly minimises the peak of the auto‑convolution.

The routine respects the supplied time budget (`budget_s` seconds) and
returns the best sequence found.
"""

import time

import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  When Σa = √(2·n) this equals max(a*a)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return the (i+1)^(-p) sequence of length n, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient‑descent on a soft‑max approximation of max(conv)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p  (only the central n entries are needed)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):

    Projected gradient descent on the soft‑max approximation.
    β grows up to 8192, momentum is used together with occasional jitter.
    """

    n = a.size
    target_sum = math.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 8192.0
    step_base = target_sum * 0.25
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.01,
                         max_window: int = 50) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing improvement of a sequence.
    Moves:
        – mass transfer from a “sensitive’’ index (large contribution to the
          current maximal convolution entry) to a low‑sensitivity index,
        – short‑window smoothing,
        – random swaps.
    Temperature and allowed transfer fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())

    # Pre‑compute convolution once, then refresh after each accepted move
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # location of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # sensitivity of each a_j to the entry at k_max (∂conv_kmax/∂a_j)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------------------------------------------------------
        # 1) mass transfer (most frequent)
        # ------------------------------------------------------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------------------------------------------------------
        # 2) short‑window smoothing
        # ------------------------------------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------------------------------------------------------
        # 3) swap
        # ------------------------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # new convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape stagnation
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (math.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “mass‑inflation’’ step (small n only)
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float):
    """
    Solve: maximise Σ g_j  subject to  g_j ≥ 0  and  (f*g)_i ≤ M   for all i.
    Returns (g, success_flag).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build the (2n‑1) × n constraint matrix (Toeplitz‐like)
    for i in range(2 * n - 1):
        lo = max(0, i - (n - 1))
        hi = min(n - 1, i)
        # j ranges from lo to hi inclusive; coefficient for g_j is f[i‑j]
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            data.append(f[i - j])

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_j  → minimise −Σ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs')
    if not res.success:
        return None, False
    g = np.maximum(res.x, 0.0)
    return g, True


def _lp_improve(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Repeatedly apply the LP “mass‑inflation’’ step and move a fraction t
    towards the LP solution, always staying on the ℓ¹‑simplex.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    t = 0.30                           # convex‑combination weight
    while time.time() < deadline:
        M = _max_convolution_fft(a)
        g, ok = _lp_step(a, M)
        if not ok or g is None:
            break

        # convex combination and projection
        new = (1.0 - t) * a + t * g
        new = _project_to_simplex(new, target_sum)

        val = _score(new)
        if val < best_val:
            best_val = val
            best = new.copy()
            print(f"[{time.time():.2f}] LP improve score={best_val:.6f}")

        a = new
    return best, best_val


# ----------------------------------------------------------------------
# One optimisation run for a fixed length
# ----------------------------------------------------------------------


def _run_one_instance(n: int, init_seq: np.ndarray, deadline: float):
    """
    Optimise a single length‑n instance.
    1. Gradient descent (≈30 % of the allocated time).
    2. LP improvement for n≤3000 (≈20 % of the remaining time).
    3. Simulated‑annealing for the rest.
    Returns (best_sequence, best_score).
    """
    target_sum = math.sqrt(2.0 * n)
    a = init_seq.astype(np.float64, copy=True)
    if a.size != n:
        a = _resize_sequence(a, n)
    a = _project_to_simplex(a, target_sum)

    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a, _score(a)

    # --------------------------------------------------------------
    # 1) Gradient descent phase
    # --------------------------------------------------------------
    gd_time = min(30.0, remaining * 0.30)
    if gd_time > 5.0:
        gd_deadline = now + gd_time
        a, gd_score = _annealed_gradient_descent(a, gd_deadline)
    else:
        gd_score = _score(a)

    # --------------------------------------------------------------
    # 2) LP improvement (only for moderate sizes)
    # --------------------------------------------------------------
    lp_score = float('inf')
    if n <= 3000:
        now = time.time()
        remaining = deadline - now
        if remaining > 10.0:
            lp_deadline = now + min(15.0, remaining * 0.30)
            a, lp_score = _lp_improve(a, target_sum, lp_deadline)

    # --------------------------------------------------------------
    # 3) Simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a, sa_score = _simulated_annealing(a, deadline)
        final_score = min(gd_score, lp_score, sa_score)
    else:
        final_score = min(gd_score, lp_score)

    return a, final_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with a low benchmark score.
    The routine works through several scales (≈1500 → 3000 → 6000 → 12000).
    The best sequence found is returned as a plain Python list.
    """
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # Scales and the fraction of the remaining budget given to each scale
    scales = [(1500, 0.12), (3000, 0.18), (6000, 0.30), (12000, 0.40)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # Warm‑start: up‑sample the previous result if available,
        # otherwise fall back to a power‑law or a globally supplied sequence.
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            if "height_sequence_1" in globals():
                init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                                dtype=np.float64), n)
            else:
                init = _power_law_sequence(n, 0.55, math.sqrt(2.0 * n))

        seq, score = _run_one_instance(n, init, stage_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Scale n={n} improved best score={best_score:.6f}")

        cur_seq = seq  # warm‑start for the next (larger) scale

    # ------------------------------------------------------------------
    # Final refinement on the current best sequence if time permits
    # ------------------------------------------------------------------
    now = time.time()
    if best_seq is not None and now < deadline - 5.0:
        n = best_seq.shape[0] if isinstance(best_seq, np.ndarray) else len(best_seq)
        best_arr = np.array(best_seq, dtype=np.float64)
        best_arr, final_score = _run_one_instance(n, best_arr, deadline)
        if final_score < best_score:
            best_seq = best_arr
            best_score = final_score
            print(f"[{time.time():.2f}] Final refinement improved best to {best_score:.6f}")

    if best_seq is None:
        best_seq = np.array([], dtype=float)

    # Trim trailing near‑zeros for neatness
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    # One final evaluation using the provided benchmark (optional)
    try:
        val = evaluate_sequence(arr.tolist())   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return arr.tolist()
