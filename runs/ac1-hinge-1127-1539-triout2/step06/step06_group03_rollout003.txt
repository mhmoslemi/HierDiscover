"""
Optimisation of a non‑negative step‑function for the “evaluate_sequence”
objective.

The algorithm proceeds in three phases:

1. **Exploratory sampling** – generate many candidate vectors from a
   variety of parametric families (power‑law, exponential, truncated
   linear, Gaussian, edge‑heavy).  Each candidate is normalised to the
   required ℓ¹‑norm (√(2 n)).  A small collection of permutations (sorted,
   reversed, alternating, and a few random shuffles) is tried for each
   multiset.  The best few candidates (the “elite” set) are retained.

2. **Continuous refinement** – each elite vector is fed to a projected
   sub‑gradient descent that directly minimises the maximal convolution
   entry (the objective after normalisation).  The sub‑gradient of the
   max‑convolution is easy to compute: for a maximal index *k*,
   ∇ max = 2·a shifted by *k*.  After each gradient step the vector is
   projected onto the simplex {x ≥ 0, Σx = √(2 n)}.

3. **Permutation annealing** – the refined vectors keep their multiset
   of values but the ordering is optimised by a simulated‑annealing
   process that swaps entries.  The maximal convolution is evaluated
   after each swap (FFT based) and worsening moves are occasionally
   accepted to escape local minima.

Throughout the run the current best objective and the elapsed time are
printed.  The routine respects the supplied time budget (default 1000 s)
and finally returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper to scale a raw vector to the required ℓ¹‑norm
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _gaussian_sequence(n: int, sigma: float, centre: float) -> np.ndarray:
    """Discrete Gaussian centred at *centre* with standard deviation *sigma*. """
    idx = np.arange(n, dtype=float)
    raw = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _edgeheavy_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """
    Heavy mass near the two edges: sum of two power‑law tails.
    a_i = (i+shift)^(-p) + (n-1-i+shift)^(-p)
    """
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p) + (n - 1 - idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Ordering heuristics (a small set of permutations for a given multiset)
# ----------------------------------------------------------------------


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a handful of orderings intended to reduce large‑by‑large overlaps."""
    n = base.size
    asc = np.sort(base)          # smallest → largest
    desc = asc[::-1]             # largest → smallest
    out = [asc, desc]

    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # a few random shuffles
    for _ in range(5):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple sub‑gradient descent on  max_k (a*a)_k .
    After each update the vector is projected onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    iteration = 0
    alpha0 = 1.0                     # initial step‑size factor

    while time.time() < deadline:
        # full convolution via FFT
        size = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

        max_val = conv.max()
        max_idxs = np.flatnonzero(conv == max_val)
        k = int(rng.choice(max_idxs))

        # sub‑gradient of the max entry (2 * shifted copy of a)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        g = np.zeros_like(a)
        if low <= high:
            idx_range = np.arange(low, high + 1)
            g[low:high + 1] = 2.0 * a[k - idx_range]

        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # step‑size decays like 1/√t
            lr = alpha0 * target_sum / (norm_g * np.sqrt(iteration + 1))
            a = a - lr * g

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if iteration % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        iteration += 1

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improve the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # initial temperature proportional to the current objective
    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # swap

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score

        accept = False
        if cur_score <= best_score:
            accept = True
        elif T > 0.0 and rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # occasional tiny reshuffle of a random subset (global jitter)
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best

                        elite.sort(key=lambda x: x[0])
                        elite = elite[:9]

        # ----- Edge‑heavy -----------------------------------------------
        for p in np.linspace(0.44, 0.60, 3):
            for shift in np.linspace(0.0, 2.0, 3):
                base = _edgeheavy_sequence(n, p, shift)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (edge) n={n} p={p:.3f} shift={shift:.2f} obj={score:.6f}")
                    if len(elite) < 9 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:9]

        # ----- Up‑sample a previously discovered construction ----------
        try:
            prev = np.array(height_sequence_1, dtype=float)
            if prev.size:
                old_n = prev.size
                old_x = np.linspace(0.0, 1.0, old_n)
                new_x = np.linspace(0.0, 1.0, n)
                up = np.interp(new_x, old_x, prev)
                up = _project_to_simplex(up, target_sum)
                for cand in _candidate_orderings(up, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] New global best (prev) n={n} obj={score:.6f}")
                    if len(elite) < 9 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:9]
        except NameError:
            pass

        # ----- short random sampling for diversity --------------------
        rnd_end = min(deadline, time.time() + 1.0)
        while time.time() < rnd_end:
            rand = rng.random(n)
            rand = _normalize_to_sum(rand, target_sum)
            score = _max_convolution_fft(rand)
            if score < best_obj:
                best_obj = score
                best_seq = rand.copy()
                print(f"[{time.time():.1f}] New global best (rand) n={n} obj={score:.6f}")
            if len(elite) < 9 or score < elite[-1][0]:
                elite.append((score, rand.copy()))
                elite.sort(key=lambda x: x[0])
                elite = elite[:9]

    # ------------------------------------------------------------------
    # 2. Refinement of elite candidates (sub‑gradient + permutation SA)
    # ------------------------------------------------------------------
    for score0, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size
        sub_dead = min(deadline, time.time() + 25.0)   # up to 25 s per elite
        refined = _subgradient_descent_opt(seed_seq, sub_dead)

        # final permutation optimisation (uses the remaining global budget)
        refined = _permutation_annealing(refined, deadline)

        obj = _max_convolution_fft(refined)
        if obj < best_obj:
            best_obj = obj
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing negligible entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Report the final evaluate_sequence value for verification
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception as e:
        print("evaluate_sequence error:", e)

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
