def _adam_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    beta = 0.5
    beta_max=16384
    beta_factor=1.5
    # Adam parameters:
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr_base=0.1  # but scale later
    beta1=0.9
    beta2=0.999
    eps=1e-8

    while time.time() < deadline:
        g, b = _gradient_softmax(a, beta)
        t += 1
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        if t % 200 == 0:
            # jitter
            jitter = np.random.default_rng().normal(scale=target_sum*0.001/(beta+1), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        if t % 400 == 0 and beta < beta_max:
            beta = min(beta*beta_factor, beta_max)

    return a

def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute convolution
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        # compute gradient
        # gradient = 2 * shift(a, k_max)
        grad = np.zeros_like(a)
        # for i where 0 <= k_max - i < n
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)  # i <= k_max and i in [0, n-1] and k_max-i in [0,n-1]
        # we can vectorize:
        i = np.arange(lo, hi+1, dtype=int)
        j = k_max - i
        grad[i] = 2.0 * a[j]
        # step size schedule:
        it += 1
        step = 1.0 / np.sqrt(it) * target_sum * 0.05  # tune factor
        # move opposite to gradient
        a -= step * grad
        # projection
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
    return a

def _deterministic_hill_climb(a0, b0, deadline):

    b = b0.copy()

    cur_max = b.max()

        j_arr = k_max - np.arange(n)

        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

"""
Hybrid optimisation for the self‑convolution max problem.

The search proceeds in stages over a schedule of increasing lengths.
For each length we (i) initialise a candidate (upsampled best or random),
(ii) run a projected gradient descent on a smooth log‑sum‑exp proxy of the
    max, (iii) run a simple sub‑gradient descent that directly targets the
    current maximal convolution entry, and (iv) perform a deterministic
    hill‑climber that moves mass from the source contributing most to the
    peak towards the destination contributing the least.
All steps respect the ℓ¹‑simplex Σa = √(2 n) and the non‑negativity
constraints.  Remaining budget is spent on a final polishing pass.
The best sequence found (as a plain list of floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers and up‑sampling
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    # small Gaussian noise and projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) gradient utilities
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (via FFT)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Momentum, β‑annealing and occasional jitter are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # annealing schedule for β
    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    # step‑size base (relative to target ℓ¹‑norm)
    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92                       # Nesterov‑style momentum decay

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to leave plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent targeting the exact max
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple sub‑gradient descent on the exact max entry of the self‑convolution.
    Gradient = 2·shift(a, k_max).  Step size decays with √iteration.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()
    step_factor = target_sum * 0.25
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # gradient = 2 * a shifted by k_max (out‑of‑range entries give 0)
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k_max - i
        grad[i] = 2.0 * a[j]

        gn = np.linalg.norm(grad)
        if gn > 0.0:
            step = step_factor / (gn * np.sqrt(it + 1))
            a -= step * grad

        # projection back to the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / np.sqrt(it + 1),
                                size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Deterministic hill‑climber (mass transfer)
# ----------------------------------------------------------------------


def _deterministic_hill_climb(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Move mass from the index that contributes most to the current peak
    towards the index that contributes least.  The convolution is updated
    analytically, so each iteration is cheap.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max
    rng = np.random.default_rng()
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each index to the current peak
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        i = int(np.argmax(contrib))          # source (most damaging)
        j = int(np.argmin(contrib))          # destination (least damaging)

        if i == j:
            # fall back to a random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-15:
            continue
        delta = max_delta                 # move as much as allowed

        # analytical update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        if new_max < cur_max - 1e-12:
            # accept the move
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] deterministic hill improvement: {best_max:.6f}")

    return best_a, best_max

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously discovered construction if available.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * a0.size))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 "
              f"(len={a0.size}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule (larger lengths give finer granularity).
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 25000, 32000]
    # Relative time allocation for each length (sums to ≈1.0)
    fractions = [0.05, 0.07, 0.11, 0.15, 0.20, 0.20, 0.22]

    total_time = budget_s - 5.0
    time_allocs = [total_time * f for f in fractions]

    for idx, (n, alloc) in enumerate(zip(lengths, time_allocs)):
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Allocate time for this length (split into phases)
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left <= 0.0:
            break
        # proportion of the allocated block we can actually spend now
        alloc = min(alloc, time_left)

        # Phase A – projected gradient descent on the smooth proxy
        phase_a_dead = time.time() + 0.45 * alloc
        a_gd = _gradient_descent_opt(a_init, phase_a_dead)
        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

        # Phase B – sub‑gradient descent targeting the exact max entry
        phase_b_dead = time.time() + 0.35 * alloc
        a_sub = _subgradient_descent(best_seq, phase_b_dead)
        score_sub = _max_convolution_fft(a_sub)
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.1f}] Sub‑grad improvement (n={n}) score={score_sub:.6f}")

        # Phase C – deterministic hill climbing
        if time.time() < deadline:
            # compute convolution once for the current best
            b_full = _full_convolution(best_seq)
            phase_c_dead = time.time() + 0.25 * alloc
            a_hc, max_hc = _deterministic_hill_climb(best_seq, b_full, phase_c_dead)
            if max_hc < best_score:
                best_score = max_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # a short sub‑gradient polishing followed by a final hill climb
        sub_dead = time.time() + 0.4 * (deadline - time.time())
        a_final = _subgradient_descent(best_seq, sub_dead)

        if time.time() < deadline:
            b_final = _full_convolution(a_final)
            a_final, max_final = _deterministic_hill_climb(a_final, b_final, deadline)
        else:
            a_final = a_final.copy()

        final_score = _max_convolution_fft(a_final)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for a tidy representation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
