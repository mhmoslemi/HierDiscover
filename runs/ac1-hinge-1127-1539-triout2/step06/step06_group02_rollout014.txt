        conv = _conv_fft(a)  # vector length 2n-1

        a = _project_to_simplex(a, target)

"""
Hybrid local search for the step‑function minimisation problem.

The algorithm works in three stages:
1. Initialise a non‑negative sequence on the ℓ¹‑simplex Σa = √(2·n)
   (power‑law, random or a warm‑start from the global variable
   `height_sequence_1` if it exists).
2. A long simulated‑annealing phase explores the space while always
   respecting the simplex constraint.
3. Deterministic polishing: repeatedly move a small amount of mass from
   indices that contribute most to the current worst convolution entry
   to indices that contribute the least, followed by a projection onto
   the simplex.  This uses the exact sub‑gradient of the max‑convolution
   and often yields a final improvement of a few 10⁻³.

All heavy operations (convolution, max‑convolution) use FFTs, and the
total wall‑clock time is limited to the supplied budget (≈1000 s).  The
best sequence found is stripped of trailing zeros before being returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of appropriate size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi et al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_normalised(a: np.ndarray) -> float:
    """
    Objective value when the sum constraint Σa = √(2·n) holds.
    In that case the evaluation function reduces to the maximal
    convolution entry.
    """
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5, end_temp: float = 1e-5,
                         max_frac_start: float = 0.40, max_frac_end: float = 0.005,
                         max_window: int = 30) -> tuple[np.ndarray, float]:
    """
    Classic SA for the max‑convolution problem.
    Moves consist of:
      – donor → receiver mass transfer,
      – local smoothing of a short window,
      – random swap.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    conv = _conv_fft(a)
    cur_val = conv.max()

    best_a = a.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current worst index
        k_max = int(conv.argmax())

        # contribution of each coordinate to the worst entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        weight = a[idx] * a[k_max - idx]          # a_i * a_{k_max-i}

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:                     # mass transfer
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=weight / total_w)
            else:
                donor = rng.integers(n)

            # pick a receiver that contributes little to the worst entry
            low_thr = np.percentile(weight, 10)
            low_idx = idx[weight <= low_thr]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:                 # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved = a[start:end].copy()
                a[start:end] = saved.mean()
                move_type = "smooth"
                saved_start, saved_end = start, end

        else:                                      # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # Evaluate the new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_val = float(conv_new.max())

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            conv = conv_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_val:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_val = float(conv.max())

        it += 1

    return best_a, best_val

# Deterministic polishing (sub‑gradient descent on the worst entry)
# ----------------------------------------------------------------------


def _polish(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from indices that contribute most
    to the current worst convolution entry to those that contribute least.
    Each step is projected onto the simplex.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score_normalised(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        cur_val = float(conv.max())
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k - idx]                # contribution to the worst entry

        # donor = index with largest contribution, receiver = smallest
        donor = idx[np.argmax(contrib)]
        receiver = idx[np.argmin(contrib)]

        # step size decays with the iteration count
        step_frac = 0.02 / (np.sqrt(it + 1) + 1.0)
        delta = a[donor] * step_frac
        if delta > 0:
            a[donor] -= delta
            a[receiver] += delta
            a = _project_to_simplex(a, target)

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] polish improved to {best_val:.6f}")

        it += 1

    return best_a, best_val

def _run_instance(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    The routine consists of:
      – initialisation,
      – a simulated‑annealing phase,
      – deterministic polishing.
    """
    target = np.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    else:
        # random power‑law initialisation (exponent sampled a bit)
        p = np.random.uniform(0.45, 0.65)
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        seq = seq * (target / seq.sum())

    seq = _project_to_simplex(seq, target)

    # ---- simulated annealing -------------------------------------------
    cur_time = time.time()
    if cur_time < deadline:
        seq, _ = _simulated_annealing(seq, deadline)

    # ---- deterministic polishing ----------------------------------------
    if time.time() < deadline:
        seq, val = _polish(seq, deadline)
    else:
        val = _score_normalised(seq)

    return seq, val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function coefficients vector that minimises the
    provided `evaluate_sequence` function.  The method explores several
    problem sizes, runs a long simulated‑annealing phase followed by a
    deterministic polishing step, and returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # problem sizes – we spend the majority of the budget on the
    # medium size (where the trade‑off between degrees of freedom and
    # computational cost is most favourable).
    sizes = [800, 1500, 3000, 5000]

    # allocate roughly 20 % of the remaining time to each size,
    # keeping the last size for a final refinement.
    for n in sizes:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        # keep at least 30 s for the final polishing of the global best
        time_for_this = min(remaining * 0.18, 250.0)

        run_deadline = now + time_for_this
        seq, score = _run_instance(n, run_deadline)

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} → new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final refinement of the overall best candidate (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        final_deadline = deadline
        best_seq, best_score = _polish(best_seq, final_deadline)

    # Strip trailing zeros (they do not affect the evaluation score)
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.empty(0, dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
