"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Initial candidates** – a few simple families are generated:
      * power‑law a_i ∝ (i+shift)^‑p   (p≈0.5, shift≈0.6)
      * “valley’’ shape a_i ∝ |cos(π·i/(n‑1))|
      * a random simplex point.
   All candidates are projected onto the ℓ¹‑simplex
   Σa_i = √(2·n) (the scaling used by the objective, so the
   evaluated objective equals the maximal convolution entry).

2. **Best seed** – the candidate with the smallest max‑convolution
   value is kept as the starting point.

3. **Continuous simulated annealing** – the entries of the seed are
   continuously perturbed while the ℓ¹‑sum is kept exactly constant.
   A move selects two indices *i, j* and transfers a small amount
   δ from a_j to a_i (δ is chosen so that both stay non‑negative).
   The new sequence is accepted if it lowers the objective or with a
   probability exp(‑Δ/T) where the temperature T slowly decays.
   The step‑size and temperature are annealed; occasional tiny
   redistributions of a random 1 % of the mass are performed to
   escape local minima.

4. **Result** – the best sequence seen during the whole run is
   returned as a plain Python list.

The whole routine respects the supplied time budget (default 1000 s)
and uses only NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast max‑convolution (linear, not circular) using FFT
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    # size = next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used only for the initial seed)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^(-p), projected to Σa = √(2·n)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return a


def _valley_sequence(n: int) -> np.ndarray:
    """
    a_i ∝ |cos(π·i/(n‑1))|, projected to Σa = √(2·n).
    The shape is large at the two ends and small in the centre.
    """
    idx = np.arange(n, dtype=float)
    raw = np.abs(np.cos(np.pi * idx / (n - 1)))
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return a


# ----------------------------------------------------------------------
# Continuous simulated annealing on the values (sum is kept exact)
# ----------------------------------------------------------------------


def _continuous_annealing(
    a0: np.ndarray, deadline: float, seed: int = 0
) -> np.ndarray:
    """
    Simulated‑annealing optimisation that keeps Σa constant.
    The objective is the maximal entry of a*a (because Σa = √(2·n)).
    """
    rng = np.random.default_rng(seed)
    a = a0.copy()
    n = a.size
    # reference sum (exact, will be preserved by the moves)
    target_sum = np.sqrt(2.0 * n)

    # initial objective
    best = a.copy()
    best_obj = _max_convolution_fft(best)

    # temperature schedule
    T = best_obj * 0.05 + 1e-12          # start relatively low, allows some uphill moves
    cooling = 0.9999

    # step size (typical magnitude of a single transfer)
    avg = target_sum / n
    step = avg * 0.5                     # start with a moderate step

    it = 0
    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a transfer δ from a[j] → a[i]
        delta = (rng.random() - 0.5) * 2.0 * step   # uniform in [-step, step]
        # enforce non‑negativity
        delta = max(-a[i], min(delta, a[j]))
        if abs(delta) < 1e-15:
            continue

        # apply move
        a[i] += delta
        a[j] -= delta

        cur_obj = _max_convolution_fft(a)
        dE = cur_obj - best_obj

        accept = False
        if cur_obj < best_obj:
            accept = True
        else:
            # Metropolis criterion
            if T > 0.0 and rng.random() < np.exp(-dE / T):
                accept = True

        if accept:
            if cur_obj < best_obj:
                best_obj = cur_obj
                best = a.copy()
        else:
            # revert
            a[i] -= delta
            a[j] += delta

        # schedule updates
        it += 1
        if it % 2000 == 0:
            T *= cooling
            step *= 0.995          # slowly shrink steps

        # occasional tiny redistribution of a random 1 % of the mass
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.01
            if mask.any():
                sub = a[mask]
                sub_sum = sub.sum()
                # keep total mass of the mask unchanged but randomise its split
                rand = rng.random(sub.size)
                rand = rand / rand.sum() * sub_sum
                a[mask] = rand
                # recompute objective after the reshuffle
                cur_obj = _max_convolution_fft(a)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best = a.copy()

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the target
    ``evaluate_sequence`` objective.  The routine uses a small family of
    seed vectors and then applies a continuous simulated‑annealing
    optimisation while respecting the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    best_seq = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # 1. Generate a few seed sequences (different families, a few lengths)
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000]          # lengths that keep FFT cheap
    for n in lengths:
        if time.time() >= deadline:
            break

        # power‑law seed (p≈0.5, shift≈0.6) – this family performed best earlier
        a_pow = _powerlaw_sequence(n, p=0.5, shift=0.6)
        obj_pow = _max_convolution_fft(a_pow)
        if obj_pow < best_obj:
            best_obj = obj_pow
            best_seq = a_pow.copy()
            print(f"[{time.time():.1f}] seed (power‑law) n={n} obj={best_obj:.6f}")

        # valley (large‑at‑ends) seed
        a_val = _valley_sequence(n)
        obj_val = _max_convolution_fft(a_val)
        if obj_val < best_obj:
            best_obj = obj_val
            best_seq = a_val.copy()
            print(f"[{time.time():.1f}] seed (valley) n={n} obj={best_obj:.6f}")

        # a short random simplex point (adds diversity)
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, np.sqrt(2.0 * n))
        obj_rnd = _max_convolution_fft(rnd)
        if obj_rnd < best_obj:
            best_obj = obj_rnd
            best_seq = rnd.copy()
            print(f"[{time.time():.1f}] seed (random) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine the current best seed with continuous annealing
    # ------------------------------------------------------------------
    if best_seq is not None:
        # give the majority of the remaining budget to annealing
        anneal_deadline = deadline
        refined = _continuous_annealing(best_seq, anneal_deadline, seed=seed + 1)
        refined_obj = _max_convolution_fft(refined)

        if refined_obj < best_obj:
            best_obj = refined_obj
            best_seq = refined

        print(f"[{time.time():.1f}] after annealing best obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity / tiny clean‑up
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (objective = 2)
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing tiny entries – pure cosmetics
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Report the true evaluate_sequence value for verification (optional)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective ≈ {best_obj:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
