"""
Adversarial step‑function optimizer – sub‑gradient Adam version

The algorithm searches for a non‑negative vector `a` of length `n`
with ℓ₁‑norm `S = sqrt(2·n)` that minimises the peak of the
self‑convolution `a*a`.  It consists of three phases:

1. **Exploration** – many short Adam restarts with diverse
   lengths and three families of initial shapes (uniform,
   Dirichlet, edge‑heavy power/exp).  The edge‑heavy shapes are
   parametrised by a random decay law, which gives a rich supply
   of candidates.

2. **Optimisation** – Adam is run on the *log‑softmax* parametrisation
   of the vector.  The gradient used is the *exact sub‑gradient* of
   the convolution peak (no τ‑soft‑max approximation): for the current
   peak positions `k` we have  
   ∂max_b/∂a = 2·conv(δ_{k∈peak}, a[::-1]) restricted to the original
   length.  This gives a more faithful descent direction than the
   previous soft‑max based gradient.

3. **Final refinement** – the best candidate obtained in phase 1 is
   refined for the remaining time budget.

The routine keeps the best sequence seen in the global variable
`height_sequence_1`.  It aims to push the evaluation score (which,
under the normalisation `∑a = √(2n)`, equals the convolution peak)
below the previous best of 1.5113 and hopefully under the target
1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    total_len = n + m - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    B = np.fft.rfft(b, n=f_len)
    conv = np.fft.irfft(A * B, n=f_len)[:total_len]
    return conv


def _full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a."""
    return _fft_convolve(a, a)


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Exact sub‑gradient of the convolution peak
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray):
    """
    Returns (grad, max_b) where grad = ∂ max(b) / ∂ a,
    b = a * a (full convolution).
    """
    n = a.shape[0]
    b = _full_convolution_self(a)
    max_b = b.max()
    # mask of all (near‑)maximal entries (tolerance for floating errors)
    eps = 1e-12
    mask = (b >= max_b - eps).astype(float)          # shape (2n‑1,)

    # gradient = 2 * conv(mask, a[::-1]) restricted to length n
    a_rev = a[::-1]
    conv_res = _fft_convolve(mask, a_rev)           # length = 3n‑2
    start = n - 1
    grad = 2.0 * conv_res[start:start + n]
    return grad, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize_subgrad(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise the convolution peak using Adam on the log‑softmax
    parametrisation, employing the exact sub‑gradient.
    Returns the best vector encountered and its hard peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # required ℓ₁‑norm

    # Initialise logits p such that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)               # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.30                      # initial learning rate

    # bookkeeping
    best_a = a0.copy()
    best_peak = _full_convolution_self(a0).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- gradient of hard max (exact sub‑gradient) ----
        g, peak = _max_subgradient(a)

        # keep the best hard peak
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- chain rule: gradient w.r.t. logits ----
        mu = np.dot(a, g) / S      # scalar ⟨a, g⟩ / S
        grad_p = a * (g - mu)      # shape (n,)

        # optional clipping (helps stability for large n)
        max_abs = np.max(np.abs(grad_p))
        clip_thr = 5.0 * S
        if max_abs > clip_thr:
            grad_p = grad_p * (clip_thr / max_abs)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules ----
        lr *= 0.9999                # very slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak

def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a feasible start vector of length n.
    The shape is chosen randomly from two families:
      * power‑law decay (distance+offset)^{-p}
      * exponential decay  exp(-λ·distance)
    A uniform background can be mixed in, then the vector is
    normalised to the required ℓ₁‑norm.
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0        # distance to nearest edge (1‑based)

    # pick a decay family
    if rng.random() < 0.55:
        # ----- power‑law -----
        p = rng.uniform(0.2, 2.0)                 # exponent
        offset = rng.uniform(0.0, 5.0)            # denominator shift
        core = (d + offset) ** (-p)
    else:
        # ----- exponential -----
        lam = rng.uniform(0.001, 0.2)             # decay rate
        core = np.exp(-lam * d)

    mix = rng.uniform(0.0, 1.0)                  # how much of the edge‑heavy part
    base = np.ones(n)                           # uniform background
    raw = mix * core + (1.0 - mix) * base

    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs many short Adam restarts with diverse
    initial vectors, keeps the best candidate and finally refines it
    with the remaining time budget.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if available
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Helper to build an initial feasible vector for a given length
    # ------------------------------------------------------------------
    def _initial_vector(n):
        # Occasionally warm‑start from the global best (scaled)
        if base_seq is not None and rng.random() < 0.15:
            a = _scale_sequence(base_seq, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * np.sqrt(2.0 * n)
            return a

        style = rng.choice(['uniform', 'dirichlet', 'edge'])
        S = np.sqrt(2.0 * n)

        if style == 'uniform':
            a = np.full(n, S / n)
        elif style == 'dirichlet':
            a = rng.random(n)
            a = a / a.sum() * S
        else:   # edge‑heavy with random parameters
            a = _edge_initial(n, rng)

        return a

    # ------------------------------------------------------------------
    # Phase 1 – many short restarts (exploration)
    # ------------------------------------------------------------------
    short_budget = 8.0                     # seconds per restart
    while time.time() + short_budget < end_time - 30.0:
        attempt += 1

        # Choose a length: small/medium/large mix
        r = rng.random()
        if r < 0.55:
            n = rng.integers(1500, 8000)
        elif r < 0.85:
            n = rng.integers(8000, 20000)
        else:
            n = rng.integers(20000, 35000)

        a_start = _initial_vector(n)

        slice_deadline = min(time.time() + short_budget, end_time - 30.0)
        a_opt, peak = _adam_optimize_subgrad(a_start, slice_deadline, rng)
        score = peak                     # with our normalisation, score == max_b

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart] attempt={attempt} n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final long refinement on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 5.0:
            # Ensure exact ℓ₁‑norm before refinement
            S = np.sqrt(2.0 * len(best_seq))
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum() * S

            a_refined, peak = _adam_optimize_subgrad(best_seq, end_time, rng)
            refined_score = peak
            if refined_score < best_score:
                best_score = refined_score
                best_seq = a_refined.copy()
                print(f"[refine] final refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
