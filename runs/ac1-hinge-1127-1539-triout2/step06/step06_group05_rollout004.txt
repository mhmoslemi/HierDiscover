"""
A hierarchical, multi‑phase optimizer for the self‑convolution peak problem.

The algorithm works as follows

1. **Diverse initialisation** – draws a random shape (uniform, geometric,
   sinusoidal window, triangular, edge‑heavy reciprocal, sparse spikes or a
   random convex blend) and projects it onto the ℓ¹‑simplex
   Σa = √(2·n).

2. **Smooth Adam optimisation** – minimises a log‑sum‑exp (soft‑max)
   approximation of the convolution peak.  The temperature α is slowly
   increased up to a large value (≈200) so the gradient approaches the
   true max‑gradient.  After each step the vector is projected onto the
   simplex; optional mirror symmetry can be enforced.

3. **Sub‑gradient descent on the exact max** – identifies the index of the
   current worst convolution entry and takes a sub‑gradient step that reduces
   the contributing coefficients.  The step size decays as 1/√t.

4. **Local random‑swap hill‑climbing** – repeatedly moves a tiny amount of
   mass between two random positions, keeping the simplex constraint.

5. **Hierarchical up‑sampling** – the best sequence found so far is
   interpolated to a larger length and re‑optimised, allowing finer control
   of the convolution shape.

The whole search runs inside the allocated time budget (default 1000 s).  The
best sequence discovered is returned and also stored in the global variable
``height_sequence_1`` for possible reuse by future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n-1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Optional symmetry enforcement
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry v[i] = v[-1-i]."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) – used when we enlarge the problem size.
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to new length n."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Initial sequence generator – a rich family of analytic and random shapes.
# ----------------------------------------------------------------------
def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with ℓ¹‑norm √(2·n)."""
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 7)   # 0–6

    if typ == 0:
        # Uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 1:
        # Geometric decay from both ends
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1])

    elif typ == 2:
        # Sinusoidal window (half‑wave) with random exponent
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        power = rng.uniform(0.5, 1.5)
        a = a ** power

    elif typ == 3:
        # Triangular shape
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    elif typ == 4:
        # Edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 5:
        # Sparse random spikes (exponential masses)
        k = rng.integers(5, max(6, min(30, n // 10) + 1))
        pos = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[pos] = rng.exponential(scale=1.0, size=k)

    else:
        # Random convex combination of a few bases
        bases = []
        bases.append(rng.random(n))                                 # uniform
        rate = rng.uniform(0.9, 0.999)
        left = rate ** np.arange(n)
        bases.append((left + left[::-1]))                           # geometric
        a_sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        bases.append(a_sin)                                         # sinusoidal
        w = rng.random(len(bases))
        w /= w.sum()
        a = sum(wi * bi for wi, bi in zip(w, bases))

    # Clip negatives (just in case) and normalise to the required ℓ¹‑norm.
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp approximation of max.
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 0.5
    alpha_mul = 1.05
    max_alpha = 200.0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.02 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score:
                    a = cand

        # Temperature / learning‑rate schedules
        alpha = min(alpha * alpha_mul, max_alpha)
        lr *= 0.9999

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent directly on the true max entry.
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the maximal entry
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g = np.zeros_like(a)
        g[idx_i] = a[idx_j]

        # decreasing step size
        eta = 0.18 / np.sqrt(it)
        a = a - eta * g

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_val = np.max(_full_convolution(cand))
                if cand_val < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_val
                    if cand_val < best_score:
                        best_score = cand_val
                        best_a = a.copy()

        # early stagnation break
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined three‑phase optimiser (Adam → sub‑gradient → local search)
# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 60 % Adam, 30 % sub‑gradient, 10 % local swap
    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _adam_optimize(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _subgrad_optimize(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_optimize(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # Choose the best among the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – time‑aware hierarchical search.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function whose self‑convolution peak
    (the evaluation metric) is as small as possible.
    Returns the best sequence found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # try to reuse a previously discovered construction
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    best_overall_n = None
    attempts = 0
    last_improve_time = start_time

    # ------------------------------------------------------------------
    # Main timed loop – repeatedly restart with a fresh length and initialisation.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Choose a problem size.
        if best_overall_seq is None:
            n = int(rng.integers(1800, 12000))
        else:
            if rng.random() < 0.55:
                # perturb the current best length by ±10 %
                scale = rng.uniform(0.9, 1.1)
                n = max(800, int(best_overall_n * scale))
            else:
                # occasionally double the length (hierarchical up‑sampling)
                n = int(best_overall_n * 2)
            n = min(n, 15000)

        # Randomly decide whether to enforce symmetry for this attempt.
        enforce_sym = rng.random() < 0.48

        # Generate a fresh initial vector.
        if base_seq is not None and rng.random() < 0.3:
            a0 = _upsample_seq(base_seq, n)
            # tiny jitter to avoid getting stuck on the exact up‑sampled shape
            a0 += rng.normal(scale=0.005 * a0.mean(), size=n)
            a0 = np.clip(a0, 0.0, None)
            S = np.sqrt(2.0 * n)
            a0 = _project_simplex(a0, S)
        else:
            a0 = _initial_sequence(n, rng)

        if enforce_sym:
            a0 = _symmetrize(a0)

        # Allocate a slice of the remaining budget for this attempt.
        slice_deadline = min(time.time() + rng.uniform(12.0, 22.0), deadline)

        # Run the three‑phase optimiser.
        a_opt, score = _refine_sequence(a0, slice_deadline, rng, enforce_sym)

        # Update global best.
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            best_overall_n = n
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_overall_score:.6f}")

        # Early exit if we have already reached the target bound.
        if best_overall_score <= 1.503:
            break

        # If we have not improved for a while, discard the current base sequence.
        if time.time() - last_improve_time > 45.0:
            base_seq = None
        else:
            # keep the current best as a possible base for up‑sampling
            base_seq = best_overall_seq.copy()

    # ------------------------------------------------------------------
    # Final fine‑tuning on a larger scale (optional up‑sampling).
    # ------------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline:
        # try one extra up‑sampling step if time permits
        upscale_factor = 1.5
        n_big = int(best_overall_n * upscale_factor)
        if n_big <= 15000:
            a_big = _upsample_seq(best_overall_seq, n_big)
            S_big = np.sqrt(2.0 * n_big)
            a_big = _project_simplex(a_big, S_big)
            if rng.random() < 0.5:
                a_big = _symmetrize(a_big)
            t_big = min(time.time() + 12.0, deadline)
            a_big_opt, sc_big = _refine_sequence(a_big, t_big, rng, enforce_sym=False)
            if sc_big < best_overall_score:
                best_overall_score = sc_big
                best_overall_seq = a_big_opt
                best_overall_n = n_big
                print(f"[upscale] factor={upscale_factor:.1f} new best={best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} "
          f"internal_best={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    # expose the best sequence as a global variable for possible up‑sampling later
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
