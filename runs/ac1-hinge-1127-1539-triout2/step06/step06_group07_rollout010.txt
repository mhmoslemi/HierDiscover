def _local_search(a0, deadline, rng):

def _optimize_adam(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.1
    alpha_max = 5000.0

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = _project_simplex(a, S)

        # random mass swap for exploration (rare)
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # increase alpha
        alpha = min(alpha * 1.07, alpha_max)

    return best_a, best_score

def _optimize_subgrad(a0, deadline, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # subgradient: for i in overlapping region
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # occasional random perturbation: move small mass
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()

"""
Propose a non‑negative step function (vector) that minimizes
evaluate_sequence() – i.e. the peak value of its self‑convolution.
The optimizer works in several stages with increasing vector length.
Each stage runs a warm‑started Adam optimiser on a smooth max‑approximation,
followed by a true sub‑gradient descent and a cheap local hill‑climbing
search.  The later stages use larger problems (up to 6000 entries) where
the extra degrees of freedom typically lower the optimum.
All updates keep the ℓ₁‑norm fixed to √(2·n) by projection onto the simplex.
"""
import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max‑norm
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # hyper‑parameters (tuned for speed / quality)
    lr = 0.6
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.1          # start temperature (smoothness)
    alpha_max = 5000.0   # final temperature – very sharp

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps Σa = S)
        a = _project_simplex(a, S)

        # occasional tiny random mass‑swap for extra exploration
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # increase temperature (sharpen the soft‑max)
        alpha = min(alpha * 1.07, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent directly reducing the worst convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.8  # initial step‑size (will decay as 1/√t)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max entry (only the active shift contributes)
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # light random perturbation (helps escape plateaus)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur:
                    a = cand
                    cur = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Combine Adam + sub‑gradient (fixed split of the allocated time)
# ----------------------------------------------------------------------
def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.7) -> tuple[np.ndarray, float]:
    """Run Adam for a fraction of the time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


# ----------------------------------------------------------------------
# Simple hill‑climbing local search (pairwise mass moves)
# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap stochastic hill‑climbing that preserves the ℓ₁‑norm."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    step_scale = 0.12   # maximal proportion of a[i] that may be moved
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = a[i] * step_scale * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cur = np.max(_full_convolution(cand))
        if cur < best_score:
            best_score = cur
            best_a = cand.copy()
            a = cand
            # a tiny boost of the step size after a success
            step_scale = min(step_scale * 1.03, 0.25)
        else:
            # accept a worse move with a tiny probability to keep diversity
            if rng.random() < 0.008:
                a = cand

        # slowly decay step size
        step_scale *= 0.99994
        if it % 8000 == 0:
            step_scale = 0.12

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine performs a multi‑stage optimisation
    with increasing problem sizes (800 → 2000 → 4000 → 6000) and combines
    Adam, sub‑gradient descent and a final local search.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if available
    # ------------------------------------------------------------------
    best_seq = None
    if 'height_sequence_1' in globals():
        try:
            best_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            best_seq = None

    best_score = np.inf

    # ------------------------------------------------------------------
    # Progressive optimisation schedule (sizes & time fractions)
    # ------------------------------------------------------------------
    sizes = [800, 2000, 4000, 6000]
    # fractions of the *total* budget (they sum to < 1 to keep a safety margin)
    fractions = [0.09, 0.18, 0.27, 0.36]   # total 0.90

    total_budget = deadline - start
    cumulative = 0.0

    for stage, n in enumerate(sizes):
        if time.time() >= deadline:
            break

        # allocate time for this stage
        cumulative += fractions[stage]
        stage_deadline = start + cumulative * total_budget
        stage_deadline = min(stage_deadline, deadline)

        # initialise the vector for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)

        # small jitter to avoid exact repetitions
        jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
        a0 = np.clip(a0 + jitter, 0.0, None)

        # Project onto the ℓ₁‑simplex (Σa = √(2·n))
        S = np.sqrt(2.0 * n)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0)
        a0 = a0 / a0.sum() * S

        # 1️⃣ Adam → sub‑gradient phase
        a_opt, sc_opt = _optimize_combined_adam(a0, stage_deadline, rng,
                                                fraction_adam=0.72)

        # 2️⃣ Short local‑search refinement (if time remains)
        if time.time() < stage_deadline - 0.5:
            a_opt, sc_opt = _local_search(a_opt, stage_deadline, rng)

        # keep the globally best vector
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[stage] n={n} new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
