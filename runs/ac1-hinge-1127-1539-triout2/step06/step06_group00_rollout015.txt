"""
step‑function optimizer – LP‑boost + Adam refinement

The search works in two phases for a given length `n`:

1. **LP‑boost** – given a feasible sequence `f` (ℓ₁‑norm = √(2n)),
   we solve a linear program
        max   Σ g_j
        s.t.  f * g ≤ M   (pointwise) ,   g ≥ 0
   where `M = max (f*f)`.  The solution `g₀` is scaled to the correct
   ℓ₁‑norm and mixed with `f` (convex combination with a step size `t`).
   The mixture is accepted if it lowers the peak; otherwise `t` is
   reduced.  Repeating this quickly drives the peak down.

2. **Adam fine‑tuning** – after the LP phase we run a projected Adam
   optimiser that minimises a τ‑soft‑max approximation of the peak.
   This removes residual irregularities.

Multiple random restarts with different lengths are performed until the
global time budget (≈ 1000 s) expires.  The best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import numpy as np

# SciPy is optional – the optimizer falls back to pure‑Adam if unavailable.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
except Exception:          # pragma: no cover
    linprog = None
    csr_matrix = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear (full) self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a."""
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Project Adam onto the ℓ₁‑simplex (sum = √(2·n))."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3

    # initial temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # gradient (soft‑max peak)
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # chain rule to logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        tau *= 0.9997
        lr *= 0.9999

        if it % 800 == 0:
            p += rng.normal(scale=0.015, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Parameterised edge‑heavy initial shapes
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a feasible start vector – power‑law decay from the edges."""
    S = np.sqrt(2.0 * n)

    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    p = rng.uniform(0.2, 1.0)          # exponent
    offset = rng.uniform(0.0, 5.0)     # denominator shift
    mix = rng.uniform(0.0, 1.0)        # edge vs uniform mix

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base

    raw = raw.clip(min=0.0)
    raw = raw / raw.sum() * S
    return raw


# ----------------------------------------------------------------------
# Linear‑programming boost (single length)
# ----------------------------------------------------------------------
def _lp_boost(f: np.ndarray,
              S: float,
              deadline: float,
              rng: np.random.Generator,
              max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑boost step described in the reference paper.
    Returns a (potentially) improved sequence and its peak value.
    """
    if linprog is None:                     # pragma: no cover
        # SciPy not available – skip the LP phase.
        return f, np.max(_full_convolution_fft(f))

    n = f.shape[0]
    # Pre‑compute the index pattern of the Toeplitz convolution matrix.
    rows = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    cols = np.repeat(np.arange(n, dtype=np.int32), n)

    best_f = f.copy()
    best_M = np.max(_full_convolution_fft(f))

    t = 0.3                # initial mixing weight
    it = 0

    # Main LP‑boost loop.
    while time.time() < deadline - 0.1:
        it += 1

        conv_f = _full_convolution_fft(f)
        M = conv_f.max()

        # Build the (A_ub, b_ub) pair for the current ‑M bound.
        data = np.tile(f, n)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M + 1e-12)
        c = -np.ones(n)               # maximise Σg ⇔ minimise −Σg
        bounds = [(0, None)] * n

        # Solve the LP.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # Infeasibility or numerical trouble – abort the boost.
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # Scale to the correct ℓ₁‑norm.
        scale = S / sum_g0
        g = g0 * scale

        # Convex combination.
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum() * S

        conv_new = _full_convolution_fft(f_new)
        M_new = conv_new.max()

        # Accept only if the peak decreased.
        if M_new < M:
            f = f_new
            if M_new < best_M:
                best_M = M_new
                best_f = f_new.copy()
                print(f"[LP boost] n={n} iter={it} new peak={M_new:.6f}")
            # Slightly increase step size to accelerate movement.
            t = min(t * 1.05, 0.8)
        else:
            # No improvement – shrink the step size.
            t *= 0.5
            if t < 1e-5:
                break

        if it >= max_iters:
            break

    return best_f, best_M


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search runs many short restarts, each consisting of an LP‑boost phase
    followed by a short Adam refinement.  The best sequence seen is stored in
    the global variable ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction, if any.
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Helper: quick internal evaluation (the same objective as ``evaluate_sequence``).
    # ------------------------------------------------------------------
    def _internal_score(seq: np.ndarray) -> float:
        n = len(seq)
        b = _full_convolution_fft(seq)
        max_b = b.max()
        sum_a = seq.sum()
        if sum_a < 0.01:
            return np.inf
        return float(2.0 * n * max_b / (sum_a ** 2))

    # ------------------------------------------------------------------
    # Main optimisation loop – many independent restarts.
    # ------------------------------------------------------------------
    while time.time() < end_time - 10.0:
        attempt += 1

        # Random length (more points give a richer search space).
        n = int(rng.integers(1500, 3500))

        # Initialise a feasible vector.
        if base_seq is not None and rng.random() < 0.2:
            a_start = _scale_sequence(base_seq, n)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum() * np.sqrt(2.0 * n)
        else:
            a_start = _edge_initial(n, rng)          # already normalised

        S = np.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # Phase 1 – LP‑boost (aggressive peak reduction)
        # ----------------------------------------------------------------
        boost_deadline = min(end_time - 5.0, time.time() + 8.0)   # allocate a few seconds
        f_boost, peak_boost = _lp_boost(a_start, S, boost_deadline, rng)

        # ----------------------------------------------------------------
        # Phase 2 – short Adam refinement (polish)
        # ----------------------------------------------------------------
        if time.time() < end_time - 2.0:
            refine_deadline = end_time - 0.5
            f_refined, peak_refined = _adam_optimize(f_boost, refine_deadline, rng)
        else:
            f_refined, peak_refined = f_boost, peak_boost

        # Compute the (internal) score – because sum = √(2n) this equals the peak.
        cur_score = peak_refined

        if cur_score < best_score:
            best_score = cur_score
            best_seq = f_refined.copy()
            print(f"[attempt={attempt}] n={n}  new best internal peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal peak = {best_score:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
