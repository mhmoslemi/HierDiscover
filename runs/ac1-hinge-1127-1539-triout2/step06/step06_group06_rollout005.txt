"""
Optimiser for the convolution‐max benchmark.

The algorithm works in three stages (increasing number of steps):
1. Generate a diverse set of seed sequences (power‑law, exponential,
   random Dirichlet, sinusoidal, triangular, plus the previous best).
2. For each seed run a short Adam optimiser on a soft‑max approximation
   of the max‑convolution.  After each update the vector is projected onto
   the ℓ¹‑simplex Σa_i = √(2·n) to keep the objective invariant.
3. Continue from the best seed with a simulated‑annealing (SA) phase that
   performs mass‑transfer, short‑window smoothing and swaps.  The SA uses
   the current maximum‑convolution index to compute a sensitivity vector
   and preferentially moves mass away from the most harmful entries.
4. A final deterministic local‑improvement loop moves a small fraction of
   the biggest contributor to the current maximum to the smallest
   contributor; this often removes a remaining “spike”.

The whole search respects the overall time budget (default 1000 s).  The
best sequence found (trimmed of trailing near‑zeros) is returned as a
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numerical helpers
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact benchmark score: 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Optimisation primitives

def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max approximation of max‑convolution.
    Returns the best sequence and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Learning‑rate scale – calibrated to the typical magnitude of a
    alpha = target_sum * 0.15

    beta = 1.0
    max_beta = 8192.0
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] ADAM n={n} it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase soft‑max sharpness gradually
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.01,
                         max_window: int = 50) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement on the simplex.
    Moves: mass transfer, local smoothing, random swaps.
    """
    a = a0.copy()
    n = a.size
    size_fft = 1 << ((2 * n - 1).bit_length())
    cur_conv = _conv_fft(a)
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # ----- identify the most critical index of the convolution -----
        k_max = int(np.argmax(cur_conv))

        # ----- compute sensitivities (derivative of the max entry) -----
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None
        saved = None  # placeholder for reverted data

        # -------------------- mass transfer --------------------
        if rng.random() < 0.68:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # -------------------- short‑window smoothing --------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved = a[start:end].copy()
                avg = saved.mean()
                a[start:end] = avg
                move_type = "smooth"

        # -------------------- random swap --------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ----- evaluate new state -----
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # ----- Metropolis acceptance -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            cur_conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[start:end] = saved
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape deep plateaus
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            # re‑normalise to the correct ℓ¹‑norm
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            cur_conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _deterministic_local_improve(a0: np.ndarray,
                                 deadline: float,
                                 max_transfer_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simple greedy cleanup: move a small fraction of the biggest contributor
    to the current maximum convolution entry towards the smallest contributor.
    """
    a = a0.copy()
    n = a.size
    size_fft = 1 << ((2 * n - 1).bit_length())
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))

        # contributions to the current max entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]   # element‑wise product
        if contrib.size == 0:
            break

        i_max = idx[np.argmax(contrib)]
        # pick a low‑contribution index (preferably zero)
        i_min = idx[np.argmin(contrib)]

        if i_max == i_min or a[i_max] <= 0:
            break

        delta = a[i_max] * max_transfer_frac
        a[i_max] -= delta
        a[i_min] += delta

        new_score = _score(a)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] LOCAL improve score={best_score:.6f}")
        else:
            # revert
            a[i_max] += delta
            a[i_min] -= delta
            break   # no further improvement possible

    return best_a, best_score


def _generate_seeds(n: int, target_sum: float) -> list[np.ndarray]:
    """Create a diverse collection of initial sequences."""
    rng = np.random.default_rng()
    seeds = []

    # power‑law family
    for p in [0.4, 0.5, 0.55, 0.6, 0.65, 0.7]:
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = a * (target_sum / a.sum())
        seeds.append(a)

    # exponential decays
    for lam in [0.001, 0.005, 0.01, 0.02, 0.05]:
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a = a * (target_sum / a.sum())
        seeds.append(a)

    # random Dirichlet / Gamma draws
    for _ in range(5):
        a = rng.gamma(1.0, 1.0, size=n)
        a = a * (target_sum / a.sum())
        seeds.append(a)

    # sinusoidal shape (positive half‑wave)
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * x / (n + 1))
    a = a * (target_sum / a.sum())
    seeds.append(a)

    # simple triangular (high at the ends, low in the middle)
    tri = np.concatenate([np.linspace(1, 0, n // 2),
                          np.linspace(0, 1, n - n // 2)])
    tri = tri * (target_sum / tri.sum())
    seeds.append(tri)

    # include the previous best if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(globals()['height_sequence_1'], dtype=np.float64)
            prev = _resize_sequence(prev, n)
            prev = _project_to_simplex(prev, target_sum)
            seeds.insert(0, prev)
        except Exception:
            pass

    return seeds


def _optimise_length(n: int,
                     deadline: float,
                     init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length n.
    Returns the best sequence and its exact score.
    """
    target_sum = np.sqrt(2.0 * n)

    # initialise the seed pool
    if init_seq is not None:
        init_resized = _resize_sequence(init_seq, n)
        init_resized = _project_to_simplex(init_resized, target_sum)
        seeds = [init_resized] + _generate_seeds(n, target_sum)
    else:
        seeds = _generate_seeds(n, target_sum)

    best_seq = None
    best_score = float("inf")
    now = time.time()
    remaining_seeds = len(seeds)

    for idx, seed in enumerate(seeds):
        if time.time() >= deadline - 5.0:
            break
        # allocate a modest slice of the remaining budget to this seed
        now = time.time()
        remaining = deadline - now
        # at least 7 s per seed, otherwise split remaining evenly
        alloc = max(7.0, remaining / (remaining_seeds - idx + 1))
        gd_deadline = now + alloc
        a_opt, s_opt = _adam_optimize(seed, gd_deadline, target_sum)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Length {n} seed {idx} new best score={best_score:.6f}")

        # early exit if we are already very close to the target
        if best_score < 1.505:
            # continue a little longer to see if we can push below 1.503
            pass

    # ---- simulated‑annealing on the current best ----
    if best_seq is not None and time.time() < deadline - 5.0:
        a_sa, s_sa = _simulated_annealing(best_seq, deadline)
        if s_sa < best_score:
            best_score, best_seq = s_sa, a_sa.copy()
            print(f"[{time.time():.2f}] Length {n} after SA best score={best_score:.6f}")

    # ---- final greedy clean‑up (tiny leftover time) ----
    if best_seq is not None and time.time() < deadline - 2.0:
        a_loc, s_loc = _deterministic_local_improve(best_seq, deadline)
        if s_loc < best_score:
            best_score, best_seq = s_loc, a_loc.copy()
            print(f"[{time.time():.2f}] Length {n} final local improve best={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence with low benchmark score.
    The routine works on three increasing lengths (≈2000 → 4000 → 8000)
    re‑using the best solution of the previous scale as a warm start.
    """
    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # increasing scales – fractions roughly sum to the whole budget
    scales = [(2000, 0.20), (4000, 0.30), (8000, 0.50)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # initialise the starting point for this scale
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            # fall back to a reasonable power‑law seed
            init = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

        seq, score = _optimise_length(n, stage_deadline, init_seq=init)

        if seq is not None and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Scale n={n} improved global best score={best_score:.6f}")

        cur_seq = seq   # warm start for the next (larger) scale

    # Trim trailing near‑zeros for a tidy result
    if best_seq is None:
        best_seq = []
    else:
        arr = np.array(best_seq, dtype=float)
        nz = np.where(arr > 1e-12)[0]
        if nz.size:
            arr = arr[: nz[-1] + 1]
        best_seq = arr.tolist()

    # Final optional evaluation using the provided benchmark
    try:
        val = evaluate_sequence(best_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"Final evaluation failed: {exc}")

    return best_seq
