assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max (a\!*a)}{(\sum_i a_i)^2}
\]

for a non‑negative vector `a`.  
If the vector is kept on the simplex  

\[
\sum_i a_i=\sqrt{2n},
\]

the denominator is constant (`2 n`) and the objective reduces to the
peak of the self‑convolution, i.e. `max(a*a)`.  
The programme therefore works **directly on the simplex** and repeatedly
tries to lower the convolution peak.

The search consists of three stages:

1. **Fast exploration** – many short runs with a variety of analytic
   initial shapes (uniform, one‑sided geometric, half‑circle, cosine,
   …).  The best candidate found is kept as a seed for the next phase.
2. **Intensive optimisation** – a three‑phase optimiser
   * **Adam on a smooth approximation** of the max (soft‑max),
   * **sub‑gradient descent** that uses all currently active constraints,
   * **cheap local swaps** that fine‑tune the solution.
   The time slice for a single run is split 60 %/30 %/10 % among the
   three phases.
3. **Final annealing** – a long simulated‑annealing‑style hill‑climb
   (pairwise mass swaps) until the global budget expires.

All updates are projected onto the simplex to keep the sum fixed.
Symmetry is *optional* – the best constructions are usually non‑symmetric,
but a small fraction of runs enforce exact mirror symmetry to diversify
the search.

The whole routine is time‑aware; it stops a few seconds before the
external evaluator would time‑out, prints progress information and finally
returns the best sequence found.

```python
"""
Propose a step‑function (non‑negative vector) with a minimal
self‑convolution peak.  The algorithm consists of a fast exploration
phase (many short runs with diverse analytic shapes) followed by a
deep optimisation phase (Adam on a soft‑max approximation, sub‑gradient
descent on the true max, and a cheap local swap search).  Finally a
simulated‑annealing hill‑climb refines the best candidate until the
global budget (≈1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution a * a (length = 2*n-1) – FFT implementation
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of weighted sum Σ w_k (a*a)_k w.r.t. a.
# w must be a probability vector (len = 2*n-1)
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # convolution of a with reversed weights
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) probabilities over the convolution entries
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Optional symmetry enforcement
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation – up‑sample an existing sequence to a new length
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Rich family of analytic initial shapes
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n whose ℓ¹‑norm equals √(2·n).
    A mixture of analytic families (uniform, one‑sided geometric, cosine,
    half‑circle, triangular, random convex combinations, up‑sampling of a
    previous best) is used to provide diversity.
    """
    S = np.sqrt(2.0 * n)               # required ℓ¹‑norm

    typ = rng.integers(0, 8)           # 0 … 7 inclusive

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        # up‑sample a known good shape and add tiny noise
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # uniform Dirichlet‑like
        a = rng.random(n)

    elif typ == 2:
        # one‑sided geometric decay (large mass on the left)
        rate = rng.uniform(0.9, 0.9999)
        a = rate ** np.arange(n)

    elif typ == 3:
        # one‑sided geometric decay from the right side
        rate = rng.uniform(0.9, 0.9999)
        a = (rate ** np.arange(n))[::-1]

    elif typ == 4:
        # half‑circle (√(1‑x²) shape) centred in the middle
        centre = (n - 1) / 2.0
        x = (np.arange(n) - centre) / centre
        a = np.sqrt(np.maximum(0.0, 1.0 - x * x))

    elif typ == 5:
        # cosine bump – smooth, non‑negative
        centre = (n - 1) / 2.0
        x = (np.arange(n) - centre) / centre
        a = 0.5 * (np.cos(np.pi * x) + 1.0)

    elif typ == 6:
        # descending triangle (linear)
        a = np.linspace(1.0, 0.01, n)

    else:
        # random convex combination of several bases
        cand = []
        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9999)
        cand.append(rate ** np.arange(n))              # left‑geometric
        cand.append((rate ** np.arange(n))[::-1])      # right‑geometric
        centre = (n - 1) / 2.0
        x = (np.arange(n) - centre) / centre
        cand.append(np.sqrt(np.maximum(0.0, 1.0 - x * x)))  # half‑circle
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalise to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth (soft‑max) approximation of the peak
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 0.5
    alpha_mul = 1.02

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient of the smooth approximation
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional random jitter to escape plateaus
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.08 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.12:
                    a = cand

        # temperature & learning‑rate schedules
        alpha = min(alpha * alpha_mul, 45.0)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *true* maximum convolution entry
# (uses all currently active constraints)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        max_val = conv.max()

        # active set – entries within a tiny relative tolerance of the max
        eps_rel = 1e-9
        mask = (conv >= max_val - eps_rel * max_val).astype(float)
        mask_sum = mask.sum()
        if mask_sum == 0.0:
            # fall back to single‑max weight
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask / mask_sum

        # gradient of the (weighted) sum of the active entries
        grad = _conv_with_weights(a, w)

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing flavour)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = a.copy()

        # early stop if stagnating for many iterations
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap hill‑climbing that swaps tiny masses between random positions
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local swaps)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # 60 % Adam, 30 % sub‑gradient, 10 % local swaps
    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    # final refinement if any time left
    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # pick the best among the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point – time‑aware exploration + deep refinement
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the external evaluation
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0
    last_improve = start_time

    # ---------------------------------------------------------------
    # Phase 1 – cheap exploration (≈20 % of total budget)
    # ---------------------------------------------------------------
    explore_deadline = start_time + 0.20 * budget_s
    while time.time() < explore_deadline:
        attempts += 1
        n = int(rng.integers(2000, 8000))
        a0 = _initial_sequence(None, n, rng)
        slice_dead = min(time.time() + 2.0, explore_deadline)
        a_opt, _ = _optimize_adam(a0, slice_dead, rng, enforce_sym=False)
        sc = np.max(_full_convolution(a_opt))
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            best_n = n
            last_improve = time.time()
            print(f"[explore] attempt={attempts} n={n} new best={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 2 – intensive refinement on the current best
    # ---------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # slightly perturb the length of the current best
        scale = rng.uniform(0.95, 1.05)
        n = max(800, int(best_n * scale))

        # generate a fresh start from the current best (up‑sampling / jitter)
        a0 = _initial_sequence(best_seq, n, rng)

        # decide randomly whether to enforce exact symmetry for this run
        enforce_sym = rng.random() < 0.30

        # allocate a time slice (≈ 15–25 s) but never exceed the global deadline
        slice_deadline = min(time.time() + rng.uniform(15.0, 25.0), deadline)

        a_opt, sc = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            best_n = n
            last_improve = time.time()
            print(f"[refine] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_score:.6f}")

        # if we have already beaten a comfortable target we may stop early
        if best_score <= 1.503:
            break

        # refresh the seed if we get stuck for a while
        if time.time() - last_improve > 45.0:
            best_seq = None
            best_score = np.inf
            best_n = None
            print("[info] stagnation – discarding current seed")

    # ---------------------------------------------------------------
    # Phase 3 – final simulated‑annealing hill‑climb (uses leftover time)
    # ---------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_final, sc_final = _local_swap_search(best_seq, deadline, rng,
                                               enforce_sym=False)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain Python list) and expose it
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]                     # fallback – never happens in practice
    else:
        final_seq = best_seq.tolist()

    # expose the best construction for possible up‑sampling in the next run
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    # optional external verification (useful for debugging)
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal_best={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    return final_seq
```
