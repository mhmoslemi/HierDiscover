"""
Simulated‑annealing / targeted local search.
Starts from a known construction (if any) or from a random decreasing vector,
projects onto the L1‑simplex Σa = √(2·n) and repeatedly moves a small amount of
mass between entries.  Moves are chosen either at random or deliberately to
reduce the worst self‑convolution entry.  The temperature is lowered
exponentially; occasional restarts with jitter avoid being trapped in a local
minimum.  The algorithm runs until the time budget (≈1000 s) is exhausted and
returns the best sequence found.
"""
import time, math
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a) to new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """Search for a non‑negative step function that minimises ``evaluate_sequence``.
    
    The method is a simulated‑annealing style local search that repeatedly
    transfers a small amount of mass between two coordinates (or swaps two
    coordinates).  Half of the moves are *targeted*: they look at the shift
    giving the current worst convolution entry and move mass away from the pair
    of indices that contributes most to that entry.  The temperature is
    exponentially decreased from 0.5 to 1e‑4; if the search stagnates for a
    while it is re‑started from the best point with a tiny jitter.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # Choose problem size.  Larger n gives a tighter theoretical lower bound.
    # 4000 is a good compromise between flexibility and speed.
    n = 4000
    S = math.sqrt(2.0 * n)                 # desired L1 norm

    # ------------------------------------------------------------------
    # Initial vector: take the known construction if it exists,
    # otherwise a random decreasing vector.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            a0 = np.asarray(height_sequence_1, dtype=float)
            a = _upsample(a0, n)
        except Exception:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a, S)

    # ------------------------------------------------------------------
    # Keep the current state and the best one seen so far.
    # ------------------------------------------------------------------
    cur_a = a.copy()
    cur_score = np.max(_full_convolution(cur_a))
    best_a = cur_a.copy()
    best_score = cur_score

    last_improve = time.time()

    # ------------------------------------------------------------------
    # Main simulated‑annealing loop.
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # Linear interpolation of temperature (exponential decay).
        elapsed = time.time() - start_time
        total = deadline - start_time
        # T decays from 0.5 to 1e‑4.
        T = 0.5 * ((1e-4) / 0.5) ** (elapsed / total)

        # ------------------------------
        # Choose a move.
        # ------------------------------
        move_type = rng.random()
        if move_type < 0.3:
            # Targeted mass transfer: look at the worst convolution shift.
            conv = _full_convolution(cur_a)
            k_max = int(np.argmax(conv))

            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            i_vals = np.arange(i_low, i_high + 1, dtype=int)
            j_vals = k_max - i_vals

            # contribution of each pair (i, j) to conv[k_max]
            contrib = cur_a[i_vals] * cur_a[j_vals]
            idx = i_vals[np.argmax(contrib)]
            partner = k_max - idx

            # move a small fraction of the larger partner to a random location.
            src = idx if cur_a[idx] >= cur_a[partner] else partner
            if cur_a[src] <= 1e-12:
                continue
            delta = cur_a[src] * rng.random() * 0.03   # up to 3 % of src
            dest = rng.integers(0, n)
            while dest == src:
                dest = rng.integers(0, n)

            cand = cur_a.copy()
            cand[src] -= delta
            cand[dest] += delta
            # numerical drift – project back onto the simplex.
            cand = _project_simplex(cand, S)

        elif move_type < 0.65:
            # Simple swap of two coordinates.
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        else:
            # Random transfer of a small amount of mass.
            i = rng.integers(0, n)
            if cur_a[i] <= 0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cur_a[i] * rng.random() * 0.04   # up to 4 % of source
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand, S)

        # ------------------------------
        # Evaluate the candidate.
        # ------------------------------
        cand_score = np.max(_full_convolution(cand))

        # ------------------------------
        # Acceptance test (Metropolis).
        # ------------------------------
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / T):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                last_improve = time.time()
                print(f"[iter {it}] new best score {best_score:.6f}")

        # ------------------------------
        # Restart if stagnation lasts >30 s.
        # ------------------------------
        if time.time() - last_improve > 30.0:
            # jitter the best point and continue.
            cur_a = best_a.copy()
            jitter = rng.normal(scale=best_a.mean() * 5e-4 + 1e-12, size=n)
            cur_a = np.clip(cur_a + jitter, 0.0, None)
            cur_a = _project_simplex(cur_a, S)
            cur_score = np.max(_full_convolution(cur_a))
            last_improve = time.time()
            print(f"[restart] jittered from best – cur_score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector we have.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq

"""
Simulated‑annealing local search for a non‑negative step function.

The search works on a vector a of length *n* (here 8000) that is kept on the
L¹‑simplex Σ a_i = √(2·n).  The objective of the competition is
    2·n·max(conv(a,a)) / (Σa)²  =  max(conv(a,a))
once the vector is normalised, so we simply minimise the maximum entry of the
self‑convolution.

 * A fraction of the moves are *targeted*: the shift that currently realises
   the worst convolution value is identified and a small amount of mass is
   taken away from the pair of indices that contributes most to that shift.
 * The remaining moves are cheap random swaps or random mass transfers.
 * An exponential temperature schedule (0.5 → 1e‑4) yields a Metropolis
   acceptance rule, allowing occasional uphill moves early on.
 * If no improvement is seen for 30 s the current state is re‑initialised
   by jittering the best solution found so far.
 * The algorithm runs until the time budget (≈ 1000 s) is exhausted and the
   best vector is returned as a plain Python list.
"""

import time
import math
import numpy as np


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x ≥ 0, Σ x = s}.
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
