def _optimize_expgd(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = a.shape[0]; S = sqrt(2n). Compute initial best.

    best_a = a.copy(); best_score = np.max(_full_convolution(a))

    # hyperparameters
    alpha = 0.5
    alpha_mul = 1.03
    max_alpha = 300.0

    eta = 0.2  # initial learning rate for multiplicative update
    eta_mul = 0.9997  # decay factor

    # iteration loop
    it = 0
    while time.time() < deadline:
        it += 1

        # compute conv (full) - we need conv for computing max and weights
        conv = _full_convolution(a)

        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # compute softmax weights with temperature alpha
        w = _smooth_weights(conv, alpha)

        # gradient of smooth objective (weighted sum)
        grad = _conv_with_weights(a, w)  # shape n

        # multiplicative update: a = a * exp(-eta * grad)
        # Clip grad magnitude to avoid overflow
        # Use scaling: compute exp_factor = np.exp(-eta * grad) maybe large; we can bound:
        exp_factor = np.exp(-eta * grad)
        a = a * exp_factor

        # enforce symmetry if needed
        if enforce_sym:
            a = _symmetrize(a)

        # normalize to simplex sum S
        sum_a = a.sum()
        if sum_a == 0:
            # reinitialize
            a = np.full(n, 1.0)
        a = a * (S / sum_a)

        # temperature and learning rate schedule
        alpha = min(alpha * alpha_mul, max_alpha)
        eta *= eta_mul

        # occasional jitter: random swap small mass
        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.02 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand * (S / cand.sum())
                cand_score = np.max(_full_convolution(cand))
                if cand_score < best_score or rng.random() < 0.1:
                    a = cand

    return best_a, best_score

def _optimize_subgrad_full(a0, deadline, rng, enforce_sym=False):
    a = a0.copy()
    n = len(a); S = sqrt(2n)
    best_a= a.copy(); best_score = np.max(_full_convolution(a))

    eta0 = 0.1
    it=0
    while time.time() < deadline:
        it+=1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        # compute full gradient for that k_max:
        # gradient = a + reversed shifted a centered at k_max
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high+1)
        jdx = k_max - idx
        g[idx] = a[jdx] + a[idx]  # both terms
        # step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)  # ensures positivity and sum
        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # occasional jitter

                    a = cand
                    cur_max = cand_score
        # early stopping maybe
    return best_a, best_score

"""
Algorithm: multi‑phase continuous optimisation on the simplex.

*   Randomly initialise a non‑negative vector a of length n with ℓ¹‑norm
    √(2n).  Families include uniform, geometric, edge‑heavy reciprocal,
    triangular and U‑shapes (high at the ends, low in the centre).

*   Phase 1 – exponentiated‑gradient descent on a smooth approximation of
    the max‑convolution (log‑sum‑exp).  The temperature α is slowly
    increased (up to ≈300) so the surrogate converges to the true max.
    Updates are multiplicative (entropy‑mirror) and a Euclidean simplex
    projection (normalisation) is performed each step.

*   Phase 2 – sub‑gradient descent on the *exact* L∞‑objective.
    The full sub‑gradient of the worst convolution entry
    (∂(a*a)ₖ/∂aᵢ = aₖ₋ᵢ + aᵢ) is used, with a decaying step‑size and
    projection onto the simplex.

*   Phase 3 – cheap random mass‑swap hill‑climbing for fine‑tuning.

The optimisation is repeated for many different lengths (≈2 k–25 k)
within the allocated 1000 s budget, keeping the best sequence seen.
The final candidate is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution a * a (length 2*n-1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2*n-1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of weighted sum of convolution entries:  Σ_k w_k (a*a)_k
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a*a)_k.
    w must be a (2*n-1)-vector of weights that sum to 1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n - 2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) weights for a vector of values
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force v[i] == v[-1-i]."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Linear interpolation (upsampling) of a vector to a new length
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Diverse initialisers
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None, n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative vector of length n with ℓ¹‑norm √(2n)."""
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 9)   # 0‑8

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)

    elif typ == 1:
        a = rng.random(n)                     # uniform Dirichlet(1)

    elif typ == 2:
        # geometric decay from both ends
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) / 2.0

    elif typ == 3:
        # edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 4:
        # triangular (peak centre)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    elif typ == 5:
        # U‑shape (high at the edges, low in the centre)
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        power = rng.uniform(0.5, 2.0)
        a = (dist + 1) ** power

    elif typ == 6:
        # cosine‑bell, but we invert it to be higher at ends
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 - np.cos(x))
        a = np.clip(a, 0.0, None)

    elif typ == 7:
        # random convex combination of a few bases (as before)
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    else:
        # pure Dirichlet with small concentration -> spiky ends
        concentration = rng.uniform(0.05, 0.2)
        a = rng.gamma(concentration, 1.0, size=n)

    # Normalise to required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    a = np.clip(a, 0.0, None)
    return a


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent on a smoothed max.
# ----------------------------------------------------------------------
def _optimize_expgd(a0: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # hyper‑parameters
    alpha = 0.5               # temperature for soft‑max
    alpha_mul = 1.025
    max_alpha = 300.0

    eta = 0.20                # step size for multiplicative update
    eta_mul = 0.9997

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights (smooth approximation of the max)
        w = _smooth_weights(conv, alpha)

        # gradient of the smooth objective Σ w_k (a*a)_k
        grad = _conv_with_weights(a, w)

        # multiplicative (exponentiated) update – keep positivity
        # clip gradient to avoid overflow in exp
        grad_clipped = np.clip(grad, -50.0, 50.0)
        a = a * np.exp(-eta * grad_clipped)

        if enforce_sym:
            a = _symmetrize(a)

        # renormalise onto the simplex Σ a = S
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, S / n)
        else:
            a = a * (S / sum_a)

        # schedule
        alpha = min(alpha * alpha_mul, max_alpha)
        eta *= eta_mul

        # occasional random jitter (helps escape flat spots)
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand * (S / cand.sum())
                cand_score = np.max(_full_convolution(cand))
                if cand_score < best_score or rng.random() < 0.13:
                    a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – sub‑gradient descent on the true L∞‑objective.
# Uses the *full* sub‑gradient of the max convolution entry.
# ----------------------------------------------------------------------
def _optimize_subgrad_full(a0: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator,
                           enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.12
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # build the full gradient for the active shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1)
        jdx = k_max - idx
        g[idx] = a[jdx] + a[idx]   # a_{k‑i} + a_i

        eta = eta0 / np.sqrt(it)
        a = a - eta * g

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < best_score or rng.random() < 0.18:
                    a = cand
                    best_score = cand_score
                    best_a = a.copy()

        # early exit if stuck for a long while
        if it % 7000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – cheap random swap hill‑climbing (fine tuning)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.018 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser: 60 % exp‑GD, 30 % sub‑grad, 10 % local swaps
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # Phase 1 – exponentiated gradient
    t1 = now + 0.60 * remaining
    a1, sc1 = _optimize_expgd(a0, t1, rng, enforce_sym)

    # Phase 2 – sub‑gradient (full)
    t2 = min(time.time() + 0.30 * (deadline - time.time()), deadline)
    a2, sc2 = _optimize_subgrad_full(a1, t2, rng, enforce_sym)

    # Phase 3 – local random swaps
    if time.time() < deadline:
        a3, sc3 = _local_swap_search(a2, deadline, rng, enforce_sym)
    else:
        a3, sc3 = a2, sc2

    # pick the best of the three
    candidates = [(a1, sc1), (a2, sc2), (a3, sc3)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score
