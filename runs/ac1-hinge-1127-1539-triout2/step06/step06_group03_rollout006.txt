"""
Hybrid search for a non‑negative step‑function that minimises
evaluate_sequence (i.e. the maximal entry of a*a with the L¹‑norm fixed).

The routine combines three complementary ideas:

1. **Parametric seeding** – many families (power‑law, exponential,
   truncated linear) are sampled for several lengths (6000‑10000).  A few
   simple permutation heuristics are applied and the best few (elite)
   candidates are kept.

2. **Max‑subgradient descent** – instead of the smooth log‑sum‑exp
   surrogate, we directly descend the true max‑objective.  At each
   iteration the index k where the convolution attains its maximum is
   found, the exact sub‑gradient   ∇a b_k = 2·a shifted by k is computed,
   a step is taken and the vector is projected back onto the positive
   simplex Σa = √(2 n).

3. **Refinement** – a short projected gradient‑descent (smooth surrogate)
   step followed by a permutation‑annealing phase further improves the
   ordering.  Finally a tiny random mass‑exchange hill‑climber is run.

All steps respect the global 1000 s deadline and the final sequence is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _convolve_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Normalisation utilities
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have L¹ sum = target."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Permutation heuristics
# ----------------------------------------------------------------------
def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    n = base.size
    asc = np.sort(base)                     # smallest → largest
    desc = asc[::-1]                        # largest → smallest
    out = [asc, desc]

    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # step‑permutations with coprime strides
    for step in (2, 3, 5, 7, 11, 13, 17, 19, 23):
        if np.gcd(step, n) != 1:
            continue
        perm = np.empty_like(base)
        pos = 0
        for i in range(n):
            perm[pos] = asc[i]
            pos = (pos + step) % n
        out.append(perm)

    # a few random shuffles
    for _ in range(7):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out


# ----------------------------------------------------------------------
# Smooth surrogate gradient descent (re‑used after sub‑gradient)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """(g, b) where b = a*a and g = ∇_a (1/β·log Σ exp(β·b))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta
        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improve the ordering of a fixed multiset by random swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[i], a[j] = a[j], a[i]   # revert

        T *= cooling

        # occasional tiny global jitter
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask].copy()
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
    return best


# ----------------------------------------------------------------------
# Direct sub‑gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected sub‑gradient descent for  max_k (a*a)[k]."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    step0 = target_sum * 0.2            # baseline step magnitude
    rho = 0.9
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        b = _convolve_fft(a)
        idx_max = int(np.argmax(b))
        k0 = idx_max - (n - 1)          # shift where max occurs

        # exact sub‑gradient of b_{k0}:  g_j = 2*a_{k0 - j} (if in range)
        idx = k0 - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # momentum
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm == 0.0:
            break

        # diminishing step size
        step = step0 / (1.0 + it * 0.03) * (target_sum / (norm + 1e-12))
        a -= step * momentum

        # enforce constraints
        np.maximum(a, 0.0, out=a)
        a = _project_to_simplex(a, target_sum)

        it += 1
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises evaluate_sequence().
    The method combines parametric seeding, max‑sub‑gradient descent,
    smooth‑gradient refinement and permutation annealing.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf

    def _record(seq: np.ndarray):
        nonlocal best_seq, best_obj
        obj = _max_convolution_fft(seq)
        if obj < best_obj:
            best_obj = obj
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST  obj={best_obj:.6f}  len={seq.size}")

    # ------------------------------------------------------------------
    # 0.  Seed from previously discovered sequence if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.array(globals()["height_sequence_1"], dtype=np.float64)
            n0 = base.size
            target = np.sqrt(2.0 * n0)
            base = _project_to_simplex(np.maximum(base, 0.0), target)
            _record(base)
            elite = [(best_obj, base.copy())]
        except Exception:
            elite = []
    else:
        elite = []

    # ------------------------------------------------------------------
    # 1.  Coarse parametric sweep (few lengths, aggressive cutoff)
    # ------------------------------------------------------------------
    lengths = [6000, 8000, 10000]
    p_grid = np.linspace(0.45, 0.62, 8)
    shift_grid = np.linspace(0.2, 3.0, 6)

    rate_grid = np.linspace(0.0005, 0.0045, 8)
    exp_shift_grid = np.linspace(0.0, 2.0, 5)

    slope_grid = np.linspace(0.0001, 0.0045, 6)
    intercept_grid = np.linspace(0.5, 3.0, 5)

    for n in lengths:
        if time.time() >= deadline:
            break
        tgt = np.sqrt(2.0 * n)

        # ---- power law -------------------------------------------------
        for p in p_grid:
            for sh in shift_grid:
                base = _powerlaw_sequence(n, p, sh)
                for cand in _candidate_orderings(base, rng):
                    _record(cand)
                    elite.append((_max_convolution_fft(cand), cand.copy()))
        # ---- exponential -----------------------------------------------
        for rate in rate_grid:
            for sh in exp_shift_grid:
                base = _exponential_sequence(n, rate, sh)
                for cand in _candidate_orderings(base, rng):
                    _record(cand)
                    elite.append((_max_convolution_fft(cand), cand.copy()))
        # ---- linear ----------------------------------------------------
        for slope in slope_grid:
            for intercept in intercept_grid:
                base = _linear_sequence(n, slope, intercept)
                for cand in _candidate_orderings(base, rng):
                    _record(cand)
                    elite.append((_max_convolution_fft(cand), cand.copy()))

        # ---- quick random sampling --------------------------------------
        rnd_end = min(deadline, time.time() + 2.0)
        while time.time() < rnd_end:
            rnd = rng.random(n)
            rnd = _normalize_to_sum(rnd, tgt)
            _record(rnd)
            elite.append((_max_convolution_fft(rnd), rnd.copy()))

    # keep only the top few elite members
    elite.sort(key=lambda x: x[0])
    elite = elite[:10]

    # ------------------------------------------------------------------
    # 2.  Refine elite members (sub‑gradient → smooth gradient → anneal)
    # ------------------------------------------------------------------
    for _score, seed_seq in elite:
        if time.time() >= deadline:
            break
        n = seed_seq.size

        # sub‑gradient descent on the true max
        sub_dead = min(deadline, time.time() + 18.0)
        refined = _subgradient_descent_max(seed_seq, sub_dead)

        # short smooth‑gradient pass (helps escape plateaus)
        gd_dead = min(deadline, time.time() + 12.0)
        refined = _gradient_descent_opt(refined, gd_dead)

        # permutation annealing for the remaining budget slice
        refined = _permutation_annealing(refined, deadline)

        _record(refined)
        elite.append((_max_convolution_fft(refined), refined.copy()))
        elite.sort(key=lambda x: x[0])
        elite = elite[:10]

    # ------------------------------------------------------------------
    # 3.  Final light hill‑climbing on the current best sequence
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
        n = a.size
        while time.time() < deadline - 5.0:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # transfer a random fraction (up to 10 %) of mass from i to j
            delta = min(a[i], a[i] * 0.1) * (rng.random() - 0.5) * 2.0
            if delta == 0.0:
                continue
            new_a = a.copy()
            new_a[i] = max(0.0, a[i] - delta)
            new_a[j] = max(0.0, a[j] + delta)   # sum unchanged
            new_obj = _max_convolution_fft(new_a)
            cur_obj = _max_convolution_fft(a)
            if new_obj < cur_obj or rng.random() < 0.001:
                a = new_a
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] NEW BEST via mass‑exchange obj={best_obj:.6f}")

        best_seq = best_seq.tolist()

    # ------------------------------------------------------------------
    # 4.  Final verification using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}  length = {len(best_seq)}")
    return best_seq
