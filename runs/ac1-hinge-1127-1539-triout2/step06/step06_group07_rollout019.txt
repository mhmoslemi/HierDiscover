def solve_lp_max_sum(a, M):
    n = len(a)

    return res.x

"""
Simple LP‑based Frank‑Wolfe optimizer for min‑max self‑convolution.

The objective  max(conv(a,a))  is convex in the non‑negative simplex
{ a ≥ 0, Σa = √(2·n) }.  For a current point a we compute
M = max(conv(a,a)).  The linear sub‑problem

    maximize   Σ g_i
    subject to conv(a,g) ≤ M      (elementwise)
               g_i ≥ 0

has a closed‑form linear program.  Its solution g is “more spread’’ than a
and moving a toward g reduces the worst convolution entry.
We repeatedly solve this LP (using SciPy’s HiGHS solver), take a line‑search
step a ← (1‑t)a + t·g and keep the best vector seen.
When the time budget is nearly exhausted we finish with a short
gradient‑based refinement (Adam on a smooth max‑approximation).

The routine works for any n up to a few thousand within the 1000 s budget.
"""

import time
import numpy as np

# optional SciPy import – the LP step is much faster if available
try:

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution utilities (full linear convolution)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    # use FFT – fast for the sizes we use
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Build the linear constraint matrix for conv(a,g) ≤ M
# ----------------------------------------------------------------------


def _make_conv_matrix(a: np.ndarray):
    """
    Build a sparse matrix A with shape (2n‑1, n) such that
        A @ g  ==  np.convolve(a, g, mode='full')
    for any g ≥ 0.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    # each column i is the vector a placed starting at row i
    A = _sp.lil_matrix((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a  # broadcast – fills exactly n rows
    return A.tocsr()


# ----------------------------------------------------------------------
# Linear‑program sub‑problem: maximise Σ g_i  subject to a*g ≤ M
# ----------------------------------------------------------------------


def _lp_max_sum(a: np.ndarray, M: float):
    """Solve  max Σ g_i  s.t. conv(a,g) ≤ M , g ≥ 0 .  Returns g or None."""
    if not _HAVE_SCIPY:                     # pragma: no cover
        return None
    n = a.shape[0]
    A = _make_conv_matrix(a)               # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # linprog minimises; -Σg ⇒ maximise Σg
    bounds = [(0.0, None)] * n
    # HiGHS is fast for dense‐ish problems and accepts sparse matrices
    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Simple line‑search on the convex combination a ← (1‑t)a + t·g
# ----------------------------------------------------------------------


def _best_step(a, g, cur_M, deadline):
    """Return (a_new, M_new, t) where t gives the smallest M_new ≤ cur_M."""
    # try a few geometrically decreasing step sizes
    for t in (1.0, 0.5, 0.25, 0.125, 0.0625):
        a_new = (1.0 - t) * a + t * g
        M_new = np.max(_full_convolution(a_new))
        if M_new < cur_M * (1 - 1e-12):
            return a_new, M_new, t
        if time.time() >= deadline:          # safety
            break
    return a, cur_M, 0.0


# ----------------------------------------------------------------------
# Global optimiser (LP → gradient finetune)
# ----------------------------------------------------------------------


def _optimise_lp_then_gradient(a0, deadline, rng):
    """
    Run the LP‑based Frank‑Wolfe steps until the bound stops improving,
    then finish with a few hundred Adam iterations on a smooth max.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # ----- 1️⃣  LP Frank‑Wolfe phase -----
    stagnation = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a)
        M = conv.max()
        if M < best_score:
            best_score = M
            best_a = a.copy()

        g = _lp_max_sum(a, M)
        if g is None:
            break

        # rescale g to the required ℓ¹ norm (the LP has no Σg constraint)
        sum_g = g.sum()
        if sum_g <= 0:
            break
        scale = S / sum_g
        g_scaled = g * scale

        a, M_new, t = _best_step(a, g_scaled, M, deadline)
        if t == 0.0:                     # no progress
            stagnation += 1
            if stagnation >= 4:
                break
        else:
            stagnation = 0
        # occasionally inject a tiny random jitter to avoid being stuck
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

    # ----- 2️⃣  Short smooth‑max Adam refinement (optional) -----
    # if a little time is left we do a few Adam steps, re‑using code from the
    # previous solution (soft‑max weights).  This often shaves a few 1e‑3.
    if time.time() < deadline - 1.0:
        a, _ = _adam_refine(a, deadline, rng)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam on a smooth approximation of max(conv) (copied from the prior code)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    return exp_shifted / exp_shifted.sum()


def _adam_refine(a0: np.ndarray, deadline: float, rng) -> tuple[np.ndarray, float]:
    """Very light Adam run on the smooth max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.3
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5
    alpha_max = 800.0

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        # gradient of the smooth max‑norm (same as in the previous code)
        grad = _grad_smooth_max(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a, S)

        cur = np.max(_full_convolution(a))
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        alpha = min(alpha * 1.07, alpha_max)

        # occasional tiny random swap to keep exploration alive
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

    return best_a, best_score


def _grad_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ w_i·conv_i  w.r.t. a (soft‑max weighted sum of conv entries).
    Derived from  g_j = 2 Σ_i w_{i+j} a_i .
    """
    n = a.shape[0]
    # we need the convolution of w (reversed) with a, multiplied by 2
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The search uses a Frank‑Wolfe style LP step
    (maximising the ℓ¹ norm under the current max‑convolution bound)
    followed by a brief Adam refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # a small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – larger sizes give a lower theoretical bound.
    # We start with 2000 (as the previous best) and, if time permits,
    # move to 4000 after a good solution for the smaller size is found.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for n in (2000, 4000):
        if time.time() >= deadline:
            break

        S = np.sqrt(2.0 * n)

        # initialise a – warm‑start from a previously found sequence if any
        if best_seq is not None:
            # up‑sample to the new size (linear interpolation)
            old = np.asarray(best_seq, dtype=float)
            x_old = np.arange(old.size, dtype=float)
            x_new = np.linspace(0, old.size - 1, n, dtype=float)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = rng.random(n)

        # project onto the ℓ¹‑simplex Σa = S
        a0 = _project_simplex(a0, S)

        # run the combined optimiser
        a_opt, sc_opt = _optimise_lp_then_gradient(a0, deadline, rng)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[stage] n={n} new global best score={best_score:.6f}")

        # if we have enough time, continue with the larger size
        # (the loop will automatically move on)

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple uniform distribution (guaranteed feasible)
        n = 2000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n, dtype=float)

    final_seq = best_seq.tolist()
    # external evaluation (the harness will also call evaluate_sequence)
    try:
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f} "
              f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    except Exception:                     # pragma: no cover
        print("[final] external evaluation failed")
    return final_seq
