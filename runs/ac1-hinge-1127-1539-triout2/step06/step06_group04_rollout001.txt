"""
Hybrid search for a non‑negative step function with small self‑convolution peak.

Algorithm
---------

1. **Initialisation** – start from the global `height_sequence_1` if it exists,
   otherwise from a random simplex‑projected vector.

2. **Length schedule** – gradually increase the number of coefficients.
   For each length `n` we:
   * up‑sample the current best vector,
   * run projected gradient descent on a smoothed max‑norm (log‑sum‑exp)
     for a time‑budgeted number of iterations,
   * (only for the smallest size) solve a linear program
     `max Σ g_i`  s.t.  `(f * g)_k ≤ M`  for all `k`, `g ≥ 0`,
     then mix the LP solution with the current vector,
   * apply a targeted hill‑climber that moves mass from indices that
     contribute most to the peak towards indices that contribute least,
     accepting occasional uphill moves (simulated‑annealing style).

3. **Final polishing** – with any remaining budget a last hill‑climb is run.

The search obeys the 1000 s wall‑clock budget, projects onto the ℓ¹‑simplex
(`Σ a_i = √(2 n)`) after every modification, and finally trims trailing zeros.
The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:  # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:  # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:  # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:  # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:  # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:  # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:  # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    Uses momentum, β‑annealing and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.13
    momentum = np.zeros_like(a)
    rho = 0.92

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Targeted hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that moves a small amount of mass from indices that
    heavily contribute to the current peak towards indices that contribute
    little.  Accepts occasional uphill moves (Metropolis rule).
    Returns best vector found and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contributions to the peak k_max
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # analytical update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < math.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)
            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max


# ----------------------------------------------------------------------
# Linear‑programming refinement (small n only)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, M: float, target_sum: float, deadline: float):
    """
    Solve  max Σ g_i   s.t.   (f * g)_k ≤ M  ∀k,   g_i ≥ 0.
    The solution is scaled to have ℓ¹‑norm = target_sum.
    Returns (g_scaled, success_flag).
    """
    n = f.size
    # Build sparse constraint matrix A (shape (2n-1, n))
    row = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    col = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    A = csr_matrix((data, (row, col)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)                     # maximize Σ g_i  ⇔  minimize -Σ g_i
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={"presolve": True, "time_limit": max(0.0, deadline - time.time())})
        if not res.success:
            return None, False
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            return None, False
        g_scaled = g * (target_sum / sum_g)
        return g_scaled, True
    except Exception as e:
        print(f"LP failed: {e}")
        return None, False


# ----------------------------------------------------------------------
# Mixing helper (simple line‑search over a few α)
# ----------------------------------------------------------------------


def _mix_and_choose(f: np.ndarray, g: np.ndarray, target_sum: float, deadline: float):
    """
    Try a few convex combinations (1‑α)·f + α·g (α∈{0.2,0.4,0.6,0.8}),
    keep the one with smallest maximal convolution.
    """
    best = f.copy()
    best_val = _max_convolution_fft(f)
    # both f and g already have the correct ℓ¹‑norm
    for alpha in (0.2, 0.4, 0.6, 0.8):
        cand = (1 - alpha) * f + alpha * g
        val = _max_convolution_fft(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
    return best, best_val

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the wall‑clock deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from an existing construction if available
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, math.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(best_seq)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start small, expand gradually.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 24000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise candidate for this length
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ------------------------------------------------------------------
        # Phase A – projected gradient descent (major share of remaining budget)
        # ------------------------------------------------------------------
        time_left = deadline - time.time()
        phaseA_dead = time.time() + min(120.0, 0.55 * time_left)
        a_gd = _gradient_descent_opt(a_init, phaseA_dead)
        a_gd = _project_to_simplex(a_gd, target_sum)
        score_gd = _max_convolution_fft(a_gd)
        if score_gd < best_score:
            best_score, best_seq = score_gd, a_gd.copy()
            print(f"[{time.time():.1f}] GD improvement (n={n}) score={score_gd:.6f}")

        # ------------------------------------------------------------------
        # Phase B – LP refinement (only for the smallest size)
        # ------------------------------------------------------------------
        if n <= 2000 and time.time() < deadline - 5.0:
            M = _max_convolution_fft(a_gd)
            g_lp, ok = _lp_refine(a_gd, M, target_sum, deadline)
            if ok:
                a_mix, mix_score = _mix_and_choose(a_gd, g_lp, target_sum, deadline)
                if mix_score < best_score:
                    best_score, best_seq = mix_score, a_mix.copy()
                    print(f"[{time.time():.1f}] LP+mix improvement (n={n}) score={mix_score:.6f}")

        # ------------------------------------------------------------------
        # Phase C – targeted hill climbing
        # ------------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a_gd)
            time_left = deadline - time.time()
            phaseB_dead = time.time() + min(80.0, 0.35 * time_left)
            a_hc, max_hc = _hill_climb_targeted(a_gd, b_full, phaseB_dead)
            if max_hc < best_score:
                best_score, best_seq = max_hc, a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_final, deadline)
        if max_final < best_score:
            best_score, best_seq = max_final, a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of length 1000
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
