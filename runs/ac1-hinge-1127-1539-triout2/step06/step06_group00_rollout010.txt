def _final_refine(a_start, deadline, rng):
    n = len(a_start)
    S = np.sqrt(2.0*n)
    # convert to logits p as before
    soft = np.maximum(a_start/S, 1e-12)
    soft /= soft.sum()
    p = np.log(soft)
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2  # maybe smaller
    it = 0
    best_a = a_start.copy()
    b = _full_convolution_fft(a_start)
    best_peak = b.max()
    while time.time() < deadline:
        it += 1
        # recover a
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # compute peak and gradient (hard)
        b = _full_convolution_fft(a)
        peak = b.max()
        idx = np.argmax(b)
        mask = np.zeros_like(b)
        mask[idx] = 1.0
        g = _mask_gradient(a, mask)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # compute gradient wrt logits
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p**2)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedule
        lr *= 0.99995
        # occasional jitter
        if it % 400 == 0:
            p += rng.normal(scale=0.005, size=n)

    return best_a, best_peak

"""
Adversarial step‑function optimizer – enhanced version

The optimizer searches a non‑negative vector `a` (length n) with ℓ₁‑norm
√(2·n) that minimises   maxₖ (a*a)(k).  The code consists of three
phases:

1. **Exploration** – many short Adam restarts from diverse initial
   families (uniform, Dirichlet, edge‑heavy power‑law, exponential).
   The best candidate is kept.

2. **Precise refinement** – a longer Adam run that uses the exact
   sub‑gradient of the hard maximum (τ → 0).  Adam operates on the
   log‑softmax parametrisation, so the ℓ₁‑norm constraint is satisfied
   automatically.

3. **Greedy hill‑climbing** – tiny random mass moves are accepted only
   if they strictly lower the convolution peak.  This often squeezes a
   few extra digits beyond what gradient descent alone can achieve.

All helper functions are defined at module level (no closures) and the
algorithm respects the 1000 s time budget.  The final sequence is stored
in the global variable ``height_sequence_1`` for reuse by later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _mask_gradient(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Gradient of <mask, a*a> with respect to a.
    mask must have length 2·n‑1.
    Returns 2 * (mask * a) convolved with reversed a, sliced to size n.
    """
    a_rev = a[::-1]
    conv_len = mask.shape[0] + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g


def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Return (grad, max_b) for the τ‑soft‑max approximation of max(b)
    where b = a*a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    g = _mask_gradient(a, mask)
    return g, max_b


def _hard_max_gradient(a: np.ndarray):
    """
    Exact sub‑gradient for the hard max (τ = 0).
    Returns (grad, max_b).
    """
    b = _full_convolution_fft(a)
    idx = int(np.argmax(b))
    mask = np.zeros_like(b)
    mask[idx] = 1.0
    g = _mask_gradient(a, mask)
    return g, b[idx]


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]

# Initial‑vector factories
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law edge‑heavy shape."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0  # distance to nearest edge (1‑based)

    p = rng.uniform(0.2, 1.0)          # exponent
    offset = rng.uniform(0.0, 5.0)     # denominator shift
    mix = rng.uniform(0.0, 1.0)        # mixing with uniform background

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _exp_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0

    p = rng.uniform(0.1, 2.0)          # decay rate
    offset = rng.uniform(0.0, 3.0)     # shift inside the exponent
    mix = rng.uniform(0.0, 1.0)        # mixing with uniform

    exp_part = np.exp(-p * (d + offset))
    base = np.ones(n)
    raw = mix * exp_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _initial_vector(n: int, base_seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    Occasionally warm‑starts from the global best (scaled).
    """
    S = np.sqrt(2.0 * n)

    # 15 % chance of scaling the previous global optimum
    if base_seq is not None and rng.random() < 0.15:
        a = _scale_sequence(base_seq, n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['uniform', 'dirichlet', 'edge', 'exp'])

    if style == 'uniform':
        a = np.full(n, S / n)
    elif style == 'dirichlet':
        a = rng.random(n)
        a = a / a.sum() * S
    elif style == 'edge':
        a = _edge_initial(n, rng)
    else:   # exponential family
        a = _exp_initial(n, rng)

    return a


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max version – good for exploration)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimises the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered and the *hard* peak value.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.35                                 # a little larger than before

    # start temperature – a few times the initial peak
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & (soft‑max) gradient -----
        g, peak = _softmax_gradient(a, tau)

        # keep the best hard peak seen so far
        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ----- chain rule to logits -----
        mu = np.dot(a, g) / S                 # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                 # shape (n,)

        # gradient clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995               # quicker tightening
        lr *= 0.9999                # very slow decay

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            p += rng.normal(scale=0.012, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Precise refinement (hard‑max gradient)
# ----------------------------------------------------------------------
def _final_refine(a_start: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam on the exact max (τ → 0).  Returns a refined vector and its peak.
    """
    n = a_start.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits from the start vector
    eps = 1e-12
    soft = np.maximum(a_start / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.22

    it = 0
    best_a = a_start.copy()
    best_peak = _full_convolution_fft(a_start).max()

    while time.time() < deadline:
        it += 1

        # ---- reconstruct a ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- exact gradient of the hard max ----
        g, peak = _hard_max_gradient(a)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits ----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)

        # optional clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 4.0 * S:
            grad_p = grad_p * (4.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules
        lr *= 0.99995

        # jitter every few hundred steps
        if it % 500 == 0:
            p += rng.normal(scale=0.003, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Greedy hill‑climbing (tiny random mass moves)
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                cur_score: float,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very simple stochastic local search that only accepts moves that
    strictly improve the hard peak.  Each move transfers a small
    fraction of mass from one index to another.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 1e-12:
            continue

        # move at most 5 % of the source entry
        max_delta = min(a[i], 0.05 * a[i])
        delta = rng.uniform(0.0, max_delta)
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _full_convolution_fft(a_new).max()
        if new_max < best_score - 1e-7:
            a = a_new
            best_seq = a_new.copy()
            best_score = new_max
            print(f"[hill] improved to {best_score:.6f}")

    return best_seq, best_score
