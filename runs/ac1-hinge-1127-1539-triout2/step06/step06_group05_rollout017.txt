"""
Improved step‑function search.

The algorithm works in two phases:

1. **Broad random sampling** – many analytic families (geometric,
   power‑law, exponential, mixed, sparse) are sampled for a wide range of
   lengths (up to 50 k).  Each candidate is projected onto the ℓ¹‑simplex
   Σa = √(2 n) and its true objective (the peak of the self‑convolution)
   is computed with an FFT‑based convolution.  The best few candidates are
   kept.

2. **Focused refinement** – the top candidates are fed to a three‑stage
   optimiser (smooth Adam → sub‑gradient → cheap local swaps) that works
   directly on the original objective.  The optimiser runs for a
   configurable slice of the remaining time budget.  The best final
   sequence is returned.

The search explores much larger problem sizes than the reference code
(e.g. n up to 50 000) and a richer family of shapes, which gives a
substantial chance to beat the previous bound of 1.511999.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution a * a (real FFT, length 2*n-1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Helper – exact mirror symmetry (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Shape generators (return raw (non‑normalised) vectors)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=float)


def _shape_geometric(n: int, rng: np.random.Generator) -> np.ndarray:
    r = rng.uniform(0.80, 0.9999)
    base = r ** np.arange(n, dtype=float)
    return base + base[::-1]


def _shape_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 4.5)
    idx = np.arange(1, n + 1, dtype=float)
    left = idx ** (-p)
    right = (n - np.arange(n) + 0) ** (-p)
    return left + right


def _shape_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    beta = rng.uniform(0.5, 5.0)
    base = np.exp(-beta * np.arange(n, dtype=float))
    return base + base[::-1]


def _shape_exp_power(n: int, rng: np.random.Generator) -> np.ndarray:
    beta = rng.uniform(0.1, 5.0)
    gamma = rng.uniform(0.5, 3.0)
    t = np.arange(n, dtype=float) / max(1, n - 1)
    base = np.exp(-beta * np.power(t, gamma))
    return base + base[::-1]


def _shape_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    # choose a support size m (between 0.5 % and 15 % of n)
    m = max(2, int(rng.integers(int(0.005 * n), max(3, int(0.15 * n) + 1))))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0
    return a


def _shape_piecewise_blocks(n: int, rng: np.random.Generator) -> np.ndarray:
    # Two identical edge blocks of length L (no overlap)
    L = max(1, int(rng.uniform(0.03, 0.12) * n))
    if 2 * L > n:                     # ensure blocks don't intersect
        L = n // 2
    a = np.zeros(n, dtype=float)
    a[:L] = 1.0
    a[-L:] = 1.0
    return a


def _shape_mixed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random convex combination of a few simple bases."""
    bases = []
    bases.append(_shape_uniform(n))
    bases.append(_shape_geometric(n, rng))
    bases.append(_shape_power(n, rng))
    bases.append(_shape_exponential(n, rng))
    bases.append(_shape_sparse(n, rng))
    w = rng.random(len(bases))
    w /= w.sum()
    a = sum(wi * bi for wi, bi in zip(w, bases))
    return a


# ----------------------------------------------------------------------
# Random initialisation wrapper
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative vector of length n with ℓ¹‑norm √(2 n)."""
    S = np.sqrt(2.0 * n)

    # occasionally up‑sample the previous best
    if base_seq is not None and rng.random() < 0.12:
        a = np.interp(np.linspace(0, 1, n),
                      np.linspace(0, 1, base_seq.size),
                      base_seq)
        a = np.clip(a, 0, None) + rng.normal(scale=0.001, size=n)
        a = np.clip(a, 0, None)
    else:
        typ = rng.integers(0, 7)       # 0‑6
        if typ == 0:
            a = _shape_uniform(n)
        elif typ == 1:
            a = _shape_geometric(n, rng)
        elif typ == 2:
            a = _shape_power(n, rng)
        elif typ == 3:
            a = _shape_exponential(n, rng)
        elif typ == 4:
            a = _shape_exp_power(n, rng)
        elif typ == 5:
            a = _shape_sparse(n, rng)
        else:
            a = _shape_piecewise_blocks(n, rng)

    # normalise to the required ℓ¹‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a * (S / total)
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities (used by Adam optimiser)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    return exp_shifted / exp_shifted.sum()


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the log‑sum‑exp smooth peak w.r.t. a."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed peak (log‑sum‑exp)
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.5                # start temperature for soft‑max
    alpha_mul = 1.025

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to jump out of plateaus
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.13:
                    a = cand

        alpha = min(alpha * alpha_mul, 50.0)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑entry of the convolution
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the maximum entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i_vals = np.arange(i_low, i_high + 1)
        j_vals = k_max - i_vals
        g[i_vals] = a[j_vals]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random swap (simulated‑annealing flavour)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.22:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # simple stagnation break
        if it % 9000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap local swap hill‑climbing
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score

def propose_candidate(seed: int = 12345,

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety buffer

    # reuse previous best if present
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_score = np.inf
    best_overall_n = None
    top_k = 8                                 # store a few elite candidates
    elite = []                                # list of (score, vector, n)
    attempts = 0
    last_improve = start_time

    # bounds for problem size
    min_n = 2000
    max_n = 50000

    while time.time() < deadline - 30.0:      # keep ~30 s for refinement
        attempts += 1

        # choose length – bias towards larger n but keep diversity
        if best_overall_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.90, 1.10)
            n = max(min_n, int(best_overall_n * scale))
        else:

        enforce_sym = rng.random() < 0.55      # many attempts keep symmetry

        a0 = _initial_sequence(base_seq, n, rng)

    return final_seq
