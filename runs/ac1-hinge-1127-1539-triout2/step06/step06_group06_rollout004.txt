"""
Subgradient+Simplex optimiser for the self‑convolution benchmark.

The objective (for a normalised sequence Σa_i = √(2·n)) is simply
    max_k (a * a)_k .
To minimise it we repeatedly:
  * evaluate the convolution by FFT,
  * locate the (few) indices attaining the current maximum,
  * compute a sub‑gradient of those max‑entries,
  * take a small step opposite to the gradient,
  * project back onto the ℓ¹‑simplex Σa_i = √(2·n), a_i ≥ 0.

The optimiser works on three increasing lengths (≈1500 → 3000 → 6000)
re‑using the best solution as a warm‑start for the next scale.
A few auxiliary tricks improve the search:
  – additive jitter and a tiny moving‑average smoothing every few
    iterations (helps escaping shallow plateaus),
  – an occasional random “mass‑transfer” move (a cheap simulated‑annealing
    style perturbation),
  – adaptive step‑size based on the gradient norm.

The routine returns a list of non‑negative heights that can be fed
directly to ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _grad_of_max_entry(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry (a*a)[k] w.r.t. a.
    For each j: ∂/∂a_j (a*a)[k] = a_{k-j} + a_j  (out‑of‑range indices give 0).
    """
    n = a.shape[0]
    g = a.copy()
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    # indices j in [lo, hi]
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    g[idx] += a[k - idx]
    return g


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    A small amount of smoothing / jitter is applied periodically.
    """
    a = a0.astype(np.float64, copy=True)
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # initialise on the simplex
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    it = 0

    # step‑size tunable constants
    base_step = target_sum * 0.005   # ≈0.5 % of total mass

    while time.time() < deadline:
        # ----- convolution & max entry -----
        a_fft = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(a_fft * a_fft, n=size_fft)[: 2 * n - 1]
        max_b = float(conv.max())
        # (for our normalisation the objective equals max_b)
        cur_val = max_b

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it:5d} best={best_val:.6f}")

        # ----- compute (averaged) sub‑gradient of the max entry -----
        # consider all indices whose value is within 0.1 % of the max
        eps = max_b * 1e-3
        idx_max = np.where(conv >= max_b - eps)[0]

        grad = np.zeros_like(a)

        # ----- adaptive step -----
        norm_g = np.linalg.norm(grad)
        if norm_g > 0:
            step = base_step / (norm_g + 1e-12)
            a -= step * grad

        # ----- keep feasibility -----
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # ----- occasional smoothing (helps flatten spikes) -----
        if it % 500 == 0 and it > 0:
            kernel = np.array([0.25, 0.5, 0.25])
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0.0)

        if it % 1500 == 0 and it > 0:

            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a += jitter

            a = _project_to_simplex(a, target_sum)

        it += 1

    # final check
    final_val = _score(a)
    if final_val < best_val:
        best_val = final_val

                                                dtype=np.float64), n)
            else:
                init = _power_law_sequence(n, 0.55, target_sum)

        # ----- optimise this scale -----
        seq_opt, score_opt = _subgradient_descent(init, stage_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] Scale n={n} improved best score={best_score:.6f}")

        cur_seq = seq_opt   # warm start for the next (larger) scale

    # ----- tidy up the final answer -----
    if best_seq is None:
        return []

    arr = np.array(best_seq, dtype=float)
    # drop trailing near‑zeros for a cleaner output
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ----- one final evaluation (mostly for logging) -----
    try:
        val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
