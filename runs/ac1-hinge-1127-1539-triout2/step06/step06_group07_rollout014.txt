"""
Hybrid optimiser for the self‑convolution max‑norm problem.

* Stage 1 – a few hundred Adam + sub‑gradient iterations give a decent
  starting point on a moderately large support (n≈4000).
* Stage 2 – a custom simulated‑annealing / mass‑transfer routine that
  works directly on the full convolution (incremental O(n) update)
  further reduces the peak.  The move consists of transferring a small
  amount ε from one coordinate to another; the resulting change of the
  convolution can be computed analytically, so each trial costs only a
  few vector operations.

The whole process respects the 1000 s budget, leaves a safety margin
and finally returns the best sequence found (as a plain Python list).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (a * a)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on smoothed max‑norm (identical to the previous code)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i ∝ exp(alpha·conv_i)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max objective: 2·(a * w_reversed)."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth max‑norm approximation."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.4
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5
    alpha_max = 1500.0

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a, S)

        # tiny random mass redistribution
        if rng.random() < 0.03:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        alpha = min(alpha * 1.05, alpha_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.03
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
    return best_a, best_score


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.7) -> tuple[np.ndarray, float]:
    """Adam for a fraction of the time, then sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))
    adam_deadline = now + fraction_adam * remaining
    a_s, sc_s = _optimize_adam(a0, adam_deadline, rng)
    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Simulated annealing / mass‑transfer optimiser
# ----------------------------------------------------------------------


def _anneal(a0: np.ndarray,
            deadline: float,
            rng: np.random.Generator,
            max_frac: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style optimisation.
    Moves a small amount ε from a source index to a destination index.
    The change in the full convolution can be updated analytically in O(n).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _full_convolution(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    # initial temperature – a fraction of the current max
    T0 = cur_max * 0.5 if cur_max > 0 else 1.0

    N = 2 * n - 1
    while time.time() < deadline:
        # linear cooling
        remaining = deadline - time.time()
        total = deadline - start
        T = T0 * remaining / total if total > 0 else 1e-12

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        src = i
        dst = j
        src_val = a[src]
        if src_val <= 0.0:
            continue
        eps = src_val * rng.random() * max_frac
        if eps <= 0.0:
            continue

        # build the delta of the convolution
        delta = np.zeros(N, dtype=conv.dtype)
        # shift contributions (factor 2 because each pair appears twice)
        delta[src: src + n] -= 2.0 * eps * a
        delta[dst: dst + n] += 2.0 * eps * a
        # quadratic (δ*δ) part
        delta[2 * src] += eps * eps
        delta[2 * dst] += eps * eps
        delta[src + dst] -= 2.0 * eps * eps

        new_conv = conv + delta
        new_max = new_conv.max()

        # Metropolis acceptance
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / (T + 1e-12)):
            # accept move
            a[src] -= eps
            a[dst] += eps
            conv = new_conv
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        # else reject – nothing to do
    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Hybrid optimiser – gradient phase (Adam + sub‑gradient) followed by
    a custom simulated‑annealing mass‑transfer phase.  Works on a single
    support size (n = 4000) which gives enough degrees of freedom to
    push the self‑convolution peak below the previously reported 1.513.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the external call
    deadline = start_time + budget_s - 5.0

    best_seq = None
    if 'height_sequence_1' in globals():
        try:
            best_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            best_seq = None

    # ------------------------------------------------------------------
    # Stage 1 – gradient optimiser on n = 4000
    # ------------------------------------------------------------------
    n = 4000
    S = np.sqrt(2.0 * n)

    if best_seq is not None:
        a0 = _upsample(best_seq, n)
    else:
        a0 = rng.random(n)

    # jitter + simplex projection
    jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
    a0 = np.clip(a0 + jitter, 0.0, None)
    if a0.sum() == 0.0:
        a0 = np.full(n, 1.0)
    a0 = a0 / a0.sum() * S

    # give about 35 % of the budget to the gradient phase
    grad_deadline = min(deadline, start_time + 0.35 * (deadline - start_time))
    a_opt, sc_opt = _optimize_combined_adam(a0, grad_deadline, rng,
                                            fraction_adam=0.78)

    # ------------------------------------------------------------------
    # Stage 2 – simulated annealing refinement (uses the remainder)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_best, sc_best = _anneal(a_opt, deadline, rng, max_frac=0.25)
        if sc_best < sc_opt:
            a_opt, sc_opt = a_best, sc_best

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    # final projection to the exact simplex (removes tiny drift)
    a_opt = _project_simplex(a_opt, np.sqrt(2.0 * a_opt.shape[0]))

    final_seq = a_opt.tolist()
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {sc_opt:.6f}) length = {len(final_seq)}")
    return final_seq
