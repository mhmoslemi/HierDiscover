def build_C_matrix(f):

    rows = 2*n-1
    data = []
    row = []
    col = []
    for j in range(n):
        # column j corresponds to f shifted by j: entries from k=j to k=j+n-1 have f_{k-j}
        start_k = j
        end_k = j + n   # exclusive
        # for k in range(start_k, end_k):
        #    coeff = f[k - j]
        # add entry at (k, j) with coeff
        # Just use slice assignment: data.extend(f.tolist())
        # row.extend(range(start_k, end_k))
        # col.extend([j]*n)
        data.extend(f.tolist())
        row.extend(range(start_k, end_k))
        col.extend([j] * n)
    # Build coo matrix:
    C = scipy.sparse.coo_matrix((data, (row, col)), shape=(2*n-1, n))

"""
Frank‑Wolfe + gradient polishing for the max‑convolution problem.

We minimise
        2·n·max(a*a) / (sum a)^2
which, after the standard ℓ¹‑normalisation Σa = √(2n), is simply the
maximum entry of the self‑convolution.  The algorithm consists of

1. Initialise a non‑negative sequence and project it onto the simplex.
2. Frank‑Wolfe style updates:
     – compute the current worst convolution value M,
     – solve a linear programme that maximises the total mass of a new
       sequence g under the linear constraints (f * g)_k ≤ M,
     – rescale g to the required ℓ¹‑norm,
     – convex‑combine f and g (trying a few mixing factors) and keep the
       combination that reduces the maximum convolution.
   If progress stalls we add a small random jitter or restart from a
   power‑law.
3. When time is limited, finish with a few seconds of projected gradient
   descent on a smooth soft‑max approximation of the max (temperature
   β is increased up to 4096).  This polishes the final candidate.
4. Strip trailing zeros and return the best sequence found.

All operations are performed with NumPy; the linear programme uses SciPy’s
‘highs’ interior‑point solver (dense matrices, which are only O(n²) for the
chosen moderate n ≈ 2000).  The whole search respects the 1000 s budget.
"""
import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (gradient, raw convolution) where gradient has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    pr_fft = np.fft.rfft(p, n=size)
    ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    Returns the best iterate found before the deadline.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} maxconv={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val

def _build_conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix C ∈ ℝ^{(2n‑1)×n} such that (f * g) = C·g.
    Row k contains the coefficients f_{k‑j} for all feasible j.
    """
    n = f.size
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _solve_lp_for_direction(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to   C·g ≤ M·1,   g ≥ 0,
    where C encodes the convolution (f * g).
    Returns the optimal g (may be None if the solver fails).
    """
    n = f.shape[0]
    C = _build_conv_matrix_dense(f)                # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)    # RHS vector
    c = -np.ones(n, dtype=np.float64)              # minimise –Σg_i  ⇒ maximise Σg_i
    bounds = [(0, None)] * n

    # Use the HiGHS interior‑point method (fast for moderate sizes)
    res = scipy.optimize.linprog(
        c, A_ub=C, b_ub=b, bounds=bounds, method="highs", options={"presolve": True}
    )
    if not res.success:
        return None
    return res.x

            cur_M = _max_convolution_fft(cur_f)
            no_improve += 1
            if no_improve >= max_no_improve:
                # hard restart with a fresh power‑law
                cur_f = _power_law_sequence(n, 0.55, target_sum)
                cur_f = _project_to_simplex(cur_f, target_sum)
                cur_M = _max_convolution_fft(cur_f)
                no_improve = 0
                print(f"[{time.time():.1f}] restart after {max_no_improve} idle iterations")

        iter_count += 1

    # ------------------------------------------------------------------
    # Final polishing with gradient descent (soft‑max temperature schedule)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + remaining * 0.6
        best_f, _ = _annealed_gradient_descent(best_f, gd_deadline)
        best_M = _max_convolution_fft(best_f)

    # ------------------------------------------------------------------
    # Clean‑up and return
    # ------------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    best_f = _project_to_simplex(best_f, target_sum)

    # strip trailing zeros (they do not affect the score)
    nz = np.where(best_f > 1e-12)[0]
    if nz.size:
        best_f = best_f[: nz[-1] + 1]
    else:
        best_f = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best maxconv≈{best_M:.6f}, length={len(best_f)}")
    return best_f.tolist()
