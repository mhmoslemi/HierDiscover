    n = len(f)
    S = np.sqrt(2.0 * n)
    a = f.copy()  # current distribution
    for it in range(n_iter):
        M = np.max(_full_convolution(a))
        # Build constraint matrix A_ub (size 2n-1 x n). Use maybe dense matrix due to small n.
        rows = []
        for k in range(2*n-1):
            row = np.zeros(n)
            # i range
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            # row[ lo: hi+1] = a[lo:hi+1]
            row[lo:hi+1] = a[lo:hi+1]
            rows.append(row)
        A_ub = np.vstack(rows)  # shape (2n-1,n)

        if not res.success:

        g = res.x
        # scale g to sum S
        sumg = g.sum()
        if sumg == 0:
            break
        g = g * (S / sumg)
        # mix

def _lp_refine(a0, steps, t, rng):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    for step in range(steps):
        M = np.max(_full_convolution(a))
        # Build A_ub
        # Option: use np.tile of a ??? but will need to construct matrix.
        rows = []
        for k in range(2*n-1):
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            row = np.zeros(n)
            row[lo:hi+1] = a[lo:hi+1]
            rows.append(row)
        A = np.vstack(rows)
        b = np.full(2*n-1, M)
        # Solve LP
        c = -np.ones(n)  # maximize sum(g)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        # scale to sum S
        sumg = g.sum()
        if sumg == 0:
            break
        g = g * (S / sumg)
        # mix
        a = (1 - t) * a + t * g
    # after steps, project onto simplex just in case
    a = _project_simplex(a, S)
    return a

Multiscale optimisation for the step‑function minimising the
self‑convolution peak.  The search proceeds in two stages:
1. a coarse problem (few hundred coefficients) is solved with the
   Adam → sub‑gradient pipeline from the reference implementation.
2. the obtained shape is up‑sampled to a larger length (≈10 k) and
   refined again.  Multiple random restarts are performed within the
   allotted budget and the best candidate is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# Helper functions ...
...

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function whose self‑convolution peak
    is as low as possible.  The function respects the 1000 s budget and
    returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # ------ Stage 1 : coarse optimisation (n ≈ 800) ----------
    n_coarse = int(rng.integers(700, 1000))
    a0 = _initial_sequence(None, n_coarse, rng)   # uniform/random start
    # we run the full three‑phase optimiser for a short time slice
    slice_deadline = min(time.time() + (deadline - start) * 0.25, deadline)
    a_coarse, score_coarse = _optimize_combined(a0, slice_deadline, rng, enforce_sym=False)
    print(f"[coarse] n={n_coarse} score={score_coarse:.6f}")

    # ------ Stage 2 : up‑sample and fine optimisation (large n) -----
    # Choose a larger problem size based on the remaining budget.
    remaining = deadline - time.time()
    if remaining < 5.0:
        final_seq = a_coarse.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final early] external={external:.6f}")
        globals()['height_sequence_1'] = np.array(final_seq)
        return final_seq

    # target length – try to use as many coefficients as time allows
    target_n = int(min(15000, max(8000, int(remaining * 15))))   # heuristic
    a_big = _upsample_seq(a_coarse, target_n)
    # re‑normalise to the required ℓ¹‑norm
    a_big = _project_simplex(a_big, np.sqrt(2.0 * target_n))

    slice_deadline = deadline
    a_opt, final_score = _optimize_combined(a_big, slice_deadline, rng, enforce_sym=False)
    final_seq = a_opt.tolist()
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal={final_score:.6f} n={len(final_seq)}")
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq

"""
Multiscale optimisation for the step‑function problem.

The goal is to minimise the quantity
    2·n·max(conv(a,a)) / (sum(a)**2)
with non‑negative entries a_i.  Because scaling a does not change the
objective, we fix the ℓ¹‑norm to √(2·n) so that the value reduces to
the maximum entry of the self‑convolution.

The search proceeds in many short restarts.  Each restart

1. picks a problem size n (larger sizes give more freedom),
2. draws an initial vector from a rich family of analytic shapes,
   optionally up‑sampling a previously found good vector,
3. runs a three‑phase optimiser
   – Adam on a smooth log‑sum‑exp surrogate,
   – sub‑gradient descent on the true max,
   – cheap random‑swap hill‑climbing,
   all respecting the simplex constraint,
4. keeps the best candidate seen so far.

The routine is time‑aware: it keeps a global deadline and stops well
before the external harness terminates the program.  Progress is printed
so that the behaviour can be inspected.

The implementation deliberately disables forced symmetry (symmetry tends
to increase the convolution peak) and adds a few extra families of
initial vectors (sinusoidal, block‑pattern) that were not present in the
reference code.  These changes give the optimiser more diverse starting
points and typically improve the final bound below the previous best
≈ 1.512.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x≥0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based) – fast full linear convolution
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities (log‑sum‑exp) – gradients for the smoothed max
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1      # 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Initial sequence generators – a richer family than the reference code
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation (or decimation) of an existing sequence."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several analytic families are mixed to provide diversity,
    including two new families (sinusoidal and block‑pattern).
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 8)   # 0‑7 inclusive

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 2:
        # geometric decay from both ends (symmetric)
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0

    elif typ == 3:
        # edge‑heavy reciprocal shape
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 4:
        # triangular shape (peak in centre)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    elif typ == 5:
        # random convex combination of a few bases (as before)
        cand = []
        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)          # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                       # reciprocal
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    elif typ == 6:
        # sinusoidal positive pattern with random power
        xs = np.linspace(0.0, np.pi, n)
        a = np.sin(xs) + rng.uniform(0.0, 0.3)   # shift a little upward
        power = rng.uniform(0.5, 1.5)
        a = a ** power

    else:   # typ == 7
        # block pattern: high on short intervals, low elsewhere
        a = np.zeros(n)
        block_len = rng.integers(1, max(2, n // 10))
        high = rng.uniform(0.8, 2.0)
        low = rng.uniform(0.0, 0.3)
        pos = 0
        while pos < n:
            a[pos:pos + block_len] = high
            pos += 2 * block_len
        a[a == 0] = low

    # Normalise to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the soft‑max approximation of the peak."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters (slightly more aggressive than the reference)
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.5
    alpha_mul = 1.04

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = 0.5 * (a + a[::-1])

        a = _project_simplex(a, S)

        # occasional random jitter to help escape plateaus
        if rng.random() < 0.025:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.15:
                    a = cand

        # schedule updates
        alpha = min(alpha * alpha_mul, 50.0)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum convolution entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent focusing on the current worst shift."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # compute sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = 0.5 * (a + a[::-1])
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # a cheap stagnation check
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap local random‑swap hill‑climbing (fine‑tuning)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Swap tiny masses between random positions until time runs out."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = 0.5 * (cand + cand[::-1])
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local search)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Runs the three optimisation phases within the supplied time budget.
    The time is split 60 % / 30 % / 10 % among Adam, sub‑gradient and
    the final local search.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    t_adam = now + 0.60 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # pick the best of the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score

# Main entry point for the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with a minimal self‑convolution peak.
    Returns the best sequence (list of floats) found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a few seconds for the external evaluation
    deadline = start_time + budget_s - 5.0

    # reuse a previously discovered construction if it exists
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main timed loop – many restarts with different lengths and
    # initialisations.  Symmetry is *not* enforced because it tends to
    # increase the convolution peak.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # problem size – favour larger instances but keep them tractable
        if best_overall_seq is not None and rng.random() < 0.4:
            # try a size near the current best (±10 %)
            scale = rng.uniform(0.9, 1.1)
            n = max(800, int(len(best_overall_seq) * scale))
        else:
            n = int(rng.integers(2000, 18000))

        enforce_sym = False            # keep sequences asymmetric

        # generate a fresh start
        a0 = _initial_sequence(base_seq, n, rng)

        # allocate a slice of time for this attempt (≈ 12–22 s)
        slice_len = rng.uniform(12.0, 22.0)
        slice_deadline = min(time.time() + slice_len, deadline)

        # Run the three‑phase optimiser
        a_opt, score = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        # Update global best
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} new best={best_overall_score:.6f}")

        # Early exit if we already beat the target comfortably
        if best_overall_score <= 1.48:
            break

        # If we have not improved for a while, drop the stored base_seq to regain diversity
        if time.time() - last_improve > 45.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]               # fallback – should never happen
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal_best={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
