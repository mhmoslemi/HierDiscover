"""
Step function optimizer that mixes smooth Adam, active‑set sub‑gradient
descent and aggressive local swaps.  The search runs many short
restarts with varied lengths and initial shapes, always keeping the
best vector found.  An up‑sampling step is applied at the very end to
refine a promising construction on a larger scale.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2*n-1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k (a*a)_k w.r.t. a.
    w must be non‑negative and sum to 1 (probability vector).
    """
    n = a.shape[0]
    # w_rev needed because (a*a)_k uses a_i a_{k-i}
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad

def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over the convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w

def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])

def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)

def _initial_sequence(base_seq: np.ndarray | None, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹ norm √(2·n).
    Several families are mixed to provide diversity.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 7)

    if typ == 0 and base_seq is not None and base_seq.size > 15:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        a = rng.random(n)                     # uniform Dirichlet(1)

    elif typ == 2:
        # geometric decay from the left side only (edge-heavy)
        rate = rng.uniform(0.9, 0.9995)
        a = rate ** np.arange(n)

    elif typ == 3:
        # edge‑heavy reciprocal shape with a random power
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.15, 0.85)
        a = rec ** power

    elif typ == 4:
        # triangular (peak in centre) with optional noise
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)
        a += rng.normal(scale=0.02, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 5:
        # mixture of a few bases
        cand = []
        cand.append(rng.random(n))
        rate = rng.uniform(0.9, 0.9995)
        cand.append(rate ** np.arange(n))
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    else:
        # small random perturbation around uniform
        a = np.full(n, 1.0)
        a += rng.normal(scale=0.02, size=n)
        a = np.clip(a, 0.0, None)

    # Normalisation to required ℓ¹‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a

def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising a log‑sum‑exp approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.5
    alpha_mul = 1.03

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.2:
                    a = cand

        # temperature and learning‑rate schedules
        alpha = min(alpha * alpha_mul, 50.0)
        lr *= 0.99996

    return best_a, best_score

def _optimize_active_subgrad(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             enforce_sym: bool = False,
                             tol_factor: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that uses all convolution entries whose
    value is within a small tolerance of the maximum.
    This yields a smoother descent direction than using a single worst shift.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        max_val = conv.max()
        # active set: all indices within tol_factor * max_val of the max
        tol = max_val * tol_factor
        active = np.where(conv >= max_val - tol)[0]

        # build weight vector (uniform over active)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        grad = _conv_with_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random swap
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.025 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # early stagnation break
        if it % 9000 == 0 and best_score == cur_max:
            break

    return best_a, best_score

def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Cheap hill‑climbing that swaps tiny masses between random positions."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.015 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand

    return best_a, best_score

def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Run a three‑phase pipeline (Adam → active‑set sub‑gradient → local swaps)
    within the supplied time budget.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # Allocate slices
    t_adam = now + 0.55 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.35 * remaining, deadline)
    a_sub, sc_sub = _optimize_active_subgrad(a_adam, t_sub, rng, enforce_sym)

    # final local refinement if time remains
    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score

"""
Step‑function optimiser.

The search mixes three complementary phases:

1. **Adam on a soft‑max approximation** – a smooth surrogate of the
   convolution peak.  A simple Adam optimiser with a simplex
   projection drives the vector towards a lower peak while slowly
   increasing the temperature of the soft‑max.

2. **Active‑set sub‑gradient descent** – instead of using a single
   worst shift, all convolution entries whose value lies within a
   small tolerance of the maximum are weighted uniformly.  The
   resulting gradient (computed via a weighted convolution) provides
   a smoother descent direction and includes the correct factor 2 for
   the diagonal terms.

3. **Local random‑swap hill‑climbing** – very cheap mass‑swap moves
   finish the optimisation.  An optional up‑sampling step at the end
   refines a promising construction on a larger scale.

The routine repeatedly restarts with varied lengths and initial
shapes, always keeping the best vector found before the 1000 s
budget expires.  The final sequence is returned as a plain Python
list; it is also stored in the global variable ``height_sequence_1``
for possible reuse.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution ``a*a`` (length ``2·n‑1``)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. ``a``.
    ``w`` must be non‑negative and sum to 1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = 3 * n - 2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * full[idx]


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) probabilities over the convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    return exp_shifted / exp_shifted.sum()


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry ``v[i] = v[-1‑i]``."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of ``base`` to a new length ``n``."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                      n: int,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length ``n`` with ℓ¹‑norm √(2·n).
    Several analytic families are mixed to obtain diverse starting points.
    """
    S = np.sqrt(2.0 * n)
    typ = rng.integers(0, 7)

    if typ == 0 and base_seq is not None and base_seq.size > 15:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        a = rng.random(n)                     # Dirichlet(1) – uniform

    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        a = rate ** np.arange(n)               # geometric decay (edge‑heavy)

    elif typ == 3:
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.15, 0.85)
        a = rec ** power                       # reciprocal‑type shape

    elif typ == 4:
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)
        a += rng.normal(scale=0.02, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 5:
        cand = [rng.random(n), rate ** np.arange(n)]
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    else:
        a = np.full(n, 1.0)
        a += rng.normal(scale=0.02, size=n)
        a = np.clip(a, 0.0, None)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    return a / total * S


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth (log‑sum‑exp) surrogate."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 0.5
    alpha_mul = 1.03

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape plateaus
        if rng.random() < 0.015:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.2:
                    a = cand

        alpha = min(alpha * alpha_mul, 50.0)
        lr *= 0.99996

    return best_a, best_score


def _optimize_active_subgrad(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             enforce_sym: bool = False,
                             tol_factor: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using *all* near‑max convolution entries.
    ``tol_factor`` determines the relative tolerance to the current maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        max_val = conv.max()
        tol = max_val * tol_factor
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        grad = _conv_with_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass transfer (simulated‑annealing style)
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.025 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        if it % 9000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing – swaps tiny amounts of mass between random indices."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.015 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand

    return best_a, best_score


def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Three‑phase pipeline (Adam → active‑set sub‑gradient → local swaps)
    executed inside the allotted time slice.
    """
    now = time.time()
    if deadline - now <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # split the remaining budget: 55 % Adam, 35 % active sub‑grad, rest local swaps
    t_adam = now + 0.55 * (deadline - now)
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.35 * (deadline - now), deadline)
    a_sub, sc_sub = _optimize_active_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Find a non‑negative step‑function (list of floats) with a small
    self‑convolution peak.  The function respects the 1000 s budget
    and returns the best vector discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # small safety margin

    # reuse a previously found construction if present
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0
    last_improve = start

    # ------------------------------------------------------------------
    # Main timed loop – many short restarts with varied lengths & shapes
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # length selection (sometimes stay near the current best)
        if best_seq is not None and rng.random() < 0.45:
            scale = rng.uniform(0.85, 1.15)
            n = max(600, int(best_n * scale))
        else:
            n = int(rng.integers(1300, 10000))

        enforce_sym = rng.random() < 0.4

        a0 = _initial_sequence(base_seq, n, rng)
        if enforce_sym:
            a0 = _symmetrize(a0)

        slice_deadline = min(time.time() + rng.uniform(14.0, 24.0), deadline)
        a_opt, score = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} new best={best_score:.6f}")

        if best_score <= 1.503:                 # known target – stop early
            break

        if time.time() - last_improve > 50.0:   # stagnation → refresh base
            base_seq = None

    # ------------------------------------------------------------------
    # Optional up‑sampling and final local polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        target_len = max(best_n, 8500)
        if best_n < target_len:
            up = _upsample_seq(best_seq, target_len)
            up = _project_simplex(up, np.sqrt(2.0 * target_len))
            up, up_score = _local_swap_search(up, deadline, rng, enforce_sym=False)
            if up_score < best_score:
                best_seq = up
                best_score = up_score
                best_n = target_len

    final_seq = best_seq.tolist() if best_seq is not None else [1.0]

    # sanity check with the provided black‑box evaluator
    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal_best={best_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)
    return final_seq
