def subgradient_descent(a, deadline):
    n = len(a)
    target_sum = np.sqrt(2.0*n)
    a = project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_val = compute_score(a)
    iteration = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = compute_convolution(a)
        max_val = conv.max()
        k_max = int(conv.argmax())
        # compute gradient
        g = gradient_max_conv(a, k_max)
        # step size: could adapt as alpha / (1 + iteration*0.001)
        # Use diminishing step size: step = step0 / np.sqrt(iteration+1)
        step0 = target_sum * 0.1
        lr = step0 / (np.sqrt(iteration+1))
        a = a - lr * g
        # Project back onto simplex
        a = project_to_simplex(a, target_sum)
        # maybe occasional jitter
        if iteration % 1000 == 0:
            jitter = rng.normal(scale=target_sum*0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a = project_to_simplex(a, target_sum)
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SG it={iteration} score={best_val:.6f}")
        iteration += 1
    return best_a, best_val

def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 1.0

    best_seq = None
    best_score = float('inf')
    cur_seq = None

        if now >= deadline - 5:
            break
        remaining = deadline - now

        target_sum = np.sqrt(2.0 * n)

        # generate initial candidates list

        init_candidates.append(_arcsine_init(n, target_sum))

        # 4. warm start from previous best if available (resized)

        for init in init_candidates:

    chosen = []

    for i in range(n):
        # possible sums: i + i (double) plus i + j for each j in chosen
        conflict = False
        # self sum
        if i + i in sums_set:
            conflict = True
        else:

                    conflict = True
                    break
        if not conflict:
            # add
            chosen.append(i)
            # add sums
            sums_set.add(i + i)
            for j in chosen[:-1]:  # old chosen elements
                sums_set.add(i + j)

"""
Improved search for a low‑value step‑function sequence.

Strategy
--------
1.  Work on three increasing problem sizes (≈2000 → 5000 → 12000).  
    The larger the length, the finer the discretisation of the
    continuous optimisation problem and the lower the achievable bound.

2.  Several complementary initialisations are tried:
    * a symmetric “arcsine” distribution (the continuous arcsine law is
      known to flatten the self‑convolution);
    * a greedy Sidon set – indices whose pair‑sums are all distinct,
      with equal mass on the chosen positions;
    * a simple power‑law (the previous baseline);
    * a random Dirichlet sample (warm‑start for diversity);
    * the best solution from the previous scale (if any).

3.  Two gradient phases are performed on each instance:
    * a soft‑max approximation of the max‑convolution (β is driven up
      to 2**20).  This gives a smooth gradient that guides the iterate
      toward a good basin of attraction.
    * a pure sub‑gradient step that uses the exact (hard) max index.
      It is fast, works directly on the true objective and removes the
      remaining bias introduced by the soft‑max.

4.  A final simulated‑annealing refinement (mass‑transfer, local
    smoothing and random swaps) explores the neighbourhood more
    globally and helps to escape shallow minima.

5.  The best sequence found over all scales and all starts is returned.
    The routine respects the 1000 s budget, projects onto the ℓ¹‑simplex
    after each move (so the sum is fixed to √(2 n)) and never produces
    negative entries.

The code is deliberately self‑contained – all helper functions are
defined at top level and no external files or network access are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence of length n, normalised to ``target_sum``."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fully vectorised)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  ``a`` is assumed non‑negative."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# Special initialisers
# ----------------------------------------------------------------------


def _arcsine_init(n: int, target_sum: float) -> np.ndarray:
    """
    Symmetric arcsine‑law like start:
    a_i ∝ 1 / sqrt( (i+0.5) * (n-i-0.5) ),  (i = 0 … n‑1)
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    base = 1.0 / np.sqrt(i * (n - i))
    base *= target_sum / base.sum()
    return base


def _greedy_sidon_set(n: int) -> np.ndarray:
    """
    Greedy construction of a Sidon set (all pair‑sums are distinct)
    inside {0,…,n‑1}.  Returns a boolean mask of selected positions.
    """
    chosen = []
    sums = set()
    for idx in range(n):
        # test whether adding idx would clash
        clash = False
        if (idx + idx) in sums:
            clash = True
        else:
            for j in chosen:
                if (idx + j) in sums:
                    clash = True
                    break
        if not clash:
            chosen.append(idx)
            sums.add(idx + idx)
            for j in chosen[:-1]:
                sums.add(idx + j)
    mask = np.zeros(n, dtype=bool)
    mask[chosen] = True
    return mask


def _sidon_init(n: int, target_sum: float) -> np.ndarray:
    """
    Initialise a sequence that puts equal mass on a greedy Sidon set.
    """
    mask = _greedy_sidon_set(n)
    m = mask.sum()
    if m == 0:          # fall back to uniform – should never happen
        return _power_law_sequence(n, 0.55, target_sum)
    a = np.zeros(n, dtype=np.float64)
    a[mask] = target_sum / m
    return a

# Gradient‑based optimisation

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    Returns (grad, conv) where ``grad`` has the same shape as ``a``.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a reversed) ⋆ p   (slice the central n entries)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_beta: float = 2 ** 20,
    momentum_gamma: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate.
    ``max_beta`` controls the final sharpness of the soft‑max.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    step_base = target_sum * 0.25
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = momentum_gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # geometric increase of β
        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    base_step: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The gradient for the active index k is 2 * a_rev * δ_k .
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        k = int(conv.argmax())                     # a hard max index
        # exact sub‑gradient for this k (2 * a_{k-i})
        idx = np.arange(n)
        src = k - idx
        mask = (src >= 0) & (src < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[src[mask]]

        lr = base_step / np.sqrt(it + 1.0)
        a -= lr * g
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 1500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SG it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 1.0,
    end_temp: float = 1e-4,
    max_frac_start: float = 0.30,
    max_frac_end: float = 0.01,
    max_window: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with three elementary moves:
        * mass transfer between a “sensitive’’ (large weight) index and a
          low‑weight index,
        * short‑window averaging (local smoothing),
        * random swap of two entries.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current most‑binding convolution index
        k_max = int(np.argmax(conv))

        # sensitivity (derivative of the k_max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens
        move_type = None

        # ------------------- mass transfer -------------------
        if rng.random() < 0.65:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ------------------- smoothing -------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                a[start:end] = old_seg.mean()
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ------------------- swap ---------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution & score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # rollback
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter (helps escaping deep local minima)
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            fa = np.fft.rfft(a, n=size_fft)
            conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _run_one_instance(n: int, init_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a single length‑n problem.
    Phases:
        1. soft‑max GD (≈30 % of the time budget)
        2. hard‑max sub‑gradient GD (≈30 % of the time budget)
        3. SA (remaining time)
    Returns the best sequence and its score.
    """
    target_sum = np.sqrt(2.0 * n)

    a = init_seq.astype(np.float64, copy=True)
    if a.size != n:
        a = _resize_sequence(a, n)
    a = _project_to_simplex(a, target_sum)

    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return a, _score(a)

    # ----- phase 1 : soft‑max GD -----
    phase1_deadline = now + remaining * 0.30
    a, score1 = _annealed_gradient_descent(a, phase1_deadline)

    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return a, _score(a)

    # ----- phase 2 : hard‑max sub‑gradient -----
    phase2_deadline = now + remaining * 0.30
    a, score2 = _subgradient_descent(a, phase2_deadline)

    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return a, _score(a)

    # ----- phase 3 : simulated annealing -----
    a, score3 = _simulated_annealing(a, deadline)

    final_score = min(score1, score2, score3)
    return a, final_score

    best_score = float("inf")
    cur_seq = None

    # (length, fraction of remaining time) – larger lengths give a finer
    # approximation of the continuous optimum.
    scales = [(2000, 0.15), (5000, 0.25), (12000, 0.60)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        target_sum = np.sqrt(2.0 * n)

        # ---- build a list of initial candidates for this scale ----
        init_candidates = []

        # 1. arcsine distribution
        init_candidates.append(_arcsine_init(n, target_sum))

        # 2. Sidon set (equal mass on a greedy Sidon set)
        init_candidates.append(_sidon_init(n, target_sum))

        # 3. power‑law (the baseline that was used previously)
        init_candidates.append(_power_law_sequence(n, 0.55, target_sum))

        # 4. random Dirichlet (ensures diversity)
        rng = np.random.default_rng()
        rand = rng.random(n)
        init_candidates.append(_project_to_simplex(rand, target_sum))

        # 5. warm start from the previous scale (if any)
        if cur_seq is not None:
            init_candidates.append(_resize_sequence(cur_seq, n))

        # ---- run optimisation for each start and keep the best ----
        for init in init_candidates:
            seq, score = _run_one_instance(n, init, stage_deadline)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] Scale n={n} new best score={best_score:.6f}")

        cur_seq = best_seq  # warm‑start next (larger) scale

    # Trim trailing near‑zero entries for a tidy output
    if best_seq is None:
        return []

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # One final evaluation with the provided benchmark (optional, just for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
