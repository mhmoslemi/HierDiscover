def _max_convolution_fft(a):
    n = a.shape[0]
    size = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

def _score(a):
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v,s):
    # From Duchi et al.
    if v.sum() == s and np.all(v>=0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _power_law_sequence(n, p, target_sum):
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a

def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = _next_pow2(2*n-1)
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]

    max_b = b.max()
    # compute softmax probabilities
    # subtract max for stability
    exps = np.exp(beta * (b - max_b))
    sum_exps = exps.sum()
    p = exps / sum_exps

    # compute gradient: 2 * (reversed a) convolve p
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2*n-1]
    # the convolution result for indices n-1..2n-2 correspond to gradient components
    grad = 2.0 * conv_rev[n-1:n-1 + n]
    return grad, b

def _adam_gradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_val = _score(a)

    # Adam parameters
    m = np.zeros_like(a)
    v2 = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_scale = target_sum * 0.5  # base step

    beta = 1.0
    max_beta = 8192.0
    beta_inc = 500  # iterations between beta increases

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v2 = beta2 * v2 + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v2 / (1 - beta2 ** it)
        # Adam update (note: we want to go opposite direction of gradient)
        step = step_scale * (m_hat / (np.sqrt(v_hat) + eps))
        a -= step
        # project to simplex
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase β
        if it % beta_inc == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it} β={beta:.1f} score={best_val:.6f}")

    return best_a, best_val

def _simulated_annealing(a0, deadline,
                         start_temp=1.0,
                         end_temp=1e-5,
                         max_frac_start=0.2,
                         max_frac_end=0.005,
                         max_window=50):
    a = a0.copy()
    n = a.size
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = _next_pow2(2*n-1)
    conv = _conv_fft(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # pick move type
        move_type = rng.choice(['transfer', 'smooth', 'swap'], p=[0.6, 0.3, 0.1])

        if move_type == 'transfer':
            # compute sensitivities based on current max conv entry
            k_max = int(np.argmax(conv))
            # compute gradient contributions for each a_i to entry k_max
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            sens = np.zeros(n)
            sens[idx] = a[k_max - idx]  # derivative w.r.t a_i is a_{k-i}
            # weight = a * sens to favour large contributions
            weight = a * sens
            # choose donor proportional to weight
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight/total_w)
            else:
                donor = rng.integers(n)
            # choose receiver among low-weight indices
            low_thr = np.percentile(weight, 20)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size > 0:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a[donor] -= delta
                a[receiver] += delta
                move = ('transfer', donor, receiver, delta)
            else:
                move = None

        elif move_type == 'smooth':
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_idx = min(start + length, n)
            if end_idx - start >= 2:
                old_seg = a[start:end_idx].copy()
                avg = old_seg.mean()
                a[start:end_idx] = avg
                move = ('smooth', start, end_idx, old_seg)
            else:
                move = None

        else:  # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move = ('swap', i, j)

        # compute new conv and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new best={best_score:.6f}")
        else:
            # revert
            if move is not None:
                if move[0] == 'transfer':
                    _, donor, receiver, delta = move
                    a[donor] += delta
                    a[receiver] -= delta
                elif move[0] == 'smooth':
                    _, start_idx, end_idx, old_seg = move
                    a[start_idx:end_idx] = old_seg
                elif move[0] == 'swap':
                    _, i, j = move
                    a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape stagnation
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0*n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _run_one_instance(n, init_seq, deadline):
    target_sum = np.sqrt(2.0 * n)
    a = init_seq.astype(np.float64, copy=True)

"""
Optimise a non‑negative step‑function (a list of heights) for the benchmark
`evaluate_sequence`.  The objective is

    2·n·max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling.  After fixing the ℓ¹‑norm to
√(2·n) the objective equals the maximal entry of the self‑convolution.
The code builds a sequence of increasing lengths (≈1500 → 3000 → 6000);
for each length it runs

* a projected Adam optimiser on a soft‑max approximation of the
  max‑convolution (β is increased from 1 to 8192);
* a short simulated‑annealing phase that moves mass between indices,
  locally smooths windows and swaps entries.

All iterates are projected onto the ℓ¹‑simplex after each update, which
keeps the sum fixed and guarantees feasibility.  The best sequence over
all scales is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padded FFT)."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _constant_conv_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    Sequence whose infinite‑length self‑convolution is constant.
    a_i = C(2i,i) / 4**i  (≈ 1/√i).  The values are generated by the recurrence

        a_{i+1} = a_i * (i + 0.5) / (i + 1).

    The vector is finally scaled to have ℓ¹‑norm = target_sum.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * ((i + 0.5) / (i + 1))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the smooth‑max approximation
        f_β(a) = (1/β)·log Σ exp(β·b_k) ,  b = a*a .
    Returns (grad, b) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()                     # soft‑max weights

    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected Adam optimiser on the soft‑max surrogate of the max‑convolution.
    β starts at 1 and is multiplied by 1.5 every `beta_inc` iterations,
    up to `max_beta`.  The iterate is projected onto the ℓ¹‑simplex after each
    update.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)          # first moment
    v2 = np.zeros_like(a)         # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step_base = target_sum * 0.5   # controls raw step magnitude

    beta = 1.0
    max_beta = 8192.0
    beta_inc = 500                # increase β every this many iterations
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        grad, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v2 / (1.0 - beta2 ** it)

        # Adam update (gradient descent direction)
        step = step_base * (m_hat / (np.sqrt(v_hat) + eps))
        a -= step

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow basins
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β schedule
        if it % beta_inc == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} "
                  f"score={best_val:.6f}")

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.20,
                         max_frac_end: float = 0.005,
                         max_window: int = 50) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves mass from “sensitive’’ indices (large contribution
    to the current maximal convolution entry) to low‑sensitivity indices,
    performs short‑window averaging and random swaps.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = _next_pow2(2 * n - 1)
    conv = _conv_fft(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # -------------------------------------------------------------
        # Choose a move
        # -------------------------------------------------------------
        move_type = rng.choice(['transfer', 'smooth', 'swap'],
                               p=[0.65, 0.25, 0.10])

        if move_type == 'transfer':
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)

            sens = np.zeros(n)
            sens[idx] = a[k_max - idx]                     # ∂b_k/∂a_i at max entry

            weight = a * sens                               # higher → more likely donor
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            low_thr = np.percentile(weight, 20)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = (rng.choice(low_idx) if low_idx.size
                        else rng.integers(n))

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                a[donor] -= delta
                a[receiver] += delta
                move = ('transfer', donor, receiver, delta)
            else:
                move = None

        elif move_type == 'smooth':
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move = ('smooth', start, end, old_seg)
            else:
                move = None

        else:   # swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move = ('swap', i, j)

        # -------------------------------------------------------------
        # Evaluate the move
        # -------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[:2 * n - 1]
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert
            if move is not None:
                if move[0] == 'transfer':
                    _, donor, receiver, delta = move
                    a[donor] += delta
                    a[receiver] -= delta
                elif move[0] == 'smooth':
                    _, s, e, old = move
                    a[s:e] = old
                else:   # swap
                    _, i, j = move
                    a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _run_one_instance(n: int, init_seq: np.ndarray, deadline: float):
    """
    Optimise a single length‑n instance.
    1. Adam gradient descent (≈60 % of the allocated budget).
    2. Simulated‑annealing refinement for the rest.
    Returns (best_sequence, best_score).
    """
    target_sum = np.sqrt(2.0 * n)
    a = init_seq.astype(np.float64, copy=True)

    if a.size != n:
        a = _resize_sequence(a, n)

    a = _project_to_simplex(a, target_sum)

    now = time.time()
    remaining = deadline - now
    if remaining > 20.0:
        gd_deadline = now + min(0.6 * remaining, 60.0)
        a, gd_score = _adam_gradient_descent(a, gd_deadline)
    else:
        gd_score = _score(a)

    # SA phase with whatever time is left
    if time.time() < deadline:
        a, sa_score = _simulated_annealing(a, deadline)
        best_score = min(gd_score, sa_score)
    else:
        best_score = gd_score

    return a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a step‑function with a low benchmark score."""
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # three increasing scales; time is split roughly equally among them
    scales = [(1500, 0.30), (3000, 0.30), (6000, 0.40)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # ---------------------------------------------------------
        # Initialise the candidate for this scale
        # ---------------------------------------------------------
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            if "height_sequence_1" in globals():
                init = _resize_sequence(np.array(globals()["height_sequence_1"],
                                                dtype=np.float64), n)
            else:
                # random choice between a power‑law and the (1‑z)^‑½ seed
                rng = np.random.default_rng()
                if rng.random() < 0.5:
                    p = rng.uniform(0.45, 0.65)
                    init = _power_law_sequence(n, p, np.sqrt(2.0 * n))
                else:
                    init = _constant_conv_sequence(n, np.sqrt(2.0 * n))

        # small random jitter before optimisation
        rng = np.random.default_rng()
        init += rng.normal(scale=init.mean() * 0.005, size=n)
        init = np.maximum(init, 0.0)
        init = _project_to_simplex(init, np.sqrt(2.0 * n))

        # ---------------------------------------------------------
        # Optimise this scale
        # ---------------------------------------------------------
        seq, score = _run_one_instance(n, init, stage_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Scale n={n} improved best score={best_score:.6f}")

        cur_seq = seq      # warm‑start the next (larger) scale

    # -----------------------------------------------------------------
    # Post‑process the best sequence
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = []

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Optional final evaluation (helps with debugging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
