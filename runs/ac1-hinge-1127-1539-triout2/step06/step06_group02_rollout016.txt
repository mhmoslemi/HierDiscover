def hill_climb(a, deadline):
    n = len(a)
    target_sum = np.sqrt(2*n)
    best_a = a.copy()
    best_score = _score(a) (i.e., maxconv)
    # track temperature schedule
    start_time = time.time()
    total_time = deadline - start_time
    # parameters
    init_temp = 0.2
    final_temp = 1e-5
    max_frac_start = 0.25
    max_frac_end = 0.001
    # maybe schedule for step fraction
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temperature = init_temp * (final_temp / init_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # compute weights
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi+1)
        w = a[idx_range] * a[k_max - idx_range]

        # donor: choose with probability proportional to w
        total_w = w.sum()
        if total_w > 0 and np.random.rand() < 0.9:
            donor_idx = np.random.choice(idx_range, p=w/total_w)
        else:
            # fallback random
            donor_idx = np.random.randint(n)

        # receiver: choose index with small w (low contribution)
        # can compute w across all indices but using same mapping to offset? We'll compute weight_all = a * a_shift where shift = k_max - np.arange(n) but invalid indexes get 0.
        # However we can approximate by picking random index weighted by inverse w.
        weight_all = np.zeros(n)
        # compute vectorization:
        # For i in idx_range, weight_all[i] = a[i] * a[k_max-i]; others remain 0.
        weight_all[idx_range] = w

        # pick low weight indices (e.g., bottom 20% of weight_all)
        threshold = np.percentile(weight_all, 20)
        low_indices = np.where(weight_all <= threshold)[0]
        if low_indices.size > 0:
            receiver_idx = np.random.choice(low_indices)
        else:
            receiver_idx = np.random.randint(n)

        if donor_idx == receiver_idx:
            continue

        # Choose delta: fraction of donor mass, limited by max_frac
        max_transfer = a[donor_idx] * max_frac
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer

        # apply move
        a[donor_idx] -= delta
        a[receiver_idx] += delta

        # Evaluate new max conv
        new_conv = _conv_fft(a)
        new_max = new_conv.max()

        # Accept based on Metropolis
        accept = False
        if new_max < max_val:
            accept = True
        else:
            # probability decreasing with temperature
            prob = np.exp(-(new_max - max_val) / (temperature + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # keep new state
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                # optional print
                print(f"[{time.time():.1f}] new best {best_score:.6f}")
            # conv = new_conv for next iteration (reuse)
        else:
            # revert move
            a[donor_idx] += delta
            a[receiver_idx] -= delta

        # occasionally jitter global
        if (int(elapsed*10) % 500==0): # maybe every 50 seconds
            jitter = np.random.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.clip(a, 0, None)
            # reproject onto simplex to maintain sum
            a = _project_to_simplex(a, target_sum)

        # may also optionally perform smoothing moves occasionally

    # after loop, final projection onto simplex
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

where conv is the linear convolution (np.convolve).  

The algorithm combines three ideas:

1. **Diverse initialisations** – power‑law, sin‑window, random Dirichlet,
   and optionally the warm‑start from the global variable `height_sequence_1`.
   Each initial vector is projected onto the ℓ¹‑simplex
   Σa_i = √(2·n) (the objective is scale‑invariant).

2. **Projected Adam optimisation on the true max‑convolution**.  
   At every iteration we compute the worst offset `k* = argmax conv`,
   obtain the sub‑gradient `g_i = 2·a_{k*‑i}` (zero outside the support),
   and perform an Adam update followed by a Euclidean projection onto the
   simplex.  A small “jitter’’ is added periodically to escape plateaus.

3. **Targeted simulated‑annealing refinement** – moves that transfer mass
   from indices that contribute most to the worst convolution entry to
   indices that contribute little, short‑window smoothing and occasional
   swaps.  Temperature and maximal transfer fraction decay exponentially.

The whole budget (default 1000 s) is split among several independent runs
with different problem sizes (1500, 3000, 5000).  The best sequence found
over all runs is returned (trailing zeros are stripped).  The code uses
only NumPy and the supplied `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements Duchi et al. (2008).  Returns a new array.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _maxconv_and_index(a: np.ndarray):
    """Return (max_convolution_value, index_of_max, full_conv_array)."""
    conv = _conv_fft(a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Sequence a_i = (i+1)^{-p} normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _sin_window_sequence(n: int, target_sum: float) -> np.ndarray:
    """Sinusoidal window a_i = sin(π·i/(n+1)), normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, target_sum: float, sigma_factor: float = 0.3) -> np.ndarray:
    """Discrete Gaussian centred at n/2, normalised."""
    i = np.arange(n, dtype=np.float64)
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _subgradient_maxconv(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) at the current worst offset.
    Returns (gradient vector, max value, worst offset, full conv array).
    """
    max_val, kmax, conv = _maxconv_and_index(a)
    n = a.shape[0]
    lo = max(0, kmax - (n - 1))
    hi = min(kmax, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[kmax - idx]          # 2·a_{k*‑i}
    return grad, max_val, kmax, conv


def _adam_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises max(conv(a,a)).
    The step size decays as 1/√t and the iterate is projected onto the
    ℓ¹‑simplex Σa = √(2·n) after each update.
    """
    start = time.time()
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_val, _, _ = _maxconv_and_index(a)

    base_lr = target * 0.3    # scale of the learning rate
    t = 0

    while time.time() < deadline:
        t += 1
        # sub‑gradient at the current worst offset
        grad, cur_max, kmax, _ = _subgradient_maxconv(a)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # decreasing learning rate
        lr = base_lr / np.sqrt(t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = _project_to_simplex(a_candidate, target)

        cand_max, _, _ = _maxconv_and_index(a_candidate)

        # keep the best ever seen
        if cand_max < best_val:
            best_val = cand_max
            best_a = a_candidate.copy()
            print(f"[{time.time():.1f}] Adam iter {t} new best max={best_val:.6f}")

        # simple acceptance: take the move if it does not increase the worst value,
        # otherwise accept with a small temperature that decays linearly.
        accept = False
        if cand_max <= cur_max:
            accept = True
        else:
            tau = (time.time() - start) / max(1e-6, (deadline - start))
            temp = 0.5 * (1.0 - tau) + 1e-12
            prob = np.exp(-(cand_max - cur_max) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate

        # occasional global jitter to escape flat regions
        if t % 5000 == 0:
            jitter = np.random.normal(scale=target * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-5,
    max_frac_start: float = 0.40,
    max_frac_end: float = 0.005,
    max_window: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Refinement by simulated annealing with moves that target the current
    worst convolution entry.
    """
    a = a0.copy()
    n = a.size
    cur_max, k_max, conv = _maxconv_and_index(a)

    best_a = a.copy()
    best_score = cur_max

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # --- analyse the worst offset -------------------------------------------------
        # contributions of individual indices to the worst entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] * a[k_max - idx]                # contribution of each a_i

        # ----------------------------------------------------------------------
        # propose a move
        # ----------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:          # mass transfer (dominant move)
            total_w = w.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # choose a receiver that contributes very little to the worst entry
            low_thr = np.percentile(w, 10)
            low_candidates = idx[w <= low_thr]
            if low_candidates.size:
                receiver = int(rng.choice(low_candidates))
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:        # short‑window smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        else:                             # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ----------------------------------------------------------------------
        # evaluate new state (single FFT)
        # ----------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_max = float(conv_new.max())

        # ----------------------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            conv = conv_new
            k_max = int(np.argmax(conv))
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best max={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            fa = np.fft.rfft(a, n=size_fft)
            conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
            cur_max = float(conv.max())
            k_max = int(np.argmax(conv))

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise the step‑function within the given time budget.
    Returns the best sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes to try – larger sizes give finer granularity.
    sizes = [1500, 3000, 5000]

    # Helper to build an initial candidate for a given size.
    def _initial_candidate(n: int):
        # choose a random initialisation strategy
        mode = rng.choice(["power", "sin", "gauss", "random"])
        target = np.sqrt(2.0 * n)

        if mode == "power":
            p = rng.uniform(0.4, 0.7)
            seq = _power_law_sequence(n, p, target)
        elif mode == "sin":
            seq = _sin_window_sequence(n, target)
        elif mode == "gauss":
            seq = _gaussian_sequence(n, target, sigma_factor=rng.uniform(0.25, 0.35))
        else:   # random Dirichlet‑like
            seq = rng.random(n)
            seq = seq / seq.sum() * target

        # small uniform noise to break symmetry
        seq += rng.random(n) * target * 1e-4
        seq = _project_to_simplex(seq, target)
        return seq

    # Main optimisation loop – run independent instances until time runs out.
    size_cycle = 0
    while time.time() < deadline - 5.0:
        n = sizes[size_cycle % len(sizes)]
        size_cycle += 1

        remaining = deadline - time.time()
        # guarantee at least a few seconds for the final polishing step
        run_time = min(remaining * 0.45, 250.0)

        # initialise (warm‑start from global variable if available)
        if "height_sequence_1" in globals():
            warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
            if warm.size > n:
                a0 = warm[:n]
            else:
                a0 = np.pad(warm, (0, n - warm.size), "constant")
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n))
        else:
            a0 = _initial_candidate(n)

        run_deadline = time.time() + run_time

        # 1) Adam optimiser on the true max‑convolution
        a_opt, _ = _adam_optimize(a0, run_deadline * 0.5 + time.time())

        # 2) Simulated‑annealing refinement (uses the remaining time)
        if time.time() < deadline - 2.0:
            a_opt, _ = _simulated_annealing(a_opt, deadline)

        # final evaluation with the supplied function
        cur_score = evaluate_sequence(a_opt.tolist())
        print(f"[{time.time():.1f}] size={n} → score={cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] *** New overall best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: a short Adam pass on the best candidate (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _adam_optimize(best_seq, extra_deadline)

    # Strip trailing zeros – they do not affect the objective.
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
