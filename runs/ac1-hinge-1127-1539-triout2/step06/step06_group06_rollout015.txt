def _maxconv_and_grad(a):
    conv = _conv_fft(a)  # compute full convolution
    kmax = int(np.argmax(conv))
    # compute subgradient of max at kmax
    grad = np.zeros_like(a)
    lo = max(0, kmax - (n-1))
    hi = min(kmax, n-1)
    # indexes i in [lo, hi] where a_i multiplies a_{kmax-i}
    # So for each i in that range:
    # contribution to max = a_i * a_{kmax-i}
    # gradient wrt a_i is a_{kmax-i}, and also wrt a_{kmax-i} is a_i (since symmetric)
    # So grad_i += a_{kmax-i} ; grad_{kmax - i} += a_i
    # resolve overlapping where i == k_max-i (only when k_max even and i==k_max/2) then add 2*a_i?
    # We'll compute vectorized.
    i_arr = np.arange(lo, hi+1)
    j_arr = kmax - i_arr
    # Add contributions
    grad[i_arr] += a[j_arr]
    grad[j_arr] += a[i_arr]
    # if any index counted twice (i==j) we added double count (which is correct as derivative of a_i*a_i = 2 a_i)

"""
Propose a step‑function (height sequence) that minimises the benchmark
`evaluate_sequence`.  The objective is proportional to the maximum entry of
the self‑convolution of the sequence while the ℓ¹‑norm is fixed (the ratio is
scale‑invariant).  The algorithm works in three increasing lengths
(~2000 → 4000 → 8000).  For each length it performs

1. Several short restarts of projected gradient descent on a soft‑max
   approximation of the max‑convolution (β is increased up to 1e5).  The
   update uses a simple Adam‑style adaptive step and occasional jitter.
2. A simulated‑annealing phase that moves mass from “sensitive’’ indices
   (large contribution to the current maximal convolution entry) to low‑
   contribution indices, smoothes short windows and swaps values.
3. The best solution of the current scale is interpolated and used as a warm
   start for the next (larger) scale.

All updates keep the vector non‑negative and projected onto the simplex
Σa_i = √(2·n).  The routine always returns a sequence before the budget is
exhausted; intermediate progress and the final evaluated score are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Exponential decay a_i = exp(-lam*i) normalised to target_sum."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where `grad` has the same shape as `a`.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a reversed) ⋆ p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected Adam‑style gradient descent on the soft‑max approximation.
    β starts at 1 and is increased up to 1e5.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e5
    rng = np.random.default_rng()
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = target_sum * 0.1 / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        # increase β slowly – the higher β, the closer we are to the true max
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 2.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.35,
                         max_frac_end: float = 0.008,
                         max_window: int = 70) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement.
    Moves:
      * mass transfer from a high‑sensitivity index to a low‑sensitivity one,
      * short‑window averaging (smoothing),
      * swap of two positions.
    Temperature and allowed transfer fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # sensitivities (how much each a_i contributes to the max entry)
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]

        weight = a * sens

        move_type = None

        # ---- mass transfer -------------------------------------------------
        if rng.random() < 0.65:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # ---- local smoothing -----------------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                avg = old_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # ---- swap -----------------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # recompute convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 9000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def _initial_candidates(n: int, target_sum: float):
    """Create a diverse set of starting vectors for a given length."""
    cand = []

    # uniform
    cand.append(np.full(n, target_sum / n))

    # power‑law family
    for p in np.linspace(0.1, 2.0, 9):
        cand.append(_power_law_sequence(n, p, target_sum))

    # exponential decays
    for lam in [0.0005, 0.001, 0.002, 0.005, 0.01]:
        cand.append(_exponential_sequence(n, lam, target_sum))

    # random Dirichlet / Gamma draws
    rng = np.random.default_rng()
    for _ in range(5):
        a = rng.gamma(1.0, 1.0, size=n)
        a *= target_sum / a.sum()
        cand.append(a)

    # use the globally supplied sequence if available
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size != n:
            seq = _resize_sequence(seq, n)
        cand.append(_project_to_simplex(seq, target_sum))

    return cand

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the benchmark.
    Three increasing scales are explored (≈2000 → 4000 → 8000).  For each
    scale a suite of random restarts of projected Adam‑gradient descent
    followed by simulated‑annealing refinement is performed.
    The best sequence found before the deadline is returned.
    """
    start = time.time()
    deadline = start + budget_s - 0.5     # small safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # three scales; fractions roughly split the remaining budget
    scales = [(2000, 0.20), (4000, 0.30), (8000, 0.50)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        target_sum = np.sqrt(2.0 * n)

        # initialise candidates (warm‑start from previous scale if any)
        if cur_seq is not None:
            init_pool = [_resize_sequence(cur_seq, n)]
        else:
            init_pool = _initial_candidates(n, target_sum)

        # limit the number of restarts to what fits in the allocated time
        max_restarts = max(1, int(remaining * frac // 30.0))  # approx 30 s each
        max_restarts = min(max_restarts, len(init_pool))

        for i in range(max_restarts):
            a0 = init_pool[i].astype(np.float64, copy=True)

            # ---- gradient descent phase (short, aggressive) ----
            gd_deadline = time.time() + min(25.0, (stage_deadline - time.time()) * 0.4)
            a_gd, gd_score = _annealed_gradient_descent(a0, gd_deadline)

            # keep the best seen so far
            if gd_score < best_score:
                best_score = gd_score
                best_seq = a_gd.copy()
                print(f"[{time.time():.2f}] Scale n={n} restart {i} GD improved best={best_score:.6f}")

            # ---- simulated annealing phase (uses whatever time is left) ----
            if time.time() < stage_deadline:
                a_sa, sa_score = _simulated_annealing(a_gd, stage_deadline)
                cur_score = min(gd_score, sa_score)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_sa.copy()
                    print(f"[{time.time():.2f}] Scale n={n} restart {i} SA improved best={best_score:.6f}")

        # Warm‑start the next (larger) scale with the current best
        cur_seq = best_seq

    # ------------------------------------------------------------------
    # final polishing on the overall best (if time remains)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_seq, final_score = _simulated_annealing(best_seq, deadline)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time():.2f}] Final SA polishing improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # tidy up and return
    # ------------------------------------------------------------------
    if best_seq is None:
        return []

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    # One optional evaluation using the benchmark (for logging)
    try:
        val = evaluate_sequence(arr.tolist())
        print(f"[{time.time():.2f}] Final evaluated benchmark score = {val:.6f}")
    except Exception:
        pass

    return arr.tolist()
