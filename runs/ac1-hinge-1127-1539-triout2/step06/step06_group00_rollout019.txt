"""
Enhanced adversarial step‑function optimizer.

The routine performs three stages:

1. **Grid search over a simple parametrised family** –  
   a symmetric “edge‑heavy’’ shape a_i = (d_i+offset)^{-p}
   mixed with a uniform background.  A coarse grid of
   (p, offset, mix) together with several vector lengths
   is evaluated (using an FFT based convolution) and the
   best candidate is retained.

2. **Many short Adam restarts** – the best candidate from the
   grid and a few random initialisations are refined with a
   projected Adam optimiser that minimises a τ‑soft‑max
   approximation of the convolution peak.

3. **Final long refinement** – the current best vector receives
   the remaining time budget in a dedicated Adam run.

All steps respect the non‑negativity requirement, the
elements are clipped to the interval [0,1000] (the limits of
the black‑box evaluator) and the ℓ₁‑norm is always projected
to the required value √(2·n).  The final sequence is stored in
the global variable ``height_sequence_1`` for reuse by later
runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_to_l1(seq: np.ndarray, n: int) -> np.ndarray:
    """Scale a non‑negative vector to ℓ₁‑norm √(2·n)."""
    target = np.sqrt(2.0 * n)
    seq = np.maximum(seq, 0.0)           # safety
    if seq.sum() == 0:
        return np.full_like(seq, target / n)
    return seq / seq.sum() * target


def _edge_shape(n: int,
                p: float,
                offset: float,
                mix: float,
                rng: np.random.Generator) -> np.ndarray:
    """
    Build a symmetric edge‑heavy shape:
        a_i = mix * (d_i + offset)^{-p} + (1-mix) * 1,
    where d_i = min(i, n-1-i) + 1.
    Afterwards the vector is scaled to the required ℓ₁‑norm.
    """
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0          # distance to nearest edge (≥1)

    edge = (d + offset) ** (-p)
    base = np.ones_like(edge)
    raw = mix * edge + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    return _scale_to_l1(raw, n)


# ----------------------------------------------------------------------
# τ‑soft‑max gradient of the convolution peak
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Return (grad, max_b) where grad = ∂ τ‑soft‑max(b) / ∂a
    and b = a * a (full linear convolution).
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)      # sub‑gradient at the exact max
    else:
        diff = b - max_b
        exps = np.exp(diff / tau)
        mask = exps / exps.sum()                # soft‑max weights

    # gradient w.r.t. a:
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    g = 2.0 * conv[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parameterisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   lr_start: float = 0.4,
                   tau_decay: float = 0.9995,
                   lr_decay: float = 0.9998) -> tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector seen and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # exact ℓ₁‑norm required

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft /= soft.sum()
    p = np.log(soft)                    # logits (no explicit projection later)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr_start

    # initialise temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- recover a from logits (log‑softmax) ----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ---- convolution & gradient (soft‑max) ----
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:           # keep the *hard* peak
            best_peak = peak
            best_a = a.copy()

        # ---- gradient w.r.t. logits (chain rule) ----
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)          # shape (n,)

        # optional clipping for stability
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedules ----
        tau *= tau_decay
        lr *= lr_decay

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time
    budget, keeps the best sequence found and stores it in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 3.0    # small safety margin

    # ------------------------------------------------------------------
    # reuse a previous construction if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Helper: create a random feasible vector of given length
    # ------------------------------------------------------------------
    def _random_initial(n):
        style = rng.choice(['uniform', 'dirichlet', 'edge'])
        if style == 'uniform':
            a = np.full(n, np.sqrt(2.0 * n) / n)
        elif style == 'dirichlet':
            a = rng.random(n)
            a = _scale_to_l1(a, n)
        else:  # edge‑heavy
            a = _edge_shape(n,
                            p=rng.uniform(0.2, 1.0),
                            offset=rng.uniform(0.0, 5.0),
                            mix=rng.uniform(0.0, 1.0),
                            rng=rng)
        return a

    # ------------------------------------------------------------------
    # Stage 1 – exhaustive grid over the parametrised family
    # ------------------------------------------------------------------
    # candidate lengths (choose a few convenient powers of two)
    length_candidates = [1500, 2000, 3000, 5000, 8000]

    # coarse grids for the parameters
    p_grid = np.linspace(0.25, 0.95, 13)          # 0.25 … 0.95
    offset_grid = [0.0, 0.5, 1.0, 2.0, 5.0]
    mix_grid = [0.0, 0.3, 0.5, 0.7, 1.0]

    for n in length_candidates:
        # keep a short time budget for the exhaustive part
        slice_deadline = min(time.time() + 8.0, end_time - 30.0)
        for p in p_grid:
            for off in offset_grid:
                for mix in mix_grid:
                    a = _edge_shape(n, p, off, mix, rng)
                    b_peak = _full_convolution_fft(a).max()
                    if b_peak < best_score:
                        best_score = float(b_peak)
                        best_seq = a.copy()
                        best_n = n
                        print(f"[grid] n={n} p={p:.2f} off={off:.1f} mix={mix:.1f} "
                              f"new best={best_score:.6f}")

                    if time.time() > slice_deadline:
                        break
                if time.time() > slice_deadline:
                    break
            if time.time() > slice_deadline:
                break
        if time.time() > end_time - 30.0:
            break

    # ------------------------------------------------------------------
    # Stage 2 – many short Adam restarts (exploration)
    # ------------------------------------------------------------------
    short_budget = 13.0          # seconds per short run
    attempts = 0
    while time.time() + short_budget < end_time - 20.0:
        attempts += 1
        # choose a length (prefer the one that gave the best grid result)
        if rng.random() < 0.7 and best_n is not None:
            n = best_n
        else:
            n = rng.integers(1500, 12000)

        # initialise
        if rng.random() < 0.25 and best_seq is not None and best_seq.shape[0] == n:
            a_start = best_seq.copy()
        elif base_seq is not None and rng.random() < 0.15:
            # warm‑start from the globally stored sequence (scaled)
            a_start = np.asarray(base_seq, dtype=float)
            if a_start.shape[0] != n:
                # simple linear interpolation for scaling
                idx_old = np.linspace(0, a_start.shape[0] - 1, n)
                lo = np.floor(idx_old).astype(int)
                hi = np.clip(lo + 1, 0, a_start.shape[0] - 1)
                w = idx_old - lo
                a_start = (1.0 - w) * a_start[lo] + w * a_start[hi]
            a_start = _scale_to_l1(a_start, n)
        else:
            a_start = _random_initial(n)

        slice_deadline = min(time.time() + short_budget, end_time - 20.0)
        a_opt, peak = _adam_optimize(a_start,
                                     deadline=slice_deadline,
                                     rng=rng,
                                     lr_start=0.4,
                                     tau_decay=0.9996,
                                     lr_decay=0.9999)

        if peak < best_score:
            best_score = float(peak)
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart] attempt={attempts} n={n}  new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final long refinement on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 8.0:
            # ensure exact ℓ₁‑norm before the final run
            best_seq = _scale_to_l1(best_seq, best_seq.shape[0])
            a_refined, peak = _adam_optimize(best_seq,
                                             deadline=end_time,
                                             rng=rng,
                                             lr_start=0.3,
                                             tau_decay=0.9997,
                                             lr_decay=0.9999)
            if peak < best_score:
                best_score = float(peak)
                best_seq = a_refined.copy()
                print(f"[final‑refine] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the provided black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f} "
              f"length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
