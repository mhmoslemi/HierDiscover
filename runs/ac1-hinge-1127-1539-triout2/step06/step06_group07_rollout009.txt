def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` by directly optimising a smooth
    approximation of the max‑convolution using L‑BFGS‑B on a
    soft‑max (log‑sum‑exp) surrogate.  The smoothness parameter
    ``alpha`` is increased gradually (continuation) and the
    optimisation is performed for several problem sizes (2000,
    4000, 8000).  The best vector found is finally refined with a
    short projected sub‑gradient descent and returned as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf

    # If a previous construction exists, use it as a warm start
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=float)
            if seq0.size > 0:
                best_seq = seq0.copy()
                best_score = evaluate_sequence(best_seq.tolist())
                print(f"[init] previous sequence score={best_score:.6f}")
        except Exception:
            best_seq = None
            best_score = np.inf

    # Problem sizes we will try – larger sizes give a richer
    # search space at the cost of a longer convolution.
    sizes = [2000, 4000, 8000]

    # Continuation schedule for the smooth approximation.
    # The values are chosen empirically; a larger final alpha brings
    # the surrogate close to the true max norm.
    alpha_schedule = [0.5, 2.0, 5.0, 20.0, 100.0, 500.0, 2000.0]

    # Main search loop
    for idx, n in enumerate(sizes):
        if time.time() >= deadline:
            break

        # Allocate a share of the remaining time to this size.
        # The share shrinks as we go to larger problems.
        remaining = deadline - time.time()
        share = 0.35 if idx < len(sizes) - 1 else 0.30   # last size gets the rest
        stage_deadline = time.time() + remaining * share

        S = np.sqrt(2.0 * n)

        # Initialise the vector – up‑sample the best one we have,
        # otherwise start from a random point on the simplex.
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
            a0 = a0 / a0.sum() * S
        else:
            a0 = rng.random(n)
            a0 = a0 / a0.sum() * S

        # Convert the simplex point to log‑it parameters used by the
        # soft‑max parametrisation.
        v = np.log(a0 + 1e-30)

        # Continuation: run a short L‑BFGS for each alpha.
        for alpha in alpha_schedule:
            if time.time() >= stage_deadline:
                break

            # Set the global parameters that the objective closure reads.
            global _current_alpha, _current_S
            _current_alpha = alpha
            _current_S = S

            # The function returns (value, gradient) – L‑BFGS‑B will
            # treat the gradient as exact because we set ``jac=True``.
            def fun(v):
                val, grad = _smooth_objective_and_grad(v)
                return val, grad

            # Run the optimiser – the iteration budget is kept modest
            # because each iteration costs two FFTs.
            res = scipy.optimize.minimize(fun,
                                          v,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': 200, 'disp': False})
            v = res.x

            # Convert back to the simplex and evaluate the true objective.
            a = _softmax_to_a(v, S)
            max_conv = np.max(_full_convolution(a))
            if max_conv < best_score:
                best_score = max_conv
                best_seq = a.copy()
                print(f"[stage] n={n} alpha={alpha:.1f} new best={best_score:.6f}")

        # A quick projected sub‑gradient pass can squeeze out a few
        # more decimals after the smooth optimisation has converged.
        if time.time() < stage_deadline:
            a_start = best_seq if best_seq.shape[0] == n else _upsample(best_seq, n)
            a_start = a_start / a_start.sum() * S
            a_opt, sc_opt = _optimize_subgrad(a_start, stage_deadline, rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[subgrad] n={n} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and external evaluation
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist() if best_seq is not None else [1.0]
    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} (internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq

def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)

    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad

import time
import numpy as np
import scipy.optimize

# ---- Helper functions -------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of sum_i w_i * (a*a)_i w.r.t a (size n)."""
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1   # 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad

def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)

def _softmax_to_a(v: np.ndarray, S: float) -> np.ndarray:
    """Convert log‑it vector v to a simplex vector of total mass S."""
    vmax = np.max(v)
    e = np.exp(v - vmax)
    a = S * e / e.sum()
    return a

# globals used by the smooth‑max objective
_current_alpha = 1.0
_current_S = 1.0

def _smooth_objective_and_grad(v: np.ndarray):
    """Return (smooth‑max value, gradient) for the current alpha/S."""
    alpha = _current_alpha
    S = _current_S
    a = _softmax_to_a(v, S)
    conv = _full_convolution(a)

    # log‑sum‑exp smooth max
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    sum_exp = exp_shifted.sum()
    val = (max_scaled + np.log(sum_exp)) / alpha

    # softmax weights for the gradient
    w = exp_shifted / sum_exp
    g_a = _conv_with_weights(a, w)

    # chain rule: ∂f/∂v_j = a_j (g_j - (a·g)/S)
    dot_ga = np.dot(g_a, a)
    grad_v = a * (g_a - dot_ga / S)
    return val, grad_v

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
