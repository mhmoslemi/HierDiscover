def _hill_climb_targeted(a0, b0, deadline):
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.2  # larger initial temperature
    cooling = 0.9999  # slower
    iteration = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # compute contributions mask and values
        idx = np.arange(n)
        j_arr = k_max - idx
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        # source selection
        if rng.random() < 0.3:
            i = np.argmax(contrib)  # deterministic biggest
        else:
            total = contrib.sum()
            if total > 0:
                prob_src = contrib / total
                i = rng.choice(n, p=prob_src)
            else:
                prob_src = a / a.sum()
                i = rng.choice(n, p=prob_src)

        # destination selection: pick index with smallest contribution to peak
        if rng.random() < 0.3:
            # deterministic minimal
            inv = np.ones_like(contrib) * np.max(contrib) - contrib
            j = np.argmin(inv)  # i.e. index with max contribution? Wait invert
            # Actually we want smallest contribution: choose argmin contrib
            j = np.argmin(contrib + 1e-12)  # add small epsilon to avoid ties
        else:
            inv = contrib.max() - contrib + 1e-12
            prob_dst = inv / inv.sum()
            j = rng.choice(n, p=prob_dst)

        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # compute new convolution analytically (like earlier)
        b_new = b.copy()
        # update linear parts
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[j:j+n] += 2.0 * delta * a

        # quadratic corrections
        k2i = 2*i
        k2j = 2*j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta*delta
        if k2j < b_new.size:
            b_new[k2j] += delta*delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta*delta

        new_max = b_new.max()

        # acceptance
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            # Metropolis
            if T > 0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)
            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        # temperature decay
        T *= cooling

        # occasional jitter to avoid stall
        iteration += 1
        if iteration % 2000 == 0:
            # small jitter across all entries
            jitter = rng.normal(scale=target_sum*0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            b = _full_convolution(a)
            cur_max = b.max()
        # optional early exit if cur_max equals best_max below threshold.

    return best_a, best_max

"""
Hybrid optimisation for the self‑convolution peak problem.

* Initialise several random or simple deterministic step functions on the
  ℓ¹‑simplex Σa = √(2 n).
* For each length run a short projected gradient descent on a smooth
  log‑sum‑exp surrogate of the max‑convolution, followed by an aggressive
  targeted hill‑climber that moves mass from indices that contribute heavily
  to the current peak toward indices that contribute little.
* The hill‑climber uses a higher initial temperature and slower cooling than
  the reference implementation, performs occasional global jitter, and also
  tries deterministic source/destination moves to escape plateaus.
* Lengths are increased geometrically (2000 → 5000 → 8000 → 12000 → 18000);
  each new length is seeded by linear up‑sampling of the best solution so far.
* After the schedule a final polishing phase repeatedly jitters and re‑runs
  the hill‑climber with the remaining budget.
* All steps respect the supplied time budget (default 1000 s) and return the
  best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# Random / deterministic initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    # small Gaussian noise and final projection onto the simplex
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _deterministic_initial_sequence(n: int) -> np.ndarray:
    """A few hand‑crafted patterns (triangle, cosine, linear) projected to the simplex."""
    target = np.sqrt(2.0 * n)
    idx = np.arange(n, dtype=float)

    # triangle shape: min(i+1, n-i)
    tri = np.minimum(idx + 1, n - idx)
    tri = _project_to_simplex(tri, target)

    # cosine shape: 1 - cos(π * i / (n-1))
    cos = 1 - np.cos(np.pi * idx / (n - 1))
    cos = _project_to_simplex(cos, target)

    # linear ramp
    lin = idx + 1
    lin = _project_to_simplex(lin, target)

    # choose the one with smallest initial max‑convolution
    candidates = [tri, cos, lin]
    best = min(candidates, key=_max_convolution_fft)
    return best


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of (1/β)·log Σ exp(β·b) where b = a*a."""
    n = a.shape[0]

    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on the smoothed maximum."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    # step size base (a bit larger than the reference code)
    step_base = target_sum * 0.18
    momentum = np.zeros_like(a)
    rho = 0.92

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to break stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that continuously reduces the maximal convolution entry.
    Uses a higher initial temperature, slower cooling and occasional
    deterministic source/destination choices to escape local minima.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # start warmer than the reference (10 % of the current peak)
    T = cur_max * 0.10 + 1e-12
    cooling = 0.9998

    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contributions of each index to the current peak
        idx = np.arange(n)
        j_arr = k_max - idx
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        # source index – usually the biggest contributor
        if rng.random() < 0.30:
            i = int(np.argmax(contrib))
        else:
            total = contrib.sum()
            if total > 0.0:
                prob_src = contrib / total
                i = int(rng.choice(n, p=prob_src))
            else:
                prob_src = a / a.sum()
                i = int(rng.choice(n, p=prob_src))

        # destination index – preferably a small contributor
        if rng.random() < 0.30:
            j = int(np.argmin(contrib + 1e-15))
        else:
            inv = contrib.max() - contrib + 1e-12
            prob_dst = inv / inv.sum()
            j = int(rng.choice(n, p=prob_dst))

        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---------- analytical convolution update ----------
        b_new = b.copy()
        # linear part:   -2·δ·a shifted by i, +2·δ·a shifted by j
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---------- acceptance test ----------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            # Metropolis step
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            # Guard against tiny negatives caused by rounding errors
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max

            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling
        it += 1

        # occasional global jitter to jump out of flat plateaus
        if it % 3000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            b = _full_convolution(a)
            cur_max = b.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improvement: {best_max:.6f}")

    return best_a, best_max

        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Report the official evaluation value.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
