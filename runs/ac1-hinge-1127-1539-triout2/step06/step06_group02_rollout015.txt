def _gradient_descent_softmax(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    # Use parameters
    best_a = a.copy()
    best_val = _score(a)
    beta = 1.0
    max_beta = 1e5  # maybe higher
    step_base = target_sum * 0.05
    rng = np.random.default_rng()
    it = 0
    # Use Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = step_base / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat

        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] softmax GD β={beta:.1f} score={best_val:.6f}")

        # increase beta slowly
        if it % 100 == 0 and beta < max_beta:
            beta = min(beta * 1.3, max_beta)

    return best_a, best_val

def _subgradient_descent_max(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    it = 0
    # Learning rate schedule
    base_lr = target_sum * 0.05
    while time.time() < deadline:
        # compute conv
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_val = conv.max()
        kmaxs = np.where(conv == max_val)[0]
        # compute subgradient
        grad = np.zeros_like(a)
        for k in kmaxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
        # step size
        lr = base_lr / np.sqrt(it + 1)
        a = a - lr * grad
        a = np.maximum(a, 0)
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} score={best_val:.6f}")

        # occasional jitter
        if it % 100 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

"""
Simple power‑law grid search + two‑phase continuous refinement.

1️⃣  Grid search:
    For a few problem sizes (800 … 10 000) generate sequences
    a_i = (i+offset)^{-p}   (i = 1,…,n)  with
        p ∈ [0.55, 0.75]  (step ≈0.01)
        offset ∈ [0, 5]   (step 0.5)
    Normalise each candidate so that Σa_i = √(2·n) (the evaluation
    function then equals the maximal convolution entry).
    Keep the best candidate seen so far.

2️⃣  Continuous refinement:
    a)  Projected Adam on a soft‑max approximation of the max‑convolution.
        The temperature β is increased up to a large value, giving an
        increasingly accurate gradient of the true max.
    b)  Sub‑gradient descent on the exact max‑convolution (the gradient
        of the active entry is 2·a_{k−i} for the index k attaining the
        maximum).  Both steps keep the vector on the ℓ¹‑simplex.

The whole routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence (trailing zeros stripped).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT, O(n log n))."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective.
    If Σa = √(2·n) the function equals max(a*a); otherwise the scaling is
    taken into account automatically.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Power‑law family (with optional offset)
# ----------------------------------------------------------------------


def _power_law_offset_sequence(n: int, p: float, offset: float, target_sum: float) -> np.ndarray:
    """Generate (i+offset)^{-p}, i=1..n, then renormalise to target_sum."""
    base = np.arange(1, n + 1, dtype=np.float64)
    a = (base + offset) ** (-p)
    a = a * (target_sum / a.sum())
    return a


# ----------------------------------------------------------------------
# Gradient of a soft‑max approximation of max(convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, raw_convolution) where grad has the same shape as a.
    grad = d/d a  Σ_k softmax_β(b_k) * b_k ,  b = a*a .
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient: 2 * (a_rev * p)  on the central window
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _gradient_descent_softmax(a0: np.ndarray, deadline: float):
    """
    Projected Adam optimiser on the soft‑max surrogate.
    β starts at 1 and is multiplied by 1.3 every 100 iterations,
    up to a large cap, making the surrogate approach the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 1e5

    step_base = target * 0.05
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = step_base / (np.sqrt(v_hat) + eps)
        a = a - lr * m_hat

        # project back onto the simplex
        a = _project_to_simplex(a, target)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target * 0.0005 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD β={beta:.1f} score={best_val:.6f}")

        # increase temperature slowly
        if it % 100 == 0 and beta < max_beta:
            beta = min(beta * 1.3, max_beta)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent_max(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on max(a*a).
    The sub‑gradient at the active index k is 2·a_{k−i}.
    """
    a = a0.copy()
    n = a.shape[0]
    target = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # compute full convolution
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_val = conv.max()
        kmax = np.where(conv == max_val)[0]

        # sub‑gradient: accumulate contributions from every active k
        grad = np.zeros_like(a)
        for k in kmax:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # normalise step size
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = (target * 0.08) / np.sqrt(it + 1) / norm
            a -= step * grad

        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it} score={best_val:.6f}")

        # occasional perturbation
        if it % 200 == 0:
            jitter = rng.normal(scale=target * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
    ``evaluate_sequence``.  The strategy is:

    * coarse grid search over power‑law exponents and offsets,
    * a final Adam‑based soft‑max gradient phase,
    * a sub‑gradient phase on the true max‑convolution.

    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Grid search
    # ------------------------------------------------------------------
    # problem sizes (larger = more freedom, still cheap for FFT)
    sizes = [800, 1500, 3000, 5000, 8000, 10000]
    if "height_sequence_1" in globals():
        hs = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if hs.size > 0:
            sizes.append(hs.size)

    p_vals = np.linspace(0.55, 0.75, 21)          # step ≈0.01
    off_vals = np.linspace(0.0, 5.0, 11)          # step 0.5

    for n in sizes:
        if time.time() >= deadline - 8.0:
            break
        target_sum = np.sqrt(2.0 * n)
        base = np.arange(1, n + 1, dtype=np.float64)

        local_best_seq = None
        local_best_val = float("inf")

        for p in p_vals:
            # pre‑compute i^{-p} for all offsets at once – cheap
            for off in off_vals:
                a_raw = (base + off) ** (-p)
                a = a_raw * (target_sum / a_raw.sum())
                val = _score(a)
                if val < local_best_val:
                    local_best_val = val
                    local_best_seq = a.copy()

        # keep the globally best candidate
        if local_best_seq is not None and local_best_val < best_score:
            best_score = local_best_val
            best_seq = local_best_seq.copy()
            print(f"[{time.time():.1f}] size={n} grid best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Continuous refinement (soft‑max Adam)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        seq_ref, val_ref = _gradient_descent_softmax(best_seq, deadline)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = seq_ref.copy()
            print(f"[{time.time():.1f}] after Adam GD score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        seq_ref, val_ref = _subgradient_descent_max(best_seq, deadline)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = seq_ref.copy()
            print(f"[{time.time():.1f}] after sub‑grad score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final tidy‑up (strip trailing zeros)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.empty(0, dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
