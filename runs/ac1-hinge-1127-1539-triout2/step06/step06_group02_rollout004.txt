"""
Propose a low‑peak self‑convolution step function.

The algorithm works in three stages for each problem size:

1. **Smart initialisation** – try several power‑law exponents (and a random
   Dirichlet start) and keep the one with the smallest ‖a*a‖∞ after normalising
   Σa = √(2·n).

2. **Projected Adam optimisation** – minimise a smooth approximation of the
   max‑convolution using the soft‑max (log‑sum‑exp) with an annealing temperature
   β that grows up to 10⁶.  Gradients are obtained analytically via the FFT.
   After each Adam step the vector is projected onto the ℓ¹‑simplex.

3. **Focused simulated annealing** – repeatedly move a small amount of mass
   from indices that heavily contribute to the current worst convolution entry
   to indices that contribute little, occasionally perform short‑window smoothing
   and random swaps.  Acceptance follows a standard Metropolis rule with an
   exponentially decaying temperature.

Multiple problem sizes (up to ~12 000 coefficients) are tried within the
time budget; the best sequence found overall is returned (trailing zeros are
stripped).  All heavy numeric work uses FFTs, so each iteration is O(n log n).

The implementation is self‑contained, uses only NumPy and the provided
`evaluate_sequence`, and respects the 1000 s runtime limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using FFT (zero‑padding to the next power of two).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Score = max(a*a) for a sequence that already satisfies Σa=√(2n)."""
    return _max_convolution_fft(a)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """
    Generate a (i+1)^(-p) sequence of length n and renormalise it to have
    Σa = target_sum.
    """
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns (gradient, raw convolution array) where gradient has the same shape
    as ``a``.  The gradient is derived from the identity
        d/d a  log‑sum‑exp(β·b) = (Aᵀ·p)·2
    with ``p`` the soft‑max weights over the convolution entries.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a (see analysis above)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _adam_projected_descent(
    a0: np.ndarray,
    deadline: float,
    start_beta: float = 1.0,
    max_beta: float = 1e6,
    beta_factor: float = 1.5,
    base_lr_factor: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the soft‑max approximation of the max‑convolution.
    The temperature β starts at ``start_beta`` and is multiplied by ``beta_factor``
    every few iterations until it reaches ``max_beta``.
    The learning rate is scaled by ``base_lr_factor``·√(2·n).
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # hyper‑parameters
    lr0 = target_sum * base_lr_factor
    beta = start_beta
    it = 0
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of soft‑max (log‑sum‑exp) of the convolution
        g, _ = _gradient_softmax(a, beta)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 400 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=n
            )
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # update best
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.1f}] Adam β={beta:.1f} score={best_val:.6f}"
            )

        # temperature schedule
        if beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing_focused(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 1.0,
    end_temp: float = 1e-5,
    max_frac_start: float = 0.40,
    max_frac_end: float = 0.005,
    max_window: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Focused simulated‑annealing refinement.
    Moves mass from indices that strongly contribute to the current worst
    convolution entry to indices that contribute little, plus occasional smoothing
    and swaps.  Temperature and maximal transferred fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()

    # current full convolution (used to locate the worst entry)
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate the current worst convolution entry
        max_b = conv.max()
        k_max = int(np.argmax(conv))

        # sensitivity (contribution) of each coordinate to that entry
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens[idx] = a[k_max - idx]          # a_j where i+j = k_max
        weight = a * sens                    # contribution a_i * a_{k_max-i}

        # --------------------------------------------------------------
        # Propose a move
        # --------------------------------------------------------------
        move_type = None

        # 70 % mass‑transfer moves
        if rng.random() < 0.70:
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            # receiver aims at a low‑contributing index
            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = (
                rng.choice(low_idx) if low_idx.size else rng.integers(n)
            )

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # 20 % local smoothing moves
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        # 10 % random swaps
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # --------------------------------------------------------------
        # Evaluate the new state (single FFT)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to shake out deep minima
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            # renormalise to keep Σa unchanged (exact – sum is invariant)
            a = a * (a.sum() / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

        if "height_sequence_1" in globals():
            seq0 = np.array(
                globals()["height_sequence_1"], dtype=np.float64
            )
            if seq0.size >= n:
                seq = seq0[:n].copy()
            else:
                seq = np.pad(seq0, (0, n - seq0.size), "constant")
            seq = _project_to_simplex(seq, target_sum)
            sc_seq = _score(seq)
            if sc_seq < best_init_score:
                best_init_score = sc_seq
                best_init = seq

        a = best_init.copy()

        # ------------------------------------------------------------------
        # 2. Projected Adam (soft‑max) – two phases with different β ranges
        # ------------------------------------------------------------------
        # Phase A: moderate β (up to 4 000)
        gd_deadline = time.time() + min(30.0, time_for_this * 0.20)
        a, score_a = _adam_projected_descent(
            a,
            gd_deadline,
            start_beta=1.0,
            max_beta=4096.0,
            beta_factor=1.5,
            base_lr_factor=0.12,
        )

        # Phase B: very high β (up to 1 000 000) for a tighter approximation
        gd_deadline2 = time.time() + min(30.0, time_for_this * 0.20)
        a, score_b = _adam_projected_descent(
            a,
            gd_deadline2,
            start_beta=1000.0,
            max_beta=1e6,
            beta_factor=2.0,
            base_lr_factor=0.08,
        )

        # ------------------------------------------------------------------
        # 3. Focused simulated annealing
        # ------------------------------------------------------------------
        sa_deadline = time.time() + max(0.0, time_for_this - (gd_deadline2 - start_time))
        a, score_sa = _simulated_annealing_focused(
            a,
            sa_deadline,
            start_temp=1.0,
            end_temp=1e-5,
            max_frac_start=0.45,
            max_frac_end=0.004,
            max_window=30,
        )

        final_score = _score(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] size={n} → new overall best score={best_score:.6f}"
            )

    # ----------------------------------------------------------------------
    # Final refinement on the globally best candidate (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n_best = best_seq.size
        final_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.30)
        best_seq, _ = _adam_projected_descent(
            best_seq,
            final_deadline,
            start_beta=1.0,
            max_beta=1e6,
            beta_factor=1.5,
            base_lr_factor=0.10,
        )

    # Strip trailing zeros – they do not affect the evaluation.
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(
        f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}"
    )
    return best_seq.tolist()
