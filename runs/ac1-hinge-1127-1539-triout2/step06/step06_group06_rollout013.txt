def lp_step(a, target_sum, t=0.2):
    n = len(a)
    max_conv = _max_convolution_fft(a)
    # Build matrix A
    c = np.concatenate([a, np.zeros(n-1, dtype=a.dtype)])
    r = np.concatenate([a[:1], np.zeros(n-1, dtype=a.dtype)])
    A = toeplitz(c, r)[:2*n-1, :n]  # shape
    # Build LP: maximize sum(b)
    # Use linprog with -c objective
    lin_res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, max_conv), bounds=(0, None), method='highs')

    # Mix

def _optimize_by_lp(a0, deadline):
    a = a0.astype(np.float64, copy=True)
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    # project onto simplex
    a = _project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_score = _score(a)
    t = 0.2 # mixing parameter
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute current max conv
        max_b = _max_convolution_fft(a)
        # build A matrix
        c = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])
        r = np.concatenate([a[:1], np.zeros(n-1, dtype=np.float64)])
        A = toeplitz(c, r)[:2*n-1, :n]  # dense
        # Solve LP
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, max_b), bounds=(0, None), method='highs')
        if not res.success:
            # If LP fails, break
            print(f"LP failed at iteration {it}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        scale = target_sum / sum_b
        g = b * scale
        # mix
        a_new = (1 - t) * a + t * g
        # optional jitter
        if it % 20 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum*0.001, size=n)
            a_new += jitter
            a_new = np.maximum(a_new, 0.0)
        # project onto simplex to keep sum exact
        a_new = _project_to_simplex(a_new, target_sum)
        # compute new score
        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] it={it} score={best_score:.6f}")
        # Accept if new max conv decreased else keep old (optional)
        if new_score < best_score*1.001: # accept minor updates
            a = a_new
        # else maybe keep old a (skip)
        it += 1
        # optionally adapt t (decrease)
        if it % 50 == 0:
            t = max(0.05, t * 0.9)
    return best_a, best_score

import time, numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# Helper functions
def _max_convolution_fft(a):
    # uses FFT
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(a):
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:  # shouldn't happen
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v, s):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _power_law_sequence(n, p, target_sum):
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return a * (target_sum / a.sum())

def _lp_refine(a, deadline, mix=0.2):
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # current bound
        M = _max_convolution_fft(a)
        # build toeplitz matrix for constraints a * b <= M
        c = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])
        r = np.concatenate([a[:1], np.zeros(n-1, dtype=np.float64)])
        A = toeplitz(c, r)[: 2*n-1, : n]
        # solve LP: maximize sum(b)
        res = linprog(c=np.full(n, -1.0), A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
        if not res.success:
            # if LP fails, break out
            print(f"LP failed at iteration {it} ({res.message})")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        scale = target_sum / sum_b
        g = b * scale
        # mix old a and g
        a_new = (1 - mix) * a + mix * g
        # occasional jitter
        if it % 20 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a_new += jitter
        # project to simplex to keep sum exactly target_sum
        a_new = _project_to_simplex(a_new, target_sum)
        # evaluate
        cur_score = _score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP it={it} score={best_score:.6f}")
        # accept if improvement, else continue with old a (or occasionally accept)
        if cur_score < best_score * 1.001:
            a = a_new
        # gradually reduce mix to fine tune
        if it % 30 == 0 and it > 0:
            mix = max(0.02, mix * 0.95)
        it += 1
    return best_a, best_score

"""
Linear‑programming (Frank‑Wolfe) optimiser for the step‑function benchmark.

The objective of `evaluate_sequence` is 2·n·max(a*a)/(sum a)².
Since scaling cancels, we may fix the ℓ¹‑norm to √(2·n) and minimise
the maximum entry of the aperiodic convolution a*a.

Given a current sequence f, the linear programme

    maximise   Σ b_j
    subject to  (f * b)[k] ≤ M   for all k,
                b_j ≥ 0

(with M = max(f*f)) finds a direction that can accommodate more mass
under the same convolution bound.  After solving the LP we scale the
solution to the required ℓ¹‑norm and mix it with f.  Repeating this
step (with a small mixing factor) provably decreases the bound M and
hence the benchmark score.

The routine works on three increasing lengths (≈1500 → 3000 → 6000)
re‑using the best solution of the previous scale as a warm start.
All numbers are forced to be non‑negative and projected onto the
simplex Σa_i = √(2·n) after each update.
The total running time is limited by the supplied budget (default 1000 s).
"""
import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Benchmark score 2·n·max(a*a)/(sum a)²."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """(i+1)^(-p) sequence of length n, normalised to `target_sum`."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation step (LP‑based Frank‑Wolfe)
# ----------------------------------------------------------------------


def _lp_refine(initial: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described in the module docstring and
    mix the solution with the current iterate.
    Returns the best sequence found and its score.
    """
    a = initial.astype(np.float64, copy=True)
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # make sure we start on the simplex
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    mix = 0.25                     # initial mixing factor
    it = 0
    no_improve = 0                # early‑stop counter

    while time.time() < deadline:
        # current bound M = max(f*f)
        M = _max_convolution_fft(a)

        # Build the Toeplitz matrix that represents the convolution a * b
        c = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        r = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(c, r)[: 2 * n - 1, : n]   # shape (2n‑1) × n

        # LP: maximise Σ b_j   s.t. A·b ≤ M , b ≥ 0
        lp_res = linprog(
            c=np.full(n, -1.0),         # maximise → minimise –1ᵀb
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=(0, None),
            method="highs",
        )

        if not lp_res.success:
            # LP failure – fall back to simple gradient descent style jitter
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a = _project_to_simplex(a + jitter, target_sum)
            continue

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Scale to keep the ℓ¹‑norm fixed
        scale = target_sum / sum_b
        g = b * scale                     # feasible candidate with same norm

        # Direct replacement if it already improves the bound
        cand_score = _score(g)
        if cand_score < best_score:
            best_a = g.copy()
            best_score = cand_score
            a = g
            it += 1
            no_improve = 0
            print(f"[{time.time():.2f}] LP it={it:4d} direct replace score={best_score:.6f}")
            continue

        # Otherwise mix old iterate with the candidate
        a_new = (1.0 - mix) * a + mix * g

        # occasional random jitter to escape stagnation
        if it % 30 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=n)
            a_new += jitter

        a_new = _project_to_simplex(a_new, target_sum)

        cur_score = _score(a_new)
        if cur_score < best_score:
            best_a = a_new.copy()
            best_score = cur_score
            a = a_new
            no_improve = 0
            print(f"[{time.time():.2f}] LP it={it:4d} mixed replace score={best_score:.6f}")
        else:
            no_improve += 1
            # occasionally still accept a non‑improving step to keep moving
            if rng.random() < 0.02:
                a = a_new

        # slowly shrink the mixing factor – fine‑grained updates later
        if it % 20 == 0 and it > 0:
            mix = max(0.02, mix * 0.93)

        if no_improve >= 40:   # stop if no progress for a while
            break

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (height list) with a low benchmark score.
    Three increasing problem sizes are explored; each scale receives a
    fraction of the remaining time budget and is optimised with the LP‑based
    Frank‑Wolfe routine defined above.
    """
    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # (length, fraction of remaining time) pairs
    scales = [(1500, 0.25), (3000, 0.35), (6000, 0.40)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # initialise the sequence for the current scale
        if cur_seq is not None:
            init_seq = _resize_sequence(cur_seq, n)
        else:
            if "height_sequence_1" in globals():
                init_seq = _resize_sequence(
                    np.array(globals()["height_sequence_1"], dtype=np.float64), n
                )
            else:
                # random power‑law exponent in a reasonable range
                p = np.random.uniform(0.4, 0.8)
                init_seq = _power_law_sequence(n, p, np.sqrt(2.0 * n))

        # run the LP‑based refinement
        seq_opt, score_opt = _lp_refine(init_seq, stage_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(
                f"[{time.time():.2f}] Scale n={n} improved best score={best_score:.6f}"
            )

        # warm‑start the next (larger) scale with the current optimum
        cur_seq = seq_opt

    # ------------------------------------------------------------------
    # final post‑processing
    # ------------------------------------------------------------------
    if best_seq is None:
        return []

    arr = np.array(best_seq, dtype=float)
    nonzero = np.where(arr > 1e-12)[0]
    if nonzero.size:
        arr = arr[: nonzero[-1] + 1]

    final_seq = arr.tolist()

    # one final evaluation with the supplied benchmark (for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
