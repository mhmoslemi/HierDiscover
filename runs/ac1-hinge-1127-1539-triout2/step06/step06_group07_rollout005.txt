"""
Propose a step‑function (non‑negative vector) that minimises the value
returned by ``evaluate_sequence``.

The search proceeds in three phases:

1. **Exploratory restarts** on a modest size (n≈800).  
   Each restart runs a combined Adam → projected sub‑gradient optimiser
   (the same smooth‑max machinery as the reference code) and a brief
   genetic refinement.  The best vector among the restarts is kept.

2. **Linear‑programming refinement** of that best vector.  
   With the current maximum self‑convolution `M` we solve
   ``max Σ g_i  s.t.  f∗g ≤ M,  g≥0`` (a simple LP).  The solution is
   scaled to the simplex and mixed with the current iterate.  Repeating
   this step strictly reduces the worst convolution entry.

3. **Upsampling & final polishing** on the target length (n=2000).  
   The refined small‑scale vector is linearly interpolated, projected onto
   the L¹‑simplex (∑a = √(2n)) and run through the same combined optimiser
   (and a tiny GA) with the remaining time budget.

All operations respect the 1000 s time limit; the code leaves a small
safety margin before the harness terminates the run.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both ``a`` (len n) and ``w`` (len 2·n‑1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (smooth approximation of max)
# ----------------------------------------------------------------------


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """w_i = exp(alpha·conv_i) / Σ_j exp(alpha·conv_j)  (stable)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a to length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth max‑norm
# ----------------------------------------------------------------------


def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # tuned hyper‑parameters
    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5          # start smoothness
    alpha_max = 2000.0   # final sharpness

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a, S)

        # occasional random mass swap (exploration)
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.02
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase temperature
        alpha = min(alpha * 1.06, alpha_max)

        # cheap stagnation check (nothing else – we just keep looping)
        if t % 6000 == 0 and cur_score >= best_score * (1 + 1e-9):
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (acts on the true max‑norm)
# ----------------------------------------------------------------------


def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that directly reduces the worst entry."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.6
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # random perturbation to escape plateaus
        if rng.random() < 0.08:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.025
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.2:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

        if it % 9000 == 0 and cur_score >= best_score * (1 + 1e-9):
            pass

    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimiser (Adam → sub‑gradient)
# ----------------------------------------------------------------------


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.75) -> tuple[np.ndarray, float]:
    """Run Adam for a fraction of the allocated time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_s, sc_s = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_f, sc_f = _optimize_subgrad(a_s, deadline, rng)
        if sc_f < sc_s:
            return a_f, sc_f
    return a_s, sc_s


# ----------------------------------------------------------------------
# Genetic algorithm optimiser (convex recombination + mutation)
# ----------------------------------------------------------------------


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very light steady‑state GA on the simplex."""
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 20 if n <= 2000 else 12
    pop = []
    scores = []
    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        pop.append(a)
        scores.append(np.max(_full_convolution(a)))
    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # pick two distinct parents
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)

            # convex recombination
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            # mutation
            if rng.random() < 0.38:
                noise = rng.normal(scale=child.mean() * 0.05 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # occasional injection of a fresh random individual
        if rng.random() < 0.12:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # break a little early if we are about to exceed the deadline
        if time.time() + 0.1 >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (max‑convolution bound)
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 20) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve
        max Σ g_i   s.t.   (a * g) ≤ M ,  g ≥ 0
    where M = max(a * a).  The solution g is scaled to the simplex and
    convexly mixed with a.  The process stops when no improvement is seen
    or the deadline is reached.
    """
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(best_a)
        M = conv.max()

        # build A_ub such that (a * g)_k = Σ_i a_i g_{k-i}
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = best_a
        b = np.full(rows, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  → minimise -Σ g_i
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                          method='highs', options={'presolve': True})
        except Exception as e:
            print("[LP] optimisation error:", e)
            break

        if not res.success:
            # infeasible or numerical trouble – give up
            print("[LP] linprog failed:", res.message)
            break

        g_opt = res.x
        sum_g = g_opt.sum()
        if sum_g <= 1e-12:
            break
        g_scaled = g_opt * (S / sum_g)

        # convex mixing (beta tuned for a smooth descent)
        beta = 0.35
        candidate = (1 - beta) * best_a + beta * g_scaled
        candidate = _project_simplex(candidate, S)

        cand_score = np.max(_full_convolution(candidate))
        if cand_score < best_score - 1e-9:
            best_score = cand_score
            best_a = candidate
            print(f"[LP] iteration {it+1}: improved score {best_score:.6f}")
        else:
            # no improvement – stop early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative vector that minimises ``evaluate_sequence``.
    The routine uses several random restarts on a modest problem size,
    a linear‑programming refinement step, and a final upscale + fine‑tuning.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the external call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Small‑scale exploratory optimisation (n≈800)
    # ------------------------------------------------------------------
    n_small = 800
    S_small = np.sqrt(2.0 * n_small)

    # initialise from any historic construction if present
    best_seq = None
    if 'height_sequence_1' in globals():
        try:
            tmp = np.asarray(height_sequence_1, dtype=float)
            if tmp.size != n_small:
                tmp = _upsample(tmp, n_small)
            best_seq = _project_simplex(tmp, S_small)
        except Exception:
            best_seq = None

    best_score = np.inf
    n_restarts = 6
    # allocate roughly half the total budget to these restarts
    time_for_restarts = (deadline - start_time) * 0.45
    per_restart = time_for_restarts / n_restarts

    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        restart_deadline = min(time.time() + per_restart, deadline)

        # Dirichlet‑type random start with varying concentration
        alpha = rng.uniform(0.2, 4.5)
        a0 = rng.gamma(alpha, 1.0, size=n_small)
        a0 = _project_simplex(a0, S_small)

        # occasionally blend with the current global best
        if best_seq is not None and rng.random() < 0.3:
            a0 = 0.6 * a0 + 0.4 * best_seq
            a0 = _project_simplex(a0, S_small)

        # gradient + sub‑gradient phase
        a_opt, sc_opt = _optimize_combined_adam(a0, restart_deadline, rng,
                                                fraction_adam=0.78)

        # very short GA refinement if time permits
        remaining = restart_deadline - time.time()
        if remaining > 2.0:
            a_ga, sc_ga = _run_genetic_optimizer(a_opt,
                                                 time.time() + min(5.0, remaining),
                                                 rng)
            if sc_ga < sc_opt:
                a_opt, sc_opt = a_ga, sc_ga

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[restart] r={r} new global best (small) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # LP‑based refinement on the best small‑scale vector
    # ------------------------------------------------------------------
    if best_seq is not None:
        lp_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.10)
        best_seq, best_score = _lp_refine(best_seq, lp_deadline, rng, max_iter=30)

    # ------------------------------------------------------------------
    # Upsample to the target size (n=2000) and final polishing
    # ------------------------------------------------------------------
    final_n = 2000
    if best_seq is None:
        a_big = np.full(final_n, 1.0)
        a_big = _project_simplex(a_big, np.sqrt(2.0 * final_n))
    else:
        a_big = _upsample(best_seq, final_n)
        a_big = _project_simplex(a_big, np.sqrt(2.0 * final_n))

    # final combined optimisation with whatever time is left
    if time.time() < deadline:
        a_final, sc_final = _optimize_combined_adam(a_big, deadline, rng,
                                                    fraction_adam=0.80)
        # a fleeting GA step for extra polish
        remaining = deadline - time.time()
        if remaining > 2.0:
            a_ga, sc_ga = _run_genetic_optimizer(a_final,
                                                 time.time() + min(5.0, remaining),
                                                 rng)
            if sc_ga < sc_final:
                a_final, sc_final = a_ga, sc_ga

        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final

    final_sequence = best_seq.tolist()
    external_score = evaluate_sequence(final_sequence)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_sequence)}")
    return final_sequence
