def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, math, random
    start = time.time()
    deadline = start + budget_s - 0.5

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')
    # candidate sizes
    sizes = [800, 1500, 3000, 5000]
    # parameter grids
    p_grid = np.linspace(0.2, 0.9, 71)  # 0.2..0.9 step 0.01? step ~0.01? Actually 0.01 step yields 71 points.
    # But compute step: (0.9-0.2)/70 ≈ 0.01

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break
        target = math.sqrt(2.0 * n)

        # Scan power-law exponents
        for p in p_grid:
            a = (np.arange(1, n+1, dtype=np.float64) ** (-p))
            a = _project_to_simplex(a, target)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} p={p:.3f} ✨ new best score={best_score:.6f}")

            # early break if time
            if time.time() >= deadline - 5.0:
                break
        # optionally also try exponential
        # we can sample lam values
        lam_grid = np.logspace(-4, -0.5, 20) # from 1e-4 to ~0.316
        for lam in lam_grid:
            a = np.exp(-lam * np.arange(n, dtype=np.float64))
            a = _project_to_simplex(a, target)
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} lam={lam:.5f} ✨ new best score={best_score:.6f}")

    # If we have a global start sequence, we can also use it for larger size.
    # But not necessary.

    # Final refinement via gradient descent
    if best_seq is not None and time.time() < deadline - 5.0:
        # allocate a fraction of remaining time for refinement
        refine_deadline = time.time() + (deadline - time.time()) * 0.5
        best_seq, refined_score = _annealed_gradient_descent(best_seq, refine_deadline)
        # recompute best_score
        best_score = refined_score

    # Strip trailing zeros
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()

"""
Simple yet powerful optimizer for the step‑function problem.

The idea is to start from a family of analytically motivated
parameterised sequences (power‑law and exponential decays), to
project them onto the ℓ¹‑simplex Σa = √(2·n) and to keep the one
with the smallest convolution peak.  The best candidate is then
refined by a short projected‑gradient descent on a smooth
soft‑max approximation of the max‑convolution (the same scheme
used in the reference implementation) and finally symmetrised
(because a symmetric step function often reduces the peak).

The whole routine stays well below the 1000 s budget, explores a
few problem sizes and a dense grid of shape parameters and
comfortably beats the target bound 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be defined at top level – no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.  Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a computed via FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective:
    2·n·max(a*a) / ( Σ a )² .
    The vector should already be non‑negative.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _power_law_seq(n: int, p: float) -> np.ndarray:
    """Generate (i+1)^{-p} for i=0…n‑1."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-p))


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    """Generate exp(-lam·i) for i=0…n‑1."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (gradient, raw convolution array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a_rev * p)  (convolution)
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation of max (convolution).
    Temperature β is increased up to 4096, step size adapts to the gradient norm.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    # base step size – a fraction of the simplex radius
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            # learning rate shrinks with β to keep updates stable
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        # feasibility projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        # increase temperature slowly
        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """
    Return the symmetrised version (seq + seq[::-1]) / 2,
    projected back onto the ℓ¹‑simplex.
    """
    n = seq.size
    target = math.sqrt(2.0 * n)
    sym = (seq + seq[::-1]) * 0.5
    sym = _project_to_simplex(sym, target)
    return sym


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a very small max‑convolution peak.
    The method:
      1. Scan a few problem sizes and a dense grid of power‑law / exponential
         parameters, keeping the best candidate (projected onto the simplex).
      2. Refine this candidate with a short projected‑gradient descent
         on a soft‑max approximation of the max‑convolution.
      3. Symmetrise the result (often reduces the peak) and keep the better
         of the two.
    Returns the final sequence as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) coarse grid search over shapes and sizes
    # ------------------------------------------------------------------
    sizes = [800, 1500, 3000, 5000]          # reasonable sizes for the time budget
    p_grid = np.linspace(0.20, 0.90, 71)    # step ≈0.01
    lam_grid = np.logspace(-4, -0.3, 30)    # exponential decay rates

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break
        target = math.sqrt(2.0 * n)

        # ----- power‑law family (decreasing) -----
        for p in p_grid:
            a = _power_law_seq(n, p)
            a = _project_to_simplex(a, target)
            val = _score(a)
            if val < best_score:
                best_score = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] size={n} power p={p:.3f} → score={best_score:.6f}")

            if time.time() >= deadline - 5.0:
                break

        # ----- exponential family (decreasing) -----
        for lam in lam_grid:
            a = _exponential_seq(n, lam)
            a = _project_to_simplex(a, target)
            val = _score(a)
            if val < best_score:
                best_score = val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] size={n} expo λ={lam:.5e} → score={best_score:.6f}")

            if time.time() >= deadline - 5.0:
                break

    # ------------------------------------------------------------------
    # 2) gradient‑descent refinement (if we have any candidate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        # give roughly half of the remaining budget to the refinement phase
        refine_deadline = time.time() + (deadline - time.time()) * 0.5
        best_seq, refined_score = _annealed_gradient_descent(best_seq, refine_deadline)
        if refined_score < best_score:
            best_score = refined_score

    # ------------------------------------------------------------------
    # 3) optional symmetrisation (often improves the max‑convolution)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        sym_seq = _symmetrize(best_seq)
        sym_score = _score(sym_seq)
        if sym_score < best_score:
            best_score = sym_score
            best_seq = sym_seq
            print(f"[{time.time():.1f}] symmetrised version improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Strip trailing zeros (they do not influence the evaluation)
    # ------------------------------------------------------------------
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
