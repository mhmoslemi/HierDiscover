"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Approach
--------
* Initialise from the best known construction (``height_sequence_1``) or,
  if unavailable, from a power‑law seed.
* Normalise the vector onto the ℓ¹‑simplex Σa = √(2 n) – the objective
  then equals the maximal entry of the auto‑convolution.
* Perform a *gradient‑guided simulated annealing* on the raw values:
    – In each iteration the current convolution is computed with a
      fast FFT, its maximal index `d` is identified and the sub‑gradient
      of the peak (g_i = 2·a_{d‑i}) is obtained.
    – Mass is moved from an index with large gradient to a random
      index with small gradient; the move is accepted if it improves
      the peak or with a Metropolis probability that cools over time.
    – Occasional pure swaps of two entries explore orderings.
* After the annealing phase a short projected‑gradient descent on a
  smooth log‑sum‑exp surrogate of the maximum further refines the
  vector.
* The best sequence found within the allotted wall‑time is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: projection onto the ℓ¹‑simplex Σx = s, x≥0 (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full linear convolution using FFT (real input)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a full convolution a*a (length 2·n‑1) using given FFT size."""
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * a.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a (full convolution)."""
    conv = _full_convolution(a, size)
    return float(conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used for guided moves)
# ----------------------------------------------------------------------
def _grad_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """Sub‑gradient of b[peak_idx] where b = a*a."""
    n = a.size
    g = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    i_vals = np.arange(lo, hi + 1)
    j_vals = peak_idx - i_vals
    g[i_vals] = 2.0 * a[j_vals]
    return g


# ----------------------------------------------------------------------
# Smoothed surrogate of max(b) and its gradient (log‑sum‑exp)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float, size: int):
    """
    Returns (grad, conv) where
        conv = a*a (full)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·conv_k) )
              = 2·(a ⋆ p) , p_k ∝ exp(β·conv_k)
    """
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * a.size - 1]

    # log‑sum‑exp
    max_c = np.max(conv)
    exp_shift = np.exp(beta * (conv - max_c))
    p = exp_shift / np.sum(exp_shift)

    # convolution a ⋆ p (a reversed)
    a_rev = a[::-1]
    # the length of a_rev * p is n + (2n‑1) – 1 = 3n‑2
    conv_len = a.size + conv.size - 1
    size2 = 1 << (conv_len.bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    p_fft = np.fft.rfft(p, n=size2)
    tmp = np.fft.irfft(ar_fft * p_fft, n=size2)[:conv_len]
    grad = 2.0 * tmp[a.size - 1: a.size - 1 + a.size]
    return grad, conv


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected GD on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    size = 1 << ((2 * n - 1).bit_length())

    beta = 5.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _softmax_gradient(a, beta, size)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = 0.4 * target / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection (keeps Σa = target)
        a = _project_to_simplex(a, target)

        # anneal β
        if it % 120 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        it += 1

    return a


# ----------------------------------------------------------------------
# Gradient‑guided simulated annealing on the raw values
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Move mass between entries (guided by the sub‑gradient of the current
    convolution peak) and perform occasional swaps.  The sum is kept
    constant, non‑negativity is enforced, and acceptance follows a
    cooling schedule.
    """
    a = a0.copy()
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    target = np.sqrt(2.0 * n)

    # initial temperature – fraction of the current peak
    cur_best = _max_convolution_fft(a, size)
    T = cur_best * 0.05 + 1e-9
    cooling = 0.9997
    rng = np.random.default_rng()

    # keep a copy of the best sequence seen so far
    best_seq = a.copy()
    best_val = cur_best

    # main loop
    while time.time() < deadline:
        # compute current convolution and the index of the peak
        conv = _full_convolution(a, size)
        peak_idx = int(conv.argmax())
        peak_val = float(conv.max())

        # compute sub‑gradient for the peak
        g = _grad_of_peak(a, peak_idx)

        # pick a donor (large gradient) and a receiver (small gradient)
        donor = int(np.argmax(g))
        # pick receiver among indices with relatively small gradient
        low_candidates = np.where(g < (g[donor] * 0.1 + 1e-12))[0]
        if low_candidates.size == 0:
            receiver = rng.integers(0, n)
        else:
            receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        # amount to transfer – a fraction of the donor's mass
        max_transfer = a[donor]
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.4  # up to 40 % of donor

        # perform tentative move
        a[donor] -= delta
        a[receiver] += delta

        # evaluate new peak
        new_val = _max_convolution_fft(a, size)

        # Metropolis acceptance
        accept = False
        if new_val <= best_val:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_val - best_val) / T):
                accept = True

        if accept:
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA improvement: {best_val:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta

        # occasional pure swap (explores ordering)
        if rng.random() < 0.008:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                a[i], a[j] = a[j], a[i]
                swap_val = _max_convolution_fft(a, size)
                if swap_val <= best_val or (T > 0 and rng.random() < np.exp(-(swap_val - best_val) / T)):
                    if swap_val < best_val:
                        best_val = swap_val
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] SA swap improvement: {best_val:.6f}")
                else:
                    a[i], a[j] = a[j], a[i]  # revert swap

        # temperature decay
        T *= cooling

        # periodic re‑projection to counteract floating‑point drift
        if it % 500 == 0:
            cur_sum = a.sum()
            if abs(cur_sum - target) > 1e-10:
                a = _project_to_simplex(a, target)

        it = getattr(_simulated_annealing, "it", 0) + 1
        _simulated_annealing.it = it

    return best_seq


# ----------------------------------------------------------------------
# Helper: generate a power‑law seed (used when height_sequence_1 is absent)
# ----------------------------------------------------------------------
def _powerlaw_seed(n: int, p: float = 0.5, shift: float = 0.6) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    a = raw * (target / raw.sum())
    return a

# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # keep a safety margin

    # ------------------------------------------------------------------
    # 1. Initialise – use the previously discovered construction if
    #    available, otherwise fall back to a power‑law seed.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)
        n = init.size
    except NameError:
        n = 8000
        init = _powerlaw_seed(n)

    # Ensure exact ℓ¹‑normalisation (objective = max convolution)
    target_sum = np.sqrt(2.0 * n)
    init = _project_to_simplex(init, target_sum)

    # Evaluate initial objective
    size_fft = 1 << ((2 * n - 1).bit_length())
    best_val = _max_convolution_fft(init, size_fft)
    best_seq = init.copy()
    print(f"[{time.time():.1f}] initial max‑conv = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient‑guided simulated annealing on the raw values
    # ------------------------------------------------------------------
    sa_deadline = min(deadline, time.time() + 0.55 * (deadline - time.time()))
    refined = _simulated_annealing(best_seq, sa_deadline)
    refined_val = _max_convolution_fft(refined, size_fft)
    if refined_val < best_val:
        best_val, best_seq = refined_val, refined.copy()
        print(f"[{time.time():.1f}] after SA best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final smooth gradient descent (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        gd_deadline = deadline
        gd_refined = _gradient_descent_opt(best_seq, gd_deadline)
        gd_val = _max_convolution_fft(gd_refined, size_fft)
        if gd_val < best_val:
            best_val, best_seq = gd_val, gd_refined.copy()
            print(f"[{time.time():.1f}] after GD best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Minor post‑processing (trim trailing near‑zeros)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)            # safety
    best_seq = _project_to_simplex(best_seq, target_sum)  # ensure exact norm
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # ------------------------------------------------------------------
    # 5. Report the final value using the provided evaluation function
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] finished – best max‑conv = {best_val:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
