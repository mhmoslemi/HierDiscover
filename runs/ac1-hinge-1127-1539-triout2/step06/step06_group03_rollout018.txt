"""
Propose a non‑negative step‑function (a list of heights) that minimises the
``evaluate_sequence`` objective.

Main ideas
----------
1. Generate a few candidate multsets using simple parametric families
   (power‑law, exponential, truncated linear) and normalise them so that
   Σa = √(2 n).

2. For each multiset keep a handful of permutations (ascending,
   descending, alternating small–large).  The best ones are stored as an
   elite list.

3. Refine the elite candidates with a smooth “log‑sum‑exp” surrogate
   gradient descent (projected onto the ℓ¹‑simplex).

4. Perform an intensive *fast* permutation annealing on the best elite
   candidates.  Swaps are evaluated in O(n) time by updating the full
   convolution incrementally:

        b ← a * a   (full linear convolution)
        Δ = a[j] – a[i]
        b[i : i+n]   += Δ * a
        b[j : j+n]   -= Δ * a

   The move is accepted if it reduces the current maximum convolution
   entry, otherwise it is accepted with a temperature‑dependent
   probability.  The temperature slowly cools.

5. The best sequence found before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x≥0  (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, truncated linear)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    scale = target / np.sum(v)
    a = v * scale
    return np.maximum(a, 0.0)


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


# ----------------------------------------------------------------------
# Simple ordering heuristics
# ----------------------------------------------------------------------


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a few useful permutations of ``base``."""
    asc = np.sort(base)               # small → large
    desc = asc[::-1]                  # large → small
    out = [asc, desc]

    # alternating small–large (small first)
    n = base.size
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # a few random shuffles
    for _ in range(3):
        tmp = asc.copy()
        rng.shuffle(tmp)
        out.append(tmp)

    return out


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and projected gradient descent
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where
        b = a*a (full convolution)
        grad = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e5
    momentum = np.zeros_like(a)
    rho = 0.9
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = 0.5 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 8e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.06, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Fast O(n) update of the convolution after swapping two entries
# ----------------------------------------------------------------------


def _apply_swap(a: np.ndarray, b: np.ndarray, i: int, j: int, delta: float):
    """Apply the effect of swapping a[i] and a[j] (delta = a[j]-a[i])."""
    n = a.size
    # update the convolution
    b[i:i + n] += delta * a
    b[j:j + n] -= delta * a
    # perform the swap in-place
    a[i], a[j] = a[j], a[i]


def _revert_swap(a: np.ndarray, b: np.ndarray, i: int, j: int, delta: float):
    """Undo a previously applied swap (using the same delta)."""
    n = a.size
    # restore the original a order
    a[i], a[j] = a[j], a[i]
    # undo the convolution update (use -delta)
    b[i:i + n] -= delta * a
    b[j:j + n] += delta * a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    explores a few parametric families, refines elite candidates with
    gradient descent, and then applies a specialised permutation annealing
    that updates the convolution in O(n) per swap.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_obj = np.inf
    elite = []          # list of (objective, sequence)

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000]
    p_grid = np.linspace(0.45, 0.55, 5)
    shift_grid = np.linspace(0.3, 1.0, 5)

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        for p in p_grid:
            for shift in shift_grid:
                base = _powerlaw_sequence(n, p, shift)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL BEST (pow) n={n} p={p:.3f} shift={shift:.2f} obj={best_obj:.6f}")
                    # keep a short elite list (size ≤7)
                    if len(elite) < 7 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:7]

                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # 2. Refine elite candidates with gradient descent
    # ------------------------------------------------------------------
    for score0, seq0 in elite:
        if time.time() >= deadline:
            break
        n = seq0.size
        sub_dead = min(deadline, time.time() + 12.0)   # short refinement
        refined = _gradient_descent_opt(seq0, sub_dead)
        refined = _project_to_simplex(refined, np.sqrt(2.0 * n))

        score = _max_convolution_fft(refined)
        if score < best_obj:
            best_obj = score
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (grad) n={n} obj={best_obj:.6f}")

        # keep for later annealing
        elite.append((score, refined.copy()))
        elite.sort(key=lambda x: x[0])
        elite = elite[:7]

    # ------------------------------------------------------------------
    # 3. Intensive permutation annealing (fast O(n) updates)
    # ------------------------------------------------------------------
    # pick up to three best elite sequences for the annealing stage
    to_anneal = sorted(elite, key=lambda x: x[0])[:3]

    for score0, seq0 in to_anneal:
        if time.time() >= deadline:
            break
        a = seq0.copy()
        n = a.size
        # ensure exact ℓ¹‑norm (project again)
        a = _project_to_simplex(a, np.sqrt(2.0 * n))

        # full convolution (FFT once)
        b = _full_convolution_fft(a)
        cur_max = float(b.max())
        best_local_seq = a.copy()
        best_local_val = cur_max

        # indices of large / small values (based on value, not position)
        median_val = np.median(a)
        top_idx = np.where(a > median_val)[0]
        bottom_idx = np.where(a < median_val)[0]
        # guarantee non‑empty sets
        if top_idx.size == 0:
            top_idx = np.arange(n)
        if bottom_idx.size == 0:
            bottom_idx = np.arange(n)

        # temperature schedule
        T = cur_max * 0.1 + 1e-9
        cooling = 0.99995
        iteration = 0

        while time.time() < deadline:
            # ----------------------------------------------------------
            # choose a swap
            # ----------------------------------------------------------
            if rng.random() < 0.35:                     # focus on the current peak
                k_max = int(np.argmax(b))
                lo = max(0, k_max - (n - 1))
                hi = min(n - 1, k_max)
                i_cand = np.arange(lo, hi + 1)
                j_cand = k_max - i_cand
                products = a[i_cand] * a[j_cand]
                i = int(i_cand[np.argmax(products)])
                j = int(rng.choice(bottom_idx))
            else:
                i = int(rng.choice(top_idx))
                j = int(rng.choice(bottom_idx))
                if i == j:
                    continue

            ai, aj = a[i], a[j]
            delta = aj - ai
            if delta == 0.0:
                continue

            # ----------------------------------------------------------
            # apply the swap (fast O(n) update of b)
            # ----------------------------------------------------------
            _apply_swap(a, b, i, j, delta)
            new_max = float(b.max())

            accept = False
            if new_max <= best_local_val:
                accept = True
            else:
                # simulated‑annealing acceptance
                if rng.random() < np.exp((best_local_val - new_max) / T):
                    accept = True

            if accept:
                if new_max < best_local_val:
                    best_local_val = new_max
                    best_local_seq = a.copy()
                    if new_max < best_obj:
                        best_obj = new_max
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL BEST (anneal) n={n} obj={best_obj:.6f}")
            else:
                # revert the move
                _revert_swap(a, b, i, j, delta)

            # temperature decay
            T *= cooling
            iteration += 1

    # ------------------------------------------------------------------
    # 4. Final sanity check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (very unlikely to happen)
        n = 4000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    # trim trailing near‑zero entries (purely cosmetic)
    nz = best_seq > 1e-12
    if np.any(nz):
        best_seq = best_seq[: np.where(nz)[0][-1] + 1]

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {final_val:.6f} "
              f"(max‑conv = {best_obj:.6f}, length = {len(best_seq)})")
    except Exception as exc:
        print("Evaluation error:", exc)

    return best_seq.tolist()
