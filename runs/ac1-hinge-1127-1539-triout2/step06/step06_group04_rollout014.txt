def _subgradient_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_max = _max_convolution_fft(a) # we can compute with full convolution as max
    iteration = 0
    base_lr = target_sum * 0.2
    while time.time() < deadline:
        # compute convolution
        b = _full_convolution(a)
        # find max
        k_max = int(np.argmax(b))
        cur_max = b[k_max]
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad improvement: {best_max:.6f}")
        # compute gradient
        g = np.zeros_like(a)
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(n - 1, k_max)
        idx = np.arange(i_lo, i_hi+1)
        g[idx] = 2.0 * a[k_max - idx]
        # step size schedule
        lr = base_lr / (1.0 + iteration * 0.001)  # decaying slowly
        a = a - lr * g
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if iteration % 200 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
        iteration += 1
    # after loop compute final max
    final_max = _max_convolution_fft(a)
    if final_max < best_max:
        best_max = final_max
        best_a = a.copy()
    return best_a, best_max

"""
Hybrid sub‑gradient descent + targeted hill‑climbing for the self‑convolution
maximisation problem.

* The objective after projecting to the ℓ¹‑simplex Σa = √(2 n) reduces to
  maximise max(b) where b = a∗a (the linear convolution).  Hence the score
  we minimise is simply max(b).
* A direct sub‑gradient of max(b) is 2·a shifted by the index of the current
  peak.  Projected sub‑gradient descent with a decaying step size drives the
  peak down while preserving non‑negativity and the ℓ¹‑norm.
* After the descent a specialised hill‑climber moves a small amount of mass
  from a “source’’ index that contributes heavily to the current peak towards
  a “destination’’ that contributes little.  The convolution is updated by an
  analytical O(n) formula, so many moves can be tried quickly.
* A length schedule progressively enlarges the search space (up to 50 000)
  and each new length is seeded by linear up‑sampling of the best solution
  found so far.  The algorithm respects a hard wall‑clock budget (default
  1000 s) and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a (length 2n‑1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a∗a using an FFT (fast, slightly less accurate than full)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Sub‑gradient descent on max(b) where b = a∗a
# ----------------------------------------------------------------------


def _subgradient_opt(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for minimising max(a∗a).
    Returns the best sequence found (as vector) and its maximal convolution value.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_max = _max_convolution_fft(a)

    it = 0
    base_lr = target_sum * 0.20          # initial step size (relative to ℓ¹‑norm)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # full convolution
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        cur_max = b[k_max]

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad improvement (n={n}) "
                  f"score={best_max:.6f}")

        # gradient of max(b) = 2 * a shifted by k_max
        g = np.zeros_like(a)
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(n - 1, k_max)
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] = 2.0 * a[k_max - idx]

        # decaying step size
        lr = base_lr / (1.0 + it * 0.001)

        # gradient step + projection onto ℓ¹‑simplex
        a -= lr * g
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    # final evaluation
    final_max = _max_convolution_fft(a)
    if final_max < best_max:
        best_max = final_max
        best_a = a.copy()
        print(f"[{time.time():.1f}] subgrad final improvement (n={n}) "
              f"score={best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Targeted hill‑climbing (moves mass from “bad’’ to “good’’ indices)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Hill‑climbing that looks at the current maximal convolution entry.
    Moves a small amount of mass from a source that contributes heavily to that
    entry towards a destination that contributes little.
    Returns (best_sequence, best_max_convolution).
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    # initial temperature proportional to current peak height
    T = cur_max * 0.02 + 1e-9
    cooling = 0.9995

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each index to that peak:
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        # destination – weighted towards small contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # analytical update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # accept if we strictly improve, otherwise Metropolis criterion
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max

                  f"score={max_sgd:.6f}")

        # ------------------------------------------------------------------
        # Phase B – targeted hill‑climbing (polish the result)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a_sgd)
            time_left = deadline - time.time()
            phaseB_dead = time.time() + min(80.0, 0.35 * time_left)
            a_hc, max_hc = _hill_climb_targeted(a_sgd, b_full, phaseB_dead)
            if max_hc < best_score:
                best_score = max_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_final, deadline)
        if max_final < best_score:
            best_score = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
