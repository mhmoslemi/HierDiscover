"""
A time‑aware optimizer for the “minimal self‑convolution peak” problem.

The search proceeds in two phases:

1. **Fast exploration** – generate a diverse set of analytic shapes
   (uniform, geometric decay, power‑law, logistic, triangular and
   up‑sampled previous best), project them onto the ℓ¹‑simplex with
   Σa = √(2 n) and keep the best few candidates.

2. **Intensive refinement** – each promising candidate is fed to a
   three‑stage optimiser:
      * Adam on a soft‑max approximation of the convolution peak,
      * sub‑gradient descent on the true max entry,
      * cheap random‑swap hill‑climbing.
   The optimiser runs as long as time permits, continually updating the
   global best.  The final vector is returned (and exposed as the global
   variable ``height_sequence_1``).

The implementation uses FFT‑based convolutions, Euclidean simplex projection
and optional exact mirror symmetry to reduce the search space.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2*n‑1) using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    w must be a probability vector (soft‑max weights).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Simple up‑sampling (linear interpolation)
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Score (internal) – 2·n·max(b) / (sum(a)^2)
# ----------------------------------------------------------------------
def _ratio_score(a: np.ndarray) -> float:
    n = a.shape[0]
    S = a.sum()
    if S < 1e-12:
        return np.inf
    max_b = np.max(_full_convolution(a))
    return 2.0 * n * max_b / (S * S)


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    a = np.random.random(n)
    return a


def _shape_geometric(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.9, 0.9995)
    left = rate ** np.arange(n)
    a = left + left[::-1]
    return a


def _shape_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.05, 2.5)
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p) + (n + 1 - idx) ** (-p)
    return a


def _shape_logistic(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.3) * n
    x = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((x - centre) / sigma))
    a = a + a[::-1]          # make it symmetric
    return a


def _shape_triangular(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(np.arange(n) - centre) / centre
    power = rng.uniform(0.3, 2.0)
    a = np.clip(a, 0.0, None) ** power
    return a


# ----------------------------------------------------------------------
# Single‑step Adam optimiser on the smooth log‑sum‑exp objective
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 1.0
    alpha_mul = 1.04

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if np.max(_full_convolution(cand)) < best_score or rng.random() < 0.15:
                    a = cand

        # schedule updates
        alpha = min(alpha * alpha_mul, 100.0)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum entry
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.3
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active quadratic form
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # random mass‑swap (simulated‑annealing style)
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.25:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # occasional early break to avoid infinite loops
        if it % 7000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap hill‑climbing that swaps tiny masses between random positions
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local swaps)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # split 55% / 35% / 10%
    t_adam = now + 0.55 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.35 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # pick the best among the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score

        if base_seq is not None:
            a = _upsample_seq(base_seq, n)
            cand_list.append(_project_simplex(a, S_target))

        # evaluate each candidate
        for a in cand_list:
            sc = _ratio_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_n = n
                print(f"[explore] n={n} new best={best_score:.6f}")

        # early exit if already very good (unlikely but cheap)
        if best_score < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase B – intensive refinement of the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.time() < deadline - 8.0:
            # allocate a modest time slice for each refinement run
            slice_deadline = min(time.time() + rng.uniform(20.0, 30.0), deadline)

            enforce_sym = rng.random() < 0.7
            a0 = best_seq.copy()

            # tiny random jitter before optimisation (helps escape plateaus)
            if rng.random() < 0.6:
                i, j = rng.integers(0, best_n, size=2)
                if i != j and a0[i] > 0:
                    delta = 0.01 * a0[i] * rng.random()
                    a0[i] -= delta
                    a0[j] += delta
                    a0 = _project_simplex(a0, np.sqrt(2.0 * best_n))

            if enforce_sym:
                a0 = _symmetrize(a0)

            a_opt, sc_opt = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                best_n = len(a_opt)
                print(f"[refine] n={best_n} sym={enforce_sym} new best={best_score:.6f}")

            # occasionally try a modest change of length via up‑sampling/down‑sampling
            if rng.random() < 0.12:
                new_n = int(best_n * rng.uniform(0.85, 1.15))
                new_n = max(1000, min(20000, new_n))
                a_up = _upsample_seq(best_seq, new_n)
                a_up = _project_simplex(a_up, np.sqrt(2.0 * new_n))
                best_seq = a_up
                best_n = new_n

    # ------------------------------------------------------------------
    # Final sanity check via the provided black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal={best_score:.6f} "
          f"length={len(final_seq)} time={time.time() - start:.1f}s")

    # expose for possible later runs
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
