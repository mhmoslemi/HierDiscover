def _softmax_grad(a, beta):
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()
    # gradient: g_i = 2 * sum_{j} w_{i+j-n+1} * a_j ??? earlier version.
    # Use convolution with reversed a.
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(w_fft * a_rev_fft, n=size)[:2*n-1]
    g = 2.0 * conv_rev[n-1: n-1+n]
    return g, conv

def _adam_optimize(a0, deadline):

    n = a.size
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)
    lr = target * 0.08  # baseline
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = _score(a)
    beta = 0.5
    max_beta = 1e5
    # schedule factor per iteration: multiply by factor >1
    beta_factor = 1.01
    while time.time() < deadline:
        t += 1
        g, conv = _softmax_grad(a, beta)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection
        a = _project_to_simplex(a, target)
        # evaluate
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam iteration {t} beta={beta:.1f} score={best_score:.6f}")
        # update beta
        if beta < max_beta:
            beta = min(beta * beta_factor, max_beta)
        # maybe adjust lr adaptively, but skip.
    return best_a, best_score

"""
A fast multi‑restart optimizer for the step‑function problem.

* Uses several random initialisations (power‑law, uniform, warm‑start).
* Each run performs:
    - Adam optimisation on a soft‑max approximation of the max‑convolution.
      The temperature β is raised gradually to focus on the true maximum.
    - A simulated‑annealing refinement that moves mass away from the worst
      convolution entry, smooths short windows and swaps entries.
* The best sequence among all restarts is returned (trailing zeros stripped).
* All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = s}.  Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective used by `evaluate_sequence`.
    Equivalent to `evaluate_sequence(a)` when a is non‑negative and
    all entries are << 1000, because the scale cancels out.
    """
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _softmax_grad(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (gradient, full convolution array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    p = exps / exps.sum()

    # gradient = 2 * (a_rev * p) evaluated on the original index range
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(p_fft * a_rev_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float):
    """
    Projected Adam optimiser on the soft‑max surrogate.
    β is increased exponentially up to a large value.
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # step size (chosen relative to the simplex radius)
    lr = target * 0.08

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5          # start temperature for soft‑max
    max_beta = 1e5
    beta_factor = 1.015

    while time.time() < deadline:
        t += 1
        grad, _ = _softmax_grad(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_to_simplex(a, target)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam it={t:4d} β={beta:.1f} score={best_val:.6f}")

        # increase β slowly
        if beta < max_beta:
            beta = min(beta * beta_factor, max_beta)

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-6,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.005,
                         max_window: int = 30):
    """
    Simple simulated‑annealing refinement.
    Moves mass from indices that contribute most to the worst
    convolution entry to indices that contribute little, smooths short windows,
    and swaps pairs.  Acceptance follows the Metropolis rule.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current worst convolution entry
        k_max = int(np.argmax(conv))

        # contributions of each coordinate to that entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]

        # ------------------------------------------------------------------
        # Propose a move
        # ------------------------------------------------------------------
        move_type = None
        if rng.random() < 0.65:          # mass transfer
            tot = contrib.sum()
            if tot > 0:
                donor = rng.choice(idx, p=contrib / tot)
            else:
                donor = rng.integers(n)
            # pick a receiver that has low contribution to the worst entry
            low_thr = np.percentile(contrib, 15)
            low_idx = np.where(contrib <= low_thr)[0]
            if low_idx.size:
                receiver = idx[rng.choice(low_idx)]
            else:
                receiver = rng.integers(n)
            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:        # local smoothing
            start = rng.integers(0, n - 2)
            length = min(rng.integers(3, max_window + 1), n - start)
            seg = a[start:start + length].copy()
            avg = seg.mean()
            a[start:start + length] = avg
            saved_start, saved_seg = start, seg
            move_type = "smooth"

        else:                            # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # ------------------------------------------------------------------
        # Evaluate the new state (single FFT)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_start + saved_seg.size] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape deep minima
        if it % 4000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _optimize_instance(n: int, init_seq: np.ndarray, deadline: float):
    """
    Run Adam for roughly half the allocated time, then simulated annealing.
    Returns the best sequence obtained and its score.
    """
    now = time.time()
    remaining = deadline - now
    if remaining < 5.0:
        return init_seq, _score(init_seq)

    # split time roughly 45 % / 55 %
    split = remaining * 0.45
    adam_deadline = now + split
    seq_adam, score_adam = _adam_optimize(init_seq, adam_deadline)

    seq_sa, score_sa = _simulated_annealing(seq_adam, deadline)

    if score_adam < score_sa:
        return seq_adam, score_adam
    else:
        return seq_sa, score_sa

        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
