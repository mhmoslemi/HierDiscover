"""
Propose a step‑function that (approximately) minimises
``evaluate_sequence``.  The objective equals the maximum entry of the
self‑convolution when the ℓ¹‑norm is fixed to √(2 n).  The search proceeds
in three stages for each problem size `n`:

1. **Smooth gradient descent** – minimise a log‑sum‑exp surrogate of the
   max‑convolution using projected gradient descent with momentum.
2. **Exact sub‑gradient descent** – after the surrogate is tight the true
   max‑convolution is used; the sub‑gradient of the max is summed over all
   active indices and a projected (Adam‑style) step is taken.
3. **Mass‑transfer hill‑climbing** – a cheap O(n) update moves a random
   amount of mass from a source index (chosen proportional to its
   contribution to the current worst convolution entry) to a destination
   index (biased towards low values).  Accepted moves follow a simulated‑
   annealing schedule.

The schedule gradually increases the vector length (up to 30 000) while
respecting the total wall‑clock budget (default 1000 s).  The best vector
found is returned as a plain Python list.

All vectors stay non‑negative and are projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)} after each stochastic move, so the
``evaluate_sequence`` call sees a valid step function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b)  where  b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a convolved with p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _smooth_gradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max (log‑sum‑exp) surrogate.
    Momentum, β‑annealing and occasional jitter are employed.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 1e5
    beta_factor = 1.5

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every now and then to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Uses the (convex) sub‑gradient of the piecewise‑linear max function.
    An Adam‑style adaptive step size is employed.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = np.max(b)
        max_idxs = np.flatnonzero(b == max_b)

        # build sub‑gradient: sum of gradients of all active max entries
        g = np.zeros_like(a)
        for M in max_idxs:
            i_start = max(0, M - (n - 1))
            i_end = min(n - 1, M)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            jdx = M - idx
            # derivative contributions
            g[idx] += a[jdx]
            g[jdx] += a[idx]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.2 / (np.sqrt(v_hat) + eps)   # base step size tuned empirically
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

    return a

# Random initialisers

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Random vector on the simplex Σ a = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length and simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that moves a random amount of
    mass from a source index to a destination index while keeping the full
    self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.08 + 1e-9          # initial temperature
    cooling = 0.9992                    # slower cooling for finer search

    while time.time() < deadline:
        # pick a worst convolution index
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index weighted by its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination index – bias towards smallest a
        if rng.random() < 0.4:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update convolution efficiently
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

    the best sequence found as a plain Python list.
    """

    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = np.inf

    # Optional seeding from a previously stored construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):

            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            cur = _max_convolution_fft(a0)

    # ------------------------------------------------------------------
    # Length schedule – gradually increase the dimension
    # ------------------------------------------------------------------
    schedule = [2000, 5000, 10000, 20000, 30000]

    for n in schedule:

        target_sum = np.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # Initialise candidate for this length
        # ----------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # ----------------------------------------------------------------
        # Phase 1 – smooth gradient descent (several short restarts)
        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 8.0:

        gd_budget = min(remaining * 0.30, 250.0)   # at most 250 s for this length
        n_restarts = 4
        local_best_a = None
        local_best_score = np.inf

        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            now = time.time()
            per_restart = min((deadline - now) * 0.33, 70.0)
            gd_deadline = now + per_restart

            a_start = a_init.copy() if r == 0 else _random_initial_sequence(n, rng)
            a_gd = _smooth_gradient_descent(a_start, gd_deadline)
            score = _max_convolution_fft(a_gd)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_gd.copy()

        if local_best_a is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time():.1f}] GD improvement for n={n} score={local_best_score:.6f}")

        # ----------------------------------------------------------------

        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining < 6.0:

        subgrad_deadline = time.time() + subgrad_budget

        if best_seq.size != n:
            a_current = _upsample_sequence(best_seq, n)
        else:
            a_current = best_seq.copy()
        a_current = _project_to_simplex(a_current, target_sum)

        if score_sub < best_score:

        b = _full_convolution(a)

        # a quick round of sub‑gradient descent followed by hill climbing
        polishing_deadline = time.time() + (deadline - time.time()) * 0.6
        a = _subgradient_descent(a, polishing_deadline)

        remaining = deadline - time.time()
        if remaining > 2.0:
            a, b = _hill_climbing(a, _full_convolution(a), deadline)

        final_score = _max_convolution_fft(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
