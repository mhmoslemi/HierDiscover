def gradient_descent_opt(a0, deadline, rng):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0*n)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Softmax temperature schedule
    beta_temp = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    t_iter = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta_temp)

        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**(t_iter+1))
        v_hat = v / (1 - beta2**(t_iter+1))
        step = (target_sum * 0.1) / (np.sqrt(v_hat)+epsilon)  # scaling relative to target
        a = a - step * m_hat

        # projection
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if t_iter % 200 == 0:
            jitter_scale = target_sum * 0.001 / (beta_temp + 1.0)
            a += rng.normal(scale=jitter_scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # beta anneal
        if t_iter % 300 == 0 and beta_temp < beta_max:
            beta_temp = min(beta_temp * beta_factor, beta_max)

        t_iter += 1

    return a

"""
Hybrid multistart gradient descent + targeted hill‑climbing with a
progressive length schedule.

* Starts from the provided seed sequence (height_sequence_1) if present,
  otherwise from a random simplex point.
* Runs several short gradient‑descent restarts on each length, then a
  targeted hill‑climber that moves mass from indices that contribute most
  to the current convolution peak to indices that contribute little.
* The best solution is up‑sampled to larger lengths (up to 50000) and the
  process repeats, allocating the remaining time budget automatically.
* Finally a polishing hill‑climb uses any time left.
* All vectors are kept on the ℓ¹‑simplex Σa = √(2 n) (projection via a fast
  Euclidean‑simplex routine).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a (via FFT)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Momentum, β‑annealing and jitter are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    # annealing schedule for β (temperature of the soft‑max)
    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    # step‑size base (relative to target ℓ¹‑norm)
    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92                       # Nesterov‑style momentum decay

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β‑annealing
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    # small Gaussian noise and projection
    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Targeted mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray,
                         b0: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that always looks at the current maximal convolution entry.
    Moves a small amount of mass from a source that contributes a lot to that
    entry to a destination that contributes little.
    Returns the best sequence found and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # initial temperature for Metropolis steps
    cooling = 0.9995

    while time.time() < deadline:
        # index of the current peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak:
        #   contrib[i] = a[i] * a[k_max-i]   (zero if out of bounds)
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        # source index – weighted by contribution (remove from “bad” places)
        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        # destination index – weighted towards *small* contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        # amount of mass to move (up to 50 % of source)
        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()
        # linear part  (‑2 δ·a shifted by i, +2 δ·a shifted by j)
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance test -----------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            # apply the move
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the given wall‑clock budget
    (default 1000 s) and returns the best sequence found as a plain Python
    list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Seed from a previously stored construction if available.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_score = _max_convolution_fft(a0)
        best_seq = a0.copy()
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – larger lengths give more flexibility.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 30000, 50000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise candidate for this length (upsample previous best).
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a_init = _upsample_sequence(best_seq, n)
        else:
            a_init = _random_initial_sequence(n, rng)

        a_init = _project_to_simplex(a_init, target_sum)

        # --------------------------------------------------------------
        # Phase A – several short gradient‑descent restarts.
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        # give roughly half of the remaining budget to Phase A
        phaseA_deadline = time.time() + min(0.5 * remaining, 200.0)
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break
            # perturb the start a little to diversify
            pert = rng.normal(scale=target_sum * 0.01, size=n)
            a0 = a_init + pert
            a0 = np.maximum(a0, 0.0)
            a0 = _project_to_simplex(a0, target_sum)

            gd_dead = min(time.time() + (phaseA_deadline - time.time()) / (restarts - r),
                          deadline)
            a_gd = _gradient_descent_opt(a0, gd_dead, rng)

            score = _max_convolution_fft(a_gd)
            if score < best_score:
                best_score = score
                best_seq = a_gd.copy()
                print(f"[{time.time():.1f}] GD restart {r} len={n} score={score:.6f}")

        # --------------------------------------------------------------
        # Phase B – targeted hill climbing.
        # --------------------------------------------------------------
        if time.time() < deadline:
            time_left = deadline - time.time()
            # reserve up to 30 % of the remaining time for hill climbing
            phaseB_dead = time.time() + min(0.3 * time_left, 150.0)
            b_full = _full_convolution(best_seq)
            a_hc, max_hc = _hill_climb_targeted(best_seq, b_full, phaseB_dead)
            if max_hc < best_score:
                best_score = max_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb len={n} score={max_hc:.6f}")

        # --------------------------------------------------------------
        # Phase C – a quick refinement gradient descent (optional).
        # --------------------------------------------------------------
        if time.time() < deadline:
            time_left = deadline - time.time()
            gd_dead2 = time.time() + min(0.2 * time_left, 80.0)
            a_ref = _gradient_descent_opt(best_seq, gd_dead2, rng)
            score_ref = _max_convolution_fft(a_ref)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time():.1f}] Post‑HC GD len={n} score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _hill_climb_targeted(a_final, b_final, deadline)
        if max_final < best_score:
            best_score = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improved to {max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness.
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
