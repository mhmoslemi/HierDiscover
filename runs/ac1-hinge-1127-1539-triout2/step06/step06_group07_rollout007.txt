import scipy.sparse as sp

def convolution_matrix(f):
    n = f.shape[0]
    rows = 2*n - 1
    data = []
    offsets = []
    for d in range(n):
        data.append(np.full(n, f[d]))
        offsets.append(d)
    A = sp.diags(data, offsets, shape=(rows, n), format='csr')
    return A

def solve_max_sum_lp(f, M):
    n = len(f)
    A_ub = convolution_matrix(f)
    b_ub = np.full(2*n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max_time}) # but options may vary.

    if res.success:
        return res.x
    else:
        return f # fallback

def solve_lp_max_sum(f, M, max_time=None):
    # f: 1D np array
    # M: scalar
    import numpy as np
    import scipy.sparse as sp
    from scipy.optimize import linprog
    n = len(f)
    rows = 2 * n - 1
    # Build matrix using sp.diags
    diags_data = []
    offsets = []
    for d in range(n):
        diags_data.append(np.full(n, f[d]))
        offsets.append(d)
    A = sp.diags(diags_data, offsets, shape=(rows, n), format='csr')
    # Upper bounds: M for each row
    ub = np.full(rows, M, dtype=float)
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    # Solve with options
    # For high-level, we can set time_limit in options? Not in SciPy maybe
    # We'll just let it run.
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fall back: return original f to keep feasible
        return f.copy()

def subgradient_step(a, step):
   conv = _full_convolution(a)
   k_max = np.argmax(conv)
   n = len(a)
   g = np.zeros_like(a)
   i_low = max(0, k_max - (n - 1))
   i_high = min(n - 1, k_max)
   idx_i = np.arange(i_low, i_high + 1)

   g[idx_i] = a[idx_j]  # a_{j}

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # init best_seq
    best_seq = None
    try:
        if 'height_sequence_1' in globals():
            best_seq = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        best_seq = None

    best_score = np.inf
    # define stages
    sizes = [800, 2000, 4000]
    fractions = [0.2, 0.35, 0.4]  # sum <1

    for idx, n in enumerate(sizes):
        if time.time() >= deadline:
            break
        stage_time = (deadline - start_time) * fractions[idx]
        stage_deadline = min(time.time() + stage_time, deadline)

        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)

        # jitter small
        jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
        a0 = np.clip(a0 + jitter, 0.0, None)

        # project to simplex sum = sqrt(2n)
        S = np.sqrt(2.0 * n)
        if a0.sum() == 0:
            a0 = np.full(n, 1.0)
        a0 = a0 / a0.sum() * S

        # apply LP+subgrad optimization
        a_opt, sc_opt = _optimize_stage(a0, stage_deadline, rng)

        # update global best
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[stage] n={n} new global best score={best_score:.6f}")

        # update start_time and remaining time
        start_time = time.time()

    # final
    if best_seq is None:
        final_seq = [1.0]
        final_score = evaluate_sequence(final_seq)
    else:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)

    print(f"[final] external score = {final_score:.6f} (internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq

"""
Propose a non‑negative step‑function that minimizes the maximum entry of its
self‑convolution.  The approach is a two‑phase refinement:

1.  Linear‑programming phase – for a fixed upper bound M on the convolution,
    solve a packing LP that maximises the ℓ₁‑norm of a new vector b subject to
    the linear constraints   (f * b) ≤ M.  Scaling b back to the required
    simplex sum reduces the bound to M' = (S/∑b)²·M, where S = √(2·n).  Repeating
    this step yields a monotone decrease of the worst convolution entry.

2.  Sub‑gradient phase – when the LP cannot increase the ℓ₁‑norm any further,
    perform a few sub‑gradient descent steps on the true max‑norm.  This
    perturbs the shape of the vector so that a subsequent LP can make further
    progress.

The algorithm runs three increasing problem sizes (800 → 2000 → 4000)
and respects the 1000 s time budget.  All vectors are kept on the simplex
∑aᵢ = √(2·n) and are forced to be non‑negative.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (linear, not circular)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Linear interpolation for up‑sampling
# ----------------------------------------------------------------------


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a vector a to a new length new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        maximise   Σ b_i
        subject to (f * b)_k ≤ M  for all k,
                   b_i ≥ 0.
    Returns the optimal b (may be equal to f if no improvement is possible).
    """
    try:
        import scipy.sparse as sp
        from scipy.optimize import linprog
    except Exception as exc:
        raise RuntimeError("SciPy (sparse + linprog) is required") from exc

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the (2n‑1) × n convolution matrix A where
    #   (A @ b)_k = Σ_i f_i * b_{k-i}   (zero when index out of range).
    diags_data = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(n))                     # 0 … n‑1
    A = sp.diags(diags_data, offsets, shape=(rows, n), format='csr')

    # Linear program: maximise Σb  ⇔  minimise -Σb
    c = -np.ones(n, dtype=float)
    ub = np.full(rows, M, dtype=float)          # RHS of the inequality constraints
    bounds = [(0.0, None)] * n

    # HiGHS (the default) works with CSR matrices
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        # In the unlikely case of failure fall back to the current vector.
        return f.copy()
    return res.x


# ----------------------------------------------------------------------
# LP + sub‑gradient refinement for a single problem size
# ----------------------------------------------------------------------


def _refine_with_lp_and_subgrad(a0: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Iteratively improve a vector using LP (increase ℓ₁‑norm) and sub‑gradients."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # Limits to avoid endless loops
    lp_iterations = 0
    max_lp_iterations = 25

    while time.time() < deadline and lp_iterations < max_lp_iterations:
        lp_iterations += 1
        conv = _full_convolution(a)
        M = conv.max()

        # ---------- linear programming step ----------
        try:
            b = _solve_lp_max_sum(a, M)
        except Exception as e:
            print("[LP] solving failed:", e)
            break

        sum_b = b.sum()
        if sum_b > S * (1.0 + 1e-12):
            # scaling factor that restores the required ℓ₁‑norm
            r = S / sum_b
            a_new = b * r
            cur_score = np.max(_full_convolution(a_new))
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a_new.copy()
            a = a_new
            continue          # we made progress – start next LP iteration

        # ---------- sub‑gradient phase (LP stalled) ----------
        # Perform a short burst of true sub‑gradient descent to change the shape.
        sub_it = 0
        max_sub_it = 30
        while time.time() < deadline and sub_it < max_sub_it:
            sub_it += 1
            conv = _full_convolution(a)
            kmax = int(np.argmax(conv))

            # sub‑gradient of max‑norm w.r.t. a:
            g = np.zeros_like(a)
            i_low = max(0, kmax - (n - 1))
            i_high = min(n - 1, kmax)
            idx_i = np.arange(i_low, i_high + 1)
            idx_j = kmax - idx_i
            g[idx_i] = a[idx_j]

            # step size decay ∝ 1/√t
            eta = 0.5 / np.sqrt(sub_it)
            a = a - eta * g
            a = _project_simplex(a, S)

            # occasional random jitter to increase exploration
            if rng.random() < 0.07:
                i, j = rng.integers(0, n, size=2)
                if a[i] > 0.0:
                    delta = a[i] * rng.random() * 0.02
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a, S)

            cur_score = np.max(_full_convolution(a))
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # after a sub‑gradient burst we attempt another LP iteration
        # (the loop will repeat automatically)

    return best_a, best_score
