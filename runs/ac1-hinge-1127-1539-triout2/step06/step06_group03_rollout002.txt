def interleaved_order(values):
    n = len(values)
    order = np.empty(n)
    # recursively assign
    def assign(vals, indices):
        if len(vals) == 0:
            return
        # pick median index
        median = len(indices)//2
        order[indices[median]] = vals[0] # largest
        assign(vals[1:1+len(indices)//2], indices[:median])
        assign(vals[1+len(indices)//2:], indices[median+1:])
    assign(list(values), list(range(n)))
    return order

"""
Efficient search for a non‑negative step function (vector) that minimises
``evaluate_sequence``.  The objective after the internal normalisation of the
function is exactly the maximal entry of the linear convolution a⋆a.
Hence we try to minimise  max(a⋆a)  while keeping the ℓ¹‑norm fixed.

The algorithm works in three phases:

1. **Coarse parametric sweep** – explore power‑law sequences
   a_i = (i+shift)^(-p) for several lengths (up to 8000) and a dense
   grid of (p,shift).  For each shape we test a few orderings
   (ascending, descending, alternating) and keep the best few candidates.

2. **Continuous refinement** – pick the most promising candidate,
   upscale it to a larger length (≈12000) and run a projected Adam‑style
   gradient descent on a smooth log‑sum‑exp surrogate of the maximum
   convolution entry.  The iterate stays on the ℓ¹‑simplex
   (sum = sqrt(2·n)).

3. **Permutation / value annealing** – finally improve the ordering
   (swap moves) and the actual distribution of the mass (pair‑wise
   transfers) by simulated annealing until the time budget expires.

All steps respect the global deadline (default 1000 s) and report the
current best bound.  The returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max entry of the full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    total = 2 * n - 1
    size = 1 << ((total - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:total]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex  {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto the simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate of max(b) where b = a*a, together with its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Compute gradient of (1/beta)·log Σ_k exp(beta·b_k) 
    where b = a*a (full convolution).
    Returns (grad, b).
    """
    n = a.shape[0]
    total = 2 * n - 1
    size = 1 << ((total - 1).bit_length())

    # b = a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:total]

    # soft‑max weights
    max_b = b.max()
    shifted = np.exp(beta * (b - max_b))
    p = shifted / shifted.sum()                     # shape (2n‑1,)

    # gradient = 2·(a ⋆ p)
    # we need the linear convolution of a (reversed) with p
    a_rev = a[::-1]
    conv_len = n + total - 1                # = 3·n‑2
    size2 = 1 << ((conv_len - 1).bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # keep central n entries
    return grad, b


# ----------------------------------------------------------------------
# Projected Adam‑style gradient descent on the smoothed max
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam‑style optimisation of the smoothed max of a*a."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 1
    rng = np.random.default_rng()

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive step – we keep the ℓ¹‑norm fixed afterwards
        lr_vec = 0.5 / (np.sqrt(v_hat) + eps)       # per‑coordinate
        a -= lr_vec * m_hat

        # projection onto the simplex (keeps the scale invariant)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaux
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 6e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # increase smoothness parameter
        if it % 200 == 0:
            beta = min(beta * 1.07, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improve ordering of a fixed multiset by random swaps (simulated annealing)."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.015 + 1e-9   # initial temperature
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]               # perform swap
        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # tiny random reshuffles to keep diversity
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
    return best


# ----------------------------------------------------------------------
# Small continuous annealing that moves mass between two positions
# ----------------------------------------------------------------------


def _value_transfer_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simulated annealing that transfers a small amount of mass from one
    coordinate to another while preserving the ℓ¹‑norm.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.015 + 1e-9
    cooling = 0.9996
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        # move a random fraction of the source entry
        eps = rng.random() * min(best[i] * 0.3, best[i] * 0.3)
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps

        cur_score = _max_convolution_fft(cand)
        dE = cur_score - best_score
        if cur_score <= best_score or rng.random() < np.exp(-dE / T):
            best = cand
            best_score = cur_score
        T *= cooling
    return best


# ----------------------------------------------------------------------
# Parametric families (only the useful power‑law one is kept)
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^(-p), normalised to ℓ¹‑sum = sqrt(2·n)."""
    idx = np.arange(n, dtype=np.float64)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    scale = target / raw.sum()
    a = raw * scale
    return a


# ----------------------------------------------------------------------
# Ordering heuristics
# ----------------------------------------------------------------------


def _orderings(base: np.ndarray) -> list[np.ndarray]:
    """
    Produce a short list of candidate orderings for the same multiset.
    The three patterns work well in practice.
    """
    asc = np.sort(base)               # small → large
    desc = asc[::-1]                  # large → small
    # alternating small‑large, starting with a small entry
    n = asc.size
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for k in range(n):
        if k % 2 == 0:
            alt[k] = asc[lo]
            lo += 1
        else:
            alt[k] = asc[hi]
            hi -= 1
    return [asc, desc, alt]

        if time.time() < deadline:
            best_seq = _value_transfer_annealing(best_seq, deadline)
            obj = _max_convolution_fft(best_seq)
            if obj < best_obj:
                best_obj = obj
                print(f"[{time.time():.1f}] After value‑transfer annealing obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 6. Final sanity checks and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # trim negligible tail entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # final report using the provided evaluator
    try:
        ev = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {ev:.6f}")
    except Exception:
        ev = None

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f}  length = {best_seq.size}")
    return best_seq.tolist()
