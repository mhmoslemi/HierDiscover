"""
Propose a candidate step‑function that minimizes ``evaluate_sequence``.

The algorithm is a streamlined version of the previous state‑of‑the‑art
pipeline:

* If a previous construction ``height_sequence_1`` exists, up‑sample it
  to a relatively large length (default 40000).  Otherwise start from a
  random simplex point.
* Run several long projected‑gradient‑descent runs on a smooth
  log‑sum‑exp surrogate of the max‑convolution.  The temperature β is
  annealed up to 1e6 so the surrogate becomes very close to the true
  max.
* With the best vector from the GD phase, perform an intensive
  simulated‑annealing hill‑climb that updates the full self‑convolution
  in O(n) per move.  The move set transfers a random amount of mass
  from a source (chosen proportionally to its contribution to the
  current worst convolution entry) to a destination (biased towards
  small entries).  Temperature is slowly cooled.
* If any time budget remains, a short final polishing GD run is
  performed.
* The returned list is projected onto the ℓ¹‑simplex
  {a_i ≥ 0, Σ a_i = √(2 n)}; consequently ``evaluate_sequence`` simply
  returns the maximum entry of the self‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (full convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float,
                          beta_start: float = 0.5,
                          beta_max: float = 1e6,
                          beta_factor: float = 1.4) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max.
    Momentum, temperature annealing and jitter are used.
    The loop terminates when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = beta_start
    step_base = target_sum * 0.20          # slightly larger base step
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # simplex projection
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 300 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σa = √(2 n).  Several families (power‑law, exponential, Dirichlet,
    piecewise) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 6)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers a random amount
    of mass from a source index to a destination index while keeping the
    full self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    # start temperature a fraction of the excess over the average
    avg = b.mean()
    T = max(cur_score - avg, 1e-6) * 0.5 + 1e-9
    cooling = 0.9996  # slower cooling for more exploration

    while time.time() < deadline:
        # ----- pick a worst convolution index -----
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        M = rng.choice(max_idxs)

        # ----- source index (weighted by contribution to entry M) -----
        i_low = max(0, M - (a.size - 1))
        i_high = min(a.size - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, a.size)

        if a[i] < 1e-14:
            continue

        # ----- destination index (bias towards small a) -----
        if rng.random() < 0.35:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(a.size, p=probs_j)
        else:
            j = rng.integers(0, a.size)

        if i == j:
            continue

        # ----- amount of mass to move (up to 80 % of source) -----
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution efficiently -----
        b_new = b.copy()
        # linear contribution of moved mass
        b_new[i:i + a.size] -= 2.0 * delta * a
        b_new[j:j + a.size] += 2.0 * delta * a
        # quadratic correction terms
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- Metropolis acceptance -----
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     target_n: int = 40000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the wall‑clock budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Initialise from previous construction if possible
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        earlier = globals()["height_sequence_1"]
        if isinstance(earlier, (list, tuple, np.ndarray)):
            a0 = np.asarray(earlier, dtype=float)
            n0 = a0.size
            a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
            print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0})")
        else:
            a0 = None
    else:
        a0 = None

    # ------------------------------------------------------------------
    # Determine the working length (use target_n unless time is too short)
    # ------------------------------------------------------------------
    n = int(min(target_n, max(2000, int(target_n * 0.7))))  # just a sanity guard
    target_sum = np.sqrt(2.0 * n)

    if a0 is not None and a0.size != n:
        a_current = _upsample_sequence(a0, n)
    elif a0 is not None and a0.size == n:
        a_current = a0.copy()
    else:
        a_current = _random_initial_sequence(n, rng)

    a_current = _project_to_simplex(a_current, target_sum)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_a = a_current.copy()
    best_score = _max_convolution_fft(best_a)
    print(f"[{time.time():.1f}] Initial max convolution = {best_score:.6f}")

    # allocate ~45 % of remaining time to GD
    gd_budget = (deadline - time.time()) * 0.45
    gd_deadline = time.time() + gd_budget
    n_restarts = 4
    for r in range(n_restarts):

        a_hc, b_hc = _hill_climbing(a_cur, b_cur, hc_deadline)
        score_hc = b_hc.max()
        if score_hc < best_score:

            best_a = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb final improvement: {best_score:.6f}")

    # ------------------------------------------------------------------

    except Exception as exc:  # pragma: no cover

    return best_a.tolist()
