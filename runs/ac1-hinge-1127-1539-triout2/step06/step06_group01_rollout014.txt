def _adam_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr_base = 0.1  # maybe scale by target_sum

    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute gradient 
        g, b = _gradient_softmax(a, beta)  # returns gradient (size n) and b (conv)
        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        lr = lr_base / (np.sqrt(v_hat) + 1e-8)  # elementwise
        step = lr * m_hat
        # Normalize step size: limit magnitude maybe
        step_norm = np.linalg.norm(step)
        if step_norm > 0:
            # Scale step to have appropriate step size relative to sum magnitude
            step = step * (target_sum * 0.2) / step_norm

        a -= step
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # beta schedule
        if t % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a

def _subgradient_descent(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = sqrt(2*n)
    step_size = target_sum * 0.2  # initial step; will adapt
    t = 0
    while time.time() < deadline:
        # compute conv via FFT
        conv = _full_convolution(a)
        max_val = conv.max()
        # get indices of max
        max_idxs = np.flatnonzero(conv == max_val)
        # compute gradient as average of contributions over max indices
        g = np.zeros_like(a)
        for k in max_idxs:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            # for each i, j = k - i
            for i in range(start_i, end_i + 1):
                g[i] += a[k - i]
        # average if multiple max indices
        g /= max(1, len(max_idxs))
        # Step diminishing
        t += 1
        # step size schedule: step = step0 / sqrt(t)
        eta = step_size / np.sqrt(t)
        a -= eta * g
        # ensure non-negative then project
        a = _project_to_simplex(a, target_sum)
        # optional jitter
        if t % 500 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
    return a

def _hardmax_gradient(a, max_idxs):
    n = a.size
    p = np.zeros(2*n-1, dtype=float)
    p[max_idxs] = 1.0 / len(max_idxs)
    # compute gradient similar to _gradient_softmax but without softmax step
    a_rev = a[::-1]
    conv_len = 3*n - 2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g

def propose_candidate(seed=42,budget_s=1000,...):

    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5

    best_seq = None

    best_score = np.inf

    # seed

    if "height_sequence_1" in globals():

         ... as earlier.

    schedule = [5000, 10000, 20000, 30000]

    for n in schedule:

        if time.time() >= deadline: break

        target_sum = np.sqrt(2.0*n)

        # Build list of init candidates

        candidates = []

        if best_seq is not None:

            a_upsampled = _upsample_sequence(best_seq, n)

            candidates.append(a_upsampled)

        # exponential shape

        lam = rng.uniform(0.0005, 0.02)   # adjust.

        a_exp = np.exp(-lam * np.arange(n))

        a_exp = a_exp * target_sum / a_exp.sum()

        candidates.append(a_exp)

        # random Dirichlet

        a_dir = rng.dirichlet(alpha=np.ones(n)) * target_sum

        candidates.append(a_dir)

        # random power law

        p = rng.uniform(0.1, 1.5)

        idx = np.arange(1, n+1)

        a_pow = (idx) ** (-p)

        a_pow = a_pow * target_sum / a_pow.sum()

        candidates.append(a_pow)

        # optionally more random if time.

        # For each candidate, run gradient descent and subgradient.

        local_best_a = None

        local_best_score = np.inf

        # Determine time for this n

        remaining = deadline - time.time()

        if remaining < 10: break

        # allocate half for gradient/subgradient, half for hill climbing

        phase_time = remaining * 0.45

        # decide per candidate time

        per_candidate_time = phase_time / max(1, len(candidates))

        # iterate each candidate

        for cand in candidates:

            if time.time() >= deadline: break

            # gradient descent restart

            gd_deadline = time.time() + per_candidate_time * 0.5

            a_gd = _gradient_descent_opt(cand, gd_deadline)

            score_gd = _max_convolution_fft(a_gd)

            if score_gd < local_best_score:

                local_best_score = score_gd

                local_best_a = a_gd

            # subgradient

            sg_deadline = time.time() + per_candidate_time * 0.5

            a_sg = _hardmax_subgradient_opt(cand, sg_deadline)

            score_sg = _max_convolution_fft(a_sg)

            if score_sg < local_best_score:

                local_best_score = score_sg

                local_best_a = a_sg

        # After candidate search, update global best

        if local_best_a is not None and local_best_score < best_score:

            best_seq = local_best_a.copy()

            best_score = local_best_score

            print(f"[{time.time():.1f}] New best after optimization for n={n} score={best_score:.6f}")

        # Hill climbing phase

        remaining = deadline - time.time()

        if remaining <= 5: break

        hill_deadline = time.time() + min(remaining*0.45, 200.0)

        if best_seq.size != n:

            a_current = _upsample_sequence(best_seq, n)

        else:

            a_current = best_seq.copy()

        a_current = _project_to_simplex(a_current, target_sum)

        b_full = _full_convolution(a_current)

        a_hc, b_hc = _hill_climbing(a_current, b_full, hill_deadline)

        score_hc = b_hc.max()

        if score_hc < best_score:

            best_score = score_hc

            best_seq = a_hc.copy()

            print(f"[{time.time():.1f}] Hill‑climb improvement for n={n} score={score_hc:.6f}")

        # Possibly run another subgradient after hill climbing to polish

        remaining = deadline - time.time()

        if remaining > 5:

            polish_deadline = time.time() + min(remaining, 30.0)

            score_polish = _max_convolution_fft(a_polish)

            if score_polish < best_score:

                best_score = score_polish

                best_seq = a_polish.copy()

                print(f"[{time.time():.1f}] Polishing subgradient improvement score={score_polish:.6f}")

    # Final polishing if time remains

    if best_seq is not None and time.time() < deadline - 5.0:

        n = best_seq.size

        target_sum = np.sqrt(2.0 * n)

        a = _project_to_simplex(best_seq.copy(), target_sum)

        b = _full_convolution(a)

        a_final, b_final = _hill_climbing(a, b, deadline)

        final_score = b_final.max()

        if final_score < best_score:

            best_seq = a_final.copy()

            best_score = final_score

            print(f"[{time.time():.1f}] Final hill climb improvement: {final_score:.6f}")

        # and final subgradient

        if time.time() < deadline - 2:

            a_final2 = _hardmax_subgradient_opt(best_seq, deadline)

            score_final2 = _max_convolution_fft(a_final2)

            if score_final2 < best_score:

                best_seq = a_final2.copy()

                best_score = score_final2

                print(f"[{time.time():.1f}] Final subgradient improvement: {score_final2:.6f}")

    # Clean-up

    if best_seq is None:

        n = 1000

        best_seq = np.full(n, np.sqrt(2.0 * n) / n)

    best_seq = np.asarray(best_seq, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]

    if nz.size:

        best_seq = best_seq[: nz[-1] + 1]

    else:

        best_seq = best_seq[:1]

    # Evaluate via evaluate_sequence

    try:

        val = evaluate_sequence(best_seq.tolist())

        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={best_seq.size})")

    except Exception as e:

        print("Evaluation error:", e)

    return best_seq.tolist()

"""
Propose a candidate step‑function that minimises the official
``evaluate_sequence`` objective (the maximum entry of the self‑convolution
when the ℓ¹‑norm is normalised to √(2 n)).

Algorithm
---------
* Initialise several candidate vectors for each length:
  - up‑sample the current best sequence,
  - a pure exponential, a random Dirichlet point and a power‑law.
* Two complementary continuous optimisers are run on each start:
  – projected gradient descent on the log‑sum‑exp (smooth) surrogate,
  – projected sub‑gradient descent on the exact hard ‑max (using the
    convolution‑gradient formula that can be evaluated with a single FFT).
* The best vector found is refined by a fast hill‑climbing stage that
  performs exact O(n) updates of the full self‑convolution while moving
  mass between indices.
* A final polishing stage repeats the sub‑gradient optimiser and a
  last hill‑climb on the longest length that fits into the remaining
  time budget.
* All vectors stay non‑negative and are projected onto the simplex
  {a_i≥0, Σa_i = √(2 n)} after every move, so the official evaluator sees a
  valid step function.

The routine respects the wall‑clock budget (default 1000 s) and prints
intermediate scores.  The best sequence found is returned as a plain
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of   (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # full convolution b = a*a
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Momentum, annealed inverse temperature β and jitter are used.
    The loop terminates when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 8192.0
    beta_factor = 1.4

    step_base = target_sum * 0.15
    momentum = np.zeros_like(a)
    rho = 0.90

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 350 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Hard‑max (sub‑gradient) utilities
# ----------------------------------------------------------------------


def _hardmax_gradient(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the hard max objective.
    For a given set of indices where the convolution attains its maximum,
    the gradient is 2·(a ⋆ p) where p is a uniform distribution on those
    indices.
    """
    n = a.size
    if max_idxs.size == 0:
        return np.zeros_like(a)

    p = np.zeros(2 * n - 1, dtype=float)
    p[max_idxs] = 1.0 / float(max_idxs.size)

    a_rev = a[::-1]
    conv_len = 3 * n - 2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g


def _hardmax_subgradient_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact max (no smoothing).
    Uses a diminishing step size proportional to 1/√t and projects onto the
    simplex after each step.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    it = 0
    # base magnitude – a fraction of the ℓ¹‑norm
    base_step = target_sum * 0.5

    while time.time() < deadline:
        # compute full self‑convolution
        b = _full_convolution(a)
        max_b = b.max()
        max_idxs = np.flatnonzero(b == max_b)

        # gradient of the hard max
        g = _hardmax_gradient(a, max_idxs)

        # diminishing step size
        it += 1
        norm_g = np.linalg.norm(g)
        if norm_g > 0.0:
            step = base_step / (norm_g * np.sqrt(it))
            a -= step * g

        a = _project_to_simplex(a, target_sum)

        # small jitter every few hundred iterations
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / np.sqrt(it), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Simulated‑annealing style hill climbing that transfers a random amount
    of mass from a source index to a destination index while keeping the
    full self‑convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * 0.1 + 1e-9          # initial temperature
    cooling = 0.9995                    # slower cooling for thorough search

    while time.time() < deadline:
        # pick a location where the convolution is maximal
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # source index – proportional to its contribution to entry M
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # destination – biased towards small entries 30 % of the time
        if rng.random() < 0.3:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount of mass to move (up to 80 % of source)
        max_delta = a[i] * 0.8
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # update the full convolution efficiently (linear + quadratic terms)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

            best_seq = a_final.copy()
            best_score = final_score
            print(f"[{time.time():.1f}] Final hill‑climb improvement: {best_score:.6f}")

        # final hard‑max sub‑gradient polishing (use any remaining time)
        if time.time() < deadline - 2.0:
            a_last = _hardmax_subgradient_opt(best_seq, deadline)
            last_score = _max_convolution_fft(a_last)
            if last_score < best_score:
                best_seq = a_last.copy()
                best_score = last_score
                print(f"[{time.time():.1f}] Final sub‑gradient improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Clean‑up: trim trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # 5) Log the official evaluation
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
