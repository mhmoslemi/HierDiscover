"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The code builds several parametric families
(power‑law, exponential, linear, inverse‑sqrt), tries a variety of
orderings (including symmetric edge‑to‑center placements), keeps an
elite set of the best candidates and then refines them with a
projected gradient descent on a smooth log‑sum‑exp surrogate of the
max‑convolution, followed by a permutation‑annealing phase that swaps
entries to lower the true max‑convolution.  The whole search respects
the supplied time budget (default 1000 s) and finally returns the best
list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex Σx = s, x≥0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection onto {x_i ≥ 0, Σx_i = s} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (g, b) where
        b = a*a (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smooth maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Annealed projected gradient descent on the log‑sum‑exp smooth max."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    max_beta = 1e6
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    it = 0
    rng = np.random.default_rng(0)

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size proportional to remaining “budget” of the simplex
            lr = 0.4 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 6e-4 / (beta + 1.0), size=n
            )
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal beta (increase smoothness)
        if it % 250 == 0 and beta < max_beta:
            beta = min(beta * 1.08, max_beta)

        it += 1
    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Improve the ordering of a fixed multiset by random swaps.
    The ℓ¹‑norm stays unchanged, only the permutation varies.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    # start temperature a small fraction of the current score
    T = best_score * 0.015 + 1e-10
    cooling = 0.9994
    rng = np.random.default_rng(1)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        if cur_score <= best_score or rng.random() < np.exp(-dE / T):
            # accept
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # tiny random reshuffle now and then
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.018
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, truncated linear,
#                inverse‑sqrt, symmetric power‑law)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale to meet the ℓ¹‑sum target (preserving non‑negativity)."""
    scale = target / np.sum(v)
    a = v * scale
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _invsqrt_sequence(n: int, eps: float = 1e-12) -> np.ndarray:
    """a_i ∝ 1 / sqrt((i+1)*(n-i))  – large at the boundaries, small in the centre."""
    i = np.arange(n, dtype=float) + 1.0
    denom = np.sqrt(i * (n - i + eps))
    raw = 1.0 / denom
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


def _sym_powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    """Symmetric sum of two power‑laws (both ends heavy)."""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p) + (n - idx - 1 + shift) ** (-p)
    target = np.sqrt(2.0 * n)
    return _normalize_to_sum(raw, target)


# ----------------------------------------------------------------------
# Ordering heuristics (including symmetric edge / centre placements)
# ----------------------------------------------------------------------


def _sym_edge_ordering(base: np.ndarray) -> np.ndarray:
    """Place the largest values alternating at the left and right ends."""
    desc = np.sort(base)[::-1]          # largest → smallest
    n = desc.size
    out = np.empty_like(base)
    l, r = 0, n - 1
    for k, val in enumerate(desc):
        if k % 2 == 0:
            out[l] = val
            l += 1
        else:
            out[r] = val
            r -= 1
    return out


def _sym_center_ordering(base: np.ndarray) -> np.ndarray:
    """Place the largest values around the centre and move outward."""
    desc = np.sort(base)[::-1]
    n = desc.size
    out = np.empty_like(base)
    mid = (n - 1) // 2
    left, right = mid, mid + (n % 2)
    out[mid] = desc[0]
    idx = 1
    toggle = True
    while idx < n:
        if toggle:
            left -= 1
            out[left] = desc[idx]
        else:
            out[right] = desc[idx]
            right += 1
        toggle = not toggle
        idx += 1
    return out


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Generate a modest collection of orderings for the same multiset.
    The goal is to reshuffle the values so that large‑by‑large overlaps
    are avoided.
    """
    n = base.size
    asc = np.sort(base)          # smallest → largest
    desc = asc[::-1]             # largest → smallest
    out = [asc, desc]

    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    out.append(alt)

    # symmetric edge ordering (largest at both ends, shrinking towards centre)
    out.append(_sym_edge_ordering(base))

    # symmetric centre ordering (largest in middle, decreasing outward)
    out.append(_sym_center_ordering(base))

    # step‑permutations with coprime strides
    for step in (2, 3, 5, 7, 11, 13, 17, 19, 23):
        if np.gcd(step, n) != 1:
            continue
        perm = np.empty_like(base)
        pos = 0
        for i in range(n):
            perm[pos] = asc[i]
            pos = (pos + step) % n
        out.append(perm)

    # a few random shuffles (preserve the multiset)
    for _ in range(5):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out
