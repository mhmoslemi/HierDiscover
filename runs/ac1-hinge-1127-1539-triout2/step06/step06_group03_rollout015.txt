def _center_out_order(values):
    sorted_desc = np.sort(values)[::-1]
    n = len(values)
    out = np.empty(n, dtype=values.dtype)
    center = n // 2
    out[center] = sorted_desc[0]
    left = center - 1
    right = center + 1
    idx = 1
    while idx < n:
        if left >= 0:
            out[left] = sorted_desc[idx]; idx+=1; left-=1
        if idx < n and right < n:
            out[right] = sorted_desc[idx]; idx+=1; right+=1
    # For odd n maybe last unfilled left or right. Works.
    return out

def _value_annealing(a0, deadline):
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)
    rng = np.random.default_rng()
    T = best_score * 0.02 + 1e-9
    cooling = 0.9998
    sigma = np.sqrt(best_score) * 0.01  # scale
    while time.time() < deadline:
        # proposal: add normal noise then project
        proposal = a + rng.normal(scale=sigma, size=n)
        # ensure non-negative by projection
        proposal = np.maximum(proposal, 0.0)
        proposal = _project_to_simplex(proposal, np.sum(a))  # keep sum constant
        cur = _max_convolution_fft(proposal)
        dE = cur - best_score
        accept = False
        if cur <= best_score:
            accept = True
        elif T > 0 and rng.random() < np.exp(-dE / T):
            accept = True
        if accept:
            a = proposal
            if cur < best_score:
                best_score = cur
                best = proposal.copy()
        T *= cooling
        sigma *= 0.9999  # gradually reduce noise
    return best

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return float(conv.max())

"""
Search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The algorithm consists of:

1.  A cheap exhaustive sweep over several parametric families
    (power‑law, exponential, linear, stretched‑exponential, Gaussian)
    and a handful of simple orderings (descending, alternating, random).
    The best few candidates (the *elite* set) are kept.

2.  For every elite candidate we run a projected gradient descent on a
    smooth log‑sum‑exp surrogate of the convolution maximum
    (the surrogate becomes the true maximum as the temperature β → ∞).

3.  The ordering of the resulting multiset is refined by a simulated‑annealing
    permutation phase and by a small value‑perturbation annealing, both
    respecting the ℓ¹‑simplex constraint.

4.  The best sequence found is returned as a plain list.  All steps respect
    the global time budget (default 1000 s) and avoid any file or network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the full linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex {x≥0, Σx = s} (Duchi et al. 2008)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of size s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate of max(b) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns g, b where
        b = a*a  (full convolution)
        g = ∇_a ( (1/β)·log Σ_k exp(β·b_k) )
          = 2·(a ⋆ p) ,  p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]

    # full convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities (stable)
    max_b = np.max(b)
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / np.sum(exp_shifted)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1                 # = 3·n‑2
    size2 = 1 << ((conv_len).bit_length())
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected gradient descent on the smoothed maximum
# ----------------------------------------------------------------------


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent with momentum on the log‑sum‑exp surrogate."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 1.0
    max_beta = 1e7
    momentum = np.zeros_like(a)
    rho = 0.9                     # momentum decay
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum step
        momentum = rho * momentum + (1.0 - rho) * g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            # step size inversely proportional to β – larger β = sharper surrogate
            lr = 0.6 * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum

        # simplex projection (keeps the ℓ¹‑norm exactly target_sum)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # anneal β (makes surrogate tighter)
        if it % 150 == 0 and beta < max_beta:
            beta = min(beta * 1.10, max_beta)

        it += 1

    return a


# ----------------------------------------------------------------------
# Permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improve the ordering of a fixed multiset by random swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]                     # propose swap

        cur_score = _max_convolution_fft(a)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            a[i], a[j] = a[j], a[i]                 # revert

        T *= cooling

        # occasional tiny reshuffle to keep diversity
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if np.any(mask):
                shuffled = a[mask]
                rng.shuffle(shuffled)
                a[mask] = shuffled
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

    return best


# ----------------------------------------------------------------------
# Value‑perturbation annealing (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------


def _value_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Add small Gaussian noise to the values while preserving the ℓ¹‑norm."""
    a = a0.copy()
    n = a.size
    target_sum = np.sum(a0)       # the sum is already the desired one
    best = a.copy()
    best_score = _max_convolution_fft(best)

    rng = np.random.default_rng()
    T = best_score * 0.015 + 1e-9
    cooling = 0.9996
    sigma = target_sum * 1e-3      # initial noise scale

    while time.time() < deadline:
        proposal = a + rng.normal(scale=sigma, size=n)
        proposal = np.maximum(proposal, 0.0)
        proposal = _project_to_simplex(proposal, target_sum)

        cur_score = _max_convolution_fft(proposal)
        dE = cur_score - best_score
        accept = False
        if cur_score <= best_score:
            accept = True
        elif rng.random() < np.exp(-dE / T):
            accept = True

        if accept:
            a = proposal
            if cur_score < best_score:
                best_score = cur_score
                best = proposal.copy()

        T *= cooling
        sigma *= 0.9999

    return best


# ----------------------------------------------------------------------
# Parametric families (power‑law, exponential, linear, stretched‑exp, Gaussian)
# ----------------------------------------------------------------------


def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale non‑negative vector to have ℓ¹‑norm exactly ``target``."""
    scale = target / np.sum(v)
    a = v * scale
    return np.maximum(a, 0.0)


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    if raw.sum() == 0.0:
        raw[0] = 1.0
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _stretched_exp_sequence(n: int, p: float, lam: float, shift: float) -> np.ndarray:
    """a_i = (i+shift)^{-p} * exp(-lam * i)"""
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p) * np.exp(-lam * idx)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _gaussian_sequence(n: int, sigma_frac: float) -> np.ndarray:
    """Centered Gaussian with standard deviation = sigma_frac·n."""
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    raw = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Ordering heuristics
# ----------------------------------------------------------------------


def _alternating_order(v: np.ndarray) -> np.ndarray:
    """Small‑large alternating ordering (starting with the smallest)."""
    asc = np.sort(v)
    n = asc.size
    out = np.empty_like(v)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            out[i] = asc[lo]
            lo += 1
        else:
            out[i] = asc[hi]
            hi -= 1
    return out


def _center_out_order(v: np.ndarray) -> np.ndarray:
    """Place the largest value in the centre and walk outwards."""
    d = np.sort(v)[::-1]               # descending
    n = d.size
    out = np.empty_like(v)
    if n % 2 == 1:
        mid = n // 2
        out[mid] = d[0]
        left, right = mid - 1, mid + 1
        pos = 1
        while pos < n:
            if left >= 0:
                out[left] = d[pos]
                left -= 1
                pos += 1
            if pos < n and right < n:
                out[right] = d[pos]
                right += 1
                pos += 1
    else:
        left, right = n // 2 - 1, n // 2
        pos = 0
        while pos < n:
            if left >= 0:
                out[left] = d[pos]
                left -= 1
                pos += 1
            if pos < n and right < n:
                out[right] = d[pos]
                right += 1
                pos += 1
    return out


def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest collection of orderings for a given multiset."""
    asc = np.sort(base)          # smallest → largest
    desc = asc[::-1]             # largest → smallest
    alt = _alternating_order(asc)
    cent = _center_out_order(desc)

    out = [desc, asc, alt, cent]

    # a few random shuffles (preserve multiset)
    for _ in range(4):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        out.append(shuffled)

    return out

            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final optional polishing (long gradient on the current best)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0 and best_seq is not None:
        final_dead = deadline - 2.0
        best_seq = _gradient_descent_opt(best_seq, final_dead)
        best_obj = _max_convolution_fft(best_seq)

    # ------------------------------------------------------------------
    # 4.  Post‑processing and reporting
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 3000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_obj = _max_convolution_fft(best_seq)

    # Trim trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # Verify with the official evaluator (for debugging output)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
