"""
Improved search for a step‐function minimizing
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three layers:

1. **Diversified initialisation** – families of analytic shapes
   (uniform, exponential, power‑law, edge‑heavy reciprocal and a
   random sparse “spike” pattern) are sampled and projected onto the
   ℓ¹‑simplex Σa = √(2·n).  The current global best (if any) is also
   up‑sampled to provide a warm start for larger problem sizes.

2. **Smooth Adam optimisation** – the peak of the convolution is
   approximated by a log‑sum‑exp (soft‑max) with a slowly increasing
   temperature.  Adam updates are performed on the simplex with an
   optional symmetry projection.

3. **Simulated‑annealing refinement** – after the Adam phase a cheap
   Monte‑Carlo “mass‑swap” annealer perturbs two entries at a time,
   accepting worsening moves with a temperature schedule.  This stage
   is able to escape shallow local minima left by the gradient based
   phase.

The whole routine is time‑aware: it repeatedly restarts with different
lengths (2 k … 50 k) while keeping the overall best solution found
within the allocated budget (default 1000 s).  The final sequence is
stored in the global name ``height_sequence_1`` for possible reuse.
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x>=0, sum x = s}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a * a (length = 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Soft‑max weights for a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    return exp_shifted / exp_shifted.sum()


# ----------------------------------------------------------------------
# Gradient of the smooth objective (log‑sum‑exp) w.r.t a
# ----------------------------------------------------------------------
def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Returns the gradient of    (1/α)·log‑sum‑exp(α·(a*a))
    w must be the soft‑max probabilities over the convolution entries.
    """
    n = a.shape[0]
    # the "kernel" is the reversed weight vector
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # pick the slice that corresponds to d/d a_i
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * full[idx]


# ----------------------------------------------------------------------
# Optional symmetry enforcement (exact mirror symmetry)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Initial sequence generators – a richer family than the previous version
# ----------------------------------------------------------------------
def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n whose ℓ¹‑norm equals √(2·n).
    A mixture of analytical families is used for diversity.
    """
    S = np.sqrt(2.0 * n)

    # choose a family
    typ = rng.integers(0, 8)   # 0–7 inclusive

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 2:
        # geometric decay from the left (asymmetric)
        rate = rng.uniform(0.90, 0.9999)
        a = rate ** np.arange(n)

    elif typ == 3:
        # power‑law decay (heavy head, long tail)
        p = rng.uniform(0.3, 1.2)
        a = (np.arange(1, n + 1) ** (-p))

    elif typ == 4:
        # edge‑heavy reciprocal shape (same as before)
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 5:
        # triangular (peak in centre)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    elif typ == 6:
        # random sparse spike pattern – pick k positions, equal mass
        k = rng.integers(int(0.05 * n), int(0.15 * n) + 1)
        a = np.zeros(n)
        idxs = rng.choice(n, size=k, replace=False)
        a[idxs] = 1.0

    else:
        # convex combination of a few bases (as in the original code)
        cand = []

        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)
        cand.append(rate ** np.arange(n))               # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                       # reciprocal

        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    # Normalise to required ℓ¹‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation of the peak
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.45
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.4
    alpha_mul = 1.03
    alpha_cap = 200.0               # a larger cap than before

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max gradient
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter (helps escape plateaus)
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.15:
                    a = cand

        # temperature & learning‑rate schedules
        alpha = min(alpha * alpha_mul, alpha_cap)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max entry (single active shift)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # build the sub‑gradient for the max shift
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_max = np.max(_full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated‑annealing flavour)
        if rng.random() < 0.05:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.2:
                    a = cand
                    cur_max = cand_max
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()

        # early stop if stagnating
        if it % 6000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climbing (deterministic local search)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement – works on the exact max‑convolution
# ----------------------------------------------------------------------
def _anneal_refine(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap Metropolis–Hastings style refinement."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = np.max(_full_convolution(a))

    # initialise temperature as a fraction of the current peak
    T = best_val * 0.3
    it = 0
    while time.time() < deadline:
        it += 1
        # propose moving a small fraction of mass from i to j
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = rng.uniform(0.0, 0.06 * a[i])
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is not required – sum is unchanged and non‑negativity holds
        cand_val = np.max(_full_convolution(cand))

        # Metropolis acceptance
        if cand_val < best_val or rng.random() < np.exp(-(cand_val - best_val) / T):
            a = cand
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()

        # exponential cooling
        if it % 800 == 0:
            T *= 0.93
            if T < 1e-6:
                T = 1e-6

    return best_a, best_val


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → sub‑gradient → local → anneal)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # allocate roughly 55 % / 30 % / 10 % / 5 % of the remaining time
    t_adam = now + 0.55 * remaining
    a_adam, _ = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, _ = _optimize_subgrad(a_adam, t_sub, rng, enforce_sym)

    t_loc = min(time.time() + 0.10 * remaining, deadline)
    a_loc, _ = _local_swap_search(a_sub, t_loc, rng, enforce_sym)

    a_ref, final_val = _anneal_refine(a_loc, deadline, rng)

    # pick the best among every phase
    candidates = [
        (a_adam, np.max(_full_convolution(a_adam))),
        (a_sub,  np.max(_full_convolution(a_sub))),
        (a_loc,  np.max(_full_convolution(a_loc))),
        (a_ref,  final_val)
    ]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function defined in the problem statement.
    The routine respects the time budget and returns the best list it found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # try to reuse a previously discovered construction, if any
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main timed loop – many short restarts with varying length
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # problem size decision – gradually favour larger n as we progress
        if best_seq is not None and rng.random() < 0.57:
            scale = rng.uniform(0.9, 1.1)
            n = max(1800, int(best_n * scale))
        else:
            n = int(rng.integers(2000, 50001))

        enforce_sym = rng.random() < 0.45

        a0 = _initial_sequence(base_seq, n, rng)
        if enforce_sym:
            a0 = _symmetrize(a0)

        # give each attempt a slice of the remaining budget (12–20 s)
        slice_deadline = min(time.time() + rng.uniform(12.0, 20.0), deadline)

        a_opt, score = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_score:.6f}")

        # early exit if we already beat the target comfortably
        if best_score <= 1.45:
            break

        # reset the base when we have been idle for a while
        if time.time() - last_improve > 50.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal_best={best_score:.6f} "
          f"len={len(final_seq)} attempts={attempts}")

    # expose the best sequence for possible up‑sampling later
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
