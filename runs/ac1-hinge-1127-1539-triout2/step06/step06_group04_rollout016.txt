def _aggressive_hill_climb(a0, b0, deadline, rng):

    b = b0.copy()
    n = a.size
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    T = cur_max * 0.05  # initial temperature
    cooling = 0.9998

    while time.time() < deadline:
        k = int(np.argmax(b))
        # overlapping interval
        start = max(0, k - (n - 1))
        end = min(k, n - 1)

        # compute contributions
        idx = np.arange(start, end + 1)
        contrib = a[idx] * a[k - idx]

        # source idx: choose weighted by contribution (largest)
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            source_idx = rng.choice(idx, p=probs)
        else:
            # fallback choose random with mass
            prob_mass = a[idx] / a[idx].sum() if a[idx].sum()>0 else np.full_like(idx, 1/len(idx))
            source_idx = rng.choice(idx, p=prob_mass)

        # Choose destination index from complement (zero contribution)
        # Complement intervals:
        left_end = start - 1
        right_start = end + 1
        cand = []
        if left_end >= 0:
            cand.append(np.arange(0, left_end+1))
        if right_start < n:
            cand.append(np.arange(right_start, n))
        if cand:
            dest_pool = np.concatenate(cand)
        else:
            dest_pool = np.arange(n)  # fallback all
        dest_idx = rng.choice(dest_pool)

        delta = a[source_idx]
        if delta <= 0:

"""
Hybrid search for the minimal self‑convolution peak.

- Start from the previously found sequence (if any) or from a random
  non‑negative simplex vector.
- For each length in a schedule we
    * briefly run a projected (log‑sum‑exp) gradient descent to obtain a
      smooth shape,
    * then run an aggressive hill‑climber that repeatedly moves *all* mass
      from a source that contributes most to the current maximal convolution
      entry to a destination that does not contribute to that entry
      (i.e. lies outside the overlapping interval).  Moves are accepted
      with a Metropolis rule and a slowly cooling temperature.
- The best sequence seen so far is kept and is up‑sampled when the length
  schedule increases.
- Finally a polishing hill‑climb runs until the wall‑clock budget expires.

The algorithm stays within the 1000 s budget and returns the best
candidate as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Smoothed‑max gradient descent (kept from the previous implementation)
# --------------------------------------------------------------

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the log‑sum‑exp surrogate of max(a*a)."""
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Projected gradient descent on the smoothed maximum."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                       # power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# --------------------------------------------------------------
# Aggressive hill‑climbing (moves entire source mass)
# --------------------------------------------------------------

def _aggressive_hill_climb(a0: np.ndarray,
                           b0: np.ndarray,
                           deadline: float,
                           rng) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing that *fully* removes the source index that contributes most
    to the current maximal convolution entry and places that mass at an
    index that does not contribute to that entry.
    Acceptance follows a simple Metropolis rule with a slowly cooling
    temperature.
    Returns the best sequence seen and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    T = cur_max * 0.05 + 1e-12          # initial temperature
    cooling = 0.9995

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # Interval where a_i and a_{k_max-i} both exist (non‑zero contribution)
        start = max(0, k_max - (n - 1))
        end = min(k_max, n - 1)

        idx = np.arange(start, end + 1, dtype=int)
        # contribution of each index to the peak
        contrib = a[idx] * a[k_max - idx]

        # choose source index – the one with the largest contribution
        if contrib.sum() > 0:
            src = idx[np.argmax(contrib)]
        else:
            # fallback: any index with positive mass in the overlapping range
            pos = idx[a[idx] > 0]
            if pos.size == 0:
                src = rng.integers(0, n)
            else:
                src = rng.choice(pos)

        # destination must lie *outside* the overlapping interval,
        # guaranteeing zero contribution to the current peak.
        left = np.arange(0, start, dtype=int) if start > 0 else np.array([], dtype=int)
        right = np.arange(end + 1, n, dtype=int) if end + 1 < n else np.array([], dtype=int)
        candidates = np.concatenate((left, right))
        if candidates.size == 0:
            # shouldn't happen, but fall back to a random index
            dst = rng.integers(0, n)
        else:
            dst = rng.choice(candidates)

        if src == dst:
            continue

        delta = a[src]                     # move *all* mass from src
        if delta <= 1e-18:
            continue

        # ----- analytical update of the full convolution -----------------
        b_new = b.copy()

        # linear part: -2*delta*a shifted by src, +2*delta*a shifted by dst
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[dst:dst + n] += 2.0 * delta * a

        # quadratic correction terms
        k2s = 2 * src
        k2d = 2 * dst
        ks_d = src + dst
        if k2s < b_new.size:
            b_new[k2s] += delta * delta
        if k2d < b_new.size:
            b_new[k2d] += delta * delta
        if ks_d < b_new.size:
            b_new[ks_d] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ----- acceptance test -----------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            # guard against numerical negatives
            a[src] = max(a[src], 0.0)
            a[dst] = max(a[dst], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] aggressive hill improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimizes the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    wall‑clock budget (default 1000 s) and returns the best sequence found
    as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Use the previously saved construction, if present.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – start modest, then enlarge.
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 12000, 18000, 25000]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Initialise a candidate for this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size != n:
            a = _upsample_sequence(best_seq, n)
        else:
            a = _random_initial_sequence(n, rng)

        a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # Phase A – very short gradient descent (provides a smooth start)
        # --------------------------------------------------------------
        time_left = deadline - time.time()
        phaseA_dead = time.time() + min(15.0, 0.1 * time_left)
        a = _gradient_descent_opt(a, phaseA_dead)

        # --------------------------------------------------------------
        # Phase B – aggressive hill climbing (the heavy work)
        # --------------------------------------------------------------
        if time.time() < deadline:
            b_full = _full_convolution(a)
            time_left = deadline - time.time()
            phaseB_dead = time.time() + min(180.0, 0.85 * time_left)
            a_hc, max_hc = _aggressive_hill_climb(a, b_full, phaseB_dead, rng)
            if max_hc < best_score:
                best_score = max_hc
                best_seq = a_hc.copy()
                print(f"[{time.time():.1f}] Hill‑climb improvement (n={n}) "
                      f"score={max_hc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with any remaining budget
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final = best_seq.copy()
        b_final = _full_convolution(a_final)
        a_final, max_final = _aggressive_hill_climb(a_final, b_final, deadline, rng)
        if max_final < best_score:
            best_score = max_final
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: "
                  f"score={max_final:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
