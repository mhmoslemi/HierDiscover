def _gradient_exact_max(a):
    n = a.shape[0]
    # compute b
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    b = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
    max_b = b.max()
    # mask indices within tolerance
    eps = max_b * 1e-12 + 1e-15  # safety
    mask = (b >= max_b - eps)
    total = mask.sum()
    if total == 0:
        p = np.zeros_like(b)
    else:
        p = mask.astype(float) / total
    # gradient
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, max_b

"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
1.  Start from a previously found sequence (``height_sequence_1``) if it
    exists, otherwise random Dirichlet / exponential / power‑law
    initialisations.
2.  Run a *sub‑gradient descent* on the exact ℓ∞‑norm of the
    self‑convolution.  The gradient of ``max_k (a*a)[k]`` is obtained
    by a cheap FFT‑based convolution with a probability mass put on the
    active (maximal) convolution indices.
3.  Refine the result with a simulated‑annealing hill‑climber that
    transfers a random amount of mass from a source index to a destination
    index while keeping the full convolution up‑to‑date in O(n) per move.
    The temperature starts relatively high (≈0.5·max) and cools very
    slowly, allowing occasional uphill moves.
4.  Increase the problem size progressively (2000 → 5000 → 10000 →
    20000 → 40000 entries).  At each size the above two phases are run
    within the remaining wall‑clock budget.
5.  A final polishing hill‑climb uses the whole remaining time.
6.  Trailing near‑zero entries are stripped before returning the list.

All vectors are projected onto the ℓ¹‑simplex
{a_i≥0, Σa_i = √(2·n)} after every stochastic move, guaranteeing that
``evaluate_sequence`` sees a valid step‑function.

The routine respects the supplied ``budget_s`` (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (FFT based)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the exact max (sub‑gradient)
# ----------------------------------------------------------------------


def _gradient_exact_max(a: np.ndarray):
    """
    Returns a sub‑gradient of  max_k (a*a)[k]  and the current max value.
    The gradient is  g = 2·(a_rev ⋆ p)  where p is a uniform distribution
    over the indices attaining the maximum.
    """
    n = a.shape[0]

    # b = a*a
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    b = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]

    max_b = b.max()
    eps = max_b * 1e-12 + 1e-15
    mask = (b >= max_b - eps)
    cnt = mask.sum()
    if cnt == 0:
        p = np.zeros_like(b)
    else:
        p = mask.astype(float) / cnt

    # gradient = 2·(a_rev ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]

    return g, max_b


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected (sub‑)gradient descent on the exact max of the self‑convolution.
    Momentum, step‑size annealing and occasional jitter are used.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    rho = 0.90
    step_factor = 1.0          # initial magnitude of the step
    it = 0

    while time.time() < deadline:
        g, cur_max = _gradient_exact_max(a)
        norm_g = np.linalg.norm(g)
        if norm_g > 0.0:
            lr = step_factor * target / (norm_g + 1e-12)
            momentum = rho * momentum + (1.0 - rho) * g
            a -= lr * momentum
        else:
            # jitter when gradient vanishes
            a += rng.normal(scale=target * 0.001, size=n)

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target)

        # anneal step size slowly
        if it % 200 == 0 and step_factor > 0.05:
            step_factor *= 0.90

        # occasional jitter for escaping plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        it += 1

    return a

def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Generate a random non‑negative vector on the ℓ¹‑simplex."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 8)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # exponential decay
        lam = rng.uniform(0.015, 0.045)
        a = np.exp(-lam * idx)

    elif family == 1:                   # power‑law
        p = rng.uniform(0.2, 1.8)
        a = (idx + 1.0) ** (-p)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.2, 1.2)
        lam = rng.uniform(0.015, 0.035)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n))

    elif family == 4:                   # piecewise exponential
        split = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.uniform(0.015, 0.035)
        lam2 = rng.uniform(0.015, 0.035)
        a = np.concatenate([np.exp(-lam1 * np.arange(split)),
                            np.exp(-lam2 * np.arange(split, n))])

    elif family == 5:                   # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    elif family == 6:                   # sinusoidal modulation
        freq = rng.uniform(1.0, 3.0)
        a = (np.sin(np.pi * freq * idx / n) ** 2) + 0.1

    else:                               # absolute Gaussian draws
        a = np.abs(rng.normal(scale=1.0, size=n))

    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float,
                   T_factor: float = 0.5, cooling: float = 0.9995):
    """
    Simulated‑annealing style hill climbing.
    a0: current vector, b0: its full self‑convolution.
    Moves a random amount of mass from a source index to a destination
    while keeping the full convolution up‑to‑date in O(n) per move.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    rng = np.random.default_rng()
    T = cur_score * T_factor + 1e-9
    while time.time() < deadline:
        # ----- pick a max‑convolution index ---------------------------------
        max_idxs = np.flatnonzero(b == cur_score)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # ----- source index (weighted by contribution to entry M) ----------
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # ----- destination index (prefer small entries) --------------------
        if rng.random() < 0.35:          # more frequent bias to small entries
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # ----- transferable mass -------------------------------------------
        max_delta = a[i] * 0.85          # allow moving up to 85 % of the source
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ----- update convolution (linear + quadratic terms) ---------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_score = b_new.max()

        # ----- Metropolis acceptance ----------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

            cur = _max_convolution_fft(a0)

        score_hc = b_hc.max()
        print(f"[{time.time():.1f}] hill climbing (n={n}) score={score_hc:.6f}")

        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] New global best after hill‑climb: {best_score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        a = _project_to_simplex(best_seq.copy(), target_sum)
        b = _full_convolution(a)
        a_final, b_final = _hill_climbing(a, b, deadline,
                                          T_factor=0.2, cooling=0.9999)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: {final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        n = 5000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:  # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
