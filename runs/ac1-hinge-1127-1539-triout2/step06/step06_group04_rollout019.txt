"""
Improved search for a non‑negative step function that minimizes
   evaluate_sequence(sequence)

The algorithm consists of three stages:

1. **Fast parametric scan** (n≈4000) over a handful of simple families
   (gaussian, exponential, power‑law, beta‑shaped).  The best candidate is
   kept as a seed.

2. **Large‑scale Adam optimisation** on a longer vector (n≈18000).  The
   surrogate objective is a smooth log‑sum‑exp approximation of the
   convolution maximum; Adam’s adaptive steps together with a β‑annealing
   schedule give a fast and stable descent.  After each update the vector
   is projected onto the ℓ¹‑simplex Σa = √(2 n).

3. **Targeted hill‑climbing** that moves a tiny amount of mass from the
   indices that contribute most to the current peak towards indices that
   contribute least.  An exact update of the whole convolution is used,
   so the cost per move is O(n).

The whole routine respects the supplied wall‑clock budget (default 1000 s)
and returns the best sequence found.  The code is self‑contained; the
required ``evaluate_sequence`` function is assumed to be already imported.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new, np.sqrt(2.0 * new_n))
    return a_new


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with   g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (via FFT)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)  (correlation)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed max
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected Adam optimiser for the smoothed max objective.
    """
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)

    beta = 0.5          # start β for the log‑sum‑exp surrogate
    beta_max = 32768.0
    beta_factor = 1.5

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # a modest base learning‑rate; the schedule below rescales it.
    base_lr = 0.12 * target / np.sqrt(n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        mh = m / (1 - beta1 ** it)
        vh = v / (1 - beta2 ** it)

        # Adam step (no momentum term, just adaptive scaling)
        step = base_lr / (np.sqrt(vh) + eps)
        a -= step * mh

        # projection onto the ℓ¹‑simplex
        a = _project_to_simplex(a, target)

        # occasional jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)

        # β‑annealing (make the surrogate sharper)
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

    return a


# ----------------------------------------------------------------------
# Targeted hill‑climbing (exact convolution updates)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Move a little mass from indices that contribute a lot to the current
    maximal convolution entry towards indices that contribute little.
    Returns the best sequence encountered and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # initial temperature for Metropolis steps
    cooling = 0.9995

    while time.time() < deadline:
        # index of the current peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak:
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # acceptance test
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max


# ----------------------------------------------------------------------
# Simple parametric families (used for the initial quick scan)
# ----------------------------------------------------------------------


def _make_gaussian(n: int, sigma_factor: float) -> np.ndarray:
    """Gaussian bump centred in the middle; sigma = sigma_factor * n."""
    sigma = sigma_factor * n
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return a


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """Pure exponential decay a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    return a


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete beta‑shaped bump:
        a_i ∝ ((i+1)/n)^α * (1 - i/(n-1))^β .
    """
    i = np.arange(n, dtype=float)
    a = ((i + 1) / n) ** alpha * (1 - i / (n - 1)) ** beta
    return a


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    wall‑clock budget (default 1000 s) and returns the best sequence found
    as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # If a previously saved construction exists, use it as a seed.
    # ------------------------------------------------------------------
    earlier = globals().get("height_sequence_1")
    if isinstance(earlier, (list, tuple, np.ndarray)):
        a0 = np.asarray(earlier, dtype=float)
        n0 = a0.size
        a0 = _project_to_simplex(a0, np.sqrt(2.0 * n0))
        best_seq = a0.copy()
        best_score = _max_convolution_fft(a0)          # equals objective after projection
        print(f"[{time.time():.1f}] Seeded from height_sequence_1 (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0. QUICK PARAMETRIC SCAN (small n) --------------------------------
    # ------------------------------------------------------------------
    n_small = 4000
    target_small = np.sqrt(2.0 * n_small)

    # families to explore ------------------------------------------------
    # 1) Gaussian
    for sigma_factor in np.linspace(0.02, 0.40, 15):
        a = _make_gaussian(n_small, sigma_factor)
        a = _project_to_simplex(a, target_small)
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Gaussian σ={sigma_factor:.3f} → {score:.6f}")

    # 2) Exponential
    for lam in np.linspace(0.0003, 0.015, 15):
        a = _make_exponential(n_small, lam)
        a = _project_to_simplex(a, target_small)
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Exponential λ={lam:.5f} → {score:.6f}")

    # 3) Power‑law
    for p in np.linspace(0.2, 1.6, 15):
        a = _make_powerlaw(n_small, p)
        a = _project_to_simplex(a, target_small)
        score = _max_convolution_fft(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Power‑law p={p:.3f} → {score:.6f}")

    # 4) Beta‑shape
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta_val in [0.5, 1.0, 1.5, 2.0]:
            a = _make_beta(n_small, alpha, beta_val)
            a = _project_to_simplex(a, target_small)
            score = _max_convolution_fft(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Beta α={alpha} β={beta_val} → {score:.6f}")

    # give up early if the budget is already exhausted
    if time.time() >= deadline:
        final = best_seq.copy()
        nz = np.where(final > 1e-12)[0]
        final = final[: nz[-1] + 1] if nz.size else final[:1]
        print(f"[{time.time():.1f}] early exit, score={best_score:.6f}")
        try:
            val = evaluate_sequence(final.tolist())
            print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (len={final.size})")
        except Exception as exc:
            print(f"Error calling evaluate_sequence: {exc}")
        return final.tolist()

    # ------------------------------------------------------------------
    # 1. UPSAMPLE TO A LARGE PROBLEM SIZE
    # ------------------------------------------------------------------
    n_big = 18000
    target_big = np.sqrt(2.0 * n_big)

    if best_seq is not None and best_seq.size != n_big:
        a_big = _upsample_sequence(best_seq, n_big)
    else:
        a_big = np.full(n_big, target_big / n_big, dtype=float)

    a_big = _project_to_simplex(a_big, target_big)

    # ------------------------------------------------------------------
    # 2. ADAM OPTIMISATION (major share of remaining budget)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    phaseA_dead = time.time() + min(0.60 * time_left, 350.0)   # at most ~350 s
    a_big = _adam_opt(a_big, phaseA_dead)

    score = _max_convolution_fft(a_big)
    if score < best_score:
        best_score = score
        best_seq = a_big.copy()
        print(f"[{time.time():.1f}] Adam optimisation improvement → {score:.6f}")

    # ------------------------------------------------------------------
    # 3. TARGETED HILL‑CLIMBING (fine‑tune)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        b_full = _full_convolution(best_seq)
        a_hc, max_hc = _hill_climb_targeted(best_seq, b_full, deadline)
        if max_hc < best_score:
            best_score = max_hc
            best_seq = a_hc.copy()
            print(f"[{time.time():.1f}] Hill‑climb final improvement → {max_hc:.6f}")

    # ------------------------------------------------------------------
    # 4. Trim trailing near‑zero entries for neatness
    # ------------------------------------------------------------------
    final = np.asarray(best_seq, dtype=float)
    nz = np.where(final > 1e-12)[0]
    if nz.size:
        final = final[: nz[-1] + 1]
    else:
        final = final[:1]

    # ------------------------------------------------------------------
    # 5. Report the official evaluation value
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={final.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return final.tolist()
