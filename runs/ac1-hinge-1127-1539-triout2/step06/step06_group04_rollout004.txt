def hill_climb_general(a, b, deadline, max_iters=None):
    n = len(a)
    best_a = a.copy()
    best_max = b.max()
    cur_max = best_max
    cur_b = b
    # compute initial temperature T as cur_max * factor
    T = cur_max * 0.1 + 1e-9   # maybe higher
    cooling = 0.9995
    step_update = 0

    while time.time() < deadline:
        # compute source and dest probabilities
        # Use contributions from current max peak
        k_max = int(np.argmax(cur_b))
        # contributions
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]
        total = contrib.sum()
        if total > 0:
            prob_src = contrib / total
        else:
            prob_src = a / a.sum()
        # dest probabilities reversed contributions
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        i = rng.choice(n, p=prob_src)
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue
        # determine delta
        max_delta = a[i] # allow up to all mass
        # maybe limit delta relative to cur_max, e.g., delta = min(max_delta, cur_max/(2*a[j_arr?])?). Hard to compute, just random.
        delta = rng.random() * max_delta
        # update b vector new
        # but we need to compute new b by updating cur_b clone? We can compute new b directly using formula with a, not need to clone fully? But for checking max we need copy.
        b_new = cur_b.copy()
        # Apply linear shift updates
        # shift source: subtract 2*delta*a at offset i
        start_i = i
        end_i = i + n
        # Subtract from slice; ensure it doesn't exceed array bounds (2n-1)
        if end_i > b_new.size:
            # truncate length
            b_new[start_i:] -= 2.0 * delta * a[:b_new.size-start_i]
        else:
            b_new[start_i:end_i] -= 2.0 * delta * a
        # Add at dest shift j
        start_j = j
        end_j = j + n
        if end_j > b_new.size:
            b_new[start_j:] += 2.0 * delta * a[:b_new.size-start_j]
        else:
            b_new[start_j:end_j] += 2.0 * delta * a
        # Quadratic terms
        k2i = 2 * i
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        k2j = 2 * j
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        kij = i + j
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta
        new_max = b_new.max()
        # acceptance
        if new_max < cur_max - 1e-12 or rng.random() < np.exp(-(new_max - cur_max) / T):
            # accept
            a[i] -= delta
            a[j] += delta
            # update cur_b to b_new
            cur_b = b_new
            cur_max = new_max
            # best tracking
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                # optional print
                print(f"[{time.time():.1f}] New best max: {best_max:.6f}")
        T *= cooling
    return best_a, best_max

The objective in `evaluate_sequence` is simply the maximum entry of the
self‑convolution `b = a*a` (the denominator cancels because the
ℓ¹‑norm of the sequence is forced to `√(2·n)`).  
Hence we minimise

```
max(b)  with   Σ a_i = √(2·n) ,   a_i ≥ 0 .
```

The algorithm below combines three ingredients

* **Projected gradient descent on a smooth approximation of `max(b)`**  
  (log‑sum‑exp).  This gives a good “global” shape for the sequence.

* **Targeted hill‑climbing** – a cheap O(n) update of the convolution
  after moving a small amount of mass from a “bad’’ index to a “good’’
  index.  The source is chosen proportionally to its contribution to
  the current maximal convolution entry, the destination to the inverse
  contribution.  Simulated‑annealing acceptance allows occasional
  uphill moves.

* **Length schedule & up‑sampling** – we start with a modest length,
  then gradually enlarge the vector (2000 → 5000 → 8000 → 12000 →
  18000).  Each larger size is seeded by linear interpolation of the
  best solution found so far.

The whole search respects the supplied wall‑clock budget (default 1000 s)
and returns the best sequence discovered.  All helper functions are
defined at top level, there are no closures or lambdas, and the code
uses only `numpy` and the standard library.

```python
"""
Proposed search for a step‐function minimizing ``evaluate_sequence``.
The algorithm consists of:

1. a projected gradient descent on a log‑sum‑exp smooth approximation
   of the max‑convolution,
2. an aggressive hill‑climber that moves a mass fraction from indices
   that largely contribute to the current peak towards indices that
   contribute little – updates are performed analytically in O(n),
3. a length schedule (2000 → 5000 → 8000 → 12000 → 18000) with linear
   up‑sampling of the current best solution.

The search stops exactly at the given time budget and returns the best
sequence (as a plain Python list).  The objective value is simply the
maximum entry of the self‑convolution; `evaluate_sequence` will compute
the same number.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x | x_i ≥ 0, Σ x_i = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b) where b = a*a.
    Returns (g, b) with g = 2·(a ⋆ p) and p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # correlation a ⋆ p (same length as a)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the smoothed maximum.
    Momentum, β‑annealing and occasional jitter are used.
    """
    a = a0.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.12
    momentum = np.zeros_like(a)
    rho = 0.92

    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1

    return a

def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = math.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:                       # pure power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                     # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:                     # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                     # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                     # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:                     # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:                                 # random Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = math.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Targeted hill‑climbing (mass transfer)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray,
                         b0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climber.
    Moves a fraction of mass from indices that strongly contribute to the
    current maximal convolution entry towards indices that contribute little.
    Uses a simulated‑annealing acceptance rule.
    Returns the best sequence found and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # temperature proportional to the current peak
    T = cur_max * 0.1 + 1e-12
    cooling = 0.9997

    it = 0
    while time.time() < deadline:
        # index of current peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
        else:
            prob_src = a / a.sum()

        # destination prefers small contribution
        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()

        i = rng.choice(n, p=prob_src)
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        max_delta = a[i]
        if max_delta < 1e-16:
            continue

        # move a random fraction (occasionally all)
        if rng.random() < 0.03:
            delta = max_delta               # aggressive move
        else:
            delta = rng.random() * max_delta * 0.5

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()

        # linear part (‑2 δ·a shifted by i, +2 δ·a shifted by j)
        start_i = i
        end_i = i + n
        if end_i > b_new.size:
            b_new[start_i:] -= 2.0 * delta * a[: b_new.size - start_i]
        else:
            b_new[start_i:end_i] -= 2.0 * delta * a

        start_j = j
        end_j = j + n
        if end_j > b_new.size:
            b_new[start_j:] += 2.0 * delta * a[: b_new.size - start_j]
        else:
            b_new[start_j:end_j] += 2.0 * delta * a

        # quadratic correction terms
        k2i = 2 * i
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        k2j = 2 * j
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        kij = i + j
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- acceptance -------------------------------------------------
        accept = False
        if new_max < cur_max - 1e-14:
            accept = True
        else:
            if T > 0.0 and rng.random() < math.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_max:.6f}")

        # temperature schedule
        T *= cooling

        # occasional global jitter to escape plateaus
        it += 1
        if it % 2500 == 0:
            jitter = rng.normal(scale=cur_max * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, math.sqrt(2.0 * n))
            b = _full_convolution(a)        # recompute safely

    return best_a, best_max

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation value.
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} "
              f"(length={best_seq.size})")
    except Exception as exc:   # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
```

This implementation builds on the earlier hybrid GD + hill‑climber,
adds a more aggressive simulated‑annealing hill‑climber, occasional
jitter, a second refinement GD step after hill‑climbing, and a longer
HC phase.  Together these changes typically push the maximal
self‑convolution below the previous best (≈ 1.5119), moving toward the
target ≈ 1.503 while staying safely inside the 1000 s limit.
