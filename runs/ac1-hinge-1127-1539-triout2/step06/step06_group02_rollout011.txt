    import scipy.sparse as sp
    a_cur = a.copy()
    n = a_cur.size
    best_score = _score(a_cur)
    best_a = a_cur.copy()

    while it < max_iter:
        if deadline is not None and time.time() >= deadline - 1.0:
            break
        # compute bound M
        max_b = _max_convolution_fft(a_cur)
        # build matrix A
        # A @ g <= M
        # Build using scipy.sparse.diags
        data = np.full((n, n), np.nan) # placeholder

        A = sp.diags(diags_data, offsets=np.arange(n), shape=(2*n-1, n), format='csc')
        # formulate linear program
        c = -np.ones(n)  # maximize sum(g)
        # bounds: g >= 0, no upper bound (None)

            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # scale
        g_scaled = g * (target_sum / sum_g)
        # mix
        a_cur = (1 - t) * a_cur + t * g_scaled
        # ensure non-negative and project to simplex just in case
        a_cur = np.maximum(a_cur, 0.0)
        a_cur = _project_to_simplex(a_cur, target_sum)
        # compute score
        sc = _score(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"[{time.time():.1f}] LP iter {it} improved score={best_score:.6f}")
        it += 1
    return best_a, best_score

"""
Propose a step‑function sequence that minimises the objective
    2·n·max(conv(a,a)) / (sum(a)²)

Algorithm (≈ 1000 s budget):
1. Multi‑start: several problem sizes (1500 → 3000 → 5000) with a
   random power‑law initialisation (exponent drawn uniformly from [0.45,0.65]).
2. Short annealed projected gradient descent on a soft‑max approximation
   of the max‑convolution (temperature β is increased up to 4096).
3. For moderate sizes (n ≤ 2000) a few iterations of a *linear‑program
   improvement*:
      maximise Σ g  subject to g ≥ 0  and  (f∗g) ≤ M   (M = max (f∗f)).
   The solution g is scaled to the ℓ¹‑simplex and mixed with the current
   sequence (f ← (1‑t) f + t g).  Each LP is solved with SciPy’s HiGHS
   optimiser.
4. A long simulated‑annealing phase that moves mass between “donor” and
   “receiver” coordinates, performs local smoothing and occasional swaps.
5. The best candidate found is up‑sampled (linear interpolation) to the next
   problem size and refined again.
6. A final polish (gradient → LP → SA) on the best sequence.

All intermediate iterates are projected onto the ℓ¹‑simplex
{a_i ≥ 0, Σa_i = √(2·n)} so that `evaluate_sequence` never receives an
invalid input.  Progress is printed for monitoring.

The implementation uses only NumPy, SciPy and the provided `evaluate_sequence`
function.
"""

import time
import numpy as np
import scipy.optimize  # type: ignore
import scipy.sparse as sp  # type: ignore


# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast evaluation of the objective for a normalised sum."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (gradient, raw convolution array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    Temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g
        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    start_temp: float = 0.5,
    end_temp: float = 1e-5,
    max_frac_start: float = 0.40,
    max_frac_end: float = 0.005,
    max_window: int = 30,
) -> tuple[np.ndarray, float]:
    """A classic simulated‑annealing refinement."""
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    size_fft = 1 << ((2 * n - 1).bit_length())
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # current worst entry
        k_max = int(np.argmax(conv))

        # contribution weights for the worst entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        sens = a[k_max - idx]               # a_j where i+j = k_max
        weight = a[idx] * sens               # a_i * a_{k_max-i}

        # --------------------------------------------------------------
        # propose a move
        # --------------------------------------------------------------
        move_type = None
        if rng.random() < 0.70:                      # mass transfer
            total_w = weight.sum()
            donor = rng.choice(n, p=weight / total_w) if total_w > 0 else rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            receiver = rng.choice(low_idx) if low_idx.size else rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        elif rng.random() < 0.90:                    # local smoothing
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                saved_seg = a[start:end].copy()
                avg = saved_seg.mean()
                a[start:end] = avg
                move_type = "smooth"
                saved_start, saved_end = start, end

        else:                                        # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # --------------------------------------------------------------
        # evaluate the new state (single FFT)
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA new best = {best_score:.6f}")
        else:
            # revert move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 5000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


def _lp_improve(a: np.ndarray, target_sum: float, t: float = 0.2,
                max_iter: int = 5, deadline: float = None):
    """
    Linear‑program improvement:
    maximise Σ g   s.t.   g ≥ 0, (a∗g) ≤ M,
    then mix f←(1‑t)f + t·g_scaled.
    """
    n = a.size
    # skip if the problem is too large for a dense LP
    if n > 2500:
        return a, _score(a)

    best_a = a.copy()
    best_val = _score(a)
    it = 0

    while it < max_iter:
        if deadline is not None and time.time() >= deadline - 1.0:
            break

        M = _max_convolution_fft(a)

        # Build the Toeplitz matrix A (shape (2n‑1)×n) with constant diagonals = a[d]
        diag_vals = [np.full(n, a[d]) for d in range(n)]
        A = sp.diags(diag_vals, offsets=np.arange(n), shape=(2 * n - 1, n), format='csc')

        c = -np.ones(n)                                 # maximise Σ g  ⇔ minimise –Σ g
        bounds = [(0, None) for _ in range(n)]
        b_ub = np.full(2 * n - 1, M)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs')
        if not res.success:
            # If the LP fails we abort the LP phase.
            print(f"[{time.time():.1f}] LP failed (iter {it}): {res.message}")
            break

        g = res.x
        s = g.sum()
        if s <= 0:
            break

        # scale to meet the ℓ¹‑normalisation
        g_scaled = g * (target_sum / s)

        # mixing step
        a = (1.0 - t) * a + t * g_scaled
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it} improved score={best_val:.6f}")

        it += 1

    return best_a, best_val


def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of seq to a new length new_n."""
    old_n = seq.size
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimises the step‑function sequence within the given time budget.
    Returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # problem sizes – we start small, then upscale
    sizes = [1500, 3000, 5000]

    # first stage: optimise the smallest size fairly thoroughly
    for idx, n in enumerate(sizes):
        if time.time() >= deadline - 5.0:
            break

        target = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # initialise
        # ------------------------------------------------------------------
        if idx == 0:
            # power‑law with a random exponent
            p = rng.uniform(0.45, 0.65)
            seq = _power_law_sequence(n, p, target)
        else:
            # upscale the best sequence obtained so far
            seq = _interpolate_sequence(best_seq, n)
            seq = np.maximum(seq, 0.0)

        seq = _project_to_simplex(seq, target)

        # ------------------------------------------------------------------
        # gradient descent (short but non‑trivial)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        gd_time = min(30.0, remaining * 0.30)
        gd_deadline = time.time() + gd_time
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)

        # ------------------------------------------------------------------
        # linear‑program improvement (only for moderate dimensions)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0 and n <= 2000:
            seq, lp_score = _lp_improve(seq, target, t=0.2,
                                        max_iter=5,
                                        deadline=deadline)
        else:
            lp_score = gd_score

        # ------------------------------------------------------------------
        # simulated annealing for the rest of the allocated slice
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            seq, sa_score = _simulated_annealing(seq, deadline)
        else:
            sa_score = lp_score

        cur_score = min(gd_score, lp_score, sa_score)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} → new best score={best_score:.6f}")

        # ---------------------------------------------------------------
        # if there is still a lot of time left, keep the current best and
        # move to the next (larger) problem size
        # ---------------------------------------------------------------
        # (the loop naturally proceeds to the next n)

    # ----------------------------------------------------------------------
    # Final polish on the globally best candidate (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        target = np.sqrt(2.0 * n)

        # a final gradient phase with a longer horizon
        extra_deadline = time.time() + (deadline - time.time()) * 0.40
        best_seq, _ = _annealed_gradient_descent(best_seq, extra_deadline)

        # one more LP step (optional, cheap)
        if time.time() < deadline - 2.0 and n <= 2000:
            best_seq, _ = _lp_improve(best_seq, target,
                                      t=0.1, max_iter=2,
                                      deadline=deadline)

        # short SA to escape any final plateau
        if time.time() < deadline - 1.0:
            best_seq, _ = _simulated_annealing(best_seq, deadline)

    # strip trailing zeros (they do not affect the score)
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
