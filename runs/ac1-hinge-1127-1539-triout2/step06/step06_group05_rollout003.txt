from scipy.optimize import linprog
import numpy as np
import scipy.linalg
def solve_lp(f, M):
    n=len(f)
    # Build coefficient matrix A_ub
    first_col = np.concatenate([f, np.zeros(n-1)])
    first_row = np.zeros(n)
    first_row[0] = f[0]
    A = scipy.linalg.toeplitz(first_col[:2*n-1], first_row)
    # Solve
    c = -np.ones(n)   # maximize sum(b) => minimize -sum(b)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: return original f
        return f
    b = res.x
    return b
```

"""
Optimisation of a step‑function (non‑negative vector) with the goal
to minimise the peak of its self‑convolution.
The search consists of many short restarts, each performing three
phases:

1. **LP‑mix phase** – given a current vector `a`, solve a linear
   programme that maximises the ℓ¹‑mass of a new vector `b` under the
   constraint   (a * b)[k] ≤ M  for all shifts `k`, where `M` is the
   current maximal convolution value.  The obtained `b` is rescaled to
   the required ℓ¹‑norm and mixed with `a`.  Repeating this step quickly
   drives the maximal convolution down.

2. **Sub‑gradient phase** – a simple sub‑gradient descent on the
   true max‑entry of the convolution (plus a few random mass‑swap
   moves) refines the solution.

3. **Local swap phase** – a cheap hill‑climber that swaps tiny masses
   between random indices.

Each restart works on a fresh length (≈ 1.5 k–4 k).  The best sequence
overall is returned.  The algorithm is deterministic given the
seed, respects the time budget (≈ 1000 s) and produces only
non‑negative numbers.
"""

import time
import numpy as np

# Optional SciPy – used only for the linear programme.
try:
    import scipy.optimize as _opt
    import scipy.linalg as _linalg
except Exception:  # pragma: no cover
    _opt = None
    _linalg = None


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1) via FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry."""
    return 0.5 * (v + v[::-1])


def _upsample_seq(base: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    if base.size == n:
        return base.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, base.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, base).astype(float)


# ----------------------------------------------------------------------
# Diverse initialisers
# ----------------------------------------------------------------------
def _initial_sequence(base_seq: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n with ℓ¹‑norm √(2·n).
    Several analytic families are mixed for diversity.
    """
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 6)   # 0–5 inclusive

    if typ == 0 and base_seq is not None and base_seq.size > 10:
        a = _upsample_seq(base_seq, n)
        a += rng.normal(scale=a.mean() * 1e-3, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        a = rng.random(n)                     # uniform Dirichlet(1)

    elif typ == 2:
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        right = left[::-1]
        a = (left + right) / 2.0

    elif typ == 3:
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 4:
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    else:
        cand = []
        cand.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9995)
        left = rate ** np.arange(n)
        cand.append((left + left[::-1]) / 2.0)          # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        cand.append(rec ** power)                       # reciprocal
        w = rng.random(len(cand))
        w /= w.sum()
        a = sum(wi * ci for wi, ci in zip(w, cand))

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Linear‑programming step (LP‑mix phase)
# ----------------------------------------------------------------------
def _lp_optimize(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   f * b ≤ M   (elementwise),   b ≥ 0 .
    Returns the optimal b (may be None if the solver fails).
    """
    if _opt is None or _linalg is None:
        return None

    n = f.shape[0]
    # Build the Toeplitz matrix that implements the convolution f * b.
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    A = _linalg.toeplitz(col[: 2 * n - 1], row)   # shape (2n‑1, n)

    c = -np.ones(n)                               # maximise Σb → minimise -Σb
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # HiGHS is fast and handles dense matrices well.
    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                       method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _optimize_lp_mix(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑mix iterations until the time slice expires.
    Returns the best vector seen and its maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    max_iters = 30
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _full_convolution(a)
        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Linear programme based direction.
        b = _lp_optimize(a, cur_score)
        if b is None:
            break

        # Rescale b to the required ℓ¹‑norm.
        b_sum = b.sum()
        if b_sum > 0:
            b = b * (S / b_sum)
        else:
            b = a.copy()

        # Simple averaging (mixing) step.
        t = 0.5
        a = (1.0 - t) * a + t * b

        if enforce_sym:
            a = _symmetrize(a)

    # Final evaluation of the best vector obtained.
    final_score = np.max(_full_convolution(best_a))
    return best_a, final_score


# ----------------------------------------------------------------------
# Sub‑gradient phase (identical to the previous implementation)
# ----------------------------------------------------------------------
def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.14
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # Sub‑gradient of the active entry.
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        if enforce_sym:
            a = _symmetrize(a)
        a = _project_simplex(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Random mass‑swap (simulated‑annealing style).
        if rng.random() < 0.045:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.03 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.22:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = a.copy()

        if it % 8000 == 0 and best_score == cur_score:
            break

    return best_a, best_score

# Very cheap local swap search (fine‑tuning)
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.02 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score

# Combined optimiser (LP → sub‑gradient → local)
# ----------------------------------------------------------------------
def _optimize_full(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # allocate time slices
    t_lp = now + 0.60 * remaining

    else:
        a_loc, sc_loc = a_sub, sc_sub

    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score

        a0 = _initial_sequence(base_seq, n, rng)

        # Occasionally start from an up‑sampled version of the current best.
        if best_overall_seq is not None and rng.random() < 0.30:
            a0 = _upsample_seq(best_overall_seq, n)
            a0 = np.clip(a0 + rng.normal(scale=a0.mean() * 1e-3, size=n),
                         0.0, None)
            # re‑normalise onto the simplex
            a0 = _project_simplex(a0, np.sqrt(2.0 * n))

        if enforce_sym:
            a0 = _symmetrize(a0)

        # Allocate a time slice for this attempt (≈ 15–25 s).
        slice_deadline = min(time.time() + rng.uniform(15.0, 25.0), deadline)

        # Run the combined optimiser.
        a_opt, score = _optimize_full(a0, slice_deadline, rng, enforce_sym)

        # Update the global best if improved.
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            best_overall_n = n
            last_improve_time = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_overall_score:.6f}")

        # Early exit if we have already reached the target bound.
        if best_overall_score <= 1.503:
            break

        # If we have been stagnant for a while, forget the base shape.
        if time.time() - last_improve_time > 45.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_overall_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external={external_score:.6f} internal_best={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    # expose the best sequence as a global variable for possible up‑sampling later
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
