"""
Step‑function optimizer (mixed ADAM + LP refinement).

1.  Generate many diverse feasible starts (edge‑heavy, Dirichlet,
    uniform) and run a projected ADAM optimiser on the simplex.
    The ADAM routine uses a τ‑soft‑max approximation of the convolution
    peak, works directly with a log‑softmax parametrisation (guarantees
    non‑negativity and the ℓ₁‑normalisation) and includes learning‑rate
    decay, temperature decay and occasional jitter.

2.  Take the best ADAM candidate and (if SciPy is available) improve it
    by a small number of Frank‑Wolfe‑style LP steps:
        maximise 1ᵀg  subject to   a∗g ≤ M   and   g ≥ 0,
    where a is the current sequence, M its peak self‑convolution and
    “∗’’ denotes linear convolution.  The LP solution pushes mass
    towards indices that affect the peak the least; after scaling to the
    required ℓ₁‑norm the new sequence is mixed with the old one.

3.  A final ADAM run refines the LP‑enhanced vector.

The routine respects the 1000 s budget, stores the best sequence in the
global variable ``height_sequence_1`` and prints progress/score.
"""

import time
import numpy as np
from typing import List, Tuple

# optional SciPy – used only for the LP refinement step
try:
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:      # pragma: no cover
    _SCIPY_OK = False


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using a real FFT (size 2·len‑1)."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Parameterised edge‑heavy initial shapes
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy shape: (distance_to_edge + offset)^{‑p} optionally mixed."""
    S = np.sqrt(2.0 * n)

    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0               # 1‑based distance to nearest edge

    p = rng.uniform(0.2, 1.5)       # exponent – extended upper bound
    offset = rng.uniform(0.0, 8.0)  # denominator shift – larger range
    mix = rng.uniform(0.0, 1.0)    # proportion of edge‑heavy component

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)               # uniform background
    raw = mix * edge_part + (1.0 - mix) * base

    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """
    Minimise the τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered (sum = sqrt(2·n)) and the
    corresponding hard peak (max self‑convolution value).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)

    # initialise logits for a log‑softmax parametrisation
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                     # logits

    # ADAM state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.35                            # a little larger than the reference

    # initial temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover a from logits (log‑softmax) -----
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ----- convolution & τ‑soft‑max gradient -----
        b = _full_convolution_fft(a)
        max_b = b.max()

        if tau < 1e-12:
            mask = (b == max_b).astype(float)
        else:
            diff = b - max_b
            exps = np.exp(diff / tau)
            mask = exps / np.sum(exps)

        # gradient w.r.t. a (see analysis notes)
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv_full = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g_a = 2.0 * conv_full[n - 1:n - 1 + n]

        # keep best hard peak
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        # ----- chain rule: gradient w.r.t. logits -----
        mu = np.dot(a, g_a) / S           # ⟨a,∂b/∂a⟩/S
        grad_p = a * (g_a - mu)

        # gradient clipping (helps stability for large n)
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # ----- ADAM update -----
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- schedules -----
        tau *= 0.9995               # faster tightening of the soft‑max
        lr *= 0.9999                # gentle decay of learning rate

        # occasional jitter to escape plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.012, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Linear‑programming refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the dense matrix A (shape (2n‑1, n)) such that
    A @ g == a * g (linear convolution) for any g ≥ 0.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # each column i contains a shifted copy of a
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_refine(a0: np.ndarray,
               t: float,
               max_iter: int,
               deadline: float) -> Tuple[np.ndarray, float]:
    """
    Perform a few LP refinement steps.
    Returns the best vector found and its peak self‑convolution.
    """
    if not _SCIPY_OK:
        # SciPy not available – skip LP phase
        return a0, _full_convolution_fft(a0).max()

    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_peak = _full_convolution_fft(a).max()
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        M = best_peak

        # build convolution matrix for current best a
        A = _build_conv_matrix(best_a)

        # solve: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(A.shape[0], M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # unlikely, but abort the LP phase if it happens
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # renormalise to the required ℓ₁‑norm
        g = g * (S / sum_g)

        # convex combination
        a_new = (1.0 - t) * best_a + t * g
        new_peak = _full_convolution_fft(a_new).max()

        if new_peak + 1e-12 < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            no_improve = 0
            print(f"[LP] iter {it:3d} improved peak to {best_peak:.6f}")
        else:
            no_improve += 1
            # if nothing improves for a handful of steps, stop early
            if no_improve >= 5:
                break

    return best_a, best_peak


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The strategy mixes a broad ADAM exploration,
    an optional LP refinement (when SciPy is present) and a final ADAM
    polishing phase.
    """
    rng = np.random.default_rng(seed)

    # time handling
    safe_margin = 5.0               # seconds reserved for post‑processing
    deadline = time.time() + budget_s - safe_margin

    # ------------------------------------------------------------------
    # re‑use a previously found construction if present
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:          # pragma: no cover
            base_seq = None

    best_seq = None
    best_peak = np.inf
    attempt = 0

    # ------------------------------------------------------------------
    # Helper: create a feasible start vector of length n
    # ------------------------------------------------------------------
    def _initial_vector(n: int) -> np.ndarray:
        # Occasionally warm‑start from the global best (scaled)
        if base_seq is not None and rng.random() < 0.2:
            a = _scale_sequence(base_seq, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * np.sqrt(2.0 * n)
            return a

        style = rng.choice(['uniform', 'dirichlet', 'edge'])
        S = np.sqrt(2.0 * n)

        if style == 'uniform':
            a = np.full(n, S / n)
        elif style == 'dirichlet':
            a = rng.random(n)
            a = a / a.sum() * S
        else:   # edge‑heavy
            a = _edge_initial(n, rng)
        return a

    # ------------------------------------------------------------------
    # Phase 1 – many ADAM restarts (exploration)
    # ------------------------------------------------------------------
    # we allocate roughly 2/3 of the budget to exploration
    explore_deadline = time.time() + (deadline - time.time()) * 0.66
    short_budget = 18.0                     # seconds per short ADAM run

    while time.time() + short_budget < explore_deadline:
        attempt += 1
        # choose a length – a mixture of medium and larger sizes
        if rng.random() < 0.6:
            n = rng.integers(1200, 3000)
        else:
            n = rng.integers(3000, 6000)

        a_start = _initial_vector(n)

        slice_deadline = min(time.time() + short_budget, explore_deadline)
        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng)

        if peak < best_peak:
            best_peak = peak
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart] attempt={attempt:3d} n={n:5d}  new best peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP refinement (if possible)
    # ------------------------------------------------------------------
    if best_seq is not None and _SCIPY_OK:
        remaining = deadline - time.time()
        if remaining > 30.0:
            # down‑scale to a manageable size for the LP (≈800)
            n_lp = min(800, len(best_seq) // 2)
            a_small = _scale_sequence(best_seq, n_lp)
            a_small = np.clip(a_small, 0.0, None)
            a_small = a_small / a_small.sum() * np.sqrt(2.0 * n_lp)

            lp_deadline = time.time() + remaining * 0.5
            a_refined_small, lp_peak = _lp_refine(a_small,
                                                  t=0.45,
                                                  max_iter=25,
                                                  deadline=lp_deadline)

            # upscale back to the original length
            a_up = _scale_sequence(a_refined_small, len(best_seq))
            a_up = np.clip(a_up, 0.0, None)
            a_up = a_up / a_up.sum() * np.sqrt(2.0 * len(best_seq))

            # one more short ADAM polishing after the up‑scaling
            final_deadline = deadline
            a_final, final_peak = _adam_optimize(a_up, final_deadline, rng)

            if final_peak < best_peak:
                best_peak = final_peak
                best_seq = a_final.copy()
                print(f"[lp+adam] improved peak to {best_peak:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final ADAM refinement on the current best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            a_refined, refined_peak = _adam_optimize(best_seq, deadline, rng)
            if refined_peak < best_peak:
                best_peak = refined_peak
                best_seq = a_refined.copy()
                print(f"[final adam] refined peak={best_peak:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal peak = {best_peak:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
