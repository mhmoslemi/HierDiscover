"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  The objective is the maximum entry of the (full)
linear convolution a*a when the ℓ¹‑norm of a is fixed to √(2 n); the
objective equals that maximum value.

Algorithm
---------
1. **Parametric sweep** – generate candidates from several simple families
   (power‑law, exponential, truncated linear).  For each base shape create a
   handful of orderings (sorted, reversed, alternating, random coprime
   strides, random shuffles).  Keep an elite list of the best few vectors.

2. **Random‑gradient stage** – repeatedly draw a random length and a random
   ℓ¹‑simplex point, then run a short Adam optimisation on the smoothed
   log‑sum‑exp surrogate of the max‑convolution.  The refined vectors are
   added to the elite set.

3. **Refinement** – for every elite candidate run a longer Adam optimisation
   (large β → true max) and finish with a permutation‑annealing phase that
   swaps entries while preserving the ℓ¹‑norm.

4. **Return** – the best vector found (converted to a plain Python list).

The code respects a global deadline (default 1000 s) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (FFT) – maximum entry of a*a
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx = s, x ≥ 0  (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) surrogate of max(b) and its gradient
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Compute gradient of (1/β)·log Σ_k exp(β·(a*a)_k) w.r.t. a.
    Returns (g, b) where
        b = a*a (full convolution)
        g = 2·(a ⋆ p)    with p_k ∝ exp(β·b_k)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max probabilities on the convolution entries
    max_b = b.max()
    exp_shifted = np.exp(beta * (b - max_b))
    p = exp_shifted / exp_shifted.sum()

    # convolution a ⋆ p (full) – we need only the central n entries
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3·n‑2
    size2 = 1 << (conv_len.bit_length())
    ar_fft = np.fft.rfft(a_rev, n=size2)
    pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Adam optimiser for the soft‑max surrogate of max(conv)."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = 1.0
    max_beta = 1e7

    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        g, _ = _gradient_softmax(a, beta)

        # Adam update (scalar step size derived from moments)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # element‑wise Adam step
        step = 0.05 / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # keep on the ℓ¹‑simplex
        a = _project_to_simplex(a, target_sum)

        # schedule for β (approach the true max)
        if t % 200 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        # occasional jitter to escape plateaus
        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

    return a


# ----------------------------------------------------------------------
# Permutation‑annealing (swap moves only, ℓ¹‑norm preserved)
# ----------------------------------------------------------------------
def _permutation_annealing(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Improve ordering of a fixed multiset by simulated‑annealing swaps."""
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _max_convolution_fft(best)

    T = best_score * 0.02 + 1e-9
    cooling = 0.9995
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        cur_score = _max_convolution_fft(a)
        delta = cur_score - best_score
        if cur_score <= best_score or rng.random() < np.exp(-delta / T):
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
        else:
            # revert the unfavourable swap
            a[i], a[j] = a[j], a[i]

        T *= cooling

        # tiny global reshuffle occasionally
        if rng.random() < 0.001:
            mask = rng.random(n) < 0.02
            if mask.any():
                sub = a[mask]
                rng.shuffle(sub)
                a[mask] = sub
                cur_score = _max_convolution_fft(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
    return best


# ----------------------------------------------------------------------
# Helper: generate a few sensible orderings for a multiset
# ----------------------------------------------------------------------
def _candidate_orderings(base: np.ndarray, rng: np.random.Generator) -> list[np.ndarray]:
    n = base.size
    asc = np.sort(base)          # smallest → largest
    desc = asc[::-1]             # largest → smallest
    orderings = [asc, desc]

    # alternating small–large (small first)
    alt = np.empty_like(base)
    lo, hi = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            alt[i] = asc[lo]
            lo += 1
        else:
            alt[i] = asc[hi]
            hi -= 1
    orderings.append(alt)

    # a few random coprime‑step permutations
    attempts = 0
    while attempts < 6:
        step = rng.integers(2, n)
        if np.gcd(step, n) == 1:
            perm = np.empty_like(base)
            pos = 0
            for v in asc:
                perm[pos] = v
                pos = (pos + step) % n
            orderings.append(perm)
            attempts += 1

    # a handful of pure random shuffles
    for _ in range(4):
        shuffled = asc.copy()
        rng.shuffle(shuffled)
        orderings.append(shuffled)

    return orderings


# ----------------------------------------------------------------------
# Families of shapes (always projected onto the ℓ¹‑simplex)
# ----------------------------------------------------------------------
def _normalize_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    if v.sum() == 0:
        v[0] = 1.0
    a = v * (target / v.sum())
    a = np.maximum(a, 0.0)
    return a


def _powerlaw_sequence(n: int, p: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = (idx + shift) ** (-p)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _exponential_sequence(n: int, rate: float, shift: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.exp(-rate * (idx + shift))
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


def _linear_sequence(n: int, slope: float, intercept: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    raw = np.maximum(0.0, intercept - slope * idx)
    return _normalize_to_sum(raw, np.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine respects a global time budget,
    explores several parametric families, refines elite candidates with an
    Adam optimiser and a permutation‑annealing phase, and finally returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    best_seq = None
    best_obj = np.inf
    elite: list[tuple[float, np.ndarray]] = []       # (objective, vector)

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (multiple lengths)
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000, 12000]

    # grids (moderately coarse – we will refine later)
    p_grid = np.linspace(0.30, 0.80, 9)          # power‑law exponent
    shift_grid = np.linspace(0.1, 5.0, 9)        # shift for power‑law & exp
    rate_grid = np.linspace(0.0002, 0.006, 9)    # exponential rate
    slope_grid = np.linspace(0.00005, 0.008, 9) # linear slope
    intercept_grid = np.linspace(0.3, 3.0, 9)    # linear intercept

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)

        # ----- power‑law family -----
        for p in p_grid:
            for sh in shift_grid:
                base = _powerlaw_sequence(n, p, sh)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL BEST (pow) n={n} p={p:.3f} shift={sh:.2f} obj={best_obj:.6f}")
                    if len(elite) < 12 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:12]

        # ----- exponential family -----
        for rate in rate_grid:
            for sh in shift_grid:
                base = _exponential_sequence(n, rate, sh)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL BEST (exp) n={n} rate={rate:.5f} shift={sh:.2f} obj={best_obj:.6f}")
                    if len(elite) < 12 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:12]

        # ----- linear (truncated) family -----
        for slope in slope_grid:
            for intercept in intercept_grid:
                base = _linear_sequence(n, slope, intercept)
                for cand in _candidate_orderings(base, rng):
                    score = _max_convolution_fft(cand)
                    if score < best_obj:
                        best_obj = score
                        best_seq = cand.copy()
                        print(f"[{time.time():.1f}] NEW GLOBAL BEST (lin) n={n} slope={slope:.5f} int={intercept:.2f} obj={best_obj:.6f}")
                    if len(elite) < 12 or score < elite[-1][0]:
                        elite.append((score, cand.copy()))
                        elite.sort(key=lambda x: x[0])
                        elite = elite[:12]

        # ----- short random sampling for diversity -----
        rnd_stop = min(deadline, time.time() + 1.2)
        while time.time() < rnd_stop:
            rand = rng.random(n)
            rand = _normalize_to_sum(rand, target_sum)
            score = _max_convolution_fft(rand)
            if score < best_obj:
                best_obj = score
                best_seq = rand.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST (rand) n={n} obj={best_obj:.6f}")
            if len(elite) < 12 or score < elite[-1][0]:
                elite.append((score, rand.copy()))
                elite.sort(key=lambda x: x[0])
                elite = elite[:12]

    # ------------------------------------------------------------------
    # 2. Random‑gradient stage (pure Adam from random points)
    # ------------------------------------------------------------------
    # Allocate a modest slice of remaining time for this stage
    random_stage_dead = min(deadline, time.time() + 40.0)
    while time.time() < random_stage_dead:
        n = rng.choice(lengths)
        target = np.sqrt(2.0 * n)
        a0 = rng.random(n)
        a0 = _project_to_simplex(a0, target)

        # brief Adam run (≈5 s)
        sub_dead = min(deadline, time.time() + 5.0)
        refined = _adam_opt(a0, sub_dead)
        score = _max_convolution_fft(refined)

        if score < best_obj:
            best_obj = score
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (rand‑grad) n={n} obj={best_obj:.6f}")

        if len(elite) < 12 or score < elite[-1][0]:
            elite.append((score, refined.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]

    # ------------------------------------------------------------------
    # 3. Refinement of elite candidates (Adam + permutation annealing)
    # ------------------------------------------------------------------
    for elite_score, elite_vec in elite:
        if time.time() >= deadline:
            break
        n = elite_vec.size
        # divide remaining time evenly among remaining elites (at most 15 s each)
        remaining = deadline - time.time()
        per_elite = max(5.0, remaining / (len(elite) + 1))
        sub_dead = min(deadline, time.time() + per_elite)

        # longer Adam optimisation with large β (approaching true max)
        refined = _adam_opt(elite_vec, sub_dead)

        # permutation annealing on the refined shape
        refined = _permutation_annealing(refined, deadline)

        cur_score = _max_convolution_fft(refined)
        if cur_score < best_obj:
            best_obj = cur_score
            best_seq = refined.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (refined) n={n} obj={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sanity‑check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple power‑law with n=8000 (should never trigger)
        n = 8000
        best_seq = _powerlaw_sequence(n, p=0.5, shift=1.0)

    # strip trailing near‑zero entries (purely cosmetic)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    # optional verification (won’t affect the returned value)
    try:
        eval_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] Final evaluate_sequence = {eval_val:.6f}")
    except Exception:
        pass

    print(f"[{time.time():.1f}] Finished. Best objective = {best_obj:.6f} length = {best_seq.size}")
    return best_seq.tolist()
