Proposes a step function (non‑negative vector) with low maximum self‑convolution.
Algorithm:
1. Progressive optimisation over increasing sizes (800 → 2000 → 4000).
2. Each stage runs an Adam optimiser on a smooth log‑sum‑exp approximation of
   the max‑norm, followed by a projected sub‑gradient descent on the true max.
3. A lightweight genetic algorithm refines the result.
4. A final hill‑climbing local search performs tiny random mass swaps.
5. The best sequence found (possibly up‑sampled from a previous stage) is
   returned.  The routine respects the 1000 s budget.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv

def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient term for the smooth max‑norm:
        g_j = 2 Σ_i w_{i+j} a_i .
    Both a (len n) and w (len 2·n−1) are real vectors.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields the gradient
    grad = 2.0 * full[idx]
    return grad

def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_i = exp(alpha·conv_i) / Σ exp(alpha·conv_j)."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w

def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=float)
    return np.interp(x_new, x_old, a)

def _normalize_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Rescale a to have exact L1‑sum target_sum (while keeping non‑negativity)."""
    cur = a.sum()
    if cur == 0:
        return np.full_like(a, target_sum / a.shape[0])
    return a * (target_sum / cur)

def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth log‑sum‑exp approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    alpha = 0.5
    alpha_max = 2000.0

    while time.time() < deadline:
        t += 1
        conv = _full_convolution(a)
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex (exact L1)
        a = _project_simplex(a, S)
        a = _normalize_sum(a, S)

        # random tiny mass‑swap for exploration
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.01
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, S)
                a = _normalize_sum(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # increase temperature (make soft‑max sharper)
        alpha = min(alpha * 1.07, alpha_max)

    return best_a, best_score

def _optimize_subgrad(a0: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    eta0 = 0.8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the max entry
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_i = np.arange(i_low, i_high + 1)
        idx_j = k_max - idx_i
        g[idx_i] = a[idx_j]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g
        a = _project_simplex(a, S)
        a = _normalize_sum(a, S)

        cur_score = np.max(_full_convolution(a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation
        if rng.random() < 0.06:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.04
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = _normalize_sum(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.3:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score

def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.85) -> tuple[np.ndarray, float]:
    """Run Adam for a fraction of the time, then finish with sub‑gradient."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid

def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple steady‑state GA on the simplex."""
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 30 if n <= 2000 else 20
    # initialise population
    pop = np.empty((pop_size, n), dtype=float)
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        a = _normalize_sum(a, S)
        pop[i] = a
        scores[i] = np.max(_full_convolution(a))

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # pick two distinct parents
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            # mutation
            if rng.random() < 0.4:
                noise = rng.normal(scale=child.mean() * 0.07 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)
            child = _normalize_sum(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # diversity injection
        if rng.random() < 0.15:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            a_new = _normalize_sum(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # safety break
        if time.time() + 0.2 >= deadline:
            break

    return best_a, best_score

def _local_search(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny random mass‑swap hill‑climbing using the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    best_score = np.max(_full_convolution(a))
    best_a = a.copy()
    # step size controls maximal delta proportion
    step_factor = 0.03

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * step_factor
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand = _normalize_sum(cand, S)

        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
        else:
            # occasionally still accept to escape shallow traps
            if rng.random() < 0.01:
                a = cand

        # slowly shrink step factor to fine‑tune
        step_factor *= 0.9999
        if step_factor < 1e-5:
            step_factor = 1e-5

    return best_a, best_score

                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                cand = _normalize_sum(cand, S)
                cand_score = np.max(_full_convolution(cand))
                if cand_score < cur_score or rng.random() < 0.3:
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()

    return best_a, best_score


def _optimize_combined_adam(a0: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            fraction_adam: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Run Adam for a fraction of the allocated time and finish with
    projected sub‑gradient descent.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    adam_deadline = now + fraction_adam * remaining
    a_mid, sc_mid = _optimize_adam(a0, adam_deadline, rng)

    if time.time() < deadline:
        a_fin, sc_fin = _optimize_subgrad(a_mid, deadline, rng)
        if sc_fin < sc_mid:
            return a_fin, sc_fin
    return a_mid, sc_mid


def _run_genetic_optimizer(initial_a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Steady‑state GA that respects the simplex constraint."""
    n = initial_a.shape[0]
    S = np.sqrt(2.0 * n)

    pop_size = 30 if n <= 2000 else 20
    pop = np.empty((pop_size, n), dtype=float)
    scores = np.empty(pop_size, dtype=float)

    for i in range(pop_size):
        if i == 0:
            a = initial_a.copy()
        else:
            a = rng.random(n)
        a = _project_simplex(a, S)
        a = _normalize_sum(a, S)
        pop[i] = a
        scores[i] = np.max(_full_convolution(a))

    best_idx = int(np.argmin(scores))
    best_a = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(pop_size):
            # choose two distinct parents
            idxs = list(range(pop_size))
            idxs.remove(i)
            p1, p2 = rng.choice(idxs, size=2, replace=False)

            # convex recombination
            alpha = rng.random()
            child = alpha * pop[p1] + (1.0 - alpha) * pop[p2]

            # mutation
            if rng.random() < 0.4:
                noise = rng.normal(scale=child.mean() * 0.07 + 1e-12, size=n)
                child += noise

            child = np.maximum(child, 0.0)
            child = _project_simplex(child, S)
            child = _normalize_sum(child, S)

            child_score = np.max(_full_convolution(child))

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_a = child.copy()

        # diversity injection
        if rng.random() < 0.15:
            worst_idx = int(np.argmax(scores))
            a_new = rng.random(n)
            a_new = _project_simplex(a_new, S)
            a_new = _normalize_sum(a_new, S)
            new_score = np.max(_full_convolution(a_new))
            pop[worst_idx] = a_new
            scores[worst_idx] = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        if time.time() + 0.2 >= deadline:
            break

    return best_a, best_score


def _local_search(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny random mass‑swap hill‑climbing on the true max‑norm."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_score = np.max(_full_convolution(a))
    best_a = a.copy()
    step_factor = 0.03   # maximal proportion of a[i] that may be moved

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * step_factor
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        cand = _normalize_sum(cand, S)

        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
        else:
            if rng.random() < 0.01:
                a = cand

        # slowly shrink the step size for fine tuning
        step_factor *= 0.9999
        if step_factor < 1e-5:
            step_factor = 1e-5

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs three stages
    (800 → 2000 → 4000) mixing Adam, sub‑gradient, a GA and a final
    local hill‑climber, always respecting the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the external call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Baseline: reuse a previously discovered construction if available.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            n0 = cand.shape[0]
            S0 = np.sqrt(2.0 * n0)
            cand = _project_simplex(cand, S0)
            cand = _normalize_sum(cand, S0)
            external = evaluate_sequence(cand.tolist())
            best_seq = cand
            best_score = external
            print(f"[baseline] loaded previous best score={best_score:.6f} length={n0}")
        except Exception:
            best_seq = None

    # ------------------------------------------------------------------
    # Progressive optimisation over three problem sizes.
    # ------------------------------------------------------------------
    stage_sizes = [800, 2000, 4000]
    # allocate most of the remaining budget to the largest stage
    stage_fractions = [0.07, 0.15, 0.78]   # sum ≈ 1.0

    total_budget = deadline - start_time
    for stage_idx, n in enumerate(stage_sizes):
        if time.time() >= deadline:
            break

        stage_time = total_budget * stage_fractions[stage_idx]
        stage_deadline = min(time.time() + stage_time, deadline)

        # initialise start vector for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = rng.random(n)

        jitter = rng.normal(scale=a0.mean() * 1e-4 + 1e-12, size=n)
        a0 = np.clip(a0 + jitter, 0.0, None)

        S = np.sqrt(2.0 * n)
        a0 = _project_simplex(a0, S)
        a0 = _normalize_sum(a0, S)

        # 1️⃣ Adam → sub‑gradient
        a_opt, sc_opt = _optimize_combined_adam(a0, stage_deadline, rng,
                                                fraction_adam=0.85)

        # 2️⃣ Short GA refinement (if time permits)
        remaining = stage_deadline - time.time()
        if remaining > 1.0:
            ga_deadline = time.time() + min(8.0, remaining)
            a_ga, sc_ga = _run_genetic_optimizer(a_opt, ga_deadline, rng)
            if sc_ga < sc_opt:
                a_opt, sc_opt = a_ga, sc_ga

        # 3️⃣ Local hill‑climbing on any leftover time
        remaining = stage_deadline - time.time()
        if remaining > 0.5:
            lc_deadline = time.time() + remaining
            a_lc, sc_lc = _local_search(a_opt, lc_deadline, rng)
            if sc_lc < sc_opt:
                a_opt, sc_opt = a_lc, sc_lc

        # update global best
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[stage] n={n} new global best score={best_score:.6f}")

        total_budget = deadline - time.time()

    # ------------------------------------------------------------------
    # Final sanity check & external evaluation.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external_score = evaluate_sequence(final_seq)
    print(f"[final] external score = {external_score:.6f} "
          f"(internal best = {best_score:.6f}) length = {len(final_seq)}")
    return final_seq
