def anneal(seq, rng, deadline):
    n = len(seq)
    S = np.sqrt(2.0*n)
    # compute initial max
    max_b = _compute_max_b(seq)
    best_seq = seq.copy()
    best_score = 2.0*n*max_b/(S**2)  # But S = sqrt(2n), so score = max_b
    best_score = max_b  # because scaling ensures sum=S
    # temperature schedule
    # maybe T0 = best_score * 0.1
    # final T = best_score * 1e-6
    it = 0
    # Pre-compute step scale
    avg = S/n
    while time.time() < deadline:
        it += 1
        # propose delta move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # delta magnitude: choose gaussian, relative to avg and maybe iteration dependent
        delta = rng.normal(scale=avg*0.05)  # e.g., 5% of avg
        # ensure delta does not exceed a_i, and non-negative after move
        if delta > seq[i]:
            delta = seq[i]
        # propose new seq
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # maintain sum (exact)
        # compute max_b quickly
        new_max_b = _compute_max_b(new_seq)
        new_score = new_max_b  # as sum = S holds exactly (but slight error from floating)
        # acceptance
        if new_score < best_score:
            # accept
            seq = new_seq
            max_b = new_max_b
            best_seq = new_seq.copy()
            best_score = new_score
        else:
            # compute acceptance probability
            # compute temperature as schedule
            T = max(1e-8, best_score * 0.1 * (0.9999 ** it))
            if rng.random() < np.exp(-(new_score - best_score) / T):
                seq = new_seq
                max_b = new_max_b
        # optionally adjust step size adaptively?
    return best_seq, best_score

"""
Simulated‑annealing step‑function optimizer.

The optimizer searches the space of non‑negative vectors a (any length
n) that minimise

    2·n·max(a∗a) / (∑a)²

(the same quantity used by the reference code).  Because the objective
is invariant under a global scaling we keep the ℓ₁‑norm fixed to
sqrt(2·n) – this makes the score simply the peak of the self‑
convolution.  The search consists of two phases:

1. **Exploration** – many short annealing runs with diverse
   initialisations (uniform, Dirichlet, linear‑decay, power‑law,
   exponential).  Short runs last a few seconds; the best vector
   found is kept globally.

2. **Refinement** – a longer annealing run starting from the current
   best candidate, using a slower temperature decay.

The annealer works on the probability simplex: each move transfers a
small amount δ from one index to another while preserving the total
sum.  The acceptance probability follows the classical Metropolis rule.
All convolutions are evaluated with an FFT (O(n log n)), which makes
the inner loop fast enough to perform many‑hundred‑thousand proposals
within the 1000 s time budget.

The final best sequence is stored in the global variable
``height_sequence_1`` for reuse by future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Return the full linear self‑convolution of a using a real FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation (preserves positivity)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.clip(np.ceil(pos).astype(int), 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Initial‑vector factories
# ----------------------------------------------------------------------
def _init_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Flat distribution (the trivial baseline)."""
    S = np.sqrt(2.0 * n)
    return np.full(n, S / n)


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like random simplex."""
    S = np.sqrt(2.0 * n)
    a = rng.random(n)
    a = a / a.sum() * S
    return a


def _init_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    """Linear decay from left to right (positive at both ends)."""
    S = np.sqrt(2.0 * n)
    x = np.arange(n, dtype=float)
    a = (n - x) + rng.uniform(0.0, 0.5)          # small offset to avoid zeros
    a = a / a.sum() * S
    return a


def _init_power(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law edge‑heavy shape (d+offset)^{-p}."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0      # distance to nearest edge (1‑based)
    p = rng.uniform(0.2, 1.2)
    off = rng.uniform(0.0, 5.0)
    a = (d + off) ** (-p)
    a = a / a.sum() * S
    return a


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the left edge."""
    S = np.sqrt(2.0 * n)
    t = np.arange(n, dtype=float) / (n - 1)
    lam = rng.uniform(1.0, 4.0)
    a = np.exp(-lam * t)
    a = a / a.sum() * S
    return a


def _choose_initial_vector(n: int, rng: np.random.Generator,
                           base_seq: np.ndarray | None) -> np.ndarray:
    """
    Build a feasible start vector of length n.
    Occasionally warm‑start from the global best (scaled).
    """
    if base_seq is not None and rng.random() < 0.12:
        a = _scale_sequence(base_seq, n)
        S = np.sqrt(2.0 * n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * S
        return a

    style = rng.choice(['uniform', 'dirichlet', 'linear',
                        'power', 'exp'])
    if style == 'uniform':
        return _init_uniform(n, rng)
    elif style == 'dirichlet':
        return _init_dirichlet(n, rng)
    elif style == 'linear':
        return _init_linear(n, rng)
    elif style == 'power':
        return _init_power(n, rng)
    else:
        return _init_exponential(n, rng)


# ----------------------------------------------------------------------
# Scoring utilities (fast, independent of evaluate_sequence)
# ----------------------------------------------------------------------
def _peak_of_convolution(a: np.ndarray) -> float:
    """
    Return max(a∗a).  The ℓ₁‑norm of a is assumed to be sqrt(2·len(a)),
    therefore the evaluate_sequence score equals this peak.
    """
    b = _full_convolution_fft(a)
    return float(b.max())


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style annealer that keeps the ℓ₁‑norm fixed.
    Returns (best_vector, best_score==peak).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)          # exact sum we want to keep
    a = a0.copy()
    # ensure the sum constraint (tiny numerical drift)
    a = a / a.sum() * S

    best_a = a.copy()
    best_score = _peak_of_convolution(a)

    # temperature schedule – start relatively high, decay slowly
    init_T = max(0.02 * best_score, 1e-3)
    T = init_T

    # step scale (how much mass is transferred each proposal)
    avg = S / n
    step_sigma = avg * 0.05       # 5 % of the average height

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- propose a move -------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.normal(scale=step_sigma)

        # clip delta to keep entries non‑negative
        if delta > a[i]:
            delta = a[i]
        if delta < -a[j]:
            delta = -a[j]

        if delta == 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # ---- evaluate -------------------------------------------------------
        new_score = _peak_of_convolution(a_new)

        # ---- Metropolis acceptance -------------------------------------------
        if new_score < best_score:
            # new global best
            best_a = a_new.copy()
            best_score = new_score

        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / T):
            a = a_new

        # ---- schedule --------------------------------------------------------
        # exponential cooling of temperature
        T *= 0.9998
        # slowly shrink step size as we cool
        step_sigma *= 0.9999

        # occasional jitter to escape tiny plateaus
        if it % 2000 == 0:
            jitter = rng.normal(scale=avg * 0.01, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a = a / a.sum() * S

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine uses simulated annealing with
    diverse initialisations, keeps the best solution found and stores
    it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    end_time = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            cand = globals()['height_sequence_1']
            base_seq = np.asarray(cand, dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    run_id = 0

    # ------------------------------------------------------------------
    # Phase 1 – many short exploratory annealing runs
    # ------------------------------------------------------------------
    short_run = 10.0  # seconds per short run
    while time.time() + short_run < end_time - 30.0:
        run_id += 1

        # pick a length: a mix of moderate and larger vectors
        if rng.random() < 0.6:
            n = rng.integers(1500, 8000)
        else:
            n = rng.integers(8000, 20000)

        a_start = _choose_initial_vector(n, rng, base_seq)

        # run annealer for a short, bounded budget
        slice_deadline = min(time.time() + short_run, end_time - 30.0)
        a_opt, peak = _anneal_optimize(a_start, slice_deadline, rng)

        # the objective for a vector with ℓ₁‑norm sqrt(2n) equals the peak
        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            print(f"[explore] run={run_id} n={n}  new best peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – one long refinement run from the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 8.0:
            n = len(best_seq)
            S = np.sqrt(2.0 * n)
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum() * S

            a_refined, peak = _anneal_optimize(best_seq, end_time, rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_refined.copy()
                print(f"[refine] final refined peak={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (use the supplied black‑box evaluator)
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}   internal peak = {best_score:.6f}   length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
