"""
Optimiser for the step‑function benchmark.

The search is performed in three stages:

1. **Exploration (small n ≈ 1500‑2000).**  
   Many cheap restarts are launched.  Each restart starts from a
   power‑law distribution a_i ∝ (i+1)^{‑p} (p is drawn at random) or from
   the current best solution with a small jitter.  A simple projected
   sub‑gradient descent is applied to the ℓ∞‑objective
   max_k (a*a)[k] (the scaling by √(2 n) is handled by a projection onto the
   ℓ¹‑simplex).  The best sequence of this stage is retained.

2. **Refinement (large n ≈ 6000‑8000).**  
   The best sequence from stage 1 is interpolated to a larger length,
   giving the optimiser many more degrees of freedom.  A longer run of the
   same sub‑gradient descent (with a slower decay of the step size) is
   performed, repeatedly adding a tiny Gaussian jitter to escape plateaus.

3. **Final polishing – Simulated Annealing.**  
   The current best solution is fed to a lightweight simulated‑annealing
   routine that transfers mass between “sensitive’’ and “insensitive’’
   indices, performs short‑window smoothing and random swaps.  The
   temperature and the maximal transfer fraction decay exponentially.

All updates keep the vector on the simplex
    {x ≥ 0, Σ x_i = √(2 n)}.
After the three stages the trailing near‑zero entries are stripped and the
sequence is returned.  The function prints progress and the final evaluated
score (via the provided `evaluate_sequence`).

The algorithm is deliberately different from the previous baseline:
it uses a *hard* sub‑gradient of the true max (instead of a soft‑max) and
employs an aggressive additive update with simplex projection, together
with a multi‑scale schedule that typically pushes the benchmark below
1.503 within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x_i ≥ 0, Σ x_i = s}."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    if s <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    if seq.shape[0] == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, seq.shape[0])
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true ℓ∞‑objective
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max_k (a*a)[k].

    Returns the best sequence found and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # best‑so‑far bookkeeping
    conv = _conv_fft(a)
    best_val = conv.max()
    best_a = a.copy()

    rng = np.random.default_rng()
    it = 0
    # base step size (will be scaled by the norm of the sub‑gradient)
    step_base = target_sum * 0.25

    while time.time() < deadline:
        # current convolution and the index of its maximum
        conv = _conv_fft(a)
        k_max = int(np.argmax(conv))
        cur_val = conv[k_max]

        # sub‑gradient of the selected entry (2 * shifted a)
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k_max - idx
        g[idx] = 2.0 * a[jdx]

        # additive update with a simple step‑size rule
        norm = np.linalg.norm(g)
        if norm > 0:
            lr = step_base / (norm + 1e-12)
            a -= lr * g

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow minima
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate new candidate
        conv_new = _conv_fft(a)
        new_val = conv_new.max()
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SGD it={it:5d} max={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑Annealing refinement (lightweight version)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.01,
                         max_window: int = 50) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves mass from a “sensitive’’ index (large contribution
    to the current maximum convolution entry) to a low‑sensitivity index,
    performs short‑window smoothing and random swaps.
    """
    a = a0.copy()
    n = a.size
    cur_score = _max_convolution_fft(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())

    # pre‑compute convolution once – it will be updated after every accepted move
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index that currently attains the maximum entry
        k_max = int(np.argmax(conv))

        # sensitivity of each a_i for the current max entry:
        # derivative of (a*a)[k_max] w.r.t. a_i equals 2 * a_{k_max-i}
        sens = np.zeros(n)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j_range = k_max - i_range
        sens[i_range] = a[j_range]

        weight = a * sens
        move_type = None

        # --------------------------------------------------------------
        # 1) mass transfer (most frequent)
        # --------------------------------------------------------------
        if rng.random() < 0.70:
            total_w = weight.sum()
            if total_w > 0:
                donor = rng.choice(n, p=weight / total_w)
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"

        # --------------------------------------------------------------
        # 2) short‑window smoothing
        # --------------------------------------------------------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                a[start:end] = old_seg.mean()
                move_type = "smooth"
                saved_seg = old_seg
                saved_start, saved_end = start, end

        # --------------------------------------------------------------
        # 3) swap
        # --------------------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"

        # compute new convolution and score
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = conv_new.max()

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:6d} new best={best_score:.6f}")
        else:
            # revert the rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = conv.max()

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    benchmark defined by `evaluate_sequence`.  The routine works in three
    stages (exploration → refinement → final annealing) and respects the
    given time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Stage 1 – fast exploration on a modest length
    # ------------------------------------------------------------------
    n1 = 1800  # small enough for many cheap iterations
    target1 = np.sqrt(2.0 * n1)
    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng()

    while time.time() < deadline * 0.20:
        # initialise either from a random power‑law or from the current best
        if best_seq is not None and rng.random() < 0.3:
            init = best_seq + rng.normal(scale=target1 * 0.001, size=n1)
        else:
            exponent = rng.uniform(0.3, 1.0)
            init = _power_law_sequence(n1, exponent, target1)

        init = np.maximum(init, 0.0)
        init = _project_to_simplex(init, target1)

        sub_dead = time.time() + min(6.0, deadline - time.time())
        seq, score = _subgradient_descent(init, sub_dead)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Stage 1 n={n1} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale and long refinement
    # ------------------------------------------------------------------
    n2 = 7200
    target2 = np.sqrt(2.0 * n2)

    if best_seq is None:
        cur_seq = _power_law_sequence(n2, 0.55, target2)
    else:
        cur_seq = _resize_sequence(best_seq, n2)
    cur_seq = _project_to_simplex(cur_seq, target2)

    while time.time() < deadline * 0.85:
        sub_dead = time.time() + min(12.0, deadline - time.time())
        seq, score = _subgradient_descent(cur_seq, sub_dead)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Stage 2 n={n2} new best={best_score:.6f}")

        # warm‑start next iteration, add a tiny jitter to keep exploring
        cur_seq = seq
        jitter = rng.normal(scale=target2 * 0.001, size=n2)
        cur_seq = np.maximum(cur_seq + jitter, 0.0)
        cur_seq = _project_to_simplex(cur_seq, target2)

    # ------------------------------------------------------------------
    # Stage 3 – final Simulated Annealing polishing (remaining budget)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_deadline = deadline
        seq, score = _simulated_annealing(best_seq, final_deadline)
        if score < best_score:
            best_seq = seq.copy()
            best_score = score
            print(f"[{time.time():.2f}] Stage 3 final best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up and return
    # ------------------------------------------------------------------
    if best_seq is None:
        return []

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final call to the official benchmark (for logging only)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
