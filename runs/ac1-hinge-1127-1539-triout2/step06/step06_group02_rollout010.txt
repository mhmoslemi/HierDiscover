The algorithm mixes three complementary optimisation ideas:

1.  Multi‑scale projected gradient descent on a soft‑max

# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective value for a normalised a (sum = sqrt(2n))."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Duchi et al. (2008) Euclidean projection onto the l1‑simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple power‑law initialisation
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Generate (i+1)^(-p) of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (used in GD stage)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the β‑soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient w.r.t. a (see analysis in the description)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


# ----------------------------------------------------------------------
# Projected (annealed) gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """Projected gradient descent with increasing temperature β."""
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a -= lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP refinement (Frank‑Wolfe style)
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray, deadline: float, max_iters: int = 12):
    """
    Repeatedly solve
        maximise Σ b_j  subject to  a * b ≤ M,  b ≥ 0
    where M = max(a*a).  Scale the LP solution to the target ℓ¹‑norm,
    blend it with the current a (step size τ), project back onto the
    simplex and keep the best improvement.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()

        # Build the convolution matrix A (a * b) = A @ b
        # A has shape (2n-1, n) and constant a_i on the i‑th super‑diagonal.
        offsets = np.arange(n)                               # 0 … n‑1
        diags = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
        A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

        # Linear programme: minimise -Σ b   (i.e. maximise Σ b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        if not res.success or np.isnan(res.x).any():
            # LP failed – abort the refinement
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Scale to the required ℓ¹‑norm
        b_scaled = b * (target_sum / sum_b)

        # Try a few blending factors τ
        improved = False
        for tau in (0.5, 0.75, 1.0):
            cand = (1.0 - tau) * a + tau * b_scaled
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)
            val = _score(cand)
            if val < best_val - 1e-9:
                best_val = val
                best_a = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time():.1f}] LP refined τ={tau:.2f} size={n} score={val:.6f}")
                break

        if not improved:
            # No τ gave an improvement – stop iterating
            break

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# High‑level optimiser for a single problem size
# ----------------------------------------------------------------------


def _run_one_instance(n: int, init_mode: str, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a problem of length n.
    init_mode ∈ {'global','power','random'} determines the starting point.
    Returns the best sequence (numpy array) and its score.
    """
    target_sum = np.sqrt(2.0 * n)

    # ----- initialise -------------------------------------------------
    seq = None
    if init_mode == "global" and "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if seq.size > n:
            seq = seq[:n]
        else:
            seq = np.pad(seq, (0, n - seq.size), "constant")
    if seq is None:
        if init_mode == "power":
            seq = _power_law_sequence(n, 0.55, target_sum)
        else:  # random Dirichlet‑like
            rng = np.random.default_rng()
            seq = rng.random(n)
            seq = seq / seq.sum() * target_sum

    seq = _project_to_simplex(seq, target_sum)

    # ----- gradient descent stage ------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        gd_deadline = time.time() + min(30.0, remaining * 0.30)
        seq, gd_score = _annealed_gradient_descent(seq, gd_deadline)
    else:
        gd_score = _score(seq)

    # ----- LP refinement stage ----------------------------------------
    cur_time = time.time()
    if cur_time < deadline - 5.0:
        seq, lp_score = _lp_refine(seq, deadline)
        final_score = min(gd_score, lp_score)
    else:
        final_score = gd_score

    return seq, final_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function coefficient sequence that minimises the
    provided evaluation function.  The routine distributes the overall
    time budget across several problem sizes, runs a fast gradient
    descent, follows up with a linear‑programming refinement and finally
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes; larger n gives more freedom but also higher cost.
    sizes = [800, 1500, 2500, 3500]

    for n in sizes:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        # allocate at most 180 s to a size, keep a safety margin for later stages
        time_for_this = min(180.0, remaining * 0.45)
        run_deadline = now + time_for_this

        init_mode = rng.choice(["global", "power", "random"])
        seq, score = _run_one_instance(n, init_mode, run_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] size={n} init={init_mode} → new best score={best_score:.6f}")

    # ----- final short gradient polishing if time permits ----------
    if best_seq is not None and time.time() < deadline - 2.0:
        extra_deadline = time.time() + (deadline - time.time()) * 0.25
        best_seq, _ = _annealed_gradient_descent(best_seq, extra_deadline)

    # strip trailing almost‑zero entries (they have no effect on the score)
    if best_seq is not None:
        nz = np.where(best_seq > 1e-12)[0]
        if nz.size:
            best_seq = best_seq[: nz[-1] + 1]
        else:
            best_seq = np.array([], dtype=float)

    print(f"[{time.time():.1f}] final best score≈{best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
