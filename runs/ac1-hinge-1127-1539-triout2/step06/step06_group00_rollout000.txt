"""
Step‑function optimizer (improved)

The algorithm searches for a non‑negative vector ``a`` of length *n*
with ℓ₁‑norm `sqrt(2·n)` that minimises the peak of the self‑convolution.
It combines several ideas:

1. **Rich pool of initialisers** – uniform, Dirichlet, edge‑heavy,
   exponential decay, and an arcsine (U‑shaped) distribution.
2. **Adam on the simplex** – parametrise ``a`` with log‑softmax logits,
   minimise a τ‑soft‑max approximation of ``max(a*a)`` and gradually
   tighten τ.
3. **Multi‑scale exploration** – start with moderate lengths (≈2‑8k)
   for fast exploration, then upscale the current best candidate to
   larger lengths and refine it with a longer run.
4. **Random jitter** – occasional noise added to logits to escape
   plateaus.

The routine respects the 1000 s time budget, keeps the best sequence
found in the global variable ``height_sequence_1`` and returns a
Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector using FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


# ----------------------------------------------------------------------
# Gradient of the τ‑soft‑max approximation of max(b) where b = a*a
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, tau: float):
    """
    Return (grad, max_b) where
    grad = ∂ (τ‑soft‑max(b)) / ∂ a   and   b = a*a.
    """
    b = _full_convolution_fft(a)
    max_b = b.max()

    if tau < 1e-12:
        mask = (b == max_b).astype(float)
    else:
        diff = b - max_b
        # to avoid overflow, shift diff by its max (which is 0)
        exps = np.exp(diff / tau)
        w = exps / np.sum(exps)
        mask = w

    # gradient of aᵀ Q mask a  = 2 * (mask * a) * reversed a
    a_rev = a[::-1]
    conv_len = len(mask) + a.shape[0] - 1
    f_len = _next_pow2(conv_len)
    Fmask = np.fft.rfft(mask, n=f_len)
    Fa_rev = np.fft.rfft(a_rev, n=f_len)
    conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
    # extract the part that corresponds to derivative wrt each a_i
    g = 2.0 * conv[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    return g, max_b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a τ‑soft‑max approximation of the convolution peak.
    Returns the best vector encountered and the corresponding hard peak.
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                # exact ℓ₁‑norm required

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                    # logits

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.65                           # a bit larger than the reference

    # temperature for soft‑max approximation of the max
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 2.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- recover a from logits (log‑softmax) -------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)                # subtract max for stability
        soft = exp_p / exp_p.sum()
        a = soft * S

        # ------- convolution & soft‑max gradient -------
        g, peak = _softmax_gradient(a, tau)

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()

        # ------- chain rule: gradient w.r.t logits -------
        mu = np.dot(a, g) / S                    # ⟨a,∂b/∂a⟩ / S
        grad_p = a * (g - mu)                    # shape (n,)

        # optional clipping for stability
        clip_thr = 5.0 * S
        max_abs = np.max(np.abs(grad_p))
        if max_abs > clip_thr:
            grad_p = grad_p * (clip_thr / max_abs)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- schedules -------
        tau *= 0.9985               # tighten temperature
        lr *= 0.9998                # very slow decay

        # occasional jitter to jump out of plateaus
        if it % 600 == 0:
            p += rng.normal(scale=0.012, size=n)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------
def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑heavy power‑law shape (distance to nearest edge)."""
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0               # distance (1‑based)
    p = rng.uniform(0.2, 1.5)                           # exponent
    offset = rng.uniform(0.0, 5.0)
    mix = rng.uniform(0.0, 1.0)
    edge_part = (d + offset) ** (-p)
    base = np.ones(n)
    raw = mix * edge_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _exp_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential decay from the edges toward the centre."""
    S = np.sqrt(2.0 * n)
    decay = rng.uniform(0.001, 0.02)           # smaller = slower decay
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)
    raw = np.exp(-decay * d)
    raw = raw / raw.sum() * S
    return raw


def _arcsine_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Discrete approximation of the arcsine (Beta(½,½)) distribution.
    Very U‑shaped, putting most mass near the boundaries.
    """
    S = np.sqrt(2.0 * n)
    # use mid‑point rule to avoid zero at the exact edges
    x = (np.arange(n) + 0.5) / n
    # arcsine density: 1/(π*sqrt(x*(1-x)))
    raw = 1.0 / np.sqrt(x * (1.0 - x))
    # optional mixture with uniform to increase diversity
    mix = rng.uniform(0.0, 0.3)
    raw = (1.0 - mix) * raw + mix * np.ones_like(raw)
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


def _dirichlet_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Simple Dirichlet (flat random) initialisation."""
    S = np.sqrt(2.0 * n)
    a = rng.random(n)
    a = a / a.sum() * S
    return a


def _uniform_initial(n: int) -> np.ndarray:
    """Exactly uniform – serves as a baseline."""
    S = np.sqrt(2.0 * n)
    return np.full(n, S / n)
