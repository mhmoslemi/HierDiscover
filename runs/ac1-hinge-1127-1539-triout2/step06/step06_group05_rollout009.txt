def _optimize_subgrad_weighted(a0, deadline, rng, enforce_sym=False):

def _linprog_step(f, S, rng):
    n = len(f)
    # compute M = max of conv f*f
    conv_f = _full_convolution(f)
    M = conv_f.max()
    # Build sparse constraint matrix
    from scipy.sparse import csr_matrix
    row_inds = []
    col_inds = []
    data = []
    for k in range(2*n-1):
        # range i where f[i] is multiplied by g_j where j = k - i
        i_low = max(0, k - (n-1))
        i_high = min(n-1, k)
        # iterate i
        for i in range(i_low, i_high + 1):
            j = k - i
            row_inds.append(k)
            col_inds.append(j)
            data.append(f[i])
    A_ub = csr_matrix((data, (row_inds, col_inds)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n)  # maximize sum g
    bounds = [(0, None) for _ in range(n)]
    # Solve
    # Use method='highs' (default)
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5.0})
    # If fail, return original
    if not res.success:
        return f
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        return f
    g = g0 * (S / sum_g0)
    # Merge
    t = rng.uniform(0.5, 0.9)  # take convex combination near g
    new_f = (1 - t) * f + t * g
    new_f = _project_simplex(new_f, S)  # ensure feasible
    return new_f

Improved search for a non‑negative step function whose self‑convolution
peak (and thus the evaluation function) is as small as possible.

The algorithm combines several ideas:

1. **Rich random initialisation** – several analytic families (geometric,
   power‑law, reciprocal, triangular, mixtures) are sampled and projected
   onto the simplex Σ a_i = √(2·n).

2. **Projected Adam on a smooth log‑sum‑exp approximation** – the hard
   max‑norm is replaced by a soft‑max with a temperature that grows
   gradually.  The update is projected onto the simplex at every step.

3. **Weighted sub‑gradient descent** – instead of a single active shift
   we use *all* shifts whose value is within a tiny tolerance of the
   current maximum.  The corresponding convex combination of gradients
   gives a better descent direction.

4. **Very cheap local random swaps** – hill‑climbing that keeps the
   solution on the simplex.

5. **Optional linear‑programming refinement** (used only for moderate
   lengths, n ≤ 1800) – for a fixed vector f we solve
   `max Σ g_j  s.t. f * g ≤ M` (M = max (f*f)).  The solution g is
   rescaled to the simplex and mixed with f.  This step implements the
   Frank–Wolfe style iteration described in the reference.

The whole routine runs for the allotted wall‑clock budget (default 1000 s)
and continually keeps the best sequence seen so far.  The final sequence
is stored in the global variable ``height_sequence_1`` for possible

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s."""
    if s <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2*n-1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of (1/α)·log‑sum‑exp(α·(a*a)) w.r.t. a.
    ``w`` must be a probability vector (soft‑max weights of length 2*n-1).
    """
    n = a.shape[0]
    # reversed weights for correlation
    w_rev = w[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # extract the part that corresponds to ∂/∂a_i
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Compute soft‑max probabilities over convolution entries."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()
    return w


# ----------------------------------------------------------------------
# Symmetry utilities (optional)
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Initial sequence generators – several analytic families
# ----------------------------------------------------------------------
def _initial_sequence(_base: np.ndarray | None,
                     n: int,
                     rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector of length n with ℓ¹‑norm √(2·n)."""
    S = np.sqrt(2.0 * n)

    typ = rng.integers(0, 8)   # eight families

    if typ == 0 and _base is not None and _base.size > 12:
        # Linear up‑sampling of a previously good vector
        x_old = np.linspace(0.0, 1.0, _base.size)
        x_new = np.linspace(0.0, 1.0, n)
        a = np.interp(x_new, x_old, _base)

    elif typ == 1:
        # Uniform Dirichlet(1) – basically random positive entries
        a = rng.random(n)

    elif typ == 2:
        # One‑sided geometric decay a_i = r**i
        r = rng.uniform(0.9, 0.9995)
        a = r ** np.arange(n)

    elif typ == 3:
        # Two‑sided symmetric geometric (both ends heavy)
        r = rng.uniform(0.94, 0.998)
        left = r ** np.arange(n)
        right = left[::-1]
        a = (left + right) * 0.5

    elif typ == 4:
        # Power‑law decreasing a_i = (i+1)^{-p}
        p = rng.uniform(0.2, 1.5)
        a = (np.arange(1, n + 1, dtype=float) ** -p)

    elif typ == 5:
        # Reciprocal heavy edges (like 1/(i+1)+1/(n-i))
        idx = np.arange(n, dtype=float)
        a = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        a = a ** power

    elif typ == 6:
        # Triangular – peak in the centre
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    else:
        # Random convex combination of a few bases
        bases = []

        bases.append(rng.random(n))                     # uniform
        r = rng.uniform(0.9, 0.9995)
        bases.append(r ** np.arange(n))                 # geometric
        p = rng.uniform(0.3, 1.2)
        bases.append((np.arange(1, n + 1) ** -p))       # power‑law
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        bases.append(rec ** rng.uniform(0.2, 0.8))       # reciprocal

        w = rng.random(len(bases))
        w /= w.sum()
        a = sum(wi * bi for wi, bi in zip(w, bases))

    # Normalise to the required ℓ¹‑norm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0)
        a_sum = a.sum()
    a = a / a_sum * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp approximation
# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the soft‑max approximation of the convolution peak.
    Returns (best_vector, best_peak_value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    # Adam hyper‑parameters
    lr = 0.35
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max
    alpha = 0.7
    alpha_mul = 1.018

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max weights and gradient w.r.t. smooth objective
        w = _smooth_weights(conv, alpha)
        grad = _conv_with_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional jitter to escape flat regions
        if rng.random() < 0.014:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.045 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.13:
                    a = cand

        # schedule updates
        alpha = min(alpha * alpha_mul, 50.0)
        lr *= 0.99996

    return best_a, best_score


# ----------------------------------------------------------------------
# Weighted sub‑gradient descent (uses all near‑max shifts)
# ----------------------------------------------------------------------
def _optimize_subgrad_weighted(a0: np.ndarray,
                               deadline: float,
                               rng: np.random.Generator,
                               enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using *all* shifts that are within a tiny
    tolerance of the current maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    # tolerance for the active set (fraction of the current max)
    eps_frac = 1e-6

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # active set of indices whose value is close to the max
        thresh = cur_max - eps_frac * cur_max
        mask = conv >= thresh
        if mask.sum() == 0:
            # fallback to the single argmax case
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / mask.sum()

        # gradient of the weighted (smooth) max
        grad = _conv_with_weights(a, w)

        # step size schedule (decreasing like 1/√t)
        eta = 0.28 / np.sqrt(it)
        a = a - eta * grad

        if enforce_sym:
            a = _symmetrize(a)

        a = _project_simplex(a, S)

        # occasional random mass‑swap (small simulated‑annealing move)
        if rng.random() < 0.036:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                delta = 0.032 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand, S)
                if enforce_sym:
                    cand = _symmetrize(cand)
                cand_max = np.max(_full_convolution(cand))
                if cand_max < cur_max or rng.random() < 0.21:
                    a = cand

        # early termination if stagnating
        if it % 8000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap local random‑swap hill climbing
# ----------------------------------------------------------------------
def _local_swap_search(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """Swap tiny amounts of mass between random positions while keeping feasibility."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(_full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.018 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand, S)
        if enforce_sym:
            cand = _symmetrize(cand)
        cand_score = np.max(_full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (only for moderate lengths)
# ----------------------------------------------------------------------
def _linprog_refine(f: np.ndarray,
                    rng: np.random.Generator,
                    max_iter: int = 3) -> np.ndarray:
    """
    Frank‑Wolfe / LP refinement:
    maximise Σ g_j  subject to   f * g ≤ M   and g ≥ 0.
    The solution is rescaled to the simplex and mixed with f.
    This routine is deliberately kept lightweight and is only called
    for n ≤ 1800 (otherwise the LP becomes too large).
    """
    if not _SCIPY_AVAILABLE:
        return f

    n = f.shape[0]
    S = np.sqrt(2.0 * n)
    conv_f = _full_convolution(f)
    M = conv_f.max()

    # build the (sparse) constraint matrix A_ub such that A_ub @ g ≤ M
    row_inds = []
    col_inds = []
    data = []
    # Every shift k corresponds to a row; coefficients are f[i] with j = k-i
    for k in range(2 * n - 1):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        for i in range(i_low, i_high + 1):
            j = k - i
            row_inds.append(k)
            col_inds.append(j)
            data.append(f[i])
    A_ub = csr_matrix((data, (row_inds, col_inds)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # objective: maximise Σ g_j  → minimise -Σ g_j
    c = -np.ones(n, dtype=float)
    bounds = [(0, None) for _ in range(n)]

    for _ in range(max_iter):
        if np.max(_full_convolution(f)) - M > 1e-12:
            # update bound if M changed earlier
            M = np.max(_full_convolution(f))
            b_ub[:] = M

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 4.0})
        if not res.success:
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break

        # rescale to meet the exact ℓ¹‑norm
        g = g0 * (S / sum_g0)

        # convex combination (take a relatively large step towards g)
        t = rng.uniform(0.55, 0.87)
        f = (1.0 - t) * f + t * g
        f = _project_simplex(f, S)

        # recompute M for the next iteration
        M = np.max(_full_convolution(f))

    return f


# ----------------------------------------------------------------------
# Combined optimisation pipeline (Adam → weighted sub‑grad → local search)
# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       enforce_sym: bool = False) -> tuple[np.ndarray, float]:
    """
    Runs the three optimisation phases within the supplied time budget.
    Returns the best vector (and its peak value) found.
    """
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(_full_convolution(a0))

    # Allocate ~55 % Adam, ~35 % weighted sub‑grad, rest local search
    t_adam = now + 0.55 * remaining
    a_adam, sc_adam = _optimize_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.35 * remaining, deadline)
    a_sub, sc_sub = _optimize_subgrad_weighted(a_adam, t_sub, rng, enforce_sym)

    # final cheap local refinement (if time permits)
    if time.time() < deadline:
        a_loc, sc_loc = _local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    # Choose the best among the three phases
    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises the
    self‑convolution peak.  The routine is time‑aware and returns the
    best sequence found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Leave a small safety margin for the external evaluator.
    deadline = start_time + budget_s - 5.0

    # Keep a good previous construction around (if any).
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(height_sequence_1, dtype=float)
        except Exception:
            base_seq = None

    best_overall_seq = None
    best_overall_score = np.inf
    attempts = 0
    last_improve = start_time
    best_n = None

    # ------------------------------------------------------------------
    # Main timed loop – repeatedly restart with fresh length and
    # initialisation, keeping the global best.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Choose problem size – bias a bit towards the current best length
        if best_overall_seq is not None and rng.random() < 0.52:
            scale = rng.uniform(0.88, 1.13)
            n = max(800, int(best_n * scale))
        else:
            n = int(rng.integers(1200, 15000))

        enforce_sym = rng.random() < 0.46

        a0 = _initial_sequence(base_seq, n, rng)
        if enforce_sym:
            a0 = _symmetrize(a0)

        # Give each attempt a slice of the remaining budget (≈ 14‑23 s)
        slice_deadline = min(time.time() + rng.uniform(14.0, 23.0), deadline)

        a_opt, score = _optimize_combined(a0, slice_deadline, rng, enforce_sym)

        # Optional LP refinement for moderate lengths (helps shave a few percent)
        if n <= 1800 and _SCIPY_AVAILABLE and time.time() < deadline:
            a_opt = _linprog_refine(a_opt, rng, max_iter=2)
            score = np.max(_full_convolution(a_opt))

        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = a_opt.copy()
            best_n = n
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_overall_score:.6f}")

        # Early exit once we have beaten the target
        if best_overall_score <= 1.503:
            break

        # Occasionally forget the base shape to regain diversity
        if time.time() - last_improve > 48.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        final_seq = [1.0]
    else:
        # Final projection just to be safe (round‑off errors)
        S_final = np.sqrt(2.0 * len(best_overall_seq))
        final_seq = _project_simplex(best_overall_seq, S_final).tolist()

    # Show final external score for debugging
    try:
        external_score = evaluate_sequence(final_seq)
    except Exception:  # pragma: no cover
        external_score = float('nan')
    print(f"[final] external={external_score:.6f} internal_best={best_overall_score:.6f} "
          f"length={len(final_seq)} attempts={attempts}")

    # expose the best sequence for possible up‑sampling later
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
