Implementation details:

- We'll implement a function `project_simplex(v, S)` that returns projection onto simplex sum=S, nonnegative.

Standard algorithm (Duchi et al. 2008):

```
def project_simplex(v, z=1):
    # v is np.ndarray
    # projection onto {x>=0, sum x = z}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v, S):
    # Ensure nonnegative
    # We want to solve min ||x - v||_2^2 s.t. x>=0, sum(x) = S
    # Standard algorithm:
    n = v.shape[0]

    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]

        theta = 0.0
    else:

    return w

"""
Adversarial step‑function optimizer.

The optimizer searches for a non‑negative vector `a` of length `n`
with ℓ₁‑norm `S = sqrt(2·n)` that minimises the peak of the
self‑convolution `a*a`.  The search proceeds in three stages:

1. **Diverse initialisation** – random lengths and a collection of
   simple parametric shapes (uniform, Dirichlet, edge‑heavy and a
   mixture).  If a previous best sequence (`height_sequence_1`) exists,
   a scaled version of it is occasionally used as a warm‑start.

2. **Projected Adam** – a fast Adam optimiser on the probability simplex
   (log‑softmax parametrisation).  The loss is a τ‑soft‑max
   approximation of the convolution peak; τ is slowly cooled so that
   the optimiser first sees a smooth landscape and later focuses on
   the true maximum.

3. **Exact projected gradient descent** – after the Adam phase we run a
   short optimisation that uses the exact sub‑gradient of the max‑convolution
   (the gradient is simply a shifted copy of the current vector).  After
   each gradient step the iterate is projected back onto the ℓ₁‑simplex,
   preserving the required sum and non‑negativity.

All steps respect the constraints, therefore the external evaluator
returns exactly the convolution peak.  The routine keeps the best
candidate found and stores it in the global variable ``height_sequence_1``.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using a real FFT."""
    n = a.shape[0]
    total_len = 2 * n - 1
    f_len = _next_pow2(total_len)
    A = np.fft.rfft(a, n=f_len)
    conv = np.fft.irfft(A * A, n=f_len)[:total_len]
    return conv


def _scale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    pos = np.linspace(0, old_n - 1, new_n)
    lo = np.floor(pos).astype(int)
    hi = np.ceil(pos).astype(int)
    hi = np.clip(hi, 0, old_n - 1)
    w = pos - lo
    return (1.0 - w) * seq[lo] + w * seq[hi]


def _edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a feasible start vector of length n.
    Shape: (distance‑to‑edge + offset)^(−p) optionally mixed with a uniform background.
    The result is normalised to ℓ₁‑norm sqrt(2·n).
    """
    S = np.sqrt(2.0 * n)
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx) + 1.0   # distance to nearest edge (1‑based)

    # random parameters
    p = rng.uniform(0.2, 1.0)          # power‑law exponent
    offset = rng.uniform(0.0, 5.0)     # denominator shift
    mix = rng.uniform(0.0, 1.0)        # proportion of edge‑heavy component

    edge_part = (d + offset) ** (-p)
    base = np.ones(n)                  # uniform background
    raw = mix * edge_part + (1.0 - mix) * base
    raw = np.clip(raw, 0.0, None)
    raw = raw / raw.sum() * S
    return raw


# ---------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x) = S}
# ---------------------------------------------------------------
def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum S."""
    if S < 0:
        raise ValueError("S must be non‑negative")
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1] - 1          # zero‑based index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ---------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Minimise a τ‑soft‑max approximation of the convolution peak."""
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)                     # required ℓ₁‑norm

    # initialise logits so that softmax(p) * S == a0
    eps = 1e-12
    soft = np.maximum(a0 / S, eps)
    soft = soft / soft.sum()
    p = np.log(soft)                         # logits (no explicit projection needed later)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.25                                 # a bit more conservative than the reference

    # initialise smoothing temperature
    b0 = _full_convolution_fft(a0)
    tau = max(b0.max(), 1.0)

    best_a = a0.copy()
    best_peak = b0.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # recover a from logits (log‑softmax)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        soft = exp_p / exp_p.sum()
        a = soft * S

        # convolution & gradient (soft‑max)
        b = _full_convolution_fft(a)
        max_b = b.max()

        # sub‑gradient via τ‑soft‑max weights
        if tau < 1e-12:
            mask = (b == max_b).astype(float)
        else:
            diff = b - max_b
            exps = np.exp(diff / tau)
            mask = exps / np.expm1(0) if False else exps / np.sum(exps)

        # gradient w.r.t a: 2 * (mask * a) convolved with reversed a
        a_rev = a[::-1]
        conv_len = len(mask) + n - 1
        f_len = _next_pow2(conv_len)
        Fmask = np.fft.rfft(mask, n=f_len)
        Fa_rev = np.fft.rfft(a_rev, n=f_len)
        conv = np.fft.irfft(Fmask * Fa_rev, n=f_len)[:conv_len]
        g = 2.0 * conv[n - 1:n - 1 + n]          # gradient w.r.t a

        # keep best hard peak (not the softened one)
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        # gradient w.r.t. logits (chain rule)
        mu = np.dot(a, g) / S
        grad_p = a * (g - mu)                    # shape (n,)

        # optional gradient clipping
        clip_val = np.max(np.abs(grad_p))
        if clip_val > 5.0 * S:
            grad_p = grad_p * (5.0 * S / clip_val)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # schedules (slower cooling than the reference)
        tau *= 0.9998
        lr *= 0.99995

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            p += rng.normal(scale=0.02, size=n)

    return best_a, best_peak


# ---------------------------------------------------------------
# Exact projected gradient descent using the true max‑convolution sub‑gradient
# ---------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact peak of a*a.
    The gradient of max_k (a*a)_k is a shifted copy of a
    (averaged over all active shifts).
    """
    n = a0.shape[0]
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    best_a = a.copy()
    best_peak = _full_convolution_fft(a).max()
    it = 0

    while time.time() < deadline:
        it += 1

        # compute convolution and locate active shifts
        b = _full_convolution_fft(a)
        max_b = b.max()
        if max_b < best_peak:
            best_peak = max_b
            best_a = a.copy()

        # indices where the peak is attained
        active = np.flatnonzero(b == max_b)

        # accumulate gradient from all active shifts
        g = np.zeros_like(a)
        for k in active:
            # indices l such that 0 <= k - l < n  i.e. l in [k-(n-1), k]
            l_start = max(0, k - (n - 1))
            l_end = min(k, n - 1)
            idx = k - np.arange(l_start, l_end + 1)           # corresponds to a[idx]
            g[l_start:l_end + 1] += 2.0 * a[idx]
        g /= max(1, len(active))  # average if multiple active shifts

        # step size schedule – decreasing like 1/√t
        eta = 0.5 / np.sqrt(it)

        # gradient step + projection onto the ℓ₁‑simplex
        a = _project_simplex(a - eta * g, S)

        # occasional small random jitter to avoid stagnation
        if it % 1500 == 0:
            jitter = rng.normal(scale=0.003 * S, size=n)
            a = _project_simplex(np.maximum(a + jitter, 0.0), S)

    return best_a, best_peak


# ---------------------------------------------------------------
# Main entry point required by the harness
# ---------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs many short Adam restarts with diverse initial vectors,
    keeps the best candidate and finally refines it with a longer Adam run
    followed by an exact projected‑gradient descent.
    It stores the final sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    safety_margin = 5.0
    end_time = time.time() + budget_s - safety_margin

    # ------------------------------------------------------------------
    # Try to re‑use a previously discovered construction
    # ------------------------------------------------------------------
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempt = 0

    # ------------------------------------------------------------------
    # Helper to build an initial feasible vector for a given length
    # ------------------------------------------------------------------
    def _initial_vector(n: int) -> np.ndarray:
        # Warm‑start from the global best with a modest probability
        if base_seq is not None and rng.random() < 0.2:
            a = _scale_sequence(base_seq, n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * np.sqrt(2.0 * n)
            return a

        style = rng.choice(['uniform', 'dirichlet', 'edge', 'mix'])
        S = np.sqrt(2.0 * n)

        if style == 'uniform':
            return np.full(n, S / n)

        if style == 'dirichlet':
            a = rng.random(n)
            a = a / a.sum() * S
            return a

        if style == 'mix':
            # blend edge‑heavy and uniform components
            a_edge = _edge_initial(n, rng)
            a_unif = np.full(n, S / n)
            mix = rng.uniform(0.0, 0.6)
            a = mix * a_edge + (1.0 - mix) * a_unif
            a = a / a.sum() * S
            return a

        # edge‑heavy
        return _edge_initial(n, rng)

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam restarts (exploration)
    # ------------------------------------------------------------------
    short_budget = 8.0                     # seconds per short run
    while time.time() + short_budget < end_time - 30.0:
        attempt += 1
        # Choose a length; favour the range that historically gave good results
        if rng.random() < 0.7:
            n = rng.integers(1500, 8000)
        else:
            n = rng.integers(8000, 25000)

        a_start = _initial_vector(n)

        slice_deadline = min(time.time() + short_budget, end_time - 30.0)
        a_opt, peak = _adam_optimize(a_start, slice_deadline, rng)

        if peak < best_score:
            best_score = peak
            best_seq = a_opt.copy()
            best_n = n
            print(f"[restart] attempt={attempt} n={n}  new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam refinement on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 15.0:
            S = np.sqrt(2.0 * len(best_seq))
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq / best_seq.sum() * S

            a_refined, peak = _adam_optimize(best_seq, time.time() + remaining - 7.0, rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_refined.copy()
                print(f"[adam‑refine] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final exact projected gradient descent (fine‑tuning)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = end_time - time.time()
        if remaining > 7.0:
            a_fine, peak = _pgd_optimize(best_seq, time.time() + remaining, rng)
            if peak < best_score:
                best_score = peak
                best_seq = a_fine.copy()
                print(f"[pgd‑fine] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check via the black‑box evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]  # fallback
    else:
        final_seq = best_seq.tolist()
        external = evaluate_sequence(final_seq)
        print(f"[final] external score = {external:.6f}  internal max = {best_score:.6f}  length = {len(final_seq)}")
        globals()['height_sequence_1'] = final_seq

    return final_seq
