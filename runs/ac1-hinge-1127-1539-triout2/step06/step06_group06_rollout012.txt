def optimize_with_gd(a0, deadline):
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)
    beta = 1.0
    max_beta = 8192.0
    step_scale = target_sum * 0.2
    momentum = np.zeros_like(a)
    gamma = 0.9
    best_a = a.copy()
    best_val = _score(a)
    while time.time() < deadline:
        # gradient of softmax max
        g,_ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_scale / (norm * (beta + 1))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # maybe jitter occasionally
        # alternately increase beta
        if np.random.rand() < 0.01:
            if beta < max_beta:
                beta = min(beta * 1.5, max_beta)
        # compute score
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # break if time low
    return best_a, best_val

"""
search for a step‑function (non‑negative height sequence) that minimises
the benchmark defined by the provided `evaluate_sequence`.  The strategy is
a collection of short, independent restarts:

* each restart starts from a simple analytically‑generated shape
  (power‑law, shifted power‑law, or a random Dirichlet vector) that is
  projected onto the ℓ¹‑simplex Σa_i = √(2·n);
* a projected gradient descent on a soft‑max approximation of the
  convolution maximum is carried out (β is increased up to a large value);
* the result is refined with a lightweight simulated‑annealing phase;
* the best sequence encountered before the global time budget expires is
  returned.

The algorithm keeps the whole search inside the time limit (default 1000 s)
and never calls `evaluate_sequence` except for a final sanity check,
using a fast NumPy‑based surrogate (`_score`) during the optimisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Fast surrogate for `evaluate_sequence`: 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x | x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic initial shapes
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Return (i+1)^{-p} (i=0..n‑1) normalised to `target_sum`."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _shifted_power_law_sequence(n: int, p: float, shift: int,
                                target_sum: float) -> np.ndarray:
    """(i+shift+1)^{-p} shape, shifted by `shift` indices."""
    idx = np.arange(shift + 1, shift + n + 1, dtype=np.float64)
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient of max‑convolution
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).

    Returns (grad, conv) where
        conv = a * a      (full convolution)
        grad = ∂softmax/∂a   (shape n)
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max weights (probability distribution over all convolution indices)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (rev(a) *⋆ p)  (restricted to the original length)
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float,
                               beta_max: float = 16384.0) -> tuple[np.ndarray, float]:
    """
    Short projected gradient descent on the soft‑max approximation.
    The schedule is:
        - start β = 1.0, increase multiplicatively up to `beta_max`;
        - basic momentum + occasional jitter;
        - after every iteration project onto the ℓ¹‑simplex.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    step_scale = target_sum * 0.15
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.9
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_scale / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of shallow basins
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # increase β slowly (log‑linear schedule)
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.6)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very light simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 0.5,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.25,
                         max_frac_end: float = 0.01,
                         max_window: int = 40) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves mass between two indices, smoothes short windows
    and occasionally swaps two positions.  The temperature and the maximal
    transferred fraction decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-9)
    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # -------------------------------------------------------------
        # pick a move
        # -------------------------------------------------------------
        move_type = None
        # 60 % mass‑transfer, 30 % short smoothing, 10 % swap
        r = rng.random()
        if r < 0.60:
            # mass transfer between a “heavy” and a “light” index
            weight = a * conv  # crude sensitivity estimate
            if weight.sum() == 0:
                donor = rng.integers(n)
            else:
                donor = rng.choice(n, p=weight / weight.sum())
            receiver = rng.integers(n)
            while receiver == donor:
                receiver = rng.integers(n)

            delta = a[donor] * rng.random() * max_frac
            if delta > 0:
                a[donor] -= delta
                a[receiver] += delta
                move_type = "transfer"
        elif r < 0.90:
            # short‑window smoothing
            start_idx = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_idx = min(start_idx + length, n)
            segment = a[start_idx:end_idx]
            avg = segment.mean()
            saved = segment.copy()
            a[start_idx:end_idx] = avg
            move_type = ("smooth", start_idx, saved)
        else:
            # random swap
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = ("swap", i, j)

        # -------------------------------------------------------------
        # evaluate move
        # -------------------------------------------------------------
        new_conv = _conv_fft(a)
        new_score = _score(a)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = new_conv
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif isinstance(move_type, tuple) and move_type[0] == "smooth":
                start_idx, saved = move_type[1], move_type[2]
                a[start_idx:start_idx + saved.size] = saved
            elif isinstance(move_type, tuple) and move_type[0] == "swap":
                i, j = move_type[1], move_type[2]
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to escape stagnation
        if it % 4000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.clip(a, 0, None)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a height sequence that (hopefully) improves on the current
    best bound.  The routine runs many short independent restarts,
    each consisting of a fast gradient‑descent phase followed by a small
    simulated‑annealing refinement.  The best sequence found before the
    time budget expires is output.
    """
    start = time.time()
    deadline = start + budget_s - 2.0       # keep a small safety margin

    # ------------------------------------------------------------------
    # problem size – a compromise between solution quality and speed
    # ------------------------------------------------------------------
    n = 1800                               # length of the candidate vectors
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # pool of initialisations
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    init_pool = []

    # – power‑law families (different exponents, optional shift)
    for p in np.linspace(0.55, 0.85, 7):
        init_pool.append(_power_law_sequence(n, float(p), target_sum))
        for sh in (0, 3, 7):
            init_pool.append(_shifted_power_law_sequence(n, float(p), sh,
                                                         target_sum))

    # – a few random Dirichlet draws (different concentration)
    for conc in (0.2, 0.5, 1.0):
        for _ in range(8):
            seq = rng.dirichlet(np.full(n, conc)) * target_sum
            init_pool.append(seq)

    # – fall‑back to a previously found solution if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # simple linear resize to the current length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        init_pool.append(prev)

    # ------------------------------------------------------------------
    # main optimisation loop (restart‑driven)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for init in init_pool:
        # stop early if time is exhausted
        now = time.time()
        if now >= deadline - 5.0:
            break

        # allocate a tiny time slice for this restart
        slice_deadline = now + min(7.0, deadline - now - 1.0)

            cur_val = min(g_val, s_val)
        else:
            cur_val = g_val

        if cur_val < best_score:
            best_score = cur_val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final tidy‑up (trim trailing zeros, final evaluation)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple uniform vector
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # one final call to the official benchmark (optional, for logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
