def subgradient_descent(a, deadline):
    n = len(a)
    target_sum = np.sqrt(2*n)
    a = project_to_simplex(a, target_sum)
    best_a = a.copy()
    best_val = compute_score(a)

    iter = 0
    while time.time() < deadline:
        conv = conv_fft(a) # compute convolution via FFT
        max_c = np.max(conv)
        k_max = int(np.argmax(conv))
        # compute gradient
        g = np.zeros_like(a)
        # compute contributions
        # vectorized:
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g[valid] = 2 * a[idx[valid]]
        # step size schedule:
        step = (target_sum) * 0.2 / (np.linalg.norm(g) + 1e-12) / (1 + iter*0.01)
        a = a - step * g
        # enforce nonnegativity and sum
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        # evaluate score in each 10 iterations (to reduce cost)
        if iter % 5 == 0:
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Subgrad iter={iter} score={val:.6f}")
        iter += 1
    return best_a, best_val

"""
Multi‑scale optimisation for the benchmark defined in ``evaluate_sequence``.
The optimiser works on three increasingly large problem sizes.
For each size the routine performs

1. A projected gradient descent on a soft‑max approximation of the
   convolution maximum (β is increased gradually up to 2ⁿ⁺¹).  Momentum
   and occasional jitter help escape shallow minima.

2. A simulated‑annealing refinement that moves mass between indices,
   locally smooths short windows and swaps values.  Temperature and
   the allowed transfer fraction decay exponentially.

3. A short “random‑restart” phase that creates fresh random vectors,
   runs a few gradient steps and keeps any improvement.

All iterates are projected onto the ℓ¹‑simplex
Σ a_i = √(2·n) after each update, which preserves the objective value
but keeps the numbers well‑scaled and non‑negative.

The total runtime is bounded by the *budget_s* argument (default 1000 s).
The best sequence found across all scales is returned as a plain Python
list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra / convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:                               # safety (should never happen)
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex Σx_i = s, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x | x≥0, Σx=s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple helpers to generate and resize candidate vectors
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


# ----------------------------------------------------------------------
# Gradient on a soft‑max approximation of the max‑convolution entry
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns (grad, conv_full) where grad has the same shape as ``a``.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    # soft‑max over the convection entries
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)                     # length 2·n‑1, non‑negative, sums to 1

    # gradient = 2 * (a_rev) ⋆ p   (⋆ = linear convolution)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    # the central n entries correspond to the gradient w.r.t. each a_i
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Projected (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    β is increased up to 8192.  Simple momentum + jitter are used.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 8192.0
    step_base = target_sum * 0.5          # larger base step than the original code
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    gamma = 0.95                         # stronger momentum
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + g
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake off tiny plateaus
        if it % 350 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-4,
                         max_frac_start: float = 0.30,
                         max_frac_end: float = 0.01,
                         max_window: int = 50) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement using mass transfers, short‑window smoothing
    and random swaps.  Temperature and allowance decay exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)
    size_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / total_time
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # index delivering the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # --- mass‑transfer move (70 % of the time) -----------------
        if rng.random() < 0.70:
            # sensitivity = a * (partial derivative of current max entry)
            sens = np.zeros(n)
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            sens[idx] = a[k_max - idx]

            weight = a * sens
            if weight.sum() > 0:
                donor = rng.choice(n, p=weight / weight.sum())
            else:
                donor = rng.integers(n)

            low_thr = np.percentile(weight, 10)
            low_idx = np.where(weight <= low_thr)[0]
            if low_idx.size:
                receiver = rng.choice(low_idx)
            else:
                receiver = rng.integers(n)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    move_type = "transfer"
                else:
                    move_type = None
            else:
                move_type = None

        # --- local smoothing move (≈20 % of the time) --------------
        elif rng.random() < 0.90:
            start = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end = min(start + length, n)
            if end - start >= 2:
                old_seg = a[start:end].copy()
                a[start:end] = old_seg.mean()
                saved_seg = old_seg
                saved_start, saved_end = start, end
                move_type = "smooth"
            else:
                move_type = None

        # --- random swap (≈10 % of the time) -----------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            move_type = "swap"
            i, j = i, j   # keep names for possible revert

        # evaluate new state (full convolution via FFT)
        fa = np.fft.rfft(a, n=size_fft)
        conv_new = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        new_score = _score(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert the rejected move
            if move_type == "transfer":
                a[donor] += delta
                a[receiver] -= delta
            elif move_type == "smooth":
                a[saved_start:saved_end] = saved_seg
            elif move_type == "swap":
                a[i], a[j] = a[j], a[i]

        # occasional global jitter to avoid being stuck
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            cur_sum = a.sum()
            if cur_sum > 0:
                a = a * (np.sqrt(2.0 * n) / cur_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# One optimisation run for a given length
# ----------------------------------------------------------------------


def _run_one_instance(n: int, init_seq: np.ndarray, deadline: float):
    """
    Optimise a single instance of length ``n``.
    Steps:
      1️⃣ gradient descent on the soft‑max surrogate,
      2️⃣ simulated‑annealing refinement,
      3️⃣ a few short random‑restart gradient runs.
    Returns the best vector and its final score.
    """
    target_sum = np.sqrt(2.0 * n)

    # initialise – make sure the length matches
    a = init_seq.astype(np.float64, copy=True)
    if a.size != n:
        a = _resize_sequence(a, n)
    a = _project_to_simplex(a, target_sum)

    now = time.time()
    remaining = deadline - now
    # ---- Phase 1: gradient descent (≈30 % of remaining budget) ----
    if remaining > 30.0:
        gd_deadline = now + min(30.0, remaining * 0.30)
        a, _ = _annealed_gradient_descent(a, gd_deadline)
        now = time.time()
        remaining = deadline - now

    # ---- Phase 2: simulated annealing (use the rest) ----------
    if now < deadline:
        a, _ = _simulated_annealing(a, deadline)

    # ---- Phase 3: cheap random‑restart polish ------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 5.0:
        # create a fresh random vector
        a0 = rng.random(n)
        a0 = _project_to_simplex(a0, target_sum)

        # a very short gradient descent (≈5 s or less)
        sub_deadline = time.time() + min(5.0, deadline - time.time())
        a0, _ = _annealed_gradient_descent(a0, sub_deadline)

        val = _score(a0)
        cur = _score(a)
        if val < cur:
            a = a0
            print(f"[{time.time():.2f}] Random‑restart improved to {val:.6f}")

    final_score = _score(a)
    return a, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the benchmark.
    The routine runs three scales (≈2 k → 8 k → 12 k) re‑using the best
    solution of the previous scale as a warm start.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # tiny safety margin

    best_seq = None
    best_score = float("inf")
    cur_seq = None

    # (length, fraction of the remaining time dedicated to the scale)
    scales = [(2000, 0.15), (8000, 0.35), (12000, 0.50)]

    for n, frac in scales:
        now = time.time()
        if now >= deadline - 5.0:
            break
        remaining = deadline - now
        stage_deadline = now + remaining * frac

        # initialise the vector for this scale
        if cur_seq is not None:
            init = _resize_sequence(cur_seq, n)
        else:
            if "height_sequence_1" in globals():
                init = _resize_sequence(
                    np.array(globals()["height_sequence_1"], dtype=np.float64), n)
            else:
                init = _power_law_sequence(n, 0.55, np.sqrt(2.0 * n))

        seq, score_val = _run_one_instance(n, init, stage_deadline)

        if score_val < best_score:
            best_score = score_val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Scale n={n} new best score={best_score:.6f}")

        cur_seq = seq      # warm start for the next scale

    # Trim trailing near‑zeros for a tidy output
    if best_seq is None:
        # very unlikely – fall back to a simple power‑law vector
        best_seq = _power_law_sequence(2000, 0.55, np.sqrt(2.0 * 2000))

    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Final evaluation (optional, just for logging)
    try:
        val = evaluate_sequence(final_seq)   # provided by the harness
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
