def _apply_transfer(a, conv, donor, receiver, delta):
    # donor loses delta, receiver gains delta
    if delta <= 0:
        return
    a_before = a.copy()
    n = a.shape[0]
    conv[donor:donor+n] -= 2.0 * delta * a_before
    conv[receiver:receiver+n] += 2.0 * delta * a_before
    # quadratic terms
    conv[2*donor] += delta * delta
    conv[2*receiver] += delta * delta
    conv[donor+receiver] -= 2.0 * delta * delta
    # update a
    a[donor] -= delta
    a[receiver] += delta
    # ensure nonnegative due to numerical errors
    a[donor] = max(a[donor], 0.0)

"""
Local‑search optimizer for the step‑function heights.

The objective (after fixing the ℓ¹‑norm Σ a_i = √(2·n)) is simply the
maximum entry of the self‑convolution a*a.  The algorithm works as follows:

1.  Initialise a with a power‑law (or the previously found best sequence)
    and project it onto the simplex {a ≥ 0, Σa = √(2·n)}.
2.  Repeatedly perform cheap “mass‑transfer” moves:
        – pick the index k that attains the current maximum of the convolution,
        – choose a donor index among those that contribute most to this entry,
        – choose a receiver that currently contributes little,
        – move a random fraction of the donor’s mass to the receiver.
    The convolution is updated exactly by an O(n) incremental formula,
    so thousands of moves can be evaluated per second.
    The move is accepted if it lowers the peak or with a small
    Metropolis probability that decays (simulated‑annealing schedule).
3.  Occasionally perform a smoothing move (replace a short random window
    by its average) or add a small Gaussian jitter followed by a projection
    onto the simplex.  These moves are also accepted with the same annealing
    rule.
4.  Keep the best sequence seen so far.
5.  After the time budget for the local search is exhausted, polish the
    best candidate using the projected‑gradient method on a soft‑max
    approximation of the max‑convolution (the same code used in the
    reference solution).

The whole procedure runs within the given time limit (default 1000 s) and
returns a list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create a (i+1)^(-p) sequence of length n and renormalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (assumes Σa = √(2·n))."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Fast objective value (max convolution) – sum is already √(2·n)."""
    return _max_convolution(a)


# ----------------------------------------------------------------------
# Incremental mass‑transfer update (exact)
# ----------------------------------------------------------------------


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    donor: int,
                    receiver: int,
                    delta: float) -> None:
    """
    Transfer an amount `delta` from `donor` to `receiver`.
    Updates both the coefficient vector `a` and the convolution array `conv`
    exactly (including the quadratic `delta**2` terms).
    """
    if delta <= 0.0 or donor == receiver:
        return
    n = a.shape[0]

    # copy of a before the move – used for the linear part of the update
    a_before = a.copy()

    # linear contribution:  -2·δ·a_before shifted by donor, +2·δ·a_before shifted by receiver
    conv[donor: donor + n] -= 2.0 * delta * a_before
    conv[receiver: receiver + n] += 2.0 * delta * a_before

    # quadratic contribution from d*d
    conv[2 * donor] += delta * delta
    conv[2 * receiver] += delta * delta
    conv[donor + receiver] -= 2.0 * delta * delta

    # update the coefficients
    a[donor] -= delta
    a[receiver] += delta

    # protect against tiny negative values caused by rounding
    if a[donor] < 0.0:
        a[donor] = 0.0
    conv[:] = np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Local‑search core
# ----------------------------------------------------------------------


def _local_search(n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Perform the incremental local search for a given length `n`.
    Returns the best sequence found and its objective value (max convolution).
    """
    target = np.sqrt(2.0 * n)

    # ---- initialise ----------------------------------------------------
    if init_seq is not None:
        a = np.array(init_seq, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        else:
            a = np.pad(a, (0, n - a.size), "constant")
    else:
        # random power‑law exponent in a sensible range
        p = rng.uniform(0.4, 0.8)
        a = _power_law_sequence(n, p, target)

    a = _project_to_simplex(a, target)

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start

    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total_time

        # schedules (warm → cold)
        max_frac = 0.5 * (0.005 / 0.5) ** tau          # max fraction of donor mass to move
        temp = 0.5 * (1e-5 / 0.5) ** tau               # temperature for Metropolis test

        move_type = rng.random()

        # --------------------------------------------------------------
        # 1) mass transfer (≈80 % of the moves)
        # --------------------------------------------------------------
        if move_type < 0.80:
            k_max = int(np.argmax(conv))

            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1)
            w = a[idx] * a[k_max - idx]               # contribution of each index
            total_w = w.sum()
            if total_w > 0.0:
                donor = rng.choice(idx, p=w / total_w)
            else:
                donor = rng.integers(n)

            # bias receiver towards small coefficients
            inv = 1.0 / (a + 1e-12)
            receiver = rng.choice(np.arange(n), p=inv / inv.sum())
            if receiver == donor:
                continue

            delta = a[donor] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            _apply_transfer(a, conv, donor, receiver, delta)
            new_max = conv.max()

            # Metropolis acceptance
            accept = False
            if new_max <= cur_max:
                accept = True
            else:
                prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] new best max={best_max:.6f}")
            else:
                # revert the move
                _apply_transfer(a, conv, receiver, donor, delta)

        # --------------------------------------------------------------
        # 2) short smoothing window (≈10 % of the moves)
        # --------------------------------------------------------------
        elif move_type < 0.95:
            max_window = min(100, n // 10)
            start_win = rng.integers(0, n)
            end_win = start_win + rng.integers(2, max_window + 1)
            end_win = min(end_win, n)
            if end_win - start_win >= 2:
                saved = a[start_win:end_win].copy()
                avg = saved.mean()
                a[start_win:end_win] = avg
                conv = _conv_fft(a)
                new_max = conv.max()

                accept = False
                if new_max <= cur_max:
                    accept = True
                else:
                    prob = np.exp(-(new_max - cur_max) / max(temp, 1e-12))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a.copy()
                        print(f"[{time.time():.1f}] smoothing improved max={best_max:.6f}")
                else:
                    # revert smoothing
                    a[start_win:end_win] = saved
                    conv = _conv_fft(a)

        # --------------------------------------------------------------
        # 3) Gaussian jitter (≈10 % of the moves)
        # --------------------------------------------------------------
        else:
            scale = target * 0.001 * (1.0 - tau)
            a += rng.normal(scale=scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
            conv = _conv_fft(a)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] jitter improved max={best_max:.6f}")

    return best_a, best_max

# Projected gradient descent on a soft‑max approximation (reference code)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g, b


def _annealed_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected gradient descent on the soft‑max approximation.
    The temperature β is increased up to 4096.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _score(a)

    beta = 1.0
    max_beta = 4096.0
    step_base = target_sum * 0.12
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        norm_g = np.linalg.norm(g)
        if norm_g > 0:
            lr = step_base / (norm_g * (beta + 1.0))
            a = a - lr * g

        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.1f}] GD improvement β={beta:.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    return best_a, best_val
