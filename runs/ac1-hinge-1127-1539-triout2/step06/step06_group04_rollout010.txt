def _subgradient_descent_opt(a0, deadline):
    a = a0.copy()
    n = a.size
    target = np.sqrt(2.0 * n)
    rho = 0.9
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute full convolution using FFT; returns array length 2n-1
        b = _full_convolution(a)
        kmax = int(np.argmax(b))
        # gradient
        g = np.zeros_like(a)
        i_low = max(0, kmax - (n-1))
        i_high = min(n-1, kmax)
        # generate indices
        i = np.arange(i_low, i_high + 1)
        j = kmax - i
        g[i] = 2.0 * a[j]
        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            # step size: adapt based on target and iteration
            lr = target * 0.15 / (norm_m + 1e-12)
            a -= lr * momentum
        # projection onto simplex
        a = _project_to_simplex(a, target)
        # jitter occasionally
        if it % 200 == 0:
            jitter_scale = target * 0.001 / (1 + it * 0.001)
            a += rng.normal(scale=jitter_scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target)
        it += 1
    return a

"""
Hybrid sub‑gradient descent + targeted hill‑climbing.

* The objective (with the ℓ¹‑normalisation Σa = √(2 n)) equals the
  maximal entry of the self‑convolution a*a.
* Sub‑gradient descent directly follows the (sub‑)gradient of that
  max‑entry: if k* is the index attaining the maximum, the gradient is
  g_i = 2·a_{k*-i} (zero when the index is out of range).  We move
  opposite to this gradient, project back onto the ℓ¹‑simplex and add
  occasional jitter to escape plateaus.
* After a phase of sub‑gradient descent we run the exact hill‑climber
  from the reference implementation (moves a tiny amount of mass from a
  “bad’’ source index to a “good’’ destination, updating the convolution
  analytically).
* The search proceeds through a schedule of lengths (2000, 5000, 8000,
  12000, 18000).  Each length is initialised by up‑sampling the best
  sequence found so far, then refined by a sub‑gradient phase followed
  by hill‑climbing.  Remaining time is spent on a final polishing run.
* All sequences are kept non‑negative and projected onto the simplex
  Σa = √(2 n), guaranteeing a fair comparison of the objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (FFT based)."""
    return float(_full_convolution(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Duchi‑et‑al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialiser / up‑sampler (same ideas as the previous code)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng) -> np.ndarray:
    """Sample a non‑negative vector on the ℓ¹‑simplex Σa = √(2 n)."""
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)

    idx = np.arange(n, dtype=float)

    if family == 0:  # power‑law
        p = rng.uniform(0.10, 1.2)
        a = (idx + 1.0) ** (-p)

    elif family == 1:  # exponential decay
        lam = rng.exponential(scale=0.018)
        a = np.exp(-lam * idx)

    elif family == 2:  # mixed power‑law × exponential
        p = rng.uniform(0.12, 1.0)
        lam = rng.exponential(scale=0.018)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:  # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:  # piecewise exponential
        cut = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(cut)),
             np.exp(-lam2 * np.arange(cut, n))]
        )

    elif family == 5:  # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    else:  # Gaussian‑like bump
        centre = rng.uniform(0, 1)
        width = rng.exponential(scale=0.15) + 0.05
        a = np.exp(-((idx / n - centre) ** 2) / (2 * width ** 2))

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation onto a new length, followed by simplex projection."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (non‑smoothed)
# ----------------------------------------------------------------------


def _subgradient_descent_opt(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the maximal entry of a*a.
    Uses momentum, a simple step‑size schedule and occasional jitter.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    rho = 0.92                # momentum decay
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        # ---- compute the current maximal convolution entry ----
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # ---- sub‑gradient of the max at k_max ----
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        # vectorised assignment
        i = np.arange(i_low, i_high + 1)
        j = k_max - i
        g[i] = 2.0 * a[j]

        # ---- momentum update ----
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            # step size inversely proportional to the gradient norm
            lr = target_sum * 0.12 / (norm_m + 1e-12)
            a -= lr * momentum

        # ---- projection onto the ℓ¹‑simplex ----
        a = _project_to_simplex(a, target_sum)

        # ---- occasional jitter to leave plateaus ----
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (1 + it * 0.001),
                                size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return a


# ----------------------------------------------------------------------
# Targeted hill‑climber (unchanged from the reference implementation)
# ----------------------------------------------------------------------


def _hill_climb_targeted(a0: np.ndarray, b0: np.ndarray, deadline: float):
    """
    Move a small amount of mass from an index that contributes a lot to the
    current peak towards an index that contributes little.  Returns the
    best sequence found and its maximal convolution value.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T = cur_max * 0.02 + 1e-9          # initial temperature
    cooling = 0.9995

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each index to that peak:
        j_arr = k_max - np.arange(n)
        mask = (j_arr >= 0) & (j_arr < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[j_arr[mask]]

        total = contrib.sum()
        if total > 0.0:
            prob_src = contrib / total
            i = rng.choice(n, p=prob_src)
        else:
            prob_src = a / a.sum()
            i = rng.choice(n, p=prob_src)

        inv = contrib.max() - contrib + 1e-12
        prob_dst = inv / inv.sum()
        j = rng.choice(n, p=prob_dst)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta < 1e-16:
            continue
        delta = rng.random() * max_delta

        # ---- analytical update of the convolution --------------------
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a

        k2i = 2 * i
        k2j = 2 * j
        kij = i + j
        if k2i < b_new.size:
            b_new[k2i] += delta * delta
        if k2j < b_new.size:
            b_new[k2j] += delta * delta
        if kij < b_new.size:
            b_new[kij] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- Metropolis acceptance -------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            b = b_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_targeted improvement: {best_max:.6f}")

        T *= cooling

    return best_a, best_max
