"""
Improved heuristic for minimizing ``evaluate_sequence``.

The algorithm builds a schedule of increasing lengths.  For each length
it:

1.  Upsamples the best solution seen so far (or samples a fresh
    simplex‑point) and projects onto the ℓ¹‑simplex.
2.  Runs a longer projected gradient descent on a smooth log‑sum‑exp
    surrogate of the convolution maximum (momentum, geometric β‑annealing,
    occasional jitter).
3.  Performs an aggressive mass‑transfer hill‑climb that updates the full
    self‑convolution in O(n) per move.  The source index is chosen
    proportionally to its contribution to the current worst entry,
    while the destination is biased towards small heights.  The temperature
    schedule is slower to allow uphill moves.
4.  After hill‑climbing a short polishing gradient‑descent phase is run.

The schedule is adapted to the remaining wall‑clock time, allowing up to
~40 000 coefficients within the 1000 s budget.  The best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT, simplex projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using an FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a as a NumPy array."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of (1/β)·log Σ exp(β·b)   where   b = a*a .
    Returns (g, b) with
        g = 2·(a ⋆ p)   and   p_k ∝ exp(β·b_k).
    """
    n = a.shape[0]

    # b = a*a  (full convolution)
    size1 = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(a_fft * a_fft, n=size1)[: 2 * n - 1]

    # soft‑max probabilities over b
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient g = 2·(a ⋆ p)
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1          # = 3n‑2
    size2 = _next_pow2(conv_len)
    Ar_fft = np.fft.rfft(a_rev, n=size2)
    Pr_fft = np.fft.rfft(p, n=size2)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size2)[:conv_len]
    g = 2.0 * conv_rev[n - 1: n - 1 + n]
    return g, b


def _gradient_descent_opt(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Projected gradient descent on the smooth max.
    Momentum, geometric β‑annealing and occasional jitter are used.
    Stops when the wall‑clock deadline is reached.
    """
    a = a0.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)

    beta = 0.5
    beta_max = 16384.0
    beta_factor = 1.5

    step_base = target_sum * 0.2
    momentum = np.zeros_like(a)
    rho = 0.90

    it = 0
    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # momentum update
        momentum = rho * momentum + (1.0 - rho) * g
        norm_m = np.linalg.norm(momentum)
        if norm_m > 0.0:
            lr = step_base / (norm_m * (beta + 1.0))
            a -= lr * momentum

        # projection back to the simplex
        a = _project_to_simplex(a, target_sum)

        # jitter every few iterations
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.0008 / (beta + 1.0), size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # β annealing
        if it % 400 == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        it += 1
    return a


# ----------------------------------------------------------------------
# Random initialisers (several families)
# ----------------------------------------------------------------------


def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector on the ℓ¹‑simplex
    Σa = √(2 n).  Mixed families (power‑law, exponential, Dirichlet,
    piecewise, Gaussian) are sampled.
    """
    target = np.sqrt(2.0 * n)
    family = rng.integers(0, 7)
    idx = np.arange(n, dtype=float)

    if family == 0:                     # power‑law
        p = rng.uniform(0.15, 1.1)
        a = (idx + 1.0) ** (-p)

    elif family == 1:                   # exponential decay
        lam = rng.exponential(scale=0.015)
        a = np.exp(-lam * idx)

    elif family == 2:                   # mixed power‑law × exponential
        p = rng.uniform(0.15, 1.0)
        lam = rng.exponential(scale=0.015)
        a = (idx + 1.0) ** (-p) * np.exp(-lam * idx)

    elif family == 3:                   # Dirichlet (uniform on simplex)
        a = rng.dirichlet(alpha=np.ones(n)) * target

    elif family == 4:                   # piecewise exponential
        b = rng.integers(int(0.3 * n), int(0.7 * n))
        lam1 = rng.exponential(scale=0.02)
        lam2 = rng.exponential(scale=0.02)
        a = np.concatenate(
            [np.exp(-lam1 * np.arange(b)), np.exp(-lam2 * np.arange(b, n))]
        )

    elif family == 5:                   # Gaussian‑like bump
        mu = rng.uniform(0.25, 0.75) * n
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    else:                               # near‑uniform with tiny jitter
        a = np.full(n, target / n)

    a += rng.normal(scale=target * 0.001, size=n)
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target)
    return a


def _upsample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a sequence onto a new length, followed by a
    projection onto the appropriate simplex.
    """
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    target = np.sqrt(2.0 * new_n)
    a_new = _project_to_simplex(a_new, target)
    return a_new

def _hill_climbing(a0: np.ndarray, b0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Simulated‑annealing style hill climbing that transfers mass from a source
    index to a destination index while keeping the full self‑convolution up‑to‑date
    in O(n) per move.  The temperature decays slowly to allow occasional
    uphill moves when they help escape plateaus.
    """
    a = a0.copy()
    b = b0.copy()
    n = a.size
    cur_score = b.max()
    best_score = cur_score
    best_a = a.copy()

    # initial temperature loosely based on current worst entry
    T = cur_score * 0.12 + 1e-9
    cooling = 0.9997   # very slow cooling to keep exploration alive

    while time.time() < deadline:
        # --- pick one of the currently worst convolution indices ---
        max_val = cur_score
        max_idxs = np.flatnonzero(b == max_val)
        if max_idxs.size == 0:
            break
        M = rng.choice(max_idxs)

        # --- source index weighted by contribution to entry M ---
        i_low = max(0, M - (n - 1))
        i_high = min(n - 1, M)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        jdx = M - idx
        contrib = a[idx] * a[jdx]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = rng.choice(idx, p=probs)
        else:
            i = rng.integers(0, n)

        if a[i] < 1e-14:
            continue

        # --- destination index (bias towards small heights) ---
        if rng.random() < 0.4:
            inv_a = 1.0 / (a + 1e-12)
            probs_j = inv_a / inv_a.sum()
            j = rng.choice(n, p=probs_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # --- decide how much mass to move ---
        # try two candidates: a random fraction and the full amount
        max_delta = a[i] * 0.85
        delta_candidates = [rng.random() * max_delta, a[i] * 0.5, a[i] * 0.9]
        best_local_score = cur_score
        best_delta = 0.0
        best_b = None

        for delta in delta_candidates:
            if delta <= 0.0:
                continue
            # cheap copy and update
            b_new = b.copy()
            # linear terms
            b_new[i:i + n] -= 2.0 * delta * a
            b_new[j:j + n] += 2.0 * delta * a
            # quadratic corrections
            k2i = 2 * i
            k2j = 2 * j
            kij = i + j
            if k2i < b_new.size:
                b_new[k2i] += delta * delta
            if k2j < b_new.size:
                b_new[k2j] += delta * delta
            if kij < b_new.size:
                b_new[kij] -= 2.0 * delta * delta

            new_score = b_new.max()
            if new_score < best_local_score:
                best_local_score = new_score
                best_delta = delta
                best_b = b_new

        # --- accept the best candidate (Metropolis rule for non‑improving) ---
        if best_delta == 0.0:
            # no improving move found, maybe accept an uphill move
            delta = max_delta * 0.3
            b_new = b.copy()
            b_new[i:i + n] -= 2.0 * delta * a
            b_new[j:j + n] += 2.0 * delta * a
            k2i = 2 * i
            k2j = 2 * j
            kij = i + j
            if k2i < b_new.size:
                b_new[k2i] += delta * delta
            if k2j < b_new.size:
                b_new[k2j] += delta * delta
            if kij < b_new.size:
                b_new[kij] -= 2.0 * delta * delta
            new_score = b_new.max()
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)
        else:
            accept = best_local_score < cur_score or rng.random() < np.exp(-(best_local_score - cur_score) / T)

        if accept:
            # apply the chosen move
            if best_delta == 0.0:
                a[i] -= delta
                a[j] += delta
                b = b_new
                cur_score = new_score
            else:
                a[i] -= best_delta
                a[j] += best_delta
                b = best_b
                cur_score = best_local_score

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] hill_climb improvement: {best_score:.6f}")

        T *= cooling

    return best_a, b

        b = _full_convolution(a)
        final_deadline = deadline
        a_final, b_final = _hill_climbing(a, b, final_deadline, rng)
        final_score = b_final.max()
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time():.1f}] Final polishing improvement: score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up: drop trailing near‑zeros for neatness
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    best_seq = np.asarray(best_seq, dtype=float)
    nz = np.where(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]
    else:
        best_seq = best_seq[:1]

    # ------------------------------------------------------------------
    # Log the official evaluation (objective = max of the self‑convolution)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] evaluate_sequence → {val:.6f} (length={best_seq.size})")
    except Exception as exc:      # pragma: no cover
        print(f"Error calling evaluate_sequence: {exc}")

    return best_seq.tolist()
