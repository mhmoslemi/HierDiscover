"""
Optimise a non‑negative step function for the benchmark defined by
`evaluate_sequence`.  The approach has three stages:

1. **Project onto the ℓ¹‑simplex** – we keep the total mass fixed to
   sqrt(2·n) (any scaling leaves the objective invariant, this value is
   convenient for numerical stability).

2. **Adam based sub‑gradient descent on the true peak** – we compute the
   exact location of the maximum entry of the auto‑convolution
   (a ∗ a) and its sub‑gradient (the contribution of each coefficient to
   that peak).  An Adam optimiser with a projection step drives the peak
   down quickly.

3. **Fast simulated‑annealing polishing** – after the gradient phase we
   apply a cheap SA that moves mass between indices, locally smoothes
   short windows and swaps entries.  The SA respects the simplex
   constraint via a final projection.

The whole routine works on three increasing problem sizes
(~1200 → 2400 → 4800).  Each size receives a fraction of the total
budget (the `budget_s` argument, default 1000 s).  The best sequence
found over all scales is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq.astype(np.float64)


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """Create (i+1)^(-p) sequence of length n and normalise to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv_and_grad(a: np.ndarray):
    """
    Return (M, conv, g) where
      M  = max_{k} (a*a)[k],
      conv = full convolution a*a,
      g  = sub‑gradient of the max entry w.r.t. a.
    """
    n = a.shape[0]
    conv = _conv_full(a)
    k = int(np.argmax(conv))
    M = conv[k]

    # gradient of the k‑th entry:
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = k - i

    g = np.zeros_like(a)
    g[i] += a[j]                     # derivative w.r.t. a[i]
    mask = i != j
    # derivative w.r.t. a[j] for i != j
    g[j[mask]] += a[i[mask]]
    return M, conv, g


def _score_seq(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.size
    max_b = _conv_full(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient optimisation
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, deadline: float):
    """Adam optimisation on the peak of a*a with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = _project_to_simplex(a, target_sum)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = target_sum * 0.12          # base learning rate
    rng = np.random.default_rng()

    best_a = a.copy()
    best_score = _score_seq(a)

    it = 1
    while time.time() < deadline:
        M, _, grad = _max_conv_and_grad(a)

        # current score (sum always = target_sum after projection)
        cur_score = 2.0 * n * M / (target_sum * target_sum)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} score={best_score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1
        # keep a tiny safety buffer
        if time.time() > deadline - 2.0:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (light‑weight variant)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_temp: float = 1.0,
                         end_temp: float = 1e-5,
                         max_frac_start: float = 0.25,
                         max_frac_end: float = 0.02,
                         max_window: int = 60) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves mass between indices, smooths short windows and
    swaps entries.  Temperature and allowed transfer fraction decay
    exponentially.
    """
    a = a0.copy()
    n = a.size
    cur_score = _score_seq(a)

    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    size_fft = 1 << ((2 * n - 1).bit_length())
    conv = _conv_full(a)

    start = time.time()
    total = max(deadline - start, 1e-9)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / total
        temp = start_temp * (end_temp / start_temp) ** tau
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** tau

        # locate current peak
        k_max = int(np.argmax(conv))

        # ---- move type selection ----
        move = rng.random()
        saved = None
        if move < 0.60:      # mass transfer guided by sensitivity
            # sensitivities = a * (a reversed at the peak)
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            partner = k_max - idx
            sens = a[idx] * a[partner]

            total_w = sens.sum()
            if total_w > 0:
                donor = rng.choice(idx, p=sens / total_w)
            else:
                donor = rng.integers(n)

            # pick a receiver among low‑sensitivity indices
            low_thr = np.percentile(sens, 12)
            low = idx[sens <= low_thr]
            if low.size == 0:
                receiver = rng.integers(n)
            else:
                receiver = rng.choice(low)

            if donor != receiver:
                delta = a[donor] * rng.random() * max_frac
                if delta > 0:
                    a[donor] -= delta
                    a[receiver] += delta
                    saved = (donor, receiver, delta)

        elif move < 0.85:    # short‑window smoothing
            start_idx = rng.integers(0, n)
            length = rng.integers(2, max_window + 1)
            end_idx = min(start_idx + length, n)
            if end_idx - start_idx >= 2:
                old_seg = a[start_idx:end_idx].copy()
                a[start_idx:end_idx] = old_seg.mean()
                saved = ("smooth", start_idx, old_seg)

        else:                # swap two entries
            i, j = rng.choice(n, size=2, replace=False)
            a[i], a[j] = a[j], a[i]
            saved = ("swap", i, j)

        # recompute convolution and score
        conv_new = _conv_full(a)
        new_score = _score_seq(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it:5d} new best={best_score:.6f}")
        else:
            # revert move
            if saved is not None:
                if isinstance(saved, tuple) and len(saved) == 3 and isinstance(saved[0], str):
                    # smooth or swap
                    typ, pos, data = saved
                    if typ == "smooth":
                        a[pos:pos + data.size] = data
                    else:  # swap
                        i, j = pos, data
                        a[i], a[j] = a[j], a[i]
                else:
                    donor, receiver, delta = saved
                    a[donor] += delta
                    a[receiver] -= delta

        # occasional global jitter
        if it % 8000 == 0 and it > 0:
            jitter = rng.normal(scale=a.sum() * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))
            conv = _conv_full(a)
            cur_score = _score_seq(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Full optimisation for a given length
# ----------------------------------------------------------------------


def _optimize_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam descent (≈60 % of the time) followed by SA polishing."""
    now = time.time()
    remaining = deadline - now
    if remaining > 20.0:
        # split time: most to Adam, the rest to SA
        adam_deadline = now + remaining * 0.60
        a_mid, score_mid = _adam_optimize(a0, adam_deadline)
        # polish with SA on whatever time is left
        final_a, final_score = _simulated_annealing(a_mid, deadline)
        return final_a, min(score_mid, final_score)
    else:
        # not enough time – just run SA
        return _simulated_annealing(a0, deadline)

    """

    """
    start = time.time()
    deadline = start + budget_s - 1.0  # margin for safety

    best_seq = None
    best_score = float("inf")
    cur_seq = None
    rng = np.random.default_rng()

    # (length, portion of remaining time)
    scales = [(1200, 0.20), (2400, 0.30), (4800, 0.50)]
