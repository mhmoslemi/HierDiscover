"""
Propose a non‑negative step‑function (vector) that minimizes the peak of its
self‑convolution.  The algorithm is a multi‑scale, multi‑restart search that
combines three optimisation phases:

1. **Fast Adam on a smooth soft‑max approximation** – the max entry of the
   convolution is approximated by a log‑sum‑exp with a temperature that is
   increased slowly.  Adam with Euclidean simplex projection is used; an
   optional symmetry enforcement can be toggled per‑run.

2. **Sub‑gradient descent on the true max entry** – one step of the sub‑gradient
   of the active convolution entry is taken with a diminishing step‑size.
   Random jitter and occasional momentum improve escape from plateaus.

3. **Intensive local swap/annealing** – tiny mass exchanges between random
   indices are performed while constantly re‑projecting to the simplex.
   This refines the solution to a (near) locally optimal configuration.

The search repeatedly restarts with different lengths (2 k–30 k) and initial
shapes (uniform, geometric, reciprocal‑power, half‑sine, triangular, or an
upsampled copy of the current best).  The best sequence found before the
deadline is returned.  All operations respect the required ℓ¹‑norm
Σa = √(2 n), so the external evaluator simply returns the maximal
convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – projects onto {x >= 0, sum x = s}
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the probability simplex scaled by s."""
    if s < 0:
        raise ValueError("Target simplex sum must be non‑negative")
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find rho
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a, full length = 2*n‑1."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def grad_softmax_conv(a: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max (log‑sum‑exp) of the convolution w.r.t. a.
    weight must be a probability vector the same size as the full convolution.
    """
    n = a.shape[0]
    # Convolution of a with reversed weights (weights correspond to entries of a*a)
    w_rev = weight[::-1]
    conv_len = n + (2 * n - 1) - 1            # = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(Fa * Fw, n=fft_len)[:conv_len]
    # extract needed slice: (a * w_rev) evaluated at positions producing a*a terms
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * full[idx]
    return grad


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries with temperature 1/α."""
    scaled = alpha * conv
    m = np.max(scaled)
    exp_shift = np.exp(scaled - m)
    w = exp_shift / exp_shift.sum()
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry (v[i] = v[-1-i])."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------
def generate_initial(base: np.ndarray | None, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse non‑negative vector of length n with Σ = √(2 n)."""
    S = np.sqrt(2.0 * n)

    # Choose a family at random
    typ = rng.integers(0, 7)   # 0..6

    if typ == 0 and base is not None and base.size > 10:
        # Upsample an existing good shape and add tiny noise
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, base.size), base)
        a += rng.normal(scale=a.mean() * 5e-4, size=n)
        a = np.clip(a, 0.0, None)

    elif typ == 1:
        # Uniform Dirichlet(1)
        a = rng.random(n)

    elif typ == 2:
        # Geometric decay from both ends
        rate = rng.uniform(0.9, 0.9999)
        left = rate ** np.arange(n)
        a = (left + left[::-1]) * 0.5

    elif typ == 3:
        # Edge‑heavy reciprocal‑power shape
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.15, 0.85)
        a = rec ** power

    elif typ == 4:
        # Half‑sine shape (positive, single bump)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.clip(a, 0.0, None)

    elif typ == 5:
        # Triangular (peak in centre)
        centre = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - centre) / centre
        a = np.clip(a, 0.0, None)

    else:
        # Random convex combination of a few bases
        bases = []
        bases.append(rng.random(n))                     # uniform
        rate = rng.uniform(0.9, 0.9999)
        left = rate ** np.arange(n)
        bases.append((left + left[::-1]) * 0.5)         # geometric
        idx = np.arange(n, dtype=float)
        rec = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        power = rng.uniform(0.2, 0.8)
        bases.append(rec ** power)                      # reciprocal
        w = rng.random(len(bases))
        w /= w.sum()
        a = sum(wi * bi for wi, bi in zip(w, bases))

    # Normalise to the required ℓ¹‑norm
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0)
        total = a.sum()
    a = a / total * S
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp objective
# ----------------------------------------------------------------------
def optimise_adam(a0: np.ndarray, deadline: float, rng: np.random.Generator,
                  enforce_sym: bool) -> tuple[np.ndarray, float]:
    """Projected Adam minimising the soft‑max approximation of max‑conv."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # start temperature low, increase slowly
    alpha = 0.4
    alpha_mul = 1.018

    best_a = a.copy()
    best_score = np.max(full_convolution(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # soft‑max gradient
        w = softmax_weights(conv, alpha)
        grad = grad_softmax_conv(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if enforce_sym:
            a = symmetrize(a)

        a = project_simplex(a, S)

        # occasional random jitter to escape flat regions
        if rng.random() < 0.024:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.05 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = project_simplex(cand, S)
                cand_max = np.max(full_convolution(cand))
                if cand_max < best_score or rng.random() < 0.15:
                    a = cand

        # schedule updates
        alpha = min(alpha * alpha_mul, 50.0)
        lr *= 0.99897

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximal convolution entry
# ----------------------------------------------------------------------
def optimise_subgrad(a0: np.ndarray, deadline: float, rng: np.random.Generator,
                    enforce_sym: bool) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent targeting the worst convolution shift."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(full_convolution(a))

    eta0 = 0.18
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient for (a*a)[k_max]
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i_idx = np.arange(i_low, i_high + 1)
        j_idx = k_max - i_idx
        g[i_idx] = a[j_idx]

        eta = eta0 / np.sqrt(it)
        a = a - eta * g

        if enforce_sym:
            a = symmetrize(a)
        a = project_simplex(a, S)

        cur_max = np.max(full_convolution(a))
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # occasional random mass‑swap (simulated annealing style)
        if rng.random() < 0.053:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = 0.04 * a[i] * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = project_simplex(cand, S)
                if enforce_sym:
                    cand = symmetrize(cand)
                cand_max = np.max(full_convolution(cand))
                # accept if better, otherwise with small probability
                if cand_max < cur_max or rng.random() < 0.21:
                    a = cand
                    cur_max = cand_max
                    if cand_max < best_score:
                        best_score = cand_max
                        best_a = a.copy()

        # Early exit if stagnating
        if it % 11000 == 0 and best_score == cur_max:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Intensive local random‑swap refinement
# ----------------------------------------------------------------------
def local_swap_search(a0: np.ndarray, deadline: float, rng: np.random.Generator,
                     enforce_sym: bool) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing that swaps tiny amounts of mass."""
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = np.max(full_convolution(a))

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        delta = 0.022 * a[i] * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_simplex(cand, S)
        if enforce_sym:
            cand = symmetrize(cand)
        cand_score = np.max(full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            a = cand
    return best_a, best_score


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single restart
# ----------------------------------------------------------------------
def optimise_one_restart(a0: np.ndarray, deadline: float, rng: np.random.Generator,
                         enforce_sym: bool) -> tuple[np.ndarray, float]:
    """Run Adam → sub‑gradient → local search within the allocated slice."""
    now = time.time()
    remaining = deadline - now
    if remaining <= 0.0:
        return a0, np.max(full_convolution(a0))

    # Split the slice: 55 % Adam, 30 % sub‑grad, rest local
    t_adam = now + 0.55 * remaining
    a_adam, sc_adam = optimise_adam(a0, t_adam, rng, enforce_sym)

    t_sub = min(time.time() + 0.30 * remaining, deadline)
    a_sub, sc_sub = optimise_subgrad(a_adam, t_sub, rng, enforce_sym)

    if time.time() < deadline:
        a_loc, sc_loc = local_swap_search(a_sub, deadline, rng, enforce_sym)
    else:
        a_loc, sc_loc = a_sub, sc_sub

    candidates = [(a_adam, sc_adam), (a_sub, sc_sub), (a_loc, sc_loc)]
    best_seq, best_score = min(candidates, key=lambda kv: kv[1])
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a step‑function with minimal self‑convolution peak.
    Returns the best list of floats found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Leave a safety margin for the external evaluation call
    deadline = start_time + budget_s - 5.0

    # Attempt to use a previously discovered construction if present
    base_seq = None
    if 'height_sequence_1' in globals():
        try:
            base_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        except Exception:
            base_seq = None

    best_seq = None
    best_score = np.inf
    best_n = None
    attempts = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main timed loop – many short restarts with varied lengths and
    # symmetry settings.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Choose a problem size.
        # Frequently enlarge or shrink around the current best length.
        if best_seq is not None and rng.random() < 0.62:
            scale = rng.uniform(0.85, 1.18)
            n = max(1500, int(best_n * scale))
        else:
            n = int(rng.integers(1800, 26000))

        enforce_sym = rng.random() < 0.52

        # Initialise
        a0 = generate_initial(base_seq, n, rng)
        if enforce_sym:
            a0 = symmetrize(a0)

        # Allocate a time slice (≈ 15‑25 s or the remaining budget, whichever is smaller)
        slice_deadline = min(time.time() + rng.uniform(15.0, 26.0), deadline)

        # Optimise this restart
        a_opt, score = optimise_one_restart(a0, slice_deadline, rng, enforce_sym)

        # Update global best
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            best_n = n
            last_improve = time.time()
            print(f"[global] attempt={attempts} n={n} sym={enforce_sym} "
                  f"new best={best_score:.6f}")

        # If we have already beaten the target threshold we can stop early.
        if best_score <= 1.503:
            break

        # If we have been stuck for a while, drop the base_seq to regain diversity.
        if time.time() - last_improve > 50.0:
            base_seq = None

    # ------------------------------------------------------------------
    # Final up‑sampling & long refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        # Upsample the current best to a larger length for a final polish.
        target_n = max(best_n * 2, best_n + 4000)
        target_n = min(int(target_n), 30000)
        a_big = np.interp(np.linspace(0, 1, target_n),
                          np.linspace(0, 1, best_seq.size), best_seq)
        # Ensure exact sum constraint after up‑sampling
        S_big = np.sqrt(2.0 * target_n)
        a_big = a_big / a_big.sum() * S_big
        # Try a short but more aggressive optimisation
        final_deadline = deadline
        a_big_opt, big_score = optimise_one_restart(a_big, final_deadline,
                                                    rng, enforce_sym=False)
        if big_score < best_score:
            best_seq = a_big_opt
            best_score = big_score
            best_n = target_n
            print(f"[final‑refine] upsampled n={target_n} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result after a final external evaluation check.
    # ------------------------------------------------------------------
    if best_seq is None:
        final_seq = [1.0]
    else:
        final_seq = best_seq.tolist()

    external = evaluate_sequence(final_seq)
    print(f"[final] external={external:.6f} internal_best={best_score:.6f} "
          f"len={len(final_seq)} attempts={attempts}")

    # expose for potential later runs
    globals()['height_sequence_1'] = np.array(final_seq, dtype=float)

    return final_seq
