    best_a = a.copy()

        max_val = conv.max()

        idx_max = np.where(conv >= max_val - 1e-12)[0]
        p = np.zeros(conv.shape[0], dtype=np.float64)
        # use uniform weight on max indices

def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = x.size
    m = y.size
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    return conv

def _lp_improvement(a: np.ndarray, target_sum: float, deadline: float, t: float = 0.1, max_iters: int = 10) -> tuple[np.ndarray, float]:
    """
    Perform Frank-Wolfe style improvement using linear programming.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    while time.time() < deadline and max_iters > 0:
        # compute convolution
        conv = _conv_fft(a)
        M = conv.max()
        # Identify indices near the max (within a tolerance)
        tol = M * 1e-3  # maybe 0.1% tolerance
        idx = np.where(conv >= M - tol)[0]
        # Limit number of constraints
        if idx.size > 25:
            idx = idx[:25]  # take first few
        # Build A matrix shape (len(idx), n)
        m = idx.size
        A = np.zeros((m, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # For i in [lo, hi], a[k - i] is partner
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            A[row, i_range] = a[k - i_range]
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        # Use SciPy
        # SciPy's linprog expects A_ub @ x <= b_ub
        # Need to import scipy.optimize.linprog
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 1e-12:
            break
        # Rescale g to match target sum
        g = g0 * (target_sum / sum_g)
        # Update a
        a_new = (1 - t) * a + t * g
        # Evaluate
        score_new = _score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP improve score={best_score:.6f}")
        # Update a for next iteration
        a = a_new
        max_iters -= 1
    return best_a, best_score

def _lp_improvement(a0, target_sum, deadline, t=0.1, max_iter=5):
    from scipy.optimize import linprog
    a = _project_to_simplex(np.array(a0, copy=True), target_sum)
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]
    while time.time() < deadline and max_iter > 0:
        conv = _conv_fft(a)
        M = conv.max()
        # pick indices close to max
        tol = max(1e-8, 0.001 * M)  # 0.1% tolerance
        idx = np.where(conv >= M - tol)[0]
        # limit constraints
        if idx.size > 30:
            idx = idx[:30]
        m = idx.size
        # Build A matrix
        A = np.zeros((m, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            A[row, i_range] = a[k - i_range]
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        # b_ub = M for each constraint
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 1e-12:
            break
        # Scale g to have target sum
        g = g0 * (target_sum / sum_g)
        # Mix
        a_new = (1 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)
        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP mix improve score={best_score:.6f}")
        a = a_new
        max_iter -= 1
    return best_a, best_score

"""
Step‑function optimizer (approx‑Frank‑Wolfe + sub‑gradient + SA).

The routine combines several complementary heuristics that all respect the
simplex constraint Σ a_i = √(2 n), a_i ≥ 0:

1. **Fast parametric sweep** – evaluates a few analytic families
   (pure power‑law, power‑law with exponential tail, sinusoid) and keeps the
   best candidate.

2. **Deterministic mass‑transfer (exact DMT2)** – a greedy move that
   reduces the current maximal self‑convolution entry to the second‑largest
   one.

3. **Linear‑programming Frank‑Wolfe step** – solves a small LP
   (maximise Σ g_i subject to (f * g) ≤ current peak) and mixes the
   obtained direction into the current sequence.

4. **Simulated annealing** – stochastic mass transfers that allow the
   search to escape shallow minima.

5. **Sub‑gradient descent on the exact max‑convolution** – the gradient of
   the active (max) constraints is computed analytically via FFTs; a
   projected gradient step with a diminishing step‑size finishes the
   refinement.

All phases respect the overall time budget (default 1000 s).  The best
sequence found is trimmed of trailing near‑zero entries and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv_full


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simple objective wrapper (max convolution = objective under our normalisation)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the quick pre‑search
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------
def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # one of the maximal indices
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # values that pair with idx to make index k
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index with smallest partner among the same range, distinct from donor
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (simple stochastic mass‑transfer)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    n = a.shape[0]
    rng = np.random.default_rng()

    # temperature initialisation
    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.02
    top_k = max(1, int(top_frac * n))

    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.3
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score

# Sub‑gradient descent on the exact max‑convolution

def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:

    Projected sub‑gradient descent using the exact gradient of the active
    (max) convolution entries.  Step size is α = 0.5 * max / ||g||.
    """
    a = _project_to_simplex(np.array(a0, copy=True), target_sum)
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)  # FFT size, constant throughout this run
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx = np.where(conv >= max_val - 1e-12)[0]

        # gradient of the (average) max entry via FFT
        a_rev = a[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=size)

        p = np.zeros(conv.shape[0], dtype=np.float64)
        p[idx] = 1.0 / len(idx)          # uniform weight on all active constraints
        p_fft = np.fft.rfft(p, n=size)

        grad_full = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        norm = np.linalg.norm(grad) + 1e-12
        step_scale = 0.5
        alpha = step_scale * max_val / norm

        # candidate step
        cand = a - alpha * grad
        cand = _project_to_simplex(cand, target_sum)
        cand_score = _score(cand)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] SubGrad it={it:5d} improved score={best_score:.6f}")
        else:

                cand2 = _project_to_simplex(cand2, target_sum)
                cand2_score = _score(cand2)
                if cand2_score < best_score - 1e-12:
                    best_score = cand2_score
                    best_a = cand2.copy()
                    a = cand2
                    print(f"[{time.time():.2f}] SubGrad it={it:5d} reduced step score={best_score:.6f}")
                    improved = True
                    break
            if not improved:
                # small random jitter to escape plateaus
                jitter = rng.normal(scale=target_sum * 0.0005, size=n)
                a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming Frank‑Wolfe improvement (small LP)
# ----------------------------------------------------------------------
def _lp_improvement(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    t: float = 0.1,
                    max_iter: int = 5) -> tuple[np.ndarray, float]:
    """
    Solve a tiny LP:
        maximise Σ g_i    subject to  (a * g)_k ≤ M for all k where (a*a)_k is near the maximum.
    The solution is mixed with the current a using a small convex weight t.
    """
    from scipy.optimize import linprog
    a = _project_to_simplex(np.array(a0, copy=True), target_sum)
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]

    while time.time() < deadline and max_iter > 0:
        conv = _conv_fft(a)
        M = conv.max()
        # constraints: indices within 0.1% of the maximum
        tol = max(1e-12, 0.001 * M)
        idx = np.where(conv >= M - tol)[0]
        if idx.size > 30:                     # keep the LP tiny
            idx = idx[:30]

        m = idx.size
        if m == 0:
            break

        # Build matrix A (m × n) where A[p,i] = a_{k_p - i} if index valid
        A = np.zeros((m, n), dtype=np.float64)
        for row, k in enumerate(idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            A[row, i_range] = a[k - i_range]

        # Linear program: maximise Σ g_i  <=> minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            break

        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 1e-12:
            break

        # Scale g to keep the total mass equal to target_sum
        g = g0 * (target_sum / sum_g)

        # Mix and re‑project
        a_new = (1 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP improvement score={best_score:.6f}")

        a = a_new
        max_iter -= 1

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal
    self‑convolution.  The routine respects the time budget and returns
    the best list of heights found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Problem size (12000 works well within the time budget)
    # --------------------------------------------------------------
    n = 12000
    target_sum = np.sqrt(2.0 * n)    # Σ a = √(2 n) ⇒ objective = max‑convolution

    # --------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    p_grid = np.linspace(0.20, 1.20, 31)   # power‑law exponent
    q_grid = np.linspace(0.0, 10.0, 31)   # exponential tail strength

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) sine‑shaped candidate
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) reuse previous construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) a few jittered perturbations around the current best
    rng = np.random.default_rng()
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (quick reduction)
    # --------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.10
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After DMT1 best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – small LP‑based Frank‑Wolfe step
    # --------------------------------------------------------------
    lp_deadline = time.time() + (deadline - time.time()) * 0.10
    best_seq, best_score = _lp_improvement(best_seq,
                                            target_sum,
                                            lp_deadline,
                                            t=0.1,
                                            max_iter=5)
    print(f"[{time.time():.2f}] After LP step best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing (exploration)
    # --------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.25
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)
    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – another round of deterministic mass‑transfer
    # --------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.10
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt2_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.30
    best_seq, best_score = _subgradient_descent(best_seq,
                                                target_sum,
                                                subgrad_deadline)
    print(f"[{time.time():.2f}] After sub‑gradient descent best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – final tiny DMT clean‑up (if time permits)
    # --------------------------------------------------------------
    final_deadline = deadline
    if time.time() + 1.0 < final_deadline:
        best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                                   final_deadline,
                                                                   target_sum)
        print(f"[{time.time():.2f}] Final DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
