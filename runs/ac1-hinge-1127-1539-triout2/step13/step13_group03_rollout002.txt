def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()
    # define lengths based on remaining time
    # start with n=5000 to warm up quickly and then increase.
    lengths = [5000, 10000, 20000, 30000] # we adjust based on time left
    # adjust if not enough time.
    for n in lengths:
        target_sum = math.sqrt(2.0 * n)
        # compute how many random candidates we can try: around 250 for large n, more for smaller.
        # Determine time per eval: conv operations for n up to 30k maybe 0.02 sec each.
        per = 0.015  # approximate
        max_iters = int((deadline - time.time()) // per)
        iters = min(4000, max_iters) # limit
        for _ in range(iters):
            # generate random simplex
            a = rng.random(n)
            # optionally shape: add power law factor if random.
            shape_type = rng.integers(0,5)
            if shape_type == 0:
                # power law
                p = rng.uniform(0.1, 2.5)
                idx = np.arange(1, n+1)
                a = (idx ** -p)
                a = rng.random(n) * a  # random scaling
            elif shape_type == 1:
                lam = rng.uniform(0.0005, 0.05)
                a = np.exp(-lam * np.arange(n))
            elif shape_type == 2:
                # sinusoidal
                x = np.linspace(0, 1, n)
                a = np.sin(np.pi * x) ** rng.uniform(0.5, 2.0)
            elif shape_type == 3:
                # mixture: two gaussians
                mu1 = rng.uniform(0, 1)
                mu2 = rng.uniform(0, 1)
                sigma = rng.uniform(0.05, 0.15)
                x = np.linspace(0, 1, n)
                a = np.exp(-0.5*((x-mu1)/sigma)**2) + np.exp(-0.5*((x-mu2)/sigma)**2)
            else:
                # flat random
                pass
            # ensure non-negative
            a = np.maximum(a, 0)
            # project to simplex sum target_sum
            a = _project_to_simplex(a, target_sum)
            # symmetrize optionally
            if rng.random() < 0.2:
                a = (a + a[::-1]) / 2.0
                a = _project_to_simplex(a, target_sum)
            # evaluate via max convolution
            val = _max_convolution_fft(a)
            if val < best_score:
                best_score = val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] new best (len={n}) score={best_score:.6f}")
        # after random search, apply refinement with gradient descent if time left.
        if time.time() < deadline - 30:
            # apply annealed gradient descent
            sub_dead = time.time() + min(12.0, deadline - time.time() - 5)
            a_opt, val_opt = _annealed_gradient_descent(best_seq, sub_dead, target_sum,
                max_beta=1e6, step_factor=0.8, gamma=0.97)
            if val_opt < best_score:
                best_score = val_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.2f}] refined best score={best_score:.6f}")
        # maybe mass transfer after refinement
        if time.time() < deadline - 10:
            mt_dead = time.time() + 5.0
            a_mt, val_mt = _mass_transfer_simulated_annealing(best_seq, mt_dead, target_sum,
                                                             start_temp=0.3, end_temp=1e-5, low_frac=0.07)
            if val_mt < best_score:
                best_score = val_mt
                best_seq = a_mt.copy()
                print(f"[{time.time():.2f}] MT best score={best_score:.6f}")

        # early termination if time running low
        if time.time() >= deadline - 5:
            break

    # final trimming zeros
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-13)[0]
    if nz.size:
        arr = arr[:nz[-1]+1]
    final_list = arr.tolist()
    try:
        val= evaluate_sequence(final_list) # compute official score
        print(f"[{time.time():.2f}] Final scoring = {val:.6f}")
    except Exception as e:
        print("Eval error:", e)
    globals()["height_sequence_1"] = final_list
    return final_list

"""
Improved step‑function optimizer.

Strategy
--------
1. Choose a problem size `n` depending on the time budget.
2. Generate many candidate shapes (window functions, power‑law,
   exponentials, Gaussians, etc.).  For each shape:
   * project to the ℓ¹‑simplex with Σa = sqrt(2 n);
   * create a few deterministic orderings (original, alternating,
     even‑spread) that destroy symmetry and tend to lower the
     autocorrelation peak;
   * evaluate the maximal linear convolution using an FFT‑based
     routine.
3. Keep the best candidate seen so far.
4. Refine the best candidate with a few seconds of soft‑max
   (annealed) gradient descent and a short simulated‑annealing
   mass‑transfer pass.
5. Trim trailing zeros, store the result in a global variable for
   warm‑starts, and return it.

The objective after normalisation is simply
    max_k (a * a)[k]
so the code works directly with this quantity for speed.
"""
import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT based convolution utilities (fast max)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = s, x ≥ 0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Reordering helpers – break symmetry
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately to the left / right ends."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Place values with a step that is coprime to n (good spread)."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent – fine tuning
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 8e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Projected GD on the soft‑max approximation of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        # soft‑max distribution over the convolution entries
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

        # gradient of the smooth approximation
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer – direct reduction of the peak
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)                     # current full convolution

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))        # index of the current biggest entry
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]      # contributions to the peak
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    return np.kaiser(n, beta)


def _blackman_window(n: int) -> np.ndarray:
    return np.blackman(n)


def _hamming_window(n: int) -> np.ndarray:
    return np.hamming(n)


def _bartlett_window(n: int) -> np.ndarray:
    return np.bartlett(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    # np.tukey exists from NumPy 1.20; fallback to linear blend if missing
    try:
        return np.tukey(n, alpha)
    except AttributeError:
        # simple approximation: Hann for alpha=0, rectangular for alpha=1
        if alpha == 0:
            return np.hanning(n)
        if alpha == 1:
            return np.ones(n)
        # linear combination (rough)
        hann = np.hanning(n)
        rect = np.ones(n)
        return (1 - alpha) * hann + alpha * rect


def _gaussian_window(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.exp(-0.5 * ((x - center) / (sigma * n)) ** 2)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _sin_half(n: int) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    return np.sin(np.pi * x)

    windows.append((_sin_half, (n,)))

    # --------------------------------------------------------------
    # Evaluate each window under a few deterministic reorderings.
    # --------------------------------------------------------------
    orderings = [lambda x: x, _alternating_order, _even_spread_order]

    for gen, params in windows:
        if time.time() >= deadline - 30.0:
            break
        raw = np.asarray(gen(*params), dtype=np.float64)
        # clip possible negatives (e.g., Blackman may dip slightly below zero)
        raw = np.clip(raw, 0.0, None)
        # project to required ℓ¹‑simplex
        a = _project_to_simplex(raw, target_sum)

        for reorder in orderings:
            cand = reorder(a)
            cand = _project_to_simplex(cand, target_sum)   # re‑project after reordering
            val = _max_convolution_fft(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] New best (window) score={best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Random simple‑x simplex seeds + a cheap jittered search.
    # --------------------------------------------------------------
    random_trials = 0
    while time.time() < deadline - 20.0:
        random_trials += 1
        # draw a random point on the simplex (Dirichlet(1,…,1))
        a = rng.random(n)
        a = _project_to_simplex(a, target_sum)

        # with 30 % probability apply one of the orderings
        if rng.random() < 0.3:
            a = _alternating_order(a) if rng.random() < 0.5 else _even_spread_order(a)
            a = _project_to_simplex(a, target_sum)

        # tiny jitter to explore neighbourhood
        jitter = rng.normal(scale=target_sum * 0.002, size=n)
        a = np.maximum(a + jitter, 0.0)
        a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Random trial {random_trials} new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Fine‑tune the current best using annealed GD (few seconds).
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        sub_dead = time.time() + min(12.0, deadline - time.time() - 4.0)
        a_opt, val_opt = _annealed_gradient_descent(
            best_seq, sub_dead, target_sum,
            max_beta=1e6, step_factor=0.85, gamma=0.97)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD refinement best={best_val:.6f}")

    # --------------------------------------------------------------
    # 4) One short simulated‑annealing mass‑transfer pass.
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6.0:
        mt_dead = time.time() + min(5.0, deadline - time.time() - 2.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            best_seq, mt_dead, target_sum,
            start_temp=0.3, end_temp=1e-5, low_frac=0.07)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] MT final best={best_val:.6f}")

    # --------------------------------------------------------------
    # 5) Trim trailing near‑zero entries and return.
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-13)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        official = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Official evaluation = {official:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # expose for warm‑starts in a subsequent run
    globals()["height_sequence_1"] = final_seq
    return final_seq
