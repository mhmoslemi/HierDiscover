"""
Improved step‑function search.

The core idea is a *smooth max* approximation: replace the true
max‑convolution entry by a softmax (log‑sum‑exp) with a temperature β.
For a fixed β we minimise the smooth objective by projected gradient
descent (the gradient is obtained by a single extra FFT).  β is
gradually increased so the optimisation approaches the original
non‑smooth problem.  The resulting vector is finally refined by an
aggressive mass‑transfer simulated‑annealing stage (the same O(n)
updates as the reference implementation).  Several random restarts
(and an optional warm‑start) are tried within the time budget.

All helper functions are defined at top level; the entry point is
`propose_candidate`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using an FFT of power‑of‑2 size."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score_from_selfconv(b: np.ndarray, n: int, target_sum: float) -> float:
    """Objective 2·n·max(b) / (target_sum²) – b must be a*a."""
    max_b = b.max()
    return float(2.0 * n * max_b / (target_sum * target_sum))

# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       beta_start: float = 0.1,
                       beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution entry.
    The approximation is  (1/β)·log Σ exp(β·(a*a)_i).  After each step
    the vector is projected onto the simplex Σa = target_sum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # initial objective (exact max) – used for bookkeeping only
    best_val = _score_from_selfconv(_conv_self_fft(a), n, target_sum)

    beta = beta_start
    step0 = target_sum * 0.8          # raw step size before normalisation
    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) current self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        b = _conv_self_fft(a)                     # length 2n‑1
        max_b = b.max()
        # exact objective (for best‑track)
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth β={beta:.2f} it={it:5d} best={best_val:.6f}")

        # log‑sum‑exp weights (numerically stable)
        b_shift = b - max_b                       # max(b_shift) == 0
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                              # softmax over all 2n‑1 positions

        # --------------------------------------------------------------
        # 2) gradient = 2·(w ∗ a_rev)  (central slice)
        # --------------------------------------------------------------
        a_rev = a[::-1]
        conv_wa = _conv_fft(w, a_rev)            # length 3n‑2
        g = 2.0 * conv_wa[n - 1: n - 1 + n]       # central n entries

        # --------------------------------------------------------------
        # 3) step size – scaled by gradient norm, decays with √it
        # --------------------------------------------------------------
        norm_g = np.linalg.norm(g)
        if norm_g == 0.0:
            break
        eta = step0 / np.sqrt(it + 1.0) * (target_sum / norm_g)

        # gradient step + projection onto the simplex
        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # --------------------------------------------------------------
        # 4) slowly increase β (more focus on the true max)
        # --------------------------------------------------------------
        if it % 2000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.5)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves tiny amounts of mass from ‘bad’
    indices (large contribution to the current peak) to ‘good’
    indices (small height).  The convolution is updated in O(n) for
    each move, exactly as in the reference implementation.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)                     # exact self‑convolution a*a
    cur_val = _score_from_selfconv(b, n, target_sum)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature – fraction of the current energy
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # square‑root cooling

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak (a[i]·a[k‑i])
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum * target_sum)

        # Metropolis acceptance based on the *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score_from_selfconv(b, n, target_sum)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # normalisation used in the reference

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run the optimisation phases for every start, respecting time
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx + 1} -----")

        # ---- Phase 1 : smooth‑max projected gradient descent ------------
        sub_deadline = time.time() + 0.35 * (deadline - time.time())
        a_smooth, val_smooth = smooth_max_descent(
            init, target_sum, sub_deadline,
            beta_start=0.1, beta_max=30.0
        )
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ---- Phase 2 : aggressive annealing ---------------------------
        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(
                a_smooth, target_sum, deadline
            )
            if val_ann < best_score:
                best_score = val_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – a decreasing power‑law
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)

    # trim trailing zeros for a slightly shorter representation
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final evaluation (outside of the optimisation loop)
    try:
        final_val = evaluate_sequence(final_seq)    # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:                      # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm start in a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
