def _grad_and_conv(a, epsilon):
    # compute convolution
    conv = _conv_fft(a)
    maxc = conv.max()
    # compute normalized weights
    # avoid overflow
    # compute shift = conv - maxc
    shift = conv - maxc
    # compute exponent, careful if epsilon small
    w_exp = np.exp(shift / epsilon)
    sum_w = w_exp.sum()
    w_norm = w_exp / sum_w
    # gradient
    a_rev = a[::-1]
    size = 1 << ((len(a_rev) + len(w_norm) - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w_norm, size)
    grad_full = np.fft.irfft(fa * fw, size)[: len(a_rev) + len(w_norm) - 1]
    # slice
    n = a.shape[0]
    grad = 2.0 * grad_full[n-1 : n-1 + n]
    return grad, maxc, conv

"""
Hybrid smoothed‑gradient descent optimizer for the step‑function problem.

The algorithm works as follows:

1. **Problem size** – pick n≈8000 (larger n gives a richer search space).
2. **Seeding** – create a diverse pool of feasible sequences (power‑law,
   exponential, sinusoidal, random Dirichlet‑like and the previous best).
3. **Random restarts** – for each seed run a projected gradient descent on a
   log‑sum‑exp smoothed version of the max‑convolution.  The gradient is
   obtained efficiently with FFTs and the iterate is projected onto the
   simplex {a_i≥0, Σa_i=√(2n)}.
4. **Continuation** – the smoothing parameter ε is halved repeatedly
   (large ε gives a smooth landscape, small ε approaches the true max).
5. **Polishing** – a short deterministic mass‑transfer routine similar to the
   one used previously refines the best candidate.
6. **Reordering** – a few simple permutations (original, alternating,
   even‑spread) are tested; the ordering with the smallest max‑convolution
   is kept.
7. **Trim & return** – trailing near‑zero entries are removed, the result is
   stored globally (for the next run) and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then normalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then normalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _sinusoid_sequence(n: int, target_sum: float, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/(2n))^power  (always non‑negative)."""
    x = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / (2.0 * n))
    a = np.power(x, power)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient computation
# ----------------------------------------------------------------------


def _grad_and_maxconv(a: np.ndarray, epsilon: float):
    """
    Returns (gradient, max_convolution, full_convolution) of the
    log‑sum‑exp smoothed max‑convolution at point `a` with smoothing
    parameter `epsilon`.
    """
    conv = _conv_fft(a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    shift = conv - maxc                     # ≤ 0
    w_exp = np.exp(shift / epsilon)        # max weight = 1
    sum_w = w_exp.sum()
    w_norm = w_exp / sum_w                  # ∑ w_norm = 1

    # gradient = 2 * (a_rev * w_norm)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    size = 1 << ((len(a_rev) + len(w_norm) - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w_norm, size)
    grad_full = np.fft.irfft(fa * fw, size)[: len(a_rev) + len(w_norm) - 1]
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, maxc, conv


# ----------------------------------------------------------------------
# Smoothed gradient descent (continuation on ε)
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    target_sum: float,
                    deadline: float,
                    init_eps_factor: float = 0.3,
                    min_eps: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp smooth approximation of
    max‑convolution.  ε is halved repeatedly (continuation) until it reaches
    `min_eps` or time runs out.
    Returns the best sequence seen and its (unsmoothed) max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    # initial smoothing level
    eps = max(_max_convolution_fft(a) * init_eps_factor, 0.01)

    while time.time() < deadline and eps > min_eps:
        # a modest number of inner iterations per ε value
        inner_iters = 200
        for _ in range(inner_iters):
            if time.time() >= deadline:
                break
            grad, maxc, _ = _grad_and_maxconv(a, eps)
            norm = np.linalg.norm(grad)
            if norm == 0.0:
                break
            # step size ~ ε / ‖grad‖ (empirically stable)
            step = 0.5 * eps / norm
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new, target_sum)

            cur_score = _max_convolution_fft(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
            a = a_new
        eps *= 0.5
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing (identical to the reference)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) equals max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function challenge.

    It builds a rich set of seeds, runs several random restarts of a
    smoothed‑gradient descent (with continuation on the smoothing parameter),
    finishes with a deterministic mass‑transfer polishing, tries a few
    reorderings and returns the best sequence found before the time budget
    expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ---------------------------------------------------------------
    # Choose problem size (larger n gives a richer landscape)
    # ---------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ---------------------------------------------------------------
    # Build a pool of diverse seeds
    # ---------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 7):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 5):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # sinusoidal shapes
    for power in [0.5, 1.0, 2.0]:
        seeds.append(_sinusoid_sequence(n, target_sum, power))

    # random Dirichlet‑like points
    for _ in range(6):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best (if available)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.array(prev, dtype=np.float64)
        if prev_arr.shape[0] != n:

        seeds.append(prev_arr)

    # Slight jitter to increase diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # ---------------------------------------------------------------
    # Pick the best seed as a starting point for the first run
    # ---------------------------------------------------------------
    best_score = np.inf
    best_seq = None
    for seq in seeds:
        sc = _max_convolution_fft(seq)

            best_score = sc
            best_seq = seq.copy()
    assert best_seq is not None

    # ---------------------------------------------------------------
    # Random‑restart smoothed gradient descent
    # ---------------------------------------------------------------
    # Reserve a chunk of the budget for polishing and finalisation
    reserve = 20.0
    time_for_descent = max(0.0, deadline - time.time() - reserve)

    # We will perform at most 4 restarts (or fewer if time runs out)
    max_restarts = 4
    restarts_done = 0

    while time.time() < deadline - reserve and restarts_done < max_restarts:
        # choose a seed (cycling through the list)
        init = seeds[restarts_done % len(seeds)]
        # add a tiny random perturbation
        pert = rng.normal(scale=target_sum * 0.005, size=n)
        a0 = init + pert
        a0 = np.maximum(a0, 0.0)
        a0 = _project_to_simplex(a0, target_sum)

        # allocate a modest slice of the remaining time

            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Restart {restarts_done}: new best = {best_score:.6f}")

    # ---------------------------------------------------------------

            print(f"[{time.time():.2f}] After mass‑transfer best = {best_score:.6f}")
