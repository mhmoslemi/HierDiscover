"""
Improved search for a step‑function with a lower maximal self‑convolution.
The algorithm follows a multi‑phase approach:

0️⃣  Initialise a diverse population of candidates (parametric families,
    Dirichlet draws and the previous best if available) and resize them to a
    problem size proportional to the time budget.

1️⃣  Fast Adam optimisation on a soft‑max approximation of the max‑convolution.
    The β‑parameter of the soft‑max is increased exponentially up to a very
    large value so that the gradient focuses on the true maximum.

2️⃣  Exact sub‑gradient descent (the “true” max‑convolution) with a line‑search.

3️⃣  Deterministic mass‑transfer (the exact DMT step used in the reference).

4️⃣  Randomised mass‑transfer (global exchanges of mass).

5️⃣  Simulated‑annealing fine‑tuning that moves a small amount of mass from a
    donor (large positive gradient) to a receiver (large negative gradient).

All phases respect the simplex constraint Σa = √(2·n) and a_i ≥ 0.
The final sequence is trimmed of trailing near‑zero entries and returned as
a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of arbitrary length vectors x and y using FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for quick initialisation
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Centered Gaussian, clipped at the ends, normalised to the target sum."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑distribution shaped sequence on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a += 1e-15               # avoid exact zero for the Dirichlet‑style draw
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """∇_a  ( softmax_β( a*a ) )  – the gradient of the smooth max."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_beta: float = 1e7,
                   step_factor: float = 0.7,
                   gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Adam optimisation of the soft‑max approximated max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-12
    it = 0
    beta_inc_iter = 200

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        # adaptive step size (scaled to the simplex radius)
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat plateaux
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        gap = max_val - second_max
        if gap < 1e-14:
            break

        # pick a random maximal index
        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient for the selected max entry
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            # tiny random move to escape degeneracy
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        # step size from classic sub‑gradient theory
        step = 0.9 * gap / (norm2 + 1e-12)
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # backtracking line search (up to 6 reductions)
        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _max_convolution_fft(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} best={best_score:.6f}")
                break
            step *= 0.5
        else:
            # no improvement – jitter
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.0005, size=n), target_sum)

        it += 1

    return best_a, best_score

# Deterministic exact mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]                      # first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                # contributions to the maximal entry
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = position that contributes most to the current max
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = position that contributes least (but not the donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} best={best_score:.6f}")

        it += 1

    return best_a, best_score

def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 0.5:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score

# Simulated‑annealing fine‑tuning (mass moves guided by the soft‑max gradient)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    a_cur = a0.copy()
    cur_score = _max_convolution_fft(a_cur)
    best_a = a_cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a_cur.size
    it = 0

    # start temperature – roughly the typical improvement size we hope to gain
    T0 = 0.02
    # exponential cooling factor (chosen so that T → ~0 at the end)
    tau = max(1.0, (deadline - time.time()) / 50.0)

    while time.time() < deadline:
        # compute a soft‑max gradient for the *current* vector
        grad, _ = _gradient_softmax(a_cur, beta=5e5)   # very sharp – essentially the true max gradient
        # donor: large positive gradient (increasing it would raise the max)
        donor = np.argmax(grad)
        # receiver: large negative gradient (increasing it would lower the max)
        receiver = np.argmin(grad)

        # prevent degenerate moves
        if donor == receiver:
            continue
        max_delta = a_cur[donor] * 0.15
        if max_delta <= 0.0:
            # jitter a little and continue
            a_cur = _project_to_simplex(a_cur + rng.normal(scale=target_sum * 0.0005, size=n), target_sum)
            continue

        # temperature based step size
        t = T0 * np.exp(-it / tau)
        delta = t * max_delta * rng.random()

        a_new = a_cur.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _max_convolution_fft(a_new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(t, 1e-12)):
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] Anneal it={it:5d} T={t:.4f} best={best_score:.6f}")

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """

    self‑convolution (the evaluation score). Returns the best sequence found
    before the time budget expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives more freedom, but we keep it reasonable.
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 16000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)                    # Σa = √(2·n)

    # --------------------------------------------------------------
    # Phase 0 – Gather a pool of diverse seeds.
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # (0a) previous best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _max_convolution_fft(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()

    # (0b) parametric families
    families = []
    p_grid = np.linspace(0.25, 0.85, 28)
    q_grid = np.linspace(0.0, 6.0, 28)

    for p in p_grid:
        families.append(_power_law_sequence(n, float(p), target_sum))
    for p in p_grid:
        for q in q_grid:
            families.append(_exp_power_law_sequence(n, float(p), float(q), target_sum))
    families.append(_sin_sequence(n, target_sum))
    for sigma in np.linspace(n / 15.0, n / 4.0, 7):
        families.append(_gaussian_sequence(n, sigma, target_sum))
    for a in np.linspace(0.7, 2.5, 9):
        for b in np.linspace(0.7, 2.5, 9):
            families.append(_beta_sequence(n, a, b, target_sum))

    # evaluate all seeds (quickly)
    for seq in families:
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (0c) a few Dirichlet draws – they naturally live on the simplex.
    for _ in range(12):
        alpha = rng.random(n) * 0.8 + 0.1    # random concentration vector
        seq = rng.dirichlet(alpha) * target_sum
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    print(f"[{time.time():.2f}] After seed generation best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation with gradual β‑increase.
    # --------------------------------------------------------------
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < phase_a_deadline - 4.0:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        start_vec = best_seq + jitter
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_optimize(start_vec,
                                        target_sum,
                                        phase_a_deadline,
                                        max_beta=5e6,
                                        step_factor=0.7,
                                        gamma=0.995)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam phase improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent.
    # --------------------------------------------------------------
    phase_b_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         phase_b_deadline,
                                         max_iter=8000)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Sub‑gradient improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (core refinement).
    # --------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT improved best={best_score:.6f}")

        # Occasionally run a global random exchange to escape plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best={best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # --------------------------------------------------------------
    # Phase D – Simulated‑annealing polishing.
    # --------------------------------------------------------------
    if time.time() + 6.0 < deadline:
        a_ann, val_ann = _simulated_annealing(best_seq,
                                              target_sum,
                                              deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] Annealing improved best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase E – Final high‑β Adam polish (optional, if time permits).
    # --------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_optimize(best_seq,
                                        target_sum,
                                        deadline,
                                        max_beta=1e7,
                                        step_factor=0.6,
                                        gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and final safety check.
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (for our own logging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
```

This implementation enlarges the problem size when the time budget permits, explores a richer set of initial shapes (including Beta‑distribution families and Dirichlet draws), runs a long Adam optimisation with a very large soft‑max temperature, follows up with exact sub‑gradient and deterministic mass‑transfer steps, adds a global random mass‐transfer heuristic, and finally applies a simulated‑annealing fine‑tuning that moves mass along the true max‑gradient direction.  All operations respect the simplex constraint and stay within the 1000‑second limit, aiming to push the maximal self‑convolution below the previous record of 1.507394.
