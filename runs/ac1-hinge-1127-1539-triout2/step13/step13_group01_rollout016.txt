    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()

    b = _conv_fft(a)
    T_start = b.max() * 1.0  # maybe 1.0 times
    T_end = 1e-4
    eta_start = 0.5
    eta_end = 0.01
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        T = T_start * (1 - frac) + T_end * frac
        eta = eta_start * (1 - frac) + eta_end * frac

        b = _conv_fft(a)
        # compute softmax weights

        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / b.size
        else:

"""
Improved search for a low‑score non‑negative step function.

The algorithm consists of three stages:

1. Multi‑peak sub‑gradient descent on the exact max‑convolution entry.
   The sub‑gradient is averaged over the K largest convolution peaks,
   followed by Euclidean projection onto the simplex {x≥0, Σx = √(2·n)}.

2. Smoothed exponentiated‑gradient descent.
   The max‑convolution is approximated by a temperature‑controlled
   soft‑max (log‑sum‑exp) which yields a smooth objective.
   An exponentiated‑gradient step keeps the variables non‑negative;
   after the step a simple renormalisation restores the required sum.

3. Aggressive simulated‑annealing mass‑transfer refinement.
   Tiny amounts of mass are moved from “bad’’ indices that contribute
   strongly to the current peak toward “good’’ low‑height indices.
   The convolution is updated in O(n) per move and a Metropolis
   acceptance rule with a slowly decaying temperature is used.

Multiple random restarts (including a warm‑start) are performed within
the allotted time budget; the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – multi‑peak projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_multi(a0: np.ndarray, target_sum: float,
                     deadline: float, top_k: int = 3):
    """Sub‑gradient descent averaging over the top‑K convolution peaks."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        max_val = b.max()

        # select K indices with largest convolution values
        if top_k >= b.size:
            active = np.arange(b.size)
        else:
            active = np.argpartition(-b, top_k)[:top_k]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution a[k-idx] appears in the derivative
            grad[idx] += a[k - idx]

        # average over the selected peaks
        grad *= 2.0 / max(1, len(active))

        # step size – linearly decreasing from 2 → 0.01
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        # gradient step and projection onto the simplex
        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad multi improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------


def _correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear correlation c[i] = Σ_j a[j]·w[i+j] for i=0..len(a)-1."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    C = A * np.conj(W)
    c = np.fft.irfft(C, n=L)
    return c[:n]


def exp_gradient_descent(a0: np.ndarray, target_sum: float,
                        deadline: float):
    """Exponentiated‑gradient descent on a temperature‑smoothed max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _score(a)

    start = time.time()

    # initialise temperature from the current peak
    b = _conv_fft(a)
    T_start = max(b.max(), 1e-8)
    T_end = 1e-4
    eta_start = 0.5
    eta_end = 0.01

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total

        # linearly decaying temperature and learning rate
        T = T_start * (1.0 - frac) + T_end * frac
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # current convolution and soft‑max weights
        b = _conv_fft(a)
        maxb = b.max()
        # stabilised soft‑max
        w = np.exp((b - maxb) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # gradient of the smoothed objective: 2·corr(a, w)
        g = 2.0 * _correlation(a, w)

        # exponential update (clamped exponent to avoid overflow)
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -30.0, 30.0)
        a = a * np.exp(exp_arg)

        # renormalise to keep the prescribed sum
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full(n, target_sum / n)

        cur_val = b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exp‑grad improved best={best_val:.6f}")

    return best_a, best_val

# Phase 3 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from indices that
    heavily contribute to the current peak towards indices with low height.
    The convolution update is performed in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact self‑convolution
    cur_val = b.max()
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 3.0                     # more aggressive initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling (keeps the system hot longer)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak (a[i]·a[k‑i])
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.15, target_sum * 0.02)  # more aggressive
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = b_new.max()

        # Metropolis acceptance based on the *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if it % 2500 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)          # the usual normalisation

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run the three optimisation phases for every start, respecting time
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx+1} -----")

        # ----- Phase 1 : multi‑peak projected sub‑gradient descent ----------
        remaining = deadline - time.time()
        sub_deadline = time.time() + 0.15 * remaining
        a_sub, score_sub = subgradient_multi(init, target_sum, sub_deadline, top_k=3)
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New global best after sub‑grad: {best_score:.6f}")

        # ----- Phase 2 : smoothed exponentiated‑gradient descent ----------
        if time.time() < deadline:
            remaining = deadline - time.time()
            exp_deadline = time.time() + 0.15 * remaining
            a_exp, score_exp = exp_gradient_descent(a_sub, target_sum, exp_deadline)
            if score_exp < best_score:
                best_score = score_exp
                best_seq = a_exp.copy()
                print(f"[{time.time():.2f}] New global best after exp‑grad: {best_score:.6f}")

        # ----- Phase 3 : aggressive annealing ----------------------------
        if time.time() < deadline:
            a_ann, score_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
            if score_ann < best_score:
                best_score = score_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple decreasing power‑law
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq

Before final, let's double-check correctness of correlation gradient formula.

The soft max weighted gradient of max(b) approximated by w vector is indeed gradient of sum_k w_k * b_k = sum_k w_k * sum_i a_i a_{k-i} = sum_i a_i * sum_k w_k * a_{k-i} = sum_i a_i * (correlation of w and a at offset i). Derivative w.r.t a_i = sum_k w_k * a_{k-i} + sum_k w_k * a_{k-i} = 2 * correlation_i because each term appears twice except when i = k-i (k=2i) but that's accounted by same formula. So gradient = 2 * correlation_i, consistent.

"""
Improved search for a low‑score non‑negative step function.

The algorithm consists of three stages:

1. **Multi‑peak projected sub‑gradient descent** – the sub‑gradient is
   averaged over the K largest convolution peaks, then the iterate is
   projected onto the simplex {x≥0, Σx = √(2·n)}.

2. **Smoothed exponentiated‑gradient descent** – the max‑convolution is
   approximated by a temperature‑controlled soft‑max (log‑sum‑exp).  The
   gradient of the smooth objective is 2·corr(a, w) where *w* are the
   soft‑max weights.  Updates are performed with an exponentiated‑gradient
   step (keeping positivity) followed by a simple renormalisation.

3. **Aggressive simulated‑annealing mass‑transfer** – tiny amounts of mass
   are moved from indices that contribute heavily to the current peak
   towards low‑height indices.  The convolution is updated in O(n) per
   move, and a Metropolis rule with a slowly decaying temperature is
   used.

Multiple random restarts (including a warm‑start) are performed within the
time budget; the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – multi‑peak projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_multi(a0: np.ndarray, target_sum: float,
                     deadline: float, top_k: int = 3):
    """Sub‑gradient descent averaging over the top‑K convolution peaks."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        max_val = b.max()

        # indices of the K largest convolution entries
        if top_k >= b.size:
            active = np.arange(b.size)
        else:
            active = np.argpartition(-b, top_k)[:top_k]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += a[k - idx]          # a contribution to the derivative

        grad *= 2.0 / max(1, len(active))

        # step size – linearly decreasing from 2 → 0.01
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad multi improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------


def exp_gradient_descent(a0: np.ndarray, target_sum: float,
                        deadline: float):
    """
    Exponentiated‑gradient descent on a temperature‑smoothed max‑convolution.
    The soft‑max weights w_k = exp(b_k/T) / Σ exp(b/T) give a smooth objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()

    # one FFT length that works for both convolution (2n‑1) and correlation (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # initialise temperature from the current peak
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    T_start = max(conv_full.max(), 1e-8)
    T_end = 1e-4
    eta_start = 0.5
    eta_end = 0.01

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total

        T = T_start * (1.0 - frac) + T_end * frac
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # recompute FFT of the current a
        A = np.fft.rfft(a, n=L)

        # convolution a*a (full length 2n‑1)
        conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        maxb = conv_full.max()

        # soft‑max weights, stabilised by subtraction of the maximum
        w = np.exp((conv_full - maxb) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(conv_full) / conv_full.size
        else:
            w /= w_sum

        # FFT of the weight vector (same length L)
        W = np.fft.rfft(w, n=L)

        # correlation c[i] = Σ_j a[j]·w[i+j], i = 0..n‑1
        C = A * np.conj(W)
        c = np.fft.irfft(C, n=L)
        g = 2.0 * c[:n]                     # gradient of the smoothed objective

        # exponentiated update (clamp exponent to avoid overflow)
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -30.0, 30.0)
        a = a * np.exp(exp_arg)

        # renormalise to keep the required sum
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full(n, target_sum / n)

        cur_val = maxb
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exp‑grad improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing mass transfer.
    Moves a small amount of mass from a “bad’’ index contributing heavily to
    the current peak towards a “good’’ low‑height index.  The convolution is
    updated in O(n) per move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)               # exact self‑convolution
    cur_val = b.max()
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 3.0             # more aggressive initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling schedule
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor – weighted by contribution a[i]·a[k‑i] to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour low‑height indices
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # transfer amount, scaled by the current temperature
        max_delta = min(a[donor] * 0.15, target_sum * 0.02)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part (2·a·Δ)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # occasional full recomputation to keep numerical errors in check
        if it % 2500 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = b.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5            # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)                # the usual normalisation

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run the three optimisation phases for each start
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx+1} -----")

        # Phase 1 – multi‑peak sub‑gradient descent
        remaining = deadline - time.time()
        sub_deadline = time.time() + 0.15 * remaining
        a_sub, score_sub = subgradient_multi(
            init, target_sum, sub_deadline, top_k=3
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New global best after sub‑grad: {best_score:.6f}")

        # Phase 2 – exponentiated‑gradient descent

            exp_deadline = time.time() + 0.15 * remaining

                best_seq, target_sum, deadline
            )
            if score_ann < best_score:
                best_score = score_ann
                best_seq = a_ann.copy()
                print(
                    f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)
