def _random_phase_sequence(n, target_sum, rng):
    size = 1 << ((2 * n - 1).bit_length())
    # magnitude: set to sqrt(rand positive?), we want variable amplitude maybe random too.
    mag = np.ones(size//2+1)
    phase = rng.random(size//2+1) * 2 * np.pi
    # Ensure symmetry: built into rfft for real sequence (complex conj automatically)
    freq = mag * np.exp(1j * phase)
    a = np.fft.irfft(freq, n=size)[:n]
    # Ensure non-negative: shift
    a -= a.min()
    a = np.maximum(a, 0.0)
    # Project to simplex to sum to target_sum
    a = _project_to_simplex(a, target_sum)
    return a

"""
Hybrid optimizer for the step‑function minimisation problem.

The method builds a diverse pool of seeds (power‑law, exponential,
random simplex, random‑phase Fourier constructions and the previous best),
runs a relatively large projected Differential Evolution (DE) with three
mutation strategies, then performs an extended deterministic mass‑transfer
phase, a long sub‑gradient descent, and finally a quick random‑phase local
search.  Simple reorderings are tried and the best candidate is trimmed
and returned.

The whole routine respects the given time budget (default 1000 s) by
checking the wall‑clock frequently.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based – fast for the optimiser)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a)/(∑a)² (for ∑a=√(2n) this equals max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _random_phase_sequence(n: int, target_sum: float,
                           rng: np.random.Generator) -> np.ndarray:
    """
    Build a sequence from a random Fourier phase vector.
    The magnitude of all frequencies is set to 1, the phases are uniform.
    The resulting real signal is shifted to be non‑negative and projected
    onto the ℓ¹‑simplex.
    """
    size = 1 << ((2 * n - 1).bit_length())
    # random phases for the rfft output (size//2 + 1 entries)
    phases = rng.random(size // 2 + 1) * 2.0 * np.pi
    freq = np.exp(1j * phases)

    # enforce reality of the DC and Nyquist components
    freq[0] = rng.random()
    if size % 2 == 0:
        freq[-1] = rng.random()

    a = np.fft.irfft(freq, n=size)[:n]
    a -= a.min()                # make all entries ≥ 0
    a = np.maximum(a, 0.0)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]

        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 rng: np.random.Generator,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """
    DE loop with three mutation strategies:
    1. rand/1
    2. best/1
    3. current-to-best/1
    Each trial is projected onto the ℓ¹‑simplex.
    """
    n = population[0].shape[0]
    pop = [p.copy() for p in population]
    scores = [_score(ind) for ind in pop]
    best_idx = int(np.argmin(scores))
    best_seq = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(pop)):
            # three distinct mates different from i
            idxs = list(range(len(pop)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            # pick a mutation scheme uniformly
            scheme = rng.integers(3)
            if scheme == 0:          # rand/1
                mutant = pop[a] + F * (pop[b] - pop[c])
            elif scheme == 1:        # best/1
                mutant = best_seq + F * (pop[b] - pop[c])
            else:                    # current-to-best/1
                mutant = pop[i] + F * (best_seq - pop[i]) + F * (pop[b] - pop[c])

            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # early exit when only a few seconds remain
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser that respects the given time budget.
    Returns a non‑negative step‑function (list of floats) with a low
    evaluation score.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a slightly larger n gives a richer search space
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 9000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.1, 1.5, 12):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.03, 10):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # pure random points on the simplex
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # random‑phase Fourier seeds
    for _ in range(10):
        seeds.append(_random_phase_sequence(n, target_sum, rng))

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter every seed a little to increase diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # ------------------------------------------------------------------
    # Initialise a modest DE population
    # ------------------------------------------------------------------
    pop_size = max(30, min(80, n // 200))   # 30‑80 individuals
    population = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # ------------------------------------------------------------------
    # DE phase – about 60 % of the total budget
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.60 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline, rng)
    print(f"[{time.time():.2f}] DE finished, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer polishing – up to ~20 % of the budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        mt_deadline = time.time() + min(200.0, deadline - time.time() - 12.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient descent – use the remaining time before the final phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 7.0:
        sg_deadline = time.time() + min(200.0, deadline - time.time() - 7.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_deadline, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Quick random‑phase local search (remaining seconds)
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:
        # generate a fresh random‑phase candidate
        cand = _random_phase_sequence(n, target_sum, rng)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

        # jitter the current best a little and re‑project
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        cand2 = best_seq + jitter
        cand2 = np.maximum(cand2, 0.0)
        cand2 = _project_to_simplex(cand2, target_sum)
        sc2 = _score(cand2)
        if sc2 < best_score:
            best_score, best_seq = sc2, cand2.copy()

    # ------------------------------------------------------------------
    # Try a handful of reorderings that often lower the max‑convolution
    # ------------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        np.sort(best_seq)[::-1],  # descending
        np.sort(best_seq)         # ascending
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store for the next run and report the final evaluation
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
