def anneal(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 1.0   # initial temperature
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))  # sqrt schedule

        # find max peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contributions for each index
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)] if contrib.sum() > 0 else rng.choice(idx)

        # receiver index - choose smallest a
        receiver = int(np.argmin(a))

        if donor == receiver:
            # pick different
            receiver = int(np.argsort(a)[1])  # second smallest
            if donor == receiver:
                # skip move
                continue

        # propose delta
        max_delta = min(a[donor] * 0.2, target_sum * 0.05)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # update convolution exactly
        b_new = b.copy()
        # linear part
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum ** 2))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal best={best_val:.6f}")

        if it % 5000 == 0:
            # recompute from scratch
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val

"""Optimizes a non‑negative step function to minimise 2·n·max(a*a)/∑a².

The search combines many random restarts, a projected sub‑gradient
descent on the active convolution entry and a long simulated‑annealing
mass‑transfer phase.  The annealing uses an exact O(n) update of the
full self‑convolution, a square‑root cooling schedule and a deterministic
choice of donor (largest contribution to the current peak) and receiver
(smallest height).  Several simple parametric initialisations (uniform,
ramps, power‑laws) are tried in addition to the warm‑start stored in
``height_sequence_1``.  The routine respects the supplied time budget
and returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """Projected (momentum) sub‑gradient descent on the entry attaining the max convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        # step size decreasing as 2/√(t+1)
        eta = 2.0 / np.sqrt(it + 1)

        v = beta * v - eta * grad
        a = a + v
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            # re‑initialise the whole vector if it collapsed to zero
            a = rng.random(n)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:6d} score={best_val:.6f}")

        # occasional jitter to avoid stagnation
        if it % 1000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0:
                a *= (target_sum / s)

        it += 1

    return best_a, best_val


def _anneal(a0: np.ndarray, target_sum: float, deadline: float):
    """Aggressive simulated‑annealing mass‑transfer with exact O(n) convolution update."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = float(2.0 * n * b.max() / (target_sum ** 2))
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 1.0          # initial temperature proportional to current energy

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # square‑root cooling

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        donor = idx[np.argmax(contrib)] if contrib.sum() > 0 else rng.choice(idx)

        # receiver – the index with smallest height (deterministic, ties broken arbitrarily)
        receiver = int(np.argmin(a))
        if donor == receiver:
            # pick the second‑smallest element instead
            order = np.argsort(a)
            receiver = order[1] if order[0] == donor else order[0]

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.2, target_sum * 0.05)   # more aggressive than the baseline
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum ** 2))

        # Metropolis acceptance based on the current energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = float(2.0 * n * b.max() / (target_sum ** 2))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

    init_candidates.append(uniform)

    ramp = np.arange(1, n + 1, dtype=np.float64)           # increasing linear ramp
    ramp = _project_to_simplex(ramp, target_sum)

    rev_ramp = np.arange(n, 0, -1, dtype=np.float64)       # decreasing linear ramp

    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq

"""Optimizes a non‑negative step function to minimise 2·n·max(a*a)/∑a².

The search combines many random restarts, a projected sub‑gradient
descent on the active convolution entry and a long simulated‑annealing
mass‑transfer phase.  The annealing uses an exact O(n) update of the
full self‑convolution, a square‑root cooling schedule and a deterministic
choice of donor (largest contribution to the current peak) and receiver
(smallest height).  Several simple parametric initialisations (uniform,
ramps, power‑laws) are tried in addition to the warm‑start stored in
``height_sequence_1``.  The routine respects the supplied time budget
and returns the best sequence found as a plain list of floats.
"""

import time

# Low‑level utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w
