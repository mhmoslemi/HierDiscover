def _lp_step(f, M):
    n = f.shape[0]
    # Build sparse matrix A_ub.
    offsets = np.arange(n)  # 0..n-1
    # For each offset d, create an array of length n values = f[d].
    data = np.vstack([np.full(n, f[d]) for d in range(n)]).astype(np.float64)
    A = diags(data, offsets, shape=(2*n-1, n), format='csr')
    # b_ub array:
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # objective: minimize -sum x => c = -np.ones(n)
    c = -np.ones(n, dtype=np.float64)
    # bounds for each variable: (0, None)
    bounds = [(0, None)] * n
    # Solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros?
        return np.zeros_like(f)
    b_opt = res.x
    # Clip tiny negative values due to numeric issues
    b_opt = np.maximum(b_opt, 0.0)
    return b_opt

def _project_to_simplex(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_softmax(a, beta):
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)
    # Convolve reversed a with p
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2*n-1]
    # gradient is 2 * conv_rev[n-1 : n-1+n]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, b

def _annealed_gradient_descent(a0, deadline, target_sum, max_beta=5e5, step_factor=0.5, gamma=0.99):
    a = _project_to_simplex(np.array(a0, dtype=float).copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)
        # jitter occasional
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it} β={beta:.1f} best={best_val:.6f}")
        if it % 400 == 0 and beta < max_beta:

"""
Alternating Linear‑Program + high‑β gradient polishing.

The optimiser:
1. Builds a modest‑size seed (≈2000 entries) and normalises it to the
   ℓ¹‑simplex Σ a_i = √(2·n).
2. Repeatedly solves the linear programme
       maximise Σ b_j   subject to  b_j≥0,
                               (a * b)_k ≤ M   for all k,
   where M = max(a*a).  The optimiser then mixes the current a with
   the LP solution (a convex combination) and projects back to the simplex.
   This is the classic alternating scheme from the problem statement.
3. The refined short vector is linearly interpolated to a larger length
   (8000‑12000) and, optionally, reordered by the “alternating order”
   trick to keep large entries far apart.
4. A momentum‑based soft‑max gradient descent with a very large
   inverse‑temperature (β≈10⁶) polishes the sequence.
5. A brief deterministic mass‑transfer pass extracts any remaining
   improvements.
6. Trailing zeros are trimmed and the final list is returned.
All phases respect the global time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP + sparse matrix)
# ----------------------------------------------------------------------
try:                                   # pragma: no‑cover
    from scipy.optimize import linprog
    from scipy.sparse import diags
    _SCIPY_OK = True
except Exception:                     # pragma: no‑cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation (preserves total mass)."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest left‑most, second largest right‑most, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    pl = True
    for v in sorted_vals:
        if pl:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        pl = not pl
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth max of convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear programme that given f and M finds a b with maximal ℓ¹‑mass
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """Solve   maximise Σ b_j   s.t.   (f*b)_k ≤ M , b≥0."""
    if not _SCIPY_OK:                     # pragma: no‑cover
        return np.zeros_like(f)

    n = f.shape[0]
    offsets = np.arange(n)                # 0,…,n‑1
    # each diagonal d gets the constant f[d] on all n entries
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)     # minimise –Σb_j  ⇔ maximise Σb_j
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return np.zeros_like(f)
    b_opt = np.maximum(res.x, 0.0)
    return b_opt


# ----------------------------------------------------------------------
# Alternating LP optimisation (the “state‑of‑the‑art” update)
# ----------------------------------------------------------------------


def _alternating_optimization(f0: np.ndarray,
                              target_sum: float,
                              t: float = 0.2,
                              iters: int = 6,
                              deadline: float = 1e30) -> tuple[np.ndarray, float]:
    """Repeatedly mix f with the LP solution (mixing factor t)."""
    a = _project_to_simplex(f0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    for _ in range(iters):
        if time.time() >= deadline:
            break
        M = _max_convolution_fft(a)
        b_opt = _lp_step(a, M)
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt * (target_sum / sum_b)
        a = (1.0 - t) * a + t * g
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Alternating LP iter score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1
    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose a modest size for the LP‑phase
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_small = 2000
    elif budget_s >= 500:
        n_small = 1500
    else:
        n_small = 1000
    target_small = np.sqrt(2.0 * n_small)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seed = _resize_sequence(prev, n_small)
        seed = _project_to_simplex(seed, target_small)
    else:
        seed = _power_law_sequence(n_small, p=0.6, target_sum=target_small)

    rng = np.random.default_rng()
    # a couple of random perturbations enrich the pool
    seeds = [seed]
    for _ in range(2):
        rand = rng.random(n_small)
        rand = rand * target_small / rand.sum()
        seeds.append(rand)

    # --------------------------------------------------------------
    # 3. Alternating LP optimisation (use up to ~30 % of the budget)
    # --------------------------------------------------------------
    lp_deadline = deadline - max(30.0, budget_s * 0.2)   # keep a healthy reserve
    best_seq = None
    best_val = np.inf

    for s in seeds:
        if time.time() >= lp_deadline:
            break
        seq, val = _alternating_optimization(s,
                                              target_small,
                                              t=0.25,
                                              iters=8,
                                              deadline=lp_deadline)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New LP‑best score={best_val:.6f}")

    if best_seq is None:                     # pragma: no‑cover
        best_seq = seed.copy()
        best_val = _score(best_seq)

    # --------------------------------------------------------------
    # 4. Upscale to a larger length (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120 and budget_s >= 800:
        n_final = 12000
    elif remaining > 80:
        n_final = 10000
    else:
        n_final = 8000

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings

    best_seq = seq_big

    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_val:

            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled n={n_final} start score={best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline - 6.0:

    # 6. Brief deterministic mass‑transfer (final nudging)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(3.0, deadline - time.time() - 0.5)
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,

            target_final,

            end_temp=1e-5,

        )

            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)

    final_seq = arr.tolist()

    # 8. Log final evaluation (optional)

        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
