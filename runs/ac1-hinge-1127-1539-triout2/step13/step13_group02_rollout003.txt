"""
Advanced step‑function search (alternative strategy).

The algorithm consists of three stages:

1. **Fast exploration** – many short sub‑gradient runs on a moderate
   size (≈2000–4000).  Seeds are drawn from power‑law, exponential and
   random Dirichlet families and from the previous best sequence (if any).

2. **Upscaling & intensive permutation search** – the best candidate
   is linearly interpolated to a larger length (≈12000–15000) and a
   massive random‑permutation hill‑climbing phase is performed.  Each
   permutation keeps the ℓ¹‑norm fixed, therefore only the ordering
   influences the convolution peak.

3. **Fine‑grained mass‑transfer annealing** – starting from the best
   ordering a simple simulated‑annealing that moves a small amount of
   mass from a high‑mass index to a low‑mass index is run until the
   time budget expires.

All iterates stay on the simplex Σaᵢ = √(2·n) and respect non‑negativity.
The routine never returns ``np.inf`` and trims trailing zeros before
returning the list of heights.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT utilities (max‑convolution and full convolution)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (the value we want to minimise)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a)/(Σa)².
    When Σa = √(2·n) this equals the maximal convolution entry.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of sequences (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then projected onto the simplex."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then projected onto the simplex."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest leftmost, second largest rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (very cheap)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact peak."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Random permutation hill‑climbing (massive ordering search)
# ----------------------------------------------------------------------


def _random_permutation_search(a0: np.ndarray,
                               deadline: float) -> tuple[np.ndarray, float]:
    """Try many random reorderings of the same multiset."""
    best_a = a0.copy()
    best_score = _score(a0)
    rng = np.random.default_rng()
    n = a0.shape[0]

    while time.time() < deadline:
        perm = rng.permutation(n)
        cand = a0[perm]
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[{time.time():.2f}] Perm search new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (fine polishing)
# ----------------------------------------------------------------------


def _mass_transfer_annealing(a0: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from a heavy index to a light one."""
    a = a0.copy()
    best_a = a.copy()
    cur_score = best_score = _score(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    start = time.time()

    while time.time() < deadline:
        # donor selected proportionally to its mass (high mass more likely)
        total = a.sum()
        if total == 0:
            break
        prob = a / total
        donor = rng.choice(n, p=prob)
        if a[donor] < 1e-14:
            continue
        # receiver uniformly random, different from donor
        receiver = rng.integers(n - 1)
        if receiver >= donor:
            receiver += 1

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta

        a[donor] -= delta
        a[receiver] += delta

        new_score = _score(a)

        # linear cooling schedule
        elapsed = time.time() - start
        total_time = deadline - start
        T = max(0.0001, 1.0 - elapsed / total_time)

        accept = (new_score < cur_score or
                  rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT anneal new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta

        # tiny jitter every few thousand iterations to stay out of plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=cur_score * 0.0005, size=n)
            a += jitter
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a, np.sqrt(2.0 * n))

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The search consists of fast sub‑gradient exploration on a modest size,

    """

    # --------------------------------------------------------------

    elif budget_s >= 500:

        n_expl = 2000

    rng = np.random.default_rng()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    init_pool: List[np.ndarray] = []

        init_pool.append(a)

    return final_seq
