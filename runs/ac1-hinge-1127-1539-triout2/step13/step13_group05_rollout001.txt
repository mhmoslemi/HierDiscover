def lp_step(a, M):
    n = len(a)
    # Build constraints: for each k in 0..2n-2: dot( f_shift[k], g ) <= M
    # Where f_shift[k] is length n vector with entries:
    # For j in [0,n-1], if 0 <= k - j < n: f_shift[k][j] = a[k-j]
    # else 0
    # Build matrix rows for k: we can allocate list of tuples (row, col, value)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # j from max(0,k-(n-1)) to min(k,n-1)
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # For these j, value f[k - j] constant
        # So we could iterate j across this range; it's O(n^2) loops.
        for j in range(lo, hi+1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    C = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # solve maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = scipy.optimize.linprog(c, A_ub=C, b_ub=np.full(2*n-1, M), bounds=bounds,
                                method='highs', options={'presolve': True})
    if not res.success: return None
    g = res.x
    return g

"""
Propose a step‑function that minimizes the max self‑convolution.
The method combines three ideas:

1. **Parametric warm‑start** – a quick scan of power‑law /
   exponentially tilted families provides a decent initial a.

2. **Linear‑programming (LP) refinement** – for a fixed a we solve
   a linear program that maximises the total mass of a new vector g
   under the linear constraints   a * g ≤ M   (M = max(a*a)).
   After scaling g to the required ℓ1‑norm we mix a←(1‑t)·a + t·g.
   Repeating this step provably never increases the peak of
   a*a and usually reduces it.

3. **Final polishing** – a short Adam‑style projected gradient descent
   (soft‑max approximation of max) removes the remaining tiny slack.

The LP is solved with `scipy.optimize.linprog` (HiGHS) using a sparse
Toeplitz constraint matrix, which makes each solve O(n²) but fast for
n≈1500–2000.  The whole routine respects the supplied time budget
(and returns the best sequence found before the deadline).
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(b)/(sum(a)²).  With sum(a)=√(2n) this equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s,  x ≥ 0 (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (max‑sum g under a * g ≤ M)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(a: np.ndarray,
                      M: float,
                      target_sum: float,
                      deadline: float) -> np.ndarray | None:
    """
    Solve  max Σ g_i   s.t.   a * g ≤ M,   g ≥ 0.
    Returns a scaled vector g with Σ g = target_sum, or None on failure.
    """
    n = a.shape[0]

    # Build the Toeplitz constraint matrix a * g ≤ M
    # For each offset d = 0..n‑1 the diagonal consists of the whole vector a.
    # Using scipy.sparse.diags is O(n²) in memory but fast enough for n≈2000.
    offsets = np.arange(n, dtype=np.int32)
    C = sp.diags([a] * n, offsets, shape=(2 * n - 1, n), format="csr")

    c = -np.ones(n)                     # maximise Σ g_i  ⇒  minimise -Σ g_i
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Remaining time for the solver (HiGHS respects a \"time_limit\" option)
    remaining = max(0.0, deadline - time.time())
    options = {"presolve": True}
    if remaining > 0.0:
        # recent SciPy versions accept a time limit for HiGHS
        options["time_limit"] = remaining

    try:
        res = opt.linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                          method="highs", options=options)
    except Exception as e:   # any unexpected issue → fall back
        print(f"linprog exception: {e}")
        return None

    if not res.success:
        print(f"LP not successful: {res.message}")
        return None

    g0 = np.maximum(res.x, 0.0)
    s0 = g0.sum()
    if s0 <= 0.0:
        return None

    # Scale to the required ℓ1‑norm
    scale = target_sum / s0
    g = g0 * scale
    g = _project_to_simplex(g, target_sum)   # tiny numerical drift correction
    return g

        new_a = (1.0 - t) * a + t * g
        new_a = _project_to_simplex(new_a, target_sum)

        new_score = _score(new_a)
        if new_score < best_score - 1e-9:
            best_score, best_seq = new_score, new_a.copy()
            print(f"[{time.time():.2f}] LP step improved to {best_score:.6f}")
            continue

        # try a smaller mixing factor if the first attempt was not good
        t = 0.1
        new_a = (1.0 - t) * a + t * g
        new_a = _project_to_simplex(new_a, target_sum)
        new_score = _score(new_a)
        if new_score < best_score - 1e-9:
            best_score, best_seq = new_score, new_a.copy()
            print(f"[{time.time():.2f}] LP step (t=0.1) improved to {best_score:.6f}")
            continue

        # No improvement – stop the LP loop
        print(f"[{time.time():.2f}] LP loop stalled (score={best_score:.6f})")
        break

    # --------------------------------------------------------------
    # Phase 2 – final Adam‑GD polishing (soft‑max with a very large β)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=5e5,
                                                step_factor=0.7,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polishing improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional sanity check (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
