"""
Hybrid optimiser for the step‑function attack.

The routine works in three stages:

1. **Global seeding + projected differential evolution** – a modest
   population of diverse simplex‑projected candidates (power‑law,
   exponential, random, previous best) is evolved with classic DE.
   The population stays feasible thanks to a Euclidean projection onto
   the ℓ¹‑simplex Σa = √(2 n).

2. **Simulated‑annealing local refinement** – starting from the best DE
   individual we run a long SA that either swaps two entries (preserving
   the multiset of heights) or transfers a small amount of mass from one
   index to another.  The objective is the maximal entry of the
   linear convolution a*a; the temperature is geometrically cooled until
   it becomes negligible.

3. **Permutation polishing** – a handful of deterministic reorderings
   (alternating, even‑spread) together with a few random shuffles are
   tested; the ordering with the smallest convolution peak is kept.
   Trailing near‑zero entries are trimmed and the final sequence is stored
   in the global ``height_sequence_1`` variable for the next run.

The whole procedure respects the 1000 s time budget and continually
adjusts the time allocated to each phase.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For sequences normalised to Σa = √(2n) this equals the max‑convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    take_left = True
    for v in sorted_vals:
        if take_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        take_left = not take_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    # ensure coprime step
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Projected DE loop."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # three distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # leave a little margin for the next phases
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local optimiser
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         init_frac: float = 0.07,
                         final_frac: float = 1e-6,
                         swap_prob: float = 0.6) -> tuple[np.ndarray, float]:
    """
    SA that mixes full swaps and small mass transfers.
    The temperature is geometrically cooled from init_frac·max to final_frac·max.
    """
    a = a0.copy()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # temperature scaling (relative to the current max)
    init_T = max(cur_max * init_frac, 1e-4)
    final_T = max(cur_max * final_frac, 1e-12)

    while time.time() < deadline:
        # linearised progress in [0,1]
        t = (time.time() - start) / total
        T = init_T * (final_T / init_T) ** t

        if rng.random() < swap_prob:
            # full swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # small mass transfer
            i = rng.integers(0, n)
            if a[i] < 1e-12:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # transfer up to 10 % of donor
            delta = a[i] * rng.random() * 0.10
            # keep within the 0‑1000 caps used by the evaluator
            delta = min(delta, 1000.0 - a[j])
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta

        cand_max = _max_convolution_fft(cand)
        d = cand_max - cur_max
        if d < 0.0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            a = cand
            cur_max = cand_max
            if cand_max < best_max:
                best_max = cand_max
                best_a = cand.copy()

    return best_a, best_max

        seeds.append(seq_j)

    # --------------------------------------------------------------

    pop_size = max(20, int(n / 400))   # roughly 20‑30 individuals

    # --------------------------------------------------------------

        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
