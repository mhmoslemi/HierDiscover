"""
search for a low‑score non‑negative step sequence.

The algorithm uses three stages:

1️⃣  *smooth softmax gradient descent* – we minimise a smoothed
    approximation of  max(a⋆a)  (log‑sum‑exp) while keeping the
    ℓ1‑norm fixed (projection onto the simplex).  The smoothing
    parameter is increased gradually, turning the method into a
    sub‑gradient descent on the true max.

2️⃣  *aggressive simulated‑annealing mass‑transfer* – the same
    routine as in the reference implementation, but started from
    the result of the first stage.

3️⃣  *multiple warm‑starts* – a few random Dirichlet vectors,
    a few power‑law candidates and the previous best sequence are
    evaluated; the best among them seeds the optimisation.

All heavy work (convolutions) is performed with FFTs.
The routine respects the given time budget and returns the best
list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a power‑of‑2 FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·n‑1)."""
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
# Objective, projection and helpers
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a⋆a) / (sum(a)²); no normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_self_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Stage 1 – smooth softmax gradient descent
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Gradient descent on a smoothed version of max(a*a).

    The smoothing uses a soft‑max (log‑sum‑exp) with a temperature that
    grows from t0≈0.5 to t_max≈30.0.  After each step the iterate is
    projected onto the simplex Σa = target_sum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    it = 0

    # pre‑compute the FFT size needed for the a_rev ⋆ w convolution
    fft_len = 1 << ((3 * n - 2 - 1).bit_length())

    while time.time() < deadline:
        # ----- compute current convolution and soft‑max weights -----
        b = _conv_self_fft(a)                       # length 2·n‑1
        max_b = b.max()
        elapsed = time.time() - start
        frac = elapsed / total

        # temperature schedule for the soft‑max (increase → sharper)
        t0, t_max = 0.5, 30.0
        t = t0 + (t_max - t0) * frac

        # stable soft‑max
        w = np.exp(t * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient = 2·Σ_i w_i·a[i‑j]  (j‑th component) -----
        a_rev = a[::-1]
        A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        W_fft = np.fft.rfft(w, n=fft_len)
        conv_rev = np.fft.irfft(A_rev_fft * W_fft, n=fft_len)[: 3 * n - 2]

        # term1_j = Σ_p a[p]·w[p + j]  = conv_rev[n‑1+j]
        term = conv_rev[n - 1 : n - 1 + n]          # length n
        grad = 2.0 * term

        # ----- step size (decreases linearly) -----
        eta0, eta_min = 0.5, 0.001
        eta = eta0 * (1.0 - frac) + eta_min

        # gradient step + projection
        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape flat regions
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SmoothGD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from a ‘bad’
    index (large contribution to the current peak) to a ‘good’
    index (small height).  The convolution is updated exactly in O(n).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_self_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                         # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_self_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)       # normalisation Σa = √(2n)

    # --------------------------------------------------------------
    # Build a list of initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # warm start from previous run, if available
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # random Dirichlet (exponential) vectors
    for _ in range(4):
        vec = rng.exponential(scale=1.0, size=n)
        vec = _project_to_simplex(vec, target_sum)
        init_candidates.append(vec)

    # power‑law families (a_i ∝ (i+1)^{-p})
    for p in np.linspace(0.30, 1.00, 8):
        vec = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        vec = vec * (target_sum / vec.sum())
        init_candidates.append(vec)

    # --------------------------------------------------------------
    # Evaluate all candidates and keep the best
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, cand in enumerate(init_candidates):
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
        print(f"[{time.time():.2f}] init #{idx+1} score={val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – smooth gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.55 * (deadline - time.time())
        seq_gd, score_gd = _smooth_gradient_descent(best_seq, target_sum, sub_deadline)
        if score_gd < best_score:
            best_score = score_gd
            best_seq = seq_gd.copy()
            print(f"[{time.time():.2f}] after SmoothGD best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – aggressive simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq_ann, score_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if score_ann < best_score:
            best_score = score_ann
            best_seq = seq_ann.copy()
            print(f"[{time.time():.2f}] after Anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, trimming trailing near‑zero entries
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # keep for the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
