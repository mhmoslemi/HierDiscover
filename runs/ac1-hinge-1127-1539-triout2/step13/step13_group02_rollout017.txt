def _gradient_softmax(a, beta):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
    maxconv = conv.max()
    exps = np.exp(beta * (conv - maxconv))
    p = exps / exps.sum()
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(np.fft.rfft(a_rev, n=size) * p_fft, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad, conv

def _adam_optimize(a0, deadline, target_sum, max_beta=1e7, lr=None):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    if lr is None:
        lr = target_sum * 0.02

    start = time.time()
    t_iter = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        progress = min(1.0, elapsed / total)
        beta = 0.5 * (max_beta / 0.5) ** progress  # start 0.5
        # compute gradient
        grad, _ = _gradient_softmax(a, beta)
        t_iter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)
        # occasional jitter
        if t_iter % 1000 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum*0.0005, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t_iter} beta={beta:.1f} score={best_val:.6f}")

    return best_a, best_val

"""
Advanced optimiser for the step‑function benchmark.

The algorithm consists of three stages:

1. **Population based global search** – a small Differential‑Evolution (DE)
   population (size ≈ 12) is evolved on the original objective
   (maximal entry of a*a).  The individuals are always projected onto the
   ℓ¹‑simplex Σa = √(2·n) (which makes the objective equal to the peak of
   the convolution).

2. **Fine‑grained Adam optimisation** – the best DE individual is refined
   with a projected Adam optimiser that minimises a soft‑max approximation
   of the peak.  The soft‑max temperature (β) is annealed from 0.5 up to a
   huge value (≈10⁸) so that the final steps target the true max.

3. **Short deterministic mass‑transfer polishing** – a cheap “mass‑transfer”
   move (the same routine used in the reference solution) finishes the
   optimisation.

All steps respect the 1000 s budget and never return `np.inf`.  The final
sequence is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    With Σa = √(2·n) this equals the peak of the convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of sequences (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, then projected onto the simplex."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), then projected onto the simplex."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize (keeps total mass)."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_c = np.max(conv)
    exps = np.exp(beta * (conv - max_c))
    p = exps / np.sum(exps)                     # soft‑max weights over the 2n‑1 entries

    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   max_beta: float = 5e7,
                   lr: float = None) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max surrogate of the peak.
    β is annealed exponentially from 0.5 to ``max_beta``.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    if lr is None:
        lr = target_sum * 0.02

    start = time.time()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # progress‑dependent temperature
        prog = (time.time() - start) / max(1e-12, deadline - start)
        beta = 0.5 * (max_beta / 0.5) ** prog

        grad, _ = _gradient_softmax(a, beta)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} β={beta:8.1f} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑Evolution (DE) on the true peak (max‑convolution)
# ----------------------------------------------------------------------


def _differential_evolution(pop: list[np.ndarray],
                            target_sum: float,
                            deadline: float,
                            F: float = 0.5,
                            CR: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Simple DE with a fixed population.
    Keeps the population on the ℓ¹‑simplex and minimises the exact peak.
    """
    popsize = len(pop)
    scores = np.array([_score(ind) for ind in pop])
    best_idx = int(np.argmin(scores))
    best_seq = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    rng = np.random.default_rng()

    while time.time() < deadline:
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover
            cross = rng.random(mutant.shape) < CR
            if not np.any(cross):
                cross[rng.integers(0, mutant.shape[0])] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
                    print(f"[{time.time():.2f}] DE swap i={i} new best={best_score:.6f}")

        # safety break when time is about to run out
        if deadline - time.time() < 0.2:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.3,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Same routine as the reference – cheap polishing."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / max(1e-12, deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine performs:
      1. Differential‑Evolution on the exact peak for a few seconds.
      2. Adam optimisation on a soft‑max surrogate (high β) for the rest.
      3. Final deterministic mass‑transfer polishing.
    The best sequence found is stored globally as ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size that balances expressiveness and speed.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Initialise a modest DE population.
    # ------------------------------------------------------------------
    popsize = 12
    population = []

    # previous best (if any)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev_resized = _resize_sequence(prev, n)
        prev_resized = _project_to_simplex(prev_resized, target_sum)
        population.append(prev_resized)

    # remaining slots – a mix of random Dirichlet, power‑law and exponential.
    while len(population) < popsize:
        typ = rng.integers(0, 3)
        if typ == 0:                      # random Dirichlet (uniform alphas)
            a = rng.random(n)
            a = a / a.sum() * target_sum
        elif typ == 1:                    # power‑law
            p = rng.uniform(0.3, 0.8)
            a = _power_law_sequence(n, p, target_sum)
        else:                             # exponential
            lam = rng.uniform(0.001, 0.02)
            a = _exponential_sequence(n, lam, target_sum)
        population.append(a)

    # ------------------------------------------------------------------
    # Stage 1 – Differential Evolution (global exploration)
    # ------------------------------------------------------------------
    de_deadline = deadline - 120.0  # reserve ~2 minutes for later phases
    if time.time() < de_deadline:
        best_seq, best_score = _differential_evolution(population,
                                                       target_sum,
                                                       de_deadline,
                                                       F=0.5,
                                                       CR=0.9)
    else:
        # fallback – just pick the current best in the pool
        scores = [_score(ind) for ind in population]
        best_idx = int(np.argmin(scores))
        best_seq = population[best_idx].copy()
        best_score = scores[best_idx]

    print(f"[{time.time():.2f}] After DE best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on the soft‑max surrogate (local refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        adam_deadline = deadline - 5.0   # keep a tiny buffer for the final pass
        best_seq, best_score = _adam_optimize(best_seq,
                                              adam_deadline,
                                              target_sum,
                                              max_beta=5e7,
                                              lr=target_sum * 0.015)
        print(f"[{time.time():.2f}] After Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Deterministic mass‑transfer polishing (final touch)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = deadline - 1.0
        best_seq, best_score = _deterministic_mass_transfer(best_seq,
                                                            mt_deadline,
                                                            target_sum,
                                                            start_temp=0.2,
                                                            end_temp=1e-6,
                                                            low_frac=0.05)
        print(f"[{time.time():.2f}] After MT polishing best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Log final evaluation (optional)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
