"""
Hybrid optimizer for the step‑function max‑convolution problem.

The routine works in three phases:

1) **Diverse seeding + a short projected Differential Evolution (DE)**
   – builds a modest population of feasible vectors (power‑law,
     exponential, random simplex, and the previous best) and
   – runs DE for a few seconds while keeping every individual on the
     ℓ¹‑simplex Σa = √(2 n).

2) **Intensive local polishing**
   – a long deterministic mass‑transfer phase (simulated‑annealing style)
     that repeatedly moves mass from indices that contribute heavily to
     the current maximal convolution entry to indices that contribute
     little;
   – a long projected sub‑gradient descent that uses the exact sub‑gradient
     of the max‑convolution (summing over all active constraints) and a
     momentum term.

3) **Reordering & trimming**
   – tests several simple permutations (alternating, even‑spread, sorted)
     that often lower the peak of the convolution;
   – removes trailing near‑zero entries for compactness.

The whole algorithm respects the 1000 s budget, automatically dividing the
time between the phases.  It loads the previous best sequence (if any) from
the global variable ``height_sequence_1`` and stores the new best there for
future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For Σa = √(2 n) this equals max‑convolution, but the function works for
    any feasible vector (the ratio is scale‑invariant).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (simple parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to Σ = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple deterministic reorderings (often lower the peak)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second‑largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _descending_order(seq: np.ndarray) -> np.ndarray:
    return np.sort(seq)[::-1]


def _ascending_order(seq: np.ndarray) -> np.ndarray:
    return np.sort(seq)


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves with a decaying Metropolis acceptance rule.
    Parameters
    ----------
    a0 : initial vector (already on the simplex)
    deadline : absolute time at which the routine must stop
    target_sum : required ℓ¹‑norm (used for re‑projection)
    start_temp, end_temp : annealing schedule
    low_frac : fraction of low‑contributing indices considered as receivers
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        # temperature schedule (linear in log‑scale)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)               # ∈[0,1)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where convolution attains its maximum
        kmax = int(np.argmax(conv))

        # range of indices that pair to give the entry kmax
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # products contributing to the current maximum
        prod = a[idx] * a[kmax - idx]

        # donor = index with the *largest* product (most responsible)
        donor = idx[np.argmax(prod)]

        # choose a receiver among the low‑contributing indices
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        # move a modest amount of mass
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that uses the exact max‑convolution.
    A simple momentum term (exponential moving average) is employed.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        # sub‑gradient of max‑convolution
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑Evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Classic DE loop with projection onto the ℓ¹‑simplex after mutation and
    crossover.  The population is kept small (≈20‑30 individuals) to stay
    fast for large n.
    """
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # three distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one donor)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # leave a couple of seconds for the polishing phases
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid global‑local search for a step‑function with a low evaluation score.
    The routine respects the given time budget (default 1000 s) and stores the
    best sequence found in the global variable ``height_sequence_1`` for use in
    subsequent runs.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives a richer search space)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Initialise a diverse pool of seeds
    # --------------------------------------------------------------
    seeds = []

    # parametric families
    for p in np.linspace(0.2, 1.0, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))
    for lam in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))
    # random simplex points
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best (if any) – resize if length differs
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        seeds.append(_project_to_simplex(prev, target_sum))

    # jitter each seed a little to increase diversity
    jittered = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------------------------------------------------
    # Build a modest DE population (≈20‑30 individuals)
    # --------------------------------------------------------------
    pop_size = max(20, int(n / 400))   # around 20‑30
    population = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())
    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------------------------------------------------
    # Very short DE phase – just enough to escape a possibly bad local
    # --------------------------------------------------------------
    de_deadline = start_time + 0.07 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Main local‑search loop: mass‑transfer → sub‑gradient → reorder
    # --------------------------------------------------------------
    while True:
        now = time.time()
        remaining = deadline - now
        if remaining < 30.0:                     # keep a small buffer for final steps
            break

        # ---- deterministic mass‑transfer (≈40% of remaining time) ----
        mt_time = min(0.40 * remaining, 300.0)    # cap to avoid overly long single runs
        mt_deadline = now + mt_time
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum,
            start_temp=0.4, end_temp=1e-6, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] MT improved to {best_score:.6f}")

        # ---- sub‑gradient descent (≈40% of remaining time) ----
        now = time.time()
        remaining = deadline - now
        if remaining < 30.0:
            break
        sg_time = min(0.40 * remaining, 300.0)
        sg_deadline = now + sg_time
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_deadline, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_seq, best_score = seq_sg, score_sg
            print(f"[{time.time():.2f}] SG improved to {best_score:.6f}")

        # ---- simple reorderings (cheap) ----
        candidates = [
            best_seq,
            _alternating_order(best_seq),
            _even_spread_order(best_seq),
            _descending_order(best_seq),
            _ascending_order(best_seq),
        ]
        seq_best = min(candidates, key=_max_convolution_fft)
        score_best = _score(seq_best)
        if score_best < best_score:
            best_seq, best_score = seq_best, score_best
            print(f"[{time.time():.2f}] Reorder improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final tidy‑up: trim trailing near‑zeros and store globally
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
