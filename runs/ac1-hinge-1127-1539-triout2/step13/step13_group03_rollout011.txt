import time
import math
import numpy as np
from typing import List

def _max_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _project_to_simplex(v, s):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n, p, target_sum):
    a = np.arange(1, n+1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n, lam, target_sum):
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _cosine_sequence(n, target_sum):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a

def _sin_half_sequence(n, target_sum):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a

def _random_simplex(n, target_sum, rng):
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq):
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n-1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out

def _even_spread_order(seq):
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(a0, deadline, target_sum,
                               max_beta=8e5, step_factor=0.9, gamma=0.98):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # print progress
            print(f"[{time.time():.2f}] GD it={it} β={beta:.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0, deadline, target_sum,
                         step_factor=0.45, gamma=0.995):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _mass_transfer_simulated_annealing(a0, deadline, target_sum,
                                      start_temp=0.4, end_temp=1e-5, low_frac=0.07):
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Multi‑scale optimizer for the max‑self‑convolution problem.
    Starts from the previous best sequence (or a random one),
    progressively upsamples, and refines it with
    annealed‑gradient, sub‑gradient and a short mass‑transfer SA.
    """
    start = time.time()
    deadline = start + budget_s - 0.5
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Base sequence (modest size)
    # ------------------------------------------------------------------
    n_base = 5000
    target_base = math.sqrt(2.0 * n_base)

    if "height_sequence_1" in globals():
        base_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if base_seq.shape[0] != n_base:
            base_seq = _resize_sequence(base_seq, n_base)
    else:
        base_seq = _random_simplex(n_base, target_base, rng)

    base_seq = _project_to_simplex(base_seq, target_base)

    # quick low‑cost annealed GD to clean up base if time permits
    if time.time() < deadline - 30.0:
        a_tmp, v_tmp = _annealed_gradient_descent(
            base_seq, min(deadline - 5.0, time.time() + 10.0),
            target_base, max_beta=5e5, step_factor=0.9, gamma=0.98)
        if v_tmp < _max_convolution_fft(base_seq):
            base_seq = a_tmp
            print(f"[{time.time():.2f}] Base refinement score={v_tmp:.6f}")

    # ------------------------------------------------------------------
    # Choose final size based on remaining time
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # conservative mapping from remaining seconds to target length
    if remaining > 300:
        n_final = 40000
    elif remaining > 200:
        n_final = 30000
    elif remaining > 120:
        n_final = 25000
    elif remaining > 80:
        n_final = 20000
    elif remaining > 50:
        n_final = 15000
    else:
        n_final = n_base

    target_final = math.sqrt(2.0 * n_final)
    seq = _resize_sequence(base_seq, n_final)
    seq = _project_to_simplex(seq, target_final)

    # random permutation to break any undesirable ordering
    if rng.random() < 0.5:
        seq = _alternating_order(seq)
    else:
        seq = _even_spread_order(seq)

    # ------------------------------------------------------------------
    # Phase A – long annealed gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        gd_deadline = min(deadline - 30.0, time.time() + (deadline - time.time()) * 0.45)
        seq, val = _annealed_gradient_descent(
            seq, gd_deadline, target_final,
            max_beta=2e6, step_factor=0.9, gamma=0.98)
        print(f"[{time.time():.2f}] After GD max={val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = min(deadline - 10.0, time.time() + (deadline - time.time()) * 0.5)
        seq, val = _subgradient_descent(seq, sub_deadline, target_final,
                                        step_factor=0.45, gamma=0.995)
        print(f"[{time.time():.2f}] After SubGrad max={val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – short simulated‑annealing mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = deadline - 2.0
        seq, val = _mass_transfer_simulated_annealing(
            seq, mt_deadline, target_final,
            start_temp=0.25, end_temp=1e-6, low_frac=0.05)
        print(f"[{time.time():.2f}] Final MT max={val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing negligible entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq, dtype=np.float64)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # optional verification print
    try:
        print(f"[{time.time():.2f}] Final evaluated score = {evaluate_sequence(final_seq):.6f}")
    except Exception:
        pass

    return final_seq

"""
Multi‑scale optimiser for the max‑self‑convolution problem.

The objective after normalising Σa = √(2 n) is simply
    max_k (a * a)[k] .
The algorithm proceeds as follows:

1.  Obtain a decent seed – the previous best sequence is reused if
    available, otherwise a random simplex point is generated (size ≈5 k).
2.  The seed is optionally cleaned with a short annealed‑gradient run.
3.  The seed is up‑sampled to a larger size (chosen from 15 k … 40 k
    depending on the remaining time) and projected back onto the
    ℓ¹‑simplex.
4.  A long annealed‑gradient descent (soft‑max approximation of the
    max‑convolution) reduces the peak aggressively.
5.  Exact sub‑gradient descent with momentum refines the solution.
6.  A brief simulated‑annealing mass‑transfer phase removes remaining
    spikes.
7.  A few random permutations and deterministic reorderings
    (alternating / even‑spread) are examined and the best layout is kept.
8.  Trailing near‑zero entries are stripped, the sequence is stored in
    the global variable ``height_sequence_1`` (for reuse in later calls)
    and returned.

All steps respect the non‑negativity and ℓ¹‑simplex constraints.
The routine adapts the target length to the remaining wall‑clock time,
so it comfortably fits the 1000 s budget.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Hann window – peaks in the centre, zero at the ends."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """sin(π·x) – also centre‑heavy."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from old length to new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Distribute values evenly using a step that is coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 8e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max surrogate of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv gradient)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from a strong‑pair donor to a weak‑pair receiver."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine adapts its workload to the remaining time, works from a
    previously found solution if present, up‑samples to a larger scale,
    and then applies an annealed gradient, an exact sub‑gradient and a
    short simulated‑annealing phase.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Base sequence (moderate size ≈5 k)
    # ------------------------------------------------------------------
    n_base = 5000
    target_base = math.sqrt(2.0 * n_base)

    if "height_sequence_1" in globals():
        base_seq = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if base_seq.shape[0] != n_base:
            base_seq = _resize_sequence(base_seq, n_base)
    else:
        base_seq = _random_simplex(n_base, target_base, rng)

    base_seq = _project_to_simplex(base_seq, target_base)

    # A quick, cheap gradient refinement (if time permits)
    if time.time() < deadline - 30.0:
        a_tmp, v_tmp = _annealed_gradient_descent(
            base_seq,
            min(deadline - 5.0, time.time() + 10.0),
            target_base,
            max_beta=5e5,
            step_factor=0.9,
            gamma=0.98,
        )
        if v_tmp < _max_convolution_fft(base_seq):
            base_seq = a_tmp
            print(f"[{time.time():.2f}] Base refinement score={v_tmp:.6f}")

    # ------------------------------------------------------------------
    # 2. Choose final length according to the time left
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 300:
        n_final = 40000
    elif remaining > 200:
        n_final = 30000
    elif remaining > 120:
        n_final = 25000
    elif remaining > 80:
        n_final = 20000
    elif remaining > 50:
        n_final = 15000
    else:
        n_final = n_base

    target_final = math.sqrt(2.0 * n_final)

    seq = _resize_sequence(base_seq, n_final)
    seq = _project_to_simplex(seq, target_final)

    # Break any pathological ordering early
    if rng.random() < 0.5:
        seq = _alternating_order(seq)
    else:
        seq = _even_spread_order(seq)

    # ------------------------------------------------------------------
    # 3. Phase A – annealed gradient descent (soft‑max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        gd_deadline = min(deadline - 30.0,
                          time.time() + (deadline - time.time()) * 0.45)
        seq, val = _annealed_gradient_descent(
            seq,
            gd_deadline,
            target_final,
            max_beta=2e6,
            step_factor=0.9,
            gamma=0.98,
        )
        print(f"[{time.time():.2f}] After GD max={val:.6f}")

    # ------------------------------------------------------------------
    # 4. Phase B – exact sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = min(deadline - 10.0,
                           time.time() + (deadline - time.time()) * 0.5)
        seq, val = _subgradient_descent(
            seq,
            sub_deadline,
            target_final,
            step_factor=0.45,
            gamma=0.995,
        )
        print(f"[{time.time():.2f}] After SubGrad max={val:.6f}")

    # ------------------------------------------------------------------
    # 5. Phase C – short simulated‑annealing mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = deadline - 2.0
        seq, val = _mass_transfer_simulated_annealing(
            seq,
            mt_deadline,
            target_final,
            start_temp=0.25,
            end_temp=1e-6,
            low_frac=0.05,
        )
        print(f"[{time.time():.2f}] Final MT max={val:.6f}")

    # ------------------------------------------------------------------
    # 6. Tiny permutation sweep (cheap, can shave a bit)
    # ------------------------------------------------------------------
    best_seq = seq
    best_val = _max_convolution_fft(seq)
    for _ in range(30):
        perm = rng.permutation(seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm
    seq = best_seq
    print(f"[{time.time():.2f}] After random perm sweep max={best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Deterministic re‑ordering trial
    # ------------------------------------------------------------------
    candidates = [seq, _alternating_order(seq), _even_spread_order(seq)]
    seq = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Best deterministic order max={_max_convolution_fft(seq):.6f}")

    # ------------------------------------------------------------------
    # 8. Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq, dtype=np.float64)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # Optional final verification (does not affect the returned value)
    try:
        print(f"[{time.time():.2f}] Final evaluated score = {evaluate_sequence(final_seq):.6f}")
    except Exception:
        pass

    return final_seq
