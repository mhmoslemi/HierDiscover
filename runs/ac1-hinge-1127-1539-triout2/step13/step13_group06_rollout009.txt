"""
Advanced step‑function optimizer with an additional LP‑based refinement.

The routine proceeds through the following stages (all respecting the
global time budget):

0.  Coarse parametric sweep (pure power‑law and exponentially tilted) to
    obtain a good initial guess (including the previously stored global
    best if available).

1.  Projected Adam gradient descent with a soft‑max approximation of the
    max‑convolution (several restarts).

2.  **New phase** – linear‑programming refinement.
    For the current shape `f` we solve
        max 1ᵀg   s.t.   f * g ≤ M  (component‑wise) , g ≥ 0
    where `M = max(f*f)`.  The LP produces a direction that can increase
    the L¹‑mass without increasing the current peak.  A convex combination
    of `f` and the LP solution is projected back onto the simplex.
    Repeating this step quickly lowers the objective further.

3.  Exact deterministic mass‑transfer (DMT2) – never increases the
    objective, but drives the solution toward an optimal extreme point.

4.  Greedy pairwise transfers with larger donor/receiver pools.

5.  Simulated‑annealing style local search for escaping shallow minima.

6.  Final short Adam polish with a very high soft‑max β.

Trailing near‑zero entries are trimmed before returning the sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective value as defined by ``evaluate_sequence``."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s ,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a)))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Several short Adam runs started from jittered variants of ``init``."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the previous DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-13) -> tuple[np.ndarray, float]:
    """The classic DMT2 step – never increases the objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # Find second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick one maximal index (any)
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass‑transfer (enhanced)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.03,
                               receiver_frac: float = 0.03,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise transfer with larger candidate pools."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        # cache shifted views for all candidates (lightweight view, no copy)
        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)          # effect per unit delta
                d_k = d[k]
                if d_k >= 0.0:
                    continue            # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # keep base up‑to‑date (cheap slice assignment)
        base[pad_len + i] = a[i]
        base[pad_len + j] = a[j]

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement (new component)
# ----------------------------------------------------------------------


def _simulated_annealing_refinement(a0: np.ndarray,
                                    deadline: float,
                                    target_sum: float) -> tuple[np.ndarray, float]:
    """Mass‑move SA that can escape shallow local minima."""
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    rng = np.random.default_rng()
    start_time = time.time()
    T0 = cur_max * 0.02 + 1e-4          # initial temperature scale
    iter_cnt = 0

    refresh_interval = 500
    donor_frac = 0.02
    recv_frac = 0.02

    donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
    receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

    while time.time() < deadline:
        if iter_cnt % refresh_interval == 0:
            donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
            receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # compute shift difference
        shift_i = base[i:i + 2 * n - 1]
        shift_j = base[j:j + 2 * n - 1]
        diff = shift_j - shift_i

        new_conv = conv + delta * diff
        new_max = new_conv.max()

        # acceptance rule (temperature decays linearly)
        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-9)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            base[pad_len + i] = a[i]
            base[pad_len + j] = a[j]

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={iter_cnt} new_best={best_score:.6f}")

        # occasional full recomputation to avoid drift
        if iter_cnt % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            base = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])

        iter_cnt += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase – unchanged)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (new component)
# ----------------------------------------------------------------------


def _lp_refinement(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   topk: int = 200,
                   max_iters: int = 8) -> tuple[np.ndarray, float]:
    """Linear‑programming step that increases mass without raising the current peak."""
    try:
        from scipy.optimize import linprog  # type: ignore
    except Exception as e:
        print("SciPy not available – skipping LP refinement.")
        return a0, _score(a0)

    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        M = conv.max()
        # select the most critical constraints
        if topk >= conv.size:
            idx_active = np.arange(conv.size)
        else:
            idx_active = np.argpartition(-conv, topk - 1)[:topk]

        # Build inequality matrix A (each row corresponds to a convolution index)
        rows = []
        for k in idx_active:
            row = np.zeros(n, dtype=np.float64)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            row[i] = a[k - i]
            rows.append(row)
        A = np.vstack(rows)                     # shape (len(idx_active), n)
        b = np.full(A.shape[0], M)

        # linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # bounds: g_i >= 0
        bounds = [(0, None)] * n

        # solve
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.2f}] LP failed (it={it}) – aborting LP refinement.")
            break

        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()

        # If we cannot increase the L1‑mass, stop.
        if sum_g <= sum_a * (1.0 + 1e-6):
            print(f"[{time.time():.2f}] LP did not increase mass (it={it}).")
            break

        # Take a convex step towards g and re‑project onto the simplex.
        alpha = 0.5
        a_candidate = (1 - alpha) * a + alpha * g
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        cand_score = _score(a_candidate)
        if cand_score < best_score - 1e-9:
            a = a_candidate
            best_score = cand_score
            print(f"[{time.time():.2f}] LP refinement it={it} improved score={best_score:.6f}")
        else:
            # No improvement – try a smaller step if possible
            if alpha > 0.05:
                alpha = 0.25
                a_candidate = (1 - alpha) * a + alpha * g
                a_candidate = _project_to_simplex(a_candidate, target_sum)
                cand_score = _score(a_candidate)
                if cand_score < best_score - 1e-9:
                    a = a_candidate
                    best_score = cand_score
                    print(f"[{time.time():.2f}] LP refinement (small step) it={it} improved score={best_score:.6f}")
                else:
                    print(f"[{time.time():.2f}] LP step did not improve (it={it}). Stopping LP phase.")
                    break
            else:
                print(f"[{time.time():.2f}] LP step did not improve (it={it}). Stopping LP phase.")
                break

        it += 1

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the supplied time budget and returns
    the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size based on the budget (larger budget → larger n)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)  →  score = max(conv)

    # ------------------------------------------------------------------
    # Phase 0 – coarse parametric search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 31)          # coarse power‑law grid
    q_grid = np.linspace(0.0, 6.0, 31)           # exponential tilt

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # incorporate a previously stored construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style projected gradient descent with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – LP‑based refinement (new)
    # ------------------------------------------------------------------
    lp_deadline = time.time() + (deadline - time.time()) * 0.20
    if lp_deadline > time.time() + 5.0:
        a_lp, val_lp = _lp_refinement(best_seq,
                                      target_sum,
                                      lp_deadline,
                                      topk=200,
                                      max_iters=8)
        if val_lp < best_score:
            best_score = val_lp
            best_seq = a_lp.copy()
            print(f"[{time.time():.2f}] LP refinement improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Exact deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # short random move to escape plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase D – Enhanced pairwise optimal mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.03,
                                                      receiver_frac=0.03)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final polishing with a higher β for the soft‑max
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase F – Simulated‑annealing refinement (new)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_sa, val_sa = _simulated_annealing_refinement(best_seq,
                                                       deadline,
                                                       target_sum)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] SA refinement improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
