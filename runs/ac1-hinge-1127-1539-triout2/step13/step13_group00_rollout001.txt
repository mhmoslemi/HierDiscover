    best_score = current_score
    t = 0.5  # mixing factor
    eps = 0.1  # initial tolerance for active set (1 - eps fraction)
    while time.time() < deadline:

        conv = _conv_fft(best_f)
        M = conv.max()
        # select active indices (top k indexes)
        # compute indices of top values
        # sort conv descending get top_k = maybe max(200, int(0.01 * len(conv))) but limit.
        top_k = min(300, len(conv))
        idx_sorted = np.argpartition(conv, -top_k)[-top_k:]
        # maybe filter to those near M
        # but we use top_k directly.
        # create sparse matrix A_ub
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(idx_sorted):
            start = max(0, k - (len(f) - 1))
            end = min(k, len(f) - 1)
            # j runs from start to end inclusive
            for j in range(start, end+1):
                rows.append(row_idx)
                cols.append(j)
                data.append(f[k - j])
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(idx_sorted), len(f)))
        # RHS: M for each row
        b_ub = np.full(len(idx_sorted), M)
        # objective: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(len(f))
        # bounds: b_j >= 0
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*len(f), method='highs')
        if not res.success:
            # cannot improve further or LP fail; break
            break
        b = res.x
        sum_b = b.sum()
        # compute improvement ratio
        if sum_b > target_sum * (1 + 1e-6):
            # we have found b with larger sum; combine
            g = b * (target_sum / sum_b)
            # mixing
            best_f = (1 - t) * best_f + t * g

            best_f = _project_to_simplex(best_f, target_sum)
            # evaluate new score

            # no improvement; maybe reduce eps set or break
            break
        # time check

"""
Hybrid LP‑+‑local‑search optimiser for the non‑negative step‑function problem.

The routine follows these stages:

1. **Initial seeding** – generate several feasible candidates (power‑law,
   exponential, uniform Dirichlet‑like, constant and the previous best) and
   project them onto the ℓ¹‑simplex Σa = √(2 n).

2. **Linear‑programming improvement** – for the current sequence f,
   compute M = max(f∗f).  Build a small set of “active’’ convolution
   constraints (the indices where f∗f is largest) and solve the linear
   program

        maximise   Σ b_j
        subject to b_j ≥ 0  and  f∗b ≤ M   (only on the active indices)

   The solution b often has a larger ℓ¹‑norm than f.  Rescale it to the
   target sum and mix it with f (convex combination).  Repeat while time
   permits.

3. **Local polishing** – a deterministic “mass‑transfer’’ routine
   (simulated‑annealing style) and a short projected sub‑gradient descent
   are applied to the best sequence obtained from step 2.

4. **Reordering** – three simple permutations (original, alternating,
   even‑spread) are tested; the ordering with the smallest max‑convolution
   is kept.

5. **Trimming** – trailing near‑zero entries are removed for compactness,
   the result is stored in the global variable `height_sequence_1` and
   returned.

The whole algorithm respects a 1000‑second budget.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a∗a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _uniform_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _constant_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.full(n, target_sum / n, dtype=np.float64)
    return a


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray,
                target_sum: float,
                current_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    One‑direction improvement: maximise Σb subject to f∗b ≤ M on an active set.
    Returns a potentially improved sequence and its score.
    """
    best_f = f.copy()
    best_score = current_score
    t_mix = 0.5               # initial mixing factor
    # active‑set size (tuned for speed vs quality)
    max_active = 300

    while time.time() < deadline:
        conv = _conv_fft(best_f)
        M = conv.max()
        # pick the largest entries (active constraints)
        k = min(max_active, conv.size)
        idx_active = np.argpartition(conv, -k)[-k:]

        # Build sparse constraint matrix A_ub * b ≤ M
        rows = []
        cols = []
        data = []
        n = best_f.shape[0]
        for row_idx, kk in enumerate(idx_active):
            lo = max(0, kk - (n - 1))
            hi = min(kk, n - 1)
            # j runs from lo..hi, coefficient is f[kk - j]
            # vectorised loop over this small range
            for j in range(lo, hi + 1):
                rows.append(row_idx)
                cols.append(j)
                data.append(best_f[kk - j])
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(idx_active), n))
        b_ub = np.full(len(idx_active), M)

        # maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n)

        res = scipy.optimize.linprog(c,
                                    A_ub=A,
                                    b_ub=b_ub,
                                    bounds=[(0, None)] * n,
                                    method='highs',
                                    options={'presolve': True})
        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= target_sum * (1 + 1e-8):
            # no improvement possible under current active set
            break

        # Scale b to meet the ℓ¹‑norm target and mix with current f
        g = b * (target_sum / sum_b)
        new_f = (1.0 - t_mix) * best_f + t_mix * g
        new_f = _project_to_simplex(new_f, target_sum)

        new_score = _score(new_f)
        if new_score < best_score - 1e-12:
            best_f = new_f
            best_score = new_score
            # slowly reduce the mixing factor to stabilise
            t_mix = max(0.05, t_mix * 0.9)
        else:
            # no improvement – reduce mixing or stop
            t_mix = max(0.05, t_mix * 0.9)
            # if mixing is already tiny, break
            if t_mix <= 0.07:
                break

        if time.time() > deadline - 1.0:
            break

    return best_f, best_score

            seeds.append(_project_to_simplex(prev_resized, target_sum))

    # pick the best seed according to the exact score
