def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5
    rng = np.random.default_rng()

    # set n
    if budget_s >= 800:
        n = 14000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000
    target_sum = np.sqrt(2.0 * n)

    # precompute base index arrays
    idx = np.arange(n, dtype=np.float64)
    i1 = idx + 1.0

    # function to construct candidate
    def make_candidate(p,q,alpha,beta,w):
        # base shapes
        # power
        pow_shape = i1 ** (-p) if w[0] > 0 else None
        # exponential
        exp_shape = np.exp(-q * idx / n) if w[1] > 0 else None
        # sine
        sin_shape = np.sin(np.pi * (idx + 0.5) / (n + 1.0)) if w[2] > 0 else None
        # beta shape
        beta_shape = (i1 ** alpha) * ((n - idx) ** beta) if w[3] > 0 else None

        a = np.zeros(n, dtype=np.float64)
        if w[0] > 0:
            a += w[0] * pow_shape
        if w[1] > 0:
            a += w[1] * exp_shape
        if w[2] > 0:
            a += w[2] * sin_shape
        if w[3] > 0:
            a += w[3] * beta_shape
        # ensure nonnegatives
        # scaling to target sum
        s = a.sum()
        if s == 0:
            return None
        a *= target_sum / s
        return a

    best_seq = None
    best_score = np.inf

    # Random search loop
    while time.time() < deadline - 5.0:   # leave 5 secs for final refinements
        # Decide number of shapes (randomly)
        num_shapes = rng.integers(1, 5)   # 1 to 4 inclusive

        # Sample weights for num_shapes
        w = rng.dirichlet(np.ones(num_shapes))
        # Pad to 4 weights
        w_full = np.zeros(4)
        # pick random indices for active shapes
        shape_indices = rng.choice(4, size=num_shapes, replace=False)
        w_full[shape_indices] = w

        # sample parameters:
        p = rng.uniform(0.0, 2.0)
        q = rng.uniform(0.0, 10.0)
        alpha = rng.uniform(0.0, 5.0)
        beta = rng.uniform(0.0, 5.0)

        cand = make_candidate(p,q,alpha,beta,w_full)
        if cand is None:
            continue

        # compute max convolution via FFT
        maxb = _max_convolution_fft(cand)
        if maxb < best_score:
            best_score = maxb
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New best score {best_score:.6f}")

    # Final refinement using Adam gradient descent
    if best_seq is not None and time.time() < deadline - 1.0:
        # run Adam for remaining time
        remaining_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      remaining_deadline,
                                                      target_sum,
                                                      max_beta=5e5,
                                                      step_factor=0.7,
                                                      gamma=0.992)

    # Trim trailing zeros and return list
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    return arr.tolist()

Simple multi‑stage optimizer for the maximal self‑convolution bound.
- Phase 0: direct parametric optimisation of power‑law (+ optional
  exponential tail) using a coarse‑to‑fine grid and golden‑section.
- Phase 1: stochastic mixture of a few analytic basis shapes; keep the
  best candidate found while respecting the time budget.
- Phase 2: projected Adam optimisation on a soft‑max approximation of
  the max‑convolution to polish the best candidate.

All intermediate vectors are projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)}.
The routine respects the given time budget (default 1000 s).
"""

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (complex)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the probability simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic basis shapes
# ----------------------------------------------------------------------


def _shape_power(p: float, n: int) -> np.ndarray:
    """a_i = (i+1)^(-p) (non‑negative, decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _shape_exp(q: float, n: int) -> np.ndarray:
    """a_i = exp(-q * i / n)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-q * i / n)


def _shape_sine(n: int) -> np.ndarray:
    """a_i = sin(π (i+0.5) / (n+1)). Always positive."""
    i = np.arange(n, dtype=np.float64) + 0.5
    return np.sin(np.pi * i / (n + 1.0))


def _shape_beta(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric beta‑like shape: (i+1)^α (n‑i)^β."""
    i = np.arange(n, dtype=np.float64)
    # add 1 to avoid zero at boundaries when α or β = 0
    return (i + 1) ** alpha * (n - i) ** beta

# Gradient of soft‑max surrogate and Adam optimiser
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return gradient of β·softmax(b) where b = a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max probabilities
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * conv_rev(p) restricted to first n entries
    # conv_rev = a_rev * p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation on the soft‑max surrogate of max(b)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam improvement: score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _golden_section(f, a, b, tol=1e-4, max_iter=30):
    """Return (x_opt, f_opt) minimising f on [a,b] using golden‑section."""
    phi = (1 + np.sqrt(5)) / 2
    invphi = 1 / phi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a, fc = c, fd
            c = d
            d = a + invphi * (b - a)
            fd = f(d)
    # choose the better of c and d
    if fc < fd:
        return c, fc
    else:
        return d, fd

    else:
        return d, fd


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution.  The routine follows three phases (parametric,
    random mixture, Adam polish) and respects the supplied time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # problem size – larger n gives a tighter theoretical lower bound
    if budget_s >= 800:
        n = 14000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optional: start from the previously found construction, if any.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = prev * (target_sum / prev.sum())
        val = _max_convolution_fft(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()
            print(f"[{time.time():.2f}] Loaded previous best, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – direct optimisation of power‑law (and with exponential tail)
    # ------------------------------------------------------------------
    def evaluate_power(p):
        seq = _shape_power(p, n)
        seq = seq * (target_sum / seq.sum())
        return _max_convolution_fft(seq)

    # coarse grid for p
    p_grid = np.linspace(0.05, 2.5, 25)
    for p in p_grid:
        val = evaluate_power(p)
        if val < best_score:
            best_score = val
            best_seq = _shape_power(p, n) * (target_sum / _shape_power(p, n).sum())
            best_p = p

    # refine best p with golden‑section
    lo = max(0.0, best_p - 0.6)
    hi = best_p + 0.6
    p_opt, p_val = _golden_section(evaluate_power, lo, hi, tol=1e-4, max_iter=25)
    if p_val < best_score:
        best_score = p_val
        best_seq = _shape_power(p_opt, n) * (target_sum / _shape_power(p_opt, n).sum())
        best_p = p_opt

    # ------------------------------------------------------------------
    # Phase 0b – power‑law + exponential tail (two‑parameter search)
    # ------------------------------------------------------------------
    def eval_power_exp(p, q):
        seq = _shape_power(p, n) * np.exp(-q * np.arange(n) / n)
        seq = seq * (target_sum / seq.sum())
        return _max_convolution_fft(seq)

    # coarse grid around the previous best p
    best_p2, best_q2 = best_p, 0.0
    for p in np.linspace(best_p - 0.4, best_p + 0.4, 9):
        for q in np.linspace(0.0, 8.0, 9):
            val = eval_power_exp(p, q)
            if val < best_score:
                best_score = val
                best_seq = (_shape_power(p, n) *
                            np.exp(-q * np.arange(n) / n))
                best_seq = best_seq * (target_sum / best_seq.sum())
                best_p2, best_q2 = p, q

    # refine q for a few p values using golden‑section
    for p_fixed in np.linspace(best_p2 - 0.2, best_p2 + 0.2, 5):
        lo_q, hi_q = max(0.0, best_q2 - 4.0), best_q2 + 4.0
        q_opt, q_val = _golden_section(
            lambda qq: eval_power_exp(p_fixed, qq), lo_q, hi_q,
            tol=1e-3, max_iter=20)
        if q_val < best_score:
            best_score = q_val
            best_seq = (_shape_power(p_fixed, n) *
                        np.exp(-q_opt * np.arange(n) / n))
            best_seq = best_seq * (target_sum / best_seq.sum())
            best_p2, best_q2 = p_fixed, q_opt

    # ------------------------------------------------------------------
    # Phase 0c – quick scan of beta‑like shapes (α,β)
    # ------------------------------------------------------------------
    for _ in range(200):
        alpha = rng.uniform(0.0, 5.0)
        beta = rng.uniform(0.0, 5.0)
        a = _shape_beta(alpha, beta, n)
        a = a * (target_sum / a.sum())
        val = _max_convolution_fft(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta shape improved: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random mixtures of analytic bases
    # ------------------------------------------------------------------
    sine_base = _shape_sine(n)

    # run until a few seconds remain for polishing
    while time.time() < deadline - 5.0:
        use_power = rng.random() < 0.6
        use_exp = rng.random() < 0.4
        use_sine = rng.random() < 0.5
        use_beta = rng.random() < 0.3

        a = np.zeros(n, dtype=np.float64)
        total_w = 0.0

        if use_power:
            p = rng.uniform(0.0, 2.5)
            w = rng.random()
            a += w * _shape_power(p, n)
            total_w += w
        if use_exp:
            q = rng.uniform(0.0, 10.0)
            w = rng.random()
            a += w * np.exp(-q * np.arange(n) / n)
            total_w += w
        if use_sine:
            w = rng.random()
            a += w * sine_base
            total_w += w
        if use_beta:
            alpha = rng.uniform(0.0, 5.0)
            beta = rng.uniform(0.0, 5.0)
            w = rng.random()
            a += w * _shape_beta(alpha, beta, n)
            total_w += w

        if total_w == 0.0:
            continue

        a = a * (target_sum / a.sum())
        val = _max_convolution_fft(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Random mixture new best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final Adam polish on the remaining time
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.2:
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      max_beta=5e5,
                                                      step_factor=0.7,
                                                      gamma=0.992)
        print(f"[{time.time():.2f}] Final Adam best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final validation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
