    n = len(a)
    a = a.copy()
    # ensure sum matches target_sum (maybe not exactly)
    current_sum = a.sum()

    best = a.copy()

    cur_max = best_max
    start_time = time.time()
    rng = np.random.default_rng()

def _conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]

"""
Optimiser for the non‑negative step‑function problem.

The algorithm consists of several phases:

1. **Seed generation** – a diverse pool of sequences (power‑law,
   exponential, random Dirichlet, previous best) is created and
   normalised to the ℓ¹‑simplex (∑a = √(2·n)).

2. **Exploration** – many short soft‑max gradient‑descent runs are
   performed on the initial length.  The best candidate is retained.

3. **Upscaling** – the best sequence is linearly interpolated to a
   larger length and a few deterministic reorderings (alternating
   order) are tested.

4. **Pairwise mass‑transfer local search** – a simulated‑annealing
   routine repeatedly moves a small amount of mass from an index that
   contributes strongly to the current peak of the auto‑convolution to
   an index with little mass.  The peak (the objective) is recomputed
   after each move and accepted according to a temperature schedule.

5. **Sub‑gradient descent on the exact peak** – using the exact
   gradient of the max‑convolution entry the sequence is refined
   further.

6. **High‑β soft‑max GD polishing** – a final long run of the smooth
   approximation with a very large β finishes the optimisation.

7. **Post‑processing** – trailing zeros are trimmed and the result is
   stored in ``height_sequence_1`` for possible reuse.

All vectors stay non‑negative and are projected onto the ℓ¹‑simplex,
hence the objective value returned by ``evaluate_sequence`` equals the
maximum entry of the auto‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple helper functions
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} then projected to the simplex."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = a * (target_sum / a.sum())
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then projected to the simplex."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a = a * (target_sum / a.sum())
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest left‑most, second largest right‑most, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    left = True
    for v in sorted_vals:
        if left:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        left = not left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution a*a."""
    return float(_conv_fft(a).max())


def _max_conv(a: np.ndarray) -> float:
    """Alias for clarity – returns the peak of the auto‑convolution."""
    return _max_convolution_fft(a)


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (equals the peak when Σa=√(2·n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution (smooth approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Momentum annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected GD on the smooth max‑convolution (soft‑max) objective."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _pairwise_local_search(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """Move small mass from a high‑contributing index to a low‑mass one."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_max = _max_conv(a)

    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # compute convolution and identify the peak(s)
        conv = _conv_fft(a)
        max_val = conv.max()
        peak_idxs = np.flatnonzero(conv == max_val)
        k = rng.choice(peak_idxs)

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = k - idx
        contrib = a[idx] * a[partner]
        donor = idx[np.argmax(contrib)]

        # choose a receiver among low‑mass indices
        low_thr = np.percentile(a, 20)
        low_cands = np.where(a <= low_thr)[0]
        if low_cands.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = rng.choice(low_cands)

        if donor == receiver:
            continue

        max_delta = 0.5 * a[donor]            # keep donor non‑negative
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = _project_to_simplex(a_new, target_sum)
        new_max = _max_conv(a_new)

        # geometric temperature decay
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = 0.5 * (1e-5 / 0.5) ** tau

        if new_max < max_val or rng.random() < np.exp(-(new_max - max_val) / max(temp, 1e-12)):
            a = a_new
            if new_max < best_max:
                best_max = new_max
                best = a_new.copy()
                print(f"[{time.time():.2f}] Pairwise new best={best_max:.6f}")

    return best, best_max


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak of the convolution
# ----------------------------------------------------------------------


def _subgradient_descent_max(a0: np.ndarray,
                             target_sum: float,
                             deadline: float,
                             max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the peak of a*a."""
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    best = a.copy()
    best_max = _max_conv(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    step_initial = target_sum * 0.5

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            partner = k - idx
            grad[idx] += a[partner]
            # diagonal term (a_i^2) contributes twice
            if k % 2 == 0:
                mid = k // 2
                if lo <= mid <= hi:
                    grad[mid] += a[mid]

        step = step_initial / np.sqrt(it + 1)
        a = a - step * grad
        a = _project_to_simplex(a, target_sum)

        cur_max = _max_conv(a)
        if cur_max < best_max:
            best_max = cur_max
            best = a.copy()
            print(f"[{time.time():.2f}] Sub‑max it={it:5d} best={best_max:.6f}")

        it += 1

    return best, best_max


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine performs:
      1. seed generation & short soft‑max GD exploration,
      2. up‑scaling + deterministic reorderings,
      3. pairwise mass‑transfer simulated annealing,
      4. exact sub‑gradient descent on the max‑convolution,
      5. high‑β soft‑max GD polishing,
      6. final trimming and registration.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose an initial length appropriate for the remaining budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_init = 5000
    elif budget_s >= 500:
        n_init = 4000
    else:
        n_init = 3000
    target_init = np.sqrt(2.0 * n_init)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seeds
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    for p in np.linspace(0.30, 0.80, 7):
        init_pool.append(_power_law_sequence(n_init, float(p), target_init))

    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_init, float(lam), target_init))

    for _ in range(10):
        a = rng.random(n_init)
        a = a * (target_init / a.sum())
        init_pool.append(a)

    # linear ramp
    lin = np.linspace(1.0, 0.0, n_init)
    lin = lin * (target_init / lin.sum())
    init_pool.append(lin)

    # previous best if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_init))

    # jittered copies to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_init * 0.008, size=n_init)
        seq_j = np.maximum(seq + jitter, 0.0)
        seq_j = _project_to_simplex(seq_j, target_init)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration with short gradient‑descent runs
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    explore_deadline = deadline - 180.0   # reserve ~3 min for polishing

    while time.time() < explore_deadline:
        # pick a seed (bias towards current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_init * 0.004, size=n_init)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_init)
        else:
            seed = rng.choice(init_pool)

        # a short soft‑max GD run (≈10 s)
        run_dead = min(time.time() + 10.0, explore_deadline)
        a_opt, val_opt = _annealed_gradient_descent(
            seed,
            run_dead,
            target_init,
            max_beta=1e5,
            step_factor=0.7,
            gamma=0.99,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration best={best_score:.6f}")

        if time.time() >= explore_deadline:
            break

    # Fallback – should never happen, but keep a safe default
    if best_seq is None:
        best_seq = _power_law_sequence(n_init, 0.55, target_init)
        best_score = _max_conv(best_seq)

    # ------------------------------------------------------------------
    # 4. Determine final length and upscale the current best
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 300 and budget_s >= 800:
        n_final = 12000
    elif remaining > 180:
        n_final = 10000
    elif remaining > 100:
        n_final = 8000
    else:
        n_final = n_init

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]
    best_seq = seq_big
    best_score = _max_conv(seq_big)
    for cand in candidates[1:]:
        sc = _max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled length={n_final} start max={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Pairwise mass‑transfer simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        mt_dead = time.time() + min(80.0, deadline - time.time() - 60.0)
        a_mt, val_mt = _pairwise_local_search(best_seq, target_final, mt_dead)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After pairwise search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Exact sub‑gradient descent on the peak
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        sub_dead = time.time() + min(60.0, deadline - time.time() - 30.0)
        a_sub, val_sub = _subgradient_descent_max(best_seq, target_final, sub_dead)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] After sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. High‑β soft‑max GD polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        gd_dead = deadline - 20.0
        a_gd, val_gd = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=5e5,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] Final GD polishing best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 9. Store globally and (optionally) log the official evaluation
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
