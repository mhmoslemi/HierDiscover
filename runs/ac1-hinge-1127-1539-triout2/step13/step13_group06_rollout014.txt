"""
Hybrid optimizer for the non‑negative step‑function problem.

Main ideas
----------

1. **Rich parametric warm‑start.**  Power‑law, exponentially tilted,
   triangular and cosine windows are swept on the simplex
   (∑a = √(2 n)).  The best candidate from this cheap grid is kept.

2. **Exact max‑convolution sub‑gradient descent.**  For a fixed vector a
   the objective reduces to `max(conv(a,a))` because the sum is fixed.
   Its sub‑gradient can be obtained analytically:
       ∇ max(conv) = 2 · conv_rev( a_rev , p )
   where `p` is the uniform distribution over the indices attaining the
   current maximum.  A simple projected sub‑gradient scheme with a
   diminishing step size is run for a large fraction of the time budget.

3. **Adam optimisation of a soft‑max surrogate.**  After the exact
   sub‑gradient phase a few Adam runs (soft‑max approximation of the max)
   are performed to escape shallow plateaus.

4. **Deterministic mass‑transfer (DMT2) and pairwise optimal moves.**
   These never increase the objective and polish the solution further.

5. **Final high‑β Adam polish** (soft‑max with a huge β) and an optional
   simulated‑annealing refinement.

All phases respect the global 1000 s budget and the best sequence found
is trimmed of trailing near‑zero entries before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(a)² (with sum = √(2 n) -> max(conv))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Linearly decreasing (triangular) window."""
    i = np.arange(n, dtype=np.float64)
    a = n - i
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Cosine bell (half‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (2 * (n - 1)))
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # jitter to leave plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (mask) utilities
# ----------------------------------------------------------------------


def _gradient_max_mask(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv(a,a)) using the mask of maximal indices."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=size)

    p = mask.astype(np.float64)
    s = p.sum()
    if s == 0.0:
        return np.zeros_like(a)
    p = p / s
    fp = np.fft.rfft(p, n=size)

    conv_rev = np.fft.irfft(fa_rev * fp, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad


def _subgradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a)
        max_val = conv.max()
        # mask of (almost) maximal entries
        mask = conv >= max_val - 1e-12

        grad = _gradient_max_mask(a, mask)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # diminishing step size (L2‑norm of the move = step)
        step = target_sum * 0.15 / np.sqrt(it)
        a = a - step * (grad / norm)
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.6f} score={best_val:.6f}"
            )

        # occasional jitter to get out of tiny plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (the “DMT2” refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray, deadline: float, target_sum: float, tol: float = 1e-12
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index with smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy pairwise optimal mass‑transfer (enhanced)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_frac: float = 0.03,
    receiver_frac: float = 0.03,
    min_improve: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """Greedy pairwise transfers with enlarged candidate pools."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest entries
        receiver_idx = np.argsort(a)[:nr]        # smallest entries

        # cache shifted views for the current candidates
        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)          # effect of one unit of transfer
                d_k = d[k]
                if d_k >= 0.0:
                    continue            # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # keep the padded view up‑to‑date (cheap slice assignment)
        base[pad_len + i] = a[i]
        base[pad_len + j] = a[j]

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(
                f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} score={best_score:.6f}"
            )

    return a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum
    self‑convolution.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)                # Σa = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 31)          # power‑law exponent
    q_grid = np.linspace(0.0, 6.0, 31)           # exponential tilt

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # extra simple families
    seq = _triangular_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    seq = _cosine_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # reuse any previously stored construction
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score, best_seq = val, seq_j.copy()

    print(
        f"[{time.time():.2f}] Initial best after parametric sweep = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase 1 – Exact max‑convolution sub‑gradient descent
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.40
    best_seq, best_score = _subgradient_descent(best_seq, target_sum, subgrad_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style optimisation of the soft‑max surrogate (restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        run_deadline = time.time() + min(30.0, remaining * 0.3)
        a_opt, val_opt = _adam_gradient_descent(
            start_vec,
            run_deadline,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT2 improved best = {best_score:.6f}")

        a_rand, val_rand = _global_random_mass_transfer(
            best_seq, deadline, target_sum
        )
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best = {best_score:.6f}")

        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase 4 – Enhanced pairwise optimal transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_pair, val_pair = _pairwise_optimal_transfer(
            best_seq, deadline, target_sum, donor_frac=0.03, receiver_frac=0.03
        )
        if val_pair < best_score:
            best_score, best_seq = val_pair, a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final polishing with a very large β (soft‑max)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for logging purposes
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
