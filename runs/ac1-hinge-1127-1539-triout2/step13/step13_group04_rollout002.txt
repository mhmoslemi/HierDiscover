def _simulated_annealing(a0, target_sum, deadline):
    a = a0.copy()
    n = a.size
    rng = np.random.default_rng()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max

    # Temperature schedule:
    total_time = deadline - time.time()
    T0 = cur_max * 0.05  # 5% of current max
    while time.time() < deadline:
        elapsed = time.time() - start
        remaining = deadline - time.time()
        # temperature linear decay
        T = T0 * (remaining / total_time)

        # Select donor, receiver
        # top fraction:
        top_frac = 0.02
        low_frac = 0.02
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argpartition(-a, k_top-1)[:k_top]  # fastest
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argpartition(a, k_low-1)[:k_low]
        receiver = idx_asc[rng.integers(k_low)]
        if donor == receiver:
            continue
        max_delta = a[donor] * 0.1  # move up to 10% of donor's mass
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # apply move
        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta
        # ensure positivity (donor >=0 automatically)
        # compute max
        new_max = _max_convolution_fft(a_candidate)
        dscore = new_max - cur_max
        if dscore <= 0 or rng.random() < np.exp(-dscore / max(T, 1e-12)):
            a = a_candidate
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_candidate.copy()
                print(f"[{time.time():.2f}] SA new best {best_max:.6f}")

"""
Improved minimiser for the maximal self‑convolution of a non‑negative step function.

The algorithm proceeds in several stages:
1.  A short parametric sweep (power‑law, tapered power‑law, Gaussian) to obtain
    a decent starting point for the given problem size n.
2.  A multi‑active‑set sub‑gradient descent that uses the average sub‑gradient
    of *all* convolution indices attaining the current maximum.  This yields
    larger, more reliable steps than taking a single active index.
3.  A simple simulated‑annealing / hill‑climbing phase that repeatedly moves a
    small amount of mass from a “heavy” location to a “light” one and accepts
    worsening moves with a temperature schedule.  It explores directions that
    are not captured by the pure gradient steps.
4.  A final high‑β Adam optimiser (soft‑max approximation of the max) for fine‑tuning.
5.  Optional penalty‑descent (as in the previous code) to push the bound a little
    lower if time permits.

All updates keep the vector on the simplex
    {a_i ≥ 0, Σ a_i = √(2·n)},
so the value returned by `evaluate_sequence` is exactly the maximal
self‑convolution.  The routine respects the 1000 s wall‑time limit and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length 2·n‑1) via FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y (length len(x)+len(y)-1) via FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean distance)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (fast initialisation)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (max‑convolution)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective value: maximal self‑convolution entry (already the evaluation score)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv). Returns (grad, conv)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Multi‑active‑set sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _subgradient_descent_multi(a0: np.ndarray,
                               target_sum: float,
                               deadline: float,
                               max_iter: int = 8000) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that averages gradients over *all* maximal convolution indices."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second‑largest value to estimate a useful step size
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        gap = max_val - second_max
        if gap < 1e-14:
            break

        # accumulate gradient contributions from all active indices
        g = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution 2*a[k-idx] as in the single‑index case
            g[idx] += 2.0 * a[k - idx]
        # average to obtain a valid sub‑gradient
        if idx_max.size > 1:
            g /= idx_max.size

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            # inject a tiny random move
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        # step size similar to classic sub‑gradient rule
        step = 0.9 * gap / (norm2 + 1e-12)
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # back‑tracking line search (max 6 reductions)
        improved = False
        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                improved = True
                print(f"[{time.time():.2f}] SubgradMulti it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # jitter to escape flat region
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.0005, size=n), target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing / random mass transfer
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Hill‑climbing with occasional uphill moves (simple SA)."""
    a = a0.copy()
    n = a.size
    rng = np.random.default_rng()
    cur_max = _max_convolution_fft(a)
    best_a = a.copy()
    best_max = cur_max

    start = time.time()
    total_time = deadline - start
    # initial temperature as a fraction of the current max
    T0 = max(cur_max * 0.04, 1e-4)

    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear cooling
        T = T0 * (remaining / total_time)

        # pick donor with probability proportional to its mass (heavier points move first)
        prob = a / a.sum()
        donor = rng.choice(n, p=prob)

        # pick receiver uniformly (or from low‑mass region)
        receiver = rng.integers(n)
        if receiver == donor:
            continue

        # magnitude of the transfer
        max_delta = a[donor] * 0.12  # up to 12 % of donor's mass
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # apply move
        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        # Evaluate
        new_max = _max_convolution_fft(a_candidate)
        dscore = new_max - cur_max

        if dscore <= 0.0 or rng.random() < np.exp(-dscore / max(T, 1e-12)):
            a = a_candidate
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a_candidate.copy()
                print(f"[{time.time():.2f}] SA new best {best_max:.6f}")

    return best_a, best_max


# ----------------------------------------------------------------------
# Penalty‑descent (optional final push)
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    """Gradient descent on the penalty (max(conv)-target_max)_+."""
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty vector (only entries above target_max)
        p = conv - target_max
        p[p < 0] = 0.0

        # gradient = 2 * (p * a_rev)_{n-1 … n-1+n-1}
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)

        it += 1

    return best_a, best_max

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal maximal self‑convolution.
    The function respects a wall‑clock budget (seconds) and returns the best
    sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # 0.5 s safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)   # Σ a_i = √(2·n)

    # --------------------------------------------------------------
    # Phase 0 – quick parametric sweep for a decent start
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # (a) power‑law grid
    for p in np.linspace(0.30, 0.85, 15):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) tapered power‑law grid
    for p in np.linspace(0.30, 0.85, 12):
        for q in np.linspace(0.0, 4.0, 12):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) Gaussian bumps with varying σ
    for sigma in np.linspace(n / 20, n / 5, 7):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) incorporate previous construction if available
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) small jittered population around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After Phase 0 best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – multi‑active‑set sub‑gradient descent
    # --------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.30
    best_seq, best_score = _subgradient_descent_multi(best_seq,

    print(f"[{time.time():.2f}] After SubgradMulti best score = {best_score:.6f}")

    # --------------------------------------------------------------

                                                gamma=0.995)

        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
