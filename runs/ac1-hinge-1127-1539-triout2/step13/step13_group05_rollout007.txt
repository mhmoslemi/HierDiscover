def solve_lp_for_g(a, M, target_sum):
    # Build constraints.
    n = len(a)
    K = 2*n - 1

    # Build A matrix (K x n) dense.
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        A[i: i+n, i] = a[i]   # rows i...i+n-1 get a_i.
    # Upper bound constraints: A @ g <= M
    # sum(g) <= target_sum
    A_ub = np.vstack([A, np.ones((1, n))])
    b_ub = np.concatenate([np.full(K, M), [target_sum]])

    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)

    # bounds for each variable: (0, target_sum) maybe.
    bounds = [(0, target_sum) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        g = res.x
    else:
        # fallback: return a copy.
        g = a.copy()
    return g
```

"""
Improved optimizer for the max self‑convolution problem.

The algorithm consists of two stages:

1. **Small‑scale alternating LP+mixing**  
   - Work with a reduced size (≈2000) where a linear program can be
     solved quickly.  
   - Given the current sequence `a` (with Σa = √(2n)), we compute the
     current peak `M = max(a*a)`.  
   - Solve the LP: maximise Σg subject to `a * g ≤ M` (linear
     convolution constraints) and `Σg ≤ √(2n)`.  The solution `g₀` is
     scaled to have the exact sum and then mixed with `a` (convex
     combination).  The mixing coefficient `t` is chosen from a small
     grid to give the smallest new peak.  This step strictly reduces
     the objective and is repeated while time permits.

2. **Large‑scale polishing**  
   - Upsample the small‑scale solution to the target length
     (5000–12000 depending on the time budget).  
   - Apply the multi‑phase refinement from the reference code:
     projected Adam GD (soft‑max), deterministic mass‑transfer,
     global random moves, a new global‑peak transfer, pairwise optimal
     transfer and a final high‑β Adam polish.  

All phases respect the overall time budget (default 1000 s).  The
resulting sequence is projected onto the simplex Σa = √(2n),
trimmed of trailing near‑zero entries and returned as a plain Python
list.
"""

import time
import numpy as np

# optional SciPy for the LP phase
try:
    from scipy.optimize import linprog
    _SCIPY_LP = True
except Exception:                      # pragma: no cover
    _SCIPY_LP = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(b)/(sum(a)^2) – with Σa fixed this equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the cheap warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑norm."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple Adam restarts with jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy donor→receiver moves that strictly lower the current peak."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick a peak index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (new refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Reduce the convolution peak by moving a *fraction* of the total donor
    mass to a single weak receiver (computed analytically).
    """
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second maximum (stop when peak essentially flat)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donors scored by contribution to the peak
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)   # contribution of donors alone

        # candidate receivers = smallest partner values
        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # shifted view
            diff = donor_mass * shift_r - conv_donors   # linearized change
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            # predicted new peak (linear model)
            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} "
                  f"α={best_alpha:.4g} new_score={best_score:.6f}")
        it += 1

    return best_a, best_score

def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise donor→receiver moves that improve the current peak."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)       # effect of moving mass i→j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} "
                  f"Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Small‑scale LP + mixing (new stage)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(a: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """Linear program: maximise Σg subject to a*g ≤ M and Σg ≤ target_sum."""
    n = a.size
    K = 2 * n - 1
    # Build the convolution matrix A (K × n) where A[k,i] = a_i if 0≤k-i≤n-1 else 0
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]

    # Upper‑bound constraints a * g ≤ M
    A_ub = np.vstack([A, np.ones((1, n), dtype=np.float64)])
    b_ub = np.concatenate([np.full(K, M, dtype=np.float64),
                           np.array([target_sum], dtype=np.float64)])

    # Objective: maximise Σg  → minimise -Σg
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, target_sum) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – return the current vector (no improvement)
        return a.copy()
    return res.x


def _lp_mix_iteration(a: np.ndarray,
                      target_sum: float,
                      t_grid: list[float],
                      deadline: float) -> tuple[np.ndarray, float]:
    """One LP‑solve + best mixing step."""
    n = a.size
    M = _max_convolution_fft(a)
    # solve LP for maximal donor vector g
    g0 = _solve_lp_max_sum(a, M, target_sum)
    sum_g0 = g0.sum()
    if sum_g0 < 1e-12:
        return a, _score(a)

    # scale to exact sum
    scale = target_sum / sum_g0
    g = g0 * scale

    # try several mixing coefficients and keep the best
    best_score = _score(a)
    best_a = a.copy()
    for t in t_grid:
        a_new = (1.0 - t) * a + t * g
        # projection not needed if sums are exact, but numerical drift may appear
        a_new = _project_to_simplex(a_new, target_sum)
        val = _score(a_new)
        if val < best_score:
            best_score = val
            best_a = a_new.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling utilities
# ----------------------------------------------------------------------


def _upsample(seq_small: np.ndarray, N_big: int) -> np.ndarray:
    """Linear interpolation from small to big length."""
    n_small = seq_small.size
    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, N_big)
    up = np.interp(x_big, x_small, seq_small)
    return up

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the max self‑convolution.
    The routine respects the given time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Decide final problem size based on budget
    # ------------------------------------------------------------------
    if budget_s >= 800:
        N_final = 12000
    elif budget_s >= 400:
        N_final = 8000
    else:
        N_final = 5000
    target_sum_final = np.sqrt(2.0 * N_final)

    # ------------------------------------------------------------------
    # Small‑scale optimisation (LP + mixing)
    # ------------------------------------------------------------------
    n_small = min(2000, max(1000, N_final // 5))   # e.g. 2000 for N=12000
    target_sum_small = np.sqrt(2.0 * n_small)

    # 0. parametric warm‑start on the small scale
    best_small = _power_law_sequence(n_small, 0.6, target_sum_small)
    best_score = _score(best_small)

    # sweep a few power‑law / exponential families
    p_grid = np.linspace(0.30, 0.85, 13)
    q_grid = np.linspace(0.0, 6.0, 7)
    for p in p_grid:
        seq = _power_law_sequence(n_small, float(p), target_sum_small)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_small = seq.copy()
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n_small, float(p), float(q), target_sum_small)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_small = seq.copy()
    sigma_grid = np.linspace(0.15 * n_small, 0.45 * n_small, 5)
    for sigma in sigma_grid:
        seq = _gaussian_sequence(n_small, float(sigma), target_sum_small)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_small = seq.copy()

    # 1. optional: incorporate a previously saved construction if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n_small:
            # reshape by interpolation
            prev = _upsample(prev, n_small)
        prev = _project_to_simplex(prev, target_sum_small)
        val = _score(prev)
        if val < best_score:
            best_score = val
            best_small = prev.copy()

    print(f"[{time.time():.2f}] Small‑scale initial best score = {best_score:.6f}")

    # 2. LP + mixing iterations (time‑controlled)
    t_grid = [0.03, 0.06, 0.10, 0.15, 0.20]
    lp_iter = 0
    while time.time() < deadline - 40.0 and lp_iter < 30 and _SCIPY_LP:
        a_new, val_new = _lp_mix_iteration(best_small,
                                            target_sum_small,
                                            t_grid,
                                            deadline)
        if val_new < best_score - 1e-12:
            best_score = val_new
            best_small = a_new.copy()
            print(f"[{time.time():.2f}] LP‑mix it={lp_iter} improved score={best_score:.6f}")
        # occasional gradient refinement on the small scale
        if lp_iter % 5 == 0:
            a_ref, val_ref = _gradient_descent_with_restarts(best_small,
                                                             target_sum_small,
                                                             deadline - 5.0)
            if val_ref < best_score:
                best_score = val_ref
                best_small = a_ref.copy()
                print(f"[{time.time():.2f}] Small‑scale GD after LP it={lp_iter} new best={best_score:.6f}")
        lp_iter += 1

    # ------------------------------------------------------------------
    # Upsample to final size and re‑normalise
    # ------------------------------------------------------------------
    a_big = _upsample(best_small, N_final)
    a_big = _project_to_simplex(a_big, target_sum_final)

    # ------------------------------------------------------------------
    # Large‑scale polishing (stage similar to the reference code)
    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        a_tmp, val_tmp = _gradient_descent_with_restarts(a_big,
                                                         target_sum_final,
                                                         gd_deadline)
        if val_tmp < best_score:
            best_score = val_tmp
            a_big = a_tmp.copy()
            print(f"[{time.time():.2f}] Large‑scale GD improved best={best_score:.6f}")

    # Phase B – Exact deterministic mass‑transfer
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(a_big,
                                                             deadline,
                                                             target_sum_final)
        if val_ref < best_score:
            best_score = val_ref
            a_big = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best={best_score:.6f}")

        a_rand, val_rand = _global_random_mass_transfer(a_big,
                                                        deadline,
                                                        target_sum_final)
        if val_rand < best_score:
            best_score = val_rand
            a_big = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best={best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # Phase C – New global peak‑mass transfer
    if time.time() < deadline - 6.0:
        a_peak, val_peak = _global_peak_mass_transfer(a_big,
                                                      deadline,
                                                      target_sum_final,
                                                      donor_frac=0.03,
                                                      receiver_frac=0.03)
        if val_peak < best_score:
            best_score = val_peak
            a_big = a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best={best_score:.6f}")

    # Phase D – Pairwise optimal transfer
    if time.time() < deadline - 3.0:
        a_pair, val_pair = _pairwise_optimal_transfer(a_big,
                                                      deadline,
                                                      target_sum_final,
                                                      donor_frac=0.02,
                                                      receiver_frac=0.02)
        if val_pair < best_score:
            best_score = val_pair
            a_big = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best={best_score:.6f}")

    # Phase E – Final high‑β Adam polishing
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(a_big,
                                                deadline,
                                                target_sum_final,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            a_big = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(a_big, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
