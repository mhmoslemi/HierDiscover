def _grad_max(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    # compute mask of max convolve
    conv = _conv_fft(a)
    max_val = conv.max()
    mask = np.zeros_like(conv)
    eps = max_val * 1e-12 + 1e-12
    mask[conv >= max_val - eps] = 1.0
    # we can normalize mask to sum to 1 (to get subgradient)
    mask_sum = mask.sum()
    if mask_sum == 0:
        return np.zeros_like(a)
    mask /= mask_sum
    # compute gradient via convolution of mask with reversed a (or a_rev)
    a_rev = a[::-1]
    mask_fft = np.fft.rfft(mask, n=size)
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * mask_fft, n=size)[: 2*n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad

"""
Step‑function optimizer – Gradient‑driven refinement.

The optimizer builds a candidate from simple parametric families,
then improves it with four complementary phases:

1. **Gradient‑Mass‑Transfer (GMT)** – uses the sub‑gradient of the
   max‑self‑convolution (obtained from a soft‑max with β→∞) to move
   mass from indices that contribute most to the peak toward those
   that contribute least.  This is a cheap, deterministic hill‑climb.

2. **Exact Deterministic Mass‑Transfer (DMT)** – the “exact DMT2”
   routine from the reference implementation, which drops the
   current peak to the second‑largest value in a greedy fashion.

3. **Simulated Annealing (SA)** – stochastic donor/receiver moves
   on the simplex with a Metropolis acceptance rule and an exponential
   cooling schedule.

4. **Second GMT + Adam polishing** – another gradient pass followed
   by Adam on a soft‑max surrogate (large β) to fine‑tune the solution.

The time budget (default 1000 s) is split among the phases.
The final sequence is projected onto the simplex
{a_i≥0, Σa_i = √(2 n)} and trailing near‑zero entries are trimmed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used for the quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # stop if already tight
        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic mass‑transfer with Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # initial temperature (fraction of the current objective)
    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.05                # donor / receiver from top/bottom 5 %
    top_k = max(1, int(top_frac * n))

    # pre‑compute sorted orders periodically
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.3
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _max_convolution_fft(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑Mass‑Transfer (sub‑gradient of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_mass_transfer(a0: np.ndarray,
                           target_sum: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small amount of mass from a high‑gradient index
    (strongly influencing the current peak) toward a low‑gradient index.
    The gradient is the sub‑gradient of max(a∗a) obtained via a
    normalized max‑mask convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    cur_max = _max_convolution_fft(a)
    best_max = cur_max
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = np.zeros_like(conv, dtype=np.float64)
        mask[conv >= max_val - eps] = 1.0
        mask_sum = mask.sum()
        if mask_sum == 0:
            break
        mask /= mask_sum          # normalized sub‑gradient probability

        a_rev = a[::-1]
        mask_fft = np.fft.rfft(mask, n=size)
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * mask_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # sub‑gradient of max

        # donor = index with largest gradient among those that still hold mass
        pos = a > 1e-14
        if not np.any(pos):
            break
        donor_candidates = np.where(pos)[0]
        donor = donor_candidates[np.argmax(grad[donor_candidates])]

        # receiver = index with smallest gradient (different from donor)
        receiver = np.argmin(grad)
        if receiver == donor:
            # pick the second smallest
            sorted_idx = np.argsort(grad)
            for r in sorted_idx:
                if r != donor:
                    receiver = r
                    break

        # transfer fraction that shrinks over time
        frac = 0.25 / np.sqrt(it + 1)
        delta = a[donor] * frac
        if delta < 1e-14:
            break

        # perform tentative move
        a[donor] -= delta
        a[receiver] += delta
        a[a < 0] = 0.0  # safety

        new_max = _max_convolution_fft(a)

        # accept only if the peak does not increase
        if new_max <= max_val + 1e-12:
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] GMT improved max={best_max:.6f} it={it}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a[a < 0] = 0.0

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal
    self‑convolution.  The routine respects the time budget and returns
    the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – larger n gives more degrees of freedom
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n) makes the objective = max_conv

    # -------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search
    # -------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # grids for power‑law exponent and exponential tail
    p_grid = np.linspace(0.20, 1.20, 25)
    q_grid = np.linspace(0.0, 8.0, 25)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _max_convolution_fft(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) reuse previous construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _max_convolution_fft(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (e) small random jittered attempts around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best max={best_score:.6f}")

    # -------------------------------------------------------------
    # Allocate time for the remaining phases
    # -------------------------------------------------------------
    remaining = deadline - time.time()
    t_gmt1 = remaining * 0.15
    t_dmt  = remaining * 0.20
    t_sa   = remaining * 0.30
    t_gmt2 = remaining * 0.15
    t_adam = remaining * 0.15

    # -------------------------------------------------------------
    # Phase 1 – Gradient Mass Transfer (first pass)
    # -------------------------------------------------------------
    seq, cur_max = _gradient_mass_transfer(best_seq, target_sum, time.time() + t_gmt1)
    if cur_max < best_score:
        best_score = cur_max
        best_seq = seq.copy()
    print(f"[{time.time():.2f}] After GMT1 best max={best_score:.6f}")

    # -------------------------------------------------------------
    # Phase 2 – Deterministic Mass Transfer
    # -------------------------------------------------------------
    seq, cur_max = _deterministic_mass_transfer_exact2(best_seq,
                                                       time.time() + t_dmt,
                                                       target_sum)
    if cur_max < best_score:
        best_score = cur_max
        best_seq = seq.copy()
    print(f"[{time.time():.2f}] After DMT best max={best_score:.6f}")

    # -------------------------------------------------------------
    # Phase 3 – Simulated Annealing
    # -------------------------------------------------------------
    seq, cur_max = _simulated_annealing(best_seq, target_sum, time.time() + t_sa)
    if cur_max < best_score:
        best_score = cur_max
        best_seq = seq.copy()
    print(f"[{time.time():.2f}] After SA best max={best_score:.6f}")

    # -------------------------------------------------------------
    # Phase 4 – Gradient Mass Transfer (second pass)
    # -------------------------------------------------------------
    seq, cur_max = _gradient_mass_transfer(best_seq, target_sum, time.time() + t_gmt2)
    if cur_max < best_score:
        best_score = cur_max
        best_seq = seq.copy()
    print(f"[{time.time():.2f}] After GMT2 best max={best_score:.6f}")

    # -------------------------------------------------------------
    # Phase 5 – Adam polishing (soft‑max surrogate)
    # -------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        seq, cur_max = _adam_gradient_descent(best_seq,
                                              deadline,
                                              target_sum,
                                              max_beta=5e5,
                                              step_factor=0.5,
                                              gamma=0.99)
        if cur_max < best_score:
            best_score = cur_max
            best_seq = seq.copy()
        print(f"[{time.time():.2f}] Final Adam best max={best_score:.6f}")

    # -------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # -------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
