"""
Hybrid optimiser for the step‑function benchmark.

The algorithm consists of three stages:

1. **Rapid shape finding (small n).**  
   A diverse pool of analytic families (power‑law, exponential,
   Hann window, sine, random) is created for n≈5000.  Short
   annealed‑gradient runs (soft‑max approximation of the max‑convolution)
   are performed from many seeds; the best shape is kept.

2. **Scaling & ordering (large n).**  
   The best small shape is interpolated to a larger length (≈20000‑25000)
   and projected onto the ℓ¹‑simplex Σa=√(2n).  Several deterministic
   reorderings (alternating, even‑spread, reversal) are tried and the
   order with the smallest max‑convolution is selected.

3. **Long sub‑gradient refinement + final mass‑transfer.**  
   An exact sub‑gradient descent on the true max‑convolution (with
   momentum) runs for the bulk of the remaining time, followed by a
   brief simulated‑annealing mass‑transfer step to escape possible
   plateaus.

All operations use FFT‑based convolutions (O(n log n)), and every
iteration projects onto the simplex to obey the non‑negativity and
ℓ¹‑normalisation constraints required by `evaluate_sequence`.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector `v` onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (analytic families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _hann_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·i/(n‑1))."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately from left and right."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    l, r = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Scatter the sorted values with a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _reverse_order(seq: np.ndarray) -> np.ndarray:
    return seq[::-1]


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 8e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on a soft‑max surrogate of max(b).
    Returns the best vector encountered and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = conv.max()
        # soft‑max probabilities over the convolution entries
        exps = np.exp(beta * (conv - max_b))
        p = exps / np.sum(exps)

        # gradient of the soft‑max objective (chain rule)
        a_rev = a[::-1]
        ar_fft = np.fft.rfft(a_rev, n=size)
        pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)  # indicator of binding indices

        # gradient = 2 * (I * a_rev) restricted to centre n entries
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # tiny jitter every few hundred iterates
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mass‑transfer simulated‑annealing (greedy pair moves)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.3,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Move a random amount of mass from a “donor” to a “receiver” pair."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start = time.time()
    n = a.shape[0]

    # pre‑compute once (will be recomputed after each move)
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where the convolution attains its current maximum
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to rediscover new basins
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time budget,
    uses a fast small‑scale gradient phase followed by a large‑scale
    sub‑gradient refinement, and finally returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – small‑scale exploration (n≈5 k)
    # ------------------------------------------------------------------
    n_small = 5000 if budget_s >= 800 else (4000 if budget_s >= 500 else 3000)
    target_small = math.sqrt(2.0 * n_small)

    # Build a diverse initial pool
    init_pool: List[np.ndarray] = []

    for p in np.linspace(0.1, 2.2, 12):
        init_pool.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.0003, 0.04, 10):
        init_pool.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(8):
        init_pool.append(_hann_sequence(n_small, target_small))
    for _ in range(8):
        init_pool.append(_sin_half_sequence(n_small, target_small))
    for _ in range(12):
        init_pool.append(_random_simplex(n_small, target_small, rng))

    # reuse previous run if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_small))

    # jittered variants of the pool
    jittered = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_small * 0.03, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        jittered.append(seq_j)
    init_pool.extend(jittered)

    best_seq = None
    best_score = np.inf

    # allocate ~20 % of the total budget to this phase (or 150 s, whichever smaller)
    phase1_deadline = min(start_time + 150.0, start_time + 0.20 * budget_s)

    while time.time() < phase1_deadline - 10.0:
        # seed selection – either a fresh random candidate or a jittered copy of current best
        if best_seq is not None and rng.random() < 0.65:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_small * 0.005, size=n_small)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 12.0, phase1_deadline - 5.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_small,
            max_beta=9e5, step_factor=0.9, gamma=0.98)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Phase‑1 new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a larger length and reorder
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_small, 0.6, target_small)

    remaining = deadline - time.time()
    # decide target length for the large‑scale refinement
    if remaining > 300.0:
        n_large = 25000
    elif remaining > 180.0:
        n_large = 20000
    elif remaining > 120.0:
        n_large = 15000
    elif remaining > 80.0:
        n_large = 12000
    else:
        n_large = n_small

    target_large = math.sqrt(2.0 * n_large)

    # interpolate the best small‑scale shape
    seq_large = _resize_sequence(best_seq, n_large)
    seq_large = _project_to_simplex(seq_large, target_large)

    # try a few deterministic reorderings and keep the best
    candidates = [
        seq_large,
        _alternating_order(seq_large),
        _even_spread_order(seq_large),
        _reverse_order(seq_large),
    ]
    seq_large = min(candidates, key=_max_convolution_fft)
    best_score = _max_convolution_fft(seq_large)
    print(f"[{time.time():.2f}] Upscaled to {n_large} elems, start max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long sub‑gradient descent on the large problem
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_deadline = deadline - 6.0
        a_sub, val_sub = _subgradient_descent(
            seq_large, sub_deadline, target_large,
            step_factor=0.38, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_large = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad finished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final short mass‑transfer pass (optional, if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        mt_deadline = deadline - 1.0
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_large, mt_deadline, target_large,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_large = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq_large, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose sequence for a potential next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
