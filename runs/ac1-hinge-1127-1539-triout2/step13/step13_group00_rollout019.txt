"""
Convex‑analytic smooth‑max gradient descent for the non‑negative step‑function
problem.

The objective (with the ℓ¹‑normalisation ∑a_i = √(2 n)) reduces to minimising
the largest entry of the self‑convolution a*a.  By writing a_i = p_i·√(2 n)
with p being a probability vector (∑p_i=1) the score equals

    score = 2·n·max_k (p∗p)_k .

We optimise p on the simplex using a softmax parametrisation
z_i = log(p_i) (so p_i = exp(z_i)/∑exp(z)).  The nonsmooth max is
approximated by a smooth log‑sum‑exp (the “soft‑max”), which gives a
differentiable surrogate

    L(z) = 2·n·log‑sum‑exp(α·(p∗p))/α .

L is minimised by Adam on z, with a continuation schedule that raises the
temperature α from 5 to 40.  The convolution and its gradient are evaluated
with FFTs (O(n log n)).  After the long‑run Adam phase a short deterministic

def _conv_full(p: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution of ``p`` with itself, computed via FFT.
    ``size`` must be a power of two ≥ len(p) + len(p) - 1.
    Returns the real part (length 2*len(p)-1).
    """
    pf = np.fft.rfft(p, n=size)
    conv = np.fft.irfft(pf * pf, n=size)
    return conv[: 2 * p.shape[0] - 1]


def _conv_cross(p: np.ndarray, q_rev: np.ndarray, size: int) -> np.ndarray:
    """
    Full convolution of ``p`` with the reversed array ``q_rev``.
    ``size`` must be a power of two ≥ len(p)+len(q_rev)-1.
    Returns the real part (length len(p)+len(q_rev)-1).
    """
    pf = np.fft.rfft(p, n=size)
    qf = np.fft.rfft(q_rev, n=size)
    conv = np.fft.irfft(pf * qf, n=size)
    return conv[: p.shape[0] + q_rev.shape[0] - 1]


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    conv = _conv_full(a, size)
    return float(conv.max())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]

    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

        elapsed = time.time() - start_time
        tau = elapsed / max(deadline - start_time, 1e-12)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = 2.0 * n * np.max(conv)          # exact max‑convolution

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sequence factories (return probability vectors that sum to 1)
# ----------------------------------------------------------------------


def _power_law_prob(n: int, p: float) -> np.ndarray:
    """p_i ∝ (i+1)^(-p) then normalised to sum 1."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def _exponential_prob(n: int, lam: float) -> np.ndarray:
    """p_i ∝ exp(-lam·i) then normalised to sum 1."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _random_simplex_prob(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a /= a.sum()
    return a


def _resize_prob(p: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_len = p.shape[0]
    if new_len == old_len:
        return p.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    q = np.interp(new_x, old_x, p)
    q = np.maximum(q, 0.0)
    q /= q.sum() if q.sum() > 0 else 1.0
    return q


# ----------------------------------------------------------------------
# Main optimiser (smooth‑max + Adam)
# ----------------------------------------------------------------------


def _adam_optimize(
    z0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits ``z`` (softmax parametrisation of the
    probability vector).  Returns (best_logits, best_score) with ``best_score``
    being the exact maximal convolution value (the quantity returned by
    evaluate_sequence).
    """
    size2 = 1 << ((3 * n - 2).bit_length())         # for p * (rev w)
    L = 2 * n - 1
    L_minus_one = L - 1

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2

    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_z = None
    best_score = np.inf
    it = 0
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        # continuation on α
        alpha = 5.0 + (40.0 - 5.0) * min(elapsed / total, 1.0)

        # softmax to get probability vector p
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        sum_exp_z = np.sum(exp_z)
        p = exp_z / sum_exp_z

        # convolution p∗p (via FFT size2)
        pf = np.fft.rfft(p, n=size2)
        conv_full = np.fft.irfft(pf * pf, n=size2)
        conv = conv_full[:L]                     # length 2n‑1, all non‑negative up to round‑off

        # smooth‑max weights w = softmax(α·conv)
        max_ac = np.max(alpha * conv)
        exp_ac = np.exp(alpha * conv - max_ac)
        sum_exp_ac = np.sum(exp_ac)
        w = exp_ac / sum_exp_ac                  # shape L, sums to 1

        # gradient w.r.t. p: g = 2 * (p * w_rev) corr
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=size2)
        conv_rev_full = np.fft.irfft(pf * wf, n=size2)
        conv_rev = conv_rev_full[: n + L - 1]    # length 3n‑2

        g = 2.0 * conv_rev[L_minus_one - np.arange(n)]

        # gradient w.r.t. logits (softmax chain rule)
        avg = np.dot(p, g)
        grad = p * (g - avg)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

        # current exact score (max‑convolution of a = p·target_sum)
        cur_score = 2.0 * n * np.max(conv)       # because target_sum² = 2 n

        if cur_score < best_score:
            best_score = cur_score
            best_z = z.copy()
            # occasional progress print
            print(f"[{time.time():.2f}] iter {it}, score {best_score:.6f}")

        # occasional jitter to escape plateaus (tiny Gaussian)
        if it % 2000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            z += jitter

    # if no improvement happened, fall back to the initial logits
    if best_z is None:
        best_z = z0.copy()
        best_score = 2.0 * n * np.max(_conv_full(np.exp(z0 - np.max(z0)), size2))

    return best_z, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimisation: initialise from several analytic / random seeds,
    improve the logits with Adam on a smooth‑max surrogate, polish with a
    deterministic mass‑transfer routine and finally try a few reorderings.
    Returns the best step‑function (list of non‑negative heights).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives a richer design space)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)               # = √(2 n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a bank of seed probability vectors (sum = 1)
    # ------------------------------------------------------------------
    seed_probs = []

    # power‑law seeds
    for p in np.linspace(0.2, 1.0, 9):
        seed_probs.append(_power_law_prob(n, float(p)))

    # exponential seeds
    for lam in np.linspace(0.001, 0.02, 6):
        seed_probs.append(_exponential_prob(n, float(lam)))

    # uniform random Dirichlet‑like seeds
    for _ in range(10):
        seed_probs.append(_random_simplex_prob(n, rng))

    # previous best (if available)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.size > 0:
            # convert to probability, resize, re‑normalise
            prob = prev / prev.sum()
            prob = _resize_prob(prob, n)
            seed_probs.append(prob)

    # ------------------------------------------------------------------
    # Evaluate seeds (exact score) and keep the best as starting point
    # ------------------------------------------------------------------
    best_score = np.inf
    best_prob = None
    for prob in seed_probs:
        a = prob * target_sum
        score = 2.0 * n * np.max(_conv_fft(a))   # exact evaluation (same as evaluate_sequence)
        if score < best_score:
            best_score = score
            best_prob = prob.copy()

    print(f"[{time.time():.2f}] Initial best score from seeds = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert the best probability vector to logits (softmax parameterisation)
    # ------------------------------------------------------------------
    eps = 1e-12
    init_logits = np.log(best_prob + eps)

    # ------------------------------------------------------------------
    # Adam on smooth‑max surrogate (major part of the budget)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 20.0               # leave a little time for polishing
    if adam_deadline > time.time():
        best_logits, best_score = _adam_optimize(
            init_logits, n, target_sum, adam_deadline)
        best_prob = np.exp(best_logits - np.max(best_logits))
        best_prob /= best_prob.sum()
        print(f"[{time.time():.2f}] After Adam best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer polish (uses exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        polish_dead = time.time() + min(12.0, deadline - time.time() - 3.0)
        a0 = best_prob * target_sum
        polished_a, polished_score = _deterministic_mass_transfer(
            a0, polish_dead, target_sum,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_a.copy()
            print(f"[{time.time():.2f}] After polishing score = {best_score:.6f}")
        else:
            best_seq = a0.copy()
    else:
        best_seq = best_prob * target_sum

    # ------------------------------------------------------------------
    # Simple reorderings (often reduce the maximal convolution)
    # ------------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = 2.0 * n * np.max(_conv_fft(best_seq))
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store for the next run and give a final sanity check
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
