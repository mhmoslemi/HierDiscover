def _subgradient_descent(a0, deadline, target_sum, step_factor=0.4, gamma=0.995, eps=1e-12):
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    # precompute FFT size for conv and grad
    size_conv = 1 << ((2*n - 1).bit_length())
    size_grad = 1 << ((3*n - 2).bit_length())
    # prepping reversed a array each iteration; compute fft of reversed later.
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it=0
    while time.time() < deadline:
        # compute convolution a*a via FFT (size_conv)
        a_fft = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=size_conv)[:2*n-1]
        max_b = conv.max()
        I = (conv >= max_b * (1 - eps)).astype(np.float64)
        # gradient: conv of I with reversed a
        a_rev = a[::-1]
        I_fft = np.fft.rfft(I, n=size_grad)
        rev_fft = np.fft.rfft(a_rev, n=size_grad)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size_grad)[:len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev)-1 : len(a_rev)-1 + n]
        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + eps)
            a -= lr * momentum
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)
        # jitter occasionally
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:6d} best={best_val:.6f}")
        # schedule decay step_factor (optional)
        step_factor *= 0.9999
        it += 1
    return best_a, best_val

"""
Propose a low‑score step‑function sequence.

The algorithm works in three stages:

1. **Initialisation** – Build a diverse pool of candidate shapes
   (power‑law, exponential, Hann, half‑sine, random).  The best
   candidate (according to a fast FFT‑based max‑convolution) becomes
   the starting point.

2. **Long‑run sub‑gradient descent** – Optimise the ℓ¹‑simplex
   under the objective  max(a*a).  The exact sub‑gradient for the
   active (maximal) convolution entries is computed with FFTs.
   Momentum, a decaying step‑size and occasional Gaussian jitter
   keep the optimisation stable and escape shallow local minima.
   The routine runs for (almost) the whole time budget on a large
   problem size (up to 30 000 coefficients) so that the asymptotic
   optimum can be approached.

3. **Optional fine‑tuning** – A brief simulated‑annealing mass‑transfer
   phase tries to improve the result further.

All operations respect non‑negativity and the ℓ¹‑simplex constraint
∑a_i = √(2n).  The final sequence is trimmed of trailing near‑zero
entries, stored globally (`height_sequence_1`) for possible warm‑starts,
and returned as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the ℓ¹‑simplex Σx_i = s, x_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    Works in O(n log n) due to a sort; for the dimensions used here this
    is fast enough.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (all return vectors that already sum to `target_sum`)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Hann window: a_i ∝ 1−cos(2πi/(n−1))."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine window: a_i ∝ sin(π i/(n−1))."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max(a*a)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.4,
                         gamma: float = 0.995,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient (with momentum) for the problem
        minimize   max_k (a*a)_k
        subject to a_i ≥ 0, Σ a_i = target_sum.
    Returns the best vector seen and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    size_conv = 1 << ((2 * n - 1).bit_length())
    size_grad = 1 << ((3 * n - 2).bit_length())
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    cur_step = step_factor

    while time.time() < deadline:
        # ---- convolution a*a  -------------------------------------------------
        a_fft = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=size_conv)[: 2 * n - 1]

        # ---- identify active constraints (max entries) ------------------------
        max_b = conv.max()
        I = (conv >= max_b * (1 - 1e-12)).astype(np.float64)

        # ---- sub‑gradient of the max‑convolution -----------------------------
        a_rev = a[::-1]
        I_fft = np.fft.rfft(I, n=size_grad)
        rev_fft = np.fft.rfft(a_rev, n=size_grad)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size_grad)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + n]

        # ---- momentum update --------------------------------------------------
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * cur_step) / (norm + eps)
            a -= lr * momentum

        # ---- projection back to the simplex -----------------------------------
        a = _project_to_simplex(a, target_sum)

        # ---- occasional jitter to escape plateaus -----------------------------
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # ---- keep the best seen -----------------------------------------------
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:6d} best={best_val:.6f}")

        # ---- simple decay of the step size ------------------------------------
        cur_step *= 0.9997
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance (optional fine‑tune)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.3,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Very cheap local moves that shift mass from a “donor” pair to a “receiver”."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(a*a) / (∑a)²  (equivalently minimise max(a*a) after normalising Σa=√(2n)).

    The routine adapts the problem size to the available time, performs a
    short diversification phase, then runs a long sub‑gradient descent on a
    large scale (up to 30 000 coefficients).  A brief final mass‑transfer
    sweep may improve the result further.  The best sequence found is
    returned as a list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # small safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Choose problem size based on the total budget.
    # Larger n gives a better asymptotic bound but makes evaluation slower.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 30000
    elif budget_s >= 600:
        n = 25000
    elif budget_s >= 400:
        n = 20000
    elif budget_s >= 200:
        n = 15000
    else:
        n = 10000

    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Build an initial candidate pool and pick the best according to
    # the fast FFT‑based max‑convolution.
    # --------------------------------------------------------------
    candidates = []

    # Use a warm‑start if a previous solution exists.
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        candidates.append(_resize_sequence(prev, n))

    # Deterministic families
    candidates.append(_cosine_sequence(n, target_sum))
    candidates.append(_sin_half_sequence(n, target_sum))
    for p in [0.3, 0.5, 0.7, 1.0]:
        candidates.append(_power_law_sequence(n, p, target_sum))
    for lam in [0.001, 0.005, 0.01]:
        candidates.append(_exponential_sequence(n, lam, target_sum))

    # A few random draws
    for _ in range(6):
        candidates.append(_random_simplex(n, target_sum, rng))

    # Pick the strongest start.
    best_start = None
    best_val = np.inf
    for seq in candidates:
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_start = seq.copy()
    a0 = best_start

    # --------------------------------------------------------------
    # Long‑run sub‑gradient optimisation.
    # --------------------------------------------------------------
    a_opt, best_score = _subgradient_descent(a0, deadline, target_sum,
                                            step_factor=0.45, gamma=0.995)

    # --------------------------------------------------------------
    # Optional fine‑tuning with a short mass‑transfer pass.
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_opt, best_score = _mass_transfer_simulated_annealing(
            a_opt, deadline, target_sum,
            start_temp=0.25, end_temp=1e-5, low_frac=0.06)

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries (they do not affect the score
    # appreciably) and store the sequence globally for future warm starts.
    # --------------------------------------------------------------
    arr = np.array(a_opt, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # --------------------------------------------------------------
    # One final check with the official evaluator (for debugging).
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
