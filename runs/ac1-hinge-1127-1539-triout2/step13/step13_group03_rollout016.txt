def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5
    rng = np.random.default_rng()
    # Phase 1 small n
    n_small = 6000 if budget_s > 800 else 4000
    target_small = math.sqrt(2.0 * n_small)
    init_seqs = []
    # Add power law families
    for p in np.linspace(0.1, 2.0, 12):
        init_seqs.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.0005, 0.03, 10):
        init_seqs.append(_exponential_sequence(n_small, float(lam), target_small))
    for _ in range(8):
        init_seqs.append(_cosine_sequence(n_small, target_small))
    for _ in range(8):
        init_seqs.append(_sin_half_sequence(n_small, target_small))
    for _ in range(12):
        init_seqs.append(_random_simplex(n_small, target_small, rng))
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        init_seqs.append(_resize_sequence(prev, n_small))
    # add jittered versions
    jittered = []
    for seq in init_seqs:
        jitter = rng.normal(scale=target_small*0.02, size=n_small)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_small)
        jittered.append(seq_j)
    init_seqs.extend(jittered)

    best_seq = None
    best_val = np.inf

    # loop until time left > some threshold
    while time.time() < deadline - 30:
        # select seed
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_small*0.005, size=n_small)
            seed += jitter
            seed = np.maximum(seed,0)
            seed = _project_to_simplex(seed, target_small)
        else:
            seed = rng.choice(init_seqs)
        # Run gradient descent
        run_deadline = min(time.time() + 12.0, deadline - 8.0)
        if run_deadline <= time.time():
            break
        a_opt, val_opt = _annealed_gradient_descent(seed, run_deadline, target_small,
                                                    max_beta=1e6, step_factor=0.8, gamma=0.98)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration best score {best_val:.6f}")

    # Phase 2: mass transfer on small size, if time
    if best_seq is not None and time.time() < deadline - 10:
        mt_deadline = time.time() + (deadline - time.time()) * 0.15
        a_mt, val_mt = _mass_transfer_simulated_annealing(best_seq, mt_deadline, target_small,
                                                          start_temp=0.5, end_temp=1e-5, low_frac=0.06)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] MassTransfer improved to {best_val:.6f}")

    # Phase 3: upscale to larger n
    # choose n_large based on remaining time
    remaining = deadline - time.time()
    if remaining > 250:
        n_large = 25000
    elif remaining > 150:
        n_large = 20000
    elif remaining > 80:
        n_large = 15000
    else:
        n_large = n_small
    target_large = math.sqrt(2.0 * n_large)

    seq_large = _resize_sequence(best_seq, n_large)
    seq_large = _project_to_simplex(seq_large, target_large)

    # reorder deterministic candidates and pick best
    candidates = [seq_large, _alternating_order(seq_large), _even_spread_order(seq_large)]
    seq_large = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_large}, start max {_max_convolution_fft(seq_large):.6f}")

    # Phase 4: permutation random and hillclimb
    if time.time() < deadline - 20:
        perm_deadline = time.time() + min(12.0, deadline - time.time() - 8.0)
        seq_large, _ = _permutation_search(seq_large, trials=120, deadline=perm_deadline)
        print(f"[{time.time():.2f}] Random permutation improved max {_max_convolution_fft(seq_large):.6f}")

    if time.time() < deadline - 30:

"""
Hybrid optimizer for the step‑function benchmark.

The algorithm follows a staged approach (≈1000 s total):
1. Rapid exploration on a moderate size (≈5–6 k) using many families of
   analytically‑generated sequences and short annealed‑gradient runs.
2. Aggressive mass‑transfer (simulated‑annealing) on the moderate size.
3. Upscaling to a larger size (≈15–25 k) with deterministic re‑ordering.
4. Random permutation search + pairwise hill‑climb on the large scale.
5. Long projected sub‑gradient descent (exact max‑convolution gradient).
6. Final short mass‑transfer pass and trimming.

All steps respect non‑negativity and the ℓ¹‑simplex constraint
(∑a = √(2 n)).  The objective after normalisation is simply
max(conv(a,a)), so every stage tries to reduce the maximum entry of the
self‑convolution.  The code keeps the best sequence seen and stores it in
the module global ``height_sequence_1`` for reuse across runs.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ 1‑cos(2π·i/(n‑1)) – Hann window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·i/(n‑1)). Peaks in the centre, zero at the ends."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place sorted values alternately at the left and right ends."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the sorted values uniformly using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.8,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        # soft‑max over all convolution entries
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv gradient)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.998,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation search – random shuffles
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Local permutation hill‑climb (pairwise swaps)
# ----------------------------------------------------------------------


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]          # revert
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val

            best_score = val_sub
            seq_large = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – final gentle mass‑transfer
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mt_deadline = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_large, mt_deadline, target_large,
            start_temp=0.3, end_temp=1e-5, low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_large = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_large, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
