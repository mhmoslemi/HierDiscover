def propose_candidate(budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    if budget_s >= 800: n = 12000
    elif budget_s >= 400: n = 8000
    else: n = 5000

    target_sum = np.sqrt(2.0 * n)

    best_seq = None
    best_score = np.inf

    # param sweep
    def evaluate(seq):
        nonlocal best_seq, best_score
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
        return val

    # grid loops
    # 1) power law
    for p in np.linspace(0.25, 1.2, 20):  # 20 values
        seq = _power_law_sequence(n, p, target_sum)
        evaluate(seq)

    # 2) exp power law
    for p in np.linspace(0.25, 1.2, 15):
        for q in np.linspace(0.0, 12.0, 15):
            seq = _exp_power_law_sequence(n, p, q, target_sum)
            evaluate(seq)

    # 3) two-sided power law
    for p in np.linspace(0.0, 1.5, 16):
        for r in np.linspace(0.0, 1.5, 16):
            i = np.arange(n, dtype=np.float64)
            a = ((i+1) ** (-p)) * ((n - i) ** (-r))
            # avoid zeros: set at least small epsilon
            a[a < 1e-15] = 0.0
            a *= target_sum / a.sum()
            evaluate(a)

    # random restarts: generate random Dirichlet vectors
    rng = np.random.default_rng()
    while time.time() < deadline - 20.0: # reserve...
        # use random draws of Dirichlet
        alpha = np.ones(n)
        # bias to some shape: maybe sample with concentration 0.5 for spikier or 2 for flatter?

        conc = rng.uniform(0.3, 2.0)

"""
Advanced optimizer for the step‑function problem.

Goal: minimise  max(conv(a,a))  under a_i≥0 and Σa_i = √(2n).
The objective returned by evaluate_sequence() is exactly this maximum.

Strategy
--------
1. **Coarse parametric sweep** – a handful of simple families
   (pure power‑law, exponential tilt, symmetric two‑sided power‑law)
   to obtain a decent initial point quickly.

2. **Long Adam optimisation** on a soft‑max approximation of the max‑convolution.
   The temperature (β) is increased exponentially up to 1e8,
   while the Adam step‑size is projected onto the simplex.
   Occasional jitter prevents stagnation.

3. **Exact deterministic mass‑transfer (DMT2)** – a provably non‑increasing
   refinement that moves mass from a “donor” index to a “receiver”
   to lower the current peak.

4. **Simulated‑annealing refinement** – a stochastic search that can
   escape shallow local minima by accepting uphill moves with a linearly
   decaying temperature.

All phases respect the global time budget (default 1000 s).  The best
sequence found is trimmed of trailing near‑zero entries and returned.

The code contains only top‑level functions (no closures) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective value (max of the self‑convolution)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))   # with our normalisation = max_b


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used in the coarse sweep)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _two_sided_power_sequence(n: int, p: float, r: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-p)) * ((n - i) ** (-r))
    # avoid all‑zero vector when p,r are large
    a[a < 1e-30] = 0.0
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   beta_max: float = 1e8,
                   step_factor: float = 0.8,
                   gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Long Adam run on the soft‑max objective.
    Returns the best vector seen and its exact max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # gradient of soft‑max
        g, _ = _gradient_softmax(a, beta)

        # Adam moments
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        # adaptive step (scaled to keep iterates in the simplex size)
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake off plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        # periodic exact evaluation
        if it % 100 == 0:
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:.1e} score={best_val:.6f}")

        # increase β exponentially (but cap)
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta * 2.0, beta_max)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the “DMT2” refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Deterministic mass‑transfer that never increases the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick one maximal index (the first)
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner for each i: a_{k-i}
        partner = a[k - idx]

        # donor: an index among idx that has a_i>0 and large partner
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver: index with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        # compute feasible delta that would lower the peak to second_max
        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement
# ----------------------------------------------------------------------


def _simulated_annealing_refinement(a0: np.ndarray,
                                    deadline: float,
                                    target_sum: float) -> tuple[np.ndarray, float]:
    """
    SA that moves mass from a high‑contributing index to a low‑contributing one.
    Accepts uphill moves with a linearly decaying temperature.
    Returns the best vector and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    rng = np.random.default_rng()
    start_time = time.time()
    T0 = cur_max * 0.02 + 1e-4

    # pools refreshed every few hundred steps
    refresh_interval = 500
    donor_frac = 0.025
    recv_frac = 0.025

    donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
    receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

    it = 0
    while time.time() < deadline:
        if it % refresh_interval == 0:
            donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
            receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        shift_i = base[i:i + 2 * n - 1]
        shift_j = base[j:j + 2 * n - 1]
        diff = shift_j - shift_i

        new_conv = conv + delta * diff
        new_max = new_conv.max()

        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-9)

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            base[pad_len + i] = a[i]
            base[pad_len + j] = a[j]

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new_best={best_score:.6f}")

        # occasional full recomputation to avoid numerical drift
        if it % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            base = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """

    """

    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None

        val = _score(seq)

            best_seq = seq.copy()

        _maybe_update(seq)

            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)

            _maybe_update(seq)

    # cheap random perturbations around the current best
