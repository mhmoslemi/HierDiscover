"""
Hybrid optimization for minimizing max self-convolution.

- Fast parametric warm‑start over power‑law, tapered power‑law and Gaussian families.
- Projected Adam with a *binary* sub‑gradient (mask of maximal convolution entries) – 
  this is an exact sub‑gradient of the max‑norm, unlike the soft‑max used before.
- Occasional random jitter and short restarts to escape plateaus.
- Final deterministic mass‑transfer refinement (pairwise) and trimming.

The algorithm respects the provided time budget and returns the best sequence found.
"""
import time
import numpy as np

def _conv_fft(a):
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution(a):
    return float(_conv_fft(a).max())

def _project_to_simplex(v, s):
    """Euclidean projection onto the simplex {x >= 0, sum x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, target_sum):
    """Return the evaluation score (equivalent to max convolution for the chosen sum)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _power_law_sequence(n, p, target_sum):
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exp_power_law_sequence(n, p, q, target_sum):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

def _gaussian_sequence(n, sigma, target_sum):
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a

def _adam_subgradient_maxconv(a0, deadline, target_sum):
    """Projected Adam using the exact sub‑gradient of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a, target_sum)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    gamma1 = 0.9   # for mean
    gamma2 = 0.999 # for second moment
    step_factor = 0.8
    it = 0

    while time.time() < deadline:
        # compute exact sub‑gradient using mask of maximal entries
        n = a.shape[0]
        conv = _conv_fft(a)
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(mask) + n - 1).bit_length())
        mask_fft = np.fft.rfft(mask, n=size)
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(mask_fft * a_rev_fft, n=size)[: len(mask) + n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # Adam update (negative gradient because we want to *decrease* the max)
        m = gamma1 * m + (1 - gamma1) * grad
        v = gamma2 * v + (1 - gamma2) * (grad * grad)
        m_hat = m / (1 - gamma1 ** (it + 1))
        v_hat = v / (1 - gamma2 ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (np.sqrt(it + 1) + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a, target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam‑subgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def _pairwise_optimal_transfer(a0, deadline, target_sum, donor_frac=0.015, receiver_frac=0.015):
    """Same as earlier implementation – kept for a final polishing pass."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a, target_sum)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        # pre‑compute shifted views for donors/receivers
        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]
                d = 2.0 * (shift_j - shift_i)   # effect on convolution when moving 1 unit from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - 1e-12:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a, target_sum)
        if cur_score < best_score - 1e-12:

"""
Hybrid optimizer for the max self‑convolution problem.

Main ideas
-----------
* Warm‑start with cheap parametric families (power‑law, tilted‑power‑law,
  Gaussian).  The best vector is projected onto the simplex Σa = √(2 n).

* Main refinement uses a *projected Adam* optimiser that works with the **exact
  sub‑gradient** of the max‑convolution (mask of maximal entries).  This is
  equivalent to Adam on the true L∞‑norm rather than a soft‑max proxy.

* A short deterministic “global‑peak mass transfer’’ step (linearised
  feasibility) and a final pairwise greedy transfer squeeze out the last
  improvements.

All phases respect the supplied time budget (default 1000 s).  The routine
returns the best non‑negative sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s, x ≥ 0 (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring (the quantity the competition evaluates)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, target_sum: float) -> float:
    """Return the evaluation score (equivalent to max convolution when Σa = target_sum)."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for the warm‑start
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser that uses the *exact* sub‑gradient of the max‑norm
# ----------------------------------------------------------------------
def _adam_subgradient_maxconv(a0: np.ndarray,
                              deadline: float,
                              target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true L∞‑norm of the self‑convolution.
    The gradient is taken w.r.t. the mask of maximal entries (exact sub‑gradient).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a, target_sum)

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    gamma1 = 0.9   # exponential decay for first moment
    gamma2 = 0.999 # exponential decay for second moment
    step_factor = 0.8
    it = 0

    while time.time() < deadline:
        n = a.shape[0]

        # ----- exact sub‑gradient (mask of maximal entries) -----
        conv = _conv_fft(a)
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(mask) + n - 1).bit_length())
        mask_fft = np.fft.rfft(mask, n=size)
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(mask_fft * a_rev_fft, n=size)[: len(mask) + n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # ----- Adam update (note the sign: we *subtract* the gradient) -----
        m = gamma1 * m + (1 - gamma1) * grad
        v = gamma2 * v + (1 - gamma2) * (grad * grad)
        m_hat = m / (1 - gamma1 ** (it + 1))
        v_hat = v / (1 - gamma2 ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (np.sqrt(it + 1) + 1.0), size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a, target_sum)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam‑subgrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Global peak‑mass transfer (aggressive reduction of the current peak)
# ----------------------------------------------------------------------
def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """Move a fraction of the total donor mass to a single weak receiver while
    guaranteeing that no other convolution entry exceeds the previous maximum."""
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a, target_sum)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # stop if the peak is already flat enough
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)

        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donors: indices with largest a_i * w_i
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        # linear contribution of donors alone (used for feasibility test)
        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        # candidate receivers = indices with smallest partner weight w_i
        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # shifted view for this receiver
            diff = donor_mass * shift_r - conv_donors   # linearised residual
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] / (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] + donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # apply the move
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a, target_sum)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global‑peak it={it} α={best_alpha:.4g} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise greedy transfer (final polishing)
# ----------------------------------------------------------------------
def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015) -> tuple[np.ndarray, float]:
    """Move mass from a large coefficient to a small one so that the peak of the
    convolution strictly decreases."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a, target_sum)
    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        # cache shifted views for the relevant indices
        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]
                d = 2.0 * (shift_j - shift_i)   # effect of moving 1 unit from i to j
                d_k = d[k]
                if d_k >= 0.0:                  # would raise the current peak
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - 1e-12:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a, target_sum)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point – respects the time budget
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the max self‑convolution.
    Returns the best sequence found (as a plain Python list) before the budget expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a tiny safety margin

    # ---- problem size (larger n gives a lower theoretical floor) ----
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)  →  score = max convolution

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric warm‑start
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # (i) power‑law sweep
    for p in np.linspace(0.30, 0.85, 31):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq, target_sum)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (ii) tilted power‑law sweep
    for p in np.linspace(0.30, 0.85, 19):
        for q in np.linspace(0.0, 6.0, 19):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq, target_sum)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (iii) Gaussian sweep
    for sigma in np.linspace(0.1 * n, 0.5 * n, 7):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq, target_sum)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (iv) reuse a previously stored construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized, target_sum)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (v) jittered neighbourhood around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand, target_sum)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] Warm‑start best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam with exact sub‑gradient (main optimisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_a_deadline = time.time() + 0.70 * remaining   # ~70 % of the remaining budget

    while time.time() < phase_a_deadline - 5.0:
        # allocate a modest budget for each restart
        run_budget = min(30.0, (phase_a_deadline - time.time()) * 0.4)
        if run_budget < 5.0:
            break

        start_vec = best_seq + rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _adam_subgradient_maxconv(start_vec,
                                                   time.time() + run_budget,
                                                   target_sum)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Sub‑grad Adam improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Global peak‑mass transfer (aggressive peak reduction)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.025,
                                                      receiver_frac=0.025)
        if val_peak < best_score:
            best_score = val_peak
            best_seq = a_peak.copy()
            print(f"[{time.time():.2f}] Global‑peak transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Pairwise greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.02,
                                                      receiver_frac=0.02)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise final polish improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final sanity check (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
