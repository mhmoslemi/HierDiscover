assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}= \frac{2n\;\max (a*a)}{(\sum a)^2}
\]

where `a` is a non‑negative step‑function and `a*a` is its linear convolution.
If we keep the ℓ¹–norm fixed to  

\[
\sum a_i = \sqrt{2n},
\]

the score is simply the maximum entry of the convolution.  
Hence the optimisation problem becomes:

*choose a non‑negative vector on the simplex that makes its self‑convolution as flat as possible.*

The previous code already uses several heuristics (annealed gradient, sub‑gradient, mass‑transfer, random permutations).  
The missing ingredient is a **deterministic ordering of the coefficients** that guarantees the smallest possible product for the central convolution term (the term that is normally the bottleneck).  
For an even length `n` the central term is  

\[
c_{n-1}= \sum_{i=0}^{n-1} a_i a_{n-1-i}.
\]

By the rearrangement inequality the smallest possible value of this sum is obtained when the **largest coefficient is paired with the smallest one, the second largest with the second smallest, …**.  
A simple linear‑time routine (`_min_product_order`) builds exactly this ordering.

The new optimiser therefore proceeds in six stages:

1. **Broad exploration** – many random seeds, power‑law / exponential / window families, a few short annealed‑gradient runs.
2. **Aggressive mass‑transfer** – simulated‑annealing on the best seed.
3. **Upscaling** – linear interpolation to a larger `n` (≈ 20 k‑30 k) and projection onto the simplex.
4. **Deterministic ordering** – apply `_min_product_order` (and keep the old alternating / even‑spread orders for safety) and keep the best.
5. **Fine‑grained optimisation** – sub‑gradient descent + a final mass‑transfer pass.
6. **Cleanup** – strip trailing zeros, store the sequence for the next run and return it.

All stages respect the non‑negativity and ℓ¹‑simplex constraints, they use only NumPy (FFT for fast convolution) and stay well within the 1000 s budget.

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (different families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Hann window – a raised‑cosine shape."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))          # Hann window
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Peaked in the centre, zero at the ends."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _double_power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """
    Symmetric “U‑shape”: a_i ∝ i^{-p} + (n‑i+1)^{-p}.
    Large values at the two ends, small in the middle.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Deterministic re‑ordering – minimise large‑large products
# ----------------------------------------------------------------------


def _min_product_order(seq: np.ndarray) -> np.ndarray:
    """
    Place the largest half of the coefficients on the left side,
    the smallest half on the right side (mirrored).  For even ``n`` this
    yields the minimum possible value of the central convolution term
    Σ a_i a_{n‑1‑i}.
    For odd ``n`` the middle element stays the median value.
    """
    n = seq.shape[0]
    asc = np.sort(seq)                     # ascending
    half = n // 2
    out = np.empty_like(seq)

    # left side receives the largest values (descending)
    out[:half] = asc[::-1][:half]

    # right side receives the smallest values, reversed so that the
    # smallest sits at the very rightmost index (mirrored to the largest)
    out[n - half:] = asc[:half][::-1]

    if n % 2 == 1:                         # odd length → keep median in the centre
        out[half] = asc[half]
    return out


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Legacy ordering – kept for comparison."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Legacy ordering – kept for comparison."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 8e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (O(n log n) gradient)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    # pre‑compute convolution once; we will recompute after each move
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation search – random shuffles
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Local permutation hill‑climb (pairwise swaps)
# ----------------------------------------------------------------------


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            # keep the swap
        else:
            a[i], a[j] = a[j], a[i]          # revert
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 1 – rapid exploration on a modest size
    # ------------------------------------------------------------------
    n_expl = 5000 if budget_s >= 800 else (4000 if budget_s >= 500 else 3000)
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool: List[np.ndarray] = []

    # diverse deterministic families
    for p in np.linspace(0.15, 2.5, 12):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0005, 0.03, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(6):
        init_pool.append(_cosine_sequence(n_expl, target_expl))
    for _ in range(6):
        init_pool.append(_sin_half_sequence(n_expl, target_expl))
    for p in np.linspace(0.3, 2.0, 8):
        init_pool.append(_double_power_law_sequence(n_expl, float(p), target_expl))
    for _ in range(12):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # reuse previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of the above pool
    jittered = []
    for seq in init_pool:
        jitter = rng.normal(scale=target_expl * 0.03, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        jittered.append(seq_j)
    init_pool.extend(jittered)

    best_seq = None
    best_score = np.inf

    # run annealed‑gradient runs until a few seconds are left
    while time.time() < deadline - 25.0:
        # pick a seed (occasionally jitter the current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed += jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 13.0, deadline - 8.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_expl,
            max_beta=9e5, step_factor=0.9, gamma=0.98)

        # deterministic re‑ordering that minimises the central term
        a_opt = _min_product_order(a_opt)
        val_opt = _max_convolution_fft(a_opt)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – aggressive mass‑transfer refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.12
        a_ref, val_ref = _mass_transfer_simulated_annealing(
            best_seq, mt_deadline, target_expl,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        a_ref = _min_product_order(a_ref)
        val_ref = _max_convolution_fft(a_ref)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] MT refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – upscale to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.5, target_expl)

    remaining = deadline - time.time()
    if remaining > 250.0:
        n_final = 30000
    elif remaining > 170.0:
        n_final = 25000
    elif remaining > 110.0:
        n_final = 20000
    elif remaining > 70.0:
        n_final = 15000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # deterministic reorderings – keep the best
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        _min_product_order(seq_big),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation search (random shuffles + hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        perm_deadline = time.time() + min(9.0, deadline - time.time() - 6.0)
        seq_big, _ = _permutation_search(seq_big, trials=150, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After random perm search max={_max_convolution_fft(seq_big):.6f}")

    if time.time() < deadline - 14.0:
        hill_deadline = time.time() + min(7.0, deadline - time.time() - 5.0)
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_deadline)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – sub‑gradient refinement on the large scale
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sub_deadline = deadline - 5.0
        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_deadline, target_final,
            step_factor=0.45, gamma=0.995)
        a_sub = _min_product_order(a_sub)
        val_sub = _max_convolution_fft(a_sub)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final greedy mass‑transfer (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_deadline = time.time() + min(6.0, deadline - time.time() - 2.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_big, mt_deadline, target_final,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05)
        a_mt = _min_product_order(a_mt)
        val_mt = _max_convolution_fft(a_mt)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)      # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
