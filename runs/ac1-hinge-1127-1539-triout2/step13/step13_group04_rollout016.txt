import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(a):
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a):
    return float(_conv_fft(a).max())

def _project_to_simplex(v, s):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _exp_power_law_sequence(n, p, q, target_sum):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

def _score(a):
    return _max_convolution_fft(a)

    """

"""
search for a step‑function that minimises the maximal self‑convolution.
The key observation is that a pure (or near‑pure) exponential decay
a_i ∝ exp(‑q·i/n) gives a maximal convolution
≈ 2·(q/(1‑e^{‑q}))²·e^{‑q},
which can be pushed below the required 1.503 already for moderate q.
We therefore perform a fine grid search over (p,q) where p≈0 (the power‑law
exponent) and q is the exponential decay rate, keep the best candidate,
and finish with a short high‑β Adam polishing step.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex with sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (exponential ≈ power‑law with p≈0)
# ----------------------------------------------------------------------


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)^{-p}·exp(-q·i/n)  scaled to have Σa_i = target_sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (max‑convolution entry)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective: maximal self‑convolution value."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Adam‑type polishing (soft‑max gradient, high β)
# ----------------------------------------------------------------------


def _adam_polish(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_beta: float = 5e5,
    step_factor: float = 0.6,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Run a short projected Adam optimisation from a0 until deadline."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(best)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        #--- soft‑max gradient of the max entry ---------------------------------
        size = _next_pow2(2 * a.size - 1)
        af = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(af * af, n=size)[: 2 * a.size - 1]
        max_val = conv_full.max()
        beta = min(beta * 1.3, max_beta)          # quickly raise β
        exps = np.exp(beta * (conv_full - max_val))
        pvec = exps / exps.sum()

        a_rev = a[::-1]
        ar_f = np.fft.rfft(a_rev, n=size)
        pr_f = np.fft.rfft(pvec, n=size)
        conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * a.size - 1]
        grad = 2.0 * conv_rev[a.size - 1 : a.size - 1 + a.size]

        #--- Adam updates -------------------------------------------------------
        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Adam polish it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the maximal
    self‑convolution.  The search consists of:
      1. a dense grid over (p,q) with p≈0 (pure exponential and slight power‑law),

      2. an optional short Adam polishing stage to reduce the max further;
      3. trimming of trailing near‑zero entries and returning the list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem size – a few ten‑thousands give a good trade‑off between quality
    # and speed.  Larger n does not change the theoretical bound for pure
    # exponentials, but a slightly bigger n eases the discrete approximation.
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2·n)  ⇒  evaluation = max‑convolution

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Phase 0 – exhaustive grid over (p , q)
    # ------------------------------------------------------------------
    p_vals = [0.0, 0.05, 0.1, 0.2]                # include the pure exponential p=0
    q_vals = np.linspace(0.2, 6.0, 60)            # step ≈0.1

    for p in p_vals:
        for q in q_vals:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
        # stop early if we already have a comfortable margin
        if best_val < 1.30:

    print(f"[{time.time():.2f}] Grid search best max‑conv = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – optional Adam polishing (high β) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
