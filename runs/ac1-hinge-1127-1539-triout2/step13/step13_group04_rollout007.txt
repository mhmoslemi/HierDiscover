"""
Constant search for step function minimising max self‑convolution.

The algorithm:
1. Initialise from several parametric families and from any previously known
   sequence (global name ``height_sequence_1``).  The best start is kept.
2. Run an Adam optimiser on a soft‑max approximation to the max‑convolution.
   The schedule of the soft‑max temperature β is increased gradually.
3. Perform an incremental penalty‑descent loop.  At each iteration we try to
   lower the current maximal convolution by a small target amount.  The
   gradient of the penalty (max‑M)_+ is used together with projection onto the
   simplex.  After a successful reduction a few more Adam steps are applied.
   When a target cannot be met the step size is halved.
4. Final polishing with a short high‑β Adam run.
5. Trim trailing near‑zero entries and return.
"""

import time

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a * a (linear convolution)."""
    return float(_conv_fft(a).max())

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a

    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a

# ----------------------------------------------------------------------
# Soft‑max gradient used by Adam

def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))
        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _adam_gradient_descent(start,
                                                time.time() + run_budget,
                                                target_sum,
                                                max_beta=2e5,
                                                step_factor=0.8,
                                                gamma=0.995)
        if val_opt < best_val - 1e-12:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

# Penalty descent – tries to enforce max(conv) ≤ target_max
# ----------------------------------------------------------------------
def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty vector (only entries above target_max)
        p = conv - target_max
        p[p < 0] = 0.0

        # gradient = 2 * (p * a_rev)_{n-1 … n-1+n-1}
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)

        it += 1

    return best_a, best_max

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _adam_gradient_descent(start,
                                                time.time() + run_budget,
                                                target_sum,
                                                max_beta=2e5,
                                                step_factor=0.8,
                                                gamma=0.995)
        if val_opt < best_val - 1e-12:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Penalty descent – tries to enforce max(conv) ≤ target_max
# ----------------------------------------------------------------------
def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    """Gradient descent on the penalty (max(conv)-target_max)_+."""
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        p = conv - target_max
        p[p < 0] = 0.0

        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution.  The procedure follows the stages described in the
    module docstring.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Problem size – larger n gives more freedom but costs more time.
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    elif budget_s >= 300:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2·n) → evaluation score = max‑convolution

    # --------------------------------------------------------------
    # Phase 0 – parametric pre‑search
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.25, 0.90, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) Gaussian bumps
    for sigma in np.linspace(n / 20, n / 5, 9):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (e) reuse a previously saved sequence if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (f) small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam optimisation (soft‑max)
    # --------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 50.0:
        gd_deadline = time.time() + remaining_time * 0.30
        while time.time() < gd_deadline - 5.0:
            jitter = rng.normal(scale=target_sum * 0.01, size=n)
            start_vec = best_seq + jitter
            start_vec = np.maximum(start_vec, 0.0)
            start_vec = _project_to_simplex(start_vec, target_sum)

            a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                             target_sum,
                                                             gd_deadline)
            if val_opt < best_score - 1e-12:
                best_score = val_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.2f}] AdamGD restart improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Incremental penalty descent (target lowering)
    # --------------------------------------------------------------
    # Try to bring the maximal convolution down in several decreasing steps.
    reduction = min(0.0045, best_score * 0.01)   # start with ≈1 % of current score
    min_reduction = 5e-5
    while time.time() < deadline - 10.0 and reduction >= min_reduction:
        target = best_score - reduction
        sub_deadline = time.time() + min(30.0, deadline - time.time())
        a_cand, new_max = _penalty_lower_max(best_seq,
                                             target_sum,
                                             sub_deadline,
                                             target)
        if new_max < best_score - 1e-12:
            best_seq = a_cand
            best_score = new_max
            print(f"[{time.time():.2f}] Reduction {reduction:.6f} succeeded → {best_score:.6f}")

            # short Adam polishing after a successful reduction
            if time.time() < deadline - 5.0:
                a_polish, val_polish = _adam_gradient_descent(best_seq,
                                                              time.time() + 6.0,
                                                              target_sum,
                                                              max_beta=5e5,
                                                              step_factor=0.5,
                                                              gamma=0.995)
                if val_polish < best_score - 1e-12:
                    best_seq = a_polish
                    best_score = val_polish
                    print(f"[{time.time():.2f}] Post‑penalty Adam improved to {best_score:.6f}")
            continue
        else:
            reduction *= 0.5
            print(f"[{time.time():.2f}] Reduction {reduction:.6f} not met, halving step")
            if rng.random() < 0.2:
                jitter = rng.normal(scale=target_sum * 0.001, size=n)
                cand = _project_to_simplex(best_seq + jitter, target_sum)
                cand_max = _max_convolution_fft(cand)
                if cand_max < best_score - 1e-12:
                    best_seq = cand
                    best_score = cand_max
                    print(f"[{time.time():.2f}] Random jitter improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase C – Final high‑β Adam polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score - 1e-12:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for diagnostics
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # Store for future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
