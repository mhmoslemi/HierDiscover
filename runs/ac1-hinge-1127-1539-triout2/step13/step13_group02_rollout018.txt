"""
Advanced step‑function optimiser – version 2.

Main ideas
----------
1. **Brief, diverse seeding** – power‑law, exponential and random Dirichlet
   families are sampled on a moderate size (≈5 k).  A few jittered copies give
   a rich pool of starts.

2. **Fast short runs** – a mixture of soft‑max gradient descent and exact
   sub‑gradient descent is executed for a small fraction of the total budget
   (≈25 %).  The best among these runs is kept.

3. **Mild mass‑transfer polishing** – a deterministic transfer step improves
   the candidate before any up‑scaling.

4. **Upscaling to a large problem** – the current best sequence is linearly
   interpolated to 12‑20 k entries (chosen according to remaining time) and
   projected onto the ℓ¹‑simplex of the required sum √(2·n).  Several simple
   reorderings (alternating, reversed‑alternating, plain sorted) are tried.

5. **Intensive high‑β gradient polishing** – a long soft‑max GD run with a
   huge β (≈10⁶) pushes the peak of the convolution down.

6. **Heavy permutation‑annealing** – swaps are performed preferentially
   between indices that contribute strongly / weakly to the current peak.
   Temperature decays exponentially to a very low final value.

7. **Final nudges** – a short sub‑gradient pass and a tiny deterministic
   mass‑transfer pass finish the optimisation.

All steps respect the 1000 s budget, constantly checking the deadline.
The returned sequence is trimmed of trailing zeros and stored as
``height_sequence_1`` for possible reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (max‑convolution and full convolution)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) this equals the peak of the convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of sequences (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, then projected onto the simplex."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), then projected onto the simplex."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize (keeps total mass)."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value left‑most, second largest right‑most, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Momentum‑based GD on soft‑max approximation of the peak.
    """
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # progress output
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (much cheaper per step)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(convolution)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # exact sub‑gradient at the peak entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if (kmax % 2 == 0):
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]          # derivative of a_mid^2

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing (swap high/low contributors)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Simulated‑annealing that swaps a high‑contributing index with a low one."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine:
      1. builds a modest pool of diverse seeds,
      2. runs many short optimisation runs (GD / sub‑gradient),
      3. applies a quick deterministic mass‑transfer,
      4. up‑scales to a larger length,
      5. tries a few deterministic reorderings,
      6. spends the bulk of the remaining budget on
         – a long high‑β gradient descent,
         – an extensive permutation‑annealing,
         – a short sub‑gradient nudge,
         – a final tiny mass‑transfer polish,
      7. trims trailing zeros and returns the best sequence.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest exploration size
    # ------------------------------------------------------------------
    n_expl = 5000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse seed pool
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.2, 1.0, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    # exponential family
    for lam in np.linspace(0.0005, 0.02, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    # random Dirichlet‑like seeds
    for _ in range(5):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    # possibly reuse the previous best
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n_expl:
            prev = _resize_sequence(prev, n_expl)
        init_pool.append(_project_to_simplex(prev, target_expl))

    # jittered copies for extra diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Short optimisation runs (≈25 % of the total budget)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    exploration_deadline = start_time + (budget_s * 0.25)

    while time.time() < exploration_deadline:
        seed = rng.choice(init_pool)

        if rng.random() < 0.7:
            # short soft‑max GD (≈6 s)
            run_dead = min(time.time() + 6.0, exploration_deadline)
            a_opt, val_opt = _annealed_gradient_descent(
                seed,
                run_dead,
                target_expl,
                max_beta=2e5,
                step_factor=0.6,
                gamma=0.98,
            )
        else:
            # short sub‑gradient (≈5 s)
            run_dead = min(time.time() + 5.0, exploration_deadline)
            a_opt, val_opt = _subgradient_descent(
                seed,
                run_dead,
                target_expl,
                max_iter=150_000,
            )

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best {best_score:.6f}")

        if time.time() >= exploration_deadline:
            break

    # Fallback – should never happen but keep robustness
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.5, target_expl)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 4. Quick deterministic mass‑transfer (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        mt_dead = time.time() + 30.0
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_expl,
            start_temp=0.5,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] After MT {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to a larger length
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 500:
        n_final = 20000
    elif remaining > 300:
        n_final = 15000
    elif remaining > 150:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # ------------------------------------------------------------------
    # 6. Deterministic reorderings
    # ------------------------------------------------------------------
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
        np.sort(seq_big)[::-1],          # plain descending
    ]
    best_seq = candidates[0]
    best_score = _score(best_seq)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Reorder improve to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Long high‑β gradient polishing (main chunk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        gd_dead = time.time() + min(120.0,
                                    deadline - time.time() - 100.0)
        a_gd, val_gd = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=5e6,
            step_factor=0.4,
            gamma=0.995,
        )
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD polish {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Extensive targeted permutation simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        perm_dead = deadline - 5.0
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.8,
            end_temp=5e-6,
            top_frac=0.06,
            low_frac=0.06,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation SA {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Very short sub‑gradient nudge
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sub_dead = time.time() + min(5.0,
                                     deadline - time.time() - 1.0)
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=200_000,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Final sub‑grad {best_score:.6f}")

    # ------------------------------------------------------------------
    # 10. Tiny final mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(3.0,
                                    deadline - time.time() - 0.5)
        a_mt2, val_mt2 = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.2,
            end_temp=1e-5,
            low_frac=0.03,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Final MT {best_score:.6f}")

    # ------------------------------------------------------------------
    # 11. Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # 12. Store globally for possible reuse
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    return final_seq
