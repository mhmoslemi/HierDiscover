"""
Propose a low‑max‑self‑convolution step‑function.

The algorithm consists of three stages:

1️⃣  A quick parametric sweep (power‑law, exponential‑power‑law,
    sine, Gaussian) to obtain a decent starting vector
    on the simplex Σa = √(2·n).

2️⃣  One to several linear‑programming refinements.
    For the current vector ``a`` the LP
        maximise   Σ g_i
        subject to   (a * g)_k ≤ M   (∀k) ,   g_i ≥ 0
    where ``M`` is the present maximal self‑convolution.
    The solution ``g`` is scaled to the correct ℓ₁‑norm and
    convex‑combined with ``a`` (step size ``t``).  Repeating
    this usually lowers the maximal convolution quickly.

3️⃣  A short exact sub‑gradient descent (the same routine that
    was used in the previous best solution) to polish the result.

The whole search is limited to the given time budget (default 1000 s).
The returned list is a non‑negative sequence; its evaluation
value equals the maximal self‑convolution because the ℓ₁‑norm is
normalised to √(2·n).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT – O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection on the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex Σx = s, x≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used for an initial guess)
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring helper – just the maximal self‑convolution
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the objective value (max self‑convolution)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same as in the previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 4000) -> tuple[np.ndarray, float]:
    """True max‑convolution sub‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        # pick a random active index
        k = idx_max[rng.integers(len(idx_max))]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]          # sub‑gradient of max entry

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            # tiny random move to escape a stall
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        # step that would erase the gap to the second maximum
        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # backtracking line‑search (max 6 reductions)
        improved = False
        for _ in range(6):
            cand = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(cand)
            if cand_score < best_score - 1e-12:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                improved = True
                break
            step *= 0.5
        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.0005, size=n), target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
def _lp_refinement(a: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5,
                   t_start: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Perform a few LP iterations:
    maximise Σ g_i  subject to (a * g) ≤ M,  g ≥ 0.
    The obtained g is rescaled to the simplex and convex‑combined
    with the current vector a (step size t that shrinks each iteration).
    Returns the improved vector and its score.
    """
    try:
        import scipy.sparse as sp
        import scipy.optimize as opt
    except Exception as e:               # SciPy not available – skip LP
        print(f"[{time.time():.2f}] SciPy unavailable ({e}); skipping LP refinement")
        return a, _score(a)

    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    t = t_start
    it = 0

    while it < max_iters and time.time() < deadline - 2.0:
        M = _score(a)                     # current maximal convolution
        # Build convolution matrix A such that (A @ g)_k = (a * g)_k
        # each diagonal i of A contains the constant a[i]
        diags = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
        offsets = np.arange(n)            # 0 … n‑1
        A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

        # Linear program data
        c = -np.ones(n, dtype=np.float64)                 # maximise Σg ⇔ minimise -Σg
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # (a * g) ≤ M
        bounds = [(0.0, None)] * n

        # time left for the LP solver
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        options = {"time_limit": remaining}

        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options=options)

        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # scale g0 to the correct ℓ₁–norm and mix
        scale = target_sum / sum_g0
        g = g0 * scale
        a = (1.0 - t) * a + t * g
        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        print(f"[{time.time():.2f}] LP iter {it:2d}: max={cur_val:.6f} (t={t:.3f})")
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # gradually reduce the mixing factor
        t = max(t * 0.7, 0.02)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the maximal
    self‑convolution (the evaluation score).  The routine consists of:
        0. a quick parametric pre‑search,
        1. a few LP‑based refinement steps,
        2. a short exact sub‑gradient polishing.
    The final sequence is projected onto the simplex Σa = √(2·n)
    and trailing near‑zero entries are removed.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom, but LP cost ~ n²)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 1800
    elif budget_s >= 400:
        n = 1300
    else:
        n = 800
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (very cheap)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_val = np.inf

    # (a) pure power‑law sweep
    for p in np.linspace(0.2, 1.0, 21):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # (b) power‑law with exponential tail
    for p in np.linspace(0.2, 1.0, 21):
        for q in np.linspace(0.0, 5.0, 21):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()

    # (c) sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_val:
        best_val = val
        best_seq = seq.copy()

    # (d) Gaussian bumps (few σ values)
    for sigma in np.linspace(n / 15, n / 5, 7):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # (e) reuse globally supplied result if available
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to the new size
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            resized = np.interp(x_new, x_old, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_val:
            best_val = val
            best_seq = resized.copy()

    # (f) a few jittered perturbations around the current best
    for _ in range(5):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming refinement
    # ------------------------------------------------------------------
    a_lp, val_lp = _lp_refinement(best_seq, target_sum, deadline,
                                  max_iters=6, t_start=0.25)
    if val_lp < best_val:
        best_val = val_lp
        best_seq = a_lp.copy()
        print(f"[{time.time():.2f}] After LP refinement best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_sg, val_sg = _subgradient_descent(best_seq,
                                            target_sum,
                                            deadline,
                                            max_iter=3000)
        if val_sg < best_val:
            best_val = val_sg
            best_seq = a_sg.copy()
            print(f"[{time.time():.2f}] Sub‑gradient polished best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (only for diagnostics – never influences the output)
    try:
        eval_score = evaluate_sequence(final_seq)   # pylint: disable=undefined-variable
        print(f"[{time.time():.2f}] Final evaluated score = {eval_score:.6f}")
    except Exception:
        pass

    return final_seq
