def _eg_subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    eta0 = 1.0
    it = 0

        a = a * np.exp(-eta * grad)

# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
#  Objective (exactly as evaluate_sequence would compute)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Simplex projection (L1 = s, x_i ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Parametric families for the coarse pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _half_sine_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i = sin(pi*(i+1)/(n+1))  (all positive)"""
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    a *= target_sum / a.sum()
    return a


def _tent_sequence(n: int, target_sum: float) -> np.ndarray:
    """Triangular (“tent”) shape, max at centre."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((i - (n - 1) / 2) / ((n - 1) / 2))
    a[a < 0] = 0.0
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
#  Exponential‑gradient sub‑gradient descent on the true max
# ----------------------------------------------------------------------


def _eg_subgradient_descent(a0: np.ndarray,
                            target_sum: float,
                            deadline: float,
                            eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """EG descent on max(conv) using the exact sub‑gradient."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    it = 0
    n = a.size

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (k) w.r.t. a
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k - idx]
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * partner

        # decreasing learning rate
        eta = eta0 / (1.0 + it * 1e-3)

        # exponential‑gradient update
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (keeps L1 = target_sum)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, target_sum / n)
        else:
            a = a * (target_sum / total)

        # optional small jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            print(f"[{time.time():.2f}] EG iter={it:5d} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
#  Exact deterministic mass‑transfer (DMT2) – unchanged from the reference
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Pairwise optimal transfer (enhanced version)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.03,
                               receiver_frac: float = 0.03,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest entries
        receiver_idx = np.argsort(a)[:nr]        # smallest entries

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)          # effect per unit δ
                d_k = d[k]
                if d_k >= 0.0:
                    continue            # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # keep base up‑to‑date (cheap slice assignment)
        base[pad_len + i] = a[i]
        base[pad_len + j] = a[j]

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
#  Simulated‑annealing refinement (new component)
# ----------------------------------------------------------------------


def _simulated_annealing_refinement(a0: np.ndarray,
                                    deadline: float,
                                    target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    rng = np.random.default_rng()
    start_time = time.time()
    T0 = cur_max * 0.02 + 1e-4          # initial temperature

    donors = np.argsort(-a)[: max(1, int(0.02 * n))]
    receivers = np.argsort(a)[: max(1, int(0.02 * n))]

    while time.time() < deadline:
        if time.time() - start_time > deadline - start_time:
            break

        if rng.random() < 0.001:  # occasional refresh
            donors = np.argsort(-a)[: max(1, int(0.02 * n))]
            receivers = np.argsort(a)[: max(1, int(0.02 * n))]

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        shift_i = base[i:i + 2 * n - 1]
        shift_j = base[j:j + 2 * n - 1]
        diff = shift_j - shift_i

        new_conv = conv + delta * diff
        new_max = new_conv.max()

        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-9)

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            base[pad_len + i] = a[i]
            base[pad_len + j] = a[j]

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={int(elapsed)} new_best={best_score:.6f}")

        if int(elapsed) % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            base = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])

    return best_a, best_score


# ----------------------------------------------------------------------
#  Global random mass‑transfer (exploratory)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the max‑self‑convolution.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # small safety margin

    # problem size -------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    # a convenient normalisation (any positive constant works because the
    # objective is homogeneous).  Using sqrt(2*n) reproduces the scaling
    # used in the reference code.
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (including a few extra families)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.30, 0.85, 31)
    q_grid = np.linspace(0.0, 6.0, 31)

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # half‑sine and tent shapes – cheap extra candidates
    for factory in (_half_sine_sequence, _tent_sequence):
        seq = factory(n, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # incorporate a previously stored construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # jitter a few times around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Phase 0 done, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – EG sub‑gradient descent on exact max (multiple restarts)
    # ------------------------------------------------------------------
    eg_deadline = time.time() + (deadline - time.time()) * 0.25
    while time.time() < eg_deadline - 5.0:
        # jittered start for each restart
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _eg_subgradient_descent(start_vec,
                                                 target_sum,
                                                 eg_deadline,
                                                 eta0=1.0)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] EG restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # short random move to shake the system
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Enhanced pairwise optimal mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.03,
                                                      receiver_frac=0.03)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Simulated‑annealing refinement (new)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_sa, val_sa = _simulated_annealing_refinement(best_seq,
                                                       deadline,
                                                       target_sum)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] SA refinement improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final check (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
