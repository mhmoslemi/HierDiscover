"""
Improved search for a low‑score non‑negative step function.

The optimizer consists of three phases:

1. Projected sub‑gradient descent (same idea as before, but with a
   slightly more aggressive schedule).

2. Smooth log‑sum‑exp gradient descent.
   We minimise a smooth approximation of the max‑convolution
   \max_k (a*a)[k] using the function
       F(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
   The gradient can be evaluated in O(n log n) by a single FFT‑based
   convolution.  β is increased exponentially, turning the smooth
   problem into the original max‑problem.

3. Aggressive simulated‑annealing mass‑transfer, now allowing larger
   transfers (up to 50 % of the donor mass) for faster refinement.

Multiple initialisations (warm start, parametric families and random
vectors) are tried; the best sequence found overall is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (both real) using an FFT."""
    n, m = a.size, b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_projected_descent(
    a0: np.ndarray, target_sum: float, deadline: float
):
    """Sub‑gradient descent on the entry attaining the max convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        # linear decay from 2.0 to 0.01
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # small jitter every few hundred iterations
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑proj it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – smooth log‑sum‑exp gradient descent
# ----------------------------------------------------------------------


def _grad_logsumexp(a: np.ndarray, beta: float):
    """Gradient of the smooth max‑convolution (log‑sum‑exp) w.r.t. a."""
    b = _conv_fft(a)               # length 2n‑1
    b_max = b.max()
    # soft‑max weights, numerically stable
    shift = beta * (b - b_max)
    exp_shift = np.exp(shift)
    w = exp_shift / exp_shift.sum()   # length 2n‑1, sums to 1

    # g_i = 2·Σ_k w_k·a_{k‑i}
    w_rev = w[::-1]
    conv = _conv_full(w_rev, a)       # length 3n‑2
    n = a.shape[0]
    g = 2.0 * conv[n - 1 : 2 * n - 1][::-1]
    return g, b_max


def smooth_gradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float
):
    """Projected gradient descent on the smooth approx. of max‑conv."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0.0:
            break
        frac = elapsed / total

        # exponential schedule for β : from 1 up to ~50
        beta = 1.0 * (50.0 / 1.0) ** frac

        g, _ = _grad_logsumexp(a, beta)
        grad_norm = np.linalg.norm(g) + 1e-12

        # step size proportional to target_sum / ||g||, decaying linearly
        eta = (target_sum / grad_norm) * (0.6 * (1.0 - frac) + 0.02)

        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth descent best={best_val:.6f}")

    return best_a, best_val

# Phase 3 – aggressive simulated‑annealing mass‑transfer
# ----------------------------------------------------------------------


def anneal_mass_transfer_aggressive(
    a0: np.ndarray, target_sum: float, deadline: float
):
    """SA that moves fairly large chunks of mass from “bad” to “good” indices."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9          # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # donor chosen proportionally to its contribution to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver prefers small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # larger transfer magnitude (up to 50 % of donor mass)
        max_delta = min(a[donor] * 0.5, target_sum * 0.1)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- O(n) update of the convolution -----------------
        b_new = b.copy()
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        # quadratic terms
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to curb numerical drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Helper to generate parametric initial candidates
# ----------------------------------------------------------------------


def _generate_param_candidates(n: int, target_sum: float, num: int = 12):
    """Create a handful of power‑law / exponential shaped vectors."""
    rng = np.random.default_rng()
    candidates = []
    # random exponents
    for _ in range(num):
        p = rng.uniform(0.3, 1.2)
        lam = rng.uniform(0.0, 0.008)
        i = np.arange(n, dtype=np.float64)
        a = (i + 1) ** (-p) * np.exp(-lam * i)
        a = a * (target_sum / a.sum())
        candidates.append(a)
    # small systematic grid (cheap)
    for p in np.linspace(0.35, 1.0, 5):
        for lam in np.linspace(0.0, 0.01, 3):
            i = np.arange(n, dtype=np.float64)
            a = (i + 1) ** (-p) * np.exp(-lam * i)
            a = a * (target_sum / a.sum())
            candidates.append(a)
    return candidates

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # a safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)  # the usual normalisation

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + parametric + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # parametric families (power‑law / exponential)
    init_candidates.extend(_generate_param_candidates(n, target_sum, num=8))

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run the optimisation pipeline for every start, respecting time
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx+1} -----")

        # ----- Phase 1 : projected sub‑gradient descent -----------------
        sub_deadline = time.time() + 0.25 * (deadline - time.time())
        a_sub, score_sub = subgradient_projected_descent(
            init, target_sum, sub_deadline
        )
        if score_sub < best_score:
            best_score = score_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New global best after sub‑proj: {best_score:.6f}")

        # ----- Phase 2 : smooth log‑sum‑exp refinement -----------------
        if time.time() < deadline:
            smooth_deadline = time.time() + 0.25 * (deadline - time.time())
            a_smooth, score_smooth = smooth_gradient_descent(
                a_sub, target_sum, smooth_deadline
            )
            if score_smooth < best_score:
                best_score = score_smooth
                best_seq = a_smooth.copy()
                print(f"[{time.time():.2f}] New global best after smooth: {best_score:.6f}")

        # ----- Phase 3 : aggressive annealing -------------------------
        if time.time() < deadline:
            # give whatever time is left to the annealer
            a_ann, score_ann = anneal_mass_transfer_aggressive(
                best_seq if best_seq is not None else a_sub,
                target_sum,
                deadline,
            )
            if score_ann < best_score:
                best_score = score_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
