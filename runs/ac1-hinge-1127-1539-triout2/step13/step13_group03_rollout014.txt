def _lp_step(f, target_sum, t=0.5, deadline=None):
    n = len(f)
    M = np.max(np.convolve(f, f))
    # Build constraint matrix A (inequalities)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n - 1))
        i_end = min(n-1, k)
        # columns i_start .. i_end inclusive
        # For each i in range, coefficient = f[i]
        # length = i_end - i_start + 1
        # Append each column i with the same coefficient repeated for each row.
        # We'll loop over i to fill entries
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Solve LP
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        print("LP failed, using original f")
        return f
    g0 = res.x
    sum_g0 = np.sum(g0)
    if sum_g0 == 0:
        # fallback
        return f
    g = g0 * (target_sum / sum_g0)
    # mix
    f_new = (1 - t) * f + t * g
    return f_new

def _lp_refine_step(f, target_sum, deadline):
    """Perform one LP refinement and line search over mixing factor."""
    n = f.shape[0]
    M = _max_convolution_fft(f)
    # Build constraint matrix A
    # use scipy.sparse.diags? I'd use diagonal approach.
    # Let's build as described using offsets and data matrix.
    import scipy.sparse
    import scipy.optimize

    # Offsets 0..n-1
    offsets = np.arange(n, dtype=np.int32)
    # data: each row j has f[j] repeated n times -> shape (n,n)
    # Build using broadcasting.
    data = np.repeat(f[:, np.newaxis], n, axis=1)  # shape (n,n)
    A = scipy.sparse.dia_matrix((data, offsets), shape=(2*n-1, n)).tocsr()
    b = np.full(2*n-1, M + 1e-12, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g) => min -sum
    bounds = [(0.0, None)] * n

    # Solve using high-speed 'highs'
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        print("LP failed:", res.message)
        return f, M
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f, M
    # Scale to target sum
    g = g0 * (target_sum / sum_g0)

    # Line search for optimal mixing factor
    best_f = f
    best_val = M
    # candidate t's
    for t in [0.2, 0.4, 0.6, 0.8]:
        f_new = (1 - t) * f + t * g
        val = _max_convolution_fft(f_new)
        if val < best_val:
            best_val = val
            best_f = f_new
    # Also consider full replace t=1 (maybe)
    f_full = g
    val_full = _max_convolution_fft(f_full)
    if val_full < best_val:
        best_val = val_full
        best_f = f_full

    # Return best_f and corresponding max value
    return best_f, best_val

"""
Hybrid optimizer with linear‑programming refinement.

The pipeline:
1.  Build a modest‑size (≈2 k) initial pool (previous best, random,
    power‑law, exponential).
2.  For each candidate run a few LP‑based refinement steps:
       -  solve   max Σ g_j   s.t.   f * g ≤ M   (M = max(f*f))
       -  scale g to the ℓ¹‑simplex Σ g = √(2n)
       -  line‑search a convex combination (1‑t)·f + t·g.
    This exploits the classic Frank‑Wolfe style improvement.
3.  Keep the best refined small candidate.
4.  Upscale to a larger length (≈20‑30 k) by linear interpolation,
    apply deterministic reorderings, then run a fast sub‑gradient
    descent and a final mass‑transfer simulated‑annealing pass.
5.  Trim trailing near‑zeros, store the result for the next call,
    and return it.

All sequences are kept non‑negative and projected onto the ℓ¹‑simplex
so that the evaluation function reduces to the maximum entry of the
self‑convolution.
"""

import time
import math
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Sub‑gradient descent (large scale)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it} score={best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing (large scale)
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (small scale)
# ----------------------------------------------------------------------


def _lp_refine_step(f: np.ndarray,
                    target_sum: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    One LP refinement pass:
       * build constraints f * g ≤ M,
       * maximise Σ g,
       * scale, then line‑search a convex combination.
    Returns the improved sequence and its new max‑convolution value.
    """
    n = f.shape[0]
    M = _max_convolution_fft(f)

    # Build sparse matrix for constraints: each diagonal j carries constant f[j]
    offsets = np.arange(n, dtype=np.int32)                # 0 … n‑1
    # data[i] = constant f[i] repeated n times
    data = np.repeat(f[:, np.newaxis], n, axis=1)        # shape (n, n)
    A = scipy.sparse.diags(data, offsets,
                           shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M + 1e-12, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                    # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None)] * n

    # Solve LP – ``highs`` handles sparse matrices efficiently
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                method='highs')
    if not res.success:
        # fallback: keep current f
        print(f"[{time.time():.2f}] LP failed: {res.message}")
        return f, M

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f, M

    # Scale to the required ℓ¹‑simplex
    g = g0 * (target_sum / sum_g0)

    # Line‑search on mixing coefficient t
    best_f = f
    best_val = M
    # try a small grid of t values
    for t in (0.2, 0.4, 0.6, 0.8):
        cand = (1 - t) * f + t * g
        val = _max_convolution_fft(cand)
        if val < best_val:
            best_val = val
            best_f = cand
    # also consider a full replacement (t = 1)
    val_full = _max_convolution_fft(g)
    if val_full < best_val:
        best_val = val_full
        best_f = g

    return best_f, best_val
