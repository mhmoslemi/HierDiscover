def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    step0 = target_sum * 0.2  # initial step size proportionally
    iteration = 0
    while time.time() < deadline:
        # compute max conv and gradient
        conv = _conv_fft(a)
        max_val = conv.max()
        # stop if no improvement possible? Not needed.
        # compute gradient for max indices (within tolerance)
        tol = 1e-12 * max_val
        idxs = np.where(conv >= max_val - tol)[0]
        n = a.shape[0]
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=int)
            grad[i] += 2.0 * a[k - i]

        # step size
        step = step0 / np.sqrt(iteration + 1.0)
        # multiplicative update to ensure positivity
        # Using exponential: a_new = a * exp(-step * grad)
        # But large grad may overflow exponent; clip grad to reasonable range.
        exp_factor = np.exp(-step * grad)
        # For stability, limit exponent magnitude
        # Actually we can use np.maximum(np.minimum(exp_factor, 1e3), 1e-3) to avoid overflow/underflow.
        exp_factor = np.clip(exp_factor, 1e-6, 1e6)
        a = a * exp_factor
        # reproject to simplex
        a = _project_to_simplex(a, target_sum)

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={iteration} best={best_score}")

        iteration += 1

    return best_a, best_score

"""
Improved step‑function optimiser.

The routine follows several stages, each respecting the global time budget:

0️⃣  Fast parametric + random seeding.
1️⃣  Deterministic exact mass‑transfer (DMT) – greedy peak reduction.
2️⃣  Sub‑gradient descent on the max‑convolution (additive, simplex‑projected).
3️⃣  Deterministic DMT again (tightening after a different move set).
4️⃣  Simulated annealing with larger move‑ranges.
5️⃣  Final Adam optimiser on a soft‑max surrogate (very high β).

All intermediate candidates are projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)}.  The best sequence found so far is kept,
trimmed of trailing near‑zero entries and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex Σ a_i = s, a_i ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """Smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _beta_like_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^{α‑1} (n‑i)^{β‑1}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a[a < 0] = 0.0
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # Second largest (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # Move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient descent on max (conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    n = a.shape[0]

    iteration = 0
    # step0 is chosen relative to the total mass; 0.02 works well empirically
    step0 = target_sum * 0.02

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices where the maximum is attained (within a tiny tolerance)
        tol = 1e-12 * max_val
        idxs = np.where(conv >= max_val - tol)[0]

        # gradient of the max‑convolution (sub‑gradient)
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=int)
            grad[i] += 2.0 * a[k - i]

        # diminishing step size
        step = step0 / math.sqrt(iteration + 1.0)

        # additive update followed by projection onto the simplex
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={iteration:5d} score={best_score:.6f}")

        iteration += 1

    return best_a, best_score

# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic mass‑transfer with Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # Initial temperature – a fraction of the current objective
    T = max(best_score * 0.08, 1e-6)
    it = 0
    top_frac = 0.05                     # use a larger active set (5 %)
    top_k = max(1, int(top_frac * n))

    # Pre‑compute orderings; they will be refreshed periodically
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        # Refresh orderings every few thousand iterations
        if it % 3000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        # Random fraction up to 50 % of donor mass
        delta = max_delta * rng.random() * 0.5
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _max_convolution_fft(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = math.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        # Temperature cooling
        it += 1
        if it % 4000 == 0:
            T *= 0.85
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score

# Adam optimiser on a soft‑max surrogate of max (conv)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max (conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # Compute gradient = 2 * conv_rev(p, a_rev)
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)

        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to leave shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val
