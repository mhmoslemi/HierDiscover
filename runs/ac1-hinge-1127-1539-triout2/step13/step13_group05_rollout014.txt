def _simulated_annealing(a0, target_sum, deadline):
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    # Temperature schedule parameters:
    T0 = 0.1 * cur_score  # initial temperature as fraction of score?
    Tmin = 1e-6
    decay = 0.99995  # per iteration
    T = T0
    it = 0
    while time.time() < deadline:
        # choose move
        # choose donor i weighted by a_i (like choose random index with probability a_i/sum)
        # but faster: choose random non-zero index with uniform distribution
        nonzero_idx = np.flatnonzero(a > 1e-12)
        if nonzero_idx.size == 0:
            break
        i = rng.choice(nonzero_idx)
        # choose receiver j maybe from low values
        # find indices where a_j < a_i (to move to lower)
        # we can sample j uniformly from range but ensure a_j < a_i
        # simply choose random index j != i
        j = rng.integers(0, n-1)
        while j == i:
            j = rng.integers(0, n-1)
        # compute max transfer fraction
        max_delta = min(a[i] * 0.5, target_sum - a[i])  # not needed target_sum condition
        # Actually we cannot exceed a[i]; also ensure we don't move too much: limit to 0.5 of donor
        delta = rng.random() * max_delta
        if delta < 1e-12:
            continue
        # propose new a
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # Ensure non-neg
        # Compute score
        new_score = _score(a_new)
        # accept if better or with probability
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={it} new best {best_score:.6f}")
        # update temperature
        T = max(Tmin, T * decay)
        it += 1
        # Occasionally reheat (increase T) if stuck? Could implement as needed.

def _simulated_annealing(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    cur_score, cur_max, cur_conv = _calc_score_and_max(a)   # we compute conv and max
    best_a = a.copy()
    best_score = cur_score

    T0 = cur_score * 0.1  # initial temperature: fraction of score
    T = T0
    Tmin = 1e-7
    decay = 0.9999
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        # compute donor and receiver based on conv peak
        k_max = int(np.argmax(cur_conv))
        # define contribution from each i: a_i * a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi+1)
        partner = a[k_max - idx_range]  # partner values for each i in idx_range
        contrib = a[idx_range] * partner
        # pick donor from top donors
        nd = max(1, int(0.02 * n))
        donor_candidates = idx_range[np.argpartition(contrib, -nd)[-nd:]]
        donor = int(rng.choice(donor_candidates))

        # pick receiver among indices with small partner (weak contributions)
        nr = max(1, int(0.02 * n))
        receiver_candidates = idx_range[np.argsort(partner)[:nr]]
        # ensure not same as donor
        if donor in receiver_candidates:
            # remove donor
            rc = receiver_candidates[receiver_candidates != donor]
            if rc.size == 0:
                # fallback to random other index
                receiver = rng.integers(0, n)
                while receiver == donor:
                    receiver = rng.integers(0, n)
            else:
                receiver = int(rng.choice(rc))
        else:
            receiver = int(rng.choice(receiver_candidates))

        # fraction maybe random between 0 and 0.5
        max_delta = a[donor] * 0.5
        if max_delta <= 0:

            continue

def propose_candidate(budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # choose n
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)

    # Phase 0: initial candidate
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        # ensure sum
        resized = _project_to_simplex(resized, target_sum)
        init_seq = resized
    else:
        init_seq = _power_law_sequence(n, p=0.5, target_sum=target_sum)
    # perform param sweep to refine init_seq
    best_seq = init_seq.copy()
    best_score = _score(best_seq)
    # param sweeps
    p_vals = np.linspace(0.3, 0.85, 21)
    q_vals = np.linspace(0.0, 6.0, 21)
    for p in p_vals:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    for p in p_vals:
        for q in q_vals:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
    # Gaussian
    sigma_vals = np.linspace(0.1*n, 0.5*n, 7)
    for sigma in sigma_vals:
        seq = _gaussian_sequence(n, float(sigma), target_sum)

problem.

The algorithm proceeds in several stages within the allocated time budget:

1. **Warm‑start** – a cheap sweep over power‑law, exponentially‑tilted
   power‑law and Gaussian families (plus a short jittered population) is
   performed to obtain a decent initial sequence.

2. **Simulated Annealing (SA)** – a stochastic mass‑transfer process on the
   simplex.  In each step a donor index (large contribution to the current
   convolution peak) donates a random fraction of its mass to a weak
   receiver.  The move is accepted if it improves the score or with a
   Metropolis probability that decays with a temperature schedule.

3. **Exact Deterministic Mass‑Transfer (DMT2)** – a greedy local move that
   reduces the current peak by transferring mass from a donor to a receiver
   while preserving all other convolution entries.

4. **Global Peak‑Mass Transfer** – a refinement that moves a fraction of the
   total donor mass to a single weak receiver, ensuring that no other entry
   exceeds the previous maximum.

5. **Pairwise Optimal Transfer** – a fine‑grained pairwise donor/receiver
   adjustment.

6. **Final Adam polishing** – projected Adam gradient descent on a
   soft‑max approximation of the max‑norm with a very large inverse‑temperature
   (β) to polish the solution.

All phases respect the global time budget (default 1000 s).  The final
sequence is trimmed of trailing near‑zero entries before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(b)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _calc_score_and_max(a: np.ndarray):
    """Return (score, max_conv, full_conv) for a."""
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf, max_val, conv
    return float(2.0 * n * max_val / (s * s)), max_val, conv


# ----------------------------------------------------------------------
# Euclidean projection onto simplex Σx = s, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the cheap warm‑start)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (previous DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass transfer
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} "
                  f"Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Global random mass transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (new refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second maximum (stop condition)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donor scoring a_i * w_i
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        # candidate receivers – smallest w_i
        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # shifted view for receiver r
            diff = donor_mass * shift_r - conv_donors   # need α·diff ≤ 0 component‑wise
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # Apply move
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} "
                  f"α={best_alpha:.4g} new_score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated Annealing (SA) – core exploration stage
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer on the simplex.
    At each step a donor (large contribution to the current peak) gives a
    random fraction of its mass to a weak receiver.  Moves are accepted
    with a Metropolis probability that decays with a temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    # initial score & convolution
    cur_score, cur_max, cur_conv = _calc_score_and_max(a)
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule
    T0 = max(cur_score * 0.1, 1e-6)
    T = T0
    Tmin = 1e-7
    decay = 0.9999
    rng = np.random.default_rng()
    it = 0
    reproject_interval = 5000  # keep sum exactly on the simplex occasionally

    while time.time() < deadline:
        # locate peak index
        k_max = int(np.argmax(cur_conv))

        # contribution pairs for the peak
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner = a[k_max - idx_range]                # a_{k_max-i}
        contrib = a[idx_range] * partner               # a_i * a_{k_max-i}

        # choose donor among top contributions
        nd = max(1, int(0.02 * n))
        donor_candidates = idx_range[np.argpartition(contrib, -nd)[-nd:]]
        donor = int(rng.choice(donor_candidates))

        # choose receiver among smallest partner values
        nr = max(1, int(0.02 * n))
        receiver_candidates = idx_range[np.argsort(partner)[:nr]]
        # ensure donor ≠ receiver
        if donor in receiver_candidates:
            rc = receiver_candidates[receiver_candidates != donor]
            if rc.size == 0:
                # fallback to a random different index
                receiver = rng.integers(0, n)
                while receiver == donor:
                    receiver = rng.integers(0, n)
            else:
                receiver = int(rng.choice(rc))
        else:
            receiver = int(rng.choice(receiver_candidates))

        # random fraction of donor mass (up to 50 % of its current value)
        max_delta = a[donor] * 0.5
        if max_delta <= 0.0:
            # nothing to move – skip this iteration
            continue
        delta = rng.random() * max_delta

        # propose new vector
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        new_score, new_max, new_conv = _calc_score_and_max(a_new)

        # Metropolis acceptance
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12))

        if accept:
            a = a_new
            cur_score = new_score
            cur_max = new_max
            cur_conv = new_conv
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA it={it} new best={best_score:.6f}")

        # temperature decay
        T = max(Tmin, T * decay)

        # occasional re‑projection to avoid drift
        if it % reproject_interval == 0:
            a = _project_to_simplex(a, np.sqrt(2.0 * n))

        it += 1

        # simple reheat if we have been stuck for a long time
        if it % 20000 == 0:
            T = T0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the max self‑convolution.
    The routine respects the given time budget and returns the best sequence
    found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – larger budgets allow larger n
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)  →  score = max(b) for this normalization

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric warm‑start (power‑law, exp‑tilted, Gaussian)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # start from a stored construction if it exists
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # pure power‑law sweep
    p_vals = np.linspace(0.30, 0.85, 21)
    for p in p_vals:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # exponentially tilted power‑law sweep
    q_vals = np.linspace(0.0, 6.0, 21)
    for p in p_vals:
        for q in q_vals:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # Gaussian family
    sigma_vals = np.linspace(0.1 * n, 0.5 * n, 7)
    for sigma in sigma_vals:
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # jittered population around the current best
    for _ in range(5):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Warm‑start best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Simulated Annealing (stochastic mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline - 50.0:
        # allocate roughly half of the remaining budget to SA
        sa_deadline = time.time() + 0.45 * (deadline - time.time())
        seq_sa, val_sa = _simulated_annealing(best_seq, sa_deadline)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = seq_sa.copy()
            print(f"[{time.time():.2f}] SA reduced best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact deterministic mass‑transfer (DMT2)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        seq_dmt, val_dmt = _deterministic_mass_transfer_exact2(best_seq,
                                                              deadline,
                                                              target_sum)
        if val_dmt < best_score:
            best_score = val_dmt
            best_seq = seq_dmt.copy()
            print(f"[{time.time():.2f}] DMT2 improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Global peak‑mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        seq_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum,
                                                        donor_frac=0.025,
                                                        receiver_frac=0.025)
        if val_peak < best_score:
            best_score = val_peak
            best_seq = seq_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Pairwise optimal transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        seq_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                        deadline,
                                                        target_sum,
                                                        donor_frac=0.015,
                                                        receiver_frac=0.015)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = seq_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final Adam polishing (very high β)
    # ------------------------------------------------------------------
    if time.time() + 3.0 < deadline:
        seq_fin, val_fin = _adam_gradient_descent(best_seq,
                                                  deadline,
                                                  target_sum,
                                                  max_beta=1e6,
                                                  step_factor=0.6,
                                                  gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = seq_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: enforce simplex, trim trailing near‑zeros, return
    # ------------------------------------------------------------------
    best_seq = _project_to_simplex(best_seq, target_sum)
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final validation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
