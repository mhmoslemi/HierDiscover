```
import cvxpy as cp

def solve_lp(f, M, target_sum):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # create convolution matrix
    # Build a 2n-1 x n matrix A, where row k has f_i for i such that 0 <= k-i < n => i in [max(0,k-n+1), min(k, n-1)], and coefficient is f_i.
    # Equivalent to each row k: A[k, i] = f[i] if 0 <= k-i < n else 0.
    # So we can vectorize by constructing via loops.
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f[i]  # for k from i to i+n-1
    # Then constraint A @ g <= M * np.ones(2n-1)
    constraints.append(A @ g <= M)
    # g sum maximizing
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, verbose=False, warm_start=True)
    if g.value is None:
        raise RuntimeError("LP solve failed.")
    return g.value
```

```
def lp_step(f, M, target_sum):
    n = len(f)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i, fi in enumerate(f):
        A[i:i+n, i] = fi
    c = -np.ones(n)   # minimize -sum(g)
    b = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f'LP failed: {res.message}')
    g = res.x
    # compute scaling factor if sum_g > target_sum
    sum_g = g.sum()
    if sum_g <= 0:
        scale = 0.0
    else:
        scale = target_sum / sum_g
    # ensure scale <= 1? Not needed
    g_scaled = g * scale
```

```python
"""
LP‑based optimizer for the convolution peak problem.

The algorithm alternates between:
1. Solving a linear program that maximises the ℓ¹‑norm of a new
   sequence `g` while keeping the convolution `f*g` below the current
   peak `M`.
2. Updating the current sequence `f` by a convex combination with a
   scaled copy of `g`.
The simplex constraint Σ a_i = √(2·n) is kept exactly, so the
evaluation of a candidate coincides with the peak of the self‑convolution.
After a number of LP iterations the result is polished with a short
sub‑gradient descent and, if time permits, a soft‑max gradient descent.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, then normalised to the given ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), then normalised to the given ℓ¹‑norm."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Dense matrix representing the linear map g ↦ f*g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] to rows i … i+n-1
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi
    return A

def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Solve   max Σ g_j  subject to  g ≥ 0,  f*g ≤ M (component‑wise).
    Returns a scaled version of the optimum such that Σ g_j = target_sum
    (the scaling factor is never larger than 1, so feasibility is preserved).
    """
    n = f.shape[0]
    A = _conv_matrix(f)           # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise –Σ g ⇒ maximise Σ g
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g = res.x
    sum_g = g.sum()
    if sum_g == 0.0:
        return f.copy()   # fallback – should not happen
    # scale down to keep the ℓ¹‑norm exactly target_sum; the optimum
    # always satisfies sum_g ≥ target_sum, therefore scale ≤ 1.
    scale = min(1.0, target_sum / sum_g)
    return g * scale

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the peak of the self‑convolution."""
    a = a0.copy()
    # ensure exact ℓ¹‑norm
    a *= target_sum / a.sum()
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # compute exact gradient of the peak entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]          # derivative of a_mid² is 2·a_mid

        # diminishing step size
        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        # projection onto the ℓ¹‑simplex
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a *= target_sum / a_sum

        # occasional jitter to shake out plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a *= target_sum / a_sum

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point.  The routine works in three stages:

        1.  A fast linear‑programming loop that repeatedly
            enlarges the ℓ¹‑norm while keeping the convolution peak fixed.
        2.  A short sub‑gradient descent for fine‑tuning.
        3.  (Optionally) a final upscale and high‑β gradient polish
            if enough time remains.

    The entire search respects the 1000 s time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # spare margin

    # ------- 1. Choose problem size -------------------------------------------------
    # A modest size keeps the LP cheap while still allowing a good peak.
    n = 2000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------- 2. Initialise a diverse seed -------------------------------------------
    # Use a power‑law shape – this is known to be a reasonably good start.
    f = _power_law_sequence(n, p=0.58, target_sum=target_sum)

    # If a previous construction is available, inject it as an alternative seed.
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size >= n:
            f = _resize_sequence(prev[:n], n)
            f *= target_sum / f.sum()
        else:
            # up‑sample the shorter sequence
            f = _resize_sequence(prev, n)
            f *= target_sum / f.sum()

    best_seq = f.copy()
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] initial score={best_val:.6f}")

    # ------- 3. LP‑based improvement loop -------------------------------------------
    # Reserve roughly 800 s for this phase; the rest will be used for polishing.
    lp_deadline = deadline - 200.0
    step = 0.25                # convex‑combination weight
    it = 0
    while time.time() < lp_deadline:
        M = _max_convolution_fft(f)
        try:
            g = _lp_step(f, M, target_sum)
        except Exception as e:
            print(f"LP step failed at it={it}: {e}")
            break

        # convex update, preserves the ℓ¹‑norm exactly
        f = (1.0 - step) * f + step * g

        cur_val = _score(f)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"[{time.time():.2f}] LP it={it:4d} new best={best_val:.6f}")

        # slowly decay the step size – the algorithm behaves like a Frank‑Wolfe
        step = max(0.02, step * 0.995)

        it += 1

    # ------- 4. Short sub‑gradient polishing ----------------------------------------
    sub_dead = time.time() + min(30.0, deadline - time.time() - 5.0)
    if sub_dead > time.time() + 1.0:
        a_sub, val_sub = _subgradient_descent(best_seq, sub_dead, target_sum,
                                              max_iter=300_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] post‑LP sub‑grad best={best_val:.6f}")

    # ------- 5. Optional upscale + high‑β gradient polish (if time permits) ---------
    if time.time() < deadline - 5.0:
        # Upscale to a longer vector – the shape of the solution often
        # improves when we allow more degrees of freedom.
        n_big = 6000
        target_big = np.sqrt(2.0 * n_big)
        seq_big = _resize_sequence(best_seq, n_big)
        seq_big *= target_big / seq_big.sum()

        # high‑β soft‑max GD (a few hundred iterations should be cheap)
        # Re‑use the simple GD routine from the previous code base.
        def _softmax_gd(a0: np.ndarray,
                        dg_deadline: float,
                        beta_start: float = 1.0,
                        beta_max: float = 3e5,
                        step_factor: float = 0.5,
                        gamma: float = 0.99) -> tuple[np.ndarray, float]:
            a = a0.copy()
            # projection onto simplex (exact sum)
            a *= target_big / a.sum()
            best_a = a.copy()
            best_v = _score(a)
            beta = beta_start
            step_base = target_big * step_factor
            momentum = np.zeros_like(a)
            while time.time() < dg_deadline:
                # soft‑max gradient
                size = 1 << ((2 * a.shape[0] - 1).bit_length())
                af = np.fft.rfft(a, n=size)
                conv = np.fft.irfft(af * af, n=size)[: 2 * a.shape[0] - 1]
                max_c = conv.max()
                exps = np.exp(beta * (conv - max_c))
                p = exps / exps.sum()
                # gradient w.r.t. a (derived in the original script)
                a_rev = a[::-1]
                ar_fft = np.fft.rfft(a_rev, n=size)
                pr_fft = np.fft.rfft(p, n=size)
                conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * a.shape[0] - 1]
                grad = 2.0 * conv_rev[a.shape[0] - 1: a.shape[0] - 1 + a.shape[0]]

                momentum = gamma * momentum + grad
                norm = np.linalg.norm(momentum)
                if norm > 0:
                    lr = step_base / (norm * (beta + 1.0))
                    a -= lr * momentum
                # projection
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 0:
                    a *= target_big / a_sum

                if time.time() - dg_deadline > -1.0:
                    # just a cheap check every few hundred iterations
                    if int(beta) % 5000 == 0:
                        cur_v = _score(a)
                        if cur_v < best_v:
                            best_v = cur_v
                            best_a = a.copy()
                beta = min(beta * 1.25, beta_max)
            return best_a, best_v

        gd_deadline = deadline - 2.0
        a_big, val_big = _softmax_gd(seq_big, gd_deadline)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[{time.time():.2f}] final GD best={best_val:.6f}")

    # ------- 6. Trim trailing zeros -------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # store for possible reuse
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] final evaluated score = {val:.6f}")
    except Exception as e:
        print(f"evaluation error: {e}")

    return final_seq
```

        a_rev = a[::-1]
        ar_fft = np.fft.rfft(a_rev, n=size)
        pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * a.shape[0] - 1]
        grad = 2.0 * conv_rev[a.shape[0] - 1: a.shape[0] - 1 + a.shape[0]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a *= target_sum / a_sum

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Soft‑max GD score={best_val:.6f}")

"""
LP‑based optimiser for the convolution‑peak objective.

The algorithm proceeds in three phases:

1. **LP refinement** – repeatedly solves a linear program that maximises the
   ℓ¹‑norm of a new candidate `g` while keeping the peak of the convolution
   `f * g` below the current peak `M`.  The current sequence `f` is then
   updated by a convex combination with a scaled copy of `g`.  The ℓ¹‑norm
   (Σ a_i = √(2·n)) is preserved exactly, so the evaluation of a candidate
   coincides with the peak of its self‑convolution.

2. **Sub‑gradient polishing** – a short projected sub‑gradient descent on the
   true peak of `f*f` to fine‑tune the solution.

3. **Upscaling + high‑β soft‑max GD** – if time permits the best sequence is
   linearly interpolated to a larger length and polished with a soft‑max
   gradient descent that drives the peak down further.

The whole search respects the 1000 s budget.  All helper functions are
top‑level (no closures) and only standard scientific libraries are used.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple families of sequences (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to the given ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to the given ℓ¹‑norm."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation that preserves total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ℓ¹‑norm while keeping peak ≤ M)
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Dense matrix representing the linear map g ↦ f * g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   g ≥ 0 ,   f * g ≤ M   (component‑wise).
    Returns a scaled version of the optimum with Σ g_j = target_sum
    (the scaling factor is never larger than 1, preserving feasibility).
    """
    n = f.shape[0]
    A = _conv_matrix(f)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # minimise –Σ g  ⇔ maximise Σ g
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failure: {res.message}")

    g = res.x
    sum_g = g.sum()
    if sum_g == 0.0:
        return f.copy()
    # optimal sum is ≥ target_sum, so we can only shrink
    scale = min(1.0, target_sum / sum_g)
    return g * scale


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true peak
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the peak of a*a."""
    a = a0.copy()
    a *= target_sum / a.sum()                 # enforce exact ℓ¹‑norm
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # exact gradient of the selected peak entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]          # derivative of a_mid²

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a *= target_sum / a_sum

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a *= target_sum / a_sum

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# High‑β soft‑max gradient descent (smooth approximation of max)
# ----------------------------------------------------------------------


def _softmax_gd(a0: np.ndarray,
                deadline: float,
                target_sum: float,
                beta_start: float = 1.0,
                beta_max: float = 3e5,
                step_factor: float = 0.5,
                gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum GD on the soft‑max approximation of the convolution peak."""
    a = a0.copy()
    a_sum = a.sum()
    if a_sum > 0:
        a *= target_sum / a_sum

    best_a = a.copy()
    best_val = _score(a)
    beta = beta_start
    step_base = target_sum * step_factor
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        size = 1 << ((2 * a.shape[0] - 1).bit_length())
        af = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(af * af, n=size)[: 2 * a.shape[0] - 1]
        max_c = conv.max()
        exps = np.exp(beta * (conv - max_c))
        p = exps / exps.sum()

        # gradient of the smooth peak
        a_rev = a[::-1]
        ar_fft = np.fft.rfft(a_rev, n=size)
        pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * a.shape[0] - 1]
        grad = 2.0 * conv_rev[a.shape[0] - 1: a.shape[0] - 1 + a.shape[0]]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a *= target_sum / a_sum

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Soft‑max GD score={best_val:.6f}")

        beta = min(beta * 1.25, beta_max)

    return best_a, best_val
