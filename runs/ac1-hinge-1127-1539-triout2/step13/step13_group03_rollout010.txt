def _exponential_gradient_descent(a0, deadline, target_sum):
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    # Step size schedule: start eta0 = maybe 0.5, decay slowly.
    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2*n-1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        maxb = conv.max()
        # indicator for near max
        idx = (conv >= maxb * (1 - 1e-10)).astype(np.float64)   # shape 2n-1
        # compute gradient.
        a_rev = a[::-1]
        I_fft = np.fft.rfft(idx, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(idx) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev)-1: len(a_rev)-1 + n]
        # step size: maybe eta = 0.5 / sqrt(it+1)?
        eta = 0.7 / np.sqrt(it+1)
        a = a * np.exp(-eta * grad)
        # renorm to target sum
        a_sum = a.sum()
        if a_sum > 0:
            a *= target_sum / a_sum
        else:
            a = np.full_like(a, target_sum / a.shape[0])
        # maybe add jitter occasionally
        if it % 500 == 0:
            a += rng.normal(scale=target_sum*0.0005, size=n)
            a = np.maximum(a,0)
            a_sum = a.sum()
            if a_sum > 0:
                a *= target_sum / a_sum
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] EG it={it} val={best_val:.6f}")
        it += 1
    return best_a, best_val

def _adam_subgradient(a0, deadline, target_sum, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    it = 0
    while time.time() < deadline:
        # compute gradient
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[:len(I)+len(a_rev)-1]
        grad = 2.0 * grad_full[len(a_rev)-1:len(a_rev)-1+len(a)]
        # Adam updates
        it += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = lr * m_hat/(np.sqrt(v_hat)+eps)
        a -= step
        a = _project_to_simplex(a, target_sum)
        # Jitter occasionally
        if it % 1000 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum*0.0003, size=a.shape)
            a += jitter
            a = np.maximum(a,0)
            a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam subgrad it={it} score={best_val:.6f}")
    return best_a, best_val
```

"""
Hybrid optimizer for the step‑function benchmark.

Strategy
--------
1. Choose a final problem size (≈5·10⁴) – larger than the
   previous 5 000‑element runs – because the objective
   (max‑convolution) is essentially independent of the scale
   of the ℓ¹‑norm, and a finer discretisation gives extra
   degrees of freedom for flattening the autocorrelation.

2. Generate a modest pool of analytic families (power‑law,
   exponential, Hann, sine‑half) together with a few random
   simplex points and the previously best sequence (if any).
   All candidates are projected onto the ℓ¹‑simplex
   {a_i≥0, Σa_i = √(2·n)} and their maximal convolution value
   is measured with a fast FFT‐based routine.

3. Keep the best candidate (lowest max‑convolution) and apply a
   mirror‑descent / Adam style sub‑gradient method that works
   directly on the true non‑smooth objective  maxₖ (a∗a)_k.
   The gradient is the exact sub‑gradient
   2·(I⋆a_rev) where I is the indicator of the maximal
   convolution entries.  An adaptive step size and momentum
   (Adam) accelerate convergence while the ℓ¹‑simplex constraint
   is enforced by Euclidean projection after each update.

4. When the time budget is almost exhausted, perform a short
   simulated‑annealing “mass‑transfer’’ pass that moves mass
   from indices that contribute heavily to the current maximum
   towards low‑contributing ones.

5. Trim trailing near‑zero entries, store the sequence globally
   (for warm‑starts in subsequent runs) and return it.

All heavy work is done with FFTs (O(n log n)) and the programme
stays within the allotted 1000 s budget.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (all return a vector of length n with Σ = target_sum)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Hann‑window shape: a_i ∝ 1‑cos(2π·i/(n‑1))."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine shape: a_i ∝ sin(π·i/(n‑1))."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (tiny heuristics that occasionally lower the max)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the values evenly using a step that is coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient descent for max‑convolution
# ----------------------------------------------------------------------


def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      target_sum: float,
                      lr: float = 0.05,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Minimises max(a*a) using an Adam‑like update on the exact sub‑gradient.
    The sequence is projected onto the ℓ¹‑simplex after each step.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # ----- compute sub‑gradient -----
        conv = _conv_fft(a)                # full convolution a*a
        maxb = conv.max()
        # indicator of entries attaining (or almost attaining) the maximum
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr * target_sum) * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:6d} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.3,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer move: pick the pair (donor,receiver) that
    contributes most to the current maximum and move a random
    amount of mass from donor to receiver, accepting improvements
    or Metropolis‑type uphill moves.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    # current convolution (updated after each move)
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # location of the global maximum of the convolution
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product a[i] * a[kmax-i] for all i that contribute to kmax
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        # choose a low‑contributing receiver among the bottom `low_frac`
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake the system
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the strategy described in the module docstring.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # small safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1️⃣ Decide final problem size (larger n provides more freedom)
    # ------------------------------------------------------------------
    # The larger the n, the longer each iteration takes.  For a 1000 s
    # budget a length of around 50 000 works well on a single CPU.
    n_final = 50000
    target_final = math.sqrt(2.0 * n_final)

    # ------------------------------------------------------------------
    # 2️⃣ Build a modest pool of analytic candidates (all already satisfy Σa = √(2n))
    # ------------------------------------------------------------------
    candidates = []

    # power‑law family
    for p in np.linspace(0.3, 1.2, 10):
        candidates.append(_power_law_sequence(n_final, float(p), target_final))
    # exponential family
    for lam in np.linspace(0.0008, 0.025, 8):
        candidates.append(_exponential_sequence(n_final, float(lam), target_final))
    # deterministic windows
    candidates.append(_cosine_sequence(n_final, target_final))
    candidates.append(_sin_half_sequence(n_final, target_final))

    # a few random points on the simplex
    for _ in range(5):
        candidates.append(_random_simplex(n_final, target_final, rng))

    # reuse the best sequence from a previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        candidates.append(_resize_sequence(prev, n_final))
        # also try a reordered version of the previous result
        candidates.append(_alternating_order(candidates[-1]))

    # ------------------------------------------------------------------
    # 3️⃣ Quick evaluation – keep the best base shape
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for seq in candidates:
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Best raw candidate max={best_val:.6f}")

    # optional tiny re‑ordering step (usually helps a little)
    alt_seq = _alternating_order(best_seq)
    alt_val = _max_convolution_fft(alt_seq)
    if alt_val < best_val:
        best_val, best_seq = alt_val, alt_seq.copy()
    spread_seq = _even_spread_order(best_seq)
    spread_val = _max_convolution_fft(spread_seq)
    if spread_val < best_val:
        best_val, best_seq = spread_val, spread_seq.copy()

    # ------------------------------------------------------------------
    # 4️⃣ Adam‑style sub‑gradient descent on the chosen candidate
    # ------------------------------------------------------------------
    sub_deadline = deadline - 8.0                     # leave a few seconds for MT
    if sub_deadline > time.time():
        a_opt, val_opt = _adam_subgradient(best_seq,
                                           sub_deadline,
                                           target_final,
                                           lr=0.04)   # a modest learning‑rate
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
    print(f"[{time.time():.2f}] After Adam best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final short mass‑transfer simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = deadline - 0.5
        a_final, val_final = _mass_transfer_simulated_annealing(
            best_seq, mt_deadline, target_final,
            start_temp=0.2, end_temp=1e-5, low_frac=0.03)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
    print(f"[{time.time():.2f}] Final max after MT = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)

        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
