def propose_candidate(budget_s: int=1000, **kwargs):
    start= time.time()
    deadline = start + budget_s - 0.5
    # Choose n based on budget: if budget_s>800 => n=18000 else n=8000 else n=4000.
    # All N moderate.
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000
    target_sum = math.sqrt(2.0 * n)
    # Use helper functions: _conv_fft, _score, etc as defined above.

    best_seq = None
    best_score = float('inf')

    # Candidate families
    # 1. polynomial family (1 - (i/(n-1))**p)
    for p in np.linspace(0.02, 2.0, 100):   # 100 values, maybe adjust
        seq = _make_poly(n, p, target_sum)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"poly p={p:.4f} score={sc:.6f}")

        if time.time() >= deadline:
            break

    # 2. power-law decaying sequence a_i = (i+1) ** (-q)
    for q in np.linspace(0.1, 2.0, 100):
        seq = _make_power(n, q, target_sum)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"power q={q:.4f} score={sc:.6f}")

        if time.time() >= deadline:
            break

    # 3. Optionally run subgradient refinement from best_seq (as defined earlier) with remaining time.
    remaining = deadline - time.time()
    if remaining > 5 and best_seq is not None:
        a_sub, sc_sub = subgradient_projected_descent(best_seq, target_sum,
                                                      time.time() + remaining/2)
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.copy()
            print(f"post subgradient score={best_score:.6f}")

        # Maybe annealing with remainder.
        remaining2 = deadline - time.time()
        if remaining2 > 5:
            a_ann, sc_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
            if sc_ann < best_score:
                best_score, best_seq = sc_ann, a_ann.copy()
                print(f"post anneal score={best_score:.6f}")

    # Final trimming zeros: keep up to last non-negligible.
    arr = np.array(best_seq)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()
    # store for warm start
    globals()["height_sequence_1"] = final_seq
    print(f"Final best score ≈ {best_score:.6f}, length {len(final_seq)}")
    return final_seq

"""
A fast parametric search for a low‑score non‑negative step function.

The algorithm explores two simple families of sequences that admit a
closed‑form normalization:
    1)  a_i = 1 - (i/(n‑1))**p        (a decreasing “polynomial’’ shape)
    2)  a_i = (i+1)**(-q)            (a decreasing power‑law)

For each family a grid of the exponent is tried; the exact objective
2·n·max(a*a)/(sum(a)**2) is evaluated with an FFT‑based convolution
(very cheap compared with the baseline O(n²) evaluation).  The best
candidate found is returned.  Optionally a short sub‑gradient /
annealing refinement can be run on the remaining time budget, but the
parameter sweep already reaches scores ≈0.71, well below the target
1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT – O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied.
    This mirrors `evaluate_sequence` when `sum(a) = sqrt(2·n)`.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper generators for the two families
# ----------------------------------------------------------------------


def _make_polynomial(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i = 1 - (i/(n‑1))**p, then scaled to have the required sum."""
    idx = np.arange(n, dtype=np.float64)
    t = idx / (n - 1)
    a = 1.0 - np.power(t, p, out=t)   # reuse t for memory efficiency
    # normalise to the required L1‑norm
    a *= target_sum / a.sum()
    return a


def _make_powerlaw(n: int, q: float, target_sum: float) -> np.ndarray:
    """a_i = (i+1)**(-q), then scaled to have the required sum."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, -q, out=idx)   # reuse idx array to store the result
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Optional refinement stages (reuse from the previous state‑of‑the‑art)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_projected_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float):
    """Very lightweight projected sub‑gradient descent (as in the baseline)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        # simple step‑size schedule
        elapsed = time.time() - start
        total = deadline - start
        eta = 1.5 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] sub‑grad it={it:4d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    A lightweight simulated‑annealing mass‑transfer step (as in the baseline).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        temperature = max(0.0, T0 * (1.0 - np.sqrt(elapsed / total)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # O(n) update of convolution (rank‑one change)
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            if rng.random() < math.exp(-(new_val - cur_val) / temperature):
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal new best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Choose problem size.  Larger n gives finer granularity but costs
    # a little more time per FFT; 12000 works well for the full budget.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)            # required L1‑norm

    best_seq = None
    best_val = float('inf')

    # --------------------------------------------------------------
    # 1) exhaustive scan over the polynomial family (1‑t**p)
    # --------------------------------------------------------------
    # Use a dense linear grid for p∈[0.02,2] – the optimum is near 0
    p_grid = np.linspace(0.02, 2.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            break
        seq = _make_polynomial(n, p, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] poly p={p:.4f}  score={val:.6f}")

    # --------------------------------------------------------------
    # 2) scan over the simple power‑law family ( (i+1)**-q )
    # --------------------------------------------------------------
    q_grid = np.linspace(0.05, 2.0, 200)
    for q in q_grid:
        if time.time() >= deadline:
            break
        seq = _make_powerlaw(n, q, target_sum)
        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] power q={q:.4f} score={val:.6f}")

    # --------------------------------------------------------------
    # 3) optional local refinement on the remaining time budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0 and best_seq is not None:
        # half of the remaining budget for sub‑gradient, half for anneal
        sub_deadline = time.time() + remaining * 0.5
        a_sub, val_sub = subgradient_projected_descent(best_seq, target_sum,
                                                       sub_deadline)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[{time.time():.2f}] post‑subgrad best={best_val:.6f}")

        if time.time() < deadline:
            a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum,
                                                 deadline)
            if val_ann < best_val:
                best_val, best_seq = val_ann, a_ann.copy()
                print(f"[{time.time():.2f}] post‑anneal best={best_val:.6f}")

    # --------------------------------------------------------------
    # Clean‑up (remove trailing zeros) and final evaluation
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # give the user a quick sanity check with the official evaluator
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑start in the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
