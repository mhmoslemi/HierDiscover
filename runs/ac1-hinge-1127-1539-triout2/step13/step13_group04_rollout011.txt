"""
Improved search for a non‑negative step function that minimises the maximal
self‑convolution (the evaluation score).  The algorithm builds on the
four‑phase scheme from the previous state‑of‑the‑art (parametric sweep,
Adam GD, exact sub‑gradient, deterministic mass‑transfer) and adds two
new ingredients:

1. **Random mixture pre‑search** – random linear combinations of several
   parametric families (power‑law, exponential tail, Gaussian, sine) are
   generated, projected onto the simplex, and evaluated.  This enlarges
   the pool of promising initial points.

2. **Greedy stochastic local search** – after the deterministic mass‑transfer
   stage a cheap hill‑climbing loop repeatedly moves a small amount of mass
   from a “heavy” index to a “light” one.  Each move is accepted only if it
   strictly reduces the maximal self‑convolution; the step size decays
   with the remaining time.  This simple stochastic refinement is able to
   escape shallow local minima that the deterministic scheme can miss.

The search respects the time budget (default 1000 s) and returns a list of
floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of arbitrary length vectors x and y using FFT.
    Returns the first len(x)+len(y)-1 entries.
    """
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Centered Gaussian, clipped at the ends, normalised to the target sum."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _piecewise_power_sequence(
    n: int, p1: float, p2: float, split_frac: float, target_sum: float
) -> np.ndarray:
    """
    Two‑segment power‑law: exponent p1 up to split_frac·n, exponent p2 afterwards.
    """
    split = max(1, min(n - 1, int(split_frac * n)))
    a = np.empty(n, dtype=np.float64)
    a[:split] = (np.arange(1, split + 1, dtype=np.float64) ** (-p1))
    a[split:] = ((np.arange(split + 1, n + 1, dtype=np.float64)) ** (-p2))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (max‑convolution value)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective: maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # jitter every few iterations to leave shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]  # first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver: smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty‑descent (soft enforcement of a target bound)
# ----------------------------------------------------------------------


def _penalty_lower_max(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    target_max: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑descent on the penalty (max(conv) – target_max)_+.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty gradient (only violations)
        p = conv - target_max
        p[p < 0] = 0.0

        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(
                a + rng.normal(scale=target_sum * 0.001, size=n), target_sum
            )
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            a = _project_to_simplex(
                a + rng.normal(scale=target_sum * 0.001, size=n), target_sum
            )

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Stochastic greedy local search (new component)
# ----------------------------------------------------------------------


def _random_local_search(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: repeatedly move a tiny amount of mass from a “heavy”
    index to a “light” one, accepting only strictly improving moves.
    The step size decays with the remaining time.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _score(a)
    rng = np.random.default_rng()
    n = a.shape[0]

    # initial maximal epsilon as a fraction of the total mass
    max_eps_initial = target_sum * 0.02

    while time.time() < deadline:
        # remaining proportion of the budget, used to shrink the step size
        remain = max(1e-12, deadline - time.time())
        total = max(1e-12, deadline - start_time)
        eps_factor = remain / total  # linearly decaying
        max_eps = max_eps_initial * eps_factor

        # pick a donor roughly proportional to current mass
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-15:
            continue

        # pick a receiver preferring low‑mass positions
        inv = 1.0 / (a + 1e-12)
        receiver = rng.choice(n, p=inv / inv.sum())
        if donor == receiver:
            continue

        epsilon = max_eps * rng.random()
        epsilon = min(epsilon, a[donor])  # cannot give more than we have
        if epsilon <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= epsilon
        a_new[receiver] += epsilon

        new_val = _score(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new
            # early exit if we reached the target bound
            if best_val <= 1.503:
                print(f"[{time.time():.2f}] Random local search hit target {best_val:.6f}")
                break
        # otherwise reject the move (no uphill acceptance)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the maximal
    self‑convolution (the evaluation score).  The routine comprises:

    * Phase 0 – dense parametric pre‑search (power‑law, exponential tail,
      sine, Gaussian, and a two‑segment power law).
    * Phase 0b – random mixture generation.
    * Phase A – Adam‑style GD with several restarts.
    * Phase B – exact sub‑gradient descent.
    * Phase C – deterministic mass‑transfer.
    * Phase D – cheap stochastic local search.
    * Phase E – final high‑β Adam polishing.
    * Phase F – penalty descent towards a target bound.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (more budget → larger n)
    # ------------------------------------------------------------------
    if budget_s >= 950:
        n = 15000
    elif budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)  # Σa = √(2·n) → score = max‑convolution

    # ------------------------------------------------------------------
    # Phase 0 – parametric families (quick pre‑search)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.20, 1.10, 31)
    q_grid = np.linspace(0.0, 12.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (d) Gaussian bumps
    for sigma in np.linspace(n / 30, n / 5, 9):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (e) two‑segment power law (extra flexibility)
    for p1 in np.linspace(0.2, 0.8, 5):
        for p2 in np.linspace(0.8, 1.4, 5):
            split_frac = rng.uniform(0.2, 0.8)
            seq = _piecewise_power_sequence(
                n, float(p1), float(p2), float(split_frac), target_sum
            )
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (f) previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev = np.array(prev, dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (g) random mixtures of basis functions (new)
    mix_budget = min(30.0, deadline - time.time() - 5.0)
    while time.time() < start_time + mix_budget:
        # Randomly decide how many basis components (2‑4)
        k = rng.integers(2, 5)
        components = []
        weights = rng.random(k) + 0.1
        weights /= weights.sum()
        for _ in range(k):
            typ = rng.choice(["power", "exp", "gauss", "sin"])
            if typ == "power":
                p = rng.uniform(0.15, 1.2)
                comp = _power_law_sequence(n, p, 1.0)  # unit sum
            elif typ == "exp":
                p = rng.uniform(0.2, 1.0)
                q = rng.uniform(0.0, 12.0)
                comp = _exp_power_law_sequence(n, p, q, 1.0)
            elif typ == "gauss":
                sigma = rng.uniform(n / 30, n / 5)
                comp = _gaussian_sequence(n, sigma, 1.0)
            else:
                comp = _sin_sequence(n, 1.0)
            components.append(comp)
        mixed = np.zeros(n, dtype=np.float64)
        for w, comp in zip(weights, components):
            mixed += w * comp
        mixed = _project_to_simplex(mixed, target_sum)
        val = _score(mixed)
        if val < best_score:
            best_score = val
            best_seq = mixed.copy()
        # occasional tiny jitter to keep diversity
        if rng.random() < 0.1:
            jitter = rng.normal(scale=target_sum * 0.003, size=n)
            mixed_j = best_seq + jitter
            mixed_j = np.maximum(mixed_j, 0.0)
            mixed_j = _project_to_simplex(mixed_j, target_sum)
            valj = _score(mixed_j)
            if valj < best_score:
                best_score = valj
                best_seq = mixed_j.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec, target_sum, gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq, target_sum, subgrad_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 15.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # occasional global random move to shake out plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 12.0:
            break

    # ------------------------------------------------------------------
    # Phase D – Stochastic greedy local search (new)
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        local_deadline = deadline - 5.0
        a_loc, val_loc = _random_local_search(best_seq, target_sum, local_deadline, start_time)
        if val_loc < best_score:
            best_score = val_loc
            best_seq = a_loc.copy()
            print(f"[{time.time():.2f}] Random local search improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final polishing with high‑β Adam
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase F – Penalty descent to push the bound lower
    # ------------------------------------------------------------------
    if time.time() + 8.0 < deadline:
        # try to lower the bound a bit beyond the current best
        target = best_score - 0.0045
        a_pen, val_pen = _penalty_lower_max(best_seq, target_sum, deadline, target)
        if val_pen < best_score:
            best_score = val_pen
            best_seq = a_pen.copy()
            print(f"[{time.time():.2f}] Penalty descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final safety evaluation (does not affect the returned value)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
