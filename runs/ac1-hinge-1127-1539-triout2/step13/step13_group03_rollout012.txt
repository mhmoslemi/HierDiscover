def _adam_descent(a0, deadline, target_sum, lr_factor=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    a = _project_to_simplex(a0.copy(), target_sum)
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        # conv
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        maxb = conv.max()
        # I indicator for entries close to max
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)
        # compute gradient
        I_fft = np.fft.rfft(I, n=size)
        a_rev_fft = np.fft.rfft(a[::-1], n=size)
        grad_full = np.fft.irfft(I_fft * a_rev_fft, n=size)[:2*n-1]
        grad = 2.0 * grad_full[n-1 : n-1 + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr_factor * target_sum * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        # enforce nonneg via projection onto simplex
        a = _project_to_simplex(a, target_sum)
        # occasional small jitter to escape local minima
        if t % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={t} best={best_val:.6f}")
    return best_a, best_val

    n = len(f)

    from scipy.sparse import diags
    A = diags(f, offsets=range(n), shape=(size, n), format='csc')
    # set b_ub = M
    for iteration in range(max_iter):
        if time.time() >= deadline:
            break
        # Solve LP maximize sum(g) subject to A@g <= M
        c = -np.ones(n)  # minimize -sum(g)

        # recompute M
        M = _max_convolution_fft(f)
        print(f"[{time.time():.2f}] LP iter {iteration} M={M:.6f}")

"""
Hybrid large‑scale subgradient optimizer.

Strategy
---------

1. **Diverse seed generation (small n ≈ 2000).**
   Build a pool from several parametric families (power‑law,
   exponential, Gaussian, semicircle, random simplex).  Keep the
   best candidate w.r.t. the max‑convolution value.

2. **Upscale to a much larger problem (n = 20 k–50 k).**
   Linear interpolation, a tiny Gaussian jitter and an ℓ¹‑simplex
   projection give a smooth start for the large scale optimisation.

3. **Projected sub‑gradient descent (the main work).**
   Repeatedly compute the convolution, build the indicator of the
   (almost) maximal entries and obtain an exact sub‑gradient.
   Momentum (Adam‑style) and an adaptive step size keep the updates
   stable.  The loop runs until the time budget expires.

4. **Final mass‑transfer (simulated‑annealing) pass.**
   A greedy donor‑receiver move with a cooling schedule polishes the
   result.

All numbers are kept non‑negative, projected onto the ℓ¹‑simplex
∑a = √(2 n) (the normalisation that makes the evaluation equal to
max (a ∗ a)).  The best sequence found is stored globally as
`height_sequence_1` for possible reuse.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using an FFT (full linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Symmetric Gaussian centred on the middle of the array."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-((x - centre) ** 2) / (2.0 * sigma ** 2))
    a *= target_sum / a.sum()
    return a


def _semicircle_sequence(n: int, target_sum: float) -> np.ndarray:
    """Discrete approximation of a raised‑cosine (Hann) window."""
    x = np.linspace(-1.0, 1.0, n, dtype=np.float64)
    a = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple deterministic reorderings
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the sorted values evenly using a step coprime with n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (main optimiser)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.999,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exact max‑convolution sub‑gradient descent with momentum.
    Runs until `deadline` (unix timestamp) is reached.
    Returns the best sequence and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # full convolution
        conv = _conv_fft(a)
        maxb = conv.max()
        # indicator of (almost) maximal entries
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        # gradient = 2 * (I * a_rev) restricted to centre n entries
        a_rev_fft = np.fft.rfft(a[::-1], n=size)
        I_fft = np.fft.rfft(I, n=size)
        grad_full = np.fft.irfft(I_fft * a_rev_fft, n=size)[: len(I) + len(a) - 1]
        grad = 2.0 * grad_full[len(a) - 1 : len(a) - 1 + len(a)]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + eps)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 800 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.0008, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:6d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """

    # ------------------------------------------------------------------
    n_small = 2000
