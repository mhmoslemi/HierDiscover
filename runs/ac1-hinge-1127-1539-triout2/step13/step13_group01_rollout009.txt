"""
Propose a low‑score non‑negative step function.

The algorithm mixes two ideas that were not combined in the previous
implementation:

1. **Smoothed‑max projected gradient descent** –  
   The objective `max_k (a*a)[k]` is approximated by a soft‑max
   `f_β(a) = (1/β)·log Σ_k exp(β·b_k)` where `b = a*a`.  Its gradient is a
   weighted sum of the individual convolution gradients, which can be
   computed in *O(n log n)* with FFTs.  Starting with a small β gives a
   global smoothing; β is increased linearly so the algorithm finally
   concentrates on the true peak.  After each gradient step the vector is
   projected onto the simplex `{x≥0, Σx = √(2·n)}` to keep the normalisation
   fixed.

2. **Aggressive annealing refinement** –  
   After the gradient phase a fast simulated‑annealing mass‑transfer is
   applied (the routine from the previous state‑of‑the‑art).  It receives
   the current best vector and uses the remaining time budget.

Both phases respect a total wall‑clock budget (default 1000 s).  The best
sequence found is stored in the global variable `height_sequence_1` for
warm‑starting subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of *two* arrays using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is performed,
    i.e. the function assumes `sum(a)` is already the desired √(2·n).
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing (unchanged, reused)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from ‘bad’
    indices (large contribution to the current peak) to ‘good’ indices
    (small height).  The convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                    # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # exact O(n) update of the convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point – smoothed‑max gradient + annealing
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # a tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more freedom, but costs more)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 24000
    elif budget_s >= 600:
        n = 18000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)       # standard normalisation

    # --------------------------------------------------------------
    # Initialise a (warm‑start if possible, otherwise random)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        a = _project_to_simplex(warm, target_sum)
    else:
        rng = np.random.default_rng()
        a = rng.random(n)
        a = _project_to_simplex(a, target_sum)

    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng()
    start_grad = time.time()
    total_grad = deadline - start_grad
    it = 0

    # --------------------------------------------------------------
    # Smoothed‑max projected gradient descent
    # --------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_grad
        frac = elapsed / total_grad

        # 1) current convolution
        b = _conv_fft(a)                 # length 2n‑1

        # 2) soft‑max weights over the entries of b
        beta = 0.1 + 9.9 * frac          # start ≈0.1, end ≈10
        b_shift = b - b.max()            # for numerical stability
        w = np.exp(beta * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:                 # extremely unlikely, fallback
            w = np.ones_like(b) / b.size
        else:
            w = w / w_sum

        # 3) gradient of the soft‑max (2·cross‑correlation)
        a_rev = a[::-1]
        conv_g = _conv_fft_two(w, a_rev)          # length 3n‑2
        g = 2.0 * conv_g[n - 1:n - 1 + n]          # slice to size n

        # 4) step size (linearly decreasing)
        eta0 = 0.8
        eta_min = 0.001
        eta = eta0 * (1.0 - frac) + eta_min

        # 5) take a gradient step and project back onto the simplex
        a = a - eta * g
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to help escape flat regions
        if it % 1500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Grad it={it:5d} new best={best_score:.6f}")

        it += 1

        # If we are close to the deadline, break early so we reserve time
        # for the annealing refinement.
        if time.time() + 20.0 >= deadline:
            break

    # --------------------------------------------------------------
    # Aggressive annealing refinement on the best vector so far
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, score_ann = anneal_mass_transfer(
            best_a, target_sum, deadline
        )
        if score_ann < best_score:
            best_score = score_ann
            best_a = a_ann.copy()
            print(f"[{time.time():.2f}] After anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, truncation of trailing zeros and evaluation
    # --------------------------------------------------------------
    arr = np.array(best_a, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:

    final_seq = arr.tolist()

        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for possible warm‑starts in the next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
