Implementation details of swap local search:

def _swap_local_search(a, target_sum, deadline, max_trials=50000):
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()
    it=0
    while time.time() < deadline and it < max_trials:
        i, j = rng.integers(n, size=2)
        if i==j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]  # in-place swap
        cur_val = _max_convolution_fft(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            # keep this swap
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]
        it += 1
    return best_a, best_val

def _swap_local_search(a, target_sum, deadline, max_iter=20000):
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_fft(best_a)
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        i,j = rng.integers(0,n, size=2)
        if i==j:
            continue
        a[i], a[j] = a[j], a[i]  # swap
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]
        max_iter -= 1
    return best_a, best_val

"""
Propose a step‐function (non‑negative sequence) that minimises the
maximal self‑convolution.  The algorithm consists of

1.  A dense parametric pre‑search (power‑law, exponential tail, sine,
    Gaussian, logistic and random Gaussian mixtures).
2.  Iterative penalty‑descent which directly minimises the excess
    (max(conv)‑target)+ using the exact convolution.
3.  A simple random‑swap local search (hill‑climbing on the permutation
    of the entries) – this often reduces the peak further because the
    objective depends heavily on the ordering of the coefficients.
4.  An optional final Adam‑style projected gradient polish.

All phases respect the overall time budget (default 1000 s).  The
objective of the optimisation is exactly the maximal entry of the
linear self‑convolution, because the sequence is kept on the simplex
{a_i≥0, Σa_i = √(2·n)}.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of appropriate length."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (exact linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex with sum s (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (quick generators)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/(n+1))."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float,
                       target_sum: float) -> np.ndarray:
    """Centered Gaussian (clipped at the ends) normalised to target_sum."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _logistic_sequence(n: int, scale: float,
                       target_sum: float) -> np.ndarray:
    """Logistic curve a_i = 1/(1+exp((i-mid)/scale))."""
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((x - mid) / scale))
    a *= target_sum / a.sum()
    return a


def _random_mixture_gaussian(n: int, target_sum: float,
                             rng: np.random.Generator,
                             max_bumps: int = 5) -> np.ndarray:
    """Random super‑position of a few Gaussian bumps."""
    nb = rng.integers(1, max_bumps + 1)
    a = np.zeros(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    for _ in range(nb):
        centre = rng.integers(0, n)
        sigma = rng.uniform(n * 0.01, n * 0.15)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:                     # fallback – never happens in practice
        a = _sin_sequence(n, target_sum)
    else:
        a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Penalty‑descent – minimise (max(conv) – target)_+
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray, target_sum: float,
                       deadline: float, target_max: float) -> tuple[np.ndarray, float]:
    """
    Try to find a feasible vector with max(conv) ≤ target_max.
    The descent direction is the gradient of the penalty
    (max(conv) - target_max)_+.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    it = 0

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty vector (positive part only)
        p = conv - target_max
        p[p < 0] = 0.0

        # gradient = 2 * (p ⋆ a_rev)_{window of length n}
        a_rev = a[::-1]
        size = _next_pow2(p.size + a_rev.size - 1)
        fp = np.fft.rfft(p, n=size)
        farev = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(fp * farev, n=size)[: p.size + a_rev.size - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n),
                                    target_sum)
            it += 1
            continue

        # step size (sub‑gradient rule)
        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        # backtracking line‑search (max 5 reductions)
        improved = False
        for _ in range(5):
            cand = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(cand)
            if cand_max < cur_max - 1e-12:
                a = cand
                best_max = cand_max
                best_a = cand.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n),
                                    target_sum)
        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Random swap (hill‑climbing) on the permutation of entries
# ----------------------------------------------------------------------


def _swap_local_search(a0: np.ndarray, target_sum: float,
                       deadline: float, max_iter: int = 30000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly try to swap two entries.
    If the swap reduces the maximal convolution it is kept,
    otherwise it is reverted.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]           # perform swap
        cur_val = _max_convolution_fft(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            # keep the swap
        else:
            a[i], a[j] = a[j], a[i]       # revert
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Soft‑max / Adam‑style descent (used for a final polish)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray, deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution.  The search consists of a fast parametric sweep,
    a penalty‑descent phase, a random‑swap local optimisation and a final
    Adam polish.  The routine respects the global time budget.
    """
    start = time.time()
    deadline = start + budget_s - 0.5                     # safety margin

    # problem size – larger n gives more freedom but costs more time
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)                         # Σa = √(2·n)

    rng = np.random.default_rng()
    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Phase 0 – parametric families + cheap random mixtures
    # --------------------------------------------------------------
    # a) uniform (reference)
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    val = _max_convolution_fft(uniform)
    best_seq, best_val = uniform.copy(), val

    # b) power‑law
    for p in np.linspace(0.2, 1.3, 23):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_seq, best_val = seq.copy(), val

    # c) exponential‑tail power‑law
    for p in np.linspace(0.2, 1.2, 11):
        for q in np.linspace(0.0, 8.0, 11):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_val:
                best_seq, best_val = seq.copy(), val

    # d) sine shape
    seq = _sin_sequence(n, target_sum)
    val = _max_convolution_fft(seq)
    if val < best_val:
        best_seq, best_val = seq.copy(), val

    # e) Gaussian bumps
    for sigma in np.linspace(n / 20, n / 4, 7):
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_seq, best_val = seq.copy(), val

    # f) Logistic curves (different scales)
    for scale in np.linspace(0.05 * n, 0.5 * n, 9):
        seq = _logistic_sequence(n, float(scale), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_seq, best_val = seq.copy(), val

    # g) Random mixtures of a few Gaussians (cheap stochastic search)
    while time.time() < deadline - 30.0:
        seq = _random_mixture_gaussian(n, target_sum, rng,
                                       max_bumps=rng.integers(1, 6))
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_seq, best_val = seq.copy(), val

    print(f"[{time.time():.2f}] After parametric search best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Penalty descent (try to push the peak down)
    # --------------------------------------------------------------
    # Repeatedly lower the target a little bit; stop when no progress.
    while time.time() < deadline - 10.0:
        # aim for a modest reduction; never go below zero
        reduction = max(0.0008, best_val * 0.005)
        target = best_val - reduction
        if target <= 0.0:
            break
        a_pen, val_pen = _penalty_lower_max(best_seq, target_sum,
                                            deadline, target)
        if val_pen < best_val - 1e-12:
            best_seq, best_val = a_pen.copy(), val_pen
            print(f"[{time.time():.2f}] Penalty descent improved to {best_val:.6f}")
        else:
            # cannot achieve the tighter bound → stop
            break

    # --------------------------------------------------------------
    # Phase 2 – Random‑swap local optimisation on the ordering
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_swap, val_swap = _swap_local_search(best_seq, target_sum, deadline)
        if val_swap < best_val - 1e-12:
            best_seq, best_val = a_swap.copy(), val_swap
            print(f"[{time.time():.2f}] Swap search improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Final Adam‑style polish (if time permits)
    # --------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq, deadline,
                                                target_sum,
                                                max_beta=5e5,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_val - 1e-12:
            best_seq, best_val = a_fin.copy(), val_fin
            print(f"[{time.time():.2f}] Final Adam polish improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final safety evaluation (for debugging)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
