    norm = np.linalg.norm(grad)

    step = avg * 2.0 / np.sqrt(it + 1)
    a_candidate = a - (grad / norm) * step
    a_candidate = _project_to_simplex(a_candidate, target_sum)

    cand_score = _max_convolution_fft(a_candidate)

        best_a = a_candidate.copy()
        print(f"[{time.time():.2f}] Subgrad it={it} new best {best_score:.6f}")

    a = a_candidate
    cur_score = cand_score
    it+=1

"""
Step‑function optimizer (subgradient + mass‑transfer).

The algorithm works in three stages within the allotted 1000 s budget:

1. **Initialisation** – use the best known sequence (global ``height_sequence_1``)
   if available, otherwise fall back to a simple power‑law shape.  The
   sequence is resized to the chosen length and projected onto the simplex
   {a_i ≥ 0, Σ a_i = √(2 n)}.

2. **Projected subgradient descent** – minimise the maximal self‑convolution
   ``max(a*a)`` (equivalently the evaluation score when Σa = √(2 n)).
   At each iteration the active‑set mask of convolution entries attaining the
   current maximum is built; the subgradient of the max‑convolution is obtained
   by a convolution of this mask with the reversed sequence (FFT‐based, O(n log n)).
   A step of size proportional to the average entry (decreasing as 1/√t) is taken
   in the negative subgradient direction and projected back onto the simplex.
   The best sequence seen so far is recorded.

3. **Exact deterministic mass‑transfer (DMT2)** – a greedy “peak‑flattening”
   routine (same as in the reference code) is applied for a short final polishing
   phase.

All phases respect the time budget; the final list is trimmed of trailing
near‑zero entries and stored in ``globals()['height_sequence_1']`` for reuse.

The returned sequence consists of non‑negative floats and, after the final
projection, has Σa = √(2 n), so the evaluation score equals the maximal
self‑convolution value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used only for a fallback initial guess)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index
    to a “receiver’’ index so that the current peak drops to the
    second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # choose one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # coefficients that pair with idx
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # choose donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected subgradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple subgradient method for minimizing max(a*a) under the simplex constraint.
    The step size is proportional to the average entry (decreasing as 1/√t).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    avg = target_sum / n
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    it = 0
    while time.time() < deadline:
        # current convolution and active set
        conv = _conv_fft(a)
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)   # length 2n‑1

        # subgradient = 2 * (mask * a_rev) convolution, slice n‑1 … n‑1+n‑1
        a_rev = a[::-1]
        size = 1 << ((len(mask) + n - 1).bit_length())
        fm = np.fft.rfft(mask, n=size)
        fr = np.fft.rfft(a_rev, n=size)
        full = np.fft.irfft(fm * fr, n=size)[: len(mask) + n - 1]
        grad = 2.0 * full[n - 1 : n - 1 + n]      # length n

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        # step length (decreases like 1/sqrt(it+1))
        step_len = avg * 2.0 / np.sqrt(it + 1)

        # move opposite to the subgradient and project onto the simplex
        direction = -grad / norm
        a_candidate = a + direction * step_len
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        cand_score = _max_convolution_fft(a_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} new best={best_score:.6f}")

        # always move (the method is guaranteed to converge with diminishing steps)
        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal
    self‑convolution.  The function respects the time budget and returns
    the best list of heights found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom but costs more per iteration)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n) makes the evaluation score = max convolution

    # ------------------------------------------------------------------
    # Initialise from the best known construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # linear interpolation to the new size
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, prev)
        else:
            seq = prev.copy()
    else:
        # fallback: simple power‑law shape
        seq = _power_law_sequence(n, 0.8, target_sum)

    seq = _project_to_simplex(seq, target_sum)

    best_seq = seq.copy()
    best_score = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Small random jitter phase (quick improvements)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.001, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Jitter improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – projected subgradient descent (main work)
    # ------------------------------------------------------------------
    sub_deadline = time.time() + (deadline - time.time()) * 0.75
    best_seq, best_score = _subgradient_descent(best_seq, target_sum, sub_deadline)
    print(f"[{time.time():.2f}] After subgradient best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        mt_deadline = deadline
        best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                                   mt_deadline,
                                                                   target_sum)
        print(f"[{time.time():.2f}] After final DMT best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]

    except Exception:

    # store for possible reuse by subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
