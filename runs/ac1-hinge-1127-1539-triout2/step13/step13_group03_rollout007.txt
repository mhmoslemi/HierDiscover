def _golden_interleaver(seq):
    n = len(seq)
    sorted_vals = np.sort(seq)[::-1]
    phi = (np.sqrt(5) - 1) / 2
    step = int(round(phi * n))

    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

"""
Hybrid optimizer – large‑scale annealed gradient + sub‑gradient + mass‑transfer.

The pipeline:
1. Load the previous best sequence (if any) and optionally improve it on a modest size.
2. Upscale to a large length (up to ~100 k) and spread the large coefficients
   using a golden‑step permutation (low‑discrepancy interleaver).
3. Run a long annealed‑gradient descent on the soft‑max approximation of the
   max‑convolution.  The schedule drives the smoothing parameter β up while
   projecting onto the ℓ¹‑simplex (∑a = √(2n), a_i ≥ 0).
4. Refine with an exact sub‑gradient method (max‑convolution indicator).
5. Final greedy mass‑transfer simulated‑annealing pass.
All stages respect the constraints and keep track of the best sequence found.
The algorithm adapts the problem size to the remaining time budget and
uses FFT‑based convolutions (O(n log n)) so that even n≈10⁵ runs comfortably
within the 1000 s limit.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a * a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ i^{‑p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(‑λ i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) shape."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π i/(n‑1)).  Peaks in the centre."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large coefficients with a coprime stride."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _golden_interleaver(seq: np.ndarray) -> np.ndarray:
    """Low‑discrepancy interleaver based on the golden ratio."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    phi = (math.sqrt(5.0) - 1.0) / 2.0          # ~0.618
    step = int(round(phi * n))
    # ensure the step is coprime with n
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 8e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max of the convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        # soft‑max probabilities (numerically stable)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # Indicator of (almost) maximal entries
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        # grad = 2 * (I * a_rev) centered on the n positions
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter for robustness
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Move a little mass from “donor” to “receiver” pairs, accepting worse
    moves with a temperature schedule."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where convolution attains its maximum
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product a_i * a_{kmax-i} for all contributing pairs
        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # rollback
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake things up
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine expands the sequence to a large size, runs a long annealed‑gradient
    descent on the soft‑max approximation of the max‑convolution, refines with a
    sub‑gradient pass and finishes with a brief mass‑transfer simulated annealing.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Stage 0 – initialise a modest‑size seed (≈5 k)
    # --------------------------------------------------------------
    n0 = 5000
    target0 = math.sqrt(2.0 * n0)

    if "height_sequence_1" in globals():
        seed_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if seed_seq.shape[0] != n0:
            seed_seq = _resize_sequence(seed_seq, n0)
    else:
        # generic power‑law seed
        seed_seq = _power_law_sequence(n0, 0.8, target0)

    seed_seq = _project_to_simplex(seed_seq, target0)

    # quick local improvement on the modest size
    if time.time() < deadline - 150.0:
        modest_deadline = time.time() + 30.0
        seed_seq, _ = _annealed_gradient_descent(
            seed_seq, modest_deadline, target0,
            max_beta=5e5, step_factor=0.7, gamma=0.98)

    # --------------------------------------------------------------
    # Stage 1 – upscale to a large problem size
    # --------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 600:
        n_big = 100000
    elif remain > 400:
        n_big = 80000
    elif remain > 250:
        n_big = 60000
    elif remain > 150:
        n_big = 40000
    elif remain > 80:
        n_big = 30000
    else:
        n_big = max(20000, n0)

    target_big = math.sqrt(2.0 * n_big)

    seq = _resize_sequence(seed_seq, n_big)
    seq = _project_to_simplex(seq, target_big)

    # try a few deterministic interleavings and keep the best
    candidates = [
        seq,
        _alternating_order(seq),
        _even_spread_order(seq),
        _golden_interleaver(seq)
    ]
    seq = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, init max={_max_convolution_fft(seq):.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long annealed‑gradient descent on the large scale
    # --------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 200.0:
        ag_deadline = time.time() + remain * 0.55
        seq, best_val = _annealed_gradient_descent(
            seq, ag_deadline, target_big,
            max_beta=2e6, step_factor=0.8, gamma=0.98)
        print(f"[{time.time():.2f}] After annealed GD max={best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – exact sub‑gradient refinement
    # --------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 80.0:
        sg_deadline = time.time() + remain * 0.30
        seq, best_val = _subgradient_descent(
            seq, sg_deadline, target_big,
            step_factor=0.5, gamma=0.995)
        print(f"[{time.time():.2f}] After sub‑grad max={best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final mass‑transfer simulated annealing (short)
    # --------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 10.0:
        mt_deadline = time.time() + remain - 2.0
        seq, best_val = _mass_transfer_simulated_annealing(
            seq, mt_deadline, target_big,
            start_temp=0.25, end_temp=1e-5, low_frac=0.05)
        print(f"[{time.time():.2f}] Final MT max={best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and evaluate
    # --------------------------------------------------------------
    arr = np.array(seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for successive runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
