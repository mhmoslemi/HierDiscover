def _smooth_gradient_descent(a0, deadline, target_sum, alpha=5.0, init_lr=0.5, decay=0.95, min_lr=1e-6):

   a = a0.copy()

   best_a = a.copy(); best_score = _score(a)

   lr = init_lr

   while time.time() < deadline and lr > min_lr:

       conv = _conv_fft(a)   # length 2n-1

       M = conv.max()

       # compute weights w = exp(alpha * (conv - M))

       w = np.exp(alpha * (conv - M))

       Z = w.sum()
       if Z == 0.0:
           # fallback: use uniform weights
           w = np.ones_like(w)
           Z = w.sum()
       # compute convolution of w with reversed a
       a_rev = a[::-1]
       conv_w_rev = _conv_fft(w, a_rev)   # length (2n-1) + n - 1 = 3n-2
       # extract slice
       n = a.shape[0]
       Tg = conv_w_rev[n-1:n-1+n]   # shape n

       grad = a + Tg / Z

       # Compute step size based on norm
       # Common: lr = target_sum * step_factor / (norm + eps). We'll compute norm and adapt.
       norm = np.linalg.norm(grad)
       if norm == 0:
           break
       step = (target_sum * 0.5) / (norm + 1e-12)   # maybe constant factor 0.5 as step factor
       # Optionally apply lr scaling:
       step *= lr

       a_new = a - step * grad
       a_new = _project_to_simplex(a_new, target_sum)
       new_score = _score(a_new)
       if new_score < best_score:
           best_score = new_score
           best_a = a_new.copy()
           # increase lr slightly
           lr *= 1.02
       else:
           # no improvement: reduce lr
           lr *= decay
       a = a_new

   return best_a, best_score

"""
Hybrid optimiser for the step‑function max‑convolution problem.

The algorithm works in several stages:

1. **Seed generation** – a mix of power‑law, exponential,
   random simplex points and the previously best construction.  All
   seeds are projected onto the ℓ¹‑simplex Σa = √(2 n).

2. **Projected Differential Evolution** – a modest population evolves
   with classic DE mutation / binomial crossover while staying feasible.
   The population size grows with the problem dimension.

3. **Smoothed gradient descent** – instead of a sub‑gradient on the
   exact max, we minimise a smooth surrogate
   `f(a)=log_sum_exp(α·(a*a)) / α`.  The gradient can be evaluated
   efficiently by two FFT‑based convolutions, allowing many fast
   iterations that often lower the peak of the autocorrelation.

4. **Deterministic mass‑transfer polishing** – a greedy move of mass
   from a “donor” index (contributing most to the peak) to a “receiver”
   index, with a Metropolis‐style acceptance schedule.

5. **Short sub‑gradient descent** – a final fine‑tuning on the exact
   objective.

6. **Reordering** – three simple permutations (original, alternating,
   even‑spread) are tried and the best ordering is kept.

7. **Trim & return** – trailing near‑zero entries are removed for
   compactness and the result is stored globally for warm‑starts.

All steps respect the global time budget (default 1000 s) and print
progress information.  The final sequence is returned as a plain list
of floats, ready to be evaluated by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_self(a).max())


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """
    Objective `2·n·max(a*a) / (Σa)²`.  When Σa = √(2n) this equals `max(a*a)`.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a cyclic step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with Euclidean simplex projection after every step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # distinct mates a,b,c ≠ i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # small safety margin
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha: float = 5.0,
                             init_step_factor: float = 0.6,
                             decay: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimise `log_sum_exp(alpha·(a*a)) / alpha` by projected gradient descent.
    The surrogate approaches the true max as `alpha→∞`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    step_factor = init_step_factor
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)                     # length 2n‑1
        M = conv.max()
        # weights for the smooth max (never overflow because (conv-M) ≤ 0)
        w = np.exp(alpha * (conv - M))
        Z = w.sum()
        if Z == 0.0:
            w[:] = 1.0
            Z = w.size

        # term Σ_i w_i a_{i‑j}
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)    # length 3n‑2
        Tg = conv_w_rev[n - 1:n - 1 + n]         # slice of length n

        # gradient of the surrogate (see analysis)
        grad = a + Tg / Z

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        # step size inversely proportional to the gradient norm
        lr = step_factor * target_sum / (norm + 1e-12)

        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            step_factor = min(step_factor * 1.03, 2.0)   # be a little bolder
        else:
            step_factor = max(step_factor * decay, 1e-4)

        a = a_new

    return best_a, best_score

# Deterministic mass‑transfer polishing (same as previous version)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_self(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_self(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val

    if budget_s >= 800:
        n = 8000
    elif budget_s >= 600:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------- seed generation -------------------
    seeds: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.15, 1.2, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed to increase diversity
    jittered: list[np.ndarray] = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        jittered.append(seq_j)
    seeds.extend(jittered)

    # --------------------- initialise DE population ----------
    pop_size = max(30, n // 200)          # ~30‑40 individuals
    population: list[np.ndarray] = []

    # start with the best seed we have
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------- DE phase -------------------------
    de_deadline = min(start_time + 0.55 * budget_s,
                      deadline - 30.0)          # leave time for polishing
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished; best score = {best_score:.6f}")

    # --------------------- Smoothed gradient descent -------
    if time.time() < deadline - 20.0:
        smooth_deadline = time.time() + min(30.0, deadline - time.time() - 15.0)
        # start with moderate alpha, then finish with a larger one
        seq_s, score_s = _smooth_gradient_descent(best_seq,
                                                  smooth_deadline,
                                                  target_sum,
                                                  alpha=4.0,
                                                  init_step_factor=0.7)
        if score_s < best_score:
            best_score, best_seq = score_s, seq_s.copy()
            print(f"[{time.time():.2f}] After smooth (α=4) best = {best_score:.6f}")

        # if still time, a second pass with a sharper surrogate
        if time.time() < deadline - 12.0:
            smooth_deadline2 = time.time() + min(15.0,
                                                deadline - time.time() - 10.0)
            seq_s2, score_s2 = _smooth_gradient_descent(best_seq,
                                                        smooth_deadline2,
                                                        target_sum,
                                                        alpha=10.0,
                                                        init_step_factor=0.5)
            if score_s2 < best_score:
                best_score, best_seq = score_s2, seq_s2.copy()
                print(f"[{time.time():.2f}] After smooth (α=10) best = {best_score:.6f}")

    # --------------------- deterministic mass‑transfer ------
    if time.time() < deadline - 10.0:
        mt_dead = time.time() + min(12.0, deadline - time.time() - 8.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # --------------------- short sub‑gradient descent -------
    if time.time() < deadline - 5.0:
        sg_dead = time.time() + min(10.0,
                                    deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(best_seq,
                                                sg_dead,
                                                target_sum,
                                                step_factor=0.45,
                                                gamma=0.995)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # --------------------- reordering attempts ---------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        best_seq[::-1]                     # simple reversal
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # --------------------- optional up‑sampling & polish -----
    # If we still have a few seconds, try to double the length and polish.
    if time.time() < deadline - 4.0 and n < 12000:
        up_len = min(12000, n * 2)
        up_seq = _resize_sequence(np.array(best_seq), up_len)
        up_seq = _project_to_simplex(up_seq, np.sqrt(2.0 * up_len))
        # a very short smoothing run on the larger vector
        up_dead = time.time() + 3.0
        up_seq, up_score = _smooth_gradient_descent(up_seq,
                                                    up_dead,
                                                    np.sqrt(2.0 * up_len),
                                                    alpha=6.0,
                                                    init_step_factor=0.5)
        # keep whichever is better (note that scores are directly comparable)
        if up_score < best_score:
            best_seq = up_seq
            best_score = up_score
            print(f"[{time.time():.2f}] Upsampled polishing improved to {best_score:.6f}")

    # --------------------- final trimming --------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------- store & return --------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
