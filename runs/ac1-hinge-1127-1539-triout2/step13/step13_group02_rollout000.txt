"""
Propose a low‐peak step function.

The search proceeds in three stages:

1. **Exploration** – generate a diverse pool of seeds (power‑law,
   exponential, random Dirichlet, previous best).  For each seed run a
   short soft‑max gradient descent (high‑β smoothed max) or a short
   exact sub‑gradient descent.  Keep the best candidate.

2. **Upscaling & symmetry** – linearly interpolate the best candidate to
   a larger length (≈10 k) and enforce exact mirror symmetry (a[i] = a[n‑1‑i]).
   Symmetry is known to reduce the autocorrelation peak.

3. **Intensive refinement** – a long high‑β gradient descent followed by
   a deterministic mass‑transfer annealing and a long exact sub‑gradient
   descent, all while keeping the vector on the ℓ¹‑simplex
   Σa = √(2·n) and symmetric.  The final sequence is trimmed of trailing
   zeros and stored globally as ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force mirror symmetry a[i] == a[n-1-i]."""
    return 0.5 * (a + a[::-1])


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Family generators (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum annealed gradient descent on smoothed max (soft‑max)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the soft‑max approximation."""
    a = _project_to_simplex(a0.copy(), target_sum)
    a = _symmetrize(a)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _symmetrize(a)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 400 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on true peak (quadratic, cheaper)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    a = _symmetrize(a)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t. a
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]        # factor 2 is exact for all i

        # diminishing step size
        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 3000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = _symmetrize(a)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (annealed)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # rollback
            a[donor] += delta
            a[receiver] -= delta
            a = _symmetrize(a)
            a = _project_to_simplex(a, target_sum)

        if it % 1500 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _symmetrize(a)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine uses diverse random seeds, short gradient/sub‑gradient runs,
    up‑scaling with symmetry, and a long final refinement (high‑β GD,
    mass‑transfer, sub‑gradient).  The best sequence is stored in the global
    variable ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1. Choose a modest length for rapid exploration
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 500:
        n_expl = 5000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 2. Build a diverse pool of seeds
    # --------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    for p in np.linspace(0.3, 0.8, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    for lam in np.linspace(0.001, 0.02, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    for _ in range(6):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # Add jittered copies
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        seq_j = _symmetrize(seq_j)
        init_pool.append(seq_j)

    # --------------------------------------------------------------
    # 3. Exploration – short runs of GD / sub‑gradient
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    exploration_deadline = deadline - 70.0   # keep > 1 min for final phases

    while time.time() < exploration_deadline:
        # pick a seed (biased towards current best)
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
            seed = _symmetrize(seed)
        else:
            seed = rng.choice(init_pool)

        # decide optimiser
        if rng.random() < 0.55:
            # short soft‑max gradient descent (≈12 s)
            run_dead = min(time.time() + 12.0, exploration_deadline)
            a_opt, val_opt = _annealed_gradient_descent(
                seed, run_dead, target_expl,
                max_beta=3e5, step_factor=0.6, gamma=0.99)
        else:
            # short exact sub‑gradient descent (≈10 s)
            run_dead = min(time.time() + 10.0, exploration_deadline)
            a_opt, val_opt = _subgradient_descent(
                seed, run_dead, target_expl, max_iter=150_000)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

        if time.time() >= exploration_deadline:
            break

    # --------------------------------------------------------------
    # 4. Upscaling & symmetry enforcement
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    # Determine final length based on remaining time
    remaining = deadline - time.time()
    if remaining > 60.0 and budget_s >= 800:
        n_final = 12000
    elif remaining > 30.0:
        n_final = 10000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _symmetrize(seq_big)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few systematic orderings (alternating tends to flatten peaks)
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _alternating_order(_alternating_order(seq_big))  # second pass
    ]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        cand = _symmetrize(cand)
        cand = _project_to_simplex(cand, target_final)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Intensive refinement (high‑β GD)
    # --------------------------------------------------------------
    if time.time() < deadline - 45.0:
        gd_dead = deadline - 45.0
        a_ref, val_ref = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=2e6,
            step_factor=0.45,
            gamma=0.995)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] High‑β GD best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Deterministic mass‑transfer annealing (long)
    # --------------------------------------------------------------
    if time.time() < deadline - 25.0:
        mt_dead = deadline - 25.0
        a_mt, val_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.7,
            end_temp=1e-6,
            low_frac=0.04)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Final exact sub‑gradient descent (until time runs out)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_dead = deadline - 3.0
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=500_000)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Final sub‑grad best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Trim trailing zeros and finalize
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional logging with the official evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
