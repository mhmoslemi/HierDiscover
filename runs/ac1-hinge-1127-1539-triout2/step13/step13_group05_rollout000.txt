"""
Propose a non‑negative step function that minimizes the max self‑convolution
score returned by ``evaluate_sequence``.

The algorithm mixes several ideas that proved useful in earlier
implementations, but with a few new twists:

* **Symmetrisation** – the optimum appears to be (almost) symmetric,
  therefore after every modification we replace ``a`` by ``(a+a[::-1])/2``.
* **Parametric warm‑start** – dense scans of power‑law, tilted
  power‑law and Gaussian families (and a log‑normal random family) are
  used to obtain a good initial sequence.
* **Projected Adam optimisation with a temperature schedule** – we minimise
  a soft‑max approximation of the peak using Adam, slowly raising the
  ``beta`` parameter (the “temperature”) from a very smooth regime up to
  10⁶.  Multiple restarts driven by jitter improve robustness.
* **Aggressive global peak‑mass transfer** – a refined version of the
  “donor‑mass → single receiver” step that tries several donor/receiver
  fractions and keeps the best move.
* **Exact deterministic mass‑transfer (DMT2)** and **pairwise optimal
  transfer** – greedy moves that exactly respect the peak constraint.
* **Final polishing** – a short Adam run with a very high ``beta`` to
  finalise the solution.

All steps respect the global time budget (default 1000 s).  The resulting
array is trimmed of trailing near‑zero entries before being returned.
"""

import time

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT. Result length = 2*len(a)-1."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Objective value: 2·n·max(b)/(sum(a)²).  When sum(a)=target_sum
    (which equals sqrt(2·n)) the denominator equals 2·n, so the score
    is exactly max(b).  The function returns a plain float.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = S, x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σ x_i = S} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Return a symmetrised copy: (a + a[::-1]) / 2."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Parametric families (all return a normalised to target_sum)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return _symmetrize(a)


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return _symmetrize(a)


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Centered Gaussian (discrete) shape."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return _symmetrize(a)


def _lognormal_sequence(n: int, mu: float, sigma: float,
                        target_sum: float,
                        rng: np.random.Generator) -> np.ndarray:
    """Log‑normal random shape (positive, heavy‑tailed)."""
    a = rng.lognormal(mean=mu, sigma=sigma, size=n)
    a *= target_sum / a.sum()
    return _symmetrize(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(b) where b = a*a.
    Returns (grad, b) where grad is the ∂softmax/∂a vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # avoid overflow
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]   # same length as a
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max) with temperature schedule
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   beta_start: float = 0.1,
                   beta_max: float = 1e6,
                   beta_inc_iter: int = 300,
                   step_factor: float = 0.8,
                   gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam gradient descent on the soft‑max approximation of the max‑convolution.
    The temperature (beta) is doubled every ``beta_inc_iter`` iterations
    until ``beta_max`` is reached.
    Returns the best sequence found inside the time window and its score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a, target_sum)

    beta = beta_start
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1.0 - gamma) * grad
        v = gamma * v + (1.0 - gamma) * (grad * grad)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)
        a = _symmetrize(a)                     # keep symmetry

        # occasional jitter to help escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
            a = _symmetrize(a)

        cur_val = _score(a, target_sum)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < beta_max:
            beta = min(beta * 2.0, beta_max)

        it += 1

    return best_a, best_val


def _adam_restarts(start_seq: np.ndarray,
                  target_sum: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform several Adam runs with jittered starts until the deadline.
    """
    best_seq = start_seq.copy()
    best_val = _score(best_seq, target_sum)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        # allocate a short slice for this restart
        remaining = deadline - time.time()
        slice_budget = min(30.0, remaining * 0.3)
        jitter = rng.normal(scale=target_sum * 0.02, size=best_seq.shape)
        start = best_seq + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)
        start = _symmetrize(start)

        a_opt, val_opt = _adam_optimize(start,
                                        target_sum,
                                        time.time() + slice_budget,
                                        beta_start=0.1,
                                        beta_max=2e5,
                                        step_factor=0.8,
                                        gamma=0.995)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       target_sum: float,
                                       deadline: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy move that reduces the current peak of the convolution by moving
    a single donor to a single receiver while staying feasible.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a, target_sum)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest entry (used as stopping condition)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a, target_sum)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass transfer
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               target_sum: float,
                               deadline: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise mass transfer: move mass from a large coefficient
    (donor) to a small one (receiver) while reducing the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a, target_sum)

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                           a,
                           np.zeros(pad_len, dtype=np.float64)])

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        # pre‑compute shifted views for the involved indices
        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect vector when moving mass
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a, target_sum)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (single‑receiver version, with flexible fractions)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               target_sum: float,
                               deadline: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Reduce the current convolution peak by moving a *fraction* of the total
    donor mass to a single weak receiver.  The fraction α is chosen as large
    as possible while guaranteeing that no other entry exceeds the previous
    maximum.  The best receiver (and α) is kept.
    """
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_score = _score(a, target_sum)
    it = 0

    base_template = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                                   a,
                                   np.zeros(pad_len, dtype=np.float64)])

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second max (stop if peak already flat)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max - i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        # contribution of donors alone (for linear constraints)
        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        # pre‑build padded view for the current a (updated each loop)
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # shifted view for receiver r
            diff = donor_mass * shift_r - conv_donors   # α·diff ≤ 0 component‑wise
            mask_pos = diff > 0
            if np.any(mask_pos):
                # α ≤ conv_donors / (donor_mass * shift_r) for those positions
                # avoid division by zero
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _symmetrize(a)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a, target_sum)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Global peak it={it} α={best_alpha:.4g} new_score={best_score:.6f}")

        it += 1

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    max self‑convolution (equivalently the score returned by
    evaluate_sequence).  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem size ------------------------------------------------
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 13000
    else:
        n = 8000

    target_sum = math.sqrt(2.0 * n)                # Σa = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric warm‑start (including random log‑normals)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # 1) Power‑law family
    for p in np.linspace(0.30, 0.85, 27):
        a = _power_law_sequence(n, float(p), target_sum)
        val = _score(a, target_sum)
        if val < best_score:
            best_score = val
            best_seq = a.copy()

    # 2) Tilted power‑law
    for p in np.linspace(0.30, 0.85, 15):
        for q in np.linspace(0.0, 5.0, 12):
            a = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(a, target_sum)
            if val < best_score:
                best_score = val
                best_seq = a.copy()

    # 3) Gaussian family
    for sigma in np.linspace(0.15 * n, 0.45 * n, 7):
        a = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(a, target_sum)
        if val < best_score:
            best_score = val
            best_seq = a.copy()

    # 4) Random log‑normal draws (quick exploration)
    while time.time() < deadline - 30.0:
        mu = rng.uniform(-1.0, 2.0)
        sigma = rng.uniform(0.4, 1.5)
        a = _lognormal_sequence(n, mu, sigma, target_sum, rng)
        val = _score(a, target_sum)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Log‑normal seed improved best={best_score:.6f}")

        # keep the loop light – only a few hundred samples
        if rng.random() < 0.01:
            break

    # 5) If a previous construction exists, use it as a seed
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # resize linearly
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        resized = _symmetrize(resized)
        val = _score(resized, target_sum)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()
            print(f"[{time.time():.2f}] Re‑used previous construction best={best_score:.6f}")

    print(f"[{time.time():.2f}] Initial best after warm‑start = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation with restarts (smooth to sharp)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    if time.time() < gd_deadline - 5.0:
        best_seq, best_score = _adam_restarts(best_seq, target_sum, gd_deadline)
        print(f"[{time.time():.2f}] After Adam restarts best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (DMT2) & random jitter
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        # DMT2
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             target_sum,
                                                             deadline)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] DMT2 improved best = {best_score:.6f}")

        # short random jitter + projection (helps escape plateaus)
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        a_jit = best_seq + jitter
        a_jit = np.maximum(a_jit, 0.0)
        a_jit = _project_to_simplex(a_jit, target_sum)
        a_jit = _symmetrize(a_jit)
        jit_val = _score(a_jit, target_sum)
        if jit_val < best_score:
            best_score = jit_val
            best_seq = a_jit.copy()
            print(f"[{time.time():.2f}] Random jitter improved best = {best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Global peak‑mass transfer (several fraction settings)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        # try a few different donor/receiver fractions
        for df, rf in [(0.02, 0.02), (0.04, 0.02), (0.02, 0.04), (0.05, 0.05)]:
            a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                          target_sum,
                                                          deadline,
                                                          donor_frac=df,
                                                          receiver_frac=rf)
            if val_peak < best_score:
                best_score = val_peak
                best_seq = a_peak.copy()
                print(f"[{time.time():.2f}] Global peak transfer (df={df},rf={rf}) improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Pairwise optimal transfer (tight neighbourhood)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      target_sum,
                                                      deadline,
                                                      donor_frac=0.02,
                                                      receiver_frac=0.02)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Final polishing with very high beta (almost exact max)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_optimize(best_seq,
                                         target_sum,
                                         deadline,
                                         beta_start=1e3,
                                         beta_max=5e6,
                                         step_factor=0.6,
                                         gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional evaluation (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
