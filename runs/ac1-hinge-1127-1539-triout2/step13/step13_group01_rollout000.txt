def smoothed_gradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)
    start = time.time()
    rng = np.random.default_rng()
    # hyperparameters
    lam_start = 0.01
    lam_end = 10.0
    max_iter = 1000000  # placeholder
    it = 0
    while time.time() < deadline:
        # compute convolution a*a
        conv = _conv_fft(a)
        # compute lambda schedule: there is iteration count, we can use fraction of remaining time
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        lam = lam_start * ((lam_end / lam_start) ** frac)  # exponential increase
        # compute softmax weights
        # to avoid overflow, subtract max of lam*conv
        exp_arg = lam * conv
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # compute gradient via convolution with w_rev
        w_rev = w[::-1]
        conv_rev = _conv_fft_general(a, w_rev)  # length 3n-2
        # extract gradient components
        idx = 2*n - 2 - np.arange(n)
        grad = 2.0 * conv_rev[idx]
        # step size: adapt
        # compute step magnitude s = target_sum * 0.01 maybe
        # Let's compute eta = step_factor * target_sum / (np.linalg.norm(grad) + 1e-12)
        step_factor = 0.5  # maybe tune
        eta = step_factor * target_sum / (np.linalg.norm(grad) + 1e-12)
        # update
        a = a - eta * grad
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

            jitter = rng.normal(scale=target_sum * 1e-4, size=n)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Gradient it={it:5d} score={best_val:.6f}")

def anneal_mass_transfer(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.8  # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))  # sqrt cooling

        # locate current peak
        k_max = int(np.argmax(b))

        # compute contributions for indices near k_max
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi+1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        # donor selection - weighted by contribution
        if contrib.sum() > 0:
            donor = rng.choice(idx, p=contrib/ contrib.sum())
        else:
            donor = rng.integers(lo, hi+1)

        # receiver selection - choose index with small a
        # compute probability inversely proportional to a+epsilon
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # enforce donor has some mass
        max_delta = min(a[donor] * 0.12, target_sum * 0.015)  # more aggressive
        if max_delta <= 0:
            continue
        # scale delta with temperature
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0:
            continue

        # Update convolution in O(n)
        b_new = b.copy()
        # linear part: remove from donor, add to receiver
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a

        # quadratic corrections
        b_new[2*donor] += delta * delta
        b_new[2*receiver] += delta * delta
        if donor != receiver:
            b_new[donor+receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:5d} best={best_val:.6f}")

        # occasional full recompute to avoid drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val

"""
Non‑negative step function optimizer.

The routine builds a low‑score sequence for the benchmark
`evaluate_sequence`.  It works in three stages:

1.  **Coarse search on a small problem** – we scan a family of power‑law
    shapes and generate a few random Dirichlet starts, then run a
    smoothed‑max gradient descent on a reduced length `n_small`.  The
    gradient of a soft‑max of the self‑convolution is computed via
    FFTs, and after each step the vector is projected onto the
    simplex {x ≥ 0, Σx = √(2·n_small)}.

2.  **Upscaling** – the best small‑scale solution is interpolated to the
    target length `n_large` and re‑projected onto the large simplex.

3.  **Fine‑grained annealing** – an aggressive simulated‑annealing
    mass‑transfer scheme (rank‑one convolution update in O(n)) refines
    the large‑scale vector.

The whole search respects the supplied time budget (default 1000 s) and
stores the final sequence in the global variable `height_sequence_1`
for warm‑starting future calls.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (FFT, power‑of‑2 size)."""
    n, m = a.size, b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²). No clipping."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

# Smoothed‑max gradient descent (small problem)
# ----------------------------------------------------------------------


def smoothed_gradient_descent(a0: np.ndarray, target_sum: float,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed approximation of max(a*a).

    The soft‑max temperature λ is increased from λ_start to λ_end
    (exponential schedule).  After each gradient step the iterate is
    projected onto the simplex to keep the ℓ₁‑norm fixed.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()

    # schedule parameters
    lam_start = 0.01
    lam_end = 10.0
    step_factor = 0.5          # ≈ step size ∝ target_sum / ‖grad‖

    it = 0
    while time.time() < deadline:
        # 1) self‑convolution
        conv = _conv_fft(a)

        # 2) temperature λ
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)  # in [0,1]
        lam = lam_start * ((lam_end / lam_start) ** frac)

        # 3) soft‑max weights
        exp_arg = lam * conv
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4) gradient via FFT (a * w_rev)
        w_rev = w[::-1]
        conv_rev = _conv_fft_full(a, w_rev)          # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)               # indices for gradient
        grad = 2.0 * conv_rev[idx]

        # 5) step & projection
        eta = step_factor * target_sum / (np.linalg.norm(grad) + 1e-12)
        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer (large problem)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves a tiny amount of mass from a “bad’’
    index (large contribution to the current peak) to a “good’’ index
    (small height).  The full convolution is maintained with an O(n)
    rank‑one update.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.8                    # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling (keeps the system hot longer)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of transferred mass, scaled by temperature
        max_delta = min(a[donor] * 0.12, target_sum * 0.015)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part (remove from donor, add to receiver)
        b_new[donor: donor + n] -= 2.0 * delta * a
        b_new[receiver: receiver + n] += 2.0 * delta * a
        # quadratic corrections
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance (based on current energy)
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:5d} best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 5000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the supplied time budget (seconds) and returns
    the best sequence found as a plain list of floats.
    """
    start = time.time()
    # small safety margin
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_large = 18000
    elif budget_s >= 500:
        n_large = 12000
    else:
        n_large = 8000

    target_sum_large = np.sqrt(2.0 * n_large)

    # Reduced problem for fast gradient descent
    n_small = max(2000, n_large // 4)
    target_sum_small = np.sqrt(2.0 * n_small)

    # --------------------------------------------------------------
    # Build a list of initial candidates for the small problem
    # --------------------------------------------------------------
    init_candidates = []

    # 1) warm‑start, if available
    if "height_sequence_1" in globals():
        warm_big = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm_small = _resize_sequence(warm_big, n_small)
        warm_small = _project_to_simplex(warm_small, target_sum_small)
        init_candidates.append(warm_small)

    # 2) scan a family of power‑law shapes (very cheap)
    best_pow_small = None
    best_pow_val = np.inf
    for p in np.arange(0.30, 1.01, 0.05):
        seq = (np.arange(1, n_small + 1, dtype=np.float64) ** (-p))
        seq = _project_to_simplex(seq, target_sum_small)
        val = _score(seq)
        if val < best_pow_val:
            best_pow_val = val
            best_pow_small = seq.copy()
    if best_pow_small is not None:
        init_candidates.append(best_pow_small)

    # 3) a few random Dirichlet starts
    rng = np.random.default_rng()
    for _ in range(2):
        rnd = rng.random(n_small)
        rnd = _project_to_simplex(rnd, target_sum_small)
        init_candidates.append(rnd)

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent on the reduced size
    # --------------------------------------------------------------
    small_deadline = start + 0.35 * (deadline - start)   # ~35 % of budget
    best_small_seq = None
    best_small_val = np.inf

    for idx, seed in enumerate(init_candidates):
        if time.time() >= small_deadline:
            break
        # allocate roughly equal time for the remaining seeds
        remaining = small_deadline - time.time()
        seeds_left = len(init_candidates) - idx
        sub_deadline = time.time() + max(0.1, remaining / seeds_left)

        print(f"[{time.time():.2f}] ---- small‑GD start #{idx+1} ----")
        a_opt, val_opt = smoothed_gradient_descent(seed,
                                                   target_sum_small,
                                                   sub_deadline)
        if val_opt < best_small_val:
            best_small_val = val_opt
            best_small_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New best small‑scale score={best_small_val:.6f}")

    # fallback if nothing was produced
    if best_small_seq is None:
        best_small_seq = _project_to_simplex(rng.random(n_small),
                                             target_sum_small)

    # --------------------------------------------------------------
    # Phase 2 – up‑scale to the target length
    # --------------------------------------------------------------
    a_large = _resize_sequence(best_small_seq, n_large)
    a_large = _project_to_simplex(a_large, target_sum_large)

    cur_score = _score(a_large)
    print(f"[{time.time():.2f}] Score after up‑scaling: {cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – aggressive annealing on the full size
    # --------------------------------------------------------------
    a_ann, ann_score = anneal_mass_transfer(a_large,
                                            target_sum_large,
                                            deadline)

    # pick the best of the two large‑scale candidates
    if ann_score < cur_score:
        final_seq = a_ann
        final_val = ann_score
    else:
        final_seq = a_large
        final_val = cur_score

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(final_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_list = arr.tolist()

    try:
        evaluated = evaluate_sequence(final_list)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {evaluated:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑starting the next call
    globals()["height_sequence_1"] = final_list
    return final_list
```

This implementation respects the time budget, explores several promising initialisations, employs a smoothed‑max gradient descent on a reduced problem (allowing many cheap iterations), up‑scales the result, and then refines it with a fast rank‑one update annealing scheme.  The combination of a more expressive gradient step and a longer, hotter annealing phase typically drives the benchmark score below the previous 1.507 → aiming for the target 1.503.
