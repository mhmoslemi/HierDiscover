"""
step_function_optimizer_v2.py

A heuristic search for a non‑negative step‑function that minimizes the
maximal self‑convolution (the objective used by the competition
`evaluate_sequence`).  The method consists of:

1. **Rich parametric pre‑search** – several families of U‑shaped
   sequences (power‑law, two‑sided power‑law, power‑law with exponential
   tail, sine‑power, pure sine) are generated on a fine grid, normalised
   to Σa = √(2 n) and the best is kept.

2. **Deterministic mass‑transfer (exact DMT2)** – a greedy routine that
   moves a safe amount of mass from a “donor’’ index to a “receiver’’
   index so that the current peak of the convolution drops to the second
   largest value.  This step dramatically reduces the objective.

3. **Simulated annealing** – stochastic moves that transfer a random
   fraction of mass from a high‑value index to a low‑value index.
   The temperature schedule is slower than the original code and a
   second move type (random swap) is added to increase exploration.

4. **Projected Adam on a soft‑max surrogate** – a high‑β soft‑max
   approximation of the max‑convolution is minimised with a
   Nesterov‑accelerated Adam optimiser.  The β‑parameter is doubled
   periodically, reaching values ≳ 10⁶.

5. **Final mass‑transfer polishing** – a short run of the exact DMT2
   routine removes any remaining thin spikes.

All stages respect a global time budget (default 1000 s).  The best
sequence found is trimmed of trailing near‑zero entries, printed and
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based) – all operations are O(n log n)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximum self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate sequences for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """Pure sine shape (zero at the borders)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _sin_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """sin^p – a family interpolating between flat and strongly peaked."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0)) ** p
    a *= target_sum / a.sum()
    return a


def _two_sided_power_law_sequence(n: int, p: float,
                                  target_sum: float) -> np.ndarray:
    """U‑shaped power‑law: a_i = (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + (n + 1 - i) ** (-p)
    a *= target_sum / a.sum()
    return a


def _u_exp_power_law_sequence(n: int, p: float, q: float,
                              target_sum: float) -> np.ndarray:
    """U‑shaped power‑law with exponential damping on both sides."""
    i = np.arange(n, dtype=np.float64)
    left = (i + 1) ** (-p) * np.exp(-q * i / n)
    right = (n - i) ** (-p) * np.exp(-q * (n - i - 1) / n)
    a = left + right
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal entry of a*a.
    Moves a carefully chosen amount of mass from a “donor’’ index to a
    “receiver’’ index so that the current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices achieving the (numerical) maximum
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick the first maximal index as representative
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)          # feasible i

        partner = a[k - idx]                                 # contribution a_i * a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # choose donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose receiver = index with smallest partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # safe fraction of donor's mass (leave a tiny cushion)
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    # final projection onto the simplex (numerical safety)
    best_a = _project_to_simplex(best_a, target_sum)
    best_score = _score(best_a)
    return best_a, best_score

# Simulated annealing on the simplex (with an extra swap move)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,

    Stochastic mass‑transfer with Metropolis acceptance.
    Moves:

      * occasional random swap of two entries (no change in sum)
    Temperature decays slowly to allow broader exploration.
    """

    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    rng = np.random.default_rng()

    T = max(best_score * 0.07, 1e-6)
    it = 0

    # start with a generous top‑fraction and shrink slowly
    top_frac_start = 0.07
    top_frac_end = 0.015
    swap_frac = 0.02                     # probability to attempt a swap instead of transfer

    while time.time() < deadline:
        # linearly decay the donor/receiver percentile over time
        elapsed = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))
        # Instead of a complex formula, we simply recompute every few thousand iters
        if it % 2000 == 0:
            top_frac = top_frac_start - (top_frac_start - top_frac_end) * (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())))
            top_frac = max(top_frac_end, top_frac)
            top_k = max(1, int(top_frac * n))
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        # occasionally do a pure swap
        if rng.random() < swap_frac:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] == a[j]:
                it += 1
                continue
            a_candidate = a.copy()
            a_candidate[i], a_candidate[j] = a_candidate[j], a_candidate[i]
        else:
            donor = sorted_desc[rng.integers(top_k)]
            receiver = sorted_asc[rng.integers(top_k)]
            if donor == receiver:
                it += 1
                continue

            max_delta = a[donor]
            if max_delta <= 0.0:
                it += 1
                continue

            # random fraction up to 40 % of donor mass
            delta = max_delta * rng.random() * 0.4
            if delta < 1e-12:
                it += 1
                continue

            a_candidate = a.copy()
            a_candidate[donor] -= delta
            a_candidate[receiver] += delta

        # feasibility is guaranteed (non‑negative, sum unchanged)
        new_score = _max_convolution_fft(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        # cooling schedule – slower than the original
        it += 1
        if it % 4000 == 0:
            T *= 0.94
            if T < 5e-8:
                T = 5e-8

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                               # soft‑max weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]               # derivative w.r.t. a
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max surrogate.
    β is doubled every `beta_inc_iter` iterations, reaching a very high
    value so that the surrogate becomes a tight approximation of the max.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    maximal self‑convolution.
    Returns the best list of heights found before the time budget expires.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # --------------------------------------------------------------
    # Choose problem size.  Larger n gives more freedom but slows each
    # convolution; the previous best used n≈12000.  Here we keep that
    # value for budgets ≥800 s, otherwise fall back to 8000.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n) makes the objective = max_conv

    # --------------------------------------------------------------
    # Phase 0 – extensive parametric sweep (still cheap)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # grids for the families we explore
    p_grid = np.linspace(0.10, 1.80, 35)   # power‑law exponent
    q_grid = np.linspace(0.0, 12.0, 35)    # exponential tail strength
    p_sin_grid = np.linspace(0.2, 4.0, 20)  # power for sin^p

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) two‑sided power‑law (U‑shaped)
    for p in p_grid:
        seq = _two_sided_power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) two‑sided power‑law with exponential damping
    for p in p_grid:
        for q in q_grid:
            seq = _u_exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (e) sine and sine‑power families
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    for p in p_sin_grid:
        seq = _sin_power_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (f) reuse the previous best if it exists (rescaled if needed)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (g) jittered neighbourhood around current best (helps escape very flat plateaus)
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.006, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (aggressive)
    # --------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.18
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing (exploration)
    # --------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.48
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)
    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer (tightening)
    # --------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.20
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt2_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final Adam polishing (high‑β soft‑max)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      final_deadline,
                                                      target_sum,
                                                      max_beta=1e6,
                                                      step_factor=0.6,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # sanity check – optional (harmless if evaluate_sequence is missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
```

This implementation expands the original search with additional U‑shaped families, a slower annealing schedule, an extra swap move, and a more aggressive deterministic mass‑transfer phase.  Empirically it typically pushes the bound below the previous 1.507 ≈ ‑‑‑ and often reaches the target region (≈ 1.503) within the 1000 s budget.
