def _lp_iterative_refine(a, target_sum, deadline, max_iter=10):
    n = a.shape[0]
    t = 0.5
    for it in range(max_iter):
        if time.time() > deadline: break
        conv = _conv_fft(a)
        M = conv.max()
        # solve LP
        g = _solve_lp_max_sum_under_conv_bound(a, M)
        sum_g = g.sum()
        if sum_g <= target_sum * (1+1e-12):
            # no improvement possible
            break
        g_scaled = g * (target_sum / sum_g)
        # mixing
        a = (1-t) * a + t * g_scaled
        # project onto simplex to correct rounding errors
        a = _project_to_simplex(a, target_sum)
        # optionally reduce t
        t *= 0.9
    return a

    """Return the smallest power of two ≥ x."""

    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y (both real) using FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float,
                       target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(
                f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}"
            )

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step,
                   target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(
                    f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}"
                )
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]      # first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose a receiver with the smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}"
            )

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Global random new best={best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty‑descent (lower a given maximal convolution bound)
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        p = conv - target_max
        p[p < 0] = 0.0

        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(
                a + rng.normal(scale=target_sum *
                               0.001, size=n), target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step,
                   target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            a = _project_to_simplex(
                a + rng.normal(scale=target_sum *
                               0.001, size=n), target_sum)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Linear‑programming based mass‑transfer (small problem)
# ----------------------------------------------------------------------


def _solve_lp_max_sum_under_conv_bound(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve:
        max   sum(g)
        s.t.  (a * g)[k] ≤ M   for all k = 0 … 2·n‑2
              g ≥ 0
    Returns the optimal g (may be very sparse).  Uses scipy.optimize.highs.
    """
    n = a.shape[0]
    # Build sparse matrix A where row k contains the coefficients a_i that
    # multiply g_{k-i}.  This is exactly a Toeplitz matrix built from a.
    diags = []
    offsets = []
    for shift in range(n):
        length = 2 * n - 1 - shift
        diags.append(a[:length])
        offsets.append(shift)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise sum → minimise -sum
    bounds = [(0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # fallback: return current a (won’t improve bound)
        return a.copy()
    return res.x


def _lp_iterative_refine(a: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 6) -> np.ndarray:
    """Iteratively improve a by the LP‑mass‑transfer scheme on a small scale."""
    n = a.shape[0]
    t = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        M = _max_convolution_fft(a)
        g = _solve_lp_max_sum_under_conv_bound(a, M)

        sum_g = g.sum()
        if sum_g <= target_sum * (1.0 + 1e-12):
            break

        # Scale to the proper simplex size, then mix
        g_scaled = g * (target_sum / sum_g)
        a = (1.0 - t) * a + t * g_scaled
        a = _project_to_simplex(a, target_sum)
        t *= 0.9                # slowly reduce mixing weight
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution (the evaluation score).  The routine consists of:

    1. Fast parametric pre‑search over a few families.
    2. LP‑based refinement on a down‑sampled problem.
    3. Adam‑style GD with restarts.
    4. Exact sub‑gradient descent.
    5. Deterministic mass‑transfer refinement (+ occasional random moves).
    6. Final high‑β Adam polishing.
    7. Penalty‑descent to push the bound a little further.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a bit larger when we have more time
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)   # Σa = √(2·n) → score = max‑convolution

    # ---------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ---------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # helper to evaluate and possibly replace the current best
    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_score
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(
                f"[{time.time():.2f}] Pre‑search improved best score = {best_score:.6f}"
            )

    # (a) pure power‑law sweep
    p_grid = np.linspace(0.25, 0.90, 31)
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        _maybe_update(seq)

    # (b) power‑law + exponential tail
    q_grid = np.linspace(0.0, 6.0, 31)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            _maybe_update(seq)

    # (c) simple sine shape
    _maybe_update(_sin_sequence(n, target_sum))

    # (d) Gaussian bump family
    for sigma in np.linspace(n / 20, n / 5, 9):
        _maybe_update(_gaussian_sequence(n, sigma, target_sum))

    # (e) previous best if it is available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        _maybe_update(resized)

    # (f) a few jittered perturbations around the current best
    rng = np.random.default_rng()
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        _maybe_update(cand)

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase LP – small‑scale LP refinement
    # ---------------------------------------------------------------
    # Down‑sample to a modest size where the LP is cheap
    n_small = min(2000, max(500, n // 6))
    x_big = np.linspace(0.0, 1.0, n)
    x_small = np.linspace(0.0, 1.0, n_small)
    a_small = np.interp(x_small, x_big, best_seq)
    a_small = _project_to_simplex(a_small, np.sqrt(2.0 * n_small))

    a_small_ref = _lp_iterative_refine(
        a_small,
        np.sqrt(2.0 * n_small),
        deadline - 5.0,
        max_iter=6,
    )

    # Upsample the refined small vector back to full size
    a_up = np.interp(x_big, x_small, a_small_ref)
    a_up = _project_to_simplex(a_up, target_sum)
    _maybe_update(a_up)

    # ---------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (soft‑max)
    # ---------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec,
            target_sum,
            gd_deadline,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time():.2f}] AdamGD restart improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (true max)
    # ---------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.25
    a_sg, val_sg = _subgradient_descent(
        best_seq,
        target_sum,
        subgrad_deadline,
    )
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(
            f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}"
        )

    # ---------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (core refinement)
    # ---------------------------------------------------------------
    while time.time() < deadline - 10.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq,
            deadline,
            target_sum,
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}"
            )

        a_rand, val_rand = _global_random_mass_transfer(
            best_seq,
            deadline,
            target_sum,
        )
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(
                f"[{time.time():.2f}] Global random improved best = {best_score:.6f}"
            )

        if time.time() > deadline - 8.0:
            break

    # ---------------------------------------------------------------
    # Phase D – Final polishing with high‑β Adam
    # ---------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(
                f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Phase E – Penalty descent to push the bound lower
    # ---------------------------------------------------------------
    if time.time() + 10.0 < deadline:
        # aim a small improvement below the current best
        target = best_score - 0.0045
        a_pen, val_pen = _penalty_lower_max(
            best_seq,
            target_sum,
            deadline,
            target,
        )
        if val_pen < best_score:
            best_score = val_pen
            best_seq = a_pen.copy()
            print(
                f"[{time.time():.2f}] Penalty descent improved best = {best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # ---------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final safety evaluation (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
