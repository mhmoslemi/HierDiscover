Advanced step‑function optimiser (v2).

The optimiser follows a three‑phase strategy:

1. **Exploration** – many short runs of soft‑max gradient descent
   and cheap sub‑gradient descent on a moderate length (≈4000‑6000).
   A diverse seed pool (power‑law, exponential, random Dirichlet,

2. **Upscaling** – the current best shape is interpolated to a larger

3. **Intensive polishing** – a long high‑β gradient descent,

   The last few seconds are spent on a final sub‑gradient nudge

The routine respects the 1000 s budget and never returns ``np.inf``.

# FFT utilities (convolution, max‑convolution)

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring (objective) – with normalisation Σa = √(2·n) the score = max(conv)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the evaluation score for a (assumes a already non‑negative)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    # the normalisation Σa = √(2n) is ensured by the caller
    return float(max_b)


# ----------------------------------------------------------------------
# Families for seeding
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_dirichlet(n: int, target_sum: float) -> np.ndarray:
    rng = np.random.default_rng()
    a = rng.random(n) + 1e-6
    a = np.log(a)  # take log to get a spread similar to Dirichlet
    a = np.exp(a)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize that conserves total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest left, second largest right, alternating."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    le = True
    for v in sorted_vals:
        if le:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        le = not le
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max probabilities
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a_rev * p) restricted to centre n entries
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size)
    p_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Annealed soft‑max gradient descent (momentum)
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 2e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected momentum GD on the soft‑max approximation of the peak."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (cheap)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(convolution)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of the active convolution entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        # Note: the formula already accounts for the factor 2 (see analysis)

        # step size schedule
        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # jitter every few thousand steps
        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} best={best_val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (local SA)
# ----------------------------------------------------------------------


def _mass_transfer(a0: np.ndarray,
                  deadline: float,
                  target_sum: float,
                  start_temp: float = 0.5,
                  end_temp: float = 1e-5,
                  low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Small‑move SA that shifts mass from high‑contrib to low‑contrib indices."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        w = a[idx] + a[kmax - idx]   # contribution weight for each index in the active band

        donor = idx[np.argmax(w)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            # no move possible
            continue

        # propose a mass shift
        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT move new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to keep dynamics alive
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the three‑phase strategy described in the module docstring.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – exploration on a moderate size
    # ------------------------------------------------------------------
    # Choose a length that balances speed and expressiveness.
    if budget_s >= 800:
        n_expl = 6000
    elif budget_s >= 500:
        n_expl = 5000
    else:
        n_expl = 4000
    target_sum = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # Build a diverse pool of seeds
    init_pool = []

    for p in np.linspace(0.35, 0.75, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_sum))
    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_sum))
    for _ in range(6):
        a = rng.random(n_expl)
        a *= target_sum / a.sum()
        init_pool.append(a)
    # linear decreasing
    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_sum / lin.sum()
    init_pool.append(lin)

    # reuse previous global if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies to enlarge diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_sum * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # allocate ~70 % of budget to exploration
    exploration_deadline = deadline - 70.0

    while time.time() < exploration_deadline:
        # Choose seed: sometimes perturb the current best
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_sum * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_sum)
        else:
            seed = rng.choice(init_pool)

        # Randomly pick a optimiser
        if rng.random() < 0.6:
            # short soft‑max GD (≈ 12‑14 s)
            run_dead = min(time.time() + 13.0, exploration_deadline)
            a_opt, val_opt = _annealed_gradient_descent(
                seed, run_dead, target_sum,
                max_beta=5e5,
                step_factor=0.7,
                gamma=0.99,
            )
        else:
            # short sub‑gradient run (≈ 10 s)
            run_dead = min(time.time() + 10.0, exploration_deadline)
            a_opt, val_opt = _subgradient_descent(
                seed, run_dead, target_sum,
                max_iter=200_000,
            )

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

        # break if time is almost up
        if time.time() >= exploration_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – upscaling to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_sum)

    remaining = deadline - time.time()
    # decide final length based on remaining budget
    if remaining > 200 and budget_s >= 800:
        n_final = 15000
    elif remaining > 120:
        n_final = 12000
    else:
        n_final = max(8000, n_expl)

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # examine a few simple permutations
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – intensive polishing on the large instance
    # ------------------------------------------------------------------
    # 1) Long high‑β gradient descent
    if time.time() < deadline - 50.0:
        gd_dead = deadline - 30.0
        a_polish, val_polish = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=2e7,
            step_factor=0.45,
            gamma=0.995,
        )
        if val_polish < best_score:
            best_score = val_polish
            best_seq = a_polish.copy()
            print(f"[{time.time():.2f}] GD polish best={best_score:.6f}")

    # 2) Deterministic mass‑transfer local SA (medium temperature)
    if time.time() < deadline - 30.0:
        mt_dead = deadline - 20.0
        a_mt, val_mt = _mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.4,
            end_temp=5e-5,
            low_frac=0.04,
        )
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] MT after GD best={best_score:.6f}")

    # 3) Very short sub‑gradient nudge
    if time.time() < deadline - 10.0:
        sub_dead = time.time() + min(4.0, deadline - time.time() - 1.0)
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=150_000,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Final sub‑grad best={best_score:.6f}")

    # 4) Final tiny mass‑transfer polish
    if time.time() < deadline - 2.0:
        mt2_dead = time.time() + min(3.0, deadline - time.time() - 0.5)
        a_mt2, val_mt2 = _mass_transfer(
            best_seq,
            mt2_dead,
            target_final,
            start_temp=0.2,
            end_temp=1e-5,
            low_frac=0.03,
        )
        if val_mt2 < best_score:
            best_score = val_mt2
            best_seq = a_mt2.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros (compactness) and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Log final evaluation (optional, ignored by the harness)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
