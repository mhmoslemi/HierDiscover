"""
Hybrid optimizer for the step‑function “max‑convolution” problem.

Approach
--------
1.  Choose a problem size n (bigger n gives a tighter theoretical bound 1 ≈ max b).
2.  Generate a diverse pool of feasible seeds (power‑law, exponential,
    uniformly random on the ℓ¹‑simplex, and the previous best solution if any)
    and pick the best seed (lowest max‑convolution).
3.  Refine this seed by minimizing a smooth approximation of the max‑convolution:
    the Log‑Sum‑Exp (LSE) surrogate.
    –   A softmax over the convolution values furnishes weights wₖ.
    –   ∇ LSE = 2·conv(w, a_rev) (a_rev = a reversed).  The gradient is
        evaluated with FFTs in O(n log n).
    –   Adam is used for step‑size adaptation; after each step the vector is
        projected onto the simplex {aᵢ≥0, Σaᵢ = √(2 n)}.
    –   The inverse temperature β starts small (smooth landscape) and is
        increased exponentially until a large value (≈ max).
4.  When the time budget is almost exhausted, run the deterministic
    mass‑transfer routine and a short sub‑gradient descent (the same routines
    that proved useful in the previous state‑of‑the‑art implementation).
5.  Test three simple reorderings (original, alternating, even‑spread) and keep
    the ordering with the smallest max‑convolution.
6.  Trim trailing near‑zero entries, store the result for the next run, and
    return the final sequence.

The whole procedure respects a 1000 s budget and works for any size of n that
fits comfortably in memory (the default is n = 20 000 when the full budget is
available).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core FFT convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i≥0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (all generate vectors with Σa = target_sum)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple reorderings that sometimes lower the peak
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Short sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient descent on the Log‑Sum‑Exp surrogate (smooth max)
# ----------------------------------------------------------------------


def _lse_gradient_step(a: np.ndarray,
                       beta: float,
                       target_sum: float,
                       m: np.ndarray,
                       v: np.ndarray,
                       it: int,
                       lr_initial: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam step minimizing the LSE surrogate of the max‑convolution."""
    n = a.shape[0]

    # 1) compute convolution a*a
    b = _conv_fft(a)                     # length 2n‑1
    maxb = b.max()

    # 2) soft‑max weights over the convolution entries
    w = np.exp(beta * (b - maxb))        # shift by max for stability
    w_sum = w.sum()
    if w_sum == 0.0:                     # numerical safeguard
        w = np.ones_like(b) / b.size
    else:
        w /= w_sum

    # 3) gradient of the LSE: ∇ = 2·conv(w, a_rev)
    a_rev = a[::-1]
    conv_len_needed = 3 * n - 2          # length of full linear conv of w (2n‑1) and a_rev (n)
    N = 1 << ((conv_len_needed - 1).bit_length())
    fw = np.fft.rfft(w, n=N)
    fa_rev = np.fft.rfft(a_rev, n=N)
    full = np.fft.irfft(fw * fa_rev, n=N)[:conv_len_needed]
    grad = 2.0 * full[n - 1:n - 1 + n]

    # 4) Adam update on the simplex
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)

    m_hat = m / (1 - beta1 ** (it + 1))
    v_hat = v / (1 - beta2 ** (it + 1))

    # learning‑rate schedule: decay linearly to a tiny floor
    lr = lr_initial * (1.0 - it / 1e6) + 0.001
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new, target_sum)

    # return updated variables (the calling code will recompute the max‑conv)
    return a_new, m, v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of heights) that minimises
    evaluate_sequence().  The routine builds diverse seeds, runs a smooth‑max
    gradient descent (Adam + simplex projection), then refines the result with
    deterministic mass transfer and a short sub‑gradient descent, finally
    tests a few reorderings and trims near‑zero tail entries.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose sequence length (bigger = tighter theoretical lower bound)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a small pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # previous solution, resized if needed
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            seeds.append(_resize_sequence(prev, n))
        else:
            seeds.append(prev.copy())

    # power‑law family
    for p in [0.2, 0.5, 0.8, 1.0]:
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in [0.001, 0.003, 0.01]:
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random points on the simplex
    for _ in range(6):
        seeds.append(_random_simplex(n, target_sum, rng))

    # evaluate seeds (objective = max‑convolution, because sum = √(2n))
    best_seq = None
    best_obj = np.inf
    for s in seeds:
        obj = _max_convolution_fft(s)
        if obj < best_obj:
            best_obj = obj
            best_seq = s.copy()

    # ------------------------------------------------------------------
    # Gradient‑descent on smooth max (LSE) – main workhorse
    # ------------------------------------------------------------------
    a = _project_to_simplex(best_seq, target_sum)   # ensure feasibility
    m = np.zeros_like(a)      # Adam 1st moment
    v = np.zeros_like(a)      # Adam 2nd moment
    lr_initial = 1.0          # base learning rate for Adam
    beta_start = 0.5
    beta_end = 150.0          # final inverse temperature (very sharp)

    it = 0
    # allocate ~80 % of the remaining budget to this phase
    grad_deadline = deadline - 15.0
    while time.time() < grad_deadline:
        elapsed = time.time() - start_time
        total = grad_deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        beta = beta_start * (beta_end / beta_start) ** frac

        a, m, v = _lse_gradient_step(a, beta, target_sum, m, v, it, lr_initial)

        # keep the best ever seen
        cur_obj = _max_convolution_fft(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()

        # occasional tiny jitter to escape plateaus
        if it % 120 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    print(f"[{time.time():.2f}] After smooth‑max descent best max‑conv = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        mt_dead = time.time() + min(8.0, deadline - time.time() - 5.0)
        seq_mt, obj_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                      target_sum,
                                                      start_temp=0.4,
                                                      end_temp=1e-5,
                                                      low_frac=0.07)
        if obj_mt < best_obj:
            best_obj, best_seq = obj_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After mass‑transfer best = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Short sub‑gradient descent (exact max) – final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        sg_dead = time.time() + min(5.0, deadline - time.time() - 2.0)
        seq_sg, obj_sg = _subgradient_descent(best_seq, sg_dead,
                                              target_sum,
                                              step_factor=0.45,
                                              gamma=0.995)
        if obj_sg < best_obj:
            best_obj, best_seq = obj_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Simple reorderings (often lower the peak)
    # ------------------------------------------------------------------
    candidates = [best_seq,
                  _alternating_order(best_seq),
                  _even_spread_order(best_seq)]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_obj = _max_convolution_fft(best_seq)
    print(f"[{time.time():.2f}] After reordering best max‑conv = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store for the next run and return
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
