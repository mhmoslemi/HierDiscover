    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)

    init_candidates = []

    # warm start
    if "height_sequence_1" in globals():
        warm_arr = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm_arr = _resize_sequence(warm_arr, n)
        warm_arr = _project_to_simplex(warm_arr, target_sum)
        init_candidates.append(warm_arr)

    rng = np.random.default_rng()
    for _ in range(5):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    # Add some deterministic power law seeds
    for p in [0.45, 0.55, 0.65]:
        seq = np.arange(1, n+1, dtype=np.float64) ** (-p)
        seq = seq * (target_sum / seq.sum())
        init_candidates.append(seq)

    best_seq = None
    best_val = np.inf

    # iterate over each start
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] ----- start #{idx+1} -----")
        remaining = deadline - time.time()
        # allocate part of remaining time proportionally - e.g., each start gets min(remaining/ (len(init_candidates)-idx), some cap)
        per_deadline = time.time() + remaining / (len(init_candidates) - idx)

def smoothmax_gradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2.0 * n * _conv_fft(a).max() / (a.sum()**2)
    rng = np.random.default_rng()
    start = time.time()
    step0 = 1.0
    min_step = 0.01
    beta0 = 0.01
    beta_end = 30.0
    it = 0
    while time.time() < deadline:
        # compute convolution
        c = _conv_fft(a)
        max_c = c.max()
        cur_val = 2.0 * n * max_c / (a.sum()**2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Grad iter {it} new best {best_val:.6f}")

        # schedule
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 0.0
        beta = beta0 + (beta_end - beta0) * (frac**2)

        # weights
        shifted = c - max_c
        # compute exponent scaled by beta; avoid overflow
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient via convolution
        rev_w = w[::-1]
        conv_rev = conv2(a, rev_w)
        L = len(w)
        idx = L - 1 - np.arange(n)
        grad = 2.0 * conv_rev[idx]

        # normalize gradient direction
        norm = np.linalg.norm(grad) + 1e-12
        direction = grad / norm

        # step size schedule
        eta = step0 * (1.0 - frac) + min_step

        # add jitter every 1000 iterations
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)

        # update
        a = a - eta * direction
        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val

"""
Improved search for a low‑score non‑negative step function.

The optimiser works in three stages:

1.  Multiple warm‑starts (previous best, random Dirichlet draws,
    a few deterministic power‑law seeds).
2.  Smooth‑max gradient descent:
    –  The true peak max(a*a) is approximated by a soft‑max
       (log‑sum‑exp) with a temperature β that is increased
       during the run.
    –  The gradient of the smoothed objective is obtained with a single
       O(n log n) convolution (FFT).  After the step the vector is
       projected onto the simplex {x≥0, Σx = √(2n)}.
    –  Early iterations use a small β (almost uniform weights) giving a
       global picture, later iterations use a large β (behaving like a
       sub‑gradient) to polish the solution.
3.  A final short simulated‑annealing “mass‑transfer” refinement
    (the same routine as in the reference implementation).

The routine respects the overall time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution helpers
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv2_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y (both 1‑D real arrays) via FFT."""
    n1 = x.size
    n2 = y.size
    size = 1 << ((n1 + n2 - 2).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: n1 + n2 - 1]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def _smoothmax_gradient_descent(a0: np.ndarray, target_sum: float,
                               deadline: float):
    """
    Gradient descent on the log‑sum‑exp surrogate of max(a*a).
    After each step the iterate is projected onto the simplex to keep
    Σa = target_sum.
    Returns the best vector seen and its (true) objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # initial best
    c = _conv_fft(a)
    best_val = 2.0 * n * c.max() / (a.sum() ** 2)
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    # hyper‑parameters (tuned empirically)
    step_max = 1.0      # initial step magnitude
    step_min = 0.01     # final step magnitude
    beta_start = 0.01
    beta_end = 30.0

    while time.time() < deadline:
        # compute convolution and true objective
        c = _conv_fft(a)
        max_c = c.max()
        cur_val = 2.0 * n * max_c / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} new best={best_val:.6f}")

        # schedule for β (smoothness) – quadratic increase gives a slow start,
        # fast finish.
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 0.0
        beta = beta_start + (beta_end - beta_start) * (frac ** 2)

        # soft‑max weights (numerically stable)
        shifted = c - max_c          # max_c becomes 0 after shift
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective
        rev_w = w[::-1]                     # length L = 2n‑1
        conv_rev = _conv2_fft(a, rev_w)     # length 3n‑2
        L = w.size
        idx = L - 1 - np.arange(n)          # slice that yields the gradient
        grad = 2.0 * conv_rev[idx]

        # normalized direction
        norm = np.linalg.norm(grad) + 1e-12
        direction = grad / norm

        # step size (linear decay)
        eta = step_max * (1.0 - frac) + step_min

        # occasional jitter to shake out of plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)

        # gradient step + projection onto simplex
        a = a - eta * direction
        a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer (same as reference)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from a “bad’’ index
    (large contribution to the current peak) to a “good’’ index (small height).
    The convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact self‑convolution
    cur_val = 2.0 * n * b.max() / (target_sum ** 2)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                    # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer (scaled by temperature)
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = 2.0 * n * b.max() / (target_sum ** 2)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)               # normalisation Σa = √(2·n)

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + random)
    # --------------------------------------------------------------
    init_candidates = []

    # warm start from previous run, if any
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    # a handful of random Dirichlet‑like starts
    for _ in range(5):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    # deterministic power‑law seeds (different exponents)
    for p in (0.45, 0.55, 0.65):
        seq = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        seq = seq * (target_sum / seq.sum())
        init_candidates.append(seq)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run smooth‑max gradient descent for every start
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx + 1} -----")

        # allocate a fair share of the remaining budget to this start
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        per_deadline = time.time() + remaining / (len(init_candidates) - idx)

        seq, val = _smoothmax_gradient_descent(init, target_sum, per_deadline)

        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Global best after GD: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final short annealing refinement if time permits
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.2f}] Final annealing refinement")
        seq_a, val_a = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_a < best_score:
            best_score = val_a
            best_seq = seq_a.copy()
            print(f"[{time.time():.2f}] Global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
