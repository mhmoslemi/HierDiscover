"""
Advanced step‑function optimiser – LP‑guided refinement.

The optimiser works in three phases:

1. **Seeding & LP refinement** – a collection of simple
   power‑law / exponential seeds is taken.  For each seed we compute the
   current peak `M = max(a*a)`.  Then we solve a linear program

        maximise   Σ g_j
        subject to  (a * g)[k] ≤ M   for all k
                    g_j ≥ 0

   The LP finds a distribution `g` that can be added without increasing
   the current peak.  After scaling `g` to the required ℓ¹‑norm we mix it
   with the current sequence (`a ← (1‑t) a + t g`).  Repeating this a few
   times strictly pushes the peak down.

2. **Upscaling & polishing** – the best sequence from phase‑1 is
   linearly interpolated to a larger length (≈ 8000‑10000) and a short
   soft‑max gradient‑descent / momentum run finishes the optimisation.

3. **Final clean‑up** – optional tiny deterministic mass‑transfer
   passes and a trailing‑zero trim.  The resulting list is stored in the
   global variable ``height_sequence_1`` and returned.

The whole routine respects the 1000 s time budget, falls back to the
previous optimiser if SciPy is unavailable, and never returns ``np.inf``.
"""

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Return the benchmark value 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, projected onto the simplex of sum `target_sum`."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = a * (target_sum / a.sum())
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), projected onto the simplex of sum `target_sum`."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a = a * (target_sum / a.sum())
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize that preserves total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest value left‑most, second largest right‑most, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smooth (soft‑max) approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum‑based GD on the soft‑max approximation of the peak."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

# Linear‑programming based refinement (core of the new contribution)

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the dense matrix A such that A @ g = f * g (linear convolution).
    Shape of A is (2·n‑1, n).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # each column j receives a shifted copy of f
    for j in range(n):
        A[j:j + n, j] = f   # f[0] aligns with row j, etc.
    return A


def _lp_refine(f: np.ndarray,
               target_sum: float,
               mix_factor: float,
               deadline: float,
               max_iters: int = 12) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑guided refinement steps.
    Returns the improved sequence and its score.
    """
    if linprog is None:
        # SciPy missing – abort early, return original.
        return f, _score(f)

    n = f.shape[0]
    best_f = f.copy()
    best_val = _score(f)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current peak
        M = _max_convolution_fft(f)

        # build convolution matrix for the constraints
        A = _build_conv_matrix(f)

        # linear program: maximise Σ g_j   ⇔ minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, (deadline - time.time()))})
        if not res.success:
            # if LP fails we stop this refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # scale so that Σ g = target_sum
        g = g * (target_sum / sum_g)

        # convex mixing
        f = (1.0 - mix_factor) * f + mix_factor * g
        f = _project_to_simplex(f, target_sum)

        cur_val = _score(f)
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"[{time.time():.2f}] LP refine it={it:2d} score={best_val:.6f}")

        it += 1

    return best_f, best_val

    deadline = start_time + budget_s - 0.5  # keep a tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose a modest size for the LP phase (fast linear solves)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 2500
    elif budget_s >= 500:
        n_expl = 2000
    else:
        n_expl = 1500
    target_expl = math.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # 2. Build a small pool of diverse seeds
    # --------------------------------------------------------------
    seed_pool = []

    # power‑law seeds
    for p in np.linspace(0.35, 0.80, 7):
        seed = _power_law_sequence(n_expl, float(p), target_expl)
        seed = _alternating_order(seed)
        seed_pool.append(seed)

    # exponential seeds
    for lam in np.linspace(0.001, 0.020, 5):
        seed = _exponential_sequence(n_expl, float(lam), target_expl)
        seed = _alternating_order(seed)
        seed_pool.append(seed)

    # a random Dirichlet‑like seed
    for _ in range(3):
        a = rng.random(n_expl)
        a = a * (target_expl / a.sum())
        seed_pool.append(_alternating_order(a))

    # reuse the previously discovered best sequence if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        resized = _resize_sequence(prev, n_expl)
        seed_pool.append(_alternating_order(resized))

    # --------------------------------------------------------------
    # 3. LP‑guided refinement on each seed, keep the best result
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # budget for the LP phase – leave ~120 s for polishing & final work
    lp_deadline = deadline - 120.0
    mix_factor = 0.25  # how much of the LP‑solution we mix in each step

    for seed in seed_pool:
        if time.time() >= lp_deadline:
            break

        # make a fresh copy and ensure exact simplex normalisation
        a = _project_to_simplex(seed.copy(), target_expl)

        # perform a short LP refinement (few iterations)
        a_refined, val = _lp_refine(a, target_expl, mix_factor, lp_deadline, max_iters=8)

        if val < best_score:
            best_score = val
            best_seq = a_refined.copy()
            print(f"[{time.time():.2f}] New best after LP seed score={best_score:.6f}")

    # fall back to a power‑law if everything failed (unlikely)
    if best_seq is None:
        best_seq = _alternating_order(_power_law_sequence(n_expl, 0.55, target_expl))

    # --------------------------------------------------------------
    # 4. Upscale to a larger length for the final polishing step
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0 and budget_s >= 800:
        n_final = 10000
    elif remaining > 40.0:
        n_final = 8000
    else:
        n_final = n_expl  # stay at current size if time is short

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _alternating_order(seq_big)
    seq_big = _project_to_simplex(seq_big, target_final)

    # --------------------------------------------------------------
    # 5. Short high‑β soft‑max gradient polishing (uses most of the rest)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_polish, val_polish = _annealed_gradient_descent(
            seq_big,
            deadline - 3.0,
            target_final,
            max_beta=2e6,
            step_factor=0.45,
            gamma=0.995,
        )
        if val_polish < best_score:
            best_score = val_polish
            seq_big = a_polish.copy()
            print(f"[{time.time():.2f}] Polished final score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final tiny mass‑transfer pass (helps tidy up lingering peaks)
    # --------------------------------------------------------------
    # This step is deliberately lightweight – it merely nudges a few
    # entries while staying well within the time budget.
    if time.time() < deadline - 0.5:
        # simple deterministic transfer: move a tiny epsilon from the
        # largest entry to the smallest (project back each time)
        a = seq_big.copy()
        eps = target_final * 1e-5
        for _ in range(200):
            i_max = int(np.argmax(a))
            i_min = int(np.argmin(a))
            if i_max == i_min:
                break
            delta = min(eps, 0.5 * a[i_max])
            a[i_max] -= delta
            a[i_min] += delta
            a = _project_to_simplex(a, target_final)
        # update score if improvement
        final_val = _score(a)
        if final_val < best_score:
            best_score = final_val
            seq_big = a.copy()
            print(f"[{time.time():.2f}] Final mass‑transfer score={best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Trim trailing zeros for compactness
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # 8. Log the official evaluation result (optional)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
