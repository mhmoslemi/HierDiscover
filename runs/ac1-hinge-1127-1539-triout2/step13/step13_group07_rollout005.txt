def _lp_max_sum(f, M, solver='highs'):

    n = len(f)

    # Build A matrix as sparse csr.
    from scipy.sparse import csr_matrix

    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)

    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    # Solve
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return original f
        return np.copy(f)
    b_opt = res.x
    return b_opt

def _lp_max_sum(f, M):
    import numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = f.shape[0]
    # Build data arrays
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    idx = np.arange(n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos+n] = i + idx
        cols[pos:pos+n] = i
        data[pos:pos+n] = f
        pos += n
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    # Solve LP: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback
        print("LP solve failed:", res.message)
        return f.copy()
    b_opt = res.x
    # Ensure non-negative due to numeric tolerance
    b_opt = np.maximum(b_opt, 0.0)
    return b_opt

"""
Iterative LP‑based refinement for the max‑self‑convolution problem.

The search proceeds in two resolution levels:

1. **Low‑resolution LP phase (n≈2000).**
   – Start from a down‑sampled version of the best known construction (or a
     simple power‑law).
   – Repeatedly:
        * compute M = max(f * f)
        * solve a linear program  max∑b_i  s.t. f * b ≤ M (component‑wise) and b≥0
        * scale b to the required ℓ₁‑norm and move the current f towards b.
   – This is the exact analogue of the “LP‑step” described in the reference
     algorithm and quickly reduces the objective.

2. **High‑resolution polishing.**
   – Interpolate the low‑resolution result to a larger size (≈12000) and
     re‑project onto the simplex Σa_i=√(2n).
   – Apply a few passes of deterministic mass‑transfer, simulated annealing
     and a short Adam‑style projected gradient descent (soft‑max surrogate)
     to eliminate remaining peaks.

The whole routine respects a 1000 s budget and returns the best sequence
found.  All numbers are non‑negative; the final list is trimmed of trailing
near‑zeros.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used when no previous construction is available)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ i^{‑p}  (i starts at 1) and then ℓ₁‑projected to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement, same as the previous code)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index to a
    “receiver’’ index so that the current peak drops to the second‑largest
    value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer with Metropolis acceptance.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # Initial temperature: a small fraction of the current objective
    T = max(best_score * 0.05, 1e-8)
    it = 0
    top_frac = 0.03                      # top / bottom 3 %
    top_k = max(1, int(top_frac * n))

    # pre‑compute sorted orders periodically
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 1500 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.35
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.88
            if T < 1e-9:
                T = 1e-9

    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient (used for the final Adam polish)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 280

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ℓ₁‑norm of b under convolution bound)
# ----------------------------------------------------------------------


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve    maximise   Σ b_i
    subject to f * b ≤ M   (component‑wise),   b_i ≥ 0.
    Returns the optimal b (as a dense numpy array).
    """
    n = f.shape[0]

    # Build sparse convolution matrix A where (A @ b)[k] = Σ_i f_i b_{k-i}
    # Row k = i + j, column i carries f_j.
    idx = np.arange(n, dtype=np.int32)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = i + idx
        cols[pos:pos + n] = i
        data[pos:pos + n] = f
        pos += n
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear program: minimise -Σ b_i  s.t. A b ≤ M, b ≥ 0
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as exc:          # fallback: return the current f
        print("Linprog exception:", exc)
        return f.copy()

    if not res.success:
        # If LP fails, simply return f – it is a feasible point.
        print("LP failed:", res.message)
        return f.copy()

    b_opt = np.maximum(res.x, 0.0)
    return b_opt

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal
    self‑convolution.  The routine respects the time budget and returns the
    best list of heights found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Low‑resolution LP phase
    # ------------------------------------------------------------------
    n_small = 2000                     # fits comfortably into the LP solver
    target_small = float(np.sqrt(2.0 * n_small))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        xs_old = np.linspace(0.0, 1.0, prev.size)
        xs_new = np.linspace(0.0, 1.0, n_small)
        seq_small = np.interp(xs_new, xs_old, prev)
        seq_small = _project_to_simplex(seq_small, target_small)
    else:
        # a modest power‑law works as a generic starter
        seq_small = _power_law_sequence(n_small, p=0.7, target_sum=target_small)

    best_small = seq_small.copy()
    best_score_small = _score(best_small)
    print(f"[{time.time():.2f}] Initial low‑res score = {best_score_small:.6f}")

    # allocate ~30 % of the total budget to the LP phase
    lp_budget = (deadline - time.time()) * 0.30
    lp_deadline = time.time() + lp_budget

    t = 0.5                # step size towards the LP solution
    it_lp = 0
    while time.time() < lp_deadline:
        M = _max_convolution_fft(best_small)
        b_opt = _lp_max_sum(best_small, M)
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break

        # scale to required ℓ₁‑norm
        g = b_opt * (target_small / sum_b)

        # convex combination (move a fraction t towards g)
        new_seq = (1.0 - t) * best_small + t * g
        new_seq = _project_to_simplex(new_seq, target_small)

        new_score = _score(new_seq)
        if new_score < best_score_small - 1e-12:
            best_score_small = new_score
            best_small = new_seq.copy()
            print(f"[{time.time():.2f}] LP it={it_lp:3d} score={best_score_small:.6f}")

        # gradually shrink the step size
        t = max(t * 0.94, 0.03)
        it_lp += 1

    print(f"[{time.time():.2f}] After LP phase best score = {best_score_small:.6f}")

    # ------------------------------------------------------------------
    # 2) Upsample to a large resolution
    # ------------------------------------------------------------------
    # decide how large the final sequence can be (keep a safety margin)
    remaining = deadline - time.time()
    if remaining < 200:
        # not enough time for a huge final size – fall back to 8000
        n_large = 8000
    else:
        n_large = 12000

    target_large = float(np.sqrt(2.0 * n_large))

    xs_small = np.linspace(0.0, 1.0, n_small)
    xs_large = np.linspace(0.0, 1.0, n_large)
    seq_large = np.interp(xs_large, xs_small, best_small)
    seq_large = _project_to_simplex(seq_large, target_large)

    best_large = seq_large.copy()
    best_score_large = _score(best_large)
    print(f"[{time.time():.2f}] Upsampled size {n_large} – score = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # 3) Local refinement on the large scale
    # ------------------------------------------------------------------
    # Split the remaining budget: deterministic → SA → deterministic → Adam
    remaining = deadline - time.time()
    if remaining <= 0:
        final_seq = best_large
        # trimming
        arr = np.array(final_seq, dtype=float)
        nz = np.where(arr > 1e-12)[0]
        if nz.size:
            arr = arr[: nz[-1] + 1]
        return arr.tolist()

    # 3a) First deterministic mass‑transfer (tighten the peak)
    dmt1_deadline = time.time() + remaining * 0.18
    best_large, best_score_large = _deterministic_mass_transfer_exact2(
        best_large, dmt1_deadline, target_large)

    print(f"[{time.time():.2f}] After first DMT score = {best_score_large:.6f}")

    # 3b) Simulated annealing (escape possible local minima)
    remaining = deadline - time.time()
    if remaining > 0:
        sa_deadline = time.time() + remaining * 0.55
        best_large, best_score_large = _simulated_annealing(
            best_large, target_large, sa_deadline)
        print(f"[{time.time():.2f}] After SA score = {best_score_large:.6f}")

    # 3c) Second deterministic mass‑transfer (polish)
    remaining = deadline - time.time()
    if remaining > 0:
        dmt2_deadline = time.time() + remaining * 0.18
        best_large, best_score_large = _deterministic_mass_transfer_exact2(
            best_large, dmt2_deadline, target_large)
        print(f"[{time.time():.2f}] After second DMT score = {best_score_large:.6f}")

    # 3d) Final Adam‑style projected gradient descent (soft‑max surrogate)
    remaining = deadline - time.time()
    if remaining > 2.0:
        final_deadline = deadline
        best_large, best_score_large = _adam_gradient_descent(
            best_large, final_deadline, target_large,
            max_beta=1e6, step_factor=0.6, gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish score = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # 4) Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_large, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the result for potential reuse
    globals()["height_sequence_2"] = final_seq
    return final_seq
