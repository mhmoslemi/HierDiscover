"""
step_optimize.py

A fast, multi‑phase optimizer for the max‑self‑convolution objective.
The pipeline:

1. **Problem size** – choose a length `n` as large as the time budget
   permits (12000 for a full 1000 s run).

2. **Rich parametric sweep** – evaluate many analytically‑parameterised
   families (power‑law, power‑law + exp‑tail, symmetric product power,
   sum‑of‑two‑power‑laws and a discrete arcsine shape).  The sweep is
   refined around promising parameters.

3. **Random Dirichlet seed** – a few random points on the simplex are
   added to diversify the search.

4. **Deterministic mass‑transfer (Exact DMT2)** – a greedy “peak‑flatten”
   routine that repeatedly moves a small amount of mass from an index
   that contributes heavily to the current maximal convolution entry to an
   index that contributes little.

5. **Simulated annealing** – stochastic moves on the simplex with a
   Metropolis rule, temperature slowly cooled.  Donor/receiver are taken
   from the top/bottom % of the current distribution.

6. **Second DMT pass** – a final tightening of the peak.

7. **Adam on a soft‑max surrogate** – projected Adam optimisation on a
   smooth approximation of the max‑convolution (β is exponentially
   increased).  This polishes the solution.

All phases respect the global time budget (`budget_s`, default 1000 s).
The best sequence found is trimmed of trailing near‑zero entries and
returned as a plain ``list[float]``.  The routine prints progress
information and, if the surrounding harness supplies it, evaluates the
final candidate with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sym_product_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i·(n‑i+1))^(‑p) – large at the two ends, small in the centre."""
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1)
    a = prod ** (-p)
    a *= target_sum / a.sum()
    return a


def _sum_two_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i)^(‑p) + (n‑i+1)^(‑p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) + ((n - i + 1) ** (-p))
    a *= target_sum / a.sum()
    return a


def _arcsine_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ 1 / sqrt(i·(n‑i+1)) – the discrete arcsine shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick any maximal index (the first one)
        k = idx_max[0]

        # Indices i that actually participate in conv[k] are
        # i ∈ [max(0, k-(n-1)), min(k, n-1)]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner values a_{k-i}
        partner = a[k - idx]

        # Choose donor: a[idx] > 0 and partner is large
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # Choose receiver: smallest partner among the same idx set,
        # and different from donor.
        # We sort locally to avoid O(n log n) every iteration.
        sorted_loc = np.argsort(partner)
        receiver = None
        for loc in sorted_loc:
            r = idx[loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        # Amount needed to bring the current max down exactly to the second max
        delta_needed = (max_val - second_max) / denom

        # Take a safe fraction of the donor's mass (avoid emptying it completely)
        max_delta = min(a[donor] * 0.95, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    # final projection onto the simplex to eliminate tiny numerical drift
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score

def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer with Metropolis acceptance.  Each move transfers
    a random amount of mass from a high‑value index to a low‑value index.
    Temperature is exponentially cooled.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # initial temperature = 5 % of the current objective (or a tiny floor)
    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.03          # donor / receiver are taken from top/bottom 3 %
    top_k = max(1, int(top_frac * n))

    # we keep sorted lists of indices; they are refreshed regularly
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 1500 == 0:          # occasional refresh (cost O(n log n))
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        if a[donor] <= 0.0:
            it += 1
            continue

        # random amount: up to 40 % of donor’s mass
        delta = a[donor] * rng.random() * 0.4
        if delta < 1e-14:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 4000 == 0:
            T *= 0.85               # moderate cooling
            if T < 1e-7:
                T = 1e-7

    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(b) where b = a*a.
    Returns (grad, b) where grad has the same shape as a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)          # soft max weights, sum to 1

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # gradient w.r.t a (see derivation in analysis)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)

        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_val

    max‑self‑convolution objective.  The routine respects the supplied
    time budget and returns the best list of heights found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5               # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)                    # Σ a = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – extensive parametric sweep (fast)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # Helper to evaluate and keep best
    def _consider(seq: np.ndarray, label: str):
        nonlocal best_seq, best_score
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New best from {label}: {best_score:.6f}")

    # 1) Pure power‑law
    p_grid = np.linspace(0.2, 1.2, 41)               # finer than before
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        _consider(seq, f"pow(p={p:.3f})")
        if time.time() >= deadline:
            break

    # 2) Power‑law + exponential tail (coarse grid first)
    if time.time() < deadline:
        p_grid2 = np.linspace(0.2, 1.2, 21)
        q_grid2 = np.linspace(0.0, 6.0, 21)
        for p in p_grid2:
            for q in q_grid2:
                seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
                _consider(seq, f"exp_pow(p={p:.3f},q={q:.3f})")
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

    # 3) Symmetric product power‑law
    if time.time() < deadline:
        p_grid3 = np.linspace(0.05, 0.8, 26)
        for p in p_grid3:
            seq = _sym_product_power_sequence(n, float(p), target_sum)
            _consider(seq, f"sym_prod(p={p:.3f})")
            if time.time() >= deadline:
                break

    # 4) Sum of two power laws (U‑shaped)
    if time.time() < deadline:
        p_grid4 = np.linspace(0.2, 1.2, 31)
        for p in p_grid4:
            seq = _sum_two_power_sequence(n, float(p), target_sum)
            _consider(seq, f"sum2(p={p:.3f})")
            if time.time() >= deadline:
                break

    # 5) Discrete arcsine shape (no parameters)
    if time.time() < deadline:
        seq = _arcsine_sequence(n, target_sum)
        _consider(seq, "arcsine")

    # 6) Random Dirichlet seeds (few)
    if time.time() < deadline:
        for _ in range(12):
            alpha = rng.uniform(0.05, 0.5)          # heavy‑tailed Dirichlet
            vec = rng.gamma(alpha, 1.0, size=n)
            seq = _project_to_simplex(vec, target_sum)
            _consider(seq, "dirichlet")
            if time.time() >= deadline:
                break

    # 7) Jittered neighbourhood of the current best
    if best_seq is not None and time.time() < deadline:
        for _ in range(8):
            jitter = rng.normal(scale=target_sum * 0.004, size=n)
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, target_sum)
            _consider(cand, "jitter")
            if time.time() >= deadline:
                break

    if best_seq is None:
        # sanity fallback – uniform sequence
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _score(best_seq)

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (aggressive)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    mt_deadline = time.time() + remaining * 0.20
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing (longer)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    sa_deadline = time.time() + remaining * 0.45
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)
    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer (tightening)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    mt2_deadline = time.time() + remaining * 0.15
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt2_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final Adam polishing (soft‑max)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      final_deadline,
                                                      target_sum,
                                                      max_beta=5e5,
                                                      step_factor=0.6,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero elements
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)          # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
