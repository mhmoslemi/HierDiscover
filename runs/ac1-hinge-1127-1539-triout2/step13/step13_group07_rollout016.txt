"""
step_function_optimizer.py

A robust optimizer for the “max‑self‑convolution” objective used by the
competition.  The algorithm consists of four stages, all respecting the
global time budget (default 1000 s).

1.  **Fast parametric sweep** – a large random sample of simple families:
    * pure power‑law,
    * power‑law × exponential tail,
    * power‑law × stretched exponential,
    * symmetric two‑sided exponential.
    Every candidate is projected onto the simplex {a_i ≥ 0, Σa_i = √(2 n)}.
    The best candidate (lowest max‑convolution) is kept.

2.  **Deterministic mass‑transfer (exact DMT2)** – a greedy reduction of the
    current convolution peak.  This step is run for a fixed share of the
    remaining budget and usually drops the score dramatically.

3.  **Extended simulated annealing** – stochastic moves on the simplex:
    * transfer a random amount of mass from a high‑value index to a low‑value
      index,
    * *swap* the values of two random indices.
    Acceptance follows a Metropolis rule with an exponential cooling schedule.
    The temperature is re‑scaled whenever a new best score is found.

4.  **Projected Adam on a soft‑max surrogate** – the soft‑max (log‑sum‑exp)
    approximation of the max‑convolution is differentiable.  Adam with
    projection onto the simplex performs a long fine‑grained polishing run.
    The β‑parameter of the soft‑max is doubled periodically, reaching a very
    high value (≈10⁶) so that the surrogate converges to the true max.

The final sequence is trimmed of trailing near‑zero entries and returned as a
plain list of Python floats.  All helper functions are top‑level; no global
state is mutated except the optional ``height_sequence_1`` cache used for a
warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – simply the maximal entry of a*a (since Σa = √(2 n))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of parametric sequences (used in the fast sweep)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)  (pure power‑law)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(
    n: int, p: float, q: float, target_sum: float
) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p) · exp(‑q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _stretched_exp_power_law_sequence(
    n: int, p: float, q: float, r: float, target_sum: float
) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p) · exp(‑q·(i/n)^r).  The extra exponent r gives a
    “stretched’’ exponential tail."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * (i / n) ** r)
    a *= target_sum / a.sum()
    return a


def _double_sided_exponential_sequence(
    n: int, q: float, target_sum: float
) -> np.ndarray:
    """Symmetric shape a_i = exp(‑q·i) + exp(‑q·(n‑1‑i)).  Both ends are high,
    the centre is low – often useful for flattening the convolution."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-q * i) + np.exp(-q * (n - 1 - i))
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry.  The routine moves a
    carefully chosen amount of mass from a donor index to a receiver index so
    that the current peak drops to the second‑largest value."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices achieving the current maximal value (within a tiny tolerance)
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick a single maximal index (any will do)
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner = a_{k‑i}
        partner = a[k - idx]

        # donor = index among the active window with the largest partner value
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) whose partner value is the
        # smallest among the active window
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        # delta needed to bring the peak down to the second‑largest value
        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # periodic progress report
            print(
                f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}"
            )

        it += 1

    # re‑project onto the simplex (numeric drift may happen)
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score


# ----------------------------------------------------------------------
# Extended simulated annealing (transfer + swap moves)
# ----------------------------------------------------------------------


def _simulated_annealing_extended(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Stochastic moves on the simplex:
        * transfer a random amount of mass from a high‑value index to a low‑value index,
        * swap the values of two random indices.
    Acceptance follows a Metropolis rule with exponential cooling.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # Initial temperature (a fraction of the current objective)
    T = max(best_score * 0.05, 1e-8)

    it = 0
    # Define fractions of the population that are considered “high” and “low”
    top_frac = 0.02
    top_k = max(1, int(top_frac * n))

    # Pre‑compute sorted orders occasionally
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        move_type = rng.random()
        if move_type < 0.7:                     # mass transfer (dominant)
            donor = sorted_desc[rng.integers(top_k)]
            receiver = sorted_asc[rng.integers(top_k)]
            if donor == receiver:
                it += 1
                continue

            max_delta = a[donor]
            if max_delta <= 0.0:
                it += 1
                continue

            # Random fraction up to 30 % of donor mass
            delta = max_delta * rng.random() * 0.3
            if delta < 1e-12:
                it += 1
                continue

            a_candidate = a.copy()
            a_candidate[donor] -= delta
            a_candidate[receiver] += delta

        else:                                   # swap move (20 %)
            i, j = rng.integers(n, size=2)
            if i == j:
                it += 1
                continue
            a_candidate = a.copy()
            a_candidate[i], a_candidate[j] = a_candidate[j], a_candidate[i]

        # No need to re‑project – the moves preserve Σa and non‑negativity.
        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                # when a new best is found we re‑scale the temperature a bit
                T = max(T * 0.7, best_score * 0.02)
                print(
                    f"[{time.time():.2f}] SA new best={best_score:.6f} it={it}"
                )

        # cooling schedule
        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-9:
                T = 1e-9

    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max (log‑sum‑exp) approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a_rev * p)  (convolution of reversed a with p)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 1e6,
    step_factor: float = 0.6,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_score = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250          # increase β every this many iterations

    while time.time() < deadline:
        # gradient of the smoothed max
        g, _ = _gradient_softmax(a, beta)

        # Adam updates
        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape tiny plateaus
        if it % 300 == 0:
            jitter = rng.normal(
                scale=target_sum * 1e-3 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_score:
            best_score = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_score:.6f}"
            )

        # β‑schedule (slow exponential growth)
        if (it + 1) % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        it += 1

    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes the
    maximal self‑convolution.  The routine respects the time budget and
    returns the best list of heights discovered.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility, but costs more)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 600:
        n = 14000
    elif budget_s >= 300:
        n = 10000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n) makes the objective = max(b)

    # ------------------------------------------------------------------
    # Phase 0 – extensive parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # a few deterministic families
    families = [
        ("pow", _power_law_sequence, {"p": (0.2, 2.0)}),
        ("exp_pow", _exp_power_law_sequence, {"p": (0.2, 2.0), "q": (0.0, 25.0)}),
        (
            "stretched",
            _stretched_exp_power_law_sequence,
            {"p": (0.2, 2.0), "q": (0.0, 25.0), "r": (0.5, 2.0)},
        ),
        ("double_exp", _double_sided_exponential_sequence, {"q": (0.0, 25.0)}),
        ("sin", _sin_sequence, {}),
    ]

    # generate a sizable random sample (≈300 candidates) – cheap because each
    # evaluation is just one FFT.
    total_samples = 320
    for _ in range(total_samples):
        family_name, ctor, ranges = rng.choice(families)
        params = {}
        for name, (lo, hi) in ranges.items():
            params[name] = float(rng.uniform(lo, hi))
        seq = ctor(n, **params, target_sum=target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
    # also keep a very small jittered population around the best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    # warm‑start from a previously saved result, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    print(
        f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (quick tightening)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.12
    best_seq, best_score = _deterministic_mass_transfer_exact2(
        best_seq, mt_deadline, target_sum
    )
    print(
        f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase 2 – extended simulated annealing (transfer + swaps)
    # ------------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.48
    best_seq, best_score = _simulated_annealing_extended(
        best_seq, target_sum, sa_deadline
    )
    print(
        f"[{time.time():.2f}] After SA best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer (fine‑grained)
    # ------------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.15
    best_seq, best_score = _deterministic_mass_transfer_exact2(
        best_seq, mt2_deadline, target_sum
    )
    print(
        f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase 4 – long Adam polishing (soft‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(
            best_seq,
            final_deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.6,
            gamma=0.995,
        )
        print(
            f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final safety evaluation (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
