"""
Propose a non‑negative step‑function minimizing the maximal self‑convolution.
The algorithm consists of:

1. Choose a problem size `n` based on the time budget.
2. Initialise the best candidate from a previous solution (if any) or a simple
   power‑law shape.
3. Scan a rich family of parametric shapes (pure power‑law,
   power‑law·exponential, shifted power‑law, logarithmic, and random
   mixtures of a few components).  The best from this cheap sweep is kept.
4. Perform a deterministic greedy “mass‑transfer” that directly lowers the
   current convolution peak.
5. Run a simulated‑annealing (SA) phase that stochastically moves mass from
   high‑value positions to low‑value positions with a Metropolis rule.
6. A second short deterministic mass‑transfer tightens the result.
7. Finally, run a projected Adam optimiser on a soft‑max approximation of
   `max(conv)`.  The temperature (`beta`) is increased exponentially so that
   the optimisation converges to the true max‑convolution objective.
8. Trim trailing near‑zero entries and return the sequence.

All steps respect the overall time budget (default 1000 s).  The code uses
FFT‑based convolutions for speed and Euclidean projection onto the simplex
`{a_i ≥ 0, Σ a_i = sqrt(2 n)}` after each move.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (cheap pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _shifted_power_law_sequence(n: int, p: float, shift: float, target_sum: float) -> np.ndarray:
    """a_i = (i+shift)^{-p}  (shift>0 to avoid singularity)."""
    i = np.arange(n, dtype=np.float64) + shift
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _log_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i = 1 / (log(i+2)+1)  – a smooth slowly decaying shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (np.log(i + 1.0) + 1.0)
    a *= target_sum / a.sum()
    return a


def _random_mixture_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    Build a mixture of 1‑3 elementary components.
    Each component is (i+shift)^{-p} * exp(-q*i/n).
    The mixture weights are random, then the whole vector is projected onto the simplex.
    """
    k = rng.integers(1, 4)                     # number of components (1,2,3)
    weights = rng.random(k)
    weights /= weights.sum()
    seq = np.zeros(n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    for comp in range(k):
        p = rng.uniform(0.05, 3.0)
        q = rng.uniform(0.0, 30.0)
        shift = rng.uniform(0.0, 2.0)
        comp_seq = (i + shift + 1.0) ** (-p) * np.exp(-q * i / n)
        seq += weights[comp] * comp_seq
    seq = _project_to_simplex(seq, target_sum)
    return seq


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index to a
    “receiver’’ index so that the current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices achieving (almost) the maximum
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        # stop if improvement possible is below tolerance
        if max_val - second_max < tol:
            break

        # pick one maximal index (any)
        k = idx_max[0]

        # indices i that participate in b_k = Σ a_i a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)           # i values

        partner = a[k - idx]                                 # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        # we simply take the smallest partner that is not the donor
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        # move a safe fraction of the donor mass (leave a small buffer)
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    # final projection (numeric drift)
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score

# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer with Metropolis acceptance.  Each move
    transfers a random amount of mass from a high‑value index to a low‑value
    index.  Temperature is exponentially cooled.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()
    # initial temperature: fraction of the objective
    T = max(best_score * 0.05, 1e-6)

    it = 0
    top_frac = 0.02
    top_k = max(1, int(top_frac * n))

    # maintain sorted order periodically (cost O(n log n))
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.3          # up to 30 % of donor mass
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        # feasibility already guaranteed (non‑negative, sum unchanged)
        new_score = _max_convolution_fft(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(b) where b = a*a.
    Returns (grad, b) where b is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max weights (sum to 1)

    # gradient of Σ p_k b_k w.r.t a : 2 * (conv_rev of p with a_rev)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # keep only n entries
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max smoothed max‑convolution.
    `beta` is increased geometrically every `beta_inc_iter` steps.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    beta_inc_iter = 300
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)

        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to help escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    # final projection (numeric drift)
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_val

    while time.time() < mixt_deadline:
        seq = _random_mixture_sequence(n, target_sum, rng)
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (quick tightening)
    # --------------------------------------------------------------
    mt1_deadline = time.time() + 0.12 * (deadline - time.time())
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt1_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing (explorative)
    # --------------------------------------------------------------
    sa_deadline = time.time() + 0.35 * (deadline - time.time())
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)
    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer (fine‑tuning)
    # --------------------------------------------------------------
    mt2_deadline = time.time() + 0.12 * (deadline - time.time())
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt2_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Adam gradient descent on soft‑max (final polish)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      final_deadline,
                                                      target_sum,
                                                      max_beta=1e6,
                                                      step_factor=0.6,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final verification (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
