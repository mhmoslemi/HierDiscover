"""
Hybrid optimisation for the “minimum peak convolution” problem.

The objective after fixing the ℓ¹‑norm (∑a = √(2n)) is simply the
maximum entry of the auto‑convolution a*a.  The algorithm works in
phases:

1. **Seeding** – generate diverse candidates (power‑law, exponential,
   Dirichlet‑like and the previous best) and project them onto the
   ℓ¹‑simplex.

2. **Differential Evolution** – a classic DE loop with simplex
   projection; this provides a good global starting point.

3. **Simulated‑Annealing (SA)** – repeatedly move a small amount of mass
   from the index that contributes most to the current peak towards a
   low‑contributing index.  Acceptance follows a temperature schedule
   (Metropolis rule); a single FFT per move is required.

4. **Deterministic mass‑transfer** – a greedy refinement that repeatedly
   transfers mass from the largest product pair at the peak to a weak
   pair, with a decaying acceptance temperature.

5. **Sub‑gradient descent** – projected sub‑gradient steps on the exact
   max‑convolution objective.

6. **Reordering** – three simple permutations (original, alternating,
   even‑spread) are tested; the ordering with the smallest peak is kept.

The runtime budget is 1000 s; the phases share the time automatically.
The final sequence is trimmed of trailing near‑zero entries and stored
in ``globals()['height_sequence_1']`` for warm‑starting the next run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray, target_sum: float) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = target_sum = sqrt(2n) the value equals the max‑convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (different shapes)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (often lower the peak)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_place = True
    for v in sorted_vals:
        if left_place:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_place = not left_place
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Differential Evolution (projected onto simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with projection onto the ℓ¹‑simplex."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind, target_sum) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # pick three distinct mates different from i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (ensure at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial, target_sum)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()

        # stop a little early to reserve time for refinements
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    conv = _conv_fft(a)
    cur_score = float(conv.max())
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        # consider the symmetric partner as an alternative donor
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        # pick a low‑contribution receiver
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        # move a fraction of donor's mass (scaled by temperature)
        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = max_delta * rng.random() * (temp / start_temp)
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = float(conv.max())

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (projected) on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = float(_conv_fft(a).max())
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            # add a tiny jitter to escape shallow plateaus
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = float(_conv_fft(a).max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑Annealing refinement (mass‑transfer with temperature)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    One‑temperature SA loop: move a fraction of mass from the index that
    contributes most to the current peak towards a low‑contributing index.
    Only a single FFT per iteration is required.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_val = cur_max
    start_time = time.time()
    # Initial temperature as a fraction of the current peak
    T_start = max(1e-4, cur_max * 0.2)
    T_end = 1e-8

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = T_start * (1 - frac) + T_end * frac

        # locate the peak index
        kmax = int(np.argmax(conv))

        # indices that contribute to this peak
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        prod = a[idx] * a[partner]

        # donor = side of the largest product
        donor_local = int(np.argmax(prod))
        donor = int(idx[donor_local])
        # optionally switch to the partner if it holds more mass
        if a[partner[donor_local]] > a[donor]:
            donor = int(partner[donor_local])

        # receiver = an index with low contribution (median split)
        median = np.median(prod)
        low_candidates = idx[prod <= median]
        if low_candidates.size == 0:
            receiver = rng.integers(n)
        else:
            receiver = int(rng.choice(low_candidates))

        if donor == receiver:
            continue

        # propose delta proportional to temperature (larger early on)
        max_delta = a[donor] * 0.5
        delta = max_delta * rng.random() * (T / T_start)

        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = _project_to_simplex(a_new, target_sum)

        conv_new = _conv_fft(a_new)
        new_max = float(conv_new.max())

        # Metropolis acceptance
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine builds a diverse seed pool, runs DE, then performs SA,
    deterministic mass‑transfer, sub‑gradient descent and a few reorderings.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives richer search space but higher cost.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 500:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds = []

    for p in np.linspace(0.2, 1.0, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    for _ in range(8):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm start from the previous run if available

        seeds.append(_resize_sequence(prev, n))

        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # --------------------------------------------------------------
    # Initialise DE population (modest size)
    # --------------------------------------------------------------

    population = []

    best_seed = min(seeds, key=lambda s: _score(s, target_sum))
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution phase (≈ 55 % of the budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.55 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated Annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sa_deadline = deadline - 10.0   # keep a margin for later phases
        seq_sa, score_sa = _simulated_annealing(best_seq, target_sum, sa_deadline, rng)
        if score_sa < best_score:
            best_seq, best_score = seq_sa, score_sa
            print(f"[{time.time():.2f}] SA improved – new peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 7.0:
        mt_dead = time.time() + min(25.0, deadline - time.time() - 5.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.4, end_temp=1e-6, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] MT polished – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sg_dead = time.time() + min(20.0, deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_seq, best_score = seq_sg, score_sg
            print(f"[{time.time():.2f}] Sub‑grad refined – peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Reordering to possibly lower the peak further
    # --------------------------------------------------------------
    candidates = [best_seq,
                  _alternating_order(best_seq),
                  _even_spread_order(best_seq)]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq, target_sum)
    print(f"[{time.time():.2f}] After reordering – final peak = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # store for the next run
    globals()["height_sequence_1"] = final_seq

    # sanity check with the provided evaluator
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
