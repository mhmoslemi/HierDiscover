        j = k - i

"""
Advanced step‑function optimizer (≈1000 s budget).

Strategy
--------
1. **Coarse parametric sweep** – power‑law and exponential‑tilted families
   (very cheap) to obtain a decent initial shape.
2. **Small‑scale linear‑programming refinement** – on a reduced problem
   (n≈2000) solve   max ∑g   s.t.   f∗g ≤ M (M = max(f∗f)),   g≥0.
   The optimal g is mixed with f (convex combination) and the small
   problem is iterated a few times.  The resulting shape is up‑sampled
   to the full size.
3. **Projected Adam** on a soft‑max approximation of the max‑convolution.
   Several restarts with a β‑schedule up to 1e6.
4. **Exact deterministic mass‑transfer (DMT2)** – never increases the
   objective, applied until convergence.
5. **Greedy pairwise transfer** – a larger donor/receiver pool than the
   original code.
6. **Simulated‑annealing refinement** – occasional uphill moves to escape
   shallow minima.
7. **Final high‑β Adam polish**.

All phases respect the global time budget; the best sequence found is
trimmed of trailing near‑zero entries and returned as a plain list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Σx = s , x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _truncated_power_law_sequence(n: int, m: int, p: float, target_sum: float) -> np.ndarray:
    """Power‑law on the first m entries, zeros afterwards."""
    a = (np.arange(1, m + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    if m < n:
        a = np.concatenate([a, np.zeros(n - m, dtype=np.float64)])
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities (used by Adam GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smooth max (soft‑max) of a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = rfft(a, n=size)
    b = irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = rfft(a_rev, n=size)
    Pr_fft = rfft(p, n=size)
    conv_rev = irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Several short Adam runs from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Greedy pairwise transfer (enhanced)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.05,
                               receiver_frac: float = 0.05,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest entries
        receiver_idx = np.argsort(a)[:nr]        # smallest entries

        # cache shifted views for all candidates (lightweight view, no copy)
        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)          # effect per unit delta
                d_k = d[k]
                if d_k >= 0.0:
                    continue            # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        base[pad_len + i] = a[i]
        base[pad_len + j] = a[j]

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score

def _simulated_annealing_refinement(a0: np.ndarray,
                                    deadline: float,
                                    target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    rng = np.random.default_rng()
    start_time = time.time()
    T0 = cur_max * 0.02 + 1e-4
    iter_cnt = 0
    refresh_interval = 500
    donor_frac = 0.02
    recv_frac = 0.02

    donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
    receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

    while time.time() < deadline:
        if iter_cnt % refresh_interval == 0:
            donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
            receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        shift_i = base[i:i + 2 * n - 1]
        shift_j = base[j:j + 2 * n - 1]
        diff = shift_j - shift_i

        new_conv = conv + delta * diff
        new_max = new_conv.max()

        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-9)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            base[pad_len + i] = a[i]
            base[pad_len + j] = a[j]

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={iter_cnt} new_best={best_score:.6f}")

        if iter_cnt % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            base = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])

        iter_cnt += 1

    return best_a, best_score

def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score

try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Construct matrix A such that (A @ g)_k = (a * g)[k] (full linear conv)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        A[k, j] = a[i]
    return A


def _lp_refine_small(seq_large: np.ndarray,
                     n_small: int,
                     target_sum_small: float,
                     target_sum_large: float,
                     deadline: float) -> np.ndarray:
    """Iterate LP refinement on a down‑sampled problem and up‑sample back."""
    if linprog is None:
        return seq_large

    # down‑sample
    small_seq = np.interp(np.linspace(0, len(seq_large) - 1, n_small),
                          np.arange(len(seq_large)), seq_large)
    small_seq = _project_to_simplex(small_seq, target_sum_small)

    while time.time() < deadline:
        M = _max_convolution_fft(small_seq)
        A = _build_conv_matrix(small_seq)
        c = -np.ones(small_seq.shape[0])  # maximise sum(g)
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0, None)] * small_seq.shape[0]

        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})
        except Exception as e:
            print(f"[{time.time():.2f}] LP failed: {e}")
            break

        if not res.success:
            print(f"[{time.time():.2f}] LP not successful")
            break

        g = res.x
        s = g.sum()
        if s <= 0:
            break
        # scale to correct sum
        g = g * (target_sum_small / s)

        # convex combine (t = 0.2)
        t = 0.2
        new_seq = (1 - t) * small_seq + t * g
        new_seq = _project_to_simplex(new_seq, target_sum_small)

        old_score = _score(small_seq)
        new_score = _score(new_seq)
        if new_score < old_score - 1e-12:
            small_seq = new_seq
            print(f"[{time.time():.2f}] LP refinement improved score {new_score:.6f}")
        else:
            break

    # up‑sample back to the original length
    large_seq = np.interp(np.linspace(0, n_small - 1, len(seq_large)),
                          np.arange(n_small), small_seq)
    large_seq = _project_to_simplex(large_seq, target_sum_large)
    return large_seq

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    max self‑convolution.  The routine respects the given time budget
    and returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)

    # --------------------------------------------------- Phase 0 : parametric sweep
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power‑law grid
    p_grid = np.linspace(0.30, 0.85, 31)
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # exponential‑tilted grid
    q_grid = np.linspace(0.0, 6.0, 31)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # truncated power‑law variants (extra flexibility)
    for p in p_grid[::5]:            # a few representative exponents
        for frac in (0.5, 0.7, 0.9):
            m = int(frac * n)
            seq = _truncated_power_law_sequence(n, m, float(p), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # incorporate a previously stored construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # small jittered population around the current best
    for _ in range(8):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}")

    # --------------------------------------------------- Phase 0.5 : LP refinement on a small problem
    # allocate up to ~15 % of the remaining budget for this step
    lp_deadline = time.time() + (deadline - time.time()) * 0.15
    n_small = min(2000, n // 6)               # about 2000 for n=12000
    target_sum_small = np.sqrt(2.0 * n_small)

    if time.time() < lp_deadline - 5.0:
        best_seq = _lp_refine_small(best_seq,
                                    n_small,
                                    target_sum_small,
                                    target_sum,
                                    lp_deadline)
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] After LP refinement best = {best_score:.6f}")

    # --------------------------------------------------- Phase A : Adam‑GD with restarts
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # --------------------------------------------------- Phase B : DMT2 + random moves
    while time.time() < deadline - 8.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 5.0:
            break

    # --------------------------------------------------- Phase C : Enhanced pairwise transfers
    if time.time() < deadline - 2.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.06,
                                                      receiver_frac=0.06)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # --------------------------------------------------- Phase D : final high‑β Adam polish
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------- Phase E : simulated‑annealing refinement
    if time.time() + 2.0 < deadline:
        a_sa, val_sa = _simulated_annealing_refinement(best_seq,
                                                       deadline,
                                                       target_sum)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] SA refinement improved best = {best_score:.6f}")

    # --------------------------------------------------- Post‑processing
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
