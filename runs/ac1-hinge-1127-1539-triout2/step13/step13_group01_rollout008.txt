def _gradient_softmax_max(a, b, p):
    # a: vector length n
    # b: vector length 2n-1 = conv of a with itself.
    # p: smoothing parameter.
    maxb = b.max()
    # stable exponents
    diff = b - maxb
    # Clip for exponent
    clip = np.clip(p * diff, -700, 0)  # avoid underflow/overflow
    exps = np.exp(clip)
    sum_exps = exps.sum()
    if sum_exps == 0:
        w = np.ones_like(b) / b.size
    else:
        w = exps / sum_exps
    # compute gradient
    w_rev = w[::-1]
    # convolution w_rev * a via FFT
    n = a.shape[0]
    M = len(w_rev)
    size = 1 << ((M + n - 1).bit_length())
    # Use rfft (real) maybe
    # compute rfft of both padded
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(Fw * Fa, n=size)[: M + n - 1]  # linear conv length M+n-1
    # slice gradient: entries from M-1 to M-1 + n
    start = M - 1
    grad = 2.0 * conv[start : start + n]
    return grad

def smooth_descent(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _score(a)

    p = 0.5  # start smoothing factor
    p_max = 500.0
    # schedule: increase p multiplicatively each iteration or time-based.
    # Might use p = p_max * (1 - np.exp(-k / K))? For simplicity increase linearly per iteration.
    # Also step size decays like alpha0 * (1 / sqrt(iter+1)).

    # Choose step size initial:
    step0 = target_sum * 0.2 # maybe a fraction
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute convolution b = a*a
        b = _conv_fft(a)
        # compute gradient based on smooth max
        # Use p that increases gradually
        # To avoid high p early causing near subgradient, we use moderate p.
        # Compute w = softmax
        maxb = b.max()
        diff = b - maxb
        # scaling p_cur
        # Could increase p with iteration: p = p0 + (p_max - p0)* min(1, it / 20000)
        p_cur = p + (p_max - p) * min(1.0, it / 50000.0)  # if run 50000 it, p will be p_max
        # To ensure stable: if p_cur < 1e-12: treat as uniform.
        clip = np.clip(p_cur * diff, -700, 0)
        exps = np.exp(clip)
        sum_exps = exps.sum()
        if sum_exps == 0 or np.isinf(sum_exps):
            w = np.ones_like(b) / b.size
        else:
            w = exps / sum_exps

        # compute gradient
        w_rev = w[::-1]
        grad = _conv_fft_general(w_rev, a)  # length 3n-2
        M = len(w_rev)
        grad = 2.0 * grad[M-1 : M-1 + n]

        # step size schedule
        step = step0 / np.sqrt(it + 1)
        # direction negative gradient (since we minimize)
        a = a - step * grad
        # project onto simplex
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum*1e-3, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Iter {it} best={best_score:.6f} p={p_cur:.2f}")

        it += 1

    return best, best_score

"""
Propose a low‑score non‑negative step sequence.

The algorithm is a multi‑restart, multi‑phase optimisation:

1. **Warm‑start / random starts** – the current best sequence (if any) is
   resized and projected onto the simplex; additional starts are drawn from
   a Dirichlet distribution.

2. **Smoothed max‑descent** – minimise the quantity
       max_k (a*a)_k
   directly.  The nonsmooth max is approximated by a soft‑max with an
   annealing parameter `p`.  A gradient of the soft‑max is obtained
   analytically (a rank‑one convolution) and a projected gradient step
   (simplex projection to keep Σa = √(2n)) is performed.  The parameter
   `p` grows linearly with elapsed time, turning the smooth problem into
   the original one.

3. **Aggressive mass‑transfer annealing** – a fast O(n) simulated‑annealing
   refinement (the same routine as the reference solution) is applied to
   the best vector obtained from step 2.

All phases respect the global time budget (default 1000 s) and the best
sequence found is stored in the global variable `height_sequence_1` for
warm‑starting future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary vectors using an FFT."""
    size = 1 << ((x.shape[0] + y.shape[0] - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.shape[0] + y.shape[0] - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²) – identical to evaluate_sequence."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smoothed max‑descent
# ----------------------------------------------------------------------


def _smooth_max_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Projected gradient descent on a smoothed version of max(a*a).
    The smoothing parameter `p` grows from p_start to p_max linearly with
    elapsed time, turning the problem into the true max‑minimisation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start

    # hyper‑parameters
    p_start = 0.2          # initial smoothness
    p_max = 500.0          # final (almost exact) max
    lr0 = 1.0              # initial learning‑rate factor

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total

        # current smoothing parameter
        p_cur = p_start + (p_max - p_start) * frac

        # full self‑convolution and soft‑max weights
        b = _conv_fft(a)
        max_b = b.max()
        diff = b - max_b                           # ≤ 0
        exp_arg = p_cur * diff
        # avoid overflow / under‑flow
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # gradient of (soft‑max weighted) max‑convolution:
        #   g = 2 * (w_rev * a)_{2n-2 .. 2n-2+n-1}
        w_rev = w[::-1]
        conv = _conv_fft_general(w_rev, a)          # length 3n-2
        M = w_rev.shape[0]                         # 2n‑1
        grad = 2.0 * conv[M - 1 : M - 1 + n]

        # diminishing step size
        lr = lr0 / np.sqrt(it + 1.0)
        a = a - lr * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=target_sum * 1e-3, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] desc it={it:6d} best={best_val:.6f} p={p_cur:.1f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer (as in the baseline)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from a “bad’’ index
    (large contribution to the current peak) to a “good’’ index.
    The convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        b_new[donor : donor + n] -= 2.0 * delta * a
        b_new[receiver : receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # the usual normalisation for the simplex

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run the optimisation phases for every start, respecting time
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start #{idx+1} (n={n}) -----")

        # ----- Phase 1 : smoothed max‑descent -------------------------
        # Allocate roughly 30 % of the remaining budget to the descent.
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(30.0, remaining * 0.3)

        a_desc, score_desc = _smooth_max_descent(init, target_sum, sub_deadline)
        if score_desc < best_score:
            best_score = score_desc
            best_seq = a_desc.copy()
            print(f"[{time.time():.2f}] New global best after descent: {best_score:.6f}")

        # ----- Phase 2 : aggressive annealing -------------------------
        if time.time() < deadline:
            a_ann, score_ann = anneal_mass_transfer(a_desc, target_sum, deadline)
            if score_ann < best_score:
                best_score = score_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely event that no candidate was found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    # Trim trailing negligible entries (the evaluator ignores trailing zeros anyway)
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Final evaluation (for logging) and warm‑start storage
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
