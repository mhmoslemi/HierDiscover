    # Choose different lengths to explore

    best_seq = None

    # Upscale to larger final length (maybe 12000)

"""
Optimiser for step‑function benchmark.

The algorithm proceeds in three stages:
1.  Initialise a diverse pool of candidate shapes (power‑law,
    exponential, random Dirichlet) of moderate length (≈ 6 k).  The
    best candidate is refined with a long projected sub‑gradient
    descent that works directly on the true peak of the convolution.
2.  The best sequence is up‑sampled to a larger length (≈ 12 k) and
    polished with a high‑β soft‑max gradient descent (smooth
    approximation of the peak).  The β‑schedule is exponential and
    the step size is momentum‑based.
3.  A final simulated‑annealing / mass‑transfer phase swaps large
    contributors with small ones and performs tiny local mass‑moves.
   All iterates stay on the ℓ¹‑simplex (Σa = √(2·n)) and respect
   non‑negativity.

The routine respects the 1000 s time budget, never returns ``np.inf``,
and stores the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

def _max_convolution_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation (preserves total mass)."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    resized = np.interp(new_x, old_x, seq)
    # Preserve total mass (tiny numerical drift)
    resized *= seq.sum() / resized.sum()
    return resized.astype(np.float64)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest values left‑most, second‑largest right‑most, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns gradient (same shape as a) and the exact convolution array.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # stability: subtract max_b before exponentiation
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # convolution of reversed a with p gives gradient of softmax max
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    # the central n entries correspond to derivative w.r.t each a_i
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _softmax_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              target_sum: float,
                              beta_init: float = 0.5,
                              beta_max: float = 1e7,
                              step_base: float = 0.5,
                              gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate of the peak.
    β grows exponentially until beta_max.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)

    beta = beta_init
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base * target_sum / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] GD β={beta:.1e} score={best_val:.6f}")

        # increase beta smoothly
        if beta < beta_max:
            beta = min(beta_max, beta * 1.5)

    return best, best_val

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true peak."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.8

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        # diagonal contribution (i=j) appears only once, derivative is just a_i
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / math.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # jitter every few thousand steps
        if it % 4000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it} score={best_val:.6f}")

        it += 1

    return best, best_val

def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Local mass‑transfer with simulated‑annealing acceptance."""
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(conv.argmax())
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        lst = a[idx] + a[kmax - idx]   # combined weight of each pair
        donor = idx[np.argmax(lst)]

        low_cnt = max(1, int(len(idx) * low_frac))
        receivers = idx[np.argsort(lst)[:low_cnt]]
        receiver = rng.choice(receivers)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best, best_score

def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-6,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Swap a high‑contributing index with a low‑contributing one."""
    a = a0.copy()
    cur = _score(a)
    best = a.copy()
    best_score = cur
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor = rng.choice(np.argpartition(-contrib, top_k - 1)[:top_k])
        receiver = rng.choice(np.argpartition(contrib, low_k - 1)[:low_k])

        if donor == receiver:
            continue

        # swap
        a[donor], a[receiver] = a[receiver], a[donor]
        new = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new < cur or rng.random() < math.exp(-(new - cur) / max(temp, 1e-12)):
            cur = new
            if new < best_score:
                best_score = new
                best = a.copy()
                print(f"[{time.time():.2f}] Perm new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best, best_score

"""
Optimiser for the step‑function benchmark.

The algorithm works in three stages:

1. **Exploration** – a compact pool of seeds (power‑law, exponential,
   random Dirichlet, linear, previous best) of moderate length
   (≈ 6 000) is built.  The seed with the lowest initial score is refined
   with a long projected sub‑gradient descent that works directly on the
   true peak of the convolution.

2. **Upscaling & polishing** – the best sequence is linearly
   interpolated to a larger length (8 000 – 12 000) and projected onto the
   ℓ¹‑simplex (Σa = √(2·n)).  A high‑β soft‑max gradient descent with
   momentum and an exponential β‑schedule is then applied to minimise a
   smooth approximation of the peak.

3. **Final annealing** – a targeted permutation simulated‑annealing
   swaps indices that contribute heavily to the current peak with
   indices that contribute little, followed by a deterministic
   mass‑transfer refinement.  All iterates stay non‑negative and on the
   simplex.

The routine respects the 1000 s time budget, never returns ``np.inf``,
and stores the best sequence in the global variable ``height_sequence_1``.
"""

def _softmax_gradient_descent(a0: np.ndarray,

                              target_sum: float,

                              beta_max: float = 1e7,
                              step_base: float = 0.5,
                              gamma: float = 0.995) -> tuple[np.ndarray, float]:

    Momentum‑based projected gradient descent on the soft‑max surrogate.
    β grows exponentially until ``beta_max``.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()

        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] GD β={beta:.1e} score={best_val:.6f}")

            beta = min(beta_max, beta * 1.5)

# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (max convolution entry)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(a*a)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.8

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(conv.argmax())

        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:  # diagonal term appears once
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / math.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # jitter occasionally
        if it % 4000 == 0 and rng.random() < 0.3:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it} score={best_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small local moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Local mass‑transfer with simulated‑annealing style acceptance."""
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(conv.argmax())
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]
        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        receivers = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(receivers)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score or rng.random() < math.exp(
                -(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing (swap high/low contributors)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-6,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Swap an index that contributes heavily to the peak with a low contributor."""
    a = a0.copy()
    cur = _score(a)
    best = a.copy()
    best_score = cur
    n = a.shape[0]
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor = rng.choice(np.argpartition(-contrib, top_k - 1)[:top_k])
        receiver = rng.choice(np.argpartition(contrib, low_k - 1)[:low_k])

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]
        new = _score(a)

        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new < cur or rng.random() < math.exp(
                -(new - cur) / max(temp, 1e-12)):
            cur = new
            if new < best_score:
                best_score = new
                best = a.copy()
                print(f"[{time.time():.2f}] Perm new best={best_score:.6f}")
        else:
            # revert swap
            a[donor], a[receiver] = a[receiver], a[donor]

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows the three‑phase strategy described above,
    respects the 1000 s time budget and stores the final sequence in
    ``height_sequence_1`` for possible reuse.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 1. Base length for quick exploration
    # ------------------------------------------------------------------
    n_base = 6000
    target_base = math.sqrt(2.0 * n_base)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seeds
    # ------------------------------------------------------------------
    pool = []

    for p in np.linspace(0.35, 0.85, 11):
        pool.append(_power_law_sequence(n_base, float(p), target_base))

    for lam in np.linspace(0.001, 0.02, 13):
        pool.append(_exponential_sequence(n_base, float(lam), target_base))

    for _ in range(6):
        a = rng.random(n_base)
        a = a / a.sum() * target_base
        pool.append(a)

    # linear decreasing shape
    lin = np.linspace(1.0, 0.0, n_base)
    lin = lin / lin.sum() * target_base
    pool.append(lin)

    # previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        pool.append(_resize_sequence(prev, n_base))

    # jittered copies to increase diversity
    for seq in list(pool):
        jitter = rng.normal(scale=target_base * 0.007, size=n_base)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_base)
        pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Long sub‑gradient descent on the most promising seed
    # ------------------------------------------------------------------
    init_scores = [_score(s) for s in pool]
    best_idx = int(np.argmin(init_scores))
    start_seed = pool[best_idx]

    sub_deadline = time.time() + (deadline - time.time()) * 0.55
    best_seq, best_val = _subgradient_descent(start_seed,
                                              sub_deadline,
                                              target_base,
                                              max_iter=400_000)
    print(f"[{time.time():.2f}] After sub‑grad best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Upscale and polish with soft‑max GD
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 300:
        n_final = 12000
    elif remaining > 150:
        n_final = 10000
    else:
        n_final = 8000

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a deterministic re‑ordering (alternating)
    candidates = [seq_big, _alternating_order(seq_big)]
    best_seq = candidates[0]
    best_val = _score(best_seq)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_val:
            best_val = sc
            best_seq = cand

    print(f"[{time.time():.2f}] Upscaled to {n_final} start score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. High‑β soft‑max gradient polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        gd_deadline = deadline - 20
        a_gd, val_gd = _softmax_gradient_descent(best_seq,
                                                 gd_deadline,
                                                 target_final,
                                                 beta_init=0.5,
                                                 beta_max=2e7,
                                                 step_base=0.4,
                                                 gamma=0.998)
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd
            print(f"[{time.time():.2f}] GD polish best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Targeted permutation annealing (most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        perm_deadline = deadline - 5
        a_perm, val_perm = _permutation_targeted_search(best_seq,
                                                        perm_deadline,
                                                        start_temp=0.6,
                                                        end_temp=5e-7,
                                                        top_frac=0.08,
                                                        low_frac=0.08)
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm
            print(f"[{time.time():.2f}] After permutation best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Final deterministic mass‑transfer pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 1:
        mt_deadline = deadline - 0.5
        a_mt, val_mt = _deterministic_mass_transfer(best_seq,
                                                    mt_deadline,
                                                    target_final,
                                                    start_temp=0.3,
                                                    end_temp=1e-6,
                                                    low_frac=0.04)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional logging with the official evaluator
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
