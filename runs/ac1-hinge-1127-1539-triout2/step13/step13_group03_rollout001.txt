def _annealed_gradient_descent(a0, deadline, target_sum):
    a = project_to_simplex(a0, target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_factor, gamma = 0.9, 0.98
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]; size = 1 << ((2*n-1).bit_length())
        # FFT
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
        max_b = b.max()
        # Softmax weights (approx derivative)
        # To avoid overflow: b - max_b
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()
        # conv of a_rev and p
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        grad = 2.0 * np.fft.irfft(Ar_fft * Pr_fft, n=size)[:2*n-1]
        # central part of grad
        grad = grad[n-1:n-1+len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # Occasionally jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001/(beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # print log

        # increase beta gradually
        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val
```

"""
Hybrid optimizer for the step‑function attack.

Goal: minimise max( a * a ) under the ℓ1‑normalisation Σ a = √(2 n)
with a_i ≥ 0.  The evaluation score equals this maximum, so the
algorithm directly optimises max‑convolution.

Approach
--------
1. Build a diverse pool of candidate shapes (power‑law, exponential,
   cosine, half‑sine, random) for a modest size (≈5 k).
2. Repeatedly run a soft‑max (annealed) gradient descent on random
   seeds – fast, exploits the smooth approximation of the max.
3. Refine the best moderate‑size solution with a simulated‑annealing
   mass‑transfer move set.
4. Upscale the solution to a larger problem (10‑25 k elements) by
   linear interpolation and project back onto the ℓ1‑simplex.
5. Apply several large‑scale optimisation passes:
   * deterministic re‑orderings that tend to flatten the autocorrelation,
   * exact sub‑gradient descent (O(n log n)),
   * another mass‑transfer SA,
   * a final sub‑gradient sweep if time permits,
   * short random‑permutation search.
6. Trim trailing near‑zero entries, store the result for reuse and
   return it.

All stages respect the non‑negativity and ℓ1‑simplex constraints,
use FFT‑based convolutions for speed and stop automatically when the
1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine shape, zero at the ends."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, smallest, second‑largest, second‑smallest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values evenly using a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    # step roughly sqrt(n) but made coprime to n
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 8e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Momentum projected GD on the soft‑max approximation of max‑conv.
    Returns the best vector encountered and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = b.max()
        # soft‑max weights
        exps = np.exp(beta * (b - max_b))
        p = exps / exps.sum()

        # gradient of the soft‑max wrt a
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        # momentum update
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on max‑convolution (with momentum).
    Returns best vector and its max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # Indicator of (almost) maximal entries
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        # sub‑gradient: 2 * (I * a_rev) restricted to centre n entries
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass‑transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from a “donor” index (large contribution)
    to a “receiver” index (small contribution) with SA acceptance.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # position of current max
        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        # pick a weak contributor among the low‑fraction of candidates
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to shake the system
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Try a number of random permutations; keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

    for _ in range(8):

        # choose seed (biased towards current best)
        if best_seq is not None and rng.random() < 0.65:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 12.0, explore_deadline)
        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_expl,
            max_beta=9e5, step_factor=0.9, gamma=0.98)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive mass‑transfer on the modest size
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        mt_deadline = time.time() + min(30.0, deadline - time.time() - 15.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            best_seq, mt_deadline, target_expl,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Mass‑transfer refined best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a larger problem size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.5, target_expl)

    remaining = deadline - time.time()
    # decide a larger size based on remaining budget
    if remaining > 300:
        n_big = 25000
    elif remaining > 180:
        n_big = 20000
    elif remaining > 100:
        n_big = 15000
    elif remaining > 60:
        n_big = 10000
    else:
        n_big = n_expl

    target_big = math.sqrt(2.0 * n_big)
    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)

    # deterministic re‑orderings that often flatten the autocorrelation
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        seq_big[::-1]
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_big} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – large‑scale sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 40:
        sub_deadline = deadline - 30.0   # reserve time for final passes
        a_sub, val_sub = _subgradient_descent(seq_big, sub_deadline, target_big,
                                              step_factor=0.45, gamma=0.995)
        if val_sub < best_val:
            best_val = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑gradient improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – large‑scale mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        mt_big_deadline = min(time.time() + 20.0, deadline - 5.0)
        a_mt_big, val_mt_big = _mass_transfer_simulated_annealing(
            seq_big, mt_big_deadline, target_big,
            start_temp=0.3, end_temp=1e-5, low_frac=0.05)
        if val_mt_big < best_val:
            best_val = val_mt_big
            seq_big = a_mt_big.copy()
            print(f"[{time.time():.2f}] Mass‑transfer (big) best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional extra sub‑gradient pass if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        sub2_deadline = deadline - 5.0
        a_sub2, val_sub2 = _subgradient_descent(seq_big, sub2_deadline, target_big,
                                                step_factor=0.35, gamma=0.99)
        if val_sub2 < best_val:
            best_val = val_sub2
            seq_big = a_sub2.copy()
            print(f"[{time.time():.2f}] Extra sub‑gradient best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – quick random permutation search (few trials)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        perm_deadline = deadline - 2.0
        seq_big, val_perm = _permutation_search(seq_big, trials=200, deadline=perm_deadline)
        if val_perm < best_val:
            best_val = val_perm
            print(f"[{time.time():.2f}] Permutation search improved best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and return
    # ------------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # expose for successive runs
    globals()["height_sequence_1"] = final_seq

    # optional final evaluation for logging
    try:
        sc = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {sc:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation failed: {exc}")

    return final_seq
