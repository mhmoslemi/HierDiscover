"""
Optimiser for the self‑convolution peak problem.

Strategy
--------
1. Choose a problem size as large as the time budget allows (up to 20 000
   coefficients).  For a larger `n` the optimal value of the objective
   (the peak of the convolution) is smaller, because the same ℓ¹‑mass can be
   spread over more points.

2. Build a diverse pool of seeds (power‑law, exponential, random Dirichlet,
   the previous best solution, and an “alternating” ordering that already
   reduces the peak).  The best seed is selected by a cheap exact evaluation.

3. Run a long soft‑max gradient descent with a momentum scheme.
   The soft‑max approximates the `max` in the objective; the temperature
   `beta` is increased exponentially up to 5·10⁶, which yields an almost
   exact gradient for the peak.  The iterate is always projected onto the
   ℓ¹‑simplex `{x≥0, Σx = √(2·n)}`.

4. Finish with a short exact sub‑gradient descent (using the true peak
   convolution entry) to polish the result.

5. Apply a cheap “alternating‑order” re‑arrangement and keep whichever
   ordering gives the lower score.

6. Trim trailing zeros, store the result globally as `height_sequence_1`
   and return it.

All heavy work is performed with FFT‑based convolutions which run in
O(n log n) time, so the optimiser fits comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) this equals the peak of the convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of seed sequences
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, projected onto the simplex of sum `target_sum`."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), projected onto the simplex of sum `target_sum`."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """
    Place the largest value left‑most, second largest right‑most,
    third largest next‑to‑left, etc.  This spreads large entries.
    """
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    """Apply alternating order on the reversed sequence and flip back."""
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of the max entry)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, conv_values) where `grad` has shape like `a`.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max probabilities over all convolution entries
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (reverse(a) * p) restricted to first n entries
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum‑annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e6,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected gradient descent on a soft‑max approximation of the peak."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 600 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (cheaper per step)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(convolution)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # exact gradient of the peak entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if (kmax % 2 == 0):
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]               # extra term when i=j

        # diminishing step size
        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # jitter every few thousand steps
        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small local moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Small random mass transfers with a simulated‑annealing acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]               # total weight contributed through kmax

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional jitter to keep exploration alive
        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine adapts the problem size to the time budget, builds a pool
    of diverse seeds, runs a long soft‑max gradient descent on a large
    sequence (up to 20 000 entries), finishes with a short exact sub‑gradient
    pass, tries a few deterministic reorderings, and finally trims trailing
    zeros.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a sequence length that balances quality and speed.
    # Larger n generally yields a smaller achievable peak.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000

    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates.
    # ------------------------------------------------------------------
    seeds = []

    # 1) Previous best (if any) resized to the new length.
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # 2) Power‑law families.
    for p in np.linspace(0.4, 0.8, 7):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # 3) Exponential families.
    for lam in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # 4) Random Dirichlet‑like vectors.
    for _ in range(5):
        a = rng.random(n)
        a *= target_sum / a.sum()
        seeds.append(a)

    # 5) Alternating ordering of a good power‑law (often reduces peaks).
    base = _power_law_sequence(n, 0.6, target_sum)
    seeds.append(_alternating_order(base))

    # 6) Jittered copies of everything above to increase diversity.
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ------------------------------------------------------------------
    # Choose the best seed according to the cheap exact score.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for s in seeds:
        v = _score(s)
        if v < best_val:
            best_val = v
            best_seq = s.copy()
    print(f"[{time.time():.2f}] Initial best seed score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main annealed gradient descent (takes the majority of the budget).
    # ------------------------------------------------------------------
    gd_deadline = deadline - 12.0          # reserve a few seconds for polishing
    if gd_deadline < time.time():
        gd_deadline = deadline

    best_seq, best_val = _annealed_gradient_descent(
        best_seq,
        gd_deadline,
        target_sum,
        max_beta=5e6,
        step_factor=0.5,
        gamma=0.99,
    )
    print(f"[{time.time():.2f}] After GD best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Short exact sub‑gradient polishing (uses the true peak).
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        best_seq, best_val = _subgradient_descent(
            best_seq,
            sub_dead,
            target_sum,
            max_iter=300_000,
        )
        print(f"[{time.time():.2f}] After sub‑grad best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Try a couple of deterministic reorderings (alternating / reversed).
    # ------------------------------------------------------------------
    for cand in (_alternating_order(best_seq), _reversed_alternating(best_seq)):
        v = _score(cand)
        if v < best_val:
            best_val = v
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Reordering gave new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Very short mass‑transfer refinement if a tiny amount of time remains.
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_dead = time.time() + min(2.0, deadline - time.time() - 0.5)
        best_seq, best_val = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_sum,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.04,
        )
        print(f"[{time.time():.2f}] Final MT pass best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing (near‑zero) entries for compactness.
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store globally for the next round and report the final evaluation.
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
