"""
Height‑sequence optimizer for the max‑self‑convolution bound.

Approach
--------
1. Choose a problem size n based on the allocated time budget.
2. Initialise a candidate by interpolating a previously found good sequence
   (if available) or by a simple parametric family (power‑law, exponential‑tail,
   sine).
3. Perform a coarse parametric sweep (power‑law ± exponential tail) and keep the
   best among them.
4. Apply a short deterministic “mass‑transfer” refinement that moves a small
   amount of mass from a donor index to a receiver index in order to lower the
   current convolution peak.
5. Run a series of projected Adam optimiser steps on a smooth soft‑max
   approximation of the max‑convolution.  Several restarts (jittered copies of
   the current best) are attempted while time permits.
6. If time remains, a final deterministic mass‑transfer pass is executed.
7. Trim trailing near‑zero entries, store the result globally for possible reuse,
   and return it as a plain list of floats.

The algorithm respects the 1000 s budget, uses only NumPy (no external solvers)
and keeps all entries non‑negative and bounded, matching the security checks
performed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used for the initial sweep)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)"""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) · exp(-q·i/n)"""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(pi·(i+0.5)/(n+1))   (always positive)"""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """Pure exponential decay: a_i ∝ exp(-lam·i)"""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    # Make sure the sum matches the target (projection)
    best_a = _project_to_simplex(best_a, target_sum)
    best_score = _max_convolution_fft(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max surrogate and Adam optimiser
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(b) where b = a*a.
    Returns (gradient, b) where b is the full convolution.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)           # soft‑max weights (sum to 1)

    # Compute gradient: 2 * (a_rev * p) where * is convolution
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 5e6,
                           step_factor: float = 0.5,
                           gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max surrogate of max‑convolution.
    The schedule for the smoothing parameter beta doubles every ``beta_inc_iter``
    steps (default 200).  The routine stops when the deadline is reached.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 200

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:7.2e} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    best_a = _project_to_simplex(best_a, target_sum)
    best_val = _max_convolution_fft(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the maximal
    self‑convolution.  The routine adapts its problem size to the time
    budget, performs a quick parametric sweep, a short deterministic mass‑
    transfer, and then several Adam‑style projected gradient descents with
    occasional jittered restarts.  The best sequence found before the deadline
    is returned (as a plain list of floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose n based on the available budget (larger n gives more freedom)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n)  → the objective equals max‑conv

    # ------------------------------------------------------------------
    # Initialise from a previously found solution if present,
    # otherwise fall back to a simple parametric shape.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        # Linear interpolation to the new size
        old_x = np.linspace(0.0, 1.0, prev.size)
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, prev)
    else:
        a0 = _exp_power_law_sequence(n, p=0.6, q=2.0, target_sum=target_sum)
    a0 = _project_to_simplex(a0, target_sum)

    # ------------------------------------------------------------------
    # Phase 0 – coarse parametric sweep (fast, cheap)
    # ------------------------------------------------------------------
    best_seq = a0.copy()
    best_score = _max_convolution_fft(best_seq)

    p_grid = np.linspace(0.2, 1.2, 21)
    q_grid = np.linspace(0.0, 6.0, 21)

    for p in p_grid:
        cand = _power_law_sequence(n, float(p), target_sum)
        val = _max_convolution_fft(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    for p in p_grid:
        for q in q_grid:
            cand = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution_fft(cand)
            if val < best_score:
                best_score = val
                best_seq = cand.copy()

    cand = _sin_sequence(n, target_sum)
    val = _max_convolution_fft(cand)
    if val < best_score:
        best_score = val
        best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – brief deterministic mass‑transfer (quick tightening)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.08
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After initial DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – repeated Adam optimisation with jittered restarts
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 2.0:
        # allocate a modest chunk for a full Adam run
        sub_deadline = time.time() + min(60.0, deadline - time.time() - 2.0)
        cand_seq, cand_score = _adam_gradient_descent(best_seq,
                                                      sub_deadline,
                                                      target_sum,
                                                      max_beta=5e6,
                                                      step_factor=0.5,
                                                      gamma=0.99)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] New best after Adam run: {best_score:.6f}")

        # Generate a jittered copy for a fresh start
        jitter = rng.normal(scale=target_sum * 0.0015, size=n)
        jittered = best_seq + jitter
        jittered = np.maximum(jittered, 0.0)
        jittered = _project_to_simplex(jittered, target_sum)

        sub_deadline2 = time.time() + min(30.0, deadline - time.time() - 1.0)
        cand_seq2, cand_score2 = _adam_gradient_descent(jittered,
                                                        sub_deadline2,
                                                        target_sum,
                                                        max_beta=2e6,
                                                        step_factor=0.6,
                                                        gamma=0.995)
        if cand_score2 < best_score:
            best_score = cand_score2
            best_seq = cand_seq2.copy()
            print(f"[{time.time():.2f}] New best after jitter+Adam: {best_score:.6f}")

        # If we already beat the target comfortably we can stop early
        if best_score < 1.504:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final deterministic mass‑transfer (tightening edge cases)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        final_mt_deadline = time.time() + (deadline - time.time()) * 0.15
        best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                                   final_mt_deadline,
                                                                   target_sum)
        print(f"[{time.time():.2f}] After final DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # Store for possible reuse in subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
