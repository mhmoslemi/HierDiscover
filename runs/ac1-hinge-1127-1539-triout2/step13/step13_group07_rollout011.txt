"""
step‑function optimiser – subgradient descent + enhanced families.

The optimiser builds a good starting candidate from a few simple
parameterised families (pure power‑law, power‑law with an exponential
tail, “inverse‑product’’ shape a_i ∝ (i·(n+1‑i))^{-p}), chooses the
best by evaluating the maximal self‑convolution, optionally refines it
with a quick deterministic mass‑transfer pass, and then runs a
projected sub‑gradient descent on the exact max‑convolution objective.
The sub‑gradient of max_i (a*a)_i is computed analytically as a
cross‑correlation between the current argument and the mask of
maximal positions; both the convolution and the correlation are
performed with a single FFT per iteration.  The routine respects the
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two for a*a
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (fast pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _inverse_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """
    a_i ∝ [(i+1)·(n‑i)]^{-p}, i = 0 … n‑1.
    This family concentrates mass towards the two ends and is
    known to flatten the self‑convolution.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    j = np.arange(n, 0, -1, dtype=np.float64)        # n … 1
    prod = i * j                                      # i·(n+1‑i)  (≈ i·(n‑i+1))
    a = prod ** (-p)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (quick tightening)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves mass from a “donor’’ index (large partner value) to a
    “receiver’’ index (small partner value) so that the current peak
    drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on f(a)=max_i (a*a)_i.
    The sub‑gradient at a is 2·corr(a, mask) where mask marks the
    maximal convolution entries.  Each iteration performs:
        a ← Π_S (a − η·g)
    where Π_S projects onto the simplex.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)

    # FFT size that can accommodate both convolution (2n‑1) and
    # correlation (3n‑2).  Use the same size for both to reuse memory.
    size = 1 << ((3 * n - 2).bit_length())
    eps = 1e-12
    it = 0
    # base step factor – will be scaled by 1/√(it+1)
    base_factor = target_sum * 0.5

    while time.time() < deadline:
        # FFT of the current vector (once per iteration)
        A = np.fft.rfft(a, n=size)

        # ----- self‑convolution (to obtain the max value) -----
        conv_full = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- mask of maximal entries -----
        mask = (conv_full >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient via cross‑correlation -----
        M = np.fft.rfft(mask, n=size)
        corr = np.fft.irfft(np.conj(A) * M, n=size)[:n]   # linear correlation length n
        g = 2.0 * corr

        # ----- adaptive step size -----
        # Estimate the gap to the second‑largest value (if any)
        # Use np.partition for O(n) retrieval of the top two values.
        top2 = np.partition(conv_full, -2)[-2:]
        second_max = top2.min()
        if second_max >= max_val - eps:
            second_max = max_val
        gap = max_val - second_max

        norm_g = np.linalg.norm(g) + 1e-12
        if gap > 0:
            # step that would (approximately) close the gap
            eta = gap / (norm_g ** 2 + 1e-12)
        else:
            # fallback diminishing step
            eta = base_factor / np.sqrt(it + 1) / norm_g

        # safeguard: do not move more mass than we have at any coordinate
        eta = min(eta, a.max() / (norm_g + 1e-12))

        # ----- projected update -----
        a_candidate = a - eta * g
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        cur_score = _score(a_candidate)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_candidate.copy()
            print(f"[{time.time():.2f}] SubGD it={it:5d} best={best_score:.6f}")

        a = a_candidate
        it += 1

    return best_a, best_score

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution.  The routine respects the time budget and returns
    the best list of heights found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # -------------------------------------------------------------
    # Choose problem size – larger n gives more freedom.
    # -------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n) makes the objective = max_conv

    # -------------------------------------------------------------
    # Phase 0 – fast parametric sweep over a few families
    # -------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # (a) pure power‑law
    for p in np.linspace(0.20, 1.20, 15):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law + exponential tail
    for p in np.linspace(0.20, 1.20, 8):
        for q in np.linspace(0.0, 5.0, 6):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) inverse‑product family (high ends, low centre)
    for p in np.linspace(0.10, 0.70, 13):
        seq = _inverse_power_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (d) simple sine shape (still useful as a baseline)
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # (e) reuse previous global solution if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            # simple linear interpolation to the new size
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # -------------------------------------------------------------
    # Phase 1 – short deterministic mass‑transfer (quick tightening)
    # -------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.08
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # -------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (the heavy lifter)
    # -------------------------------------------------------------
    sg_deadline = deadline - 20.0                # keep a few seconds at the end
    best_seq, best_score = _subgradient_descent(best_seq,
                                                target_sum,
                                                sg_deadline)
    print(f"[{time.time():.2f}] After sub‑gradient descent best score = {best_score:.6f}")

    # -------------------------------------------------------------
    # Phase 3 – final deterministic mass‑transfer to clean up
    # -------------------------------------------------------------
    final_mt_deadline = time.time() + 12.0
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               final_mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After final DMT best score = {best_score:.6f}")

    # -------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # -------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
