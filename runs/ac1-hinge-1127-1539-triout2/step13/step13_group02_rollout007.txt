"""
Simple exponential‑family optimiser.

The algorithm searches a one‑parameter family a_i ∝ exp(−λ·i) on a
fixed length n.  λ is explored on a logarithmic grid (very small values
≈10⁻⁵ … 5·10⁻³).  For each λ the sequence is scaled to satisfy the
ℓ¹‑normalisation Σa_i = √(2·n) (the evaluation function is homogeneous,
so this fixes the objective to the peak of the auto‑convolution).
The best λ is kept, optionally reordered by an alternating placement of
large and small entries – a cheap trick that often reduces the peak
further – and finally polished by a short soft‑max gradient descent.
The whole routine respects the 1000 s budget and stores the best
construction in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # only used if optional LP refinement is desired


# ----------------------------------------------------------------------
# FFT utilities (fast max‑convolution)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a*a using FFT.
    """
    n = a.shape[0]
    # length of the convolution is 2*n-1; pick a power of two for speed
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) this equals the peak of the convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:               # safety – should never happen
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex Σx = s, x_i ≥ 0.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(convolution).
    Returns (grad, full_convolution_array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # soft‑max weights over the convolution entries
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * (a ⋆ p) centred on the original length
    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum‑annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------
def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Simple momentum based GD on the soft‑max approximation.
    Returns the best vector seen and its (exact) score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5                # start temperature for the soft‑max
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to shake out of plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Alternating order – a cheap heuristic that often lowers the peak
# ----------------------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value left‑most, second largest right‑most, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low convolution peak.
    The strategy is:

      1. Fix a length n (≈8000 for the full budget) and the normalising sum.
      2. Sweep a geometric/exponential family a_i ∝ exp(−λ·i) over a
         logarithmic grid of very small λ (down to 1e‑5).  For each λ we
         scale the vector to the ℓ¹‑simplex and evaluate the true objective.
      3. Keep the best λ; optionally apply the alternating‑order trick.
      4. Perform a short soft‑max gradient‑descent polishing step.
      5. Return the best sequence found.
    The final sequence is stored globally as ``height_sequence_1``.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Choose a problem size
    # --------------------------------------------------------------
    # Larger n gives a finer search space but costs more per FFT.
    # 8000 works well for the 1000 s budget.
    n = 8000
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 2. Scan the exponential family
    # --------------------------------------------------------------
    # λ is explored on a log‑scale.  The region of interest is roughly
    # 1e‑5 … 5e‑3 – values larger than that produce peaks >2, smaller
    # values behave almost uniformly.
    lam_grid = np.geomspace(1e-5, 5e-3, num=50)

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    for lam in lam_grid:
        # time‑out guard
        if time.time() > deadline - 3.0:
            break

        # build the exponential vector and project onto the simplex
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a *= target_sum / a.sum()                      # exact ℓ¹ normalisation
        score = _score(a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] λ={lam:.2e}  score={score:.6f}")

    # --------------------------------------------------------------
    # 3. Cheap re‑ordering heuristic (alternating order)
    # --------------------------------------------------------------
    if best_seq is not None:
        cand = _alternating_order(best_seq)
        cand_score = _score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] alternating order improved to {cand_score:.6f}")

    # --------------------------------------------------------------
    # 4. Optional short polishing with soft‑max GD
    # --------------------------------------------------------------
    # Use a very short window (≈5 s) if time permits.
    if best_seq is not None and time.time() < deadline - 5.0:
        polish_dead = time.time() + min(5.0, deadline - time.time() - 1.0)
        a_polished, val_polished = _annealed_gradient_descent(
            best_seq,
            polish_dead,
            target_sum,
            max_beta=2e5,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_polished < best_score:
            best_score = val_polished
            best_seq = a_polished.copy()
            print(f"[{time.time():.2f}] polishing GD improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final tidy‑up
    # --------------------------------------------------------------
    # Trim trailing zeros for compactness.
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    nz = np.nonzero(best_seq > 1e-12)[0]
    if nz.size:
        best_seq = best_seq[: nz[-1] + 1]

    final_seq = best_seq.tolist()

    # --------------------------------------------------------------
    # 6. Logging of the final score (optional, safe even if evaluate_sequence
    #    is not imported in the local environment)
    # --------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
