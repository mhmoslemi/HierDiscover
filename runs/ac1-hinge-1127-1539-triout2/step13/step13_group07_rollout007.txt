"""
Propose a low‑max‑self‑convolution step function.

The algorithm works in several phases:

1. **Parameter sweep** – generate a handful of analytic families
   (power‑law, power‑law + exponential tail, sine, Gaussian, linear,
   random Dirichlet).  The best candidate is kept.

2. **Deterministic mass‑transfer (DMT)** – a greedy reduction of the
   current convolution peak: move a safe amount of mass from a “donor”
   index (large partner value) to a “receiver” index (small partner
   value) until the peak drops to the second‑largest value.

3. **Simulated annealing (SA)** – stochastic mass moves with a
   Metropolis rule, allowing the search to escape local minima.

4. **Projected Adam on a soft‑max surrogate** – smooth approximation of
   the max‑convolution (soft‑max with annealed β).  The Adam‑style
   updates are run for the bulk of the remaining budget, aggressively
   increasing β to approach the true max.

5. **Final DMT polish** – another short deterministic pass.

All iterates are projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)}.  The routine respects the given 1000 s
budget and prints progress information.
"""

from __future__ import annotations
import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    # next power‑of‑two size for zero‑padding
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple analytic families for the quick pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=np.float64) ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-((i - centre) ** 2) / (2.0 * sigma ** 2))
    a *= target_sum / a.sum()
    return a


def _linear_sequence(n: int, target_sum: float) -> np.ndarray:
    a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    a *= target_sum / a.sum()
    return a


def _random_simplex_sequence(rng: np.random.Generator,
                             n: int,
                             target_sum: float) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> Tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    The routine moves a carefully chosen amount of mass from a donor
    index to a receiver index so that the current peak drops to the
    second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # pick one maximal index (the first)
        k = idx_max[0]

        # feasible donor‑receiver indices for this k
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                     # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index among feasible donors with largest partner value
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> Tuple[np.ndarray, float]:
    """Stochastic mass‑transfer with Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # initial temperature – a fraction of the current objective
    T = max(best_score * 0.05, 1e-6)
    it = 0

    # donor/receiver are taken from the top / bottom fraction of the array
    top_frac = 0.015
    top_k = max(1, int(top_frac * n))

    # periodically refreshed orderings
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 1500 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        # transfer a random fraction (up to 35 %) of the donor mass
        delta = max_delta * rng.random() * 0.35
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 4000 == 0:
            T *= 0.88
            T = max(T, 1e-7)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray]:
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = b.max()
    # soft‑max probabilities (scaled for numerical stability)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # gradient = 2 * conv_rev where rev_a is a reversed
    a_rev = a[::-1]
    Pr_fft = np.fft.rfft(p, n=size)
    Ar_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]

    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e6,
                           step_factor: float = 0.8,
                           gamma: float = 0.998) -> Tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 200       # increase β more frequently

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % 300 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """Search for a non‑negative step function minimizing max self‑convolution."""
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 850:
        n = 24000
    elif budget_s >= 600:
        n = 18000
    else:
        n = 12000

    target_sum = float(np.sqrt(2.0 * n))

    # --------------------------------------------------------------
    # Phase 0 – light parametric sweep
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # ---- power‑law ------------------------------------------------
    for p in np.linspace(0.25, 1.05, 9):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # ---- power‑law + exponential tail -----------------------------
    for p in np.linspace(0.30, 1.00, 7):
        for q in np.linspace(0.0, 6.0, 7):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # ---- simple sine -----------------------------------------------
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # ---- Gaussian --------------------------------------------------
    for sigma_factor in np.linspace(0.07, 0.25, 5):
        sigma = sigma_factor * n
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # ---- linear (triangular) ---------------------------------------
    seq = _linear_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # ---- reuse previous global solution if available ----------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # ---- random Dirichlet restarts ---------------------------------
    for _ in range(4):
        cand = _random_simplex_sequence(rng, n, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    # ---- tiny jittered perturbations --------------------------------
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (quick refinement)
    # --------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.12
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated annealing
    # --------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.30
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)
    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer
    # --------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.13
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt2_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Adam polishing (soft‑max with aggressive beta schedule)
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        adam_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      adam_deadline,
                                                      target_sum,
                                                      max_beta=2e6,
                                                      step_factor=0.85,
                                                      gamma=0.998)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final deterministic polish (very short)
    # --------------------------------------------------------------
    final_mt_deadline = time.time() + (deadline - time.time()) * 0.07
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               final_mt_deadline,
                                                               target_sum)
    print(f"[{time.time():.2f}] Final DMT polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return as plain list
    # --------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
