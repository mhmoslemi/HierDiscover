def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    start = time.time()
    deadline = start + budget_s - 0.2
    # initialize
    if 'height_sequence_1' in globals():
        seq = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n = seq.size
        target_sum = np.sqrt(2.0 * n)
        seq = _project_to_simplex(seq, target_sum)
    else:
        n = 12000 if budget_s >= 800 else 8000
        target_sum = np.sqrt(2.0 * n)
        seq = _power_law_sequence(n, 0.5, target_sum)
    best_seq = seq.copy()
    best_score = _score(best_seq)

    # Simulated annealing loop
    rng = np.random.default_rng()
    # compute normalized probabilities for donor selection
    total_sum = best_seq.sum()
    # we may recompute weights each iteration to adapt.
    # but we can compute p = best_seq/total_sum.
    # Also we compute step_factor as small fraction of target_sum.

    # Temperature schedule parameters
    init_temp = best_score * 0.05   # initial ~5% of best
    final_temp = best_score * 0.001
    # let's iterate as many as possible
    while time.time() < deadline - 5.0:
        # dynamic parameters
        elapsed = time.time() - start
        t_frac = elapsed / (deadline - start)  # 0..1
        temp = final_temp + (init_temp - final_temp) * (1 - t_frac)  # decreasing
        # choose donor weighted by current best_seq (higher mass more likely)
        if best_seq.max() == 0:
            break
        prob = best_seq / best_seq.sum()
        donor = rng.choice(n, p=prob)
        receiver = rng.integers(n)
        if donor == receiver:
            continue
        # choose delta
        max_delta = best_seq[donor] * 0.9
        if max_delta <= 0:
            continue
        # delta magnitude proportion to target_sum and t_frac
        # early stage larger steps, later smaller
        step_scale = 0.01 + 0.04 * (1 - t_frac)  # from 0.05 to 0.01
        delta = step_scale * target_sum * rng.random()
        if delta > max_delta:
            delta = max_delta
        cand = best_seq.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        # due to floating errors, ensure nonnegativity
        cand = np.maximum(cand, 0.0)
        # No need to project exactly because sum unchanged (subject small rounding)
        # but rounding errors may accumulate; we can renorm every 1000 iterations.
        # compute score
        cand_score = _score(cand)

        if cand_score < best_score - 1e-12:
            best_seq = cand
            best_score = cand_score
            print(f"[{time.time():.2f}] SA improvement score={best_score:.6f}")
        else:
            # acceptance probability
            if rng.random() < np.exp(-(cand_score - best_score) / (temp + 1e-12)):
                best_seq = cand
                best_score = cand_score

        # occasional renormalize to exact simplex to avoid drift
        if int(elapsed * 10) % 1000 == 0:
            best_seq = _project_to_simplex(best_seq, target_sum)

    # final local refinement: call deterministic mass transfer if time left
    if time.time() < deadline - 2.0:
        best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq, deadline, target_sum)

    # Trim near zeros
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[:nz[-1] + 1]

    final_seq = arr.tolist()
    # final safety eval
    try:
        val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluation = {val:.6f}")
    except Exception:
        pass

    return final_seq

"""
Proposes a non‑negative step function that (approximately) minimises the
maximum self‑convolution value.
The algorithm works in three stages:

1. Initialise from the best known sequence (global variable
   ``height_sequence_1``) or from a simple power‑law shape.
2. Run a simulated‑annealing / random‑walk search that repeatedly moves a
   small amount of mass from a high‑mass index to a low‑mass index,
   accepting improvements and occasionally uphill moves.
3. Apply the exact deterministic mass‑transfer refinement (the same
   routine used in the reference solution) on any remaining time budget.

All operations respect the simplex constraint Σa = √(2·n) and a_i ≥ 0.
The final list is trimmed of trailing near‑zero entries before being
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s} using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families (fallback initialisation)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (max‑convolution)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective: maximum entry of the full self‑convolution."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Exact mass‑transfer improvement step (identical to the reference impl)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second best value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # values that pair with idx to give k
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # choose donor with largest partner value
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose receiver with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises the maximal self‑convolution.
    The method combines a random mass‑transfer (simulated annealing) phase
    with the exact deterministic mass‑transfer refinement used in the
    reference solution.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # Initialisation (use previous best if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a0 = np.array(globals()['height_sequence_1'], dtype=np.float64)
        n = a0.size
        target_sum = np.sqrt(2.0 * n)
        a0 = _project_to_simplex(a0, target_sum)
    else:
        # fallback: a simple power‑law shape
        n = 12000 if budget_s >= 800 else (8000 if budget_s >= 400 else 5000)
        target_sum = np.sqrt(2.0 * n)
        a0 = _power_law_sequence(n, p=0.5, target_sum=target_sum)

    best_seq = a0.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Simulated annealing / random mass‑transfer
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    init_temp = best_score * 0.05          # start with a modest temperature
    final_temp = best_score * 0.001        # cool down to a tiny temperature

    it = 0
    renorm_interval = 500   # how often we re‑project onto simplex

    while time.time() < deadline - 4.0:
        # Linear cooling schedule based on elapsed proportion
        elapsed = time.time() - start_time
        total = deadline - start_time
        t_frac = elapsed / total
        temp = final_temp + (init_temp - final_temp) * (1.0 - t_frac)

        # Choose a donor proportional to current mass (high‑mass indices are more likely)
        if best_seq.max() == 0:
            break
        prob = best_seq / best_seq.sum()
        donor = rng.choice(n, p=prob)

        # Choose a receiver uniformly (favoring low‑mass positions)
        receiver = rng.integers(n)
        if donor == receiver:
            continue

        # Propose moving a small fraction of the donor's mass
        max_delta = best_seq[donor] * 0.9
        if max_delta <= 0:
            continue
        # Step size shrinks as we advance
        step_scale = 0.01 + 0.04 * (1.0 - t_frac)      # from ~0.05 down to ~0.01
        delta = step_scale * target_sum * rng.random()
        delta = min(delta, max_delta)

        cand = best_seq.copy()
        cand[donor] -= delta
        cand[receiver] += delta

        # Clip tiny negatives that may appear due to rounding
        cand = np.maximum(cand, 0.0)

        # No full projection here (sum is preserved up to floating error)
        cand_score = _score(cand)

        # Acceptance
        improved = cand_score < best_score - 1e-12
        if improved:
            best_seq = cand
            best_score = cand_score
            print(f"[{time.time():.2f}] SA it={it:6d} improved score={best_score:.6f}")
        else:
            # uphill move with probability exp(-Δ/Temp)
            if rng.random() < np.exp(-(cand_score - best_score) / (temp + 1e-12)):
                best_seq = cand
                best_score = cand_score

        # Periodically renormalise onto the simplex to avoid drift
        if it % renorm_interval == 0:
            best_seq = _project_to_simplex(best_seq, target_sum)

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Exact deterministic mass‑transfer refinement (if time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        best_seq, best_score = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum)

    # ------------------------------------------------------------------
    # Post‑processing
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final sanity check (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluation = {val:.6f}")
    except Exception:
        pass

    return final_seq
