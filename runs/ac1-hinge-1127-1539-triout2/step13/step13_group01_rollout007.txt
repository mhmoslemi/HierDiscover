def smooth_gradient_descent(a0, target_sum, deadline, alpha=30.0, step_initial=0.5):
    a = a0.copy()
    best = a.copy()
    best_val = _score(a) # compute original objective (max conv)
    start = time.time()
    while time.time() < deadline:
        # compute b, gradient using alpha
        b = _conv_fft(a)   # length 2n-1
        m = b.max()
        # compute softmax weights
        # subtract max for stability
        w = np.exp(alpha * (b - m))
        w = w / w.sum()
        # compute gradient
        # size for FFT: next power of 2 > len(a_rev) + len(w) -1 = n + (2n-1) -1 = 3n-2
        size = 1 << ((3*a.shape[0] - 2).bit_length())
        a_rev = a[::-1]
        # compute FFTs
        Fa = np.fft.rfft(a_rev, n=size)
        Fw = np.fft.rfft(w, n=size)
        conv_rev_w = np.fft.irfft(Fa * Fw, n=size)[:3*a.shape[0]-2]
        grad = 2.0 * conv_rev_w[a.shape[0]-1:a.shape[0]-1 + a.shape[0]]
        # compute step size decay
        elapsed = time.time() - start
        total = deadline - start
        eta = step_initial * (1.0 - elapsed / total) + 0.001
        # gradient descent step
        a = a - eta * grad
        # projection to simplex
        a = _project_to_simplex(a, target_sum)
        # optionally small jitter or noise
        # evaluate original objective
        val = _score(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[{time.time():.2f}] SmoothGD improvement {best_val:.6f}")
    return best, best_val

def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.1

    # choose parameters based on budget
    if budget_s >= 800:
        n_fine = 25000
        n_coarse = 8000
    elif budget_s >= 500:
        n_fine = 18000
        n_coarse = 6000
    else:
        n_fine = 10000
        n_coarse = 4000

    target_sum_coarse = np.sqrt(2.0 * n_coarse)
    target_sum_fine = np.sqrt(2.0 * n_fine)

    best_seq = None
    best_val = np.inf

    # Phase 1: coarse search
    phase1_deadline = start + 0.2 * (deadline - start)
    # generate candidates: warm start if exists, random Dirichlet (norm simplex)
    candidates = []
    rng = np.random.default_rng()

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n_coarse)
        warm = _project_to_simplex(warm, target_sum_coarse)
        candidates.append(warm)

    for _ in range(5):
        rnd = rng.random(n_coarse)
        candidates.append(_project_to_simplex(rnd, target_sum_coarse))

    for idx, init in enumerate(candidates):
        if time.time() >= phase1_deadline:
            break
        # subgradient descent
        sub_deadline = time.time() + 0.12 * (phase1_deadline - time.time())
        a_sub, val_sub = subgradient_projected_descent(init, target_sum_coarse, sub_deadline)
        # annealing
        if time.time() < phase1_deadline:
            a_ann, val_ann = anneal_mass_transfer(a_sub, target_sum_coarse, phase1_deadline)
        else:
            a_ann, val_ann = a_sub, val_sub
        # keep best coarse
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New global best coarse value {best_val:.6f}")

    # Upsample best coarse to fine size
    if best_seq is None:
        # fallback random fine
        init_fine = _project_to_simplex(rng.random(n_fine), np.sqrt(2.0*n_fine))
    else:
        init_fine = _resize_sequence(best_seq, n_fine)
        init_fine = _project_to_simplex(init_fine, target_sum_fine)

    # Phase 2: fine search
    phase2_deadline = start + 0.8 * (deadline - start)
    # subgradient refinement
    a_fine_sub, val_fine_sub = subgradient_projected_descent(init_fine, target_sum_fine, phase2_deadline)
    # annealing refinement
    a_fine_ann, val_fine_ann = anneal_mass_transfer(a_fine_sub, target_sum_fine, deadline)

    # final smooth gradient descent
    final_deadline = deadline
    a_final, val_final = smooth_gradient_descent(a_fine_ann, target_sum_fine, final_deadline)

    final_seq = a_final[:np.where(a_final > 1e-12)[0][-1]+1].tolist()
    # store
    globals()["height_sequence_1"] = final_seq
    # optional evaluation
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore
        print(f"[{time.time():.2f}] Final evaluated score {final_score:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error {e}")

    return final_seq

"""
Propose a low‑score step function.

The routine works in three stages:

1. **Coarse optimisation** (small n, many random restarts).  
   Each start is refined by projected sub‑gradient descent on the
   true maximal convolution entry followed by an aggressive
   simulated‑annealing “mass‑transfer” phase.

2. **Upscaling** – the best coarse sequence is interpolated to a
   larger problem size (larger n gives more degrees of freedom).

3. **Fine optimisation** – the up‑scaled vector is further refined
   by (a) another sub‑gradient pass, (b) annealing and finally
   (c) a smooth‑max gradient descent (soft‑max approximation of the
   max‑convolution) with projection onto the simplex
   {x≥0, Σx = √(2·n)}.

All intermediate vectors are kept feasible by Euclidean projection
onto the simplex.  The best sequence seen is stored in the global
variable `height_sequence_1` for warm‑starts in later runs.

The function respects the given time budget (default 1000 s) and
returns a plain ``list[float]`` that can be fed to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution / scoring utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    # because we keep Σa = √(2·n) the denominator equals 2·n
    # and the score is simply max_b.  Keeping the full expression is
    # harmless and safeguards against accidental sum drift.
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – projected sub‑gradient descent (max‑entry)
# ----------------------------------------------------------------------


def subgradient_projected_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the entry that achieves the
    maximum of the convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    # more gentle schedule than the original code
    while time.time() < deadline:
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k_max - idx]

        elapsed = time.time() - start
        total = deadline - start
        # step size slowly shrinks from 0.7 → 0.001
        eta = 0.7 * (1.0 - elapsed / total) + 0.001

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # tiny jitter every few hundred steps – helps escape flat regions
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 2e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d}  best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that moves a small amount of mass from a
    “bad’’ index (large contribution to the current peak) to a
    “good’’ index (small height).  The convolution is updated in O(n).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                     # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # linear cooling – keeps the system warm a bit longer than sqrt
        temperature = max(0.0, T0 * (1.0 - frac))

        # current peak (may have ties)
        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to the peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – bias towards small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – temperature‑scaled
        max_delta = min(a[donor] * 0.12, target_sum * 0.018)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # ----- O(n) convolution update (exact) -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (remove from donor, add to receiver)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improve best={best_val:.6f}")

        # occasional full recompute to avoid drift
        if it % 3000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – smooth‑max gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray, target_sum: float,
                           deadline: float,
                           alpha: float = 5.0,
                           step_initial: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of the max‑convolution
    entry:  softmax_α(b) = (1/α)·log Σ exp(α·b).  The gradient is
    2·a * w where w is the softmax weight vector.
    The iterate is projected onto the simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # full convolution
        b = _conv_fft(a)                # length 2n-1
        m = b.max()
        # soft‑max weights – stabilised by subtracting max
        w = np.exp(alpha * (b - m))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # compute gradient = 2·a ⋆ w  (cross‑correlation)
        # we use FFT on a_rev (length n) and w (length 2n-1)
        a_rev = a[::-1]
        size = 1 << ((3 * n - 2).bit_length())
        Fa = np.fft.rfft(a_rev, n=size)
        Fw = np.fft.rfft(w, n=size)
        conv_rev_w = np.fft.irfft(Fa * Fw, n=size)[: 3 * n - 2]
        grad = 2.0 * conv_rev_w[n - 1:n - 1 + n]   # slice of length n

        # step size – linear decay
        elapsed = time.time() - start
        total = deadline - start
        eta = step_initial * (1.0 - elapsed / total) + 1e-4

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # tiny random shake every few hundred iterations
        if rng.random() < 0.001:
            a += rng.normal(scale=target_sum * 5e-5, size=n)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth‑GD best={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the

    """
    start = time.time()
    deadline = start + budget_s - 0.2          # safety margin

    # --------------------------------------------------------------
    # Choose problem sizes (coarse vs fine) based on the total budget
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_fine = 25000
        n_coarse = 8000
    elif budget_s >= 500:
        n_fine = 18000
        n_coarse = 6000
    else:
        n_fine = 12000
        n_coarse = 4000

    target_sum_fine = np.sqrt(2.0 * n_fine)
    target_sum_coarse = np.sqrt(2.0 * n_coarse)

    # --------------------------------------------------------------
    # Phase 0 – prepare several initial vectors (warm start + random)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    init_candidates = []

                                         dtype=np.float64), n_coarse)
        warm = _project_to_simplex(warm, target_sum_coarse)
        init_candidates.append(warm)

    for _ in range(5):
        rnd = rng.random(n_coarse)
        init_candidates.append(_project_to_simplex(rnd, target_sum_coarse))

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 1️⃣ Coarse optimisation (sub‑gradient + anneal) on n_coarse
    # --------------------------------------------------------------
    coarse_deadline = start + 0.22 * (deadline - start)   # ≈ 22 % of total time
    for idx, init in enumerate(init_candidates):
        if time.time() >= coarse_deadline:
            break
        print(f"[{time.time():.2f}] --- coarse start #{idx + 1} ---")
        sub_dead = time.time() + 0.12 * (coarse_deadline - time.time())
        a_sub, v_sub = subgradient_projected_descent(init,

                                                target_sum_coarse,

        else:

        if v_ann < best_score:

            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New global best (coarse) = {best_score:.6f}")

    # --------------------------------------------------------------
    # sub‑gradient phase (~30 % of the remaining budget)
    sub_deadline_fine = time.time() + 0.30 * (deadline - time.time())
    a_fine_sub, v_fine_sub = subgradient_projected_descent(fine_start,
                                                          target_sum_fine,
                                                          sub_deadline_fine)

    # annealing phase (~50 % of the remaining budget)
    anneal_deadline_fine = time.time() + 0.50 * (deadline - time.time())
    a_fine_ann, v_fine_ann = anneal_mass_transfer(a_fine_sub,

                                               target_sum_fine,

    return final_seq
