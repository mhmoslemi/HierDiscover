"""
Advanced step‑function optimiser using a linear‑programming alternating scheme.

The optimiser works as follows:

1. Initialise a non‑negative sequence of length *n* on the ℓ¹‑simplex
   ( Σ aᵢ = √(2·n) ).

2. Repeatedly:
   a) compute M = max( a * a ).
   b) solve the LP   maximise   Σ gⱼ          (g ≥ 0)
                     subject to   a ∗ g ≤ M .
      (a ∗ g is a linear convolution, thus the constraints are linear.)
   c) scale g to the simplex, mix it with the current a
      (a ← (1‑t)·a + t·g,  t≈0.5) and keep the best sequence found.

   The process monotonically decreases the peak of the self‑convolution.

3. Upscale the best sequence to a larger length (up to 8000) while
   preserving the ℓ¹‑norm, then polish the result with the soft‑max
   gradient descent from the previous state‑of‑the‑art code.

4. Return the final sequence (trailing zeros trimmed).  The global
   variable ``height_sequence_1`` is updated for possible reuse.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, then projected onto the simplex."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), then projected onto the simplex."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize that conserves total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) the value equals max(a*a).
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Soft‑max gradient descent (same implementation as the previous state‑of‑art)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum‑based GD on the soft‑max approximation of the peak."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming alternating optimisation
# ----------------------------------------------------------------------


def _lp_step(f: np.ndarray, M: float, target_sum: float, t_mix: float = 0.5):
    """
    Given current sequence f and its peak M = max(f∗f), solve

        maximise   Σ g_j
        subject to   f ∗ g ≤ M ,  g ≥ 0

    Returns the mixed sequence (1‑t)·f + t·g_scaled
    where g_scaled is scaled onto the simplex Σ = target_sum.
    """
    n = f.shape[0]

    # Build the convolution matrix A (size (2n‑1) × n) such that A·g = f∗g
    # Toeplitz construction: first column = f padded with zeros, first row = [f0,0,…,0]
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(col, row)                     # shape (2n‑1, n)

    # Linear programme: minimise -1ᵀg   (i.e. maximise Σg)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})

    if not res.success:
        # If the LP fails we simply return the original f
        return f.copy(), False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return f.copy(), False

    # Scale onto the ℓ¹‑simplex
    scale = target_sum / sum_g
    g_scaled = g * scale

    # Mix and re‑project (numerical drift)
    new_seq = (1.0 - t_mix) * f + t_mix * g_scaled
    new_seq = _project_to_simplex(new_seq, target_sum)

    return new_seq, True

    while time.time() < lp_deadline:
        # current peak
        M = _conv_fft(best_seq).max()

        # one LP step
        new_seq, ok = _lp_step(best_seq, M, target_lp, t_mix=0.5)
        if not ok:
            break

        new_score = _score(new_seq)

        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
            no_improve = 0
            print(f"[{time.time():.2f}] LP iter {it:4d} – new best score = {best_score:.6f}")
        else:
            no_improve += 1

        # early stop if we see a long stretch without improvement
        if no_improve >= 15:
            break

        it += 1

    print(f"[{time.time():.2f}] LP phase finished – length {n_lp}, best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Upscale to a larger length (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 45 and budget_s >= 800:
        n_final = 8000
    elif remaining > 30:
        n_final = 6000
    elif remaining > 20:
        n_final = 4000
    else:
        n_final = n_lp

    if n_final != n_lp:
        seq_big = _resize_sequence(best_seq, n_final)
        target_big = math.sqrt(2.0 * n_final)
        seq_big = _project_to_simplex(seq_big, target_big)
    else:
        seq_big = best_seq.copy()
        target_big = target_lp

    cur_score = _score(seq_big)
    print(f"[{time.time():.2f}] Upscaled to {n_final} – start score {cur_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final high‑β gradient descent polishing (uses the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        gd_dead = deadline - 2.0
        polished, polished_score = _annealed_gradient_descent(
            seq_big,
            gd_dead,
            target_big,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if polished_score < cur_score:
            seq_big = polished
            cur_score = polished_score
            print(f"[{time.time():.2f}] Polishing improved score to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 6. Trim trailing zeros and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    print(f"[{time.time():.2f}] Final candidate length {len(final_seq)} – score approx {cur_score:.6f}")

    # store for possible reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
