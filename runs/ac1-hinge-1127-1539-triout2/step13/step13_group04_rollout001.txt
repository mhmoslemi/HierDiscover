def _subgradient_multiple(a0, target_sum, deadline, max_iter=5000):

    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0
    while time.time()<deadline and it<max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        # find indices where conv approx max
        eps = 1e-12
        idx_max = np.flatnonzero(conv >= max_val - eps)
        # find second max
        conv_tmp = conv.copy()
        conv_tmp[idx_max] = -np.inf
        second_max = conv_tmp.max()
        delta_target = max_val - second_max
        if delta_target < 1e-15:
            break
        # gradient: sum for each k in idx_max of 2 * a_rev shifted
        # Use p vector: indicator of idx_max
        p = np.zeros_like(conv)
        p[idx_max] = 1.0 / len(idx_max)  # average, perhaps equal weights
        # gradient computation via convolution
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)  # length 2n-1
        grad = 2.0 * grad_full[n-1:n-1+len(a)]

        norm2 = np.dot(grad, grad)
        if norm2 == 0:
            # jitter random
            a = _project_to_simplex(a + rng.normal(scale=target_sum*0.001, size=a.shape), target_sum)
            it+=1
            continue
        step = delta_target / norm2 # similar to earlier but not times 0.9?
        # optionally cap step
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))
        # backtracking line search:
        improved = False
        cur_score = max_val
        for _ in range(10):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_score = _max_convolution_fft(a_candidate)
            if cand_score < cur_score - 1e-12: # improvement
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                improved = True
                break
            else:
                step *= 0.5
        if not improved:
            # jitter
            a = _project_to_simplex(a + rng.normal(scale=target_sum*0.0005, size=a.shape), target_sum)

        it += 1
    return best_a, best_score

2. **Multi‑max sub‑gradient descent** – instead of reacting to a single
   convolution peak, the gradient is built from *all* current maximal
   indices.  A classic sub‑gradient step is taken and projected onto the
   simplex.  This reduces the peak much faster than the single‑index
   version used before.

3. **Penalty descent with a shrinking target** – after stage 2 the
   remaining gap is chased by solving a penalised optimisation
   (max(conv) − target)+.  The target is decreased gradually until no
   further progress is observed.

Optional random mass‑transfer moves are interleaved to escape possible
plateaus.  All operations are vectorised and use FFT‑based convolutions,
so the method easily scales to n≈30 000 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of arbitrary vectors x and y via FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for a quick pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, width_frac: float, target_sum: float) -> np.ndarray:
    """Symmetric triangular bump centred in the middle, width = width_frac·n."""
    width = max(1, int(width_frac * n))
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    dist = np.abs(i - centre)
    a = np.clip((width - dist) / width, 0.0, None)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective: maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Multi‑max sub‑gradient descent (exact max)
# ----------------------------------------------------------------------


def _multi_subgradient_descent(a0: np.ndarray,
                               target_sum: float,
                               deadline: float,
                               max_iter: int = 6000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that uses *all* active maximal convolution
    indices to form the descent direction.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices where the max is essentially attained
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # second highest value (needed for a sensible step size)
        conv_tmp = conv.copy()
        conv_tmp[idx_max] = -np.inf
        second_max = conv_tmp.max()
        gap = max_val - second_max
        if gap < 1e-14:
            break

        # build gradient: average over all maximal indices
        p = np.zeros_like(conv)
        p[idx_max] = 1.0 / idx_max.size
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)                     # length 2n‑1
        grad = 2.0 * grad_full[a.shape[0] - 1:a.shape[0] - 1 + a.shape[0]]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            # inject a tiny jitter to escape a degenerate situation
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=a.shape),
                                    target_sum)
            it += 1
            continue

        # classic sub‑gradient step size
        step = gap / norm2
        # cap the step to keep us inside the simplex neighbourhood
        step = min(step, target_sum * 0.4 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        cur_score = max_val
        for _ in range(8):
            a_cand = _project_to_simplex(a - step * grad, target_sum)
            cand_score = _max_convolution_fft(a_cand)
            if cand_score < cur_score - 1e-12:
                a = a_cand
                best_a = a_cand.copy()
                best_score = cand_score
                improved = True
                break
            step *= 0.5
        if not improved:
            # random jitter when line‑search fails
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.0005, size=a.shape),
                                    target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty descent – drive max(conv) below a prescribed target
# ----------------------------------------------------------------------


def _penalty_descent(a0: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     init_target: float,
                     target_reduction: float = 1e-4,
                     max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves the penalised problem
        minimise (max(conv) - target)+
    while gradually lowering ``target``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    target = init_target
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and it < max_iters:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target + 1e-12:
            # succeeded for current target – lower it a bit more
            best_a = a.copy()
            best_max = cur_max
            target = max(target - target_reduction, 0.0)
            # continue trying to push it further
            continue

        # penalty vector (only positive violations)
        p = conv - target
        p[p < 0] = 0.0

        # gradient = 2 * (p * a_rev) over the window of length n
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n),
                                    target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(6):
            a_cand = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_cand)
            if cand_max < cur_max - 1e-12:
                a = a_cand
                best_a = a_cand.copy()
                best_max = cand_max
                improved = True
                break
            step *= 0.5
        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n),
                                    target_sum)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Simple random mass‑transfer moves (exploration)
# ----------------------------------------------------------------------


def _random_mass_transfer(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         move_fraction: float = 0.02) -> tuple[np.ndarray, float]:
    """Very cheap stochastic move: shift a tiny amount from a large entry to a small one."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    while time.time() < deadline:
        # pick donor from upper tail, receiver from lower tail
        donor = np.argpartition(-a, int(move_fraction * n))[:1][0]
        receiver = np.argpartition(a, int(move_fraction * n))[:1][0]
        if donor == receiver:
            continue
        delta = rng.random() * a[donor] * 0.05
        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            # occasional print
            print(f"[{time.time():.2f}] Random transfer new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal maximal self‑convolution.
    The routine consists of a quick parametric sweep, a multi‑max sub‑gradient
    phase, a penalty‑descent phase and a few random mass‑transfer refinements.
    The final sequence is returned as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a small safety margin

    # ---- problem size -------------------------------------------------
    if budget_s >= 800:
        n = 25000
    elif budget_s >= 500:
        n = 18000
    elif budget_s >= 300:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)      # Σa = √(2·n)  → evaluation returns max‑conv directly

    # ---- phase 0 : parametric pre‑search -------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # power‑law family
    for p in np.linspace(0.2, 1.2, 21):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # power‑law + exponential tail
    for p in np.linspace(0.2, 1.2, 11):
        for q in np.linspace(0.0, 8.0, 11):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # sinusoidal
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # Gaussian bumps
    for sigma in np.linspace(n / 30, n / 8, 7):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # Triangular shapes
    for width in np.linspace(0.1, 0.5, 9):
        seq = _triangular_sequence(n, width, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # If a previous solution exists, interpolate it to current n
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if old.size != n:
            old_x = np.linspace(0.0, 1.0, old.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, old)
        else:
            resized = old
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    print(f"[{time.time():.2f}] Pre‑search best = {best_score:.6f}")

    # ---- Phase A : multi‑max sub‑gradient descent --------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.35
    a_sub, val_sub = _multi_subgradient_descent(best_seq, target_sum,
                                                subgrad_deadline, max_iter=8000)
    if val_sub < best_score:
        best_score, best_seq = val_sub, a_sub.copy()
        print(f"[{time.time():.2f}] Multi‑subgrad improved best = {best_score:.6f}")

    # ---- Phase B : penalty descent with slowly shrinking target -------
    # We aim to go a few thousandths lower than the current best.
    penalty_deadline = time.time() + (deadline - time.time()) * 0.30
    initial_target = best_score - 0.0035
    a_pen, val_pen = _penalty_descent(best_seq, target_sum,
                                      penalty_deadline,
                                      init_target=initial_target,
                                      target_reduction=0.0008,
                                      max_iters=15000)
    if val_pen < best_score:
        best_score, best_seq = val_pen, a_pen.copy()
        print(f"[{time.time():.2f}] Penalty descent improved best = {best_score:.6f}")

    # ---- Phase C : occasional random mass‑transfer moves -------------
    rand_deadline = time.time() + (deadline - time.time()) * 0.20
    a_rand, val_rand = _random_mass_transfer(best_seq, target_sum, rand_deadline)
    if val_rand < best_score:
        best_score, best_seq = val_rand, a_rand.copy()
        print(f"[{time.time():.2f}] Random mass‑transfer improved best = {best_score:.6f}")

    # ---- Post‑processing : trim near‑zero tail -----------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional sanity check (does not affect the returned sequence)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
