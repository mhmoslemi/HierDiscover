"""
Hybrid optimiser for the step‑function problem.

The algorithm proceeds in four stages within the allotted time budget:

1. **Seeding & DE** – generate a diverse set of feasible sequences
   (power‑law, exponential, random Dirichlet‑like, previous best) and
   run a projected Differential‑Evolution optimiser for roughly the
   first half of the time budget.  The population size is a little
   larger than in the reference implementation to increase diversity.

2. **Deterministic mass‑transfer** – a long greedy “mass‑move” phase
   that repeatedly shifts a small amount of weight from a donor (the
   pair that contributes most to the current maximal convolution entry)
   to a low‑contributing receiver.  The schedule for the Metropolis
   acceptance temperature is long enough to escape shallow local minima.

3. **Projected sub‑gradient descent** – an accelerated sub‑gradient
   method on the exact max‑convolution (approximated via a smooth log‑sum‑exp
   surrogate for robustness).  Momentum is kept and the step‑size
   decays slowly.

4. **Permutation annealing** – the final sequence is only a permutation
   of the multiset of heights found above.  A simulated‑annealing
   routine repeatedly swaps two positions; a swap is accepted if it
   reduces the max‑convolution or with a temperature‑controlled probability
   otherwise.  This often uncovers a better ordering that the earlier
   “alternating” and “even‑spread” tricks miss.

All intermediate vectors are projected onto the ℓ¹‑simplex
{a_i≥0, Σa_i = √(2 n)} after each modification, satisfying the
security checks in ``evaluate_sequence``.  The routine returns the best
sequence found before the global deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (FFT implementation)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (FFT implementation)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)².  For Σa = √(2n) it equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Seed factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of an old sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Reordering helpers

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → left, second largest → right, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        left_turn = not left_turn
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Place large values on a full‑cycle step to spread them uniformly."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.8,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        # temperature schedule (linear in log‑space)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))               # index of current max
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]               # contribution of each pair
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.8,
                         gamma: float = 0.90) -> tuple[np.ndarray, float]:
    """Momentum sub‑gradient descent with periodic jitter."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        # sub‑gradient of the max‑convolution entry
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional jitter to escape flat regions
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 3e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.7,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # pick three distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # early break when only a couple of seconds remain
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing permutation search
# ----------------------------------------------------------------------


def _anneal_permutation(a0: np.ndarray,
                       deadline: float,
                       start_temp: float = 0.2,
                       end_temp: float = 1e-5) -> tuple[np.ndarray, float]:
    """Swap‑based annealing that only changes the ordering of the current heights."""
    a = a0.copy()
    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # propose a swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]

        new_score = _score(a)

        # temperature schedule (linear in log‑space)
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert the swap
                a[i], a[j] = a[j], a[i]

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimisation for the step‑function competition.
    Returns a list of non‑negative heights that (hopefully) beats the current
    best upper bound.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Build seed pool
    # --------------------------------------------------------------
    seeds = []

    # power laws (wide range of exponents)
    for p in np.linspace(0.0, 1.5, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponentials (including very flat ones)
    for lam in np.geomspace(5e-5, 2e-2, 7):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random Dirichlet‑like points
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape[0] != n:
            prev = _resize_sequence(prev, n)
        seeds.append(prev)

    # a few jittered copies for extra diversity
    extra = []
    for seq in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        extra.append(seq_j)
    seeds.extend(extra)

    # --------------------------------------------------------------
    # Initialise DE population
    # --------------------------------------------------------------
    pop_size = max(30, int(n / 300))   # a bit larger than before
    population = []

    # start with the best seed we have
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # --------------------------------------------------------------
    # Phase 1 : Differential Evolution (≈ 50 % of budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.50 * budget_s
    de_deadline = min(de_deadline, deadline - 2.0)
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline,
                                        F=0.7, CR=0.9)
    print(f"[{time.time():.2f}] DE finished, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 : Deterministic mass‑transfer (≈ 30 % of budget)
    # --------------------------------------------------------------
    mt_deadline = de_deadline + 0.30 * budget_s
    mt_deadline = min(mt_deadline, deadline - 2.0)
    if time.time() < mt_deadline:
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum,
            start_temp=1.0, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 : Sub‑gradient descent (≈ 15 % of budget)
    # --------------------------------------------------------------
    sg_deadline = mt_deadline + 0.15 * budget_s
    sg_deadline = min(sg_deadline, deadline - 2.0)
    if time.time() < sg_deadline:
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_deadline, target_sum,
            step_factor=0.8, gamma=0.90)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 : Permutation annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        anneal_deadline = deadline - 0.5
        seq_an, score_an = _anneal_permutation(best_seq, anneal_deadline,
                                               start_temp=0.1, end_temp=1e-6)
        if score_an < best_score:
            best_score, best_seq = score_an, seq_an.copy()
            print(f"[{time.time():.2f}] After annealing best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Try a few deterministic reorderings (as in the reference)
    # --------------------------------------------------------------
    cand = [best_seq,
            _alternating_order(best_seq),
            _even_spread_order(best_seq)]
    best_seq = min(cand, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After final reordering score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Record for the next run and return
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # the harness supplies this function
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
