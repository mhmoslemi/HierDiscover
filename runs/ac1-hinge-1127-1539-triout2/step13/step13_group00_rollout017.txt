"""
Gradient‑descent optimisation of a non‑negative step function to minimise
the maximum entry of its self‑convolution (equivalently the evaluation
objective).  The algorithm starts from a diverse set of simple seeds,
projecting them onto the ℓ¹‑simplex ∑a = √(2 n).  A smooth‑max
approximation of the max‑convolution is minimised by a projected gradient
descent with momentum, using FFTs for fast convolution and for the
gradient.  The smoothing parameter α is increased gradually, turning the
smooth objective into the true max‑convolution.  After the main descent
a few cheap local mass‑transfer moves and simple reorderings are applied.
The best sequence found within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    left_turn = True
    for v in sorted_vals:
        if left_turn:
            out[left] = v

"""
Gradient‑descent optimisation of a non‑negative step function to minimise
the evaluation score (the maximum entry of its self‑convolution).
The method starts from several simple seeds, projects them onto the
ℓ¹‑simplex ∑a = √(2 n) and then runs a projected gradient descent on a
smooth approximation of the max‑convolution.  The smoothing parameter
α is increased from a small to a very large value, turning the smooth
objective into the true max‑convolution.  A momentum term stabilises the
updates and the Euclidean projection onto the simplex keeps the
variables feasible.  After the main descent a few cheap local
mass‑transfer moves and three deterministic reorderings are tried; the
best sequence found within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    For a sequence normalised to Σa = √(2 n) this equals the max‑convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then normalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then normalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly using a step coprime to the length."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def _smooth_max_descent(initial: np.ndarray,
                        target_sum: float,
                        start_time: float,
                        deadline: float,
                        alpha_start: float = 0.5,
                        alpha_end: float = 5000.0,
                        step_factor: float = 0.5,
                        momentum_beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of
    ``max_k (a*a)[k]``.  The smoothness parameter α is increased
    geometrically from ``alpha_start`` to ``alpha_end`` as time passes.
    Returns the best sequence observed and its true score.
    """
    a_cur = initial.copy()
    n = a_cur.shape[0]
    # FFT length that can contain a convolution up to length 3n‑2
    size = 1 << ((3 * n - 2).bit_length())
    best_seq = a_cur.copy()
    best_val = _score(a_cur)
    momentum = np.zeros_like(a_cur)

    total_time = max(deadline - start_time, 1e-12)

    while True:
        now = time.time()
        if now >= deadline:
            break

        fraction = (now - start_time) / total_time
        fraction = min(max(fraction, 0.0), 1.0)

        # smooth‑max parameter α
        alpha = alpha_start * (alpha_end / alpha_start) ** fraction

        # ----- convolution a*a ------------------------------------------------
        fa = np.fft.rfft(a_cur, n=size)
        conv_full = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        max_conv = conv_full.max()

        # ----- smooth weights --------------------------------------------------
        exp_term = np.exp(alpha * (conv_full - max_conv))   # max entry becomes 1
        w_norm = exp_term / exp_term.sum()                 # weights sum to 1

        # ----- gradient of the smooth‑max -------------------------------------
        # gradient = 2 * (a_rev * w_norm)  (linear convolution)
        fa_rev = np.fft.rfft(a_cur[::-1], n=size)
        fw = np.fft.rfft(w_norm, n=size)
        conv_rev_full = np.fft.irfft(fa_rev * fw, n=size)[: 3 * n - 2]
        grad = 2.0 * conv_rev_full[n - 1: 2 * n - 1]        # length n

        # ----- momentum update ------------------------------------------------
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad

        # ----- adaptive step --------------------------------------------------
        norm_m = np.linalg.norm(momentum) + 1e-12
        step = step_factor * target_sum / norm_m

        # ----- projected update ------------------------------------------------
        a_next = a_cur - step * momentum
        a_next = _project_to_simplex(a_next, target_sum)

        # ----- keep the best observed -----------------------------------------
        cur_val = _score(a_next)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_next.copy()

        a_cur = a_next

    return best_seq, best_val

                a_cur = a_candidate
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = a_candidate.copy()
        print(f"[{time.time():.2f}] Local polishing best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Try a few deterministic reorderings (they often lower the max‑conv)
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
    ]

    # one random shuffle for good measure
    rand_perm = best_seq.copy()
    rng.shuffle(rand_perm)
    candidates.append(rand_perm)

    best_seq = min(candidates, key=_max_convolution_fft)
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Record globally and return
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
