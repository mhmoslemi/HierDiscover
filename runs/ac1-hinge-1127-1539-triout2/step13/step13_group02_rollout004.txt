"""
Advanced optimiser for the convolution‑peak problem.

The algorithm consists of three main phases:

1. **Flattening (L2‑gradient descent)** – minimise the squared deviation of
   the convolution from its theoretical average value
   `t = (Σa)²/(2·n‑1)`.  This encourages a flat autocorrelation which
   usually lowers the ℓ∞‑norm.

2. **Soft‑max high‑β gradient polishing** – a momentum based gradient descent
   on a smooth soft‑max approximation of the peak.  The temperature `β`
   is increased exponentially up to a very large value, effectively
   optimising the true ℓ∞‑objective.

3. **Final refinement** – a short sub‑gradient descent followed by a
   deterministic mass‑transfer pass to clean up the remaining tiny
   irregularities.

All steps keep the vector on the ℓ¹‑simplex `Σa = √(2·n)` and enforce
non‑negativity by Euclidean projection onto the simplex.  The routine
uses only NumPy and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of sequences (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} then projected onto the simplex."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then projected onto the simplex."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize while preserving total mass."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum‑based GD on the soft‑max approximation of the peak."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (cheap per step)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(convolution)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # exact gradient of the peak entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if (kmax % 2 == 0):
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]          # derivative of a_mid^2 is 2*a_mid

        # diminishing step size
        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# L2‑flattening gradient (minimise ∑(conv‑t)²)
# ----------------------------------------------------------------------


def _l2_gradient(a: np.ndarray, target: float) -> np.ndarray:
    """
    Gradient of ½·∑(conv - target)² w.r.t. a
    where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2·n‑1
    diff = conv - target                    # same length
    a_rev = a[::-1]

    # convolution of diff with a_rev (full length = (2n‑1)+(n)‑1 = 3n‑2)
    size = 1 << ((len(diff) + len(a_rev) - 1).bit_length())
    diff_fft = np.fft.rfft(diff, n=size)
    rev_fft = np.fft.rfft(a_rev, n=size)
    conv_rev = np.fft.irfft(diff_fft * rev_fft, n=size)[: len(diff) + len(a_rev) - 1]

    # the gradient entries are the central n elements of conv_rev
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad


def _l2_flatten_descent(a0: np.ndarray,
                        deadline: float,
                        target_sum: float,
                        step_factor: float = 0.4,
                        momentum_coeff: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the L2 deviation of the convolution
    from its average value.  This quickly flattens the peak.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    n = a.shape[0]
    # theoretical average of the convolution under the ℓ¹‑constraint
    t_const = (target_sum ** 2) / (2 * n - 1)

    best_a = a.copy()
    best_val = _score(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        grad = _l2_gradient(a, t_const)
        momentum = momentum_coeff * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_factor * target_sum / (norm + 1e-12)
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if rng.random() < 0.005:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] L2‑flatten score={best_val:.6f}")

    return best_a, best_val

                                                    low_frac=0.05)
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    print(f"[{time.time():.2f}] Final evaluated score = {best_val:.6f}")

    # expose for possible later reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
