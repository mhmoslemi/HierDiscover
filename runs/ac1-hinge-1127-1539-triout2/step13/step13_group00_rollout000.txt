def _permute_anneal(seq, deadline):
    n = len(seq)
    cur = seq.copy()
    cur_score = _max_convolution_fft(cur)
    best_seq = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # normalization not needed as only reorder
    while time.time() < deadline:
        elapsed = time.time() - start
        # temperature schedule decreasing linearly from start_temp to end_temp (e.g., 0.02 to 1e-4)
        start_temp = 0.02
        end_temp = 1e-5
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)
        # propose swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        new_seq = cur.copy()
        new_seq[i], new_seq[j] = cur[j], cur[i]
        new_score = _max_convolution_fft(new_seq)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / (T + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = cur.copy()
                best_score = new_score
    return best_seq, best_score

"""
Hybrid search for a low‑max‑convolution step function.

The routine builds a diverse set of parametric seeds (power‑law,
exponential, combined power‑law × exponential × taper), projects them
onto the ℓ¹‑simplex ∑a = √(2 n), runs a modest projected Differential Evolution,
then improves the ordering of the values with a simulated‑annealing
swap‑based permutation search and finally refines the heights with
deterministic mass‑transfer and a short sub‑gradient descent.  The best
sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    The objective used by the original evaluator:
        2·n·max(a*a) / (∑a)² .
    For ∑a = √(2 n) this equals max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to target sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _combined_sequence(n: int,
                       p: float,
                       lam: float,
                       q: float,
                       target_sum: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^(-p) · exp(-lam·i) · (1 - i/(n‑1))^q,
    then renormalised.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p)
    a *= np.exp(-lam * i)
    a *= (1.0 - i / (n - 1)) ** q
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass‑moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # the index where convolution is maximal
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Projected Differential Evolution."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Permutation simulated‑annealing (swap‑based)
# ----------------------------------------------------------------------


def _permute_anneal(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing that only reorders the values."""
    n = len(seq)
    cur = seq.copy()
    cur_score = _max_convolution_fft(cur)
    best_seq = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    start_temp = 0.02
    end_temp = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start
        # linear temperature schedule
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        new_seq = cur.copy()
        new_seq[i], new_seq[j] = cur[j], cur[i]
        new_score = _max_convolution_fft(new_seq)
        delta = new_score - cur_score
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur.copy()
    return best_seq, best_score
