"""
Improved step‑function optimizer.

The routine builds several diverse initial candidates (warm‑start,
random Dirichlet, power‑law and Hann windows), then for each candidate
runs

1. *Smoothed max‑convolution gradient descent*.
   The true max is approximated by a log‑sum‑exp (soft‑max) with a
   decreasing temperature ε.  The gradient of this smooth surrogate
   can be written as a linear cross‑correlation, which we evaluate with
   two FFTs per iteration.  After each step we project onto the simplex
   {x≥0, Σx = √(2·n)}.

2. *Aggressive simulated‑annealing mass transfer* (the same fast O(n)
   rank‑one update used in the previous state‑of‑the‑art code) to
   polish the result.

Both phases respect the global time budget and keep the best sequence
found.  The final sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Alias kept for compatibility with the annealing code."""
    return _conv_full(a)


def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation r[i] = Σ_j a[j]·w[i+j] (zero‑padded)
    for i = 0,…,len(a)‑1. Implemented with FFTs.
    """
    n = a.shape[0]
    m = w.shape[0]             # always 2·n‑1 in our usage
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(A) * W, size)[:n]
    return corr


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence but without the clamping)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """2·n·max(a*a) / (sum(a)²).  No clipping performed."""
    n = a.shape[0]
    max_b = _conv_full(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient descent
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray, target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a log‑sum‑exp surrogate of the
    max‑convolution.  Returns the best vector seen and its objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # ---- current convolution and soft‑max temperature ----
        b = _conv_full(a)               # length 2·n‑1
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (target_sum * target_sum)

        # ----- schedule for the smoothing temperature ε -----
        elapsed = time.time() - start
        total = deadline - start
        prog = min(1.0, elapsed / total)
        # ε shrinks from 0.5·max_b to 0.001·max_b
        eps_factor = 0.001 + (0.5 - 0.001) * (1.0 - prog)
        eps = max_b * eps_factor + 1e-12

        # ----- soft‑max weights -----
        w = np.exp((b - max_b) / eps)   # w_i ∈ (0,1], max at the peak

        # ----- gradient of the smooth surrogate -----
        grad = 2.0 * _cross_corr(a, w)   # shape (n,)

        # ----- step‑size: decreasing linear schedule -----
        step_factor = 2.0 * (1.0 - prog) + 0.01
        norm_g = np.linalg.norm(grad) + 1e-12
        step = step_factor * target_sum / norm_g

        # ----- back‑tracking line search (max 5 attempts) -----
        improved = False
        for _ in range(5):
            a_candidate = a - step * grad
            a_candidate = _project_to_simplex(a_candidate, target_sum)
            cand_val = _score(a_candidate)
            if cand_val <= cur_val:
                a = a_candidate
                cur_val = cand_val
                improved = True
                break
            step *= 0.5

        # if no improvement, keep the current a (but still project)
        if not improved:
            a = _project_to_simplex(a - step * grad, target_sum)

        # ----- bookkeeping -----
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] grad‑descent it={it:5d} score={best_val:.6f}")

        # occasional random jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer (unchanged)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from ‘bad’
    indices (large contribution to the current peak) to ‘good’ indices
    (small height).  The convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                    # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of transfer – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper functions for candidate generation
# ----------------------------------------------------------------------


def _power_law(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-exponent)."""
    arr = np.arange(1, n + 1, dtype=np.float64) ** (-exponent)
    return arr


def _hann_window(n: int) -> np.ndarray:
    """Discrete Hann (sin²) window, zero at both ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    return np.sin(np.pi * t) ** 2


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)       # the usual normalisation

    # --------------------------------------------------------------
    # Build a list of initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # warm start from previous run (if any)
    if "height_sequence_1" in globals():
        warm = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # random Dirichlet (alpha = 1) – i.e. uniform on the simplex
    for _ in range(2):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    # power‑law seeds (different exponents)
    for exp in (0.55, 0.80, 1.10):
        pl = _power_law(n, exp)
        pl = pl * (target_sum / pl.sum())
        pl = _project_to_simplex(pl, target_sum)
        init_candidates.append(pl)

    # Hann window seed
    hann = _hann_window(n)
    hann = hann * (target_sum / hann.sum())
    hann = _project_to_simplex(hann, target_sum)
    init_candidates.append(hann)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optimise each start, respecting the global budget
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        now = time.time()
        if now >= deadline:
            break

        remaining = deadline - now
        # allocate roughly half of the remaining budget to this start
        sub_deadline = now + 0.45 * remaining
        ann_deadline = deadline   # remaining time after gradient

        print(f"[{time.time():.2f}] ----- start #{idx + 1} (n={n}) -----")

        # ----- Phase A: smoothed gradient descent -----------------
        a_g, val_g = _smooth_gradient_descent(init, target_sum, sub_deadline)
        if val_g < best_score:
            best_score = val_g
            best_seq = a_g.copy()
            print(f"[{time.time():.2f}] New global best after grad‑descent: {best_score:.6f}")

        # ----- Phase B: aggressive annealing ----------------------
        if time.time() < ann_deadline:
            a_a, val_a = anneal_mass_transfer(a_g, target_sum, ann_deadline)
            if val_a < best_score:
                best_score = val_a
                best_seq = a_a.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up, evaluation and warm‑start storage
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: simple decreasing power‑law
        fallback = _power_law(n, 0.6)
        fallback = fallback * (target_sum / fallback.sum())
        best_seq = fallback

    arr = np.array(best_seq, dtype=np.float64)

    # discard trailing near‑zero entries (they carry no mass)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
