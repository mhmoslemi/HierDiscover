"""
Hybrid smoothed‑gradient descent optimiser for the non‑negative step‑function
problem.

The algorithm keeps the ℓ¹‑norm ∑a = √(2 n) fixed (simplex projection) and
minimises the maximum entry of the linear convolution a∗a.
It repeatedly smooths the max‑operator with a soft‑max (temperature β) and
performs a projected gradient step computed via FFT‑based correlation.
β is increased continuously (continuation method) so that the optimisation
starts with a smooth landscape and ends at the true max‑objective.
A pool of diverse seeds (power‑law, exponential, random, previous best) is
explored; each seed is refined for a few seconds with the gradient method,
the best candidate is then polished by a deterministic mass‑transfer
routine and a few simple reorderings are tried.
The whole routine respects the 1000 s budget and returns the best sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x_i ≥ 0, Σ x_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scoring (objective = max‑convolution when ∑a = √(2 n))
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective value for a (assumes ∑a ≈ √(2 n))."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    return float(max_b)


# ----------------------------------------------------------------------
# Sequence factories (diverse seeds)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(−p) and renormalised to target ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(−lam·i) and renormalised to target ℓ¹‑norm."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (often reduce the peak of autocorrelation)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly using a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing (still used as a final touch)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.4,
                                end_temp: float = 1e-5,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # locate a position where convolution is maximal
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score

# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,

    """
    start = time.time()
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_score = _score(a)

    n = a.shape[0]
    conv_len = 2 * n - 1
    # length needed for linear correlation a (size n) with w (size conv_len)
    corr_len = conv_len + n - 1          # = 3n‑2
    L = 1 << ((corr_len - 1).bit_length())  # next power of two

    beta_start = 0.5
    beta_end = 200.0
    step_start = 0.5
    step_end = 0.05

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        t = elapsed / total                     # ∈[0,1]
        beta = beta_start * (beta_end / beta_start) ** t
        step_factor = step_start * (1.0 - t) + step_end * t

        # FFT of a (zero‑padded to length L)
        A_fft = np.fft.rfft(a, n=L)

        # linear convolution a*a (only first 2n‑1 entries are needed)
        conv = np.fft.irfft(A_fft * A_fft, n=L)[:conv_len]
        max_val = conv.max()

        # soft‑max weights (stable numerics)
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # FFT of weights
        W_fft = np.fft.rfft(w, n=L)

        # correlation w with a : corr[k] = Σ_i w_i a_{i+k}
        corr_full = np.fft.irfft(np.conj(W_fft) * A_fft, n=L)

        # gradient (see analysis)
        grad = 2.0 * beta * corr_full[n - 1 :: -1][:n]

        # step size proportional to target_sum / ‖grad‖
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            lr = step_factor * target_sum / (norm + 1e-12)
        else:
            lr = 0.0

        a = a - lr * grad
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

    return best, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    Uses a pool of diverse seeds, refines each with smoothed‑gradient
    descent, applies a final deterministic mass‑transfer polishing and a
    few re‑orderings, then returns the best sequence found.
    """
    start_time = time.time()
    margin = 0.5                     # safety margin before hard timeout
    deadline = start_time + budget_s - margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a modest set of seeds
    # ------------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.5, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.001, 0.02, 7):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(8):
        seeds.append(_random_simplex(n, target_sum, rng))

    # include previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # small jitter on each seed to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ------------------------------------------------------------------
    # Main optimisation loop (run gradient descent on different seeds)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate a chunk for the final polishing & reordering
    final_reserved = 40.0
    while time.time() < deadline - final_reserved:
        # time left for a single run (capped at ~12 s)
        remaining = deadline - time.time() - final_reserved
        run_time = min(12.0, remaining)
        if run_time < 1.0:
            break

        # pick a seed – occasionally take the current best to refine it
        if best_seq is not None and rng.random() < 0.3:
            seed = best_seq.copy()
        else:
            seed = rng.choice(seeds)

        seq_opt, score_opt = _smooth_gradient_descent(seed,
                                                      target_sum,
                                                      time.time() + run_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] New best after GD: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer polishing (few seconds if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        mt_deadline = time.time() + min(10.0, deadline - time.time() - 5.0)
        seq_mt, sc_mt = _deterministic_mass_transfer(best_seq,
                                                     mt_deadline,
                                                     target_sum,
                                                     start_temp=0.3,
                                                     end_temp=1e-6,
                                                     low_frac=0.08)
        if sc_mt < best_score:
            best_score = sc_mt
            best_seq = seq_mt.copy()
            print(f"[{time.time():.2f}] After MT polish: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simple reordering tricks (often lower the maximal convolution)
    # ------------------------------------------------------------------
    if best_seq is not None:
        candidates = [best_seq,
                      _alternating_order(best_seq),
                      _even_spread_order(best_seq)]
        # pick the ordering with smallest max‑convolution (not full score)
        best_seq = min(candidates, key=_max_convolution_fft)
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] After reordering: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fall‑back (should never happen)
        best_seq = np.full(n, target_sum / n, dtype=float)

    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store for the next run and report final evaluation
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
