def smooth_sequence(a, sigma_frac=0.01):
    sigma = max(1.0, sigma_frac * len(a))
    # length for kernel about 6 sigma
    L = int(6*sigma)+1
    x = np.arange(-L//2, L//2+1)
    kernel = np.exp(-0.5*(x/sigma)**2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    return a_smooth

"""
Enhanced multi‑stage optimizer for minimizing the maximal entry of a
self‑convolution.  The algorithm expands on the previous pipeline by:

* richer parametric warm‑start (power‑law, exponentially‑tilted, Gaussian,
  plus a rapid random sampling of (p,q));
* repeated short bursts of projected Adam (soft‑max) with aggressive β‑schedule;
* a more aggressive global peak‑mass‑transfer (larger donor/receiver fractions);
* occasional smoothing of the sequence with a Gaussian kernel;
* interleaved pairwise optimal transfers and random moves;
* final high‑β polishing.

All phases respect the provided time budget (default 1000 s) and keep the
vector on the simplex Σa = √(2 n) with non‑negative entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value 2·n·max(b)/(sum(a)^2).  If sum(a)=√(2n) the score equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (aggressive version)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.02,
                               receiver_frac: float = 0.02,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """Move a fraction of the mass of the strongest contributors to a single weak receiver."""
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # stop if no significant gap
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donor scores = a_i * w_i (contribution to the current peak)
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        # candidate receivers = indices with smallest partner weight
        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]
            diff = donor_mass * shift_r - conv_donors
            mask_pos = diff > 0
            if np.any(mask_pos):
                # α ≤ conv_donors / (donor_mass * shift_r) for those components
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            # predicted reduction of the peak (linearised model)
            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # apply the move
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} new_score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Pairwise optimal mass transfer (local refinement)

def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise transfer that reduces the current peak."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]
                d = 2.0 * (shift_j - shift_i)          # effect of moving mass i→j
                d_k = d[k]
                if d_k >= 0.0:
                    continue
                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i
                if delta_max <= 1e-12:
                    continue
                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Simple Gaussian smoothing (used as a heuristic)
# ----------------------------------------------------------------------


def _smooth_sequence(a: np.ndarray, sigma_frac: float = 0.01) -> np.ndarray:
    """Convolve with a normalized Gaussian kernel (σ proportional to length)."""
    n = a.shape[0]
    sigma = max(1.0, sigma_frac * n)
    radius = int(3 * sigma)
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(a, kernel, mode='same')
    return smoothed

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    maximum self‑convolution entry.  The routine respects the given
    time budget (default 1000 s) and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # problem size -------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 500:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → score = max convolution

    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Load previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()
            print(f"[{time.time():.2f}] Loaded previous best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1.  Dense parametric warm‑start (power‑law, exp‑power, Gaussian)
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.30, 0.85, 21)
    q_grid = np.linspace(0.0, 6.0, 21)

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    sigma_grid = np.linspace(0.1 * n, 0.5 * n, 7)
    for sigma in sigma_grid:
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # ------------------------------------------------------------------
    # 1b. Random sampling of (p,q) for a few seconds
    # ------------------------------------------------------------------
    random_param_deadline = start_time + min(30.0, budget_s * 0.03)
    while time.time() < random_param_deadline:
        p = rng.uniform(0.30, 0.85)
        q = rng.uniform(0.0, 6.0)
        seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()
            print(f"[{time.time():.2f}] Random param (p={p:.3f}, q={q:.3f}) improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Jittered population around current best
    # ------------------------------------------------------------------
    for _ in range(5):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.clip(cand, 0.0, None)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()
            print(f"[{time.time():.2f}] Jitter improved best to {best_score:.6f}")

    print(f"[{time.time():.2f}] Initial best after warm‑start = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Main refinement loop (alternating stochastic / deterministic)
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:
        # ---- 3a. short Adam burst from a jittered start ----
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        start_vec = best_seq + jitter
        start_vec = np.clip(start_vec, 0.0, None)
        start_vec = _project_to_simplex(start_vec, target_sum)

        sub_deadline = time.time() + 8.0
        a_adam, val_adam = _adam_gradient_descent(start_vec,
                                                  sub_deadline,
                                                  target_sum,
                                                  max_beta=2e5,
                                                  step_factor=0.8,
                                                  gamma=0.995)
        if val_adam < best_score:
            best_score, best_seq = val_adam, a_adam.copy()
            print(f"[{time.time():.2f}] Adam burst improved best to {best_score:.6f}")
            continue   # skip to next outer iteration

        # ---- 3b. aggressive global peak mass transfer (short) ----
        sub_deadline = time.time() + 6.0
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      sub_deadline,
                                                      target_sum,
                                                      donor_frac=0.04,
                                                      receiver_frac=0.04,
                                                      min_alpha=1e-7)
        if val_peak < best_score:
            best_score, best_seq = val_peak, a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best to {best_score:.6f}")
            continue

        # ---- 3c. pairwise optimal transfer ----
        sub_deadline = time.time() + 4.0
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      sub_deadline,
                                                      target_sum,
                                                      donor_frac=0.02,
                                                      receiver_frac=0.02)
        if val_pair < best_score:
            best_score, best_seq = val_pair, a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best to {best_score:.6f}")
            continue

        # ---- 3d. Gaussian smoothing + projection ----
        a_smooth = _smooth_sequence(best_seq, sigma_frac=0.015)
        a_smooth = _project_to_simplex(a_smooth, target_sum)
        val_smooth = _score(a_smooth)
        if val_smooth < best_score:
            best_score, best_seq = val_smooth, a_smooth.copy()
            print(f"[{time.time():.2f}] Smoothing improved best to {best_score:.6f}")
            continue

        # ---- 3e. random global move as a last resort ----
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best to {best_score:.6f}")
            continue

        # if none of the above produced improvement, break to avoid endless loop
        break

    # ------------------------------------------------------------------
    # 4.  Final high‑β polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=5e5,
                                                step_factor=0.6,
                                                gamma=0.999)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final polish improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
