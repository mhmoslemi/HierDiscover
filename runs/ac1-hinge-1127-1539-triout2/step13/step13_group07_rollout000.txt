def _subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    eps_abs = 1e-9
    step_scale = 1.0   # initial step factor relative to target sum
    while time.time() < deadline:
        conv = _conv_fft(a)
        max_b = conv.max()
        # update best
        if max_b < best_score - 1e-12:
            best_score = max_b
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgradient it={it} best={best_score:.6f}")
        # compute active set indicator
        # tolerance: maybe relative: max_b - 1e-12
        # Use tolerance absolute: 1e-8?
        tol = max_b * 1e-9  # small relative tolerance
        active = (conv >= max_b - tol).astype(np.float64)
        # if only one active, maybe gradient is simply 2 * a_rev reversed? but that's fine.
        grad = _gradient_from_weights(a, active)  # returns 2*conv_rev
        # compute step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        # Step size schedule: step = step_scale * target_sum / (grad_norm * sqrt(it+1))
        step = step_scale * target_sum / (grad_norm * np.sqrt(it + 1))

        a = a_new

Define _gradient_from_weights:

def _gradient_from_weights(a, w):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=size)
    Wf = np.fft.rfft(w, n=size)
    conv_rev = np.fft.irfft(Af * Wf, n=size)[:2*n-1]
    grad = 2.0 * conv_rev[n-1 : n-1+n]
    return grad

"""
Step‑function optimizer for the max‑self‑convolution problem.

The algorithm consists of:

1. **Parametric sweep** – evaluates several analytic families
   (power law, exponentially‑tailed power law, sine, inverse‑sqrt) on a
   grid of parameters and keeps the best candidate.

2. **Projected sub‑gradient descent on the max‑convolution** – iteratively
   reduces the current peak by moving along a sub‑gradient of the
   max‑convolution (computed via an active‑set indicator) and projecting
   onto the simplex Σa_i = √(2 n).

3. **Deterministic mass‑transfer (DMT)** – a greedy exact routine that
   pushes the current maximum down to the second‑largest value.

4. **Final refinement** – a short run of the DMT routine on the remaining
   budget.

All phases respect the total time budget (default 1000 s) and the
result is trimmed of trailing negligible entries before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2·n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the quick pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _inv_sqrt_sequence(n: int, exponent: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-exponent), using exponent ≈ 0.5 for inverse‑sqrt."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-exponent))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index to a
    “receiver’’ index so that the current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # find second‑largest entry
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient of weighted sum of convolution entries (used for sub‑gradient)
# ----------------------------------------------------------------------


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    w must have length 2·n‑1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=size)
    Wf = np.fft.rfft(w, n=size)
    conv_rev = np.fft.irfft(Af * Wf, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent_max(a0: np.ndarray,
                             target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the max‑convolution objective.
    At each iteration we form an active‑set indicator for the current
    maximal entries and move a step opposite to its gradient, projecting
    back onto the simplex Σ a_i = target_sum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0
    rng = np.random.default_rng()
    step_scale = 0.7  # initial relative step size

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()

        # keep best seen
        if max_val < best_score - 1e-12:
            best_score = max_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_score:.6f}")

        # active set (indices where the convolution is (almost) maximal)
        tol = max_val * 1e-7 + 1e-12
        active = (conv >= max_val - tol).astype(np.float64)

        # gradient of the sum over the active set
        grad = _gradient_from_weights(a, active)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # diminishing step size
        step = step_scale * target_sum / (grad_norm * math.sqrt(it + 1))

        a_candidate = a - step * grad
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (it + 1), size=n)
            a_candidate = _project_to_simplex(a_candidate + jitter, target_sum)

        a = a_candidate
        it += 1

    return best_a, best_score
