    n = a.shape[0]
    size = 1 << ((n + (2*n-1) - 1).bit_length())

"""
Advanced step‑function optimiser for the “max‑self‑convolution” task.

The routine proceeds in several stages, each respecting a global
time‑budget (default 1000 s):

1. **Fast parametric sweep** – a broad grid of power‑law, exponentially
   tilted, and two‑exponential mixtures yields a decent starting point.
2. **Exact sub‑gradient descent** – the true gradient of the max‑convolution
   (obtained via FFT) is used with a projected (simplex) step; this
   typically pushes the objective below the previous best.
3. **Deterministic mass‑transfer (DMT2)** – a proven exact refinement that
   never raises the objective; runs for the bulk of the remaining time.
4. **Greedy pairwise transfer** – larger donor/receiver pools to escape
   shallow minima.
5. **Simulated‑annealing polishing** – stochastic moves with a cooling
   schedule for a final touch.
6. **Post‑processing** – trailing near‑zero entries are trimmed and the
   result is returned as a plain list.

All helper functions are top‑level (no closures) and the code uses only
NumPy and the provided ``evaluate_sequence`` for final verification.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective value returned by ``evaluate_sequence``."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # the caller normally keeps sum = sqrt(2·n) → score == max_b
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s ,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x_i ≥ 0, Σx_i = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _mixture_exp_sequence(
    n: int, lam1: float, lam2: float, alpha: float, target_sum: float
) -> np.ndarray:
    """Two‑exponential mixture a_i = α·exp(-λ₁·i) + (1‑α)·exp(-λ₂·i)."""
    i = np.arange(n, dtype=np.float64)
    a = alpha * np.exp(-lam1 * i / n) + (1.0 - alpha) * np.exp(-lam2 * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _max_convolution_subgradient(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv(a,a))``.
    The gradient is 2·∑_{k∈K} shift_rev(a,k) where K are indices attaining the maximum.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()
    # mask of maximal entries (allow a tiny tolerance)
    mask = conv >= max_val - eps
    p = mask.astype(np.float64)           # length 2·n‑1
    # compute g = 2·sum_{k∈K} shift_rev(a,k) via FFT:
    # g_i = 2·∑_j a_j·p_{i+j}   (see derivation in analysis)
    revp = p[::-1]                       # reversed mask
    size = _next_pow_two(n + (2 * n - 1) - 1)   # length of a + revp - 1 = 3n‑2
    af = np.fft.rfft(a, n=size)
    pf = np.fft.rfft(revp, n=size)
    conv2 = np.fft.irfft(af * pf, n=size)[: 3 * n - 2]
    idx = (2 * n - 2) - np.arange(n)          # pull out the needed slice
    g = 2.0 * conv2[idx]
    return g


def _subgradient_descent(
    a0: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact max‑convolution.
    Uses an adaptive step size based on the gradient norm.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    it = 0
    while time.time() < deadline:
        g = _max_convolution_subgradient(a)
        norm = np.linalg.norm(g)
        if norm == 0.0:
            break
        # adaptive step: we want a move of order ~0.5·target_sum in L2 norm
        step = 0.5 * target_sum / (norm + 1e-12)
        a_new = a - step * g
        a_new = _project_to_simplex(a_new, target_sum)
        a = a_new
        cur_val = _score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")
        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the “DMT2” refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
    delta_factor: float = 0.95,
) -> tuple[np.ndarray, float]:
    """
    Deterministic mass‑transfer that never raises the objective.
    ``delta_factor`` controls how aggressively we move mass (≤1.0).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * delta_factor, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _pairwise_optimal_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_frac: float = 0.03,
    receiver_frac: float = 0.03,
    min_improve: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """Greedy pairwise transfer similar to the original DMT2 but with larger pools."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest entries
        receiver_idx = np.argsort(a)[:nr]        # smallest entries

        # cache shifted views for all candidates (lightweight view, no copy)
        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)          # effect per unit δ
                d_k = d[k]
                if d_k >= 0.0:
                    continue            # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # keep base up‑to‑date (cheap slice assignment)
        base[pad_len + i] = a[i]
        base[pad_len + j] = a[j]

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} score={best_score:.6f}")

    return a, best_score

def _simulated_annealing_refinement(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """Mass‑move SA that can escape shallow local minima."""
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    rng = np.random.default_rng()
    start_time = time.time()
    T0 = cur_max * 0.02 + 1e-4          # initial temperature scale
    iter_cnt = 0
    refresh_interval = 500
    donor_frac = 0.02
    recv_frac = 0.02

    donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
    receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

    while time.time() < deadline:
        if iter_cnt % refresh_interval == 0:
            donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
            receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        shift_i = base[i:i + 2 * n - 1]
        shift_j = base[j:j + 2 * n - 1]
        diff = shift_j - shift_i

        new_conv = conv + delta * diff
        new_max = new_conv.max()

        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-9)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            base[pad_len + i] = a[i]
            base[pad_len + j] = a[j]

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={iter_cnt} best={best_score:.6f}")

        if iter_cnt % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            base = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])

        iter_cnt += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the maximal
    self‑convolution.  The routine respects the time budget and returns the
    best sequence found (as a plain Python list).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n) → score = max(conv)

    # ------------------------------------------------------------------
    # Phase 0 – quick parametric / random family sweep
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # start from a stored construction if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        best_seq = resized
        best_score = val

    # deterministic grids (power‑law & exponential tilt)
    p_grid = np.linspace(0.2, 1.2, 21)
    q_grid = np.linspace(0.0, 8.0, 21)

    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_seq = seq
            best_score = val

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_seq = seq
                best_score = val

    # random two‑exponential mixtures (limited budget)
    time_for_random_families = min(5.0, (deadline - time.time()) * 0.10)
    while time.time() < start_time + time_for_random_families:
        lam1 = rng.uniform(0.0, 6.0)
        lam2 = rng.uniform(0.0, 6.0)
        alpha = rng.uniform(0.0, 1.0)
        seq = _mixture_exp_sequence(n, lam1, lam2, alpha, target_sum)
        val = _score(seq)
        if val < best_score:
            best_seq = seq
            best_score = val

    # some jittered copies of the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_seq = seq_j
            best_score = val

    print(f"[{time.time():.2f}] After parametric sweep best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exact sub‑gradient descent (fast, convex‑like)
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.20
    if time.time() < subgrad_deadline - 2.0:
        a_sg, val_sg = _subgradient_descent(best_seq, target_sum, subgrad_deadline)
        if val_sg < best_score:
            best_seq = a_sg
            best_score = val_sg
            print(f"[{time.time():.2f}] Sub‑grad improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Deterministic DMT2 refinement (core)
    # ------------------------------------------------------------------
    dmt_deadline = time.time() + (deadline - time.time()) * 0.35
    if time.time() < dmt_deadline - 4.0:
        a_dmt, val_dmt = _deterministic_mass_transfer_exact2(
            best_seq, dmt_deadline, target_sum, delta_factor=0.95
        )
        if val_dmt < best_score:
            best_seq = a_dmt
            best_score = val_dmt
            print(f"[{time.time():.2f}] DMT2 improved best={best_score:.6f}")

        # occasional short random moves to avoid getting stuck
        a_rand, val_rand = _global_random_mass_transfer(best_seq, dmt_deadline, target_sum)
        if val_rand < best_score:
            best_seq = a_rand
            best_score = val_rand
            print(f"[{time.time():.2f}] Random move improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Greedy pairwise transfer
    # ------------------------------------------------------------------
    pair_deadline = time.time() + (deadline - time.time()) * 0.15
    if time.time() < pair_deadline - 2.0:
        a_pair, val_pair = _pairwise_optimal_transfer(
            best_seq, pair_deadline, target_sum, donor_frac=0.04, receiver_frac=0.04
        )
        if val_pair < best_score:
            best_seq = a_pair
            best_score = val_pair
            print(f"[{time.time():.2f}] Pairwise improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing polishing (final)
    # ------------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.20
    if time.time() < sa_deadline - 1.0:
        a_sa, val_sa = _simulated_annealing_refinement(best_seq, sa_deadline, target_sum)
        if val_sa < best_score:
            best_seq = a_sa
            best_score = val_sa
            print(f"[{time.time():.2f}] SA improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final verification (harmless if evaluate_sequence is undefined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the sequence for potential future runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
