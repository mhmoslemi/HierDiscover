    best_score = _score(a)
    best_a = a.copy()
    rng = np.random.default_rng()
    n = len(a)

"""
Propose a step‑function (non‑negative heights) that minimizes the maximal
self‑convolution under the usual sum normalisation Σa_i = √(2·n).
The algorithm follows a multi‑phase scheme:

1. **Rich parametric pre‑search** – evaluate many analytic families
   (power law, exponential tail, sine, sine^p, raised‑cosine, logistic,
   Gaussian) on a problem size n that fits the time budget.

2. **Soft‑max Adam optimisation** – projected Adam on a smoothed max
   (soft‑max) with several restarts; the smoothing parameter β is
   increased to approach the true max.

3. **Exact sub‑gradient descent** – using the true max‑convolution
   gradient.

4. **Deterministic mass‑transfer refinement** – greedy transfers of mass
   between “donor” and “receiver” indices that directly lower the max
   entry.

5. **Penalty‑descent (feasibility) phase** – repeatedly try to push the
   maximal convolution below a slightly reduced target; each attempt
   runs a projected sub‑gradient on the penalty (max‑conv − target)_+.

6. **Final high‑β Adam polishing**.

All stages respect the simplex constraint (non‑negative, fixed ℓ₁‑norm)
by Euclidean projection.  The routine adapts n to the available time
(≈12 k–20 k entries) and trims trailing near‑zero entries before
returning the sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of arbitrary vectors x and y using FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used in the cheap pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _sin_power_sequence(n: int, power: float, target_sum: float) -> np.ndarray:
    """sin(iπ/(n+1)) raised to a power."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** power
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    """Centered Gaussian (clipped at the ends)."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _raised_cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """A raised‑cosine bump, zero at the ends."""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(2.0 * np.pi * x / (n - 1))
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


def _logistic_sequence(n: int, scale: float, target_sum: float) -> np.ndarray:
    """Logistic (sigmoid) shape, symmetric."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((x - centre) / scale))
    a -= a.min()                         # shift to start at zero
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring utilities
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Objective: maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient used by Adam
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max approximation of max(conv) w.r.t. a.
    Returns (grad, conv_array) where grad has the same length as a.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)

    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(conv)
    exps = np.exp(beta * (conv - max_b))
    p = exps / np.sum(exps)                     # soft‑max distribution

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]

    grad = 2.0 * conv_rev[n - 1: n - 1 + n]     # slice the central n entries
    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser with projection onto the simplex."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to exit tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """Greedy sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy transfer of mass to equalise the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        # use the first maximal index (any works)
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                      # values that pair with idx
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose a receiver with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Simple stochastic mass‑transfer to escape plateaus."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty‑descent: try to enforce a target maximal convolution bound
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    """
    Reduce max(conv) below target_max by minimising the penalty
    (max(conv) – target_max)_+.  Gradient follows from the penalty term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty vector (only entries above target_max)
        p = conv - target_max
        p[p < 0] = 0.0

        # gradient = 2 * (p * a_rev)_{central n entries}
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)

        it += 1

    return best_a, best_max

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution.  The routine contains a rich multi‑phase pipeline
    and adapts the problem size to the available time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # Choose problem size (larger n gives more flexibility but is slower)
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)               # Σa_i = √(2·n) ⇒ score = max‑conv

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Phase 0 – parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # (a) pure power‑law
    for p in np.linspace(0.2, 1.0, 21):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (b) power‑law with exponential tail
    for p in np.linspace(0.2, 1.0, 13):
        for q in np.linspace(0.0, 8.0, 13):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # (c) pure sine
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # (d) sin^p families
    for power in [0.5, 1.0, 1.5, 2.0, 3.0]:
        seq = _sin_power_sequence(n, power, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (e) raised cosine
    seq = _raised_cosine_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # (f) logistic shapes
    for scale in [n / 30.0, n / 20.0, n / 15.0]:
        seq = _logistic_sequence(n, scale, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (g) Gaussian bumps with varying sigma
    for sigma in np.linspace(n / 15.0, n / 5.0, 5):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # (h) jittered refinements of the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] Pre‑search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam (soft‑max) with restarts
    # ------------------------------------------------------------------
    phase_a_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < phase_a_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         phase_a_deadline)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam phase improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent
    # ------------------------------------------------------------------
    phase_b_deadline = time.time() + 0.20 * (deadline - time.time())
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         phase_b_deadline)
    if val_sg < best_score:
        best_score, best_seq = val_sg, a_sg.copy()
        print(f"[{time.time():.2f}] Sub‑gradient improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer + occasional random move
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # deterministic mass‑transfer (short sub‑deadline)
        sub_dead = min(deadline, time.time() + 25.0)
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             sub_dead,
                                                             target_sum)
        if val_ref < best_score - 1e-6:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT2 improved best={best_score:.6f}")

        # global random perturbation (short sub‑deadline)
        sub_dead = min(deadline, time.time() + 10.0)
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        sub_dead,
                                                        target_sum)
        if val_rand < best_score - 1e-6:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best={best_score:.6f}")

        if time.time() > deadline - 8.0:
            break

    # ------------------------------------------------------------------
    # Phase D – Penalty descent (attempt to lower the bound)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # try to shave a small slice off the current best
        target = best_score - 0.0015   # modest decrement
        if target <= 1.0:
            break

        sub_dead = min(deadline, time.time() + 20.0)
        a_pen, max_pen = _penalty_lower_max(best_seq,
                                            target_sum,
                                            sub_dead,
                                            target)
        if max_pen < best_score - 1e-5:
            best_score, best_seq = max_pen, a_pen.copy()
            print(f"[{time.time():.2f}] Penalty descent achieved {best_score:.6f}")
        else:
            # if we cannot meet the stricter target, stop trying finer reductions
            break

    # ------------------------------------------------------------------
    # Phase E – Final high‑β Adam polishing
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish gives {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing – trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final check (evaluate_sequence is supplied by the environment)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
