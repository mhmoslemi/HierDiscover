"""

Algorithm
---------
1. **Small‑scale LP “tightening”** (≈30 % of total budget)
   * Work with a modest size `n_small` (≈2000) for tractability.
   * Initialise the sequence with a few simple families (power‑law,

   * For the current sequence `a` compute `M = max(a∗a)`.

                     g_j ≥ 0

     peak `M` but increase the total sum.

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b via FFT."""
    n = a.size
    m = b.size
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Simple parametric families (used for a quick seed)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a

                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer with Metropolis acceptance.  Each move transfers
    a random amount of mass from a high‑value index to a low‑value index.
    Temperature is exponentially cooled.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # initial temperature (fraction of the current objective)
    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.02                      # donor / receiver selected from top/bottom 2 %
    top_k = max(1, int(top_frac * n))
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.3
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _max_convolution_fft(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP improvement for a modest‑size sequence (optional)
# ----------------------------------------------------------------------


def _lp_improve_once(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    One iteration of the linear‑programming tightening step.
    Solves
        maximise   Σ g_j
        subject to a∗g ≤ M,   g ≥ 0
    where M = max(a∗a).  The solution g is rescaled to the target simplex,
    and a 1‑D line search yields the best mixing factor t.
    """
    if not _LP_AVAILABLE:
        # fallback: no LP capability – simply return the input unchanged
        return a

    n = a.shape[0]
    # current peak
    conv_aa = _conv_fft(a)
    M = conv_aa.max()

    # Build the Toeplitz matrix that yields the linear convolution a∗g
    c = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    r = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    T = toeplitz(c, r)          # shape (2n‑1, n)

    # Linear program
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:          # pragma: no cover
        return a

    if g.value is None:
        return a

    g_opt = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    sum_g = g_opt.sum()
    if sum_g < 1e-12:
        return a

    # Rescale to the required simplex sum
    g_rescaled = g_opt * (target_sum / sum_g)

    # Convolutions needed for the quadratic mixing formula
    conv_ag = _conv_full_fft(a, g_rescaled)
    conv_gg = _conv_fft(g_rescaled)

    # 1‑D line search for the optimal mixing coefficient t∈[0,1]
    best_t = 0.0
    best_val = conv_aa.max()
    # coarse grid
    for t in np.linspace(0, 1, 51):
        mixed = ((1 - t) ** 2) * conv_aa + 2 * t * (1 - t) * conv_ag + (t ** 2) * conv_gg
        val = mixed.max()
        if val < best_val:
            best_val = val
            best_t = t
    # refined ternary search in a small neighbourhood
    left = max(0.0, best_t - 0.02)
    right = min(1.0, best_t + 0.02)
    for _ in range(30):
        t1 = left + (right - left) / 3.0
        t2 = right - (right - left) / 3.0
        mixed1 = ((1 - t1) ** 2) * conv_aa + 2 * t1 * (1 - t1) * conv_ag + (t1 ** 2) * conv_gg
        mixed2 = ((1 - t2) ** 2) * conv_aa + 2 * t2 * (1 - t2) * conv_ag + (t2 ** 2) * conv_gg
        val1 = mixed1.max()
        val2 = mixed2.max()
        if val1 < val2:
            right = t2
            if val1 < best_val:
                best_val = val1
                best_t = t1
        else:
            left = t1
            if val2 < best_val:
                best_val = val2
                best_t = t2

    # Mix and re‑project onto the simplex
    a_new = (1 - best_t) * a + best_t * g_rescaled
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal self‑convolution.
    The routine respects the time budget and returns the best list of heights found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Small LP stage (≈30 % of total budget)
    # --------------------------------------------------------------
    n_small = 2000
    target_small = np.sqrt(2.0 * n_small)

    # start from a few simple families
    best_small = _power_law_sequence(n_small, 0.55, target_small)
    best_score = _score(best_small)

    # also try the pure sine family
    seq = _sin_sequence(n_small, target_small)
    val = _score(seq)
    if val < best_score:
        best_score, best_small = val, seq.copy()

    # reuse historic best if present
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n_small:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n_small)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_small)
        val = _score(resized)
        if val < best_score:
            best_score, best_small = val, resized.copy()

    print(f"[{time.time():.2f}] Small‑stage init best score = {best_score:.6f}")

    # LP tightening loop
    small_stage_deadline = start_time + (deadline - start_time) * 0.30
    lp_iter = 0
    while time.time() < small_stage_deadline:
        cand = _lp_improve_once(best_small, target_small)
        cand_score = _score(cand)
        if cand_score < best_score - 1e-7:
            best_score, best_small = cand_score, cand.copy()
            print(f"[{time.time():.2f}] LP iter {lp_iter:02d} improved score = {best_score:.6f}")
        else:
            # stop early if no improvement for several iterations
            lp_iter += 1
            if lp_iter >= 15:
                break
        lp_iter += 1

    # --------------------------------------------------------------
    # Upscale to the final problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n_big = 12000
    elif budget_s >= 400:
        n_big = 8000
    else:
        n_big = 5000
    target_big = np.sqrt(2.0 * n_big)

    old_x = np.linspace(0.0, 1.0, n_small)
    new_x = np.linspace(0.0, 1.0, n_big)
    seq_big = np.interp(new_x, old_x, best_small)
    seq_big = _project_to_simplex(seq_big, target_big)
    best_seq = seq_big.copy()
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After upscale best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Deterministic mass‑transfer (quick refinement)
    # --------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.15
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt_deadline,
                                                               target_big)

    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing (longer exploration)
    # --------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.45
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_big,
                                                sa_deadline)

    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Deterministic mass‑transfer (tightening)
    # --------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.20
    best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                               mt2_deadline,
                                                               target_big)

    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Final Adam polishing
    # --------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      deadline,
                                                      target_big,
                                                      max_beta=1e6,
                                                      step_factor=0.6,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional sanity check with the competition evaluator
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    # expose the result for potential reuse by a later run
    globals()["height_sequence_1"] = final_seq
    return final_seq
