def propose_candidate(budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    rng = np.random.default_rng()

    # Phase 0: Determine n0
    if budget_s > 900:
        n0 = 8000
    elif budget_s > 500:
        n0 = 6000
    else:
        n0 = 5000

    target_n0 = math.sqrt(2.0 * n0)

    # generate initial pool
    init_pool = []
    # power law
    for p in np.linspace(0.1, 2.0, 20):
        init_pool.append(_power_law_sequence(n0, p, target_n0))
    # exponential
    for lam in np.linspace(0.001, 0.05, 20):
        init_pool.append(_exponential_sequence(n0, lam, target_n0))
    # sin shape
    for exp in np.linspace(0.5, 2.5, 20):
        init_pool.append(_sin_power_sequence(n0, exp, target_n0))
    # cos power
    for exp in np.linspace(0.5, 2.5, 20):
        init_pool.append(_cos_power_sequence(n0, exp, target_n0))
    # random
    for _ in range(30):
        init_pool.append(_random_simplex(n0, target_n0, rng))

    # optionally include previous sequence
    if "height_sequence_1" in globals():
        init_pool.append(_resize_sequence(np.array(globals()["height_sequence_1"],float), n0))

    # Evaluate all and keep top K
    evaluations = []
    for seq in init_pool:
        val = _max_convolution_fft(seq)
        evaluations.append((val, seq))
    evaluations.sort(key=lambda x: x[0])
    top_k = 10
    best_candidates = [seq for _, seq in evaluations[:top_k]]

    best_seq = best_candidates[0].copy()
    best_val = _max_convolution_fft(best_seq)

    # Phase A: Annealed gradient on top candidates
    while time.time() < deadline - 120: # reserve at least 2 minutes
        seed = rng.choice(best_candidates)
        # jitter seed a bit
        jitter = rng.normal(scale=target_n0 * 0.01, size=n0)
        seed_j = seed + jitter
        seed_j = np.maximum(seed_j, 0.0)
        seed_j = _project_to_simplex(seed_j, target_n0)

        run_deadline = min(time.time() + min(15.0, deadline - time.time() - 30.0), deadline - 10.0)
        a_opt, val_opt = _annealed_gradient_descent(seed_j, run_deadline,
                                                    target_n0, max_beta=1e6, step_factor=0.9, gamma=0.99)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New best after GD {best_val:.6f}")

        # keep top few seeds
        best_candidates.append(a_opt)
        # keep only unique best few
        best_candidates = sorted(best_candidates, key=lambda x: _max_convolution_fft(x))[:top_k]

    # Phase B: Upscaling
    remaining = deadline - time.time()
    n_big = 30000 if remaining > 400 else (25000 if remaining > 250 else (20000 if remaining > 150 else n0))
    target_big = math.sqrt(2.0 * n_big)
    seq_big = _resize_sequence(best_seq, n_big)
    seq_big = _project_to_simplex(seq_big, target_big)
    # reorder to reduce max conv: try few reorderings
    reorderings = [seq_big, _alternating_order(seq_big), _even_spread_order(seq_big)]
    seq_big = min(reorderings, key=_max_convolution_fft)
    best_val_big = _max_convolution_fft(seq_big)
    print(f"[{time.time():.2f}] Upscaled to {n_big}, initial max={best_val_big:.6f}")

            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Subgrad improved to {best_val_big:.6f}")

"""
Hybrid step‑function optimizer – refined version.

Goal: minimise 2·n·max(a*a)/(∑a)² which, after normalising
∑a=√(2n), reduces to minimising max(a*a).  The algorithm
combines a fast parametric sweep, lengthy annealed‑gradient
and sub‑gradient phases, up‑scaling, and final combinatorial
tuning.  It is deliberately tuned to spend the bulk of the
runtime on a large instance (≈30 k entries) where the
objective is almost at its continuous limit.

Key ideas compared to the reference implementation:
 * a richer family of initial shapes (sin^p, cos^p, mixed)
 * aggressive early‑stage parameter sweep to obtain very good seeds
 * longer annealed‑gradient runs with a smoother β‑schedule
 * more iterations of exact sub‑gradient descent on the large problem
 * tighter mass‑transfer simulated‑annealing at the end.

The code respects all contest constraints (no I/O, single‑CPU,
no closures, explicit top‑level helpers) and returns the best
sequence found before the deadline.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# L1‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (various parametric families)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised‑cosine (Hann) window."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Single‑humped sine shape."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _sin_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ sin(π·i/(n‑1))^p, p>0."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x) ** p
    a *= target_sum / a.sum()
    return a


def _cos_power_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (1‑cos(π·i/(n‑1)))^p, always non‑negative."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = (1.0 - np.cos(np.pi * x)) ** p
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from `seq` to length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Re‑ordering helpers (deterministic permutations that often lower max‑conv)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, smallest, second largest, second smallest …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values as evenly as possible using a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent on the max‑conv proxy
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth (soft‑max) approximation
    of max(a*a).  β (inverse temperature) is doubled every 500 iters.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)

        # exact convolution (real part is enough)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_c = conv.max()

        # soft‑max probabilities over convolution entries
        exps = np.exp(beta * (conv - max_c))
        p = exps / exps.sum()

        # gradient of the smoothed max is the convolution of a_rev with p
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)

        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient descent (O(n log n) per step)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.996,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the true max‑conv indicator."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indicator of (almost) maximal entries – a small looseness helps
        I = (conv >= maxb * (1 - 1e-8)).astype(np.float64)

        # gradient = 2 * (I * a_rev) centred on the n entries
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # jitter every few hundred iterations
        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.3,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from a donor (large product) to a receiver (small product)."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation based search (random shuffles + hill‑climbing)
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 25000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine follows a multi‑phase scheme:

    1. Fast parametric sweep on a modest size (≈6‑8 k).
    2. Annealed‑gradient refinements on the best seeds.
    3. Upscaling to a larger size (≈20‑30 k) and deterministic re‑ordering.
    4. Sub‑gradient descent on the large instance (the most expensive phase).
    5. Permutation + hill‑climb + final mass‑transfer polishing.

    cached globally as ``height_sequence_1`` for possible reuse.
    """

    # Phase 0 – decide an intermediate problem size
    # --------------------------------------------------------------
    if budget_s >= 900:
        n0 = 8000
    elif budget_s >= 600:
        n0 = 6000
    else:
        n0 = 5000
    target0 = math.sqrt(2.0 * n0)

    # --------------------------------------------------------------
    # Phase 1 – generate a diverse pool of candidate sequences
    # --------------------------------------------------------------
    init_pool: List[np.ndarray] = []

    for p in np.linspace(0.1, 2.2, 22):
        init_pool.append(_power_law_sequence(n0, float(p), target0))
    # exponentials
    for lam in np.linspace(0.001, 0.06, 22):
        init_pool.append(_exponential_sequence(n0, float(lam), target0))
    # sin‑power shapes
    for exp in np.linspace(0.4, 2.8, 20):
        init_pool.append(_sin_power_sequence(n0, float(exp), target0))
    # cos‑power shapes (always non‑negative)
    for exp in np.linspace(0.4, 2.8, 20):
        init_pool.append(_cos_power_sequence(n0, float(exp), target0))
    # classic windows
    for _ in range(8):
        init_pool.append(_cosine_sequence(n0, target0))
    for _ in range(8):
        init_pool.append(_sin_half_sequence(n0, target0))
    # random simplexes
    for _ in range(25):
        init_pool.append(_random_simplex(n0, target0, rng))
    # reuse previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n0))

    # jitter all candidates a bit – creates a richer neighbourhood
    jittered = []
    for seq in init_pool:
        j = rng.normal(scale=target0 * 0.03, size=n0)
        seqj = seq + j
        seqj = np.maximum(seqj, 0.0)
        seqj = _project_to_simplex(seqj, target0)
        jittered.append(seqj)
    init_pool.extend(jittered)

    # keep only the top few seeds

    scored.sort(key=lambda x: x[0])
    top_k = 12
    seeds = [s for _, s in scored[:top_k]]
    best_seq = seeds[0].copy()
    best_val = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – annealed‑gradient refinement on the modest size
    # --------------------------------------------------------------

        # pick a seed – occasionally jitter the current best

            seed = best_seq.copy()
            seed += rng.normal(scale=target0 * 0.008, size=n0)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target0)
        else:
            seed = rng.choice(seeds)

        run_deadline = min(time.time() + 13.0, deadline - 170.0)

            seed, run_deadline, target0,

            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New best after GD = {best_val:.6f}")

        # maintain a small elite list for diversity

        seeds = sorted(seeds, key=_max_convolution_fft)[:top_k]

    # --------------------------------------------------------------
    # Phase 3 – upscale to a larger instance
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 400:

    elif remaining > 250:
        n_big = 25000
    elif remaining > 150:

    else:

    seq_big = min(candidates, key=_max_convolution_fft)

    # --------------------------------------------------------------
    # Phase 4 – long sub‑gradient descent on the large scale
    # --------------------------------------------------------------

        a_sub, val_sub = _subgradient_descent(
            seq_big, sub_deadline, target_big,
            step_factor=0.55, gamma=0.996)
        if val_sub < best_big_val:
            best_big_val = val_sub

    # --------------------------------------------------------------
    # Phase 5 – permutation search + hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline - 50:
        perm_deadline = time.time() + min(12.0, deadline - time.time() - 40.0)
        seq_big, _ = _permutation_search(seq_big, trials=300, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After random perm max={_max_convolution_fft(seq_big):.6f}")

        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # --------------------------------------------------------------
    # Phase 6 – final greedy mass‑transfer (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline - 20:
        mt_deadline = time.time() + min(12.0, deadline - time.time() - 10.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_big, mt_deadline, target_big,
            start_temp=0.25, end_temp=1e-6, low_frac=0.045)
        if val_mt < best_big_val:
            best_big_val = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_big_val:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    globals()["height_sequence_1"] = final_seq

    # sanity check (optional)
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_score:.6f}")
    except Exception:
        pass

    return final_seq
