# -*- coding: utf-8 -*-
"""Hybrid optimizer for step function minimization.

We use a multi‑phase search:
1. Build diverse initial sequences (power law, exponential, trigonometric,
   two‑cluster and random) and improve each with a soft‑max annealed gradient.
2. Refine the best candidate by a simulated‑annealing mass‑transfer step.
3. Upscale the solution to a larger length and try several deterministic
   permutations (alternating, even‑spread, golden‑ratio) to minimise the
   autocorrelation peak.
4. Perform a large number of random permutations followed by a local
   hill‑climb that swaps pairs.
5. Apply an exact sub‑gradient descent on the full convolution to polish.
6. Final short mass‑transfer pass and trimming of negligible tail.

The objective after normalising the ℓ¹‑norm is simply the maximal entry of
`np.convolve(a, a)`.  All steps respect non‑negativity and the ℓ¹‑simplex.
"""

import time
import math
import numpy as np
from typing import List

# Helper functions:
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥0, Σx = s}."""
    # Implementation based on Duchi et al. (2008)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution_fft(a: np.ndarray) -> float:
    """Return max of linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# Sequence factories ---------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))   # Hann window
    a *= target_sum / a.sum()
    return a

def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a

def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a

def _two_cluster_sequence(n: int, p: float, alpha: float, target_sum: float) -> np.ndarray:
    """Two half‑intervals each following a power‑law tail.
    alpha fraction of total ℓ¹‑mass is placed in the first half.
    """
    half = n // 2
    first_len = half
    second_len = n - half
    a1 = (np.arange(1, first_len + 1, dtype=np.float64) ** (-p))
    a2 = (np.arange(1, second_len + 1, dtype=np.float64) ** (-p))
    a1 *= (target_sum * alpha) / a1.sum()
    a2 *= (target_sum * (1 - alpha)) / a2.sum()
    return np.concatenate([a1, a2])

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to length new_len."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_len, dtype=np.float64)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# Reordering helpers ---------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = sorted_vals.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out

def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _golden_permutation(seq: np.ndarray) -> np.ndarray:
    """Permutation based on the golden‑ratio step."""
    n = seq.shape[0]
    phi = (1 + 5 ** 0.5) / 2
    step = int(round(phi * n)) % n
    if step == 0:
        step = 1
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Projected gradient descent on a soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_c = conv.max()
        # soft‑max weights
        exps = np.exp(beta * (conv - max_c))
        p = exps / np.sum(exps)

        # gradient of ⟨p, conv⟩ w.r.t a
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        grad_full = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val

def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from a “donor’’ index to a “receiver’’ index,
    accepting worsening moves with a temperature schedule."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # Find lag with maximal convolution value
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Try a limited number of random permutations."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val

def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    """Hill‑climb by swapping random pairs."""
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            # keep swap
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val

"""Hybrid optimizer for the step‑function benchmark.

The objective after ℓ¹‑normalisation simplifies to the maximal entry of
`np.convolve(a, a)`.  The algorithm proceeds in several time‑bounded
phases:

1.  Build a diverse pool of candidate shapes (power‑law, exponential,
    cosine, sine‑half, two‑cluster and random) on a modest size and
    improve each with a soft‑max annealed gradient descent.
2.  Refine the best candidate by a simulated‑annealing mass‑transfer
    move that shifts a small amount of mass from a “donor’’ position to a
    “receiver’’ one.
3.  Upscale the solution to a larger problem size and apply several
    deterministic permutations (alternating, even‑spread, golden‑ratio)
    to lower the convolution peak.
4.  Perform a short random‑permutation search followed by a pairwise
    hill‑climb.
5.  Polished by an exact sub‑gradient descent on the full convolution.
6.  Final brief mass‑transfer pass and trimming of negligible tail
    entries.

All updates respect non‑negativity and the ℓ¹‑simplex constraint.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT)
# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a

def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))      # Hann window
    a *= target_sum / a.sum()
    return a

def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a

def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a

def _two_cluster_sequence(n: int, p: float, alpha: float,
                          target_sum: float) -> np.ndarray:
    """Two halves each follow a power‑law tail, share the total ℓ¹‑mass."""
    half = n // 2
    first = half
    second = n - half
    a1 = (np.arange(1, first + 1, dtype=np.float64) ** (-p))
    a2 = (np.arange(1, second + 1, dtype=np.float64) ** (-p))
    a1 *= (target_sum * alpha) / a1.sum()
    a2 *= (target_sum * (1 - alpha)) / a2.sum()
    return np.concatenate([a1, a2])

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of seq to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len, dtype=np.float64)
    new_x = np.linspace(0.0, 1.0, new_len, dtype=np.float64)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# ----------------------------------------------------------------------
# Deterministic permutations
# ----------------------------------------------------------------------
def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = sorted_vals.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out

def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _golden_permutation(seq: np.ndarray) -> np.ndarray:
    """Permutation based on a golden‑ratio step (ensuring a full cycle)."""
    n = seq.shape[0]
    phi = (1 + 5 ** 0.5) / 2
    step = int(round(phi * n)) % n
    if step == 0:
        step = 1
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    sorted_vals = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Projected gradient descent on a soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_c = conv.max()
        exps = np.exp(beta * (conv - max_c))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=size)
        p_fft = np.fft.rfft(p, n=size)
        grad_full = np.fft.irfft(rev_fft * p_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.45,
                                      end_temp: float = 5e-6,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Move a small amount of mass from a donor index to a receiver index."""
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]

    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:          # occasional global jitter
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
        else:
            a[i], a[j] = a[j], a[i]
        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """Search for a non‑negative step‑function with a low max‑convolution."""
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin
    total_time = deadline - start_time
    rng = np.random.default_rng()

    # ---- Phase 1: broad exploration (≈35 % of total time) ----
    phase1_deadline = start_time + total_time * 0.35
    n_expl = 8000 if budget_s >= 900 else (6000 if budget_s >= 700 else 5000)
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool = []

    for p in np.linspace(0.1, 2.5, 14):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    for lam in np.linspace(0.0003, 0.04, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    for _ in range(6):
        init_pool.append(_cosine_sequence(n_expl, target_expl))
    for _ in range(6):
        init_pool.append(_sin_half_sequence(n_expl, target_expl))
    for alpha in (0.3, 0.5, 0.7):
        for p in (0.5, 1.0, 1.5):
            init_pool.append(_two_cluster_sequence(n_expl, p, alpha, target_expl))
    for _ in range(12):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies of everything
    jittered = []
    for seq in init_pool:
        seq_j = seq + rng.normal(scale=target_expl * 0.025, size=n_expl)
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        jittered.append(seq_j)
    init_pool.extend(jittered)

    best_seq = None
    best_score = np.inf

    while time.time() < phase1_deadline - 2.0:
        if best_seq is not None and rng.random() < 0.6:
            seed = best_seq.copy()
            seed += rng.normal(scale=target_expl * 0.005, size=n_expl)
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 6.0, phase1_deadline - 1.0)
        if run_deadline <= time.time():
            break

        a_opt, val_opt = _annealed_gradient_descent(
            seed, run_deadline, target_expl,
            max_beta=9e5, step_factor=0.9, gamma=0.98)

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best={best_score:.6f}")

    # ---- Phase 2: aggressive mass‑transfer (≈10 % of total time) ----
    phase2_deadline = start_time + total_time * 0.45
    if best_seq is not None and time.time() < phase2_deadline - 1.0:
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            best_seq, phase2_deadline, target_expl,
            start_temp=0.45, end_temp=5e-6, low_frac=0.07)
        if val_mt < best_score:
            best_score = val_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] MT refinement score={best_score:.6f}")

    # ---- Phase 3: upscale to larger problem size (≈15 % of total time) ----
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.7, target_expl)

    phase3_deadline = start_time + total_time * 0.60
    remaining = phase3_deadline - time.time()
    if remaining > 150.0:
        n_final = 25000
    elif remaining > 100.0:
        n_final = 20000
    elif remaining > 70.0:
        n_final = 15000
    elif remaining > 40.0:
        n_final = 12000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try several deterministic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
        _golden_permutation(seq_big)
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elems, start max={_max_convolution_fft(seq_big):.6f}")

    # ---- Phase 4: permutation search and hill‑climb (≈20 % of total time) ----
    phase4_deadline = start_time + total_time * 0.80
    if time.time() < phase4_deadline - 2.0:
        perm_deadline = min(time.time() + 12.0, phase4_deadline - 1.0)
        seq_big, _ = _permutation_search(seq_big, trials=200, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After random perms max={_max_convolution_fft(seq_big):.6f}")

    if time.time() < phase4_deadline - 2.0:
        hill_deadline = min(time.time() + 10.0, phase4_deadline - 1.0)
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_deadline)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # ---- Phase 5: sub‑gradient polishing (≈15 % of total time) ----
    phase5_deadline = start_time + total_time * 0.95
    if time.time() < phase5_deadline - 1.0:
        a_sub, val_sub = _subgradient_descent(
            seq_big, phase5_deadline, target_final,
            step_factor=0.5, gamma=0.995)
        if val_sub < best_score:
            best_score = val_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # ---- Phase 6: final short mass‑transfer pass (remaining time) ----
    if time.time() < deadline - 0.5:
        a_mt, val_mt = _mass_transfer_simulated_annealing(
            seq_big, deadline, target_final,
            start_temp=0.25, end_temp=5e-6, low_frac=0.05)
        if val_mt < best_score:
            best_score = val_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # Trim negligible tail entries
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
