def _smooth_gradient_descent(a0, deadline, target_sum, initial_alpha=2.0, final_alpha=30.0):
   a = a0.copy()
   best_a = a.copy()
   best_score = _score(a)
   # use Adam
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   epsilon = 1e-8
   t = 0
   start = time.time()
   while time.time() < deadline:
       elapsed = time.time() - start
       tau = elapsed / (deadline - start)
       alpha = initial_alpha * (final_alpha/initial_alpha)**tau # exponent
       # compute conv
       conv = _conv_fft(a)  # could reuse existing function that returns conv array
       # compute softmax weights
       # to avoid overflow, subtract max from conv to stabilize
       maxc = conv.max()
       exp_vals = np.exp(alpha * (conv - maxc))
       w = exp_vals / np.sum(exp_vals)
       # compute gradient via correlation
       # compute correlation w with a
       # correlation length n
       # Use FFT: conv_w_a = correlation(w, a)
       # compute convolution of w and a_rev
       size = 1 << ((len(w) + len(a) - 1).bit_length())
       # compute w_f and a_rev_f
       wf = np.fft.rfft(w, n=size)
       af_rev = np.fft.rfft(a[::-1], n=size)
       conv_corr = np.fft.irfft(wf * af_rev, n=size)
       # extract correlation entries for shifts 0..n-1
       corr = conv_corr[len(a)-1 : len(a)-1 + len(a)]
       # compute gradient
       grad = 2.0 * corr - w[2*np.arange(len(a))] * a
       # Adam update
       t += 1
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad * grad)
       m_hat = m / (1 - beta1**t)
       v_hat = v / (1 - beta2**t)
       # step size: adapt based on target_sum and magnitude
       # compute lr = step_factor * target_sum / (np.linalg.norm(m_hat) + 1e-12)
       lr = 0.5 * target_sum / (np.linalg.norm(m_hat) + 1e-12)
       a = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
       # project to simplex
       a = _project_to_simplex(a, target_sum)
       cur_score = _score(a)
       if cur_score < best_score:
           best_score = cur_score
           best_a = a.copy()
   return best_a, best_score

def _smooth_gradient_descent(a0, deadline, target_sum, alpha_start=2.0, alpha_end=30.0):

    a = np.array(a0, copy=True)
    best_a = a.copy()
    best_val = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()
    # pre-allocate size for FFT correlation (size = 1 << ((len(w)+len(a)-1).bit_length()))
    # but w length changes each iter? conv length constant = 2n-1, so w length constant. Set size_corr = 1 << ((2*len(a)-1 + len(a) -1).bit_length()) = 1 << ((3*len(a)-2).bit_length()), maybe 32768 for n=8000.
    n = len(a)
    size_corr = 1 << ((3*n - 2).bit_length())
    while time.time() < deadline:
        # compute conv
        conv = _conv_fft(a)  # length 2n-1
        # compute softmax weights
        maxc = conv.max()
        # linear interpolation for alpha based on progress
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        alpha = alpha_start * (alpha_end / alpha_start) ** tau
        # stable exp
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # correlation via FFT
        # compute FFT of w (size size_corr)
        wf = np.fft.rfft(w, n=size_corr)
        af_rev = np.fft.rfft(a[::-1], n=size_corr)

        corr = corr_full[n-1 : n-1 + n]

        # Adam update
        t += 1
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

"""
Hybrid optimizer for the step‑function problem.

Main ideas
----------
* A diverse seed pool (power‑law, exponential, random simplex, previous best).
* Projected Differential Evolution (DE) to explore the space quickly.
* A smooth‑max gradient descent (Adam) that follows the max‑convolution
  with a temperature schedule (α grows from ~2 to ~30).  This gives a
  smoother landscape early on and focuses on the true maximum later.
* Deterministic mass‑transfer polishing (as in the reference code).
* Simple re‑ordering heuristics (alternating, even‑spread) and a final
  trim of near‑zero tail entries.

The routine respects a 1000‑second budget, automatically allocating time
to each phase.  The best sequence found is stored in the global variable
`height_sequence_1` for warm‑starts in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max (Adam) gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float,
                             alpha_start: float = 2.0,
                             alpha_end: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam optimizer on the smooth‑max of the self‑convolution.
    The temperature alpha grows from `alpha_start` to `alpha_end`,
    turning the smooth‑max into a true max near the end.
    """
    a = a0.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # FFT size for the correlation (w * a_rev)
    size_corr = 1 << ((3 * n - 2).bit_length())
    start = time.time()
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        # ---- compute convolution and soft‑max weights ----
        conv = _conv_fft(a)                 # length 2n‑1
        maxc = conv.max()
        # linearly increase α
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        alpha = alpha_start * (alpha_end / alpha_start) ** tau

        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()       # shape (2n‑1,)

        # ---- gradient via correlation (w with a) ----
        wf = np.fft.rfft(w, n=size_corr)
        af_rev = np.fft.rfft(a[::-1], n=size_corr)
        corr_full = np.fft.irfft(wf * af_rev, n=size_corr)
        # extract the n‑length correlation needed for the gradient
        corr = corr_full[n - 1:n - 1 + n]

        # gradient of the smooth‑max
        grad = 2.0 * corr - w[2 * np.arange(n)] * a

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive step size (scaled to the ℓ¹‑norm target)
        step = 0.5 * target_sum / (np.linalg.norm(m_hat) + 1e-12)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps Σa = target_sum)
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape flat regions
        if it % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=n)
            a = _project_to_simplex(a + jitter, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classical DE with a projection onto the ℓ¹‑simplex after each step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # three distinct mates different from i
            idxs = list(range(len(population)))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            mutant = population[a_idx] + F * (population[b_idx] - population[c_idx])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover (guarantee at least one mutant component)
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # early exit when only a few seconds remain
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
