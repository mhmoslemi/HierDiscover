"""
Hybrid optimiser for the step‑function max‑convolution problem.

The routine builds a diverse set of seeds (power‑law, exponential,
random on the ℓ¹‑simplex, previous best) and runs a projected
Differential‑Evolution (DE) for the majority of the time budget.
After DE the candidate is polished by three complementary local
procedures:

1. deterministic mass‑transfer (greedy reduction of the current
   max‑convolution);
2. smooth‑max gradient descent – the non‑smooth objective
   max(b) is replaced by a log‑sum‑exp surrogate with a gradually
   increasing temperature (γ).  The gradient of the surrogate can
   be computed by a single FFT‑based convolution, allowing fast
   projected gradient steps;
3. a short swap‑based ordering annealing that improves the placement
   of the already optimised values.

Finally a few simple reorderings (alternating, even‑spread) are tried,
the best ordering is kept, trailing near‑zero entries are trimmed,
and the sequence is returned.

All operations respect the 1000 s time limit and keep the sequence
non‑negative with sum √(2 n) (the normalisation used by the scorer).

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two arbitrary 1‑D arrays (FFT)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n.bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv

def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (∑a)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (existing implementation)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑Evolution optimiser (projected on the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (new)
# ----------------------------------------------------------------------


def _grad_smoothed_max(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of   (log Σ exp(gamma * (a*a))) / gamma
    with respect to a (for non‑negative a).  Uses FFT for speed.
    """
    n = a.shape[0]
    conv = _conv_fft(a)                     # length 2n‑1
    max_b = conv.max()
    # stable soft‑max weights
    w = np.exp(gamma * (conv - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # convolution w * a_rev  (length 3n‑2)
    a_rev = a[::-1]
    conv_wa = _conv_fft_two(w, a_rev)
    grad = 2.0 * gamma * conv_wa[n - 1:n - 1 + n]
    return grad


def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        target_sum: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a log‑sum‑exp surrogate of max(b).
    The temperature (gamma) is increased gradually.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    start = time.time()

    # schedule of gamma values – from smooth to almost exact
    gammas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    for gamma in gammas:
        # a modest number of steps per gamma; we stop early if out of time
        for _ in range(200):
            if time.time() >= deadline:
                break
            grad = _grad_smoothed_max(a, gamma)
            norm = np.linalg.norm(grad)
            if norm == 0:
                continue
            # step size – a fraction of the feasible ℓ¹‑norm
            step = 0.45 * target_sum / (norm + 1e-12)
            if rng.random() < 0.02:           # occasional jitter
                step *= 1.0 + 0.3 * rng.normal()
            a_new = a - step * grad
            a_new = _project_to_simplex(a_new, target_sum)

            new_score = _score(a_new)
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
            # accept if improvement; otherwise keep current a (no uphill moves)
            if new_score < _score(a):
                a = a_new
    return best, best_score


# ----------------------------------------------------------------------
# Simple swap‑based ordering annealing (optional)
# ----------------------------------------------------------------------


def _ordering_annealing(a0: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Random pairwise swaps with a tiny Metropolis acceptance to improve ordering.
    Works directly on the max‑convolution (fast FFT evaluation).
    """
    a = a0.copy()
    best = a.copy()
    best_val = _max_convolution_fft(a)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        cur = _max_convolution_fft(a)
        if cur < best_val:
            best_val = cur
            best = a.copy()
            # keep the swap
        else:
            # tiny probability to accept an uphill move
            if rng.random() < 0.001:
                pass
            else:
                # revert
                a[i], a[j] = a[j], a[i]
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser that returns a non‑negative step function with a low
    evaluation score.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a pool of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    for p in np.linspace(0.2, 1.0, 9):
        seeds.append(_power_law_sequence(n, float(p), target_sum))
    for lam in np.linspace(0.001, 0.02, 6):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ------------------------------------------------------------------
    # Initialise DE population
    # ------------------------------------------------------------------
    pop_size = max(25, int(n / 350))   # modest but diverse
    population = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # ------------------------------------------------------------------
    # DE phase (≈55 % of total budget)
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.55 * budget_s
    if de_deadline > deadline - 30.0:
        de_deadline = deadline - 30.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 25.0:
        mt_dead = time.time() + min(20.0, deadline - time.time() - 20.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] After MT – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient descent (short)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        sg_dead = time.time() + min(12.0, deadline - time.time() - 10.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] After sub‑grad – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Smooth‑max gradient descent (new phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sm_dead = time.time() + min(8.0, deadline - time.time() - 6.0)
        seq_sm, score_sm = _smooth_max_descent(
            best_seq, sm_dead, target_sum)
        if score_sm < best_score:
            best_score, best_seq = score_sm, seq_sm.copy()
            print(f"[{time.time():.2f}] After smooth‑max – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Ordering annealing (optional, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        ord_dead = time.time() + min(4.0, deadline - time.time() - 2.0)
        seq_ord, val_ord = _ordering_annealing(best_seq, ord_dead)
        if val_ord < _max_convolution_fft(best_seq):
            best_seq = seq_ord
            best_score = _score(best_seq)
            print(f"[{time.time():.2f}] After ordering anneal – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Try a few simple reorderings that often lower the max‑convolution
    # ------------------------------------------------------------------
    candidates = [best_seq,
                  _alternating_order(best_seq),
                  _even_spread_order(best_seq)]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] Final reordering – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store for the next run and (optionally) report the evaluation
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
