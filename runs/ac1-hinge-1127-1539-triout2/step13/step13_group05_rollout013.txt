"""
Step‑function optimizer for the max self‑convolution problem.

The implementation builds on the previously successful multi‑stage
optimizer but allocates more time to the most promising refinements
and adds a few extra techniques:

* a broader parametric warm‑start (power‑law, tilted power‑law,
  Gaussian and a simple Beta‑family);
* longer projected Adam runs with a higher soft‑max temperature;
* repeated deterministic mass‑transfer (DMT2) passes;
* a more aggressive global‑peak mass‑transfer;
* several pairwise‑transfer passes with increasing donor/receiver
  fractions;
* a deterministic symmetrisation step (averaging with the reversed
  sequence) and a tiny smoothing/gradient‑free polishing loop;
* a final high‑β Adam polishing.

All phases respect the global 1000 s budget and the simplex
constraint Σa = √(2 n).  The returned list is trimmed of trailing
near‑zero entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(b)/(sum(a)^2).  With sum(a)=√(2n) the
    denominator equals 2·n, hence the score is exactly max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^α·(n‑i)^β   (a simple Beta‑like shape)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the previous DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass‑transfer (new refinement)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise mass transfer.
    In each iteration the current peak of the convolution is reduced by
    moving mass from a large coefficient (donor) to a small coefficient
    (receiver).  The amount moved is the largest delta that keeps the
    peak index as the maximal entry.  The best donor‑receiver pair is
    applied.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection to avoid drift
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} "
                  f"Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration phase – unchanged)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (new refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Reduce the current convolution peak by moving a *fraction* of the
    total donor mass to a single weak receiver.  Donors are chosen as the
    indices with largest a_i·a_{k−i} (where k is the peak index);
    receivers are indices with smallest a_{k−i}.  The maximal feasible
    fraction α is computed analytically from the linear constraints
    α·(M·shift_r – C_d) ≤ 0 for all positions, guaranteeing that no
    other entry exceeds the previous maximum in the linear model.
    The move is accepted only if the true score improves.
    """
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second maximum (used only as a stop‑condition)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)   # contribution of donors

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # a shifted for receiver r
            diff = donor_mass * shift_r - conv_donors   # we need α·diff ≤ 0 component‑wise
            mask_pos = diff > 0
            if np.any(mask_pos):
                # α ≤ conv_donors / (donor_mass * shift_r) for those positions
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # Apply the move (fraction α of donor mass)
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move

        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} "
                  f"α={best_alpha:.4g} new_score={best_score:.6f}")
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic symmetrisation and tiny smoothing refinements
# ----------------------------------------------------------------------


def _symmetrize(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Average a with its reverse and re‑project onto the simplex."""
    b = (a + a[::-1]) * 0.5
    return _project_to_simplex(b, target_sum)


def _smooth_update(a: np.ndarray, lam: float, target_sum: float) -> np.ndarray:
    """One smoothing step: a ← (1‑λ)·a + λ·(a * K)  with a small kernel K."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    a_s = np.convolve(a, kernel, mode='same')
    a_new = (1.0 - lam) * a + lam * a_s
    a_new = np.maximum(a_new, 0.0)
    return _project_to_simplex(a_new, target_sum)


def _tiny_smoothing_refine(a0: np.ndarray,
                           deadline: float,
                           target_sum: float) -> tuple[np.ndarray, float]:
    """Very small deterministic polishing using a few smoothing passes."""
    a = a0.copy()
    best_score = _score(a)
    best_a = a.copy()
    # Progressive λ values – start relatively large then shrink
    lambdas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for lam in lambdas:
        # at most a few attempts per λ
        attempts = 0
        while time.time() < deadline and attempts < 5:
            a_candidate = _smooth_update(a, lam, target_sum)
            val = _score(a_candidate)
            if val < best_score - 1e-12:
                best_score = val
                best_a = a_candidate.copy()
                a = a_candidate.copy()
                print(f"[{time.time():.2f}] Smoothing λ={lam:.3f} improved to {best_score:.6f}")
            attempts += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximum
    self‑convolution (equivalently the score returned by evaluate_sequence).
    The routine respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)  →  score = max convolution

    # ------------------------------------------------------------------
    # Phase 0 – expanded parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # broadened grids
    p_grid = np.linspace(0.20, 1.10, 31)          # power‑law exponent
    q_grid = np.linspace(-3.0, 9.0, 31)          # exponential tilt
    sigma_grid = np.linspace(0.05 * n, 0.5 * n, 9)  # Gaussian widths
    beta_alpha_grid = np.linspace(-0.5, 1.5, 7)   # alpha for beta‑family
    beta_beta_grid = np.linspace(-0.5, 1.5, 7)    # beta for beta‑family

    # Helper to try a candidate and keep the best
    def _maybe_update(cand):
        nonlocal best_seq, best_score
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New best from parametric = {best_score:.6f}")

    # power‑law
    for p in p_grid:
        if time.time() >= deadline:
            break
        _maybe_update(_power_law_sequence(n, float(p), target_sum))

    # tilted power‑law
    for p in p_grid:
        for q in q_grid:
            if time.time() >= deadline:
                break
            _maybe_update(_exp_power_law_sequence(n, float(p), float(q), target_sum))

    # Gaussian family
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        _maybe_update(_gaussian_sequence(n, float(sigma), target_sum))

    # Beta‑like family
    for alpha in beta_alpha_grid:
        for beta in beta_beta_grid:
            if time.time() >= deadline:
                break
            _maybe_update(_beta_sequence(n, float(alpha), float(beta), target_sum))

    # incorporate a previously stored construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        _maybe_update(resized)

    # jittered population around the current best (few rounds)
    for _ in range(6):
        if time.time() >= deadline:
            break
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        _maybe_update(cand)

    print(f"[{time.time():.2f}] Parametric stage finished, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Projected Adam GD with many restarts (≈40 % remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + remaining * 0.40
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD restart gave new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Repeated deterministic mass‑transfer (core refinement)
    # ------------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # short random move to escape plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        # break early if we are near the end
        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Aggressive global peak‑mass transfer (two passes)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.05,
                                                      receiver_frac=0.05)
        if val_peak < best_score:
            best_score = val_peak
            best_seq = a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer (first) best = {best_score:.6f}")

    if time.time() < deadline - 6.0:
        a_peak2, val_peak2 = _global_peak_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum,
                                                        donor_frac=0.03,
                                                        receiver_frac=0.03)
        if val_peak2 < best_score:
            best_score = val_peak2
            best_seq = a_peak2.copy()
            print(f"[{time.time():.2f}] Global peak transfer (second) best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Pairwise transfers with increasing aggressiveness
    # ------------------------------------------------------------------
    donor_fractions = [0.015, 0.03, 0.06]
    receiver_fractions = [0.015, 0.03, 0.06]
    for df, rf in zip(donor_fractions, receiver_fractions):
        if time.time() < deadline - 4.0:
            a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                          deadline,
                                                          target_sum,
                                                          donor_frac=df,
                                                          receiver_frac=rf)
            if val_pair < best_score:
                best_score = val_pair
                best_seq = a_pair.copy()
                print(f"[{time.time():.2f}] Pairwise (df={df}) improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Symmetrisation (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_sym = _symmetrize(best_seq, target_sum)
        val_sym = _score(a_sym)
        if val_sym < best_score:
            best_score = val_sym
            best_seq = a_sym.copy()
            print(f"[{time.time():.2f}] Symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase F – Tiny deterministic smoothing passes
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_smooth, val_smooth = _tiny_smoothing_refine(best_seq,
                                                      deadline,
                                                      target_sum)
        if val_smooth < best_score:
            best_score = val_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] Tiny smoothing improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase G – Final high‑β Adam polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final evaluation for logging (harmless if evaluate_sequence
    # is not present in the environment).
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
