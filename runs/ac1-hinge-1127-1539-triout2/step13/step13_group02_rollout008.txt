"""
Optimiser for the convolution‑peak benchmark.

The programme performs three progressive phases:

1. **Coarse Adam optimisation** – a fast Adam optimiser on a
   soft‑max approximation of the peak of the self‑convolution.
   The temperature (beta) is increased exponentially so the
   approximation becomes tighter while the optimiser keeps
   improving the true peak.

2. **Upscaling & fine Adam** – the current best sequence is linearly
   interpolated to a larger length (more degrees of freedom) and the
   Adam optimiser is continued with a much larger beta (≈1e7) to
   polish the solution.

3. **Final exact sub‑gradient & ordering refinement** – a short
   projected sub‑gradient descent on the exact peak followed by a
   deterministic “alternating‑order” rearrangement (largest values
   put at the ends) and a final cheap sub‑gradient pass.

All vectors are kept on the ℓ¹‑simplex
( Σ a_i = √(2·n) ) using Euclidean projection.  The routine respects
the 1000 s time budget, stores the best sequence in the global variable
``height_sequence_1`` and returns it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full convolution and max entry)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simple ℓ¹‑simplex projection (Σx = s, x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring (exact objective)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Exact benchmark score 2·n·max(a*a) / (Σa)²."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of the peak
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Returns (grad, b) where b = a*a (full convolution) and
    grad = ∂/∂a  ( softmax_β (b) ).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())

    # full convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # soft‑max weights on the convolution entries
    max_b = b.max()
    exps = np.exp(beta * (b - max_b))
    p = exps / exps.sum()

    # gradient = 2 * (a_rev  *  p)  (restricted to the original length)
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max objective (projected onto simplex)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   beta_start: float = 1.0,
                   beta_max: float = 1e7,
                   beta_mul: float = 1.03,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   eval_every: int = 200) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max approximation of the peak.
    Returns the best sequence seen and its exact score.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = beta_start

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_to_simplex(a, target_sum)

        # increase beta (make the approximation sharper)
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # occasional exact evaluation
        if t % eval_every == 0:
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Adam it={t:5d} β={beta:.1e} score={val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true peak
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(a*a).
    Uses the exact gradient of the active (max) entry.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.3

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # exact sub‑gradient of the peak entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]

        # diagonal term (i=j) contributes twice in the ordered convolution,
        # but the derivative of a_i^2 w.r.t a_i is 2*a_i, already accounted
        if (kmax % 2) == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        # small random jitter every few thousand steps to escape plateaus
        if it % 5000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple deterministic re‑ordering (large values to the ends)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value left‑most, second largest right‑most, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out

        a_sub, val_sub = _subgradient_descent(a_opt, sub_dead, target_big,
                                               max_iter=200_000)
        if val_sub < best_val:
            best_val = val_sub
            a_opt = a_sub

    # ------------------------------------------------------------------
    # Deterministic alternating ordering (often reduces the peak)
    # ------------------------------------------------------------------
    alt = _alternating_order(a_opt)
    alt_val = _score(alt)
    if alt_val < best_val:
        best_val = alt_val
        a_opt = alt

    # ------------------------------------------------------------------
    # Final tiny sub‑gradient pass for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        tiny_dead = time.time() + 0.8
        a_last, val_last = _subgradient_descent(a_opt, tiny_dead, target_big,
                                                max_iter=50_000)
        if val_last < best_val:
            best_val = val_last
            a_opt = a_last

    # ------------------------------------------------------------------
    # Trim trailing zeros for compactness
    # ------------------------------------------------------------------
    arr = np.array(a_opt, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    # expose for possible reuse
    globals()["height_sequence_1"] = final_seq

    # optional logging of the final benchmark value
    try:
        print(f"[{time.time():.2f}] Final evaluated score = {evaluate_sequence(final_seq):.6f}")
    except Exception:
        pass

    return final_seq
