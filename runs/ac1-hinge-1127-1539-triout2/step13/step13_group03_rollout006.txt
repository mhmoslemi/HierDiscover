def _subgradient_descent(a0, deadline, target_sum, step_factor=0.7, gamma=0.95, eps=1e-12):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # mask where close to max
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)
        # compute grad = 2 * (I * rev(a)) restricted to centre n entries
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[:len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + eps)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad iter={it} best={best_val:.6f}")

        it += 1

    return best_a, best_val

3. A brief permutation‑hill‑climb to polish the final sequence.

The core optimisation is a projected sub‑gradient method that works
directly on the true objective `max(conv(a,a))`.  All sequences are
kept on the ℓ¹‑simplex `∑a = √(2·n)`, so the evaluation score equals the
maximum convolution entry.
"""
import time
import math
import numpy as np
from typing import List

# --------------------------------------------------------------
# Convolution utilities (FFT based)
# --------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------
# ℓ¹‑simplex projection (Euclidean)
# --------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Sequence factories
# --------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))          # Hann window
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# --------------------------------------------------------------
# Permutation helpers – pair large with small (anti‑symmetric)
# --------------------------------------------------------------


def _pairwise_large_small_order(seq: np.ndarray) -> np.ndarray:
    """
    Produce a permutation where each position i is paired with n‑1‑i:
    the largest remaining value is placed at i, the smallest at n‑1‑i.
    """
    vals = np.sort(seq)[::-1]          # descending
    n = len(vals)
    out = np.empty_like(seq)
    left = 0
    right = n - 1
    i = 0
    j = n - 1
    while left < right:
        out[left] = vals[i]
        out[right] = vals[j]
        i += 1
        j -= 1
        left += 1
        right -= 1
    if left == right:
        out[left] = vals[i]            # middle element (if any)
    return out


# --------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# --------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.7,
                         gamma: float = 0.95,
                         eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on `max(conv(a,a))` with momentum.
    All iterates are projected onto the ℓ¹‑simplex of sum `target_sum`.
    Returns the best sequence seen and its max‑convolution value.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indicator of entries attaining (or almost attaining) the max
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        # gradient = 2 * (I ⋆ a_rev) restricted to the centre n entries
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1: len(a_rev) - 1 + len(a)]

        # momentum + step
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + eps)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Subgrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# --------------------------------------------------------------
# Main entry point required by the competition harness
# --------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The algorithm uses a fast exploratory phase on a modest sized sequence,
    then upscales and performs a long exact sub‑gradient descent.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 0 – modest size exploration
    # --------------------------------------------------------------
    # Choose a size that allows many quick trials.
    if budget_s >= 800:
        n0 = 6000
    elif budget_s >= 500:
        n0 = 5000
    else:
        n0 = 4000
    target0 = math.sqrt(2.0 * n0)

    # Build a diverse pool of candidate sequences
    init_pool: List[np.ndarray] = []
    for p in np.linspace(0.2, 2.5, 12):
        init_pool.append(_power_law_sequence(n0, float(p), target0))
    for lam in np.linspace(0.0005, 0.025, 12):
        init_pool.append(_exponential_sequence(n0, float(lam), target0))
    for _ in range(8):
        init_pool.append(_cosine_sequence(n0, target0))
    for _ in range(8):
        init_pool.append(_sin_half_sequence(n0, target0))
    for _ in range(12):
        init_pool.append(_random_simplex(n0, target0, rng))

    # reuse previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n0))

    # evaluate all seeds (objective = max convolution, because sum == target0)
    best_seq = None
    best_val = np.inf
    for seq in init_pool:
        val = _max_convolution_fft(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[{time.time():.2f}] Phase 0 best maxb={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – brief sub‑gradient refinement on the modest size
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.2)
        best_seq, best_val = _subgradient_descent(
            best_seq, sub_deadline, target0,
            step_factor=0.7, gamma=0.95)
        print(f"[{time.time():.2f}] Phase 1 refined maxb={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – upscale to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200:
        n_final = 50000
    elif remaining > 120:
        n_final = 40000
    elif remaining > 80:
        n_final = 30000
    else:
        n_final = max(n0, int(remaining * 50))  # fallback modest size

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # Apply the large‑scale anti‑symmetric ordering (large ↔ small)
    seq_big = _pairwise_large_small_order(seq_big)
    seq_big = _project_to_simplex(seq_big, target_final)

    cur_val = _max_convolution_fft(seq_big)
    print(f"[{time.time():.2f}] Upscaled to n={n_final}, start maxb={cur_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – long sub‑gradient descent on the large size
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        sub_big_deadline = deadline - 15.0   # reserve a few seconds for polishing
        seq_big, best_big_val = _subgradient_descent(
            seq_big, sub_big_deadline, target_final,
            step_factor=0.7, gamma=0.95)
        print(f"[{time.time():.2f}] Large‑scale subgrad best maxb={best_big_val:.6f}")
    else:
        best_big_val = cur_val

    # --------------------------------------------------------------
    # Phase 4 – final permutation hill‑climb (very short)
    # --------------------------------------------------------------
    if time.time() < deadline - 2:
        hill_deadline = deadline - 0.5
        # simple pairwise swap hill‑climb
        a = seq_big.copy()
        best_val_h = best_big_val
        n = a.shape[0]
        attempts = 0
        while time.time() < hill_deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            val = _max_convolution_fft(a)
            if val < best_val_h:
                best_val_h = val
                best_seq_h = a.copy()
            else:
                a[i], a[j] = a[j], a[i]   # revert
            attempts += 1
            if attempts % 20000 == 0:
                print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val_h:.6f}")
        seq_big = best_seq_h
        print(f"[{time.time():.2f}] After hill‑climb final maxb={best_val_h:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # evaluate with the supplied function (only for logging)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")

    # store for potential warm‑starts in a subsequent call
    globals()["height_sequence_1"] = final_seq
    return final_seq
