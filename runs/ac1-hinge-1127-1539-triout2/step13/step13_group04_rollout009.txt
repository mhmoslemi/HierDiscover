"""
Propose a step‑function with a very low maximal self‑convolution.
The routine follows several stages:

1)  A rich parametric pre‑search (power‑law, exponential tail,
    sine, Gaussian, triangular and a flexible Beta‑type shape).
2)  A set of Adam‑style projected gradient restarts on the simplex.
3)  An optional exact deterministic mass‑transfer refinement.
4)  A final high‑β Adam polish.
5)  A penalty‑descent phase that repeatedly pushes the current
    maximal convolution downwards.

All work is performed on the simplex
{ a_i ≥ 0 , Σ a_i = √(2·n) } so the evaluation score is exactly
the maximal convolution entry.
The code adapts the problem size to the available time budget,
tries many constructions and keeps the best one found before the
deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution) --------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors (FFT based)."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s, x ≥ 0 --------------------
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (dense pre‑search) ---------------------------------
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float,
                       target_sum: float) -> np.ndarray:
    """Centered Gaussian, clipped at the ends."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, target_sum: float) -> np.ndarray:
    """Symmetric triangular shape (zero at the ends, peak in the middle)."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        right = left[::-1]
        a = np.concatenate([left, right])
    else:
        right = left[::-1]
        a = np.concatenate([left, np.array([left[-1] + 1.0]), right])
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float,
                   target_sum: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: a_i ∝ (i+1)^alpha * (n-i)^beta.
    Both exponents can be fractional.
    """
    i = np.arange(n, dtype=np.float64) + 1.0          # 1 … n
    j = np.arange(n, dtype=np.float64)[::-1] + 1.0   # n … 1
    a = (i ** alpha) * (j ** beta)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring ---------------------------------------------------------------
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the current objective (max self‑convolution)."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (for Adam) -----------------------------------------
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max) -----------------------
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the simplex using a soft‑max approximation of max."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                               size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement) --------------------
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Exact mass‑transfer that reduces the maximal convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second best value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        # pick the first maximal index (any works)
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                      # the element paired with idx
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner among donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index with smallest partner (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty‑descent (push bound down) --------------------------------------
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    """
    Starting from a0, try to reach a feasible point with
    max(conv) ≤ target_max by minimizing the penalty
    (max(conv) - target_max)_+.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty = convex hinge
        p = conv - target_max
        p[p < 0] = 0.0

        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = grad.dot(grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001,
                                                   size=n),
                                    target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            cand = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(cand)
            if cand_max < cur_max - 1e-12:
                a = cand
                best_max = cand_max
                best_a = cand.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001,
                                                   size=n),
                                    target_sum)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises the maximal self‑convolution.
    The function adapts the problem size to the available budget, runs a
    parametric pre‑search, several Adam restarts, an exact deterministic
    mass‑transfer refinement and a final penalty‑descent polishing.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (larger n → more freedom, but also slower)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000
    target_sum = math.sqrt(2.0 * n)          # Σa = √(2·n) → objective = max(conv)

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search (including new Beta family)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power‑law
    for p in np.linspace(0.25, 0.90, 21):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # power‑law + exponential tail
    for p in np.linspace(0.25, 0.90, 15):
        for q in np.linspace(0.0, 4.0, 15):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # sine
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # Gaussian families
    for sigma in np.linspace(n / 30, n / 8, 7):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score, best_seq = val, seq.copy()

    # triangular shape
    seq = _triangular_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score, best_seq = val, seq.copy()

    # Beta‑type shapes (flexible symmetric or skewed)
    alphas = np.linspace(0.2, 4.0, 9)
    betas = np.linspace(0.2, 4.0, 9)
    for a in alphas:
        for b in betas:
            seq = _beta_sequence(n, float(a), float(b), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score, best_seq = val, seq.copy()

    # Re‑use a previous construction if it exists (global variable)
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score, best_seq = val, resized.copy()

    # Small random jittered population around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score, best_seq = val, cand.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (≈30 % of remaining time)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score - 1e-12:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (≈25 % of remaining time)
    # ------------------------------------------------------------------
    dmt_deadline = time.time() + 0.25 * (deadline - time.time())
    a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                         dmt_deadline,
                                                         target_sum)
    if val_ref < best_score - 1e-12:
        best_score, best_seq = val_ref, a_ref.copy()
        print(f"[{time.time():.2f}] Deterministic DMT improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Final high‑β Adam polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=8e5,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score - 1e-12:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final high‑β Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Iterative penalty descent to push the bound lower
    # ------------------------------------------------------------------
    while time.time() + 7.0 < deadline:
        # aim for a modest reduction of the current best
        target = best_score - max(0.001, 0.05 * (best_score - 1.0))
        a_pen, val_pen = _penalty_lower_max(best_seq,
                                            target_sum,
                                            deadline,
                                            target)
        if val_pen < best_score - 1e-12:
            best_score, best_seq = val_pen, a_pen.copy()
            print(f"[{time.time():.2f}] Penalty descent improved best = {best_score:.6f}")
        else:
            # if no progress, break to avoid endless loop
            break

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.asarray(best_seq, dtype=float)
    nz = np.where(arr > 1e-13)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # sanity check with the provided evaluator (optional)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
