"""
Propose a low‑peak self‑convolution step function.

The algorithm works in two scales.

* **Stage 1 – small scale (n≈⅓ of the final size).**
  A dense scan of simple families (power‑law, exponentially‑tilted,
  Gaussian) supplies an initial candidate.  A few restarts of an
  Adam‑style projected gradient descent (soft‑max approximation of the
  max‑norm) and an exact deterministic mass‑transfer improve the
  solution quickly because the problem is cheap at this size.

* **Stage 2 – up‑scale and refine.**
  The best small sequence is interpolated to the final length (≈12000)
  and projected onto the simplex Σa = √(2 n).  The longer vector is then
  refined with a richer toolbox:
    – Adam‑GD with a large soft‑max temperature,
    – deterministic mass‑transfer,
    – global peak‑mass transfer,
    – pairwise greedy transfer,
    – a final high‑β Adam polish.

All phases respect an overall time budget (default 1000 s).  The
returned list is trimmed of trailing near‑zero entries and is ready to
be passed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(b)/(sum(a)^2)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search (small scale)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return (gradient, full convolution) of the soft‑max approximation."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max probabilities

    a_rev = a[::-1]
    ar_fft = np.fft.rfft(a_rev, n=size)
    pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_fft * pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]     # gradient w.r.t. a
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (new refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.02,
                               receiver_frac: float = 0.02,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """Move a fraction of the total donor mass to a single weak receiver."""
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second maximum (stop‑condition)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i} (zero out‑of‑range)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donors: indices with large a_i * w_i
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)       # contribution of donors only

        # receivers: indices with smallest partner weight w_i
        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        # padded view for fast shifted access
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # a shifted for receiver r
            diff = donor_mass * shift_r - conv_donors   # linear constraint
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # apply the move
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} α={best_alpha:.4g} new_score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass transfer (greedy)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise move that reduces the current convolution peak."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]
        receiver_idx = np.argsort(a)[:nr]

        # padded view for fast shifted access
        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i:i + 2 * n - 1]     # length 2n‑1 view

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection to avoid drift
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    max self‑convolution (equivalently the score returned by
    ``evaluate_sequence``).  The routine respects the given time budget
    and returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem sizes
    # ------------------------------------------------------------------
    if budget_s >= 800:
        final_n = 12000
    elif budget_s >= 400:
        final_n = 8000
    else:
        final_n = 5000

    # a smaller scale for the cheap exploratory phase
    small_n = max(1000, final_n // 3)

    target_sum_small = np.sqrt(2.0 * small_n)
    target_sum_final = np.sqrt(2.0 * final_n)

    # ------------------------------------------------------------------
    # Stage 1 – cheap exploration on the small scale
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    best_small = None
    best_small_val = np.inf

    # parametric families (dense scan)
    p_grid = np.linspace(0.30, 0.85, 26)          # fewer points for speed
    q_grid = np.linspace(0.0, 6.0, 21)

    for p in p_grid:
        seq = _power_law_sequence(small_n, float(p), target_sum_small)
        val = _score(seq)
        if val < best_small_val:
            best_small_val = val
            best_small = seq.copy()

    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(small_n, float(p), float(q), target_sum_small)
            val = _score(seq)
            if val < best_small_val:
                best_small_val = val
                best_small = seq.copy()

    sigma_grid = np.linspace(0.1 * small_n, 0.5 * small_n, 7)
    for sigma in sigma_grid:
        seq = _gaussian_sequence(small_n, float(sigma), target_sum_small)
        val = _score(seq)
        if val < best_small_val:
            best_small_val = val
            best_small = seq.copy()

    # incorporate previous construction if available (down‑sample it)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != small_n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, small_n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum_small)
        val = _score(resized)
        if val < best_small_val:
            best_small_val = val
            best_small = resized.copy()

    # jittered population around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum_small * 0.004, size=small_n)
        cand = best_small + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum_small)
        val = _score(cand)
        if val < best_small_val:
            best_small_val = val
            best_small = cand.copy()

    print(f"[{time.time():.2f}] Small‑scale initial best = {best_small_val:.6f}")

    # Adam‑GD with restarts on the small scale
    gd_deadline_small = time.time() + (deadline - time.time()) * 0.22
    a_small_opt, val_small_opt = _gradient_descent_with_restarts(best_small,
                                                                 target_sum_small,
                                                                 gd_deadline_small)
    if val_small_opt < best_small_val:
        best_small_val = val_small_opt
        best_small = a_small_opt.copy()
        print(f"[{time.time():.2f}] Small‑scale GD improved best = {best_small_val:.6f}")

    # Exact deterministic mass‑transfer (small scale)
    dmt_deadline_small = time.time() + (deadline - time.time()) * 0.15
    a_small_dmt, val_small_dmt = _deterministic_mass_transfer_exact2(best_small,
                                                                     dmt_deadline_small,
                                                                     target_sum_small)
    if val_small_dmt < best_small_val:
        best_small_val = val_small_dmt
        best_small = a_small_dmt.copy()
        print(f"[{time.time():.2f}] Small‑scale DMT improved best = {best_small_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑scale to the final size and heavy refinement
    # ------------------------------------------------------------------
    # Interpolate to final length
    x_old = np.linspace(0.0, 1.0, small_n)
    x_new = np.linspace(0.0, 1.0, final_n)
    best_seq = np.interp(x_new, x_old, best_small)
    best_seq = _project_to_simplex(best_seq, target_sum_final)
    best_val = _score(best_seq)
    print(f"[{time.time():.2f}] After up‑scale score = {best_val:.6f}")

    # Allocate remaining budget among refinement phases
    remaining = deadline - time.time()
    # Order of phases (proportional time slices)
    phases = [
        ("AdamGD (high β)", 0.28,
         lambda a, dl: _adam_gradient_descent(a, dl, target_sum_final,
                                              max_beta=5e5,
                                              step_factor=0.7,
                                              gamma=0.995)),
        ("Exact DMT", 0.20,
         lambda a, dl: _deterministic_mass_transfer_exact2(a, dl, target_sum_final)),
        ("Global peak transfer", 0.20,
         lambda a, dl: _global_peak_mass_transfer(a, dl, target_sum_final,
                                                 donor_frac=0.03,
                                                 receiver_frac=0.03)),
        ("Pairwise transfer", 0.16,
         lambda a, dl: _pairwise_optimal_transfer(a, dl, target_sum_final,
                                                  donor_frac=0.02,
                                                  receiver_frac=0.02)),
        ("Random moves", 0.10,
         lambda a, dl: _global_random_mass_transfer(a, dl, target_sum_final)),
    ]

    for name, fraction, worker in phases:
        if time.time() >= deadline - 0.7:
            break
        sub_deadline = time.time() + (deadline - time.time()) * fraction
        a_new, val_new = worker(best_seq, sub_deadline)
        if val_new < best_val - 1e-10:
            best_val = val_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {name} improved best = {best_val:.6f}")

    # Final polishing (very high β, tiny steps) if time permits
    if time.time() < deadline - 1.0:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum_final,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
