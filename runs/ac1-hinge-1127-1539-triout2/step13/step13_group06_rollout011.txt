"""
Advanced step‑function optimizer (v2).

The optimizer builds a candidate by combining several ideas:

1. **Rich parametric seeds** – power‑laws, exponentially‑tilted,
   sinusoidal and elliptic shapes, plus a few random spike patterns.
2. **Flatten‑the‑auto‑correlation** – minimise the L2 distance between
   the convolution `a*a` and its average value (the theoretical lower
   bound).  This is a smooth objective; we optimise it with Adam
   while always projecting onto the simplex {a≥0, Σa = √(2n)}.
3. **Soft‑max max‑convolution minimisation** – the original Adam‑GD on a
   smooth approximation of the max, now with a much larger β schedule
   (up to 1e7).
4. **Exact deterministic mass‑transfer (DMT2)**, aggressive pairwise
   transfers and a final simulated‑annealing refinement to escape any
   remaining shallow minima.
5. **Time‑aware orchestration** – each phase consumes a fraction of the
   total 1000 s budget, ensuring that the routine always returns a
   valid sequence before the deadline.

The returned list contains non‑negative heights; trailing near‑zero
entries are stripped.  The algorithm is deterministic apart from the
random jitter used for exploration.  It has been tuned to beat the
previous best score (≈1.5074) and to target a value below 1.503.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Σx = s , x ≥ 0)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (used for the initial population)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sinusoidal_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a *= target_sum / a.sum()
    return a


def _elliptic_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt((i + 1) * (n - i))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities (max‑approximation)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(conv(a,a))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam optimisation of a *smooth* loss
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max approximation of the max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}"
            )

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray, target_sum: float, deadline: float
) -> tuple[np.ndarray, float]:
    """Multiple short Adam runs from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] GD restart improved best={best_val:.6f}"
            )

    return best_a, best_val

# Exact deterministic mass‑transfer (DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}"
            )

        it += 1

    return best_a, best_score

def _pairwise_optimal_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_frac: float = 0.03,
    receiver_frac: float = 0.03,
    min_improve: float = 1e-7,
) -> tuple[np.ndarray, float]:
    """Greedy pairwise transfer (larger donor/receiver pools)."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate(
        [np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)]
    )

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))

        # donor based on contribution to the current maximum
        partner = a[k - np.arange(max(0, k - (n - 1)), min(k, n - 1) + 1)]
        idx_range = np.arange(max(0, k - (n - 1)), min(k, n - 1) + 1)
        contrib = a[idx_range] * partner
        donor_idx = idx_range[np.argsort(-contrib)[:nd]]

        # receivers with small partner values
        receiver_idx = idx_range[np.argsort(partner)[:nr]]

        shifted = {}
        cand = np.unique(np.concatenate((donor_idx, receiver_idx)))
        for i in cand:
            shifted[i] = base[i : i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]
                d = 2.0 * (shift_j - shift_i)          # effect per unit delta
                d_k = d[k]
                if d_k >= 0.0:
                    continue            # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # keep base up‑to‑date (cheap slice assignment)
        base[pad_len + i] = a[i]
        base[pad_len + j] = a[j]

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(
                f"[{time.time():.2f}] Pairwise transfer i={i} j={j} Δ={best_delta:.3e} new_score={best_score:.6f}"
            )

    return a, best_score

def _simulated_annealing_refinement(
    a0: np.ndarray, deadline: float, target_sum: float
) -> tuple[np.ndarray, float]:
    """Mass‑move SA that can escape shallow local minima."""
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    pad_len = n - 1
    base = np.concatenate(
        [np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)]
    )

    rng = np.random.default_rng()
    start_time = time.time()
    T0 = cur_max * 0.02 + 1e-4  # initial temperature scale
    iter_cnt = 0
    refresh_interval = 500
    donor_frac = 0.02
    recv_frac = 0.02

    while time.time() < deadline:
        if iter_cnt % refresh_interval == 0:
            donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
            receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        shift_i = base[i : i + 2 * n - 1]
        shift_j = base[j : j + 2 * n - 1]
        diff = shift_j - shift_i

        new_conv = conv + delta * diff
        new_max = new_conv.max()

        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-9)
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            base[pad_len + i] = a[i]
            base[pad_len + j] = a[j]

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(
                    f"[{time.time():.2f}] SA iter={iter_cnt} new_best={best_score:.6f}"
                )

        if iter_cnt % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            base = np.concatenate(
                [np.zeros(pad_len), a, np.zeros(pad_len)]
            )

        iter_cnt += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Flattening gradient (L2 loss vs. average convolution)
# ----------------------------------------------------------------------


def _gradient_flatten(a: np.ndarray, target_val: float) -> np.ndarray:
    """
    Gradient of  ∑_k (conv_k - target)^2  w.r.t. a,
    where conv = a * a (full linear convolution).
    """
    n = a.shape[0]
    # ensure FFT size at least 3n-2 for the full cross‑correlation
    size = 1 << ((3 * n - 2).bit_length())

    # convolution a*a
    a_fft = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    # C = conv - target
    C = conv - target_val

    # cross‑correlation C ⋆ a = (C * a_rev)[n-1 : n-1+n]
    a_rev_fft = np.fft.rfft(a[::-1], n=size)
    C_fft = np.fft.rfft(C, n=size)
    corr_full = np.fft.irfft(C_fft * a_rev_fft, n=size)[: 3 * n - 2]

    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


def _adam_flatten(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    target_val: float,
    step_factor: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the L2 flattening loss.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_score = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        grad = _gradient_flatten(a, target_val)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Flatten Adam it={it:5d} score={best_score:.6f}"
            )
        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """

    """

    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size selection
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 18000
    elif budget_s >= 800:
        n = 15000
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)  # Σa = √(2 n)

    # --------------------------------------------------------------
    # Phase 0 – diverse parametric seeds
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf

        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in np.linspace(0.2, 1.0, 7):
        for q in np.linspace(0.0, 8.0, 5):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # sinusoidal & elliptic shapes
    seq = _sinusoidal_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    seq = _elliptic_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # random sparse spikes
    for _ in range(5):
        seq = np.zeros(n, dtype=np.float64)
        k_spikes = rng.integers(low=2, high=max(2, n // 100))
        pos = rng.choice(n, size=k_spikes, replace=False)
        vals = rng.random(k_spikes) + 0.1
        seq[pos] = vals
        seq = _project_to_simplex(seq, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # previous construction if available
    if "height_sequence_1" in globals():
        prev = np.array(
            globals()["height_sequence_1"], dtype=np.float64
        )
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # jittered population around the current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] Phase0 best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – flatten the convolution (L2 loss)
    # --------------------------------------------------------------
    # target average value of the convolution
    target_avg = target_sum ** 2 / (2 * n - 1)

    flatten_deadline = min(
        deadline - 5.0, time.time() + (deadline - time.time()) * 0.20
    )
    for _ in range(3):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_flat, val_flat = _adam_flatten(
            start_vec, flatten_deadline, target_sum, target_avg, step_factor=0.6
        )
        if val_flat < best_score:
            best_score = val_flat
            best_seq = a_flat.copy()
            print(
                f"[{time.time():.2f}] Flattening improved best = {best_score:.6f}"
            )
        # shrink deadline for the next restart if time runs low
        flatten_deadline = min(flatten_deadline, deadline - 5.0)

    # --------------------------------------------------------------
    # Phase 2 – minimise the max via soft‑max ADAM (high β)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_gd, val_gd = _gradient_descent_with_restarts(
            best_seq, target_sum, deadline
        )
        if val_gd < best_score:
            best_score = val_gd
            best_seq = a_gd.copy()
            print(
                f"[{time.time():.2f}] Max‑softmax ADAM improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase 3 – deterministic mass‑transfer (DMT2)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_dmt, val_dmt = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_dmt < best_score - 1e-12:
            best_score = val_dmt
            best_seq = a_dmt.copy()
            print(
                f"[{time.time():.2f}] DMT2 improved best = {best_score:.6f}"
            )
        else:
            break

    # --------------------------------------------------------------
    # Phase 4 – aggressive pairwise optimisation
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_pair, val_pair = _pairwise_optimal_transfer(
            best_seq, deadline, target_sum, donor_frac=0.05, receiver_frac=0.05
        )
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(
                f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase 5 – final simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a_sa, val_sa = _simulated_annealing_refinement(
            best_seq, deadline, target_sum
        )
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(
                f"[{time.time():.2f}] SA refinement improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Post‑processing
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final sanity check (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
