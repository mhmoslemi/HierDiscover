```
def smooth_gradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    best = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    start = time.time()
    rng = np.random.default_rng()
    # schedule for alpha: start small, increase linearly to max_alpha
    max_alpha = 10.0
    min_alpha = 1.0
    while time.time() < deadline:
        b = _conv_fft(a)
        maxb = b.max()
        # compute alpha based on progress
        elapsed = time.time() - start
        total = deadline - start
        alpha = min_alpha + (max_alpha - min_alpha) * (elapsed / total)
        # compute weights
        w = np.exp(alpha * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # We'll implement via FFT correlation.
```

assistantfinal**Solution Overview**

The score to minimise is  

\[
S(a)=\frac{2n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where \((a*a)_k\) denotes the linear self‑convolution of the non‑negative step‑function
\(a=(a_0,\dots ,a_{n-1})\).
The factor \(\sum_i a_i\) is fixed by the projection onto the simplex
\(\{x\ge0,\;\sum x=\sqrt{2n}\}\); after the projection the objective is
\(\displaystyle 2n\max_k (a*a)_k /(\sqrt{2n})^2\).

The algorithm consists of three independent phases, each respecting the
global time budget:

| Phase | Idea |
|------|------|
| **1 – Smooth gradient descent** | Approximate the max‑convolution by a soft‑max, obtain a dense gradient via FFT and descend while staying on the simplex. |
| **2 – Multi‑active sub‑gradient descent** | Use the exact sub‑gradient of the max‑convolution, but average over *all* active constraints (all indices where the peak is attained). |
| **3 – Aggressive simulated‑annealing mass transfer** | Local mass moves (donor → receiver) are performed with an exact \(O(n)\) update of the convolution; a square‑root cooling schedule keeps the system hot for a long time. |

Multiple random restarts (uniform, Dirichlet, power‑law, Gaussian) and a
warm‑start from the previous best solution are tried.  
The best sequence found is returned and stored in the global variable
`height_sequence_1` for the next run.

All heavy work (convolutions, gradients) is performed with FFTs; each
iteration is therefore \(O(n\log n)\) and easily fits into the 1000 s budget.

```python
"""
Non‑negative step function optimiser.

Three phase optimiser:
1. smooth (soft‑max) gradient descent on the max‑convolution,
2. multi‑active sub‑gradient descent on the true max,
3. aggressive simulated‑annealing mass‑transfer refinement.

Multiple restarts (warm‑start, uniform, Dirichlet, power‑law, Gaussian)
are tried; the best sequence is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Gradient descent on a soft‑max approximation of the max‑convolution.
    The gradient is obtained by a convolution of the weight vector with `a`
    (exactly via FFT).  After each step the iterate is projected onto the
    simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    it = 0
    rng = np.random.default_rng()

    # schedule for the smoothing parameter α (soft‑max "temperature")
    min_alpha = 0.5
    max_alpha = 12.0

    while time.time() < deadline:
        # convolution a*a
        b = _conv_fft(a)
        maxb = b.max()

        # linear interpolation of α from min_alpha → max_alpha
        elapsed = time.time() - start
        total = deadline - start
        alpha = min_alpha + (max_alpha - min_alpha) * (elapsed / total)

        # soft‑max weights (numerically stable)
        w = np.exp(alpha * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / b.size
        else:
            w /= w_sum

        # gradient g_i = 2 * sum_j w_{i+j} a_j
        # realised as correlation of w with a (via FFT)
        size = 1 << ((len(w) + n - 1).bit_length())
        fw = np.fft.rfft(w, n=size)
        far = np.fft.rfft(a[::-1], n=size)          # a reversed
        conv = np.fft.irfft(fw * far, n=size)
        # usable part: indices n‑1 … n‑1+n‑1
        start_idx = n - 1
        grad = 2.0 * conv[start_idx:start_idx + n]

        # step‑size schedule (linear decay)
        eta = 1.5 * (1.0 - elapsed / total) + 0.01
        a = a - eta * grad
        a = _project_to_simplex(np.maximum(a, 0.0), target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth it={it:5d} score={best_val:.6f}")

        # occasional jitter to avoid stagnation
        if it % 800 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – multi‑active sub‑gradient descent
# ----------------------------------------------------------------------


def multiactive_subgradient_descent(a0: np.ndarray, target_sum: float,
                                    deadline: float):
    """
    Sub‑gradient descent on the exact max‑convolution.
    The gradient is the *average* of the sub‑gradients of all active
    constraints (all indices where the convolution reaches its maximum).
    After each step we project back onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        b = _conv_fft(a)
        maxb = b.max()
        # treat all indices within a tiny tolerance as active
        tol = maxb * 1e-8 + 1e-12
        active = np.where(b >= maxb - tol)[0]

        # build averaged sub‑gradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]          # 2·a_{k‑i}
        if active.size > 0:
            grad /= active.size

        # step‑size (linear decay)
        elapsed = time.time() - start
        total = deadline - start
        eta = 1.8 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(np.maximum(a, 0.0), target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Multi‑sub it={it:5d} score={best_val:.6f}")

        # jitter every few hundred iterations
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – aggressive simulated‑annealing mass transfer
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves a tiny amount of mass from a
    “bad’’ index (large contribution to the current peak) to a
    “good’’ index (low height).  The convolution is updated in O(n)
    for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact self‑convolution
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                    # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling (slow decay)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak (a[i]·a[k‑i])
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour low heights
        inv = 1.0 / (a + 1e-12)
        inv /= inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of transfer – aggressive but still bounded
        max_delta = min(a[donor] * 0.4, target_sum * 0.03)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear term: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # choose a problem size (larger n → finer control, but slower)
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # usual normalisation

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + diverse randoms)
    # --------------------------------------------------------------
    init_candidates = []

    # warm start from previous run (if any)
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()

    # uniform start (just a baseline)
    uniform = np.full(n, target_sum / n)
    init_candidates.append(uniform)

    # Dirichlet (random simplex) starts
    for _ in range(2):
        rnd = rng.dirichlet(alpha=np.ones(n))
        rnd = rnd * target_sum
        init_candidates.append(rnd)

    # power‑law starts with random exponent
    for _ in range(2):
        exp = rng.uniform(0.3, 0.9)
        a = np.arange(1, n + 1, dtype=np.float64) ** (-exp)
        a = a / a.sum() * target_sum
        init_candidates.append(a)

    # Gaussian‑shaped start
    sigma = n / 10.0
    x = np.arange(n, dtype=np.float64)
    gauss = np.exp(- (x - n / 2) ** 2 / (2.0 * sigma * sigma))
    gauss = gauss / gauss.sum() * target_sum
    init_candidates.append(gauss)

    # --------------------------------------------------------------
    # Optimisation loop over the candidates
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start candidate {idx+1}/{len(init_candidates)} -----")

        # ---- Phase 1 : smooth gradient descent --------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        phase1_deadline = time.time() + remaining * 0.25
        a1, score1 = smooth_gradient_descent(init, target_sum, phase1_deadline)
        if score1 < best_score:
            best_score = score1
            best_seq = a1.copy()
            print(f"[{time.time():.2f}] New global best after smooth GD: {best_score:.6f}")

        # ---- Phase 2 : multi‑active sub‑gradient descent ---------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        phase2_deadline = time.time() + remaining * 0.20
        a2, score2 = multiactive_subgradient_descent(a1, target_sum, phase2_deadline)
        if score2 < best_score:
            best_score = score2
            best_seq = a2.copy()
            print(f"[{time.time():.2f}] New global best after multi‑subgrad: {best_score:.6f}")

        # ---- Phase 3 : simulated‑annealing refinement -------------
        if time.time() < deadline:
            a3, score3 = anneal_mass_transfer(a2, target_sum, deadline)
            if score3 < best_score:
                best_score = score3
                best_seq = a3.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the unlikely case that nothing was generated
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = _score(best_seq)

    # Trim trailing near‑zero entries (evaluate_sequence does this internally)
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Final evaluation (for logging only – does not affect the result)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
```
