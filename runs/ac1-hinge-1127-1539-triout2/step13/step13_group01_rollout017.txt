"""
Propose a low‑score non‑negative step sequence.

The search proceeds in three stages within the allotted time budget:

1. **Guided initialization** – generate a few deterministic families that are
   analytically known to give low max‑convolution (power‑law a_i∝(i+1)^(-p)
   with p≈0.5‑0.7, exponential decay, and a product of both).  The best of
   these is kept as a warm–start.

2. **Projected sub‑gradient descent** – optimise the true max‑convolution
   entry directly, projecting after every step onto the simplex
   {x≥0, Σx = √(2·n)}.  This phase receives about 40 % of the remaining
   budget.

3. **Aggressive simulated‑annealing mass transfer** – a fast O(n) rank‑one
   update moves a small amount of mass from “bad” indices (large contribution
   to the current peak) to “good” ones.  The temperature follows a square‑root
   cooling schedule and this phase uses the rest of the time.

All helper functions are top‑level, no closures or lambdas are used.
The final sequence is clipped to the allowed range by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied.
    """
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape families for guided initialisation
# ----------------------------------------------------------------------


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (i = 0,…,n-1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i / (n-1))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-lam * x)
    return a


def _shape_powerexp(n: int, p: float, lam: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) * exp(-lam * i/(n-1))."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) / (n - 1))
    a = idx ** (-p) * np.exp(-lam * x)
    return a


def _shape_beta(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape: a_i ∝ (x)^{α-1} (1-x)^{α-1}
    where x = (i+0.5)/n ∈ (0,1).
    """
    # avoid zero to keep finite values
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (alpha - 1.0))
    return a


# ----------------------------------------------------------------------
# Phase 1 – projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_projected_descent(a0: np.ndarray, target_sum: float,
                                 deadline: float):
    """
    Sub‑gradient descent on the entry achieving the max convolution.
    After each step the vector is projected onto the simplex to keep the
    sum fixed (√(2·n)).  Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        # full convolution – we need the index of the max only
        b = _conv_fft(a)
        k_max = int(np.argmax(b))

        # build the sub‑gradient (active peak)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        # derivative of a_i*a_{k-i} w.r.t a_i is a_{k-i} + a_i (but symmetric)
        grad[idx] = 2.0 * a[k_max - idx]

        # step size – linearly decreasing
        elapsed = time.time() - start
        total = deadline - start
        eta = 2.0 * (1.0 - elapsed / total) + 0.01

        a = a - eta * grad
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑proj it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from ‘bad’
    indices (large contribution to the current peak) to ‘good’ indices
    (small height).  The convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature – a sizable fraction of the current energy
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling (keeps the system hot longer)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak (a[i]·a[k‑i])
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on the *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # problem size – larger n gives finer granularity at higher cost
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # the usual normalisation

    # ------------------------------------------------------------------
    # Guided initialisation (shape families)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Warm start from previous run, if any
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        val = _score(warm)
        best_seq = warm.copy()
        best_score = val
        print(f"[{time.time():.2f}] Warm start score = {best_score:.6f}")

    # deterministic shape sweep (cheap, gives solid baselines)
    # power‑law exponents
    for p in np.linspace(0.45, 0.75, 7):
        a = _shape_powerlaw(n, p)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:   # respect the hard bound from evaluate_sequence
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.3f} score={val:.6f}")

    # exponential decay sweep
    for lam in np.linspace(0.5, 8.0, 7):
        a = _shape_exponential(n, lam)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exponential λ={lam:.3f} score={val:.6f}")

    # combined power‑law + exponential
    for _ in range(8):
        p = np.random.random() * 0.8 + 0.1   # 0.1 … 0.9
        lam = np.random.random() * 8.0
        a = _shape_powerexp(n, p, lam)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] PowerExp p={p:.3f} λ={lam:.3f} score={val:.6f}")

    # symmetric Beta‑type family (U‑shaped when α<1)
    for alpha in np.linspace(0.25, 0.9, 7):
        a = _shape_beta(n, alpha)
        a = a * (target_sum / a.sum())
        if a.max() > 1000.0:
            continue
        val = _score(a)
        if val < best_score:
            best_score = val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta α={alpha:.3f} score={val:.6f}")

    # Random Dirichlet starts (a few)
    rng = np.random.default_rng()
    for _ in range(4):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        val = _score(rnd)
        if val < best_score:
            best_score = val
            best_seq = rnd.copy()
            print(f"[{time.time():.2f}] Random start score={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 : projected sub‑gradient descent (more time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.45 * (deadline - time.time())
        a_sub, val_sub = subgradient_projected_descent(best_seq, target_sum,
                                                       sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] New global best after sub‑proj {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 : aggressive annealing mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ann, val_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if val_ann < best_score:
            best_score = val_ann
            best_seq = a_ann.copy()
            print(f"[{time.time():.2f}] New global best after anneal {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clean‑up and evaluation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # final check using the provided evaluate_sequence
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for warm‑start in the next invocation
    globals()["height_sequence_1"] = final_seq
    return final_seq
