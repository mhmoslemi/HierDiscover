def propose_candidate(budget_s=1000, **kwargs):
    start=time.time()
    deadline=start+budget_s-0.5
    # Determine n
    if budget_s >= 900:
        n=12000
    elif budget_s >= 600:
        n=10000
    else:
        n=8000
    target_sum=math.sqrt(2.0*n)
    rng=np.random.default_rng()
    seeds=[]
    # power law p values (maybe 0.1 to 2)
    for p in np.linspace(0.1, 1.8, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))
    # exponential lambda
    for lam in np.linspace(0.001, 0.02, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))
    # random Dirichlet
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))
    # sinusoid mixtures
    for _ in range(8):
        seeds.append(_sinusoidal_sequence(n, target_sum, rng))
    # previous best
    if "height_sequence_1" in globals():
        prev=np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))
    # jitter seeds
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum*0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)
    # init population
    pop_size = max(30, int(n/250))   # e.g. 30-50
    population=[]
    # start with the best seed
    best_seed=min(seeds, key=_score)
    population.append(best_seed.copy())
    while len(population)<pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))
    # DE phase (70% of budget)
    de_dead = start + 0.70*budget_s
    if de_dead > deadline-10:  # leave some for later
        de_dead = deadline-10
    best_seq, best_score = _de_optimize(population, target_sum, de_dead)
    print(f"[{time.time():.2f}] DE best score {best_score:.6f}")
    # mass transfer polish (max 0.2 budget)
    mt_dead = time.time() + min(0.20*budget_s, deadline - time.time() - 5)
    if mt_dead > time.time():

                                                       start_temp=0.4, end_temp=1e-6, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] After mass transfer score {best_score:.6f}")
    # subgradient descent (remainder)
    sg_dead = deadline - 2.0
    if sg_dead > time.time():
        seq_sg, score_sg = _subgradient_descent(best_seq, sg_dead, target_sum,

        if score_sg < best_score:
            best_seq, best_score = seq_sg, score_sg

    # reordering

"""
Hybrid evolutionary optimizer for the step‑function minimisation problem.

The algorithm works in three stages:

1. **Seeding** – a diverse pool of feasible sequences is created
   (power‑law, exponential, random Dirichlet, sinusoidal mixtures and
   a resized copy of the previously best solution).  Every seed is
   projected onto the ℓ¹‑simplex Σa = √(2 n).

2. **Projected Differential Evolution** – a relatively large population
   is evolved using a mixture of “rand/1” and “best/1” mutation
   strategies.  After mutation a binomial crossover is applied and the
   trial vector is projected back onto the simplex.  The best individual
   is kept throughout the run.

3. **Local polishing** – deterministic mass‑transfer (simulated‑annealing
   style) followed by a short projected sub‑gradient descent are applied
   to the best DE individual.  Finally three simple reorderings are
   tried and the one with the smallest maximum autoconvolution is kept.

The total runtime is limited to the supplied budget (default 1000 s).
The function returns a list of non‑negative heights that achieves a
lower evaluation score than the previous state‑of‑the‑art implementation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a * a using FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (length 2·n‑1) using FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2 n) this equals the maximum convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (all produce points on the ℓ¹‑simplex)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to meet Σa = target_sum."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _sinusoidal_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """
    A smooth mixture of a few random cosine waves.  The absolute value is taken
    to stay non‑negative and the result is projected onto the simplex.
    """
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    a = np.zeros(n, dtype=np.float64)
    num_terms = rng.integers(3, 7)  # 3‑6 harmonics
    for _ in range(num_terms):
        freq = rng.uniform(0.5, n / 10.0)  # low‑frequency components
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.1, 1.0)
        a += amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.abs(a) + 1e-8          # ensure strict positivity
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple reordering helpers (often lower the max‑convolution)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest element leftmost, second largest rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _random_permutation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of the sequence."""
    perm = rng.permutation(seq.shape[0])
    return seq[perm]


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-6,
                                low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        # index where the convolution attains its maximum
        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # product a[i] * a[kmax-i]
        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0.0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected Differential Evolution (mixed strategies)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected DE loop.  Two mutation strategies are mixed:
      * “rand/1” (≈70 % of the time)
      * “best/1” (≈30 % of the time)
    """
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]

    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # Choose mutation strategy
            if rng.random() < 0.7:
                # rand/1
                idxs = list(range(len(population)))
                idxs.remove(i)
                a, b, c = rng.choice(idxs, size=3, replace=False)
                mutant = population[a] + F * (population[b] - population[c])
            else:
                # best/1
                a, b = rng.choice([j for j in range(len(population)) if j != i],
                                  size=2, replace=False)
                mutant = best_seq + F * (population[a] - population[b])

            mutant = _project_to_simplex(mutant, target_sum)

            # Binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # leave a short safety margin
        if time.time() > deadline - 0.5:
            break
    return best_seq, best_score

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with low evaluation score.
    The routine builds a large, diverse seed pool, runs a projected
    differential‑evolution for the bulk of the allotted time and then
    refines the best individual with deterministic mass‑transfer and a
    short sub‑gradient descent.  Finally a handful of simple reorderings
    are tested and the best ordering is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility but is slower)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 12000
    elif budget_s >= 600:
        n = 10000
    else:
        n = 8000
    target_sum = math.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a diverse collection of seeds
    # ------------------------------------------------------------------
    seeds = []

    # Power‑law families
    for p in np.linspace(0.1, 1.8, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # Exponential families
    for lam in np.linspace(0.001, 0.02, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # Random Dirichlet points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # Sinusoidal mixtures (smooth shapes)
    for _ in range(8):
        seeds.append(_sinusoidal_sequence(n, target_sum, rng))

    # Previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # Small jitter to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ------------------------------------------------------------------
    # Initialise the DE population
    # ------------------------------------------------------------------
    pop_size = max(30, int(n / 250))   # roughly 30‑50 individuals
    population = []

    # start with the best seed we have
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # ------------------------------------------------------------------
    # Differential Evolution phase (≈70 % of the budget)
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.70 * budget_s
    # keep a small headroom for polishing
    if de_deadline > deadline - 12.0:
        de_deadline = deadline - 12.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Deterministic mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_deadline = time.time() + min(0.20 * budget_s,
                                        deadline - time.time() - 6.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_deadline, target_sum,
            start_temp=0.4, end_temp=1e-6, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] After mass‑transfer – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Short sub‑gradient descent refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        sg_deadline = deadline - 2.0
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_deadline, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_seq, best_score = seq_sg, score_sg
            print(f"[{time.time():.2f}] After sub‑gradient – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simple reordering permutations (often helpful)
    # ------------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        _random_permutation(best_seq, rng)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for a compact representation
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:

    final_seq = arr.tolist()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
