"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(a*a) / (sum(a)^2)

The algorithm consists of several independent stages:

1.  Choose a problem size n depending on the time budget.  Larger n yields
    a richer search space and (empirically) a lower attainable bound.

2.  **Parametric warm‑start** – a dense sweep over several simple families
    (power‑law, offset‑power‑law, exponentially‑tilted power‑law,
    stretched‑exponential and a few Gaussians).  The best sequence from
    this stage serves as the initial point.

3.  **Projected Adam (soft‑max) GD** – the max‑norm is approximated by a
    soft‑max with an increasing inverse temperature β.  Multiple restarts
    are performed; the best iterate is kept.

4.  **Exact deterministic mass transfer (DMT2)** – a greedy move that
    reduces the current convolution peak by moving a single donor mass to a
    weak receiver while respecting all other constraints.

5.  **Global peak‑mass transfer** – for the current peak index k* a set of
    strong donors (large a_i·a_{k*−i}) and weak receivers (small a_{k*−i})
    is assembled.  A fraction α of the total donor mass is moved to the
    best receiver; the analytical α guarantees that no other entry exceeds
    the previous maximum.  This is repeated until no improvement.

6.  **Pairwise optimal transfer** – an exhaustive scan over a limited set
    of donor/receiver pairs to squeeze out the remaining improvement.

7.  **Final high‑β Adam polishing** – a last Adam run with a very large
    β to finish the optimisation.

All stages respect the global time budget (default 1000 s) and periodically
project the vector onto the simplex {x ≥ 0, Σx = √(2 n)}.
The final sequence is trimmed of trailing near‑zero entries and returned
as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(b) / (sum(a)^2).  With the normalisation
    Σa = √(2 n) the denominator equals 2·n, hence the score equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _power_law_offset_sequence(n: int, p: float, offset: float,
                               target_sum: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}  (offset>0 avoids an extremely large a_0)."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _stretched_exponential_sequence(n: int, alpha: float, beta: float,
                                   target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-α·(i/n)^β).  0 < β ≤ 2."""
    i = np.arange(n, dtype=np.float64) / n
    a = np.exp(-alpha * (i ** beta))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Multiple Adam‑GD restarts with random jitter."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (the previous DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy single‑donor / single‑receiver move that reduces the current peak."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Pairwise optimal mass transfer (new refinement)
# ----------------------------------------------------------------------


def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.015,
                               receiver_frac: float = 0.015,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy pairwise move that reduces the current peak."""
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        # Pre‑extract shifted views for the relevant indices
        shifted = {}
        relevant = np.unique(np.concatenate((donor_idx, receiver_idx)))
        for i in relevant:
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)          # effect of moving mass
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection to suppress numerical drift
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise transfer i={i} j={j} "
                  f"Δ={best_delta:.3e} new_score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Random donor→receiver moves with projection after each step."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random move new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global peak‑mass transfer (new refinement)
# ----------------------------------------------------------------------


def _global_peak_mass_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.025,
                               receiver_frac: float = 0.025,
                               min_alpha: float = 1e-6) -> tuple[np.ndarray, float]:
    """Move a fraction of the total donor mass to a single weak receiver,
    guaranteeing that the current peak does not increase (linearised model)."""
    a = a0.copy()
    n = a.shape[0]
    pad_len = n - 1
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # second maximum (stop condition)
        conv2 = conv.copy()
        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max - i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donor scoring = a_i * w_i
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)   # contribution of donors alone

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate([np.zeros(pad_len, dtype=np.float64),
                               a,
                               np.zeros(pad_len, dtype=np.float64)])

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r:r + 2 * n - 1]            # shifted view for receiver r
            diff = donor_mass * shift_r - conv_donors   # linear constraint: α·diff ≤ 0
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] /
                                   (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (-conv_donors[k_max] +
                                            donor_mass * shift_r[k_max])
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # apply the move
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global peak transfer it={it} "
                  f"α={best_alpha:.4g} new_score={best_score:.6f}")
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the max
    self‑convolution bound.  The routine respects the given time budget and
    returns the best sequence discovered as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ---------- problem size ----------
    if budget_s >= 800:
        n = 20000               # more freedom → typically lower bound
    elif budget_s >= 400:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)   # Σa = √(2 n)  →  score = max(convolution)

    # ---------- Phase 0 : parametric warm‑start ----------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power‑law
    p_grid = np.linspace(0.30, 0.85, 23)      # 23 values
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # offset power‑law (extra flexibility)
    offset_grid = [0.0, 0.3, 0.8, 1.5, 3.0]
    for p in p_grid:
        for off in offset_grid:
            seq = _power_law_offset_sequence(n, float(p), float(off), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # exponentially‑tilted power‑law
    q_grid = np.linspace(0.0, 6.0, 19)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # stretched exponential
    alpha_grid = [2.0, 5.0, 10.0, 20.0]
    beta_grid = [0.5, 0.8, 1.0, 1.2]
    for alpha in alpha_grid:
        for beta in beta_grid:
            seq = _stretched_exponential_sequence(n, float(alpha), float(beta), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # Gaussians (mostly for sanity)
    sigma_grid = np.linspace(0.15 * n, 0.45 * n, 7)
    for sigma in sigma_grid:
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # existing construction (if provided by the harness)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # jittered population around the best so far
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        val = _score(seq_j)
        if val < best_score:
            best_score = val
            best_seq = seq_j.copy()

    print(f"[{time.time():.2f}] After parametric search best score = {best_score:.6f}")

    # ---------- Phase A : Adam‑GD with restarts ----------
    gd_deadline = time.time() + (deadline - time.time()) * 0.35
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.009, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # ---------- Phase B : Exact deterministic mass‑transfer ----------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # brief random exploration to avoid being stuck
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # ---------- Phase C : Global peak‑mass transfer ----------
    if time.time() < deadline - 6.0:
        a_peak, val_peak = _global_peak_mass_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.04,
                                                      receiver_frac=0.04)
        if val_peak < best_score:
            best_score = val_peak
            best_seq = a_peak.copy()
            print(f"[{time.time():.2f}] Global peak transfer improved best = {best_score:.6f}")

    # ---------- Phase D : Pairwise optimal transfer ----------
    if time.time() < deadline - 3.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.02,
                                                      receiver_frac=0.02)
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}")

    # ---------- Phase E : Final high‑β Adam polishing ----------
    if time.time() + 2.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # ---------- Post‑processing ----------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional evaluation with the provided function (harmless if missing)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
