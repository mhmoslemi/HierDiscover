"""
Hybrid optimiser for the step‑function self‑convolution problem.

The procedure uses a large simplex‑projected population, a long
differential‑evolution (DE) phase, a temperature‑controlled random walk
(simulated‑annealing), deterministic mass‑transfer polishing,
sub‑gradient descent on the exact max‑convolution, and a short
smooth‑maximum gradient descent.  Simple reorderings are tried at the
end and the resulting sequence is trimmed and returned.

All elements stay non‑negative and the ℓ¹‑norm is forced to
√(2 n), so the objective value equals the maximal entry of the
self‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution tools (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (max‑convolution) – after ℓ¹‑normalisation this equals the
# max‑convolution itself.
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """2·n·max(a*a) / (Σa)² .  When Σa = √(2n) this equals max(a*a)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the ℓ¹‑simplex {x_i ≥ 0, Σx_i = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (different families of initial points)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simple re‑ordering helpers (often reduce the max‑convolution)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    to_left = True
    for v in sorted_vals:
        if to_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        to_left = not to_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer polishing (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)

        a[receiver] += delta

            accept = rng.random() < prob

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑maximum gradient (useful for a quick final polishing)
# ----------------------------------------------------------------------


def _smooth_max_grad(a: np.ndarray, tau: float) -> np.ndarray:
    """Gradient of τ·log‑sum‑exp(τ·conv) w.r.t. a."""
    n = a.shape[0]
    conv = _conv_fft(a)                 # length 2n‑1
    exp = np.exp(tau * conv)
    Z = exp.sum()
    w = exp / Z                         # softmax weights, length 2n‑1

    rev_w = w[::-1]
    tmp = np.convolve(a, rev_w)         # length 3n‑2
    offset = len(w) - 1                 # = 2n‑2
    grad = 2.0 * tmp[offset - np.arange(n)]
    return grad


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             target_sum: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on a smooth approximation of the max."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    tau = 0.5
    while time.time() < deadline:
        tau = min(tau * 1.015, 12.0)          # slowly increase smoothing
        grad = _smooth_max_grad(a, tau)
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = (target_sum * 0.2) / (norm + 1e-12)
            a -= step * grad
            a = _project_to_simplex(a, target_sum)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser that builds a diverse seed pool, runs a long DE
    phase, then refines the best individual with simulated annealing,
    deterministic mass‑transfer, sub‑gradient descent and a smooth‑max
    gradient step.  Simple reorderings are finally tested and the
    trimmed sequence is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # ----- problem size -------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)
    rng = np.random.default_rng()

    # ----- seed generation -------------------------------------------------
    seeds = []

    for p in np.linspace(0.1, 1.4, 13):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    for lam in np.linspace(0.0002, 0.025, 12):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    for _ in range(20):
        seeds.append(_random_simplex(n, target_sum, rng))

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter all seeds a little
    jittered = []
    for s in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        sj = s + jitter
        sj = np.maximum(sj, 0.0)
        sj = _project_to_simplex(sj, target_sum)
        jittered.append(sj)
    seeds.extend(jittered)

    # ----- initialise DE population ---------------------------------------
    pop_size = max(30, int(n / 400))            # ~30‑50 individuals
    population = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        population.append(_project_to_simplex(seeds[rng.integers(len(seeds))].copy(),
                                             target_sum))

    # ----- DE phase (major part of the budget) ---------------------------
    de_deadline = start_time + 0.73 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline,
                                        F=0.6, CR=0.9)
    print(f"[{time.time():.2f}] DE finished – best score {best_score:.6f}")

    # ----- Simulated annealing (global random walk) -----------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_seq = cur_seq.copy()
    best_score = cur_score

    # temperature schedule (starts relative to current score)
    T_start = max(cur_score * 0.08, 0.01)
    sa_end = deadline - 8.0
    if time.time() < sa_end:
        sa_start = time.time()
        while time.time() < sa_end:
            # linear cooling
            elapsed = time.time() - sa_start
            tau = elapsed / (sa_end - sa_start)
            T = T_start * (1.0 - tau) + 1e-8

            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if cur_seq[i] < 1e-12:
                continue
            # move a random fraction (up to 50%) of mass from i to j
            delta = rng.random() * cur_seq[i] * 0.5
            cand = cur_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_to_simplex(cand, target_sum)

            cand_score = _score(cand)      # uses exact max‑convolution
            if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                cur_seq = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
            # occasional light re‑ordering to escape symmetry
            if rng.random() < 0.001:
                cur_seq = _alternating_order(cur_seq)
                cur_seq = _project_to_simplex(cur_seq, target_sum)
                cur_score = _score(cur_seq)

    print(f"[{time.time():.2f}] SA finished – best score {best_score:.6f}")

    # ----- Deterministic mass‑transfer polishing -------------------------
    if time.time() < deadline - 10.0:
        mt_dead = time.time() + min(12.0, deadline - time.time() - 7.0)
        seq_mt, score_mt = _deterministic_mass_transfer(best_seq, mt_dead,
                                                        target_sum,
                                                        start_temp=0.35,
                                                        end_temp=1e-5,
                                                        low_frac=0.06)
        if score_mt < best_score:
            best_score, best_seq = score_mt, seq_mt.copy()
            print(f"[{time.time():.2f}] MT polished – best score {best_score:.6f}")

    # ----- Sub‑gradient descent -------------------------------------------
    if time.time() < deadline - 6.0:
        sg_dead = time.time() + min(8.0, deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(best_seq, sg_dead, target_sum,
                                                step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_score, best_seq = score_sg, seq_sg.copy()
            print(f"[{time.time():.2f}] Sub‑grad refined – best score {best_score:.6f}")

    # ----- Smooth‑max gradient descent ------------------------------------
    if time.time() < deadline - 3.0:
        sg2_dead = time.time() + min(4.0, deadline - time.time() - 1.0)
        seq_sg2, score_sg2 = _smooth_gradient_descent(best_seq, sg2_dead, target_sum)
        if score_sg2 < best_score:
            best_score, best_seq = score_sg2, seq_sg2.copy()
            print(f"[{time.time():.2f}] Smooth‑grad refined – best score {best_score:.6f}")

    # ----- Try a few simple reorderings -----------------------------------
    candidates = [best_seq,
                  _alternating_order(best_seq),
                  _even_spread_order(best_seq)]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering – best score {best_score:.6f}")

    # ----- Trim trailing near‑zero entries ---------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ----- Record for the next run (global cache) -------------------------
    globals()["height_sequence_1"] = final_seq

    # ----- Final sanity check (optional) ----------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
