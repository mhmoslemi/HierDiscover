"""
Multi‑stage optimizer for the maximal self‑convolution problem.

The algorithm consists of:
  1. An extensive parametric sweep over several families
     (power‑law, power‑law×exponential, beta‑distribution,
     truncated triangular).  The best candidate is kept.
  2. Deterministic “exact” mass‑transfer (greedy reduction of the
     current convolution peak).
  3. Simulated annealing on the simplex with larger donor/receiver
     pools and adaptive step sizes.
  4. A second mass‑transfer pass.
  5. Projected Adam optimisation on a smooth‑max surrogate (soft‑max
     with a gradually increasing β).
All phases respect a global time budget (default 1000 s) and the final
sequence is trimmed of trailing near‑zero entries before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (returns length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Simplex projection  {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)·exp(-q·i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float,
                   target_sum: float) -> np.ndarray:
    """Beta‑distribution shape on the interval (0,1)."""
    t = (np.arange(n, dtype=np.float64) + 0.5) / n                     # (0,1)
    # Guard against non‑positive powers
    eps = 1e-12
    t = np.clip(t, eps, 1 - eps)
    a = (t ** (alpha - 1)) * ((1 - t) ** (beta - 1))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0)
    a *= target_sum / a.sum()
    return a


def _triangular_sequence(n: int, frac: float,
                         target_sum: float) -> np.ndarray:
    """Truncated decreasing triangle of length m = frac·n."""
    m = max(1, int(frac * n))
    a = np.arange(m, 0, -1, dtype=np.float64)          # m, m‑1, …, 1
    a = np.concatenate([a, np.zeros(n - m, dtype=np.float64)])
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray,
                                                                     float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # safe fraction of donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic mass‑transfer with a Metropolis rule."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    T = max(best_score * 0.05, 1e-8)          # initial temperature
    it = 0
    top_frac = 0.05                           # donor/receiver from top/bottom 5 %
    top_k = max(1, int(top_frac * n))

    # periodic sorting
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0:
            it += 1
            continue

        # random fraction up to 40 % of donor mass
        delta = max_delta * rng.random() * 0.4
        if delta < 1e-14:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.85
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on a soft‑max surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth‑max objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # Occasionally add tiny jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal
    self‑convolution.  The routine respects the wall‑clock budget and
    returns the best list found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5                         # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom)
    # ------------------------------------------------------------------
    if budget_s >= 850:
        n = 16000
    elif budget_s >= 450:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)                                 # Σ a = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – extensive parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # ---- power‑law ----------------------------------------------------
    p_grid = np.linspace(0.15, 1.30, 56)         # finer grid
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # ---- exp‑power‑law ------------------------------------------------
    q_grid = np.linspace(0.0, 12.0, 41)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # ---- beta distribution --------------------------------------------
    alpha_grid = np.linspace(0.2, 4.0, 38)
    beta_grid = np.linspace(0.2, 4.0, 38)
    for a in alpha_grid:
        for b in beta_grid:
            seq = _beta_sequence(n, float(a), float(b), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # ---- truncated triangle --------------------------------------------
    frac_grid = np.linspace(0.10, 1.00, 19)
    for frac in frac_grid:
        seq = _triangular_sequence(n, float(frac), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # ---- simple sine (baseline) ----------------------------------------
    sin_seq = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) /
                     (n + 1.0))
    sin_seq *= target_sum / sin_seq.sum()
    val = _score(sin_seq)
    if val < best_score:
        best_score = val
        best_seq = sin_seq.copy()

    # ---- use previous global result if present -------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # ---- jittered neighbourhood ----------------------------------------
    for _ in range(12):
        jitter = rng.normal(scale=target_sum * 0.003, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = "
          f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (quick refinement)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.15
    best_seq, best_score = _deterministic_mass_transfer_exact2(
        best_seq, mt_deadline, target_sum)

    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing (exploratory)
    # ------------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.45
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)

    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer (tightening)
    # ------------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.15
    best_seq, best_score = _deterministic_mass_transfer_exact2(
        best_seq, mt2_deadline, target_sum)

    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final Adam polishing (soft‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(
            best_seq, final_deadline, target_sum,
            max_beta=1e6, step_factor=0.6, gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional sanity check (harmless if evaluate_sequence is absent)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated max‑conv = {val:.6f}")
    except Exception:
        pass

    return final_seq
