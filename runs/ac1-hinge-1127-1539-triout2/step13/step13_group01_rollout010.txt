def gradient_descent_smooth(a0, target_sum, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a) # actual max(b) after projection.
    # parameters
    start = time.time()
    iteration = 0
    # schedule for alpha and lr pattern
    # we can use piecewise schedule: choose alpha based on iteration count or just exponentially increase.
    # Let's use initial alpha = 0.5, then increase when it seems stuck.
    # We can call _smooth_step.
    while time.time() < deadline:
        elapsed = time.time() - start
        # compute alpha: start small and grow to maybe 30
        fraction = elapsed / (deadline - start)
        alpha = 0.5 + 29.5 * fraction**0.5 # sqrt schedule to increase early slower then faster? maybe linear.
        # step size: decreasing with sqrt(iteration+1)
        lr = 0.5 / np.sqrt(iteration + 1)  # or maybe 0.2?
        # compute convolution b
        b = _conv_fft(a)
        max_b = b.max()
        # compute weights p
        # shift for numeric stability
        exp_arg = alpha * (b - max_b)
        # avoid overflow: clip exp_arg
        # exp_arg <= 0 always, because b <= max_b; min may be -alpha*range.
        # exp_arg min ~ -alpha*max(b) maybe large negative, but np.exp handles.
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to uniform
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum
        # compute gradient using correlation
        # compute appropriate FFT length
        L = 1 << ((3 * n - 2).bit_length())
        fa = np.fft.rfft(a, n=L)
        fp = np.fft.rfft(p, n=L)
        # correlation: conj(fa) * fp
        corr = np.fft.irfft(np.conj(fa) * fp, n=L)
        grad = 2.0 * corr[:n]
        # step
        a = a - lr * grad
        # projection onto simplex
        a = _project_to_simplex(a, target_sum)
        # compute new score (max(b))
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD iter {iteration} best={best_val:.6f}")
        iteration += 1
        # maybe add jitter slowly
        if iteration % 1000 == 0:
            jitter = np.random.default_rng().normal(scale=target_sum * 1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)
    return best_a, best_val

"""
Implementation of a two‑phase optimizer for the step‑function
max‑convolution problem. Phase 1 performs a smoothed gradient‑
descent on the log‑sum‑exp approximation of the max entry, staying
on the simplex by Euclidean projection. Phase 2 refines the
solution by a fast O(n) simulated‑annealing mass‑transfer move
similar to the previous state‑of‑the‑art. Multiple random
restarts and a warm‑start from a previous run are employed.
"""

import time
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _score(a: np.ndarray) -> float:
    """Objective = max self‑convolution entry (since the sum is fixed)."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    # denominator (sum^2) = 2n because s = sqrt(2n)
    return float(max_b)

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, target_sum: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    """Gradient descent on a log‑sum‑exp surrogate of the max convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    # Pre‑compute FFT length (>= 3n‑2) – enough for both convolution and correlation.
    L = 1 << ((3 * n - 2).bit_length())

    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() < deadline:
        # forward FFT of current a (used for both conv and correlation)
        fa = np.fft.rfft(a, n=L)

        # full self‑convolution (first 2n‑1 entries are relevant)
        conv_full = np.fft.irfft(fa * fa, n=L)
        b = conv_full[: 2 * n - 1]
        max_b = b.max()

        # log‑sum‑exp weights (smooth approximation of the argmax)
        # alpha grows from 0.5 to 20 as the run progresses.
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        alpha = 0.5 + 19.5 * frac   # range [0.5, 20.0]

        w = np.exp(alpha * (b - max_b))      # max entry gets weight 1, others <1
        w_sum = w.sum()
        if w_sum == 0.0:
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum

        # gradient = 2 * (a ⋆ p) where ⋆ is the (non‑circular) correlation
        fp = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(np.conj(fa) * fp, n=L)   # length L
        grad = 2.0 * corr[:n]

        # step size – decreases like 1/√t
        lr = 0.8 / np.sqrt(it + 1.0)

        # gradient step + projection onto the simplex
        a = a - lr * grad
        a = _project_to_simplex(a, target_sum)

        # occasional small jitter to avoid stagnation
        if it % 1000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 1e-5, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

        # evaluate true objective (max convolution entry)
        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:6d} best={best_val:.6f}")

        it += 1

    return best_a, best_val

def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """Simulated‑annealing that moves tiny amounts of mass between entries."""
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact self‑convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9   # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))   # square‑root cooling

        # locate current peak of b
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – more aggressive than before
        max_delta = min(a[donor] * 0.12, target_sum * 0.015)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(b_new.max())

        # Metropolis acceptance
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it} improved best={best_val:.6f}")

        # periodic full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute it={it} best={best_val:.6f}")

        it += 1

    return best_a, best_val

        print(f"[{time.time():.2f}] Final refinement on global best")
        extra_deadline = time.time() + 0.2 * (deadline - time.time())
        best_seq, best_score = _smooth_gradient_descent(best_seq, target_sum, extra_deadline)
        if time.time() < deadline:
            best_seq, best_score = anneal_mass_transfer(best_seq, target_sum, deadline)

    # ------------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – power‑law if everything else failed
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # Log the final evaluated score (only for information)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
