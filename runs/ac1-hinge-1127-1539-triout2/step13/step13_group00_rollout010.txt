"""
Hybrid optimisation for the step‑function challenge.

The optimiser builds a diverse set of feasible seeds (power‑law,
exponential, uniform, random simplex and the previous best), runs a
projected Differential Evolution (DE) on the ℓ¹‑simplex, refines the
best individual with a deterministic mass‑transfer routine and a short
sub‑gradient descent, and then performs an aggressive permutation
search (random full shuffles, simulated‑annealing‑style pair swaps and a
few handcrafted reorderings such as alternating, even‑spread and a
binary‑tree layout).  A final tiny polishing mass‑transfer finishes the
solution before the 1000 s budget expires.

The goal is to minimise 2·n·max(conv(a,a))/sum(a)², which for the
normalisation sum(a)=√(2n) reduces to minimising max(conv(a,a)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (Σa)² (for Σa=√(2n) this equals max‑conv)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised to the required ℓ¹‑norm."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniformly random point on the ℓ¹‑simplex (Dirichlet‑like)."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers (often lower the max‑convolution)
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _binary_tree_order(seq: np.ndarray) -> np.ndarray:
    """
    Place the largest values hierarchically: recursively put the next
    value in the middle of the largest remaining interval.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]  # descending
    out = np.empty_like(seq)
    intervals = [(0, n - 1)]
    idx = 0
    while intervals and idx < n:
        start, end = intervals.pop(0)   # BFS order gives good spread
        if start > end:
            continue
        mid = (start + end) // 2
        out[mid] = sorted_vals[idx]
        idx += 1
        intervals.append((start, mid - 1))
        intervals.append((mid + 1, end))
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]
        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert the move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that uses the exact max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]
        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 5e-4, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Differential‑evolution loop with simplex projection after each step."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser for the step‑function problem.

    Returns a list of non‑negative heights that (hopefully) improves on the
    previous best bound.  The routine respects the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives finer granularity)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 500:
        n = 6000
    else:
        n = 4000
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Seed generation
    # --------------------------------------------------------------
    seeds = []

    # uniform baseline
    seeds.append(np.full(n, target_sum / n, dtype=np.float64))

    # power‑law family
    for p in np.linspace(0.15, 1.2, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(5e-4, 0.02, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(10):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best (if available)
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter each seed a little for extra diversity
    jittered = []
    for s in seeds:
        jitter = rng.normal(scale=target_sum * 0.02, size=n)
        s2 = s + jitter
        s2 = np.maximum(s2, 0.0)
        s2 = _project_to_simplex(s2, target_sum)
        jittered.append(s2)
    seeds.extend(jittered)

    # --------------------------------------------------------------
    # Initialise DE population
    # --------------------------------------------------------------
    pop_size = max(30, n // 250)            # about 30‑35 individuals
    population = []

    # start with the best seed we have
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # --------------------------------------------------------------
    # Differential Evolution phase (≈70 % of the budget)
    # --------------------------------------------------------------
    de_deadline = start_time + 0.70 * budget_s
    if de_deadline > deadline - 5.0:
        de_deadline = deadline - 5.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)
    print(f"[{time.time():.2f}] DE finished, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic mass‑transfer polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 12.0:
        mt_dead = time.time() + min(18.0, deadline - time.time() - 9.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.5, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Short sub‑gradient descent refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 7.0:
        sg_dead = time.time() + min(12.0, deadline - time.time() - 3.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_seq, best_score = seq_sg, score_sg
            print(f"[{time.time():.2f}] After sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Aggressive permutation search (random full shuffles)
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        perm_dead = deadline - 2.0
        attempts = 0
        while time.time() < perm_dead and attempts < 5000:
            perm = rng.permutation(n)
            cand = best_seq[perm]
            cand_score = _max_convolution_fft(cand)
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                # keep the best_{seq} for further phases
            attempts += 1
        print(f"[{time.time():.2f}] After random shuffles best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing style pair‑swap local search
    # --------------------------------------------------------------
    if time.time() < deadline - 3.0:
        swap_dead = deadline - 1.0
        cur_seq = best_seq.copy()
        cur_max = best_score
        total_iters = 0
        while time.time() < swap_dead and total_iters < 20000:
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            # perform swap
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            new_max = _max_convolution_fft(cur_seq)
            if new_max < cur_max:
                cur_max = new_max
                best_seq = cur_seq.copy()
                best_score = new_max
                # keep swap
            else:
                # linear cooling temperature
                elapsed = time.time() - start_time
                remaining = deadline - start_time
                temp = max(1e-4, (remaining - elapsed) / remaining)
                prob = np.exp(-(new_max - cur_max) / temp)
                if rng.random() < prob:
                    cur_max = new_max
                    # keep sub‑optimal swap (helps exploration)
                else:
                    # revert
                    cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            total_iters += 1
        print(f"[{time.time():.2f}] After swap SA best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Additional handcrafted reorderings
    # --------------------------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq),
        _binary_tree_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After reordering best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Tiny final polishing (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        fin_dead = time.time() + min(3.0, deadline - time.time() - 1.0)
        seq_fin, score_fin = _deterministic_mass_transfer(
            best_seq, fin_dead, target_sum,
            start_temp=0.3, end_temp=1e-5, low_frac=0.06)
        if score_fin < best_score:
            best_seq, best_score = seq_fin, score_fin
            print(f"[{time.time():.2f}] Final MT best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # --------------------------------------------------------------
    # Store for the next run and return
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # noqa: F821
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
