Hybrid optimizer for the step‑function max‑convolution problem.

The routine builds many diverse seeds, runs a projected Differential‑Evolution
(DE) phase, then alternates two deterministic refinements:
    • greedy mass‑transfer (simulated‑annealing style),
    • sub‑gradient descent on the exact max‑convolution.
When time permits a fast random‑permutation search explores better orderings.
All steps respect the 1000 s budget and keep the sequence feasible
(non‑negative, values ≤ 1000).  The best sequence found is stored in the global
variable ``height_sequence_1`` for warm‑starting future runs.
"""

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    If Σa is very small we return +inf (invalid).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x_i ≥ 0, Σx_i = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories (all return vectors that sum to `target_sum`)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _triangle_sequence(n: int, target_sum: float) -> np.ndarray:
    """Symmetric triangular (tent) shape, normalised to target_sum."""
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * x - 1.0)      # peak = 1, linearly down to 0
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Ordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread large values uniformly by a coprime step."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


def _random_ordering_search(base_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast random permutation search.  The multiset of values is kept
    (sorted descending) and many random permutations are tried until the
    deadline is reached.  The best ordering (smallest max‑convolution) is
    returned.
    """
    rng = np.random.default_rng()
    sorted_vals = np.sort(base_seq)[::-1]  # descending
    n = sorted_vals.shape[0]
    best_seq = sorted_vals.copy()
    best_val = _max_convolution_fft(best_seq)

    # try a few deterministic orderings first (they are cheap)
    for fn in (_alternating_order, _even_spread_order):
        candidate = fn(sorted_vals)
        val = _max_convolution_fft(candidate)
        if val < best_val:
            best_val, best_seq = val, candidate.copy()

    # random permutations while time permits
    while time.time() < deadline:
        perm = rng.permutation(n)
        cand = sorted_vals[perm]
        val = _max_convolution_fft(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy mass moves with a decaying Metropolis acceptance rule.
    Moves a small amount of mass from the index contributing the most
    to the current peak of the convolution to one contributing the least.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.5,
                         gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the (exact) set of maximising
    convolution indices.  Momentum and occasional jitter help to escape
    shallow plateaus.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        max_idxs = np.where(conv >= maxb * (1 - 1e-12))[0]

        grad = np.zeros_like(a)
        n = a.shape[0]
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 400 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Differential‑Evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Classic DE loop with simplex projection after mutation and crossover."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            # three distinct mates
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        # leave a few seconds for the final polishing phase
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid search for a non‑negative step function with a low evaluation score.
    The routine builds a modest population, runs a projected differential‑evolution,
    then repeatedly applies deterministic mass‑transfer, sub‑gradient descent,
    and a fast random‑ordering search.  The best sequence found before the deadline
    is returned (and stored globally as ``height_sequence_1`` for warm‑starting).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # ------------------------------------------------------------------
    # Choose a problem size (larger n gives richer search space)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8000
    elif budget_s >= 400:
        n = 6000
    else:
        n = 4000
    target_sum = math.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # Build a pool of diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.2, 11):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.025, 9):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # a single triangular shape (often good)
    seeds.append(_triangle_sequence(n, target_sum))

    # random Dirichlet‑like points
    for _ in range(6):
        seeds.append(_random_simplex(n, target_sum, rng))

    # warm‑start from the previous run if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jitter every seed a little to increase diversity
    jitter_scale = target_sum * 0.02
    for seq in list(seeds):
        seq_j = seq + rng.normal(scale=jitter_scale, size=n)
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)

    # ------------------------------------------------------------------
    # Initialise the DE population
    # ------------------------------------------------------------------
    pop_size = max(30, int(n / 350))   # about 20‑30 individuals
    population = []

    # best seed goes first
    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        s = rng.choice(seeds)
        population.append(_project_to_simplex(s.copy(), target_sum))

    # ------------------------------------------------------------------
    # DE phase (roughly 45 % of the total budget)
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.45 * budget_s
    de_deadline = min(de_deadline, deadline - 5.0)
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline)

    print(f"[{time.time():.2f}] DE finished, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Alternating refinement loops (mass‑transfer + sub‑gradient)
    # ------------------------------------------------------------------
    while True:
        remaining = deadline - time.time()
        if remaining < 15.0:
            break

        # ---- deterministic mass‑transfer (≈12 s) ----
        mt_dead = time.time() + min(12.0, remaining - 8.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.4, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt.copy(), score_mt
            print(f"[{time.time():.2f}] Mass‑transfer improved to {best_score:.6f}")

        remaining = deadline - time.time()
        if remaining < 8.0:
            break

        # ---- sub‑gradient descent (≈6 s) ----
        sg_dead = time.time() + min(6.0, remaining - 4.0)
        seq_sg, score_sg = _subgradient_descent(
            best_seq, sg_dead, target_sum,
            step_factor=0.45, gamma=0.995)
        if score_sg < best_score:
            best_seq, best_score = seq_sg.copy(), score_sg
            print(f"[{time.time():.2f}] Sub‑gradient improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Random ordering search (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        order_dead = deadline - 1.0
        seq_ord, val_ord = _random_ordering_search(best_seq, order_dead)
        if val_ord < _score(best_seq):
            best_seq = seq_ord.copy()
            best_score = val_ord
            print(f"[{time.time():.2f}] Ordering search gave {best_score:.6f}")

    # ------------------------------------------------------------------
    # Trim trailing near‑zero entries for compactness
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Store for the next run and report the final evaluated score
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = final_seq
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
