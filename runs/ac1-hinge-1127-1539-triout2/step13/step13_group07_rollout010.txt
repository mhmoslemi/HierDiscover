    m = np.zeros_like(a)

    it=0
    best_a = a.copy()
    best_score = np.inf
    start = time.time()
    while time.time() - start < time_budget and it < 5000:

"""
Step‑function optimizer – improved search.

The algorithm combines several complementary ideas:

1.  **Fast parametric sweep** – evaluates a few analytic families
    (power‑law, power‑law + exponential, sine) and keeps the best.

2.  **Random restarts** – generates many Dirichlet‑distributed seeds
    (both uniform and sparse) and runs an Adam optimizer on a *spectral‑variance*
    surrogate.  Flattening the squared‑magnitude of the Fourier transform
    tends to flatten the self‑convolution, thus lowering the true max‑convolution.

3.  **Deterministic mass‑transfer** – a greedy “exact DMT2’’ routine that
    moves mass from a donor index to a receiver index until the current peak
    equals the second‑largest convolution entry.

4.  **Soft‑max Adam polish** – final refinement using the smooth log‑sum‑exp
    approximation of the max to drive the true objective down.

All candidates are projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)} after each update.
Trailing near‑zero entries are stripped before returning.
The routine respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real output)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Parametric families (used for the initial quick sweep)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                           target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy reduction of the maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Spectral‑variance gradient (Adam optimiser)
# ----------------------------------------------------------------------


def _grad_variance(a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Gradient of the variance of |FFT(a)|² w.r.t. a.
    a is assumed to be of length n ≤ fft_size; entries beyond n are zero‑padded.
    """
    n = a.shape[0]
    pad = np.zeros(fft_size, dtype=np.float64)
    pad[:n] = a
    A = np.fft.fft(pad)                     # complex FFT of length fft_size
    B = np.abs(A) ** 2                      # |A|², real
    mu = B.mean()
    C = (B - mu) * np.conj(A)               # (B‑μ)·A*
    G_full = 4.0 * np.real(np.fft.fft(C))   # gradient for the padded vector
    return G_full[:n]                       # only the first n components are needed


def _adam_variance(a0: np.ndarray,
                   target_sum: float,
                   fft_size: int,
                   deadline: float,
                   max_iters: int = 5000,
                   step_factor: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the spectral‑variance surrogate.
    Returns the best vector seen (projected onto the simplex) and its max‑convolution.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_fft(a)

    start = time.time()
    it = 0
    while time.time() < deadline and it < max_iters:
        grad = _grad_variance(a, fft_size)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional evaluation (every 30 iters)
        if it % 30 == 0:
            cur_score = _max_convolution_fft(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] Variance‑Adam it={it:5d} best={best_score:.6f}")

        it += 1

    return best_a, best_score

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(b)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_softmax(a0: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  max_beta: float = 2e5,
                  step_factor: float = 0.8,
                  gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max surrogate of the max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * grad
        v = gamma * v + (1 - gamma) * (grad * grad)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_to_simplex(a, target_sum)

        # jitter to avoid plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Softmax‑Adam it={it:5d} β={beta:6.1f} best={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal self‑convolution.
    The routine respects the supplied time budget and returns the best list found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom)
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 16000
    elif budget_s >= 600:
        n = 13000
    elif budget_s >= 400:
        n = 10000
    else:
        n = 6000

    target_sum = np.sqrt(2.0 * n)          # Σ a = √(2 n) makes the objective = max‑conv
    fft_size = 1 << ((2 * n - 1).bit_length())   # size for all FFT‑based routines

    # ------------------------------------------------------------------
    # Phase 0 – fast parametric sweep (same families as before)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # grids for power‑law exponent and exponential tail strength
    p_grid = np.linspace(0.20, 1.20, 31)
    q_grid = np.linspace(0.0, 10.0, 31)

    for p in p_grid:                                   # pure power‑law
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _max_convolution_fft(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    for p in p_grid:                                   # power‑law + exponential tail
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    seq = _sin_sequence(n, target_sum)                 # simple sine shape
    val = _max_convolution_fft(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # incorporate previously found construction if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    rng = np.random.default_rng()
