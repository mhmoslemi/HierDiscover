"""
Hybrid SGD + mass‑transfer + adaptive penalty descent for the
max‑self‑convolution problem.

The sequence a=(a0,…,a_{n‑1}) is kept on the simplex
    Σ a_i = √(2·n),   a_i ≥ 0.
The objective is  max_k (a*a)_k  (the maximal entry of the linear
convolution).  The code proceeds in several stages:

* 0️⃣  Quick parametric sweep (power‑law, exponential‑tail,
  Gaussian, sine, semi‑circle, raised‑cosine) to obtain a decent starter.
* A️⃣  Adam‑style projected gradient descent on a smooth soft‑max
  approximation of the max‑convolution.
* B️⃣  Exact sub‑gradient descent that moves mass from the donor index
  that contributes most to the current maximum toward a receiver.
* C️⃣  Efficient random mass‑transfer steps that update the full
  convolution in O(n) using the explicit shift formula.
* D️⃣  Adaptive penalty‑descent: repeatedly try to push the maximal
  convolution below a shrinking target (binary‑search‑style).

All operations respect the simplex constraint (exactly by moving mass
or by a Euclidean projection).  The algorithm automatically chooses a
problem size n up to 40 000 depending on the allotted time budget.
The final sequence is trimmed of trailing near‑zero entries and returned.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (used for the heavy gradient phases)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of arbitrary length vectors x and y using FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution) via FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    inde = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / inde > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (quick pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _semicircle_sequence(n: int, target_sum: float) -> np.ndarray:
    """Discretised semi‑circle (arc) shape."""
    i = np.arange(n, dtype=np.float64)
    # centre at (n‑1)/2, radius = (n‑1)/2
    radius = (n - 1) / 2.0
    dist = np.abs(i - radius)
    a = np.sqrt(np.maximum(radius ** 2 - (i - radius) ** 2, 0.0))
    a *= target_sum / a.sum()
    return a


def _raised_cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Raised cosine, non‑negative on the whole interval."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(np.pi * (x - (n - 1) / 2.0) / (n / 2.0)))
    a = np.maximum(a, 0.0)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (max‑convolution value)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective: maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    # Prevent overflow for very large beta by shifting
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape tiny plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)
            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        for _ in range(6):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        else:
            jitter = rng.normal(scale=target_sum * 0.0005, size=a.shape)
            a = _project_to_simplex(a + jitter, target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty‑descent (push max below a given target)
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float) -> tuple[np.ndarray, float]:
    """
    Try to drive the maximal convolution entry down to ≤ target_max.
    Uses the penalty (max(conv)‑target_max)_+ with a sub‑gradient step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and best_max > target_max + 1e-12 and it < 20000:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty vector (zero if below target)
        p = conv - target_max
        p[p < 0] = 0.0

        # gradient = 2 * (p * a_rev)_{n‑1 … n‑1+n‑1}
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Fast random mass‑transfer (O(n) update of the full convolution)
# ----------------------------------------------------------------------


def _random_mass_transfer(a: np.ndarray,
                         conv: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         current_best: float) -> tuple[np.ndarray, float]:
    """
    Monte‑Carlo style transfer of a small amount of mass from a heavy
    index to a light one, updating the full convolution in O(n) using the
    explicit shift formula (see analysis).  Accept only strictly
    improving moves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_conv = conv.copy()
    best_val = current_best
    rng = np.random.default_rng()

    # fractions of the population used to pick donors / receivers
    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 0.5:
        k_top = max(1, int(top_frac * n))
        k_low = max(1, int(low_frac * n))

        # pick donor among the k_top largest entries
        idx_desc = np.argpartition(-a, k_top - 1)[:k_top]
        donor = rng.choice(idx_desc)

        # pick receiver among the k_low smallest entries
        idx_asc = np.argpartition(a, k_low - 1)[:k_low]
        receiver = rng.choice(idx_asc)

        if donor == receiver:
            continue

        # max fraction of donor's mass to move
        max_delta = a[donor] * 0.08
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # --- update convolution ---
        conv_new = conv.copy()
        # subtract contribution of donor
        conv_new[donor:donor + n] -= 2.0 * delta * a
        # add contribution of receiver
        conv_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic adjustments
        conv_new[2 * donor] += delta * delta
        conv_new[2 * receiver] += delta * delta
        conv_new[donor + receiver] -= 2.0 * delta * delta

        # --- update coefficients ---
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        cur_max = conv_new.max()
        if cur_max + 1e-12 < best_val:
            best_val = cur_max
            best_a = a_new
            best_conv = conv_new
            a = a_new
            conv = conv_new
            print(f"[{time.time():.2f}] Random transfer improved best={best_val:.6f}")

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution.  The routine combines a parametric sweep,
    Adam‑style gradient descent, exact sub‑gradient steps, fast random
    mass transfers, and an adaptive penalty‑descent phase.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more flexibility)
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 40000
    elif budget_s >= 500:
        n = 30000
    elif budget_s >= 300:
        n = 20000
    else:
        n = 12000

    target_sum = float(np.sqrt(2.0 * n))

    # ------------------------------------------------------------------
    # Phase 0 – dense parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # pure power‑law
    for p in np.linspace(0.25, 0.90, 17):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # power‑law + exponential tail
    for p in np.linspace(0.25, 0.90, 13):
        for q in np.linspace(0.0, 8.0, 9):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # Gaussian bumps
    for sigma in np.linspace(n / 20.0, n / 5.0, 7):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # semi‑circle
    seq = _semicircle_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # raised cosine
    seq = _raised_cosine_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # if a previous global solution exists, resize it
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # small jittered population around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After pre‑search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (soft‑max approximation)
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.008, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Adam restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (using true max)
    # ------------------------------------------------------------------
    subgrad_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         subgrad_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Fast random mass‑transfer polishing
    # ------------------------------------------------------------------
    # Pre‑compute the convolution once for the first call
    conv = _conv_fft(best_seq)
    rand_deadline = time.time() + (deadline - time.time()) * 0.20
    a_rand, val_rand = _random_mass_transfer(best_seq,
                                             conv,
                                             target_sum,
                                             rand_deadline,
                                             best_score)
    if val_rand < best_score:
        best_score = val_rand
        best_seq = a_rand.copy()
        print(f"[{time.time():.2f}] Random mass‑transfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Adaptive penalty descent (push bound lower)
    # ------------------------------------------------------------------
    penalty_deadline = time.time() + (deadline - time.time()) * 0.15
    # try to shrink the bound iteratively
    while time.time() < penalty_deadline - 2.0:
        # target is a modest fraction below current best, but never below 1.5
        gap = best_score - 1.5005
        if gap <= 0:
            break
        target = best_score - max(0.001, gap * 0.4)
        a_pen, val_pen = _penalty_lower_max(best_seq,
                                            target_sum,
                                            penalty_deadline,
                                            target)
        if val_pen < best_score - 1e-12:
            best_score = val_pen
            best_seq = a_pen.copy()
            print(f"[{time.time():.2f}] Penalty descent improved best = {best_score:.6f}")
        else:
            # couldn't meet the tighter target – stop trying smaller ones
            break

    # ------------------------------------------------------------------
    # Phase E – Final high‑β Adam polishing (optional)
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e6,
                                                step_factor=0.6,
                                                gamma=0.995)
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # ------------------------------------------------------------------
    # Final safety evaluation (optional – will be printed if available)
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
