"""
Propose candidate step function minimizing max self‑convolution.
Algorithm:
0. Choose problem size n based on time budget, set target sum = sqrt(2n).
1. Fast parametric sweep over several families (power‑law, power‑law+exp,
   sine) to obtain an initial feasible sequence.
2. Precise deterministic mass‑transfer: enumerate a few high‑value ("donor")
   indices and low‑value ("receiver") indices, compute the exact feasible
   step size that reduces the current convolution peak without creating a
   new larger peak, and perform the best move.  This loop runs until the
   deadline fraction allocated for phase 1.
3. Simulated annealing on the simplex to escape local minima.
4. Second deterministic mass‑transfer (same routine) for final tightening.
5. Adam‑style projected gradient descent on a soft‑max smoothed objective
   for polishing.
6. Trim trailing near‑zero entries and return the list of heights.
"""

import time
import numpy as np

# Helper functions ---------------------------------------------------------

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return float(_conv_fft(a).max())

def _score(a: np.ndarray) -> float:
    """Objective value (max convolution)."""
    return _max_convolution_fft(a)

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _shifted_array(a: np.ndarray, shift: int) -> np.ndarray:
    """Return array of length 2*len(a)-1 equal to a shifted right by `shift`."""
    n = a.shape[0]
    out_len = 2 * n - 1
    out = np.zeros(out_len, dtype=a.dtype)
    out[shift:shift + n] = a
    return out

# Parametric families -----------------------------------------------------

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a

def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a

def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a

def _exp_sequence(n: int, rate: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i / n)
    a *= target_sum / a.sum()
    return a

# Deterministic precise mass transfer -------------------------------------

def _deterministic_mass_transfer_precise(a0: np.ndarray,
                                         deadline: float,
                                         target_sum: float,
                                         top_frac: float = 0.02,
                                         bottom_frac: float = 0.02,
                                         tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Greedy reduction of the max convolution with exact feasibility check."""
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_fft(a)
    best_a = a.copy()
    best_score = conv.max()
    it = 0
    # pre‑allocate workspace for shifted arrays (filled each iteration)
    while time.time() < deadline:
        max_val = conv.max()
        mask_max = conv >= max_val - 1e-12
        # second largest entry
        conv2 = conv.copy()
        conv2[mask_max] = -np.inf
        second_max = conv2.max()

                delta_needed = (max_val - second_max) / (2.0 * min_diff)

# Adam gradient descent on soft‑max surrogate ------------------------------

def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on a smoothed max‑convolution objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # small jitter occasionally
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

"""

Algorithm
---------
0️⃣ Choose a problem size `n` based on the time budget; the L¹ norm is fixed
   to `Σ a = √(2 n)`, which makes the objective equal to `max(a*a)`.

1️⃣ Fast parametric sweep over several simple families (pure power‑law,
   power‑law + exponential tail, pure exponential, sinusoidal).  The best
   candidate from this inexpensive grid search seeds the subsequent phases.

2️⃣ Precise deterministic mass‑transfer:
   •Select a short list of “donor’’ indices (largest heights) and
    “receiver’’ indices (smallest heights).  
   •For every donor–receiver pair compute the exact feasible step size
    `δ` that reduces the current convolution peak while guaranteeing that
    no other entry exceeds the next‑largest value.  The pair giving the
    largest reduction is applied.  This loop runs until the allotted time
    expires or the peak cannot be lowered further.

3️⃣ Simulated annealing on the simplex to escape local minima.

4️⃣ Second pass of the precise deterministic mass‑transfer for final tightening.

5️⃣ Adam‑style projected gradient descent on a soft‑max smoothed surrogate
   of the max‑convolution, using an exponentially increasing β to
   approach the true max.

6️⃣ Trim trailing near‑zero heights and return the list.

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value – the maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the quick pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _exp_sequence(n: int, rate: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Helper: shifted version of a for a given index
# ----------------------------------------------------------------------


def _shifted_array(a: np.ndarray, shift: int) -> np.ndarray:
    """Return array of length 2·n‑1 equal to a shifted right by `shift`."""
    n = a.shape[0]
    out_len = 2 * n - 1
    out = np.zeros(out_len, dtype=a.dtype)
    out[shift:shift + n] = a
    return out

    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the soft‑max smoothed objective."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal self‑convolution.
    The function respects the time budget and returns the best sequence it discovered.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --- problem size ---------------------------------------------------
    if budget_s >= 900:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 12000
    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – fast parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    p_grid = np.linspace(0.05, 1.5, 31)
    q_grid = np.linspace(0.0, 15.0, 31)

    # (a) pure power‑law
    for p in p_grid:
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) pure exponential
    for rate in np.linspace(0.0, 10.0, 21):
        seq = _exp_sequence(n, float(rate), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (d) sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (e) previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # (f) jittered refinements around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – precise deterministic mass‑transfer (first tightening)
    # ------------------------------------------------------------------
    mt1_deadline = time.time() + (deadline - time.time()) * 0.15
    best_seq, best_score = _deterministic_mass_transfer_precise(best_seq,
                                                                mt1_deadline,
                                                                target_sum)
    print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing
    # ------------------------------------------------------------------
    sa_deadline = time.time() + (deadline - time.time()) * 0.45
    best_seq, best_score = _simulated_annealing(best_seq,
                                                target_sum,
                                                sa_deadline)
    print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – second precise deterministic mass‑transfer
    # ------------------------------------------------------------------
    mt2_deadline = time.time() + (deadline - time.time()) * 0.20
    best_seq, best_score = _deterministic_mass_transfer_precise(best_seq,
                                                                mt2_deadline,
                                                                target_sum)
    print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final Adam polishing
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      final_deadline,
                                                      target_sum,
                                                      max_beta=1e6,
                                                      step_factor=0.6,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # Optional final verification (harmless if evaluate_sequence is unavailable)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
