def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # f_i contributes to rows k = i .. i+n-1
        row_start = i
        row_end = i + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    # Convert to csr matrix shape (2n-1, n)
    C = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return C

"""
Advanced step‑function optimiser with LP‑based refinement.

The optimiser works in four stages:

1. **Exploration** – many short runs of soft‑max GD or exact sub‑gradient
   descent starting from a diverse pool of seeds (power‑law, exponential,
   random Dirichlet, previous best).  This quickly finds a decent candidate.

2. **LP refinement** – the current candidate is down‑scaled (≈1024 entries)
   and a small linear program is solved:
        maximise   Σ b_i
        subject to  (a * b)_k ≤ M   ∀k ,   b_i ≥ 0
   where M = max(a*a).  The LP produces a direction that can reduce the
   convolution peak.  The solution is scaled to the simplex, mixed with the
   current vector and projected back.  A few such iterations give a
   non‑trivial improvement.

3. **Upscaling & polishing** – the refined low‑dimensional sequence is
   interpolated to a larger length (≈12000) and polished with a high‑β
   soft‑max gradient descent followed by a short sub‑gradient pass.

4. **Final clean‑up** – trailing zeros are trimmed and the result is stored
   in the global variable ``height_sequence_1`` for reuse.

All steps respect the 1000 s budget, use only a single CPU, and keep the
vector on the ℓ¹‑simplex (Σ a_i = √(2·n)) to make the evaluation equal to
the convolution peak.
"""

import time
import numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT utilities (max‑convolution and full convolution)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) this equals the peak of the convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of sequences (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum‑annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum‑based GD on soft‑max approximation of the peak."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (much cheaper per step)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(convolution)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # exact gradient of the peak entry
        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = kmax - idx
        grad[idx] = a[partner]
        if kmax % 2 == 0:
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]          # derivative of a_mid^2 is 2*a_mid

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP‑based refinement (max‑convolution bound)
# ----------------------------------------------------------------------


def _conv_matrix_from_f(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix C such that (C @ b) = f * b (linear convolution).
    Row k contains the coefficient f_i for every i with 0≤i<n and 0≤k‑i<n.
    """
    n = f.shape[0]
    # rows: i + j for all i,j in [0,n)
    row = (np.arange(n)[:, None] + np.arange(n)).ravel()
    col = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    return csr_matrix((data, (row, col)), shape=(2 * n - 1, n))


def _lp_refinement(a0: np.ndarray,
                   deadline: float,
                   target_sum: float,
                   max_iters: int = 8) -> tuple[np.ndarray, float]:
    """
    LP‑based descent:
        maximise Σ b_i
        s.t.   (a * b)_k ≤ M  ∀k,   b_i ≥ 0,
    where M = max(a*a).  The solution is scaled to the simplex and mixed back.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)

    n = a.shape[0]
    t = 0.5                       # mixing weight
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        M = _max_convolution_fft(a)                 # current peak
        C = _conv_matrix_from_f(a)                  # (2n‑1)×n matrix

        b = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b)),
                           [C @ b <= M])            # broadcasting is fine
        try:
            prob.solve(solver='ECOS', warm_start=True,
                       max_iters=3000, abstol=1e-8, reltol=1e-6)
        except Exception as e:
            print(f"[{time.time():.2f}] LP solve error: {e}")
            break

        if b.value is None:
            print(f"[{time.time():.2f}] LP infeasible at iter {it}")
            break

        b_val = np.asarray(b.value).flatten()
        sum_b = b_val.sum()
        if sum_b <= 0:
            break

        # Scale to keep the ℓ¹‑norm fixed
        scale = target_sum / sum_b
        g = b_val * scale

        # Simple mixing and projection
        a_new = (1.0 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP iter={it} new best={best_score:.6f}")

        # Accept if we improved, otherwise shrink the step size
        if new_score < _score(a):
            a = a_new
        else:
            t = max(t * 0.5, 0.05)   # become more cautious

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing (swap high/low contributors)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.07,
                                 low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """Swap a high‑contributing index with a low‑contributing one."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low benchmark score.
    The routine performs:
      1. Exploration (short soft‑max GD / sub‑gradient runs)
      2. LP‑based refinement on a modest‑size down‑scaled version
      3. Upscaling + high‑β GD polishing + permutation annealing
      4. Final trimming.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Determine problem size for the exploration phase
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n_expl = 8000
    elif budget_s >= 500:
        n_expl = 6000
    else:
        n_expl = 4000
    target_expl = np.sqrt(2.0 * n_expl)

    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seeds
    # ------------------------------------------------------------------
    init_pool: list[np.ndarray] = []

    for p in np.linspace(0.35, 0.78, 9):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))

    for lam in np.linspace(0.0015, 0.015, 7):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))

    for _ in range(6):
        a = rng.random(n_expl)
        a *= target_expl / a.sum()
        init_pool.append(a)

    lin = np.linspace(1.0, 0.0, n_expl)
    lin *= target_expl / lin.sum()
    init_pool.append(lin)

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jittered copies
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.008, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    # ------------------------------------------------------------------
    # 3. Exploration (short GD / sub‑gradient runs)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # allocate ~30 % of the budget for exploration
    exploration_deadline = start_time + 0.30 * budget_s

    while time.time() < exploration_deadline:
        # pick a seed
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.004, size=n_expl)
            seed = seed + jitter
            seed = np.maximum(seed, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        if rng.random() < 0.6:
            run_dead = min(time.time() + 12.0, exploration_deadline)
            a_opt, val_opt = _annealed_gradient_descent(
                seed,
                run_dead,
                target_expl,
                max_beta=5e5,
                step_factor=0.7,
                gamma=0.99,
            )
        else:
            run_dead = min(time.time() + 10.0, exploration_deadline)
            a_opt, val_opt = _subgradient_descent(
                seed,
                run_dead,
                target_expl,
                max_iter=150_000,
            )

        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Exploration new best score={best_score:.6f}")

        if time.time() >= exploration_deadline:
            break

    # ------------------------------------------------------------------
    # 4. LP‑based refinement on a down‑scaled version
    # ------------------------------------------------------------------
    if best_seq is not None:
        # down‑scale to a manageable size for the LP (≈1024)
        n_lp = 1024
        seq_lp = _resize_sequence(best_seq, n_lp)
        target_lp = np.sqrt(2.0 * n_lp)
        seq_lp = _project_to_simplex(seq_lp, target_lp)

        lp_deadline = time.time() + 0.20 * budget_s   # give ~20 % to LP
        seq_lp, lp_score = _lp_refinement(
            seq_lp,
            lp_deadline,
            target_lp,
            max_iters=8,
        )
        if lp_score < best_score:
            best_score = lp_score
            best_seq = _resize_sequence(seq_lp, n_expl)  # bring back to exploration size
            best_seq = _project_to_simplex(best_seq, target_expl)

        print(f"[{time.time():.2f}] After LP refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscaling to a larger length for final polishing
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.55, target_expl)

    remaining = deadline - time.time()
    if remaining > 70.0 and budget_s >= 800:
        n_final = 12000
    elif remaining > 45.0:
        n_final = 10000
    else:
        n_final = n_expl

    target_final = np.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _reversed_alternating(seq_big)]
    best_seq = seq_big
    best_score = _score(seq_big)
    for cand in candidates[1:]:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
    print(f"[{time.time():.2f}] Upscaled to {n_final} items, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. High‑β GD polishing (takes the bulk of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        gd_deadline = deadline - 8.0
        a_final, val_final = _annealed_gradient_descent(
            best_seq,
            gd_deadline,
            target_final,
            max_beta=2e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_final < best_score:
            best_score = val_final
            best_seq = a_final.copy()
            print(f"[{time.time():.2f}] GD polish best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Targeted permutation annealing (uses the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        perm_deadline = deadline - 0.5
        a_perm, val_perm = _permutation_targeted_search(
            best_seq,
            perm_deadline,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.07,
            low_frac=0.07,
        )
        if val_perm < best_score:
            best_score = val_perm
            best_seq = a_perm.copy()
            print(f"[{time.time():.2f}] Permutation search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Tiny post‑processing sub‑gradient pass
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sub_dead = time.time() + min(3.0, deadline - time.time() - 1.0)
        a_sub, val_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=100_000,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(f"[{time.time():.2f}] Final sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 9. Trim trailing zeros and return
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional logging of the official evaluator
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
