def build_constraint_matrix(f, indices):
    n = len(f)
    rows = len(indices)
    C = np.zeros((rows, n), dtype=np.float64)
    for row_idx, k in enumerate(indices):
        # Compute contributions: g_j participates with weight f_{k-j} if 0 <= k-j < n
        # So for each j in [0,n):
        # weight = f[k - j] if 0 <= k-j < n else 0, which corresponds to reversed f slice.
        # Equivalent: C[row_idx, :] = np.concatenate( (f[:k+1][::-1], zeros(n-(k+1))) ) but careful.
        # Actually conv formula: conv[k] = sum_{i=0}^{n-1} f[i] * g[k - i], with g index out-of-range yields 0.
        # So g_j appears with coefficient f[k - j] if j <= k and k-j < n.
        # For each j from max(0, k-(n-1)) to min(k, n-1), coefficient = f[k - j].
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # for j in lo..hi inclusive:
        # coefficient f[k - j]
        # subarray of f: f[lo':hi'+1 reversed]
        # Actually need to assign to row slice j range: indices lo..hi inclusive
        j_range = np.arange(lo, hi + 1)
        coeffs = f[k - j_range]
        C[row_idx, lo:hi+1] = coeffs

    a = a0.copy()
    best_a = a.copy()
    best_max = _score(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute conv, M
        conv = _conv_fft(a)
        M = conv.max()
        # select K indices with largest conv values
        # we ensure K <= len(conv)
        idx_top = np.argpartition(-conv, min(K, conv.size)-1)[:K]  # unsorted top K
        # Build matrix
        n = a.size
        rows = idx_top.size
        C = np.zeros((rows, n), dtype=np.float64)
        for r, k in enumerate(idx_top):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j_range = np.arange(lo, hi + 1, dtype=np.int64)
            C[r, lo:hi+1] = a[k - j_range]
        # Solve LP

def _subgradient_descent(a0, target_sum, deadline, max_iter=1000):
    a = a0.copy()
    best_a = a.copy()
    best_max = _score(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]
        # Build gradient: use p indicator equally weighted at top indices
        p = np.zeros_like(conv)
        p[idx_max] = 1.0 / len(idx_max)
        # Compute gradient using FFT (as earlier in gradient_softmax)
        n = a.size
        size = 1 << ((2 * n - 1).bit_length())
        a_rev_fft = np.fft.rfft(a[::-1], n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2 * n - 1]
        grad = 2.0 * conv_rev[n-1:n-1+n]

        norm_sq = np.dot(grad, grad)
        if norm_sq == 0:
            break
        # step size

"""
Step‑function optimizer for the max‑self‑convolution objective.

The algorithm consists of:
1. Fast parametric sweep over power‑law families.
2. LP‑based refinement: given the current shape f, solve a linear
   program that finds a distribution g with maximal total mass while
   respecting the current convolution peak on a selected set of
   “binding’’ indices.  A convex combination (or a small step) with g
   provably cannot increase the peak.
3. Deterministic mass‑transfer (exact DMT) and simulated annealing
   as in the reference implementation.
4. A short sub‑gradient descent on the true ‖f∗f‖∞ to polish the
   result.
All phases respect the overall time budget (default 1000 s) and the
returned list is projected onto the simplex {a_i≥0, Σa_i=√(2 n)}.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Convenient wrapper returning the peak of the self‑convolution."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (used for the initial guess)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """A smooth positive sine shape."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact DMT) – same as reference code
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        # donor = index with largest partner value among viable donors
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # receiver = index (different from donor) with smallest partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # move a safe fraction of the donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (reference)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic mass‑transfer with Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.02
    top_k = max(1, int(top_frac * n))
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.3
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based refinement
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iter: int = 30,
               K: int = 250) -> tuple[np.ndarray, float]:
    """
    Iteratively improve the peak of the self‑convolution by solving a
    linear programme that maximises the total mass of a new candidate g
    under the current peak bound on a selected set of K largest
    convolution indices.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _score(a)
    n = a.size
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        M = conv.max()
        # pick up to K indices with largest convolution values
        K_eff = min(K, conv.size)
        # use argpartition for speed (unsorted)
        idx_top = np.argpartition(-conv, K_eff - 1)[:K_eff]

        # build constraint matrix C (shape K_eff × n)
        rows = idx_top.size
        C = np.zeros((rows, n), dtype=np.float64)
        for row, k in enumerate(idx_top):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j_start = lo
            j_end = hi + 1
            # coefficients are a[k - j] for j in [lo, hi]
            C[row, j_start:j_end] = a[k - np.arange(j_start, j_end, dtype=np.int64)]

        # linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # abort if the LP fails – likely numerical issues
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # rescale g to meet the simplex sum constraint
        g = g * (target_sum / sum_g)

        # convex combination; start with a fairly aggressive step
        t = 0.5 if it == 0 else 0.3
        a_new = (1.0 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_max = _score(a_new)
        if new_max < best_max - 1e-12:
            best_max = new_max
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP it={it} best max={best_max:.6f}")

        # accept only if the peak does not increase
        if new_max <= _score(a) + 1e-12:
            a = a_new
        else:
            # try a smaller step as a fallback
            t_small = 0.1
            a_new = (1.0 - t_small) * a + t_small * g
            a_new = _project_to_simplex(a_new, target_sum)
            new_max2 = _score(a_new)
            if new_max2 < _score(a) - 1e-12:
                a = a_new
                print(f"[{time.time():.2f}] LP it={it} small‑t improved max={new_max2:.6f}")
            else:
                # no progress – stop the LP loop
                break

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Sub‑gradient descent on the max‑convolution (exact subgradient)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 500) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true ℓ∞‑norm of the
    self‑convolution.  The sub‑gradient is obtained from the set of
    maximal indices and a simple line‑search is performed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _score(a)
    n = a.size
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # build the sub‑gradient: p is uniform over the maximal indices
        p = np.zeros_like(conv)
        p[idx_max] = 1.0 / len(idx_max)

        size = 1 << ((2 * n - 1).bit_length())
        a_rev_fft = np.fft.rfft(a[::-1], n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2 * n - 1]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        norm_sq = float(np.dot(grad, grad))
        if norm_sq < 1e-18:
            break

        # step size that would (in the linearised model) drive the peak to zero
        step = max_val / norm_sq * 0.5
        a_candidate = a - step * grad
        a_candidate = _project_to_simplex(a_candidate, target_sum)
        new_max = _score(a_candidate)

        if new_max < best_max - 1e-12:
            best_max = new_max
            best_a = a_candidate.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it} best max={best_max:.6f}")

        # accept if we improve, otherwise back‑off the step
        if new_max < max_val:
            a = a_candidate
        else:
            # try halving the step a few times
            improved = False
            for _ in range(3):
                step *= 0.5
                a_candidate = a - step * grad
                a_candidate = _project_to_simplex(a_candidate, target_sum)
                new_max = _score(a_candidate)
                if new_max < max_val:
                    a = a_candidate
                    improved = True
                    break
            if not improved:
                # no progress – terminate
                break

        it += 1

    return best_a, best_max

    if val < best_score:

    # previous global best if available

        if prev.size != n:
            # interpolate to new length
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)

        val = _score(cand)

"""
Step‑function optimizer for the max‑self‑convolution objective.

The algorithm proceeds in a few phases, each respecting the global
time budget (default 1000 s):

1. **Fast parametric sweep** – evaluates simple power‑law families and a
   sine shape.  The best candidate is retained.

2. **LP‑based refinement** – given the current shape `f`, a linear
   programme maximises the total mass of a new candidate `g` under the
   current convolution peak `M = max(f*f)` on a selected set of the
   `K` largest convolution indices.  A convex combination
   `(1‑t)·f + t·g` (with a moderate step size) provably cannot increase
   the peak, and usually reduces it.

3. **Deterministic mass‑transfer (exact DMT)** – a greedy move that
   transfers mass from a “donor’’ index to a “receiver’’ index until the
   peak drops to the second‑largest value.

4. **Simulated annealing** – stochastic mass transfers with a Metropolis
   rule to escape local minima.

5. **Second DMT pass** – further tightening.

6. **Exact sub‑gradient descent** – projected sub‑gradient steps on the
   true ℓ∞‑norm of the self‑convolution to polish the result.

Each intermediate candidate is projected onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)}.  Finally trailing near‑zero entries are
removed and the list is returned.  The routine prints progress messages
and the final evaluated objective.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Peak of the self‑convolution (objective when Σa = √(2 n))."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families (initial guess)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """Positive sine‑shaped candidate."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact DMT) – reference implementation
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore all current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # pick one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (reference)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Stochastic mass‑transfer with Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.02
    top_k = max(1, int(top_frac * n))
    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.3
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _score(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based refinement
# ----------------------------------------------------------------------


def _lp_refine(a0: np.ndarray,
               target_sum: float,
               deadline: float,
               max_iter: int = 30,
               K: int = 250) -> tuple[np.ndarray, float]:
    """
    Iteratively improve the peak of the self‑convolution by solving a
    linear programme that maximises the total mass of a new candidate g
    under the current peak bound on a selected set of the K largest
    convolution indices.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _score(a)
    n = a.size
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        M = conv.max()
        K_eff = min(K, conv.size)
        idx_top = np.argpartition(-conv, K_eff - 1)[:K_eff]

        rows = idx_top.size
        C = np.zeros((rows, n), dtype=np.float64)
        for row, k in enumerate(idx_top):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j_start = lo
            j_end = hi + 1
            C[row, j_start:j_end] = a[k - np.arange(j_start,
                                                    j_end,
                                                    dtype=np.int64)]

        c = -np.ones(n, dtype=np.float64)               # maximise sum(g)
        b = np.full(rows, M, dtype=np.float64)          # (f * g) ≤ M
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Scale g to satisfy the simplex sum constraint
        g = g * (target_sum / sum_g)

        # Convex combination – aggressive first step, then moderate
        t = 0.5 if it == 0 else 0.3
        a_new = (1.0 - t) * a + t * g
        a_new = _project_to_simplex(a_new, target_sum)

        new_max = _score(a_new)
        if new_max < best_max - 1e-12:
            best_max = new_max
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] LP it={it} best max={best_max:.6f}")

        # Accept the move only if the peak does not increase.
        if new_max <= _score(a) + 1e-12:
            a = a_new
        else:
            # Try a smaller step as a safeguard.
            t_small = 0.1
            a_new = (1.0 - t_small) * a + t_small * g
            a_new = _project_to_simplex(a_new, target_sum)
            new_max2 = _score(a_new)
            if new_max2 < _score(a) - 1e-12:
                a = a_new
                print(f"[{time.time():.2f}] LP it={it} small‑t improved max={new_max2:.6f}")
            else:
                break

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the ℓ∞‑norm of the self‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 500) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(f*f)."""
    a = a0.copy()
    best_a = a.copy()
    best_max = _score(a)
    n = a.size
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # Build the sub‑gradient (uniform weight on all maximal indices)
        p = np.zeros_like(conv)
        p[idx_max] = 1.0 / len(idx_max)

        size = 1 << ((2 * n - 1).bit_length())
        a_rev_fft = np.fft.rfft(a[::-1], n=size)
        p_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(a_rev_fft * p_fft, n=size)[:2 * n - 1]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        norm_sq = float(np.dot(grad, grad))
        if norm_sq < 1e-18:
            break

        # Step size derived from the linearised model
        step = max_val / norm_sq * 0.5
        a_candidate = a - step * grad
        a_candidate = _project_to_simplex(a_candidate, target_sum)
        new_max = _score(a_candidate)

        if new_max < best_max - 1e-12:
            best_max = new_max
            best_a = a_candidate.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it} best max={best_max:.6f}")

        if new_max < max_val:
            a = a_candidate
        else:
            # Back‑off the step a few times
            improved = False
            for _ in range(3):
                step *= 0.5
                a_candidate = a - step * grad
                a_candidate = _project_to_simplex(a_candidate, target_sum)
                new_max = _score(a_candidate)
                if new_max < max_val:
                    a = a_candidate
                    improved = True
                    break
            if not improved:
                break

        it += 1

    return best_a, best_max
