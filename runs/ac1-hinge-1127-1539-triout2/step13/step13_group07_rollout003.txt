"""
Step‑function optimizer for the max‑self‑convolution problem.

The algorithm proceeds in the following stages (time‑budget aware):

1. **Rich parametric sweep** – evaluates a variety of simple families:
   * shifted power‑law,
   * power‑law with exponential tail,
   * beta‑shaped (i^{α‑1}(n‑i)^{β‑1}) sequences.
   A coarse grid of parameters is explored and the best candidate kept.
   The previous best construction (global ``height_sequence_1``) is also
   considered.

2. **Deterministic mass‑transfer (exact DMT2)** – a greedy routine that
   moves mass from a “donor’’ index (large partner value at the current
   convolution peak) to a “receiver’’ index (small partner value) so that
   the current peak drops to the second‑largest value.

3. **Simulated annealing** – stochastic mass moves (donor↔receiver) with
   Metropolis acceptance, allowing the search to escape local minima.

4. **Second deterministic mass‑transfer** – further tighten the peak.

5. **Projected Adam on a soft‑max surrogate** – smooth approximation of
   the max‑convolution (log‑sum‑exp) with an annealed temperature β.
   The iterates are projected onto the simplex Σa_i=√(2 n).

6. **Optional final DMT** – if time permits, a last greedy pass.

All phases respect the overall time budget (default 1000 s) and the
returned list is trimmed of trailing near‑zero entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _shifted_power_law_sequence(n: int, p: float, shift: float, target_sum: float) -> np.ndarray:
    """a_i = (i+shift)^{-p}, i starting from 1."""
    i = np.arange(1, n + 1, dtype=np.float64) + shift
    a = i ** (-p)
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    """a_i = (i)^{-p} * exp(-q * (i-1)/n)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _beta_shape_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑shaped sequence: (i)^{α‑1} * (n‑i+1)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    """Smooth sine‑shaped candidate (always positive)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (core refinement)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Greedy reduction of the maximal convolution entry.
    Moves a carefully chosen amount of mass from a “donor’’ index (large
    partner value) to a “receiver’’ index (small partner value) so that the
    current peak drops to the second‑largest value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # indices achieving the maximum (within tolerance)
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest entry (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        # choose one of the maximal indices (first)
        k = idx_max[0]

        # indices that contribute to k are i in [max(0,k-(n-1)), min(k,n-1)]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # partner values a_{k-i}
        partner = a[k - idx]

        # choose donor: index among idx where a[idx] > 0 and partner is large
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break
        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # choose receiver: index (different from donor) with smallest partner
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom

        # take a safe fraction of donor mass
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _max_convolution_fft(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    # re‑project to the simplex just in case numerical drift occurred
    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score

# Simulated annealing on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         target_sum: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer with Metropolis acceptance.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a)
    cur_score = best_score

    n = a.shape[0]
    rng = np.random.default_rng()

    # initial temperature (fraction of the current objective)
    T = max(best_score * 0.05, 1e-6)
    it = 0
    top_frac = 0.02                      # donor / receiver from top/bottom 2 %
    top_k = max(1, int(top_frac * n))

    sorted_desc = np.argsort(-a)
    sorted_asc = np.argsort(a)

    while time.time() < deadline:
        # refresh sorted lists occasionally (cost O(n log n))
        if it % 2000 == 0:
            sorted_desc = np.argsort(-a)
            sorted_asc = np.argsort(a)

        donor = sorted_desc[rng.integers(top_k)]
        receiver = sorted_asc[rng.integers(top_k)]
        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor]
        if max_delta <= 0.0:
            it += 1
            continue

        delta = max_delta * rng.random() * 0.3  # up to 30 % of donor mass
        if delta < 1e-12:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[donor] -= delta
        a_candidate[receiver] += delta

        new_score = _max_convolution_fft(a_candidate)
        delta_obj = new_score - cur_score

        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA improved best={best_score:.6f} it={it}")

        # cooling
        it += 1
        if it % 5000 == 0:
            T *= 0.9
            if T < 1e-7:
                T = 1e-7

    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_score

def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max (log‑sum‑exp) approximation of max(b).
    Returns (grad, b) where b = a*a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)                     # soft‑max weights

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]     # only the n components we keep
    return grad, b


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the log‑sum‑exp smoothed objective.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _max_convolution_fft(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    best_a = _project_to_simplex(best_a, target_sum)
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the maximal
    self‑convolution.  The function respects the time budget and returns
    the best list of heights found.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives more freedom, but costs more time)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 12000
    elif budget_s >= 400:
        n = 8000
    else:
        n = 5000
    target_sum = np.sqrt(2.0 * n)   # Σ a = √(2 n) makes the objective = max(b)

    # --------------------------------------------------------------
    # Phase 0 – enriched parametric sweep
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # grids
    p_grid = np.linspace(0.2, 1.2, 21)          # power‑law exponent
    shift_grid = np.linspace(0.0, 5.0, 11)      # shift for power‑law
    q_grid = np.linspace(0.0, 10.0, 11)        # exponential tail strength
    alpha_grid = np.linspace(0.2, 1.0, 9)      # beta‑shape left parameter
    beta_grid = np.linspace(1.0, 3.0, 9)       # beta‑shape right parameter

    # (a) shifted power‑law
    for p in p_grid:
        for sh in shift_grid:
            seq = _shifted_power_law_sequence(n, float(p), float(sh), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
    # (b) power‑law + exponential tail
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
    # (c) beta‑shape family
    for a in alpha_grid:
        for b in beta_grid:
            seq = _beta_shape_sequence(n, float(a), float(b), target_sum)
            val = _max_convolution_fft(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
    # (d) sine shape (simple baseline)
    seq = _sin_sequence(n, target_sum)
    val = _max_convolution_fft(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (e) previous best construction if available
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        prev = _project_to_simplex(prev, target_sum)
        val = _max_convolution_fft(prev)
        if val < best_score:
            best_score = val
            best_seq = prev.copy()
    except Exception:
        pass

    if best_seq is None:
        # fall back to uniform distribution (should never happen)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    print(f"[{time.time():.2f}] After parametric sweep best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – deterministic mass‑transfer (quick tightening)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    mt_deadline = time.time() + remaining * 0.15
    if time.time() < mt_deadline:
        best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                                   mt_deadline,
                                                                   target_sum)
        print(f"[{time.time():.2f}] After first DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated annealing (exploratory)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    sa_deadline = time.time() + remaining * 0.45
    if time.time() < sa_deadline:
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    target_sum,
                                                    sa_deadline)
        print(f"[{time.time():.2f}] After SA best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – second deterministic mass‑transfer
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    mt2_deadline = time.time() + remaining * 0.15
    if time.time() < mt2_deadline:
        best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                                   mt2_deadline,
                                                                   target_sum)
        print(f"[{time.time():.2f}] After second DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Adam gradient descent on soft‑max surrogate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        final_deadline = deadline
        best_seq, best_score = _adam_gradient_descent(best_seq,
                                                      final_deadline,
                                                      target_sum,
                                                      max_beta=1e6,
                                                      step_factor=0.6,
                                                      gamma=0.995)
        print(f"[{time.time():.2f}] Final Adam polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional final deterministic pass (if any time left)
    # --------------------------------------------------------------
    if time.time() + 1.0 < deadline:
        final_mt_deadline = deadline
        best_seq, best_score = _deterministic_mass_transfer_exact2(best_seq,
                                                                   final_mt_deadline,
                                                                   target_sum)
        print(f"[{time.time():.2f}] After final DMT best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return list
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence missing)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
