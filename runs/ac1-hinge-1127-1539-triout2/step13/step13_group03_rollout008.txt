"""
Hybrid optimizer for minimizing max convolution of a non‑negative step function.

The algorithm proceeds in several stages:
1. Generate a diverse pool of candidate sequences (power‑law, exponential,
   Hann window, sinusoidal, random) of modest size (≈ 8000) and run short
   annealed‑gradient + sub‑gradient refinements to obtain a good seed.
2. Upscale the best seed to a larger length (up to 40 000) using linear
   interpolation and project onto the ℓ¹‑simplex (∑a = √(2 n)).
3. Apply a few deterministic reorderings and an intensive permutation search
   (random shuffles + hill‑climbing with simulated‑annealing acceptance) to
   spread the large coefficients evenly.
4. Perform a long projected sub‑gradient descent on the large problem, followed
   by a final greedy mass‑transfer step.
5. Trim trailing zeros, store the result for possible reuse, and return it.

All operations respect the non‑negativity constraints and are timed so that
the function finishes before the allocated budget (default 1000 s).
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Return the maximum entry of the linear convolution a*a."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# ℓ¹‑simplex projection
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    """Hann window shape."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * x))  # Hann
    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine shape, zero at the ends, peak in the middle."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread the sorted values evenly using a step that is coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    # ensure step is coprime to n
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 9e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on a soft‑max approximation of the
    max‑convolution. The objective is approximated by
        f(a) = log Σ exp(β·(a∗a - max)).
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        # gradient of soft‑max term
        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AGD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         step_factor: float = 0.45,
                         gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with momentum on max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        # indices achieving the maximum (within tolerance)
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        # tiny jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from a pair (i, j) that contributes heavily to
    the current maximal convolution index toward a pair that contributes little.
    Accept moves according to a temperature schedule (simulated annealing).
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            # revert move
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        # occasional global jitter to explore new basins
        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

# Permutation search – random shuffles with optional hill‑climb
# ----------------------------------------------------------------------


def _permutation_search(base_seq: np.ndarray,
                        trials: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Try a number of random permutations and keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for t in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float,
                                 temperature_start: float = 0.3,
                                 temperature_end: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Pairwise swap hill‑climb with a simple annealing schedule.
    Swaps that improve the score are accepted; otherwise they are accepted
    with probability exp(-(Δ)/T).
    """
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        T = temperature_start * (temperature_end / temperature_start) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # try swap
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        delta = val - best_val
        if delta < 0:
            best_val = val
            # keep swap
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                best_val = val  # accept worsening move
            else:
                # revert
                a[i], a[j] = a[j], a[i]

    return a, best_val

    a *= target_sum / a.sum()
    return a


def _sin_half_sequence(n: int, target_sum: float) -> np.ndarray:
    """Half‑sine shape, zero at the ends, peak in the middle."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.sin(np.pi * x)
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_len``."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Spread sorted values evenly using a step coprime to n."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Annealed (soft‑max) gradient descent
# ----------------------------------------------------------------------


def _annealed_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 9e5,
    step_factor: float = 0.9,
    gamma: float = 0.98,
) -> tuple[np.ndarray, float]:
    """
    Momentum projected GD on a soft‑max approximation of max‑convolution.
    The soft‑max temperature β is increased exponentially up to ``max_beta``.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AGD it={it:5d} β={beta:7.1f} best={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    step_factor: float = 0.45,
    gamma: float = 0.995,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with momentum on max‑convolution."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        maxb = conv.max()
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = (target_sum * step_factor) / (norm + 1e-12)
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.0004, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] SubGrad it={it:5d} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy mass transfer with simulated‑annealing acceptance
# ----------------------------------------------------------------------


def _mass_transfer_simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp: float = 0.4,
    end_temp: float = 1e-5,
    low_frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Transfer a small chunk of mass from a heavy contributing pair
    (responsible for the current maximal convolution entry) to a lighter pair.
    Accept moves according to an annealing schedule.
    """
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation search – random shuffles + hill‑climb
# ----------------------------------------------------------------------


def _permutation_search(
    base_seq: np.ndarray,
    trials: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Try a number of random permutations; keep the best."""
    best_seq = base_seq.copy()
    best_val = _max_convolution_fft(best_seq)
    rng = np.random.default_rng()

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(base_seq)
        val = _max_convolution_fft(perm)
        if val < best_val:
            best_val = val
            best_seq = perm.copy()
    return best_seq, best_val


def _local_permutation_hillclimb(
    seq: np.ndarray,
    deadline: float,
    temperature_start: float = 0.35,
    temperature_end: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Pairwise swap hill‑climb with a simple annealing schedule.
    Improves the ordering of a fixed multiset of coefficients.
    """
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        T = temperature_start * (temperature_end / temperature_start) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]
        val = _max_convolution_fft(a)
        delta = val - best_val
        if delta < 0:
            best_val = val
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if rng.random() >= prob:
                a[i], a[j] = a[j], a[i]

    return a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a small max‑convolution.
    The function respects the given time budget and returns a list of floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    rng = np.random.default_rng()

    # --------------------------------------------------------------
    # Phase 0 – build a diverse pool of modest‑size seeds
    # --------------------------------------------------------------
    n_expl = 8000 if budget_s >= 900 else (6000 if budget_s >= 700 else 5000)
    target_expl = math.sqrt(2.0 * n_expl)

    init_pool: list[np.ndarray] = []

    # power‑law family
    for p in np.linspace(0.1, 2.2, 14):
        init_pool.append(_power_law_sequence(n_expl, float(p), target_expl))
    # exponential family
    for lam in np.linspace(0.0003, 0.04, 12):
        init_pool.append(_exponential_sequence(n_expl, float(lam), target_expl))
    # windowed families
    for _ in range(10):
        init_pool.append(_cosine_sequence(n_expl, target_expl))
    for _ in range(10):
        init_pool.append(_sin_half_sequence(n_expl, target_expl))
    # random simplex
    for _ in range(15):
        init_pool.append(_random_simplex(n_expl, target_expl, rng))

    # reuse previous best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        init_pool.append(_resize_sequence(prev, n_expl))

    # jitter each seed to increase diversity
    for seq in list(init_pool):
        jitter = rng.normal(scale=target_expl * 0.025, size=n_expl)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_expl)
        init_pool.append(seq_j)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Phase 1 – quick annealed‑gradient + sub‑gradient refinements
    # --------------------------------------------------------------
    while time.time() < deadline - 40.0:
        if best_seq is not None and rng.random() < 0.55:
            seed = best_seq.copy()
            jitter = rng.normal(scale=target_expl * 0.006, size=n_expl)
            seed = np.maximum(seed + jitter, 0.0)
            seed = _project_to_simplex(seed, target_expl)
        else:
            seed = rng.choice(init_pool)

        run_deadline = min(time.time() + 14.0, deadline - 6.0)
        if run_deadline <= time.time():
            break

        # a) annealed gradient descent
        a1, v1 = _annealed_gradient_descent(
            seed, run_deadline, target_expl, max_beta=9e5, step_factor=0.9, gamma=0.98
        )

        # b) short sub‑gradient refinement
        sub_deadline = min(time.time() + 6.0, deadline - 2.0)
        a2, v2 = _subgradient_descent(
            a1, sub_deadline, target_expl, step_factor=0.45, gamma=0.995
        )

        if v2 < v1:
            val_opt, seq_opt = v2, a2
        else:
            val_opt, seq_opt = v1, a1

        if val_opt < best_score:
            best_score = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] Phase 1 new best={best_score:.6f}")

    # Fallback if nothing succeeded
    if best_seq is None:
        best_seq = _power_law_sequence(n_expl, 0.7, target_expl)
        best_score = _max_convolution_fft(best_seq)

    # --------------------------------------------------------------
    # Phase 2 – mass‑transfer refinement on modest size
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        mt_deadline = time.time() + (deadline - time.time()) * 0.12
        a_mt, v_mt = _mass_transfer_simulated_annealing(
            best_seq,
            mt_deadline,
            target_expl,
            start_temp=0.4,
            end_temp=1e-5,
            low_frac=0.07,
        )
        if v_mt < best_score:
            best_score = v_mt
            best_seq = a_mt.copy()
            print(f"[{time.time():.2f}] MT refinement score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – upscale to a larger problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        n_final = 40000
    elif remaining > 80.0:
        n_final = 30000
    elif remaining > 50.0:
        n_final = 25000
    else:
        n_final = n_expl

    target_final = math.sqrt(2.0 * n_final)

    seq_big = _resize_sequence(best_seq, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # deterministic reorderings, keep the most promising
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _even_spread_order(seq_big),
    ]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(
        f"[{time.time():.2f}] Upscaled to {n_final} elems, start max={_max_convolution_fft(seq_big):.6f}"
    )

    # --------------------------------------------------------------
    # Phase 4 – permutation search (random shuffles + hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline - 12.0:
        perm_deadline = time.time() + min(10.0, deadline - time.time() - 5.0)
        seq_big, val_perm = _permutation_search(seq_big, trials=120, deadline=perm_deadline)
        print(f"[{time.time():.2f}] After random perms max={val_perm:.6f}")

    if time.time() < deadline - 14.0:
        hill_deadline = time.time() + min(8.0, deadline - time.time() - 4.0)
        seq_big, val_hill = _local_permutation_hillclimb(
            seq_big,
            hill_deadline,
            temperature_start=0.35,
            temperature_end=1e-5,
        )
        print(f"[{time.time():.2f}] After hill‑climb max={val_hill:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – large‑scale sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_deadline = deadline - 5.0
        a_sub, v_sub = _subgradient_descent(
            seq_big,
            sub_deadline,
            target_final,
            step_factor=0.44,
            gamma=0.996,
        )
        if v_sub < best_score:
            best_score = v_sub
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad polished best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – final greedy mass‑transfer
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        mt_deadline = time.time() + min(6.0, deadline - time.time() - 2.0)
        a_mt, v_mt = _mass_transfer_simulated_annealing(
            seq_big,
            mt_deadline,
            target_final,
            start_temp=0.25,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if v_mt < best_score:
            best_score = v_mt
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # --------------------------------------------------------------
    # Trim negligible tail, store for potential reuse and return
    # --------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    globals()["height_sequence_1"] = final_seq
    return final_seq
