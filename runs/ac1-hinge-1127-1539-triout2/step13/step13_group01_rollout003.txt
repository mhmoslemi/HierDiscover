def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Linear cross-correlation of w (len L) with a (len n), returning first n entries for lags 0..n-1."""
    L = w.shape[0]
    n = a.shape[0]
    size = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr_full = np.fft.irfft(W * np.conj(A), n=size)
    start = n - 1
    # shape may be larger than start + n; ensure we have enough
    return corr_full[start:start + n]

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    start_time = time.time()
    deadline = start_time + budget_s - 0.2
    # select n
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 600:
        n = 15000
    else:
        n = 10000
    target_sum = np.sqrt(2.0 * n)
    # build init candidates
    candidates = []
    # warm start
    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"], dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        candidates.append(warm)
    # power law
    pl = np.arange(1, n+1, dtype=np.float64) ** (-0.55)
    pl = pl * (target_sum / pl.sum())
    candidates.append(pl)
    # random
    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        candidates.append(rnd)
    best_seq = None
    best_score = np.inf
    # for each candidate
    for idx, init in enumerate(candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] Starting candidate {idx+1}/{len(candidates)}")
        # allocate portion of remaining time to smooth descent
        remaining = deadline - time.time()
        # sub-allocate 70% for smooth optimization, rest for anneal
        opt_deadline = time.time() + 0.7 * remaining
        a_opt, score_opt = _smooth_optimize(init, target_sum, opt_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] New global best after smooth opt: {best_score:.6f}")
        # remaining time for anneal
        if time.time() < deadline:
            a_ann, score_ann = anneal_mass_transfer(a_opt, target_sum, deadline)
            if score_ann < best_score:
                best_score = score_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")
    if best_seq is None:
        # fallback: simple decreasing power law
        best_seq = (np.arange(1, n+1, dtype=np.float64) ** (-0.6))
        best_seq = best_seq * (target_sum / best_seq.sum())
    # Trim trailing near-zero entries
    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1]+1]
    final_seq = arr.tolist()
    # store warm start
    globals()["height_sequence_1"] = final_seq
    # final evaluation
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as e:
        print(f"[{time.time():.2f}] Evaluation error: {e}")

def _smooth_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # compute initial score via max conv
    b = _conv_fft(a)
    max_b = b.max()
    best_score = 2.0 * n * max_b / (target_sum ** 2)
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_counter = 0
    start = time.time()
    # schedule parameters
    eta_start = 0.5
    eta_end = 40.0
    while time.time() < deadline:
        # compute FFT of a
        size = 1 << ((2 * n - 1).bit_length())
        # Use size that covers convolution + correlation; we might need size >= 3n approx
        conv_size = 1 << ((2 * n - 1).bit_length())
        # For correlation we need size >= (2n-1) + n - 1 = 3n - 2.
        size_corr = 1 << ((3 * n - 2).bit_length())
        size_fft = max(conv_size, size_corr)
        A_fft = np.fft.rfft(a, n=size_fft)
        # convolution b
        B_fft = A_fft * A_fft
        b_full = np.fft.irfft(B_fft, n=size_fft)[: 2 * n - 1]
        # smoothing parameter eta according to elapsed time
        elapsed = time.time() - start
        total = deadline - start
        eta = eta_start + (eta_end - eta_start) * (elapsed / total)
        # compute softmax weights for b
        b_max = b_full.max()
        exp_vals = np.exp(eta * (b_full - b_max))
        w = exp_vals / exp_vals.sum()
        # correlation gradient
        W_fft = np.fft.rfft(w, n=size_fft)
        corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=size_fft)
        start_idx = n - 1
        grad = 2.0 * corr_full[start_idx:start_idx + n]
        # Adam update
        step_counter += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step_counter)
        v_hat = v / (1 - beta2 ** step_counter)
        # step size lr (maybe decreasing)
        lr = 0.5  # base learning rate
        a = a - lr * (m_hat / (np.sqrt(v_hat) + eps))
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # evaluate score
        cur_score = 2.0 * n * b_max / (target_sum ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth opt new best {best_score:.6f}")
    return best_a, best_score

"""
Smooth‑max gradient descent + annealing for the self‑convolution step‑function
problem.

The objective (after fixing Σa = √(2·n)) is to minimise the peak of the
linear self‑convolution a*a.  Directly minimising max(b) is hard because the
max is non‑smooth.  Instead we minimise a smoothed approximation

    F(a) = (1/η)·log Σ_k exp(η·b_k)  with b = a*a

which converges to max(b) as η → ∞.  The gradient of F is a weighted sum of
the individual convolution gradients; the weights are the soft‑max of the
entries of b.  The gradient can be evaluated in O(n log n) using FFTs:
* b = a*a   via FFT,
* w = softmax(η·b),
* ∇F = 2·(w ⋆ a)   (linear cross‑correlation).

We use an Adam‐style update, project the iterate onto the simplex
{a ≥ 0, Σa = √(2·n)} after each step and linearly increase η during the run.
After the smooth optimisation a short aggressive simulated‑annealing (the
mass‑transfer code from the previous state‑of‑the‑art) is applied for fine
tuning.

The routine respects the supplied time budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution / correlation utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) for non‑negative lags.
    Returns the first n entries: (w⋆a)[i] = Σ_j w[i+j]·a[j], i = 0…n‑1.
    """
    n = a.size
    L = w.size            # = 2·n‑1
    size = _next_pow2(L + n - 1)   # ≥ 3·n‑2
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr_full = np.fft.irfft(W * np.conj(A), n=size)
    start = n - 1
    return corr_full[start : start + n]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimisation
# ----------------------------------------------------------------------


def _smooth_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best vector encountered and its corresponding score
    (the true objective 2·n·max(b)/target_sum²).
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑compute FFT size large enough for both convolution and correlation
    size_fft = _next_pow2(3 * n - 2)   # ≥ 2·n‑1 + n‑1

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    # score of the current a (computed inside the loop on first iteration)
    best_score = np.inf

    start = time.time()
    it = 0
    while time.time() < deadline:
        # ----- evaluate current a -----
        A_fft = np.fft.rfft(a, n=size_fft)
        B_fft = A_fft * A_fft
        b = np.fft.irfft(B_fft, n=size_fft)[: 2 * n - 1]

        cur_score = 2.0 * n * b.max() / (target_sum ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Smooth opt it={it:6d} new best {best_score:.6f}")

        # ----- smoothing parameter η (linear schedule) -----
        elapsed = time.time() - start
        total = deadline - start
        eta = 0.5 + (40.0 - 0.5) * (elapsed / total)   # 0.5 → 40

        # ----- soft‑max weights -----
        b_max = b.max()
        exp_vals = np.exp(eta * (b - b_max))   # shift for numerical stability
        w = exp_vals / exp_vals.sum()

        # ----- gradient = 2·(w ⋆ a) -----
        W_fft = np.fft.rfft(w, n=size_fft)
        corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=size_fft)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # first n lags

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.5                     # base learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                 # descend
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape shallow plateaus
        if it % 4000 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=target_sum * 1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing mass‑transfer (same as previous code)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Small‑mass transfers guided by the current peak of the convolution.
    Keeps the best solution seen.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)
    cur_val = 2.0 * n * b.max() / (target_sum ** 2)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    T0 = cur_val * 0.9          # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak and its contributing indices
        k_max = int(b.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # favour small heights as receivers
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta * (temperature / T0 if T0 > 0 else 0.0)
        if delta <= 0.0:
            continue

        # exact O(n) update of the full convolution
        b_new = b.copy()
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max() / (target_sum ** 2)

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal it={it:6d} improved best {best_val:.6f}")

        # periodic recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = 2.0 * n * b.max() / (target_sum ** 2)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the

    """
    start = time.time()
    deadline = start + budget_s - 0.3   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n = finer granularity, but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 500:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)   # normalisation Σa = √(2·n)

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + diverse seeds)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(
            np.array(globals()["height_sequence_1"], dtype=np.float64), n
        )
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    # deterministic power‑law start (often a good baseline)
    power = np.arange(1, n + 1, dtype=np.float64) ** (-0.55)
    power = power * (target_sum / power.sum())
    init_candidates.append(power)

    # a few random Dirichlet‑like starts
    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Optimise each start, respecting the global deadline
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:

        print(f"[{time.time():.2f}] ----- start #{idx + 1} of {len(init_candidates)} -----")

        # allocate ~70 % of the remaining budget to the smooth optimiser
        remain = deadline - time.time()
        sub_deadline = time.time() + 0.70 * remain

        if score_opt < best_score:
            best_score = score_opt

            if score_ann < best_score:
                best_score = score_ann

                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # Fallback in the unlikely event that nothing was produced
