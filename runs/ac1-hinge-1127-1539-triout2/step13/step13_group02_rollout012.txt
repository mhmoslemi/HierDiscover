def _solve_lp_max_sum(a, M):
    n = a.shape[0]
    m = 2*n - 1
    # Build A matrix
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    # Setup linprog
    c = -np.ones(n)  # minimize -sum => maximize sum
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    # using hi
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    g0 = res.x
    sum_g0 = g0.sum()
    return g0, sum_g0

def _solve_lp_max_sum(a: np.ndarray, M: float):
    n = a.shape[0]
    m = 2 * n - 1
    # Build data arrays
    # rows: i + j for i in 0..n-1, j in 0..n-1
    i = np.arange(n).reshape(-1, 1)  # column vector

    if not res.success:

    g0 = res.x

"""
Hybrid optimiser for the step‑function challenge.

Main ideas
----------

1. **Linear‑programming refinement** –  starting from a modest
   length (≈1500) we repeatedly solve a LP that maximises the ℓ¹‑norm of a
   partner sequence `g` under the constraint ‖f∗g‖∞ ≤ M, where
   `M = max (f∗f)`.  After scaling `g` back to the required ℓ¹‑norm we
   convex‑combine it with `f`.  This is the “LP‑step” described in the
   paper and is very effective at reducing the peak of the autocorrelation.

2. **Upscaling** –  the refined short sequence is linearly interpolated to
   a larger length (≈10 k – 12 k) and projected onto the ℓ¹‑simplex.

3. **Fine polishing** –  a high‑β soft‑max gradient descent,
   a targeted permutation annealing and a short sub‑gradient pass are
   applied to the large sequence.  These phases are essentially the same
   as in the previous state‑of‑the‑art code but run on a much better
   starting point.

The whole routine respects the 1000 s time budget, stores the best
sequence in the global ``height_sequence_1`` and returns it as a plain
list of floats.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# FFT utilities (max‑convolution and full convolution)
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2·n) this equals the peak of the convolution.
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple families of sequences (used for seeding)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, then projected onto the simplex."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), then projected onto the simplex."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation resize (keeps total mass)."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Place the largest value left‑most, second largest right‑most, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _reversed_alternating(seq: np.ndarray) -> np.ndarray:
    rev = _alternating_order(seq[::-1])
    return rev[::-1]


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max(convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Momentum‑annealed gradient descent on the soft‑max objective
# ----------------------------------------------------------------------


def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 5e5,
                               step_factor: float = 0.5,
                               gamma: float = 0.99) -> tuple[np.ndarray, float]:
    """Momentum‑based GD on soft‑max approximation of the peak."""
    a = _project_to_simplex(a0.copy(), target_sum)

    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        grad, _ = _gradient_softmax(a, beta)
        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum
        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true peak (cheap)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         target_sum: float,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(convolution)."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    base_step = target_sum * 0.5

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        grad = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = kmax - idx
        grad[idx] = a[partner]
        if (kmax % 2 == 0):
            mid = kmax // 2
            if lo <= mid <= hi:
                grad[mid] += a[mid]

        step = base_step / np.sqrt(it + 1.0)
        a -= step * grad
        a = _project_to_simplex(a, target_sum)

        if it % 4000 == 0:
            jitter = rng.normal(scale=target_sum * 0.0005, size=n)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Sub‑grad it={it:5d} score={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (small moves)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-4,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    conv = _conv_fft(a)
    n = a.shape[0]

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        w = a[idx] + a[kmax - idx]

        donor = idx[np.argmax(w)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_idxs = idx[np.argsort(w)[:low_cnt]]
        receiver = rng.choice(low_idxs)

        if donor == receiver:
            it += 1
            continue

        max_delta = a[donor] * 0.5
        weight_diff = w[np.argmax(w)] - w[np.argmin(w)]
        if weight_diff > 0:
            max_delta = min(max_delta, weight_diff * 0.25)

        delta = rng.random() * max_delta
        if delta <= 0:
            it += 1
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT it={it:5d} new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if it % 2000 == 0 and it > 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _score(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted permutation simulated annealing (swap high/low contributors)
# ----------------------------------------------------------------------


def _permutation_targeted_search(a0: np.ndarray,
                                 deadline: float,
                                 start_temp: float = 0.5,
                                 end_temp: float = 1e-5,
                                 top_frac: float = 0.05,
                                 low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing that swaps an index with a high contribution to the
    current peak with an index with a low contribution.
    """
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    n = a.shape[0]
    start_time = time.time()

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        kmax_candidates = np.flatnonzero(conv == max_val)
        kmax = rng.choice(kmax_candidates)

        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        top_k = max(2, int(top_frac * n))
        low_k = max(2, int(low_frac * n))

        donor_candidates = np.argpartition(-contrib, top_k - 1)[:top_k]
        receiver_candidates = np.argpartition(contrib, low_k - 1)[:low_k]

        donor = rng.choice(donor_candidates)
        receiver = rng.choice(receiver_candidates)

        if donor == receiver:
            continue

        a[donor], a[receiver] = a[receiver], a[donor]

        new_score = _score(a)

        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] PermSearch new best={best_score:.6f}")
        else:
            a[donor], a[receiver] = a[receiver], a[donor]

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximising ℓ¹‑norm under a peak bound)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(a: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_i   subject to   a * g ≤ M  (entrywise) , g ≥ 0 .
    Returns the optimal g and its ℓ¹‑norm.
    """
    n = a.shape[0]
    m = 2 * n - 1

    # Build the Toeplitz‑like constraint matrix:
    # A_{k,j} = a_{k-j}  (with out‑of‑range indices interpreted as 0)
    i = np.arange(n).reshape(-1, 1)          # shape (n,1)
    j = np.arange(n).reshape(1, -1)          # shape (1,n)
    rows = (i + j).ravel()                  # size n*n, values 0 … 2n‑2
    cols = np.tile(np.arange(n), n)         # repeat 0..n‑1 for each i
    data = np.repeat(a, n)                  # a_i repeats for each column j
    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))

    c = -np.ones(n)                          # minimise –Σg  ⇒ maximise Σg
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g0 = res.x
    sum_g0 = g0.sum()
    return g0, sum_g0


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser:
      1. LP‑based refinement on a short sequence (≈1500).
      2. Upscale to a longer sequence (≈10‑12k).
      3. High‑β gradient descent, permutation annealing and sub‑gradient polishing.
      4. Trim trailing zeros and return the best found sequence.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Small‑scale LP refinement
    # ------------------------------------------------------------------
    n_small = 1500
    target_small = np.sqrt(2.0 * n_small)

    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        f = _resize_sequence(prev, n_small)
        f = _project_to_simplex(f, target_small)
    else:
        rng = np.random.default_rng()
        f = rng.random(n_small)
        f = f * target_small / f.sum()

    best_f = f.copy()
    best_score = _score(best_f)
    print(f"[{time.time():.2f}] Initial small‑scale score={best_score:.6f}")

    # allocate time for LP refinement, keep ~40 s for later phases
    lp_deadline = deadline - 40.0
    t_mix = 0.2                    # convex‑mixing weight
    lp_iter = 0
    while time.time() < lp_deadline:
        # current peak
        M = _max_convolution_fft(best_f)

        # LP step
        try:
            g0, sum_g0 = _solve_lp_max_sum(best_f, M)
        except Exception as e:
            print(f"[{time.time():.2f}] LP failed ({e}); breaking.")
            break

        # scale to the required ℓ¹‑norm and mix
        scale = target_small / sum_g0 if sum_g0 > 0 else 0.0
        g = g0 * scale
        best_f = (1.0 - t_mix) * best_f + t_mix * g
        best_f = _project_to_simplex(best_f, target_small)

        new_score = _score(best_f)
        if new_score < best_score:
            best_score = new_score
            print(f"[{time.time():.2f}] LP iter {lp_iter} new best small score={best_score:.6f}")

        lp_iter += 1
        if lp_iter % 5 == 0:
            t_mix = max(0.02, t_mix * 0.9)      # slowly reduce mixing weight

    # ------------------------------------------------------------------
    # 2. Upscale to a larger problem size
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 80.0:
        n_final = 12000
    elif remaining > 50.0:
        n_final = 10000
    elif remaining > 30.0:
        n_final = 8000
    else:
        n_final = n_small

    target_final = np.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_f, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a couple of systematic reorderings
    candidates = [
        seq_big,
        _alternating_order(seq_big),
        _reversed_alternating(seq_big),
    ]

    best_seq = None
    best_score = np.inf
    for cand in candidates:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time():.2f}] Upscaled length={n_final} start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine polishing (high‑β GD → permutation annealing → sub‑grad → MT)
    # ------------------------------------------------------------------
    # 3a. High‑β gradient descent
    if time.time() < deadline - 12.0:
        gd_dead = deadline - 6.0
        seq_gd, sc_gd = _annealed_gradient_descent(
            best_seq,
            gd_dead,
            target_final,
            max_beta=5e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if sc_gd < best_score:
            best_score = sc_gd
            best_seq = seq_gd.copy()
            print(f"[{time.time():.2f}] GD polish best={best_score:.6f}")

    # 3b. Targeted permutation simulated annealing
    if time.time() < deadline - 5.0:
        perm_dead = deadline - 0.5
        seq_perm, sc_perm = _permutation_targeted_search(
            best_seq,
            perm_dead,
            start_temp=0.5,
            end_temp=1e-5,
            top_frac=0.07,
            low_frac=0.07,
        )
        if sc_perm < best_score:
            best_score = sc_perm
            best_seq = seq_perm.copy()
            print(f"[{time.time():.2f}] After permutation search best={best_score:.6f}")

    # 3c. Very short sub‑gradient pass
    if time.time() < deadline - 2.0:
        sub_dead = time.time() + min(3.0, deadline - time.time() - 1.0)
        seq_sub, sc_sub = _subgradient_descent(
            best_seq,
            sub_dead,
            target_final,
            max_iter=100_000,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] Sub‑grad final best={best_score:.6f}")

    # 3d. Final deterministic mass‑transfer polishing
    if time.time() < deadline - 1.0:
        mt_dead = time.time() + min(4.0, deadline - time.time() - 0.5)
        seq_mt, sc_mt = _deterministic_mass_transfer(
            best_seq,
            mt_dead,
            target_final,
            start_temp=0.3,
            end_temp=1e-5,
            low_frac=0.05,
        )
        if sc_mt < best_score:
            best_score = sc_mt
            best_seq = seq_mt.copy()
            print(f"[{time.time():.2f}] Final MT best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Trim trailing zeros and finish
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.flatnonzero(arr > 1e-12)
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    # optional final evaluation log
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
