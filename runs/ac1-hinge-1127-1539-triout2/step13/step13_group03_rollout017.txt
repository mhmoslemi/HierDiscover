def _adam_subgradient_descent(a0, deadline, target_sum):
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a)  # length 2n-1
        maxb = conv.max()
        # indicator of near-max entries (within tolerance)
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        # compute subgradient: 2 * convolution of I and reversed a
        a_rev = a[::-1]
        size = 1 << ((len(I) + len(a_rev) - 1).bit_length())
        I_fft = np.fft.rfft(I, n=size)
        a_rev_fft = np.fft.rfft(a_rev, n=size)
        grad_full = np.fft.irfft(I_fft * a_rev_fft, n=size)[: len(I) + len(a_rev) - 1]
        grad = 2.0 * grad_full[len(a_rev) - 1 : len(a_rev) - 1 + len(a)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = (target_sum * 0.4) / (np.sqrt(v_hat) + eps) * m_hat
        a -= step
        a = _project_to_simplex(a, target_sum)

        # optional jitter to escape local minima
        if t % 2000 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)
        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD iter={t} val={best_val:.6f}")
    return best_a, best_val

def _lp_iterative_improvement(f0, target_sum, deadline, max_iters=6, t=0.2):
    f = _project_to_simplex(f0.copy(), target_sum)
    best_f = f.copy()
    best_val = _max_convolution_fft(f)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        n = f.shape[0]
        A_ub = _build_conv_constraint_matrix(f)
        b_ub = np.zeros(2*n-1, dtype=np.float64)
        A_eq = np.zeros((1, n+1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum], dtype=np.float64)
        c = np.zeros(n+1, dtype=np.float64)
        c[-1] = 1.0  # minimize M
        bounds = [(0, None)] * n + [(0, None)]
        # Solve
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed: {res.message}")
            break
        g = res.x[:n]  # shape (n,)
        M_opt = res.x[-1]
        # Combine
        f_new = (1 - t) * f + t * g
        # Project onto simplex to counteract numeric drift
        f_new = _project_to_simplex(f_new, target_sum)
        val = _max_convolution_fft(f_new)
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            print(f"[{time.time():.2f}] LP iter={it} t={t:.2f} max={val:.6f}")
        f = f_new
        # optionally adapt t: maybe decrease.
        t = max(t * 0.9, 0.05)
    return best_f, best_val

"""
Hybrid optimizer for the step‑function benchmark.

Phase 0 – small‑scale exploration (n≈300) with random families and a
few seconds of annealed‑gradient descent.
Phase 1 – linear‑programming refinement that directly minimises the
maximum of the convolution f * g under the ℓ¹‑simplex constraint.
Phase 2 – upscale to a large size (8k‑30k) by linear interpolation,
apply deterministic re‑orderings and keep the best layout.
Phase 3 – long‑run Adam‑style sub‑gradient descent on the exact max‑
convolution (‖f * f‖∞) until the time budget expires.
Phase 4 – very short mass‑transfer / hill‑climb passes and final trim.

All iterates are projected onto the simplex {x≥0, Σx = √(2 n)} so the
objective reduces to the maximal entry of the linear convolution.
The LP formulation introduces an auxiliary variable M and solves

    minimise   M
    s.t.       (f * g)[k] − M ≤ 0   (∀k)
               Σ g_i = √(2 n),  g_i ≥ 0 .

Iterating this LP with a convex combination of the previous iterate
produces a monotone reduction of the bound.
"""

def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection onto {x | x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …"""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """Place values on a co‑prime step to spread them evenly."""
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]
    step = max(1, int(round(math.sqrt(n))))
    while math.gcd(step, n) != 1:
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out

def _annealed_gradient_descent(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               max_beta: float = 8e5,
                               step_factor: float = 0.9,
                               gamma: float = 0.98) -> tuple[np.ndarray, float]:
    """Momentum projected GD on the soft‑max approximation of max‑conv."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    beta = 0.5
    step_base = target_sum * step_factor
    rng = np.random.default_rng()
    momentum = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        n = a.shape[0]
        size = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=size)
        b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

        max_b = np.max(b)
        exps = np.exp(beta * (b - max_b))
        p = exps / np.sum(exps)

        a_rev = a[::-1]
        Ar_fft = np.fft.rfft(a_rev, n=size)
        Pr_fft = np.fft.rfft(p, n=size)
        conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]

        momentum = gamma * momentum + grad
        norm = np.linalg.norm(momentum)
        if norm > 0:
            lr = step_base / (norm * (beta + 1.0))
            a -= lr * momentum

        a = _project_to_simplex(a, target_sum)

        if it % 250 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD it={it:5d} β={beta:7.1f} score={best_val:.6f}")

        if it % 500 == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val

def _adam_subgradient_descent(a0: np.ndarray,
                              deadline: float,
                              target_sum: float,
                              step_factor: float = 0.4,
                              beta1: float = 0.9,
                              beta2: float = 0.999,
                              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Exact max‑convolution sub‑gradient descent with Adam momentum."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _max_convolution_fft(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a)
        maxb = conv.max()
        # indicator of entries attaining the max (within a tiny tolerance)
        I = (conv >= maxb * (1 - 1e-12)).astype(np.float64)

        a_rev = a[::-1]

        v = beta2 * v + (1 - beta2) * (grad * grad)

        v_hat = v / (1 - beta2 ** t)
        step = (target_sum * step_factor) * (m_hat / (np.sqrt(v_hat) + eps))
        a -= step
        a = _project_to_simplex(a, target_sum)

        if t % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        cur_val = _max_convolution_fft(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={t:6d} score={best_val:.6f}")

def _mass_transfer_simulated_annealing(a0: np.ndarray,
                                      deadline: float,
                                      target_sum: float,
                                      start_temp: float = 0.4,
                                      end_temp: float = 1e-5,
                                      low_frac: float = 0.07) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_fft(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    n = a.shape[0]
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        tau = elapsed / (deadline - start_time)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]
        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _max_convolution_fft(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] MT new best={best_score:.6f}")
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

        if rng.random() < 0.001:
            jitter = rng.normal(scale=target_sum * 0.001, size=a.shape)
            a += jitter
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)
            conv = _conv_fft(a)
            cur_score = _max_convolution_fft(a)

    return best_a, best_score

def _permutation_search(base_seq: np.ndarray,
                        trials: int,

def _local_permutation_hillclimb(seq: np.ndarray,
                                 deadline: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    best_val = _max_convolution_fft(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    attempts = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:

        a[i], a[j] = a[j], a[i]

        else:

        attempts += 1
        if attempts % 20000 == 0:
            print(f"[{time.time():.2f}] Hill‑climb attempts={attempts} best={best_val:.6f}")
    return a, best_val

def _build_conv_constraint_matrix(f: np.ndarray) -> "lil_matrix":
    """
    Return a sparse matrix A (shape (2n‑1, n+1)) representing the constraints
        (f * g)[k] - M ≤ 0
    where the last column corresponds to the auxiliary variable M.
    """
    n = f.shape[0]
    A = lil_matrix((2 * n - 1, n + 1), dtype=np.float64)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A[k, i] = f[k - i]
        A[k, n] = -1.0  # coefficient for M
    return A.tocsr()


def _lp_iterative_improvement(f0: np.ndarray,
                              target_sum: float,
                              deadline: float,
                              max_iters: int = 6,
                              t: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve
        minimise   M
        s.t.       (f * g)[k] - M ≤ 0
                   Σ g_i = target_sum , g_i ≥ 0
    for the current f, then replace f ← (1‑t)·f + t·g.
    Returns the best iterate encountered.
    """
    f = _project_to_simplex(f0.copy(), target_sum)
    best_f = f.copy()
    best_val = _max_convolution_fft(f)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        n = f.shape[0]
        A_ub = _build_conv_constraint_matrix(f)
        b_ub = np.zeros(2 * n - 1, dtype=np.float64)

        # Equality Σ g_i = target_sum
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum], dtype=np.float64)

        # Objective: minimise M
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.2f}] LP iteration {it} failed: {res.message}")
            break

        g = res.x[:n]
        M_opt = res.x[-1]

        # convex combination
        f_new = (1 - t) * f + t * g
        f_new = _project_to_simplex(f_new, target_sum)

        val = _max_convolution_fft(f_new)
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            print(f"[{time.time():.2f}] LP iter={it:2d} t={t:.2f} max={val:.6f}")

        f = f_new
        t = max(t * 0.9, 0.05)          # slowly reduce the step size

    return best_f, best_val

def propose_candidate(budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the benchmark score.
    The routine respects the 1000 s budget and stores the best sequence in
    the global variable ``height_sequence_1`` for warm‑starts in subsequent runs.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin
    rng = np.random.default_rng()

    # ------------------------------------------------------------
    # Phase 0 – tiny exploration (n≈300)
    # ------------------------------------------------------------
    n_small = 300
    target_small = math.sqrt(2.0 * n_small)

    pool_small = []

    # diverse parametric families
    for p in np.linspace(0.2, 2.0, 6):
        pool_small.append(_power_law_sequence(n_small, float(p), target_small))
    for lam in np.linspace(0.004, 0.02, 6):
        pool_small.append(_exponential_sequence(n_small, float(lam), target_small))
    for sigma in np.linspace(5.0, 30.0, 5):
        pool_small.append(_gaussian_sequence(n_small, float(sigma), target_small))
    pool_small.append(_cosine_sequence(n_small, target_small))
    pool_small.append(_sin_half_sequence(n_small, target_small))
    for _ in range(8):
        pool_small.append(_random_simplex(n_small, target_small, rng))

    # reuse previous solution if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        pool_small.append(_project_to_simplex(_resize_sequence(prev, n_small), target_small))

    # quick annealed‑gradient runs on a few random seeds
    best_small = None
    best_val_small = np.inf
    phase0_deadline = min(deadline - 30, time.time() + 12.0)

    while time.time() < phase0_deadline:
        seed = rng.choice(pool_small)
        # small jitter before optimisation
        seed = seed + rng.normal(scale=target_small * 0.02, size=n_small)
        seed = np.maximum(seed, 0.0)
        seed = _project_to_simplex(seed, target_small)

        gd_deadline = min(time.time() + 5.0, deadline - 8.0)
        if gd_deadline <= time.time():
            break
        a_opt, val_opt = _annealed_gradient_descent(seed, gd_deadline,
                                                    target_small,
                                                    max_beta=5e5,
                                                    step_factor=0.9,
                                                    gamma=0.98)
        if val_opt < best_val_small:
            best_val_small = val_opt
            best_small = a_opt.copy()
            print(f"[{time.time():.2f}] Phase0 new best={best_val_small:.6f}")

    if best_small is None:
        best_small = _random_simplex(n_small, target_small, rng)

    # ------------------------------------------------------------
    # Phase 1 – LP refinement on the small problem
    # ------------------------------------------------------------
    if time.time() < deadline - 50:                     # keep enough time for large scale work
        lp_deadline = deadline - 35
        best_small, best_val_small = _lp_iterative_improvement(
            best_small, target_small, lp_deadline,
            max_iters=6, t=0.3)
        print(f"[{time.time():.2f}] After LP small max={best_val_small:.6f}")

    # ------------------------------------------------------------
    # Phase 2 – upscale to a larger size
    # ------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n_final = 30000
    elif remaining > 400:
        n_final = 25000
    elif remaining > 250:
        n_final = 20000
    elif remaining > 150:
        n_final = 15000
    elif remaining > 80:
        n_final = 12000
    else:
        n_final = 8000

    target_final = math.sqrt(2.0 * n_final)
    seq_big = _resize_sequence(best_small, n_final)
    seq_big = _project_to_simplex(seq_big, target_final)

    # try a few deterministic reorderings and keep the best
    candidates = [seq_big,
                  _alternating_order(seq_big),
                  _even_spread_order(seq_big)]
    seq_big = min(candidates, key=_max_convolution_fft)
    print(f"[{time.time():.2f}] Upscaled to {n_final} elements, start max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------
    # Phase 3 – long‑run Adam sub‑gradient descent (uses the whole rest)
    # ------------------------------------------------------------
    if time.time() < deadline - 10:
        sub_deadline = deadline - 5.0
        a_sub, val_sub = _adam_subgradient_descent(seq_big, sub_deadline, target_final,
                                                   step_factor=0.45)
        if val_sub < _max_convolution_fft(seq_big):
            seq_big = a_sub.copy()
            print(f"[{time.time():.2f}] Adam‑GD final max={val_sub:.6f}")

    # ------------------------------------------------------------
    # Phase 4 – final short mass‑transfer and hill‑climb
    # ------------------------------------------------------------
    if time.time() < deadline - 4:
        mt_deadline = time.time() + min(4.0, deadline - time.time() - 2.0)
        a_mt, val_mt = _mass_transfer_simulated_annealing(seq_big, mt_deadline,
                                                          target_final,
                                                          start_temp=0.3,
                                                          end_temp=1e-5,
                                                          low_frac=0.07)
        if val_mt < _max_convolution_fft(seq_big):
            seq_big = a_mt.copy()
            print(f"[{time.time():.2f}] Final MT max={val_mt:.6f}")

    if time.time() < deadline - 2:
        hill_deadline = time.time() + min(2.0, deadline - time.time())
        seq_big, _ = _local_permutation_hillclimb(seq_big, hill_deadline)
        print(f"[{time.time():.2f}] After hill‑climb max={_max_convolution_fft(seq_big):.6f}")

    # ------------------------------------------------------------
    # Trim trailing near‑zero entries and return
    # ------------------------------------------------------------
    arr = np.array(seq_big, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f} (len={len(final_seq)})")
    except Exception as e:
        print("Evaluation error:", e)

    globals()["height_sequence_1"] = final_seq
    return final_seq
```

This implementation adds an explicit linear‑programming refinement (Phase 1) that directly minimises the convolution bound, uses an Adam‑style exact sub‑gradient method for the remaining time, and scales the solution to a much larger dimension than the original 5 000‑element run. All iterates are projected onto the ℓ¹‑simplex, guaranteeing non‑negative heights, and the routine respects the 1000 s time budget.
