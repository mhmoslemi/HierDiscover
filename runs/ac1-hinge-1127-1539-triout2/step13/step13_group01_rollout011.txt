"""
Propose a low‑score non‑negative step function.

The routine builds several initial candidates (warm start + random Dirichlet
vectors), then for each start it runs two optimisation phases:

1) **Variance‑reduction gradient descent** – minimise the variance of the
   self‑convolution (i.e. make the convolution as flat as possible).  The
   exact gradient of the squared‑error objective is obtained by a single FFT
   convolution, and the iterate is projected onto the simplex
   {x≥0, Σx = √(2·n)} after each step.

2) **Aggressive mass‑transfer simulated annealing** – a fast O(n) update
   scheme that moves a small amount of mass from an index that contributes
   heavily to the current peak towards a low‑height index.  A square‑root
   cooling schedule keeps the system hot for a long time.

Both phases respect the overall time budget (default 1000 s) and keep the
best sequence found so far.  The final sequence is trimmed of trailing zeros
and stored globally as ``height_sequence_1`` for warm‑starting the next run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – variance‑reduction gradient descent
# ----------------------------------------------------------------------


def _gradient_variance(a: np.ndarray) -> np.ndarray:
    """
    Gradient of the squared‑error   Σ_k ( (a*a)[k] - μ )²   w.r.t. a,
    where μ is the mean of the convolution entries.
    The gradient is computed by a single FFT‑based convolution.
    """
    n = a.shape[0]
    # self‑convolution
    b = _conv_fft(a)                     # length 2·n‑1
    mu = b.mean()
    diff = b - mu                        # length 2·n‑1

    # convolution diff * reversed(a)
    a_rev = a[::-1]
    L = 1 << ((3 * n - 2).bit_length())  # >= (2n‑1)+(n‑1)
    fdiff = np.fft.rfft(diff, n=L)
    fa_rev = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(fdiff * fa_rev, n=L)[: 3 * n - 2]
    grad = 2.0 * conv[n - 1 - np.arange(n)]
    return grad


def variance_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Gradient descent on the variance of the self‑convolution.
    After each step the iterate is projected onto the simplex so that
    Σa = target_sum is preserved.
    """
    a = a0.copy()
    n = a.shape[0]

    # best seen so far
    best_a = a.copy()
    best_val = _score(a)

    start = time.time()
    # a modest step‑size factor; we adapt it by the gradient norm
    step_factor = 0.4

    while time.time() < deadline:
        grad = _gradient_variance(a)
        grad_norm = np.linalg.norm(grad) + 1e-12
        eta = step_factor * target_sum / grad_norm     # ensure scale‐invariance
        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new, target_sum)

        cur_val = _score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] variance‑descent improved score={best_val:.6f}")

        a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – aggressive simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from “bad’’ indices
    (large contribution to the current peak) to “good’’ indices (small height).
    The full convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    # initial temperature – a sizable fraction of the current energy
    T0 = cur_val * 0.9

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        # square‑root cooling (keeps the system hot longer)
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # locate the current peak of the convolution
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # contribution of each index to that peak (a[i]·a[k‑i])
        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour indices with small height
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        # size of the transfer – scaled by temperature
        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        # Metropolis acceptance based on the *current* energy
        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # periodic full recomputation to avoid numerical drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose problem size (balance granularity vs. speed)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)          # the usual normalisation

    # --------------------------------------------------------------
    # Build a list of initial candidates (warm start + random)
    # --------------------------------------------------------------
    init_candidates = []

    if "height_sequence_1" in globals():
        warm = _resize_sequence(np.array(globals()["height_sequence_1"],
                                         dtype=np.float64), n)
        warm = _project_to_simplex(warm, target_sum)
        init_candidates.append(warm)

    rng = np.random.default_rng()
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project_to_simplex(rnd, target_sum)
        init_candidates.append(rnd)

    # a simple decreasing power‑law – often a good starting shape
    powlaw = np.arange(1, n + 1, dtype=np.float64) ** (-0.55)
    powlaw = powlaw * (target_sum / powlaw.sum())
    init_candidates.append(powlaw)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run the two optimisation phases for each start, respecting time
    # --------------------------------------------------------------
    for idx, init in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        print(f"[{time.time():.2f}] ----- start candidate #{idx + 1} -----")

        # ----- Phase 1 : variance‑reduction gradient descent -------------
        # allocate roughly 30 % of the remaining budget to this phase
        remaining = deadline - time.time()
        sub_deadline = time.time() + 0.30 * remaining
        a_var, score_var = variance_descent(init, target_sum, sub_deadline)

        if score_var < best_score:
            best_score = score_var
            best_seq = a_var.copy()
            print(f"[{time.time():.2f}] New global best after variance descent: {best_score:.6f}")

        # ----- Phase 2 : aggressive annealing --------------------------
        if time.time() < deadline:
            a_ann, score_ann = anneal_mass_transfer(a_var, target_sum, deadline)
            if score_ann < best_score:
                best_score = score_ann
                best_seq = a_ann.copy()
                print(f"[{time.time():.2f}] New global best after anneal: {best_score:.6f}")

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a plain decreasing power‑law
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** (-0.55))
        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()

    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
