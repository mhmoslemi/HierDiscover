"""
Hybrid optimization for the step‑function max‑convolution problem.

The pipeline (all stages respect the global budget):
  1. Choose a problem size n proportional to the time budget.
  2. Generate a diverse set of feasible seeds (power‑law, exponential,
     random simplex, previous best) and project them onto the ℓ¹–simplex
     Σa = √(2 n).
  3. Projected Differential Evolution (DE) searches the continuous space.
  4. Deterministic mass‑transfer polishing (a greedy simulated‑annealing
     style move that directly reduces the peak of the auto‑convolution).
  5. Permutation annealing – random swaps of entries (the ℓ¹‑norm is
     invariant) with a temperature schedule; this often improves the
     ordering beyond the simple “alternating” and “even‑spread” heuristics.
  6. Final re‑ordering trial, trimming of trailing near‑zero entries,
     storage of the result for the next run and return.

The objective after normalising Σa = √(2 n) is simply
    max_k (a * a)[k],
so we minimise the peak of the auto‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution a*a using FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (Σa)².
    When Σa = √(2 n) the value coincides with max(a*a).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = s}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence factories
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) then renormalised."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exponential_sequence(n: int, lam: float, target_sum: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) then renormalised."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a *= target_sum / a.sum()
    return a


def _random_simplex(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform random point on the ℓ¹‑simplex."""
    a = rng.random(n)
    a *= target_sum / a.sum()
    return a


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_len`."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Reordering helpers
# ----------------------------------------------------------------------


def _alternating_order(seq: np.ndarray) -> np.ndarray:
    """Largest → leftmost, second largest → rightmost, etc."""
    sorted_vals = np.sort(seq)[::-1]
    n = seq.shape[0]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        place_left = not place_left
    return out


def _even_spread_order(seq: np.ndarray) -> np.ndarray:
    """
    Spread the large values uniformly by placing them on a cyclic sequence
    with a step coprime to the length.
    """
    n = seq.shape[0]
    sorted_vals = np.sort(seq)[::-1]          # descending
    step = max(1, int(round(np.sqrt(n))))
    while np.gcd(step, n) != 1:               # ensure a full cycle
        step = (step + 1) % n or 1
    out = np.empty_like(seq)
    for i, v in enumerate(sorted_vals):
        out[(i * step) % n] = v
    return out


# ----------------------------------------------------------------------
# Deterministic mass‑transfer refinement (simulated‑annealing style)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                deadline: float,
                                target_sum: float,
                                start_temp: float = 0.5,
                                end_temp: float = 1e-4,
                                low_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Greedy mass moves with a decaying Metropolis acceptance rule."""
    a = a0.copy()
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    conv = _conv_fft(a)

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        temp = start_temp * (end_temp / start_temp) ** tau

        kmax = int(np.argmax(conv))

        lo = max(0, kmax - (a.shape[0] - 1))
        hi = min(kmax, a.shape[0] - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        prod = a[idx] * a[kmax - idx]

        donor = idx[np.argmax(prod)]

        low_cnt = max(1, int(len(idx) * low_frac))
        low_candidates = idx[np.argsort(prod)[:low_cnt]]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.5, a[receiver] * 0.5)
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        conv = _conv_fft(a)
        new_score = _score(a)

        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a[donor] += delta
            a[receiver] -= delta
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Differential‑evolution optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _de_optimize(population: list[np.ndarray],
                 target_sum: float,
                 deadline: float,
                 F: float = 0.5,
                 CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Projected DE loop."""
    n = population[0].shape[0]
    rng = np.random.default_rng()
    scores = [_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(len(population)):
            idxs = list(range(len(population)))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = _project_to_simplex(mutant, target_sum)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            trial = _project_to_simplex(trial, target_sum)

            trial_score = _score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
        if time.time() > deadline - 2.0:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Permutation annealing (swap‑based local search on the ordering)
# ----------------------------------------------------------------------


def _permutation_annealing(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random swaps with a temperature schedule, keeping Σa constant.
    Returns the best ordering found (objective = max convolution).
    """
    rng = np.random.default_rng()
    cur = a0.copy()
    cur_max = _max_convolution_fft(cur)
    best = cur.copy()
    best_max = cur_max
    start = time.time()
    n = cur.shape[0]

    while time.time() < deadline:
        elapsed = time.time() - start
        tau = elapsed / (deadline - start)
        # temperature decays from ~0.3 to ~1e-4
        temp = 0.3 * (1e-4 / 0.3) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a swap
        cur[i], cur[j] = cur[j], cur[i]
        new_max = _max_convolution_fft(cur)

        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

    return best, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser that returns a non‑negative step function with a low
    evaluation score. The routine adapts the problem dimension to the
    allocated budget and combines DE, mass‑transfer polishing and a
    permutation‑annealing stage.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ----- problem size -------------------------------------------------
    # n grows roughly linearly with the budget – more variables give a richer search space.
    n = max(4000, min(20000, int(budget_s * 15)))  # 1000 → 15000, 800 → 12000, …
    target_sum = np.sqrt(2.0 * n)

    rng = np.random.default_rng()

    # ----- seed generation -----------------------------------------------
    seeds = []

    # power‑law family
    for p in np.linspace(0.2, 1.2, 10):
        seeds.append(_power_law_sequence(n, float(p), target_sum))

    # exponential family
    for lam in np.linspace(0.0005, 0.02, 8):
        seeds.append(_exponential_sequence(n, float(lam), target_sum))

    # random simplex points
    for _ in range(12):
        seeds.append(_random_simplex(n, target_sum, rng))

    # previous best, if any
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        seeds.append(_resize_sequence(prev, n))

    # jittered copies and simple reorderings to increase diversity
    for seq in list(seeds):
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        seq_j = seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        seeds.append(seq_j)
        seeds.append(_alternating_order(seq_j))
        seeds.append(_even_spread_order(seq_j))

    # ----- initialise DE population ---------------------------------------
    pop_size = max(30, n // 300)          # ≈30‑50 individuals
    population = []

    best_seed = min(seeds, key=_score)
    population.append(best_seed.copy())

    while len(population) < pop_size:
        population.append(_project_to_simplex(rng.choice(seeds).copy(), target_sum))

    # ----- DE stage -------------------------------------------------------
    de_deadline = start_time + 0.55 * budget_s
    if de_deadline > deadline - 6.0:
        de_deadline = deadline - 6.0
    best_seq, best_score = _de_optimize(population, target_sum, de_deadline,
                                        F=0.7, CR=0.9)
    print(f"[{time.time():.2f}] DE finished, best score = {best_score:.6f}")

    # ----- Deterministic mass‑transfer polishing -------------------------
    if time.time() < deadline - 12.0:
        mt_dead = time.time() + min(12.0, deadline - time.time() - 7.0)
        seq_mt, score_mt = _deterministic_mass_transfer(
            best_seq, mt_dead, target_sum,
            start_temp=0.5, end_temp=1e-5, low_frac=0.07)
        if score_mt < best_score:
            best_seq, best_score = seq_mt, score_mt
            print(f"[{time.time():.2f}] After MT best = {best_score:.6f}")

    # ----- Permutation annealing -----------------------------------------
    if time.time() < deadline - 5.0:
        pa_dead = time.time() + min(15.0, deadline - time.time() - 4.0)
        seq_pa, max_pa = _permutation_annealing(best_seq, pa_dead)
        # _permutation_annealing returns the max convolution directly
        if max_pa * 2 * n / (target_sum ** 2) < best_score:
            best_seq = seq_pa
            best_score = max_pa * 2 * n / (target_sum ** 2)
            print(f"[{time.time():.2f}] After perm‑anneal best = {best_score:.6f}")

    # ----- final ordering trial -------------------------------------------
    candidates = [
        best_seq,
        _alternating_order(best_seq),
        _even_spread_order(best_seq)
    ]
    best_seq = min(candidates, key=_max_convolution_fft)
    best_score = _score(best_seq)
    print(f"[{time.time():.2f}] After final reordering best score = {best_score:.6f}")

    # ----- trimming & return -----------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.nonzero(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    globals()["height_sequence_1"] = final_seq
    # one final evaluation for reporting (may be slow for large n)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass
    return final_seq
```

This implementation expands the search space (larger n), enriches the initial seed pool, adds a dedicated permutation‑annealing stage to fine‑tune the ordering, and carefully partitions the time budget among the different optimisation phases. The combination of continuous DE, greedy mass transfer and swap‑based annealing is designed to push the max‑convolution below the target 1.503 within the 1000 s budget.
