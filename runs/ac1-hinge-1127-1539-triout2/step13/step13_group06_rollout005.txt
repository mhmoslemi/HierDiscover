"""
Advanced step‑function optimizer (v2).

The search proceeds in several stages, all respecting a 1000 s wall‑clock
budget:

0.  Choose a problem size n (10 k–20 k) and a target ℓ₂‑norm
    Σa = √(2 n).  The objective then equals max(conv(a,a)).

1.  Coarse parametric sweep over power‑law and exponentially‑tilted
    power‑law families → good starting point.

2.  Jittered local search + a symmetrisation step (a←(a+a[::-1])/2).

3.  Projected Adam descent on a soft‑max approximation of the
    max‑convolution, with many restarts and a β schedule up to 1e7.

4.  Exact deterministic mass‑transfer (DMT2) – never increases the
    objective.

5.  Global random mass moves to escape plateaus.

6.  Greedy pairwise transfer with enlarged donor / receiver pools.

7.  Final polishing: Adam with a very high β and a short simulated‑annealing
    refinement.

All transformations keep the vector non‑negative and preserve the
normalisation by projecting onto the simplex Σa = √(2 n).  The best
sequence found is trimmed of trailing near‑zero entries and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv) / (sum(a)²).
    With the chosen normalisation Σa = √(2 n) this reduces to max(conv).
    """
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the simplex Σx = s , x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float,
                            target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max utilities (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Gradient of the smooth max of conv(a,a) via soft‑max."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    # conv of reversed a with p
    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]

    # gradient of each entry of a (2·shifted vector, then keep centre)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.995) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max approximation of max(conv).
    The schedule for β doubles every ``beta_inc_iter`` iterations up to
    ``max_beta``.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        # Adam update
        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0),
                                 size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam β={beta:.1e} iter={it} "
                  f"score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several short Adam runs from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a_opt, val_opt = _adam_gradient_descent(start,
                                                time.time() + run_budget,
                                                target_sum,
                                                max_beta=2e5,
                                                step_factor=0.8,
                                                gamma=0.995)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.2f}] Restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact deterministic mass‑transfer (DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(a0: np.ndarray,
                                       deadline: float,
                                       target_sum: float,
                                       tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Classic DMT2 refinement: move a small amount of mass from a donor
    that heavily contributes to the current peak towards a receiver that
    contributes little.  The procedure never increases the objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second‑largest value
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]  # pick one peak index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]                     # a_{k-i}
        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Global random mass transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Light‑weight stochastic moves that keep Σa fixed (via projection)."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Random move new best={best_score:.6f}")

    return best_a, best_score

def _pairwise_optimal_transfer(a0: np.ndarray,
                               deadline: float,
                               target_sum: float,
                               donor_frac: float = 0.05,
                               receiver_frac: float = 0.05,
                               min_improve: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Greedy pairwise transfer with larger candidate pools.
    For each peak index k we consider moving mass from a large entry
    (donor) to a small entry (receiver) and pick the most beneficial
    move that actually reduces the peak.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest entries
        receiver_idx = np.argsort(a)[:nr]        # smallest entries

        # pre‑extract the shifted windows for all candidates (cheap view)
        shifted = {}
        cand = np.unique(np.concatenate((donor_idx, receiver_idx)))
        for i in cand:
            shifted[i] = base[i:i + 2 * n - 1]

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d_k = d[k]
                if d_k >= 0.0:
                    continue            # would not lower the peak

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = feasible.min()
                    delta_max = min(delta_max, a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # keep the padded view up‑to‑date
        base[pad_len + i] = a[i]
        base[pad_len + j] = a[j]

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.2f}] Pairwise i={i} j={j} Δ={best_delta:.3e} "
                  f"score={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (new component)
# ----------------------------------------------------------------------


def _simulated_annealing_refinement(a0: np.ndarray,
                                    deadline: float,
                                    target_sum: float) -> tuple[np.ndarray, float]:
    """
    SA that moves mass between many indices while accepting uphill
    moves with a linearly decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]

    conv = _conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = cur_max

    pad_len = n - 1
    base = np.concatenate([np.zeros(pad_len, dtype=np.float64), a,
                           np.zeros(pad_len, dtype=np.float64)])

    rng = np.random.default_rng()
    start_time = time.time()
    T0 = cur_max * 0.02 + 1e-4          # initial temperature scale
    iter_cnt = 0
    refresh_interval = 500
    donor_frac = 0.02
    recv_frac = 0.02

    donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
    receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

    while time.time() < deadline:
        if iter_cnt % refresh_interval == 0:
            donors = np.argsort(-a)[: max(1, int(donor_frac * n))]
            receivers = np.argsort(a)[: max(1, int(recv_frac * n))]

        i = rng.choice(donors)
        j = rng.choice(receivers)
        if i == j:
            continue

        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # compute impact on the convolution
        shift_i = base[i:i + 2 * n - 1]
        shift_j = base[j:j + 2 * n - 1]
        diff = shift_j - shift_i          # effect per unit delta

        new_conv = conv + delta * diff
        new_max = new_conv.max()

        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / (deadline - start_time)), 1e-9)

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            base[pad_len + i] = a[i]
            base[pad_len + j] = a[j]

            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA iter={iter_cnt} new_best={best_score:.6f}")

        # occasional full recomputation to avoid accumulated drift
        if iter_cnt % 5000 == 0:
            conv = _conv_fft(a)
            cur_max = conv.max()
            base = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])

        iter_cnt += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: symmetrisation (optional)
# ----------------------------------------------------------------------


def _symmetrize(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Replace a by (a+a[::-1])/2 and re‑project onto the simplex."""
    sym = (a + a[::-1]) * 0.5
    return _project_to_simplex(sym, target_sum)

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under a 1000 s budget.
    Returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – trade‑off between quality and time
    if budget_s >= 800:
        n = 20000
    elif budget_s >= 400:
        n = 15000
    else:
        n = 10000

    target_sum = np.sqrt(2.0 * n)                  # Σa = √(2 n)

    # ------------------------------------------------------------------
    # Phase 0 – coarse parametric sweep
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # power‑law only
    for p in np.linspace(0.30, 0.90, 13):
        seq = _power_law_sequence(n, float(p), target_sum)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()

    # exponentially‑tilted power law
    for p in np.linspace(0.30, 0.90, 13):
        for q in np.linspace(0.0, 6.0, 13):
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            sc = _score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()

    # small jittered population around the current best
    for _ in range(12):
        jitter = rng.normal(scale=target_sum * 0.005, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()

    print(f"[{time.time():.2f}] After parametric search best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts
    # ------------------------------------------------------------------
    gd_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.01, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         gd_deadline)
        if val_opt < best_score:
            best_score, best_seq = val_opt, a_opt.copy()
            print(f"[{time.time():.2f}] GD improvement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A‑sym – optional symmetrisation
    # ------------------------------------------------------------------
    sym_seq = _symmetrize(best_seq, target_sum)
    sym_score = _score(sym_seq)
    if sym_score < best_score:
        best_score, best_seq = sym_score, sym_seq.copy()
        print(f"[{time.time():.2f}] Symmetrisation improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic mass‑transfer (DMT2) + random escapes
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(best_seq,
                                                             deadline,
                                                             target_sum)
        if val_ref < best_score:
            best_score, best_seq = val_ref, a_ref.copy()
            print(f"[{time.time():.2f}] DMT2 improved best={best_score:.6f}")

        # a few random moves to shake the system
        a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                        deadline,
                                                        target_sum)
        if val_rand < best_score:
            best_score, best_seq = val_rand, a_rand.copy()
            print(f"[{time.time():.2f}] Random move improved best={best_score:.6f}")

        # early exit if we are within the last few seconds
        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Phase C – Enhanced pairwise transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_pair, val_pair = _pairwise_optimal_transfer(best_seq,
                                                      deadline,
                                                      target_sum,
                                                      donor_frac=0.05,
                                                      receiver_frac=0.05)
        if val_pair < best_score:
            best_score, best_seq = val_pair, a_pair.copy()
            print(f"[{time.time():.2f}] Pairwise improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Final Adam polish with a very high β
    # ------------------------------------------------------------------
    if time.time() + 4.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(best_seq,
                                                deadline,
                                                target_sum,
                                                max_beta=1e7,
                                                step_factor=0.5,
                                                gamma=0.99)
        if val_fin < best_score:
            best_score, best_seq = val_fin, a_fin.copy()
            print(f"[{time.time():.2f}] Final Adam improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() + 4.0 < deadline:
        a_sa, val_sa = _simulated_annealing_refinement(best_seq,
                                                       deadline,
                                                       target_sum)
        if val_sa < best_score:
            best_score, best_seq = val_sa, a_sa.copy()
            print(f"[{time.time():.2f}] SA refinement improved best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
