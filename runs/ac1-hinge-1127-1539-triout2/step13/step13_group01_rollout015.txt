                cur_val = 2.0 * n * np.max(_conv_fft(cur_seq)) / (target_sum**2)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()

        while time.time() < ann_deadline:

            hi = min(k_max, n - 1)

            a_new[receiver] += delta

        # else break
        if time.time() + 5.0 > deadline:

        # Random restart: new init random
        rng = np.random.default_rng()
        cur_seq = _project_to_simplex(rng.random(n), target_sum)
        # Continue loop

    except Exception as exc:
        print(f"[{time.time():.2f}] Evaluation error: {exc}")
    globals()["height_sequence_1"] = final_seq
    return final_seq

"""
Propose a low‑score non‑negative step sequence.

The method works in two scales.
* A modest-sized problem (≈2000 points) is first optimised with
  - a smoothed sub‑gradient descent on the maximal self‑convolution,
  - a linear‑programming “mass‑increase” step (maximise the L1‑norm of a
    partner vector g that respects the current peak bound).
  The LP is solved with ``scipy.optimize.linprog`` (highs solver).
* The resulting shape is linearly interpolated to the final problem size
  (8000–18000 points, depending on the time budget) and refined by a
  classic projected sub‑gradient descent (using the exact gradient of the
  active peak) together with a cheap simulated‑annealing mass‑transfer.
The whole routine respects the supplied time budget and stores the best
sequence in the global variable ``height_sequence_1`` for warm‑starts.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Phase 1 – smoothed sub‑gradient descent on a small problem
# ----------------------------------------------------------------------


def _smoothed_subgrad_step(a: np.ndarray, target_sum: float,
                           beta: float, eta: float) -> np.ndarray:
    """
    One iteration of a smoothed sub‑gradient descent.
    Uses a softmax over the full convolution to obtain a smoother
    direction than the pure max‑gradient.
    """
    n = a.shape[0]
    b = _conv_fft(a)
    max_b = b.max()
    # softmax weights (numerically stable)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum
    # gradient = 2 * correlation(a, w)
    # np.correlate gives sum_i a[i] * w[i+lag]
    corr = np.correlate(a, w, mode='full')
    grad = 2.0 * corr[len(w) - 1: len(w) - 1 + n]
    a_new = a - eta * grad
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
# Phase 2 – LP based mass‑increase step (small problem)
# ----------------------------------------------------------------------


def _lp_mass_increase(a: np.ndarray, M: float, target_sum: float):
    """
    Solve   max Σ g_i   s.t.   (a * g)_k ≤ M  (for all k) , g ≥ 0.
    Returns a feasible vector g (scaled to the required simplex sum)
    or ``None`` if the LP fails.
    """
    n = a.shape[0]
    eps = 1e-9
    coeffs = np.maximum(a, eps)          # avoid zero columns → unbounded LP

    # Build the dense constraint matrix A (size (2n‑1) × n)
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A.fill(0.0)
    for i in range(n):
        A[i:i + n, i] = coeffs[i]

    c = -np.ones(n)                       # maximise sum ⇒ minimise –sum
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    sum_g = g.sum()
    if sum_g < 1e-12:
        return None
    # Scale to the required simplex sum
    scale = target_sum / sum_g
    return g * scale


# ----------------------------------------------------------------------
# Phase 3 – classic sub‑gradient descent on the large problem
# ----------------------------------------------------------------------


def _max_peak_subgrad_step(a: np.ndarray, target_sum: float, eta: float) -> np.ndarray:
    """Sub‑gradient descent on the *active* peak of a*a."""
    n = a.shape[0]
    b = _conv_fft(a)
    k_max = int(np.argmax(b))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k_max - idx]
    a_new = a - eta * grad
    a_new = _project_to_simplex(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
# Simple simulated‑annealing mass transfer (large problem)
# ----------------------------------------------------------------------


def _anneal_step(a: np.ndarray, target_sum: float,
                 cur_val: float, T: float) -> tuple[np.ndarray, float]:
    """
    Perform a single mass‑transfer move with Metropolis acceptance.
    Returns (new_a, new_val) – either the updated state or the original.
    """
    n = a.shape[0]
    b = _conv_fft(a)
    k_max = int(np.argmax(b))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)

    contrib = a[idx] * a[k_max - idx]
    if contrib.sum() > 0.0:
        donor = np.random.default_rng().choice(idx,
                                               p=contrib / contrib.sum())
    else:
        donor = np.random.default_rng().integers(lo, hi + 1)

    inv = 1.0 / (a + 1e-12)
    inv /= inv.sum()
    receiver = np.random.default_rng().choice(n, p=inv)

    if donor == receiver:
        return a, cur_val

    max_delta = min(a[donor] * 0.08, target_sum * 0.02)
    if max_delta <= 0.0:
        return a, cur_val
    delta = np.random.random() * max_delta * (T / (T + 1e-12))
    if delta <= 0.0:
        return a, cur_val

    a_new = a.copy()
    a_new[donor] -= delta
    a_new[receiver] += delta
    val_new = 2.0 * len(a_new) * _conv_fft(a_new).max() / (target_sum ** 2)

    # Metropolis acceptance
    if val_new <= cur_val or np.random.random() < np.exp(-(val_new - cur_val) / (T + 1e-12)):
        return a_new, val_new
    return a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem sizes (small for LP, large for final score)
    # --------------------------------------------------------------
    if budget_s >= 800:
        large_n = 18000
        small_n = 2000
    elif budget_s >= 500:
        large_n = 12000
        small_n = 1500
    else:
        large_n = 8000
        small_n = 1000

    target_sum_small = np.sqrt(2.0 * small_n)
    target_sum_large = np.sqrt(2.0 * large_n)

    # --------------------------------------------------------------
    # Initialise the small‑scale problem (warm start if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        warm = _resize_sequence(warm, small_n)
    else:
        warm = np.random.random(small_n)

    a_small = _project_to_simplex(warm, target_sum_small)

    best_small = a_small.copy()
    best_val_small = 2.0 * small_n * _conv_fft(a_small).max() / (target_sum_small ** 2)

    # --------------------------------------------------------------
    # Phase 1 – optimise the small problem (LP + smoothed GD)
    # --------------------------------------------------------------
    phase_end = start + min(150.0, 0.35 * (deadline - start))

    rng = np.random.default_rng()
    it_counter = 0
    while time.time() < phase_end:
        # ---- a few smoothed sub‑gradient steps ----
        for _ in range(200):
            if time.time() >= phase_end:
                break
            it_counter += 1
            beta = 5.0 + 30.0 * (it_counter % 200) / 200.0   # slowly increase
            eta = 0.4 * (1.0 - (it_counter % 200) / 200.0) + 0.02
            a_small = _smoothed_subgrad_step(a_small, target_sum_small,
                                             beta, eta)
            cur_val = 2.0 * small_n * _conv_fft(a_small).max() / (target_sum_small ** 2)
            if cur_val < best_val_small:
                best_val_small = cur_val
                best_small = a_small.copy()
                print(f"[{time.time():.2f}] Small smoothed GD improvement: {best_val_small:.6f}")

        # ---- LP mass‑increase step ----
        M = _conv_fft(a_small).max()
        g_scaled = _lp_mass_increase(a_small, M, target_sum_small)
        if g_scaled is not None:
            t_mix = 0.2
            a_small = (1.0 - t_mix) * a_small + t_mix * g_scaled
            a_small = _project_to_simplex(a_small, target_sum_small)
            cur_val = 2.0 * small_n * _conv_fft(a_small).max() / (target_sum_small ** 2)
            if cur_val < best_val_small:
                best_val_small = cur_val
                best_small = a_small.copy()
                print(f"[{time.time():.2f}] LP‑mix improved: {best_val_small:.6f}")

    # --------------------------------------------------------------
    # Upscale to the large problem size
    # --------------------------------------------------------------
    a_large = _resize_sequence(best_small, large_n)
    a_large = _project_to_simplex(a_large, target_sum_large)

    best_large = a_large.copy()
    best_val_large = 2.0 * large_n * _conv_fft(a_large).max() / (target_sum_large ** 2)

    # --------------------------------------------------------------
    # Phase 2 – refined optimisation on the large scale
    # --------------------------------------------------------------
    while time.time() < deadline:
        # Sub‑gradient descent on the active peak (fast, O(n log n))
        elapsed = time.time() - start
        total = deadline - start
        eta = 1.0 * (1.0 - elapsed / total) + 0.01
        a_large = _max_peak_subgrad_step(a_large, target_sum_large, eta)
        cur_val = 2.0 * large_n * _conv_fft(a_large).max() / (target_sum_large ** 2)
        if cur_val < best_val_large:
            best_val_large = cur_val
            best_large = a_large.copy()
            print(f"[{time.time():.2f}] Large sub‑gradient best: {best_val_large:.6f}")

        # Occasionally perform a cheap annealing move to escape plateaus
        if int(time.time() - start) % 30 == 0:
            # initialise temperature from current energy
            T0 = cur_val
            a_tmp, cur_tmp = a_large, cur_val
            # one move (the full O(n) recompute is affordable at this scale)
            a_tmp, cur_tmp = _anneal_step(a_tmp, target_sum_large, cur_tmp, T0)
            if cur_tmp < best_val_large:
                best_val_large = cur_tmp
                best_large = a_tmp.copy()
                a_large = a_tmp
                print(f"[{time.time():.2f}] Large anneal best: {best_val_large:.6f}")

        # stop quickly if only a few seconds are left
        if time.time() + 4.0 > deadline:
            break

    # --------------------------------------------------------------
    # Clean‑up, final evaluation and warm‑start storage
    # --------------------------------------------------------------
    arr = np.array(best_large, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # final external evaluation (may clip / normalise internally)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
