"""
Search for a step function (non‑negative sequence) that minimises the
maximal self‑convolution.  The routine builds on the previous four‑phase
scheme, adds a richer parametric pre‑search (including beta‑type families),
uses a tighter deterministic mass‑transfer step that selects the donor
according to the true sub‑gradient, and finishes with a greedy hill‑climb
that swaps tiny amounts of mass between random indices.  All operations stay
on the ℓ1‑simplex {a≥0, Σa = √(2·n)} so the returned objective is exactly
the maximal convolution value.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    condition = u - cssv / rho_arr > 0
    if not np.any(condition):
        theta = 0.0
    else:
        rho = np.where(condition)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (fast pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^{α} (n-i)^{β}, α,β ≤ 0 give a bowl shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # shift exponents to avoid overflow for very negative values
    a = (i ** alpha) * ((n - i + 1) ** beta)
    # Clip negatives to zero just in case
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        # fallback to uniform
        a = np.ones_like(a)
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (max convolution)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective: maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    """Return (sub‑gradient of the soft‑max surrogate, raw convolution)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max surrogate, with projection onto the simplex."""
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            # occasional jitter to avoid exact plateaus
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.30)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,
                                         target_sum,
                                         max_beta=2e5,
                                         step_factor=0.8,
                                         gamma=0.995)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         target_sum: float,
                         deadline: float,
                         max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient method for the max‑convolution objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        # compute second max
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        # pick a random maximal index
        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient for selected k
        g = np.zeros_like(a)
        # for each i in idx the contribution is a_{k-i}
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:
            # inject tiny jitter and continue
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=a.shape), target_sum)
            it += 1
            continue

        # step size derived from target reduction
        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        # backtracking line search (up to 5 reductions)
        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * g, target_sum)
            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                improved = True
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # jitter
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.0005, size=a.shape), target_sum)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic mass‑transfer (exact, gradient‑aware)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Improved exact mass transfer using true sub‑gradient values."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second max
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        # pick the first maximal index for simplicity
        k = idx_max[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]          # values a_{k-i}
        # true sub‑gradient = 2 * partner (except possibly centre, factor 2 already correct)
        grad_vals = 2.0 * partner

        # donor: among indices with positive mass, pick one with largest gradient
        positive_mask = a[idx] > 0
        if not np.any(positive_mask):
            break
        donor_candidates = idx[positive_mask]
        donor_grad = grad_vals[np.isin(idx, donor_candidates)]
        donor = donor_candidates[np.argmax(donor_grad)]

        # receiver: pick index with smallest gradient (could be zero)
        receiver = idx[np.argmin(grad_vals)]

        # amount that would bring the current maximum down to the second maximum
        g_d = grad_vals[np.where(idx == donor)[0][0]]
        g_r = grad_vals[np.where(idx == receiver)[0][0]]
        denom = g_d - g_r
        if denom <= 0.0:
            # fallback to original DMT logic
            donor_scores = a[idx] * partner
            donor = idx[np.argmax(donor_scores)]
            # choose a receiver with minimal partner (different from donor)
            sorted_loc = np.argsort(partner)
            receiver = None
            for r_loc in sorted_loc:
                r = idx[r_loc]
                if r != donor:
                    receiver = r
                    break
            if receiver is None:
                break
            g_d = 2.0 * a[k - donor]
            g_r = 2.0 * a[k - receiver]
            denom = g_d - g_r
            if denom <= 0.0:
                # give up this iteration
                a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
                continue

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            # jitter to escape
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            continue

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] MassTransfer it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Global random mass‑transfer (exploration)
# ----------------------------------------------------------------------


def _global_random_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float) -> tuple[np.ndarray, float]:
    """Simple stochastic redistribution to escape plateaus."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)

    top_frac = 0.02
    low_frac = 0.02

    while time.time() < deadline - 1.0:
        k_top = max(1, int(top_frac * n))
        idx_desc = np.argsort(-a)
        donor = idx_desc[rng.integers(k_top)]

        k_low = max(1, int(low_frac * n))
        idx_asc = np.argsort(a)
        receiver = idx_asc[rng.integers(k_low)]

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.1
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a[donor] -= delta
        a[receiver] += delta
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Global random new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty descent (soft constraint on the maximal convolution)
# ----------------------------------------------------------------------


def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float,
                       max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Attempt to drive the maximal convolution below ``target_max``.
    Uses the convex‑concave penalty p = max(0, conv - target_max) and a
    simple projected sub‑gradient step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()
    it = 0
    n = a.shape[0]

    while time.time() < deadline and it < max_iters and best_max > target_max + 1e-12:
        conv = _conv_fft(a)
        cur_max = conv.max()
        if cur_max <= target_max:
            best_max = cur_max
            best_a = a.copy()
            break

        # penalty vector: only entries exceeding target
        p = conv - target_max
        p[p < 0] = 0.0

        # gradient of penalty wrt a: 2 * (p * a_rev)_{n-1…}
        a_rev = a[::-1]
        grad_full = _conv_full_fft(p, a_rev)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            it += 1
            continue

        step = 0.9 * (cur_max - target_max) / (norm2 + 1e-12)
        step = min(step, target_sum * 0.5 / (np.linalg.norm(grad) + 1e-12))

        improved = False
        for _ in range(5):
            a_candidate = _project_to_simplex(a - step * grad, target_sum)
            cand_max = _max_convolution_fft(a_candidate)
            if cand_max < cur_max - 1e-12:
                a = a_candidate
                best_max = cand_max
                best_a = a_candidate.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Hill‑climbing refinement (tiny random transfers that are kept only when improving)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                target_sum: float,
                deadline: float,
                max_trials: int = 100000) -> tuple[np.ndarray, float]:
    """
    Greedy local search: repeatedly pick a random donor/receiver pair,
    move a tiny fraction of the donor's mass and keep the move if it strictly
    reduces the maximal convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)
    rng = np.random.default_rng()
    trial = 0

    while time.time() < deadline and trial < max_trials:
        # weighted random donor (probability proportional to a)
        if a.sum() == 0:
            break
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            trial += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            trial += 1
            continue

        # move a small fraction (up to 5% of donor)
        epsilon = rng.random() * 0.05 * a[donor]
        if epsilon <= 0.0:
            trial += 1
            continue

        a[donor] -= epsilon
        a[receiver] += epsilon
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            # keep move
        else:
            # revert
            a[donor] += epsilon
            a[receiver] -= epsilon
            a = _project_to_simplex(a, target_sum)
        trial += 1

    return a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the maximal
    self‑convolution (the evaluation score).  The routine follows a
    multi‑phase scheme: fast parametric pre‑search, Adam optimisation,
    exact sub‑gradient descent, an improved deterministic mass‑transfer,
    optional global random moves, a final hill‑climb and a penalty
    descent polishing.  All intermediate vectors are kept on the simplex
    Σa = √(2·n) so the returned value equals the maximal convolution.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # small safety margin

    # Choose problem size; larger n gives more flexibility but also costs.
    # Use 15000 when we have enough budget, otherwise fall back.
    if budget_s >= 900:
        n = 15000
    elif budget_s >= 600:
        n = 12000
    else:
        n = 8000
    target_sum = np.sqrt(2.0 * n)   # Σa = √(2·n) → evaluation = max convolution

    # ------------------------------------------------------------------
    # Phase 0 – rich parametric pre‑search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng()

    # (a) pure power‑law
    for p in np.linspace(0.2, 1.2, 25):
        seq = _power_law_sequence(n, float(p), target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (b) power‑law + exponential tail
    p_grid = np.linspace(0.2, 1.0, 15)
    q_grid = np.linspace(0.0, 5.0, 15)
    for p in p_grid:
        for q in q_grid:
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (c) beta‑type family (bowl shaped)
    alpha_grid = np.linspace(-1.5, 0.0, 13)
    beta_grid = np.linspace(-1.5, 0.0, 13)
    for alpha in alpha_grid:
        for beta_ in beta_grid:
            seq = _beta_sequence(n, float(alpha), float(beta_), target_sum)
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()

    # (d) simple sine shape
    seq = _sin_sequence(n, target_sum)
    val = _score(seq)
    if val < best_score:
        best_score = val
        best_seq = seq.copy()

    # (e) Gaussian bumps
    for sigma in np.linspace(n / 25, n / 6, 7):
        seq = _gaussian_sequence(n, sigma, target_sum)
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()

    # (f) reuse previous global best if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev
        resized = _project_to_simplex(resized, target_sum)
        val = _score(resized)
        if val < best_score:
            best_score = val
            best_seq = resized.copy()

    # add a few jittered populations around current best
    for _ in range(6):
        jitter = rng.normal(scale=target_sum * 0.004, size=n)
        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (soft‑max surrogate)
    # ------------------------------------------------------------------
    # Allocate roughly 25 % of remaining time
    ad_deadline = time.time() + (deadline - time.time()) * 0.25
    while time.time() < ad_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         ad_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (true max)
    # ------------------------------------------------------------------
    sg_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         sg_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (gradient‑aware)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.30
    a_mt, val_mt = _deterministic_mass_transfer(best_seq,
                                                mt_deadline,
                                                target_sum)
    if val_mt < best_score:
        best_score = val_mt
        best_seq = a_mt.copy()
        print(f"[{time.time():.2f}] MassTransfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Global random redistribution (exploration)
    # ------------------------------------------------------------------
    rand_deadline = time.time() + (deadline - time.time()) * 0.15
    a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                    rand_deadline,
                                                    target_sum)
    if val_rand < best_score:
        best_score = val_rand
        best_seq = a_rand.copy()
        print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Hill climbing (tiny directed moves)
    # ------------------------------------------------------------------
    hc_deadline = time.time() + (deadline - time.time()) * 0.10
    a_hc, val_hc = _hill_climb(best_seq,
                               target_sum,
                               hc_deadline,
                               max_trials=200000)
    if val_hc < best_score:
        best_score = val_hc
        best_seq = a_hc.copy()
        print(f"[{time.time():.2f}] Hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase F – Penalty descent to push the max a little lower
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        # try to lower the bound by a modest amount (≈0.003)
        target = best_score - 0.0035
        a_pen, val_pen = _penalty_lower_max(best_seq,
                                            target_sum,
                                            deadline,
                                            target)
        if val_pen < best_score:
            best_score = val_pen
            best_seq = a_pen.copy()
            print(f"[{time.time():.2f}] Penalty descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final safety evaluation (the host will call evaluate_sequence)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception:
        pass

    return final_seq

"""
Search for a step function (non‑negative sequence) that minimises the
maximal self‑convolution.  The routine builds on the previous four‑phase
scheme, adds a richer parametric pre‑search (including beta‑type families),
uses a tighter deterministic mass‑transfer step that selects the donor
according to the true sub‑gradient, and finishes with a greedy hill‑climb
that swaps tiny amounts of mass between random indices.  All operations stay
on the ℓ₁‑simplex {a≥0, Σa = √(2·n)} so the returned objective is exactly the
maximal convolution value.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·n‑1)."""
    n = a.size
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using FFT."""
    size = _next_pow2(x.size + y.size - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: x.size + y.size - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of a*a (linear convolution)."""
    return float(_conv_fft(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    condition = u - cssv / rho_arr > 0
    if not np.any(condition):
        theta = 0.0
    else:
        rho = np.where(condition)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families (fast pre‑search)
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _sin_sequence(n: int, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0))
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _beta_sequence(n: int, alpha: float, beta: float, target_sum: float) -> np.ndarray:
    """Beta‑type bowl: a_i ∝ (i+1)^α (n-i)^β (α,β ≤ 0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** alpha) * ((n - i + 1) ** beta)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.ones_like(a)
    a *= target_sum / a.sum()
    return a

# Scoring (max convolution)
# ----------------------------------------------------------------------


def _score(a: np.ndarray) -> float:
    """Return the objective: maximal self‑convolution entry."""
    return _max_convolution_fft(a)


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=size)
    b = np.fft.irfft(af * af, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    ar_f = np.fft.rfft(a_rev, n=size)
    pr_f = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(ar_f * pr_f, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b

# Adam‑style projected gradient descent (soft‑max)

def _adam_gradient_descent(a0: np.ndarray,
                           deadline: float,
                           target_sum: float,
                           max_beta: float = 2e5,
                           step_factor: float = 0.8,
                           gamma: float = 0.999) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 250

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1.0 - gamma) * g
        v = gamma * v + (1.0 - gamma) * (g * g)
        m_hat = m / (1.0 - gamma ** (it + 1))
        v_hat = v / (1.0 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        if it % 200 == 0:
            jitter = rng.normal(scale=target_sum * 0.001 / (beta + 1.0), size=a.shape)
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(init: np.ndarray,
                                   target_sum: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """Run several Adam restarts from jittered starts."""
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.30)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)

        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(start,
                                         time.time() + run_budget,

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

def _subgradient_descent(a0: np.ndarray,

                         deadline: float,

    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.flatnonzero(conv >= max_val - 1e-12)

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        delta_target = max_val - second_max
        if delta_target < 1e-14:
            break

        k = idx_max[rng.integers(len(idx_max))]
        n = a.shape[0]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        norm2 = np.dot(g, g)
        if norm2 == 0.0:

            it += 1
            continue

        step = 0.9 * delta_target / norm2
        step = min(step, target_sum * 0.4 / (np.linalg.norm(g) + 1e-12))

        improved = False
        for _ in range(5):

            cand_score = _score(a_candidate)
            if cand_score < best_score - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_score = cand_score
                improved = True
                print(f"[{time.time():.2f}] Subgrad it={it:5d} step={step:.3e} score={best_score:.6f}")
                break
            step *= 0.5
        if not improved:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.0005, size=a.shape), target_sum)

        it += 1

    return best_a, best_score

def _deterministic_mass_transfer(a0: np.ndarray,
                                 deadline: float,
                                 target_sum: float,
                                 tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Improved exact mass transfer using the true sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take the first maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k-i}
        grad_vals = 2.0 * partner           # true sub‑gradient entries

        # donor: positive mass & largest gradient
        pos_mask = a[idx] > 0
        if not np.any(pos_mask):
            break
        donor_cands = idx[pos_mask]
        donor_grad = grad_vals[np.isin(idx, donor_cands)]
        donor = donor_cands[np.argmax(donor_grad)]

        # receiver: smallest gradient (could be zero)
        receiver = idx[np.argmin(grad_vals)]

        # reduction that would bring the current max down to the second max
        g_d = grad_vals[np.where(idx == donor)[0][0]]
        g_r = grad_vals[np.where(idx == receiver)[0][0]]
        denom = g_d - g_r
        if denom <= 0.0:
            # fallback to the original product‑based heuristic
            donor_scores = a[idx] * partner
            donor = idx[np.argmax(donor_scores)]
            sorted_loc = np.argsort(partner)
            receiver = None
            for r_loc in sorted_loc:
                r = idx[r_loc]
                if r != donor:
                    receiver = r
                    break
            if receiver is None:
                break
            g_d = 2.0 * a[k - donor]
            g_r = 2.0 * a[k - receiver]
            denom = g_d - g_r
            if denom <= 0.0:
                a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
                continue

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.9, delta_needed * 0.99)
        if max_delta <= 0.0:
            a = _project_to_simplex(a + rng.normal(scale=target_sum * 0.001, size=n), target_sum)
            continue

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] MassTransfer it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

# Penalty descent (soft constraint on the maximal convolution)

def _penalty_lower_max(a0: np.ndarray,
                       target_sum: float,
                       deadline: float,
                       target_max: float,
                       max_iters: int = 20000) -> tuple[np.ndarray, float]:

    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution_fft(a)
    rng = np.random.default_rng()

    n = a.shape[0]

# ----------------------------------------------------------------------
# Hill‑climbing refinement (tiny random transfers that are kept only when improving)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                target_sum: float,
                deadline: float,
                max_trials: int = 100000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_score = _score(a)
    rng = np.random.default_rng()
    n = a.shape[0]
    trial = 0

    while time.time() < deadline and trial < max_trials:
        donor = rng.choice(n, p=a / a.sum())
        if a[donor] <= 1e-14:
            trial += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            trial += 1
            continue

        epsilon = rng.random() * 0.05 * a[donor]   # move up to 5 % of donor's mass
        if epsilon <= 0.0:
            trial += 1
            continue

        a[donor] -= epsilon
        a[receiver] += epsilon
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            # keep the move
        else:
            # revert
            a[donor] += epsilon
            a[receiver] -= epsilon
            a = _project_to_simplex(a, target_sum)

        trial += 1

    return a, best_score

        cand = best_seq + jitter
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, target_sum)
        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()

    print(f"[{time.time():.2f}] After parametric search best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam‑style GD with restarts (soft‑max surrogate)
    # ------------------------------------------------------------------
    ad_deadline = time.time() + (deadline - time.time()) * 0.25
    while time.time() < ad_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.015, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(start_vec,
                                                         target_sum,
                                                         ad_deadline)
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] AdamGD improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Exact sub‑gradient descent (true max)
    # ------------------------------------------------------------------
    sg_deadline = time.time() + (deadline - time.time()) * 0.20
    a_sg, val_sg = _subgradient_descent(best_seq,
                                         target_sum,
                                         sg_deadline)
    if val_sg < best_score:
        best_score = val_sg
        best_seq = a_sg.copy()
        print(f"[{time.time():.2f}] Subgradient descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Deterministic mass‑transfer (gradient‑aware)
    # ------------------------------------------------------------------
    mt_deadline = time.time() + (deadline - time.time()) * 0.30
    a_mt, val_mt = _deterministic_mass_transfer(best_seq,
                                                mt_deadline,
                                                target_sum)
    if val_mt < best_score:
        best_score = val_mt
        best_seq = a_mt.copy()
        print(f"[{time.time():.2f}] MassTransfer improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Global random redistribution (exploration)
    # ------------------------------------------------------------------
    rand_deadline = time.time() + (deadline - time.time()) * 0.15
    a_rand, val_rand = _global_random_mass_transfer(best_seq,
                                                    rand_deadline,
                                                    target_sum)
    if val_rand < best_score:
        best_score = val_rand
        best_seq = a_rand.copy()
        print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Hill climbing (tiny directed moves)
    # ------------------------------------------------------------------
    hc_deadline = time.time() + (deadline - time.time()) * 0.10
    a_hc, val_hc = _hill_climb(best_seq,
                               target_sum,
                               hc_deadline,
                               max_trials=200000)
    if val_hc < best_score:
        best_score = val_hc
        best_seq = a_hc.copy()
        print(f"[{time.time():.2f}] Hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase F – Penalty descent to push the max a little lower
    # ------------------------------------------------------------------
    if time.time() + 5.0 < deadline:
        target = best_score - 0.0035
        a_pen, val_pen = _penalty_lower_max(best_seq,
                                            target_sum,
                                            deadline,
                                            target)
        if val_pen < best_score:
            best_score = val_pen
            best_seq = a_pen.copy()
            print(f"[{time.time():.2f}] Penalty descent improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries
    # ------------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final safety evaluation (the host will call evaluate_sequence)
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception:
        pass

    return final_seq
