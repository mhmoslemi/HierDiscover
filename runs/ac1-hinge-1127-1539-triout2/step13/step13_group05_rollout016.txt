def _exponential_subgradient_descent(a0, deadline, target_sum, eta0=0.01):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    it = 0
    # Exponential update: a = a * exp(-η * g)
    while time.time() < deadline:
        # Compute convolution and max index
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))
        # compute gradient (subgradient) g_i = 2 * a_{k - i} for indices i where 0 <= k-i < n
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i range: lo..hi; then partner index = k - i
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]
        # Exponential update:
        eta = eta0 / np.sqrt(it + 1)  # decreasing
        a = a * np.exp(-eta * g)
        # project to simplex
        a = _project_to_simplex(a, target_sum)
        # evaluate score
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] ExpSubgrad it={it} score={best_score:.6f}")
        # optionally break if improvement very small
        if it % 1000 == 0 and time.time() > deadline - 30:
            # maybe break early if close to deadline
            pass
        it += 1
    return best_a, best_score

def _simulated_annealing(a0, deadline, target_sum):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    rng = np.random.default_rng()
    # temperature schedule: start high, decay
    init_temp = (best_score) * 0.1  # maybe 0.1*score
    # number of steps approx = (deadline - time.time()) / time_per_move
    # we can loop while time remaining > small amount
    while time.time() < deadline:
        remaining = deadline - time.time()
        # estimate per move ~0.02 sec? just loop
        # pick donor idx weighted by a * partner at current peak
        conv = _conv_fft(a)
        max_idx = int(np.argmax(conv))
        # compute contributions for peak: partner = a[max_idx - i] as earlier
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner = np.zeros(n, dtype=np.float64)
        partner[idx_range] = a[max_idx - idx_range]
        donor_weights = a * partner
        donor_weights_sum = donor_weights.sum()
        if donor_weights_sum == 0:
            break
        # sample donor proportional to weight
        donor = rng.choice(n, p=donor_weights / donor_weights_sum)
        # sample receiver from low partner indices: choose among indices where partner small
        # choose random among lowest 5% of partner values
        low_idxs = np.argsort(partner)[: max(1, n // 20)]
        receiver = rng.choice(low_idxs)
        if donor == receiver:
            continue
        # propose delta up to 0.5 * a[donor] maybe
        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta == 0:
            continue
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # project to simplex to correct sum drift
        a_new = _project_to_simplex(a_new, target_sum)
        new_score = _score(a_new)
        # acceptance probability
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # temperature schedule: linear decay
            # calculate fraction of time elapsed
            frac = (time.time() - start_time) / (deadline - start_time)
            temp = init_temp * (1 - frac)  # linearly decreasing to 0
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True
        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best {best_score:.6f}")
    return best_a, best_score

"""
Efficient search for a non‑negative step function with a small
self‑convolution peak.  The algorithm combines:

* a wide parametric warm‑start (power‑law, exponentially‑tilted,
  Gaussian, double‑exponential, symmetric power‑law, sinusoid,
  cosine, quadratic families),

* several rounds of projected Adam/GD on a soft‑max approximation of
  the max‑norm,

* exact deterministic mass‑transfer (DMT2) and a global
  peak‑mass‑transfer move,

* pairwise optimal transfers,

* exponential‑subgradient descent on the true max‑norm (sub‑gradient
  of the peak index),

* a short simulated‑annealing phase for escaping tiny basins,

* a final high‑β Adam polishing step.

All phases respect a global time budget (default 1000 s) and the
sequence is always projected onto the simplex Σa = √(2 n), a_i ≥ 0.
The returned list is trimmed of trailing near‑zero entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a using FFT."""
    return float(_conv_fft(a).max())


def _score(a: np.ndarray) -> float:
    """Objective value: 2·n·max(b)/(sum(a)^2).  If sum(a)=√(2n) the
    denominator equals 2·n and the score equals max(b)."""
    n = a.shape[0]
    max_b = _max_convolution_fft(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex Σx = s,  x ≥ 0
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto {x_i ≥ 0, Σx_i = s} (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the fast pre‑search
# ----------------------------------------------------------------------


def _power_law_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a *= target_sum / a.sum()
    return a


def _exp_power_law_sequence(n: int, p: float, q: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) * np.exp(-q * i / n)
    a *= target_sum / a.sum()
    return a


def _gaussian_sequence(n: int, sigma: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (i / sigma) ** 2)
    a *= target_sum / a.sum()
    return a


def _double_exp_sequence(n: int, p: float, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-p * i) + np.exp(-p * (n - 1 - i))
    a *= target_sum / a.sum()
    return a


def _sym_power_law_sequence(
    n: int, p1: float, p2: float, target_sum: float
) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p1)) * ((n - i + 1) ** (-p2))
    a += 1e-12
    a *= target_sum / a.sum()
    return a


def _sinusoidal_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a += 1e-12
    a *= target_sum / a.sum()
    return a


def _cosine_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a += 1e-12
    a *= target_sum / a.sum()
    return a


def _quadratic_sequence(n: int, target_sum: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i * (n + 1 - i)
    a += 1e-12
    a *= target_sum / a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient (used by Adam‑GD)
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray, beta: float):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=size)
    b = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]

    max_b = np.max(b)
    exps = np.exp(beta * (b - max_b))
    p = exps / np.sum(exps)

    a_rev = a[::-1]
    Ar_fft = np.fft.rfft(a_rev, n=size)
    Pr_fft = np.fft.rfft(p, n=size)
    conv_rev = np.fft.irfft(Ar_fft * Pr_fft, n=size)[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, b


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _adam_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    max_beta: float = 2e5,
    step_factor: float = 0.8,
    gamma: float = 0.999,
) -> tuple[np.ndarray, float]:
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_val = _score(a)

    beta = 0.5
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    it = 0
    beta_inc_iter = 300

    while time.time() < deadline:
        g, _ = _gradient_softmax(a, beta)

        m = gamma * m + (1 - gamma) * g
        v = gamma * v + (1 - gamma) * (g * g)
        m_hat = m / (1 - gamma ** (it + 1))
        v_hat = v / (1 - gamma ** (it + 1))

        step = step_factor * target_sum / (np.sqrt(v_hat) + eps)
        a -= step * m_hat
        a = _project_to_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % 250 == 0:
            jitter = rng.normal(
                scale=target_sum * 0.001 / (beta + 1.0), size=a.shape
            )
            a += jitter
            a = _project_to_simplex(a, target_sum)

        val = _score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD it={it:5d} β={beta:6.1f} "
                  f"score={best_val:.6f}")

        if it % beta_inc_iter == 0 and beta < max_beta:
            beta = min(beta * 2.0, max_beta)

        it += 1

    return best_a, best_val


def _gradient_descent_with_restarts(
    init: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    best_a = init.copy()
    best_val = _score(best_a)
    rng = np.random.default_rng()

    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        run_budget = min(30.0, remaining * 0.35)
        if run_budget < 5.0:
            break

        jitter = rng.normal(scale=target_sum * 0.02, size=best_a.shape)
        start = best_a + jitter
        start = np.maximum(start, 0.0)
        start = _project_to_simplex(start, target_sum)

        a, val = _adam_gradient_descent(
            start,
            time.time() + run_budget,
            target_sum,
            max_beta=2e5,
            step_factor=0.8,
            gamma=0.995,
        )
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[{time.time():.2f}] AdamGD restart improved best={best_val:.6f}")

    return best_a, best_val

# Exact deterministic mass‑transfer (the previous DMT2)
# ----------------------------------------------------------------------


def _deterministic_mass_transfer_exact2(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        idx_max = np.where(conv >= max_val - 1e-12)[0]

        # second largest value (ignore current maxima)
        conv2 = conv.copy()
        conv2[idx_max] = -np.inf
        second_max = conv2.max()

        if max_val - second_max < tol:
            break

        k = idx_max[0]                     # take one maximal index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        partner = a[k - idx]               # a_{k‑i}

        donor_mask = a[idx] > 0
        if not np.any(donor_mask):
            break

        donor_scores = partner * donor_mask
        donor_local = np.argmax(donor_scores)
        donor = idx[donor_local]

        # pick a receiver with minimal partner value (different from donor)
        sorted_loc = np.argsort(partner)
        receiver = None
        for r_loc in sorted_loc:
            r = idx[r_loc]
            if r != donor:
                receiver = r
                break
        if receiver is None:
            break

        partner_donor = partner[donor_local]
        receiver_local = np.where(idx == receiver)[0][0]
        partner_receiver = partner[receiver_local]

        denom = 2.0 * (partner_donor - partner_receiver)
        if denom <= 0.0:
            break

        delta_needed = (max_val - second_max) / denom
        max_delta = min(a[donor] * 0.7, delta_needed * 0.99)
        if max_delta <= 0.0:
            break

        a[donor] -= max_delta
        a[receiver] += max_delta

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] Exact DMT2 it={it:5d} score={best_score:.6f}")

        it += 1

    return best_a, best_score

def _pairwise_optimal_transfer(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    donor_frac: float = 0.015,
    receiver_frac: float = 0.015,
    min_improve: float = 1e-7,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = _score(a)

    pad_len = n - 1

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        nd = max(1, int(donor_frac * n))
        nr = max(1, int(receiver_frac * n))
        donor_idx = np.argsort(-a)[:nd]          # largest coefficients
        receiver_idx = np.argsort(a)[:nr]        # smallest coefficients

        base = np.concatenate(
            [np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)]
        )

        shifted = {}
        for i in np.unique(np.concatenate((donor_idx, receiver_idx))):
            shifted[i] = base[i : i + 2 * n - 1]     # view of length 2n‑1

        best_new_max = max_val
        best_pair = None
        best_delta = 0.0

        for i in donor_idx:
            shift_i = shifted[i]
            a_i = a[i]
            if a_i <= 0.0:
                continue
            for j in receiver_idx:
                if i == j:
                    continue
                shift_j = shifted[j]

                d = 2.0 * (shift_j - shift_i)   # effect of moving mass from i to j
                d_k = d[k]
                if d_k >= 0.0:
                    continue

                mask = d > d_k
                if np.any(mask):
                    denom = d[mask] - d_k
                    numer = max_val - conv[mask]
                    feasible = numer / denom
                    delta_max = min(feasible.min(), a_i)
                else:
                    delta_max = a_i

                if delta_max <= 1e-12:
                    continue

                new_max = max_val + d_k * delta_max
                if new_max < best_new_max - min_improve:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta_max

        if best_pair is None:
            break

        i, j = best_pair
        a[i] -= best_delta
        a[j] += best_delta

        # occasional re‑projection (tiny drift)
        if int(time.time() * 10) % 50 == 0:
            a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(
                f"[{time.time():.2f}] Pairwise transfer i={i} j={j} "
                f"Δ={best_delta:.3e} new_score={best_score:.6f}"
            )

    return a, best_score

def _global_random_mass_transfer(

        if cur_score < best_score:

            best_a = a.copy()

        conv2[conv2 == max_val] = -np.inf
        second_max = conv2.max()
        if max_val - second_max < 1e-12:
            break

        # partner weights w_i = a_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        w = np.zeros(n, dtype=np.float64)
        w[idx_range] = a[k_max - idx_range]

        # donor scoring
        donor_score = a * w
        nd = max(1, int(donor_frac * n))
        donor_idx = np.argsort(-donor_score)[:nd]
        donor_mask = np.zeros(n, dtype=bool)
        donor_mask[donor_idx] = True
        donor_mass = a[donor_idx].sum()
        if donor_mass <= 0.0:
            break

        a_donors = np.zeros_like(a)
        a_donors[donor_idx] = a[donor_idx]
        conv_donors = _conv_fft(a_donors)

        nr = max(1, int(receiver_frac * n))
        receiver_idx = np.argsort(w)[:nr]

        base = np.concatenate(
            [np.zeros(pad_len, dtype=np.float64), a, np.zeros(pad_len, dtype=np.float64)]
        )

        best_new_max = max_val
        best_r = None
        best_alpha = 0.0

        for r in receiver_idx:
            if donor_mask[r]:
                continue
            shift_r = base[r : r + 2 * n - 1]            # view of length 2n‑1
            diff = donor_mass * shift_r - conv_donors   # α·diff ≤ 0 component‑wise
            mask_pos = diff > 0
            if np.any(mask_pos):
                alpha_max = np.min(conv_donors[mask_pos] / (donor_mass * shift_r[mask_pos]))
                alpha_max = min(alpha_max, 1.0)
            else:
                alpha_max = 1.0
            if alpha_max < min_alpha:
                continue

            delta_peak = 2.0 * alpha_max * (
                -conv_donors[k_max] + donor_mass * shift_r[k_max]
            )
            new_max = max_val + delta_peak
            if new_max < best_new_max - 1e-12:
                best_new_max = new_max
                best_r = r
                best_alpha = alpha_max

        if best_r is None:
            break

        # apply move
        move = best_alpha * donor_mass
        a[donor_idx] *= (1.0 - best_alpha)
        a[best_r] += move
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Global peak transfer it={it} "
                f"α={best_alpha:.4g} new_score={best_score:.6f}"
            )
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponential sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _exponential_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    eta0: float = 0.015,
) -> tuple[np.ndarray, float]:
    """Iteratively apply a_i ← a_i·exp(-η·g_i) with g the sub‑gradient
    of the peak entry.  Projection onto the simplex keeps the sum fixed."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.time() < deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k-i} (zero outside the overlap region)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = _project_to_simplex(a, target_sum)

        cur_score = _score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time():.2f}] Exp‑subgrad it={it} score={best_score:.6f}"
            )
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (short stochastic exploration)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    target_sum: float,
    start_temp_frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """Random donor‑receiver moves accepted with a temperature schedule."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    start_time = time.time()
    total_time = deadline - start_time
    # temperature linearly decays from start_temp to ~0
    start_temp = max(start_temp_frac * best_score, 1e-6)

    while time.time() < deadline:
        # recompute conv only when needed
        conv = _conv_fft(a)
        k = int(np.argmax(conv))

        # donor selection: weighted by contribution a_i·a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner = np.zeros(n, dtype=np.float64)
        partner[idx_range] = a[k - idx_range]
        donor_weights = a * partner
        weight_sum = donor_weights.sum()
        if weight_sum == 0.0:
            break
        donor = rng.choice(n, p=donor_weights / weight_sum)

        # receiver: pick among indices with smallest partner values
        low_candidates = np.argsort(partner)[: max(1, n // 30)]
        receiver = rng.choice(low_candidates)

        if donor == receiver:
            continue

        max_delta = a[donor] * 0.5
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = _project_to_simplex(a_new, target_sum)
        new_score = _score(a_new)

        elapsed = time.time() - start_time
        frac = elapsed / total_time
        temp = start_temp * (1.0 - frac)  # linear cooling

        accept = False
        if new_score < cur_score:
            accept = True
        elif temp > 0.0:
            prob = np.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.2f}] SA new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    max self‑convolution (equivalently the score returned by
    evaluate_sequence).  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 15000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = float(np.sqrt(2.0 * n))

    # --------------------------------------------------------------
    # Helper to update best solution
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def _try_update(seq: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_score
        val = _score(seq)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(
                f"[{time.time():.2f}] {tag} candidate improved best={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase 0 – diverse parametric warm‑start (≈10 % of budget)
    # --------------------------------------------------------------
    param_deadline = start_time + 0.10 * budget_s
    rng = np.random.default_rng()

    # basic families
    p_vals = np.linspace(0.30, 0.85, 9)
    q_vals = np.linspace(0.0, 5.0, 6)
    sigma_vals = np.linspace(0.15 * n, 0.45 * n, 5)

    # power law
    for p in p_vals:
        if time.time() > param_deadline:
            break
        seq = _power_law_sequence(n, float(p), target_sum)
        _try_update(seq, "PowerLaw")

    # exp‑power law
    for p in p_vals:
        for q in q_vals:
            if time.time() > param_deadline:
                break
            seq = _exp_power_law_sequence(n, float(p), float(q), target_sum)
            _try_update(seq, "ExpPower")

    # Gaussian
    for sigma in sigma_vals:
        if time.time() > param_deadline:
            break
        seq = _gaussian_sequence(n, float(sigma), target_sum)
        _try_update(seq, "Gaussian")

    # Double exponential
    p_vals_de = np.linspace(0.005, 0.12, 6)
    for p in p_vals_de:
        if time.time() > param_deadline:
            break
        seq = _double_exp_sequence(n, float(p), target_sum)
        _try_update(seq, "DoubleExp")

    # Symmetric power law
    p1_vals = np.linspace(0.2, 1.0, 5)
    p2_vals = np.linspace(0.2, 1.0, 5)
    for p1 in p1_vals:
        for p2 in p2_vals:
            if time.time() > param_deadline:
                break
            seq = _sym_power_law_sequence(n, float(p1), float(p2), target_sum)
            _try_update(seq, "SymPower")

    # Sinusoidal
    seq = _sinusoidal_sequence(n, target_sum)
    _try_update(seq, "Sinusoid")

    # Cosine
    seq = _cosine_sequence(n, target_sum)
    _try_update(seq, "Cosine")

    # Quadratic
    seq = _quadratic_sequence(n, target_sum)
    _try_update(seq, "Quadratic")

    # If a previous construction exists, import and adapt it
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            resized = np.interp(new_x, old_x, prev)
        else:
            resized = prev.copy()
        resized = _project_to_simplex(resized, target_sum)
        _try_update(resized, "PrevResized")

    # Jitter around the current best
    for _ in range(12):
        if time.time() > param_deadline:
            break
        jitter = rng.normal(scale=target_sum * 0.006, size=n)
        seq_j = best_seq + jitter
        seq_j = np.maximum(seq_j, 0.0)
        seq_j = _project_to_simplex(seq_j, target_sum)
        _try_update(seq_j, "Jitter")

    print(
        f"[{time.time():.2f}] Initial best after parametric search = {best_score:.6f}"
    )

    # --------------------------------------------------------------
    # Phase A – Adam GD with restarts (≈30 % of budget)
    # --------------------------------------------------------------
    gd_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < gd_deadline - 5.0:
        jitter = rng.normal(scale=target_sum * 0.012, size=n)
        start_vec = best_seq + jitter
        start_vec = np.maximum(start_vec, 0.0)
        start_vec = _project_to_simplex(start_vec, target_sum)

        a_opt, val_opt = _gradient_descent_with_restarts(
            start_vec, target_sum, gd_deadline
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] GD improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase B – Exact deterministic mass‑transfer (core refinement)
    # --------------------------------------------------------------
    while time.time() < deadline - 12.0:
        a_ref, val_ref = _deterministic_mass_transfer_exact2(
            best_seq, deadline, target_sum
        )
        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] Exact DMT2 improved best = {best_score:.6f}")

        # short random move to escape plateaus
        a_rand, val_rand = _global_random_mass_transfer(best_seq, deadline, target_sum)
        if val_rand < best_score:
            best_score = val_rand
            best_seq = a_rand.copy()
            print(f"[{time.time():.2f}] Global random improved best = {best_score:.6f}")

        if time.time() > deadline - 10.0:
            break

    # --------------------------------------------------------------
    # Phase C – Global peak‑mass transfer
    # --------------------------------------------------------------
    if time.time() < deadline - 6.0:
        a_peak, val_peak = _global_peak_mass_transfer(
            best_seq,
            deadline,
            target_sum,
            donor_frac=0.03,
            receiver_frac=0.03,
        )
        if val_peak < best_score:
            best_score = val_peak
            best_seq = a_peak.copy()
            print(
                f"[{time.time():.2f}] Global peak transfer improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase D – Pairwise optimal mass transfer
    # --------------------------------------------------------------
    if time.time() < deadline - 4.0:
        a_pair, val_pair = _pairwise_optimal_transfer(
            best_seq,
            deadline,
            target_sum,
            donor_frac=0.018,
            receiver_frac=0.018,
        )
        if val_pair < best_score:
            best_score = val_pair
            best_seq = a_pair.copy()
            print(
                f"[{time.time():.2f}] Pairwise transfer improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase E – Exponential sub‑gradient descent on the true max norm
    # --------------------------------------------------------------
    if time.time() < deadline - 3.0:
        sub_deadline = min(deadline - 0.5, time.time() + 5.0 + 0.05 * (deadline - time.time()))
        a_sub, val_sub = _exponential_subgradient_descent(
            best_seq,
            sub_deadline,
            target_sum,
            eta0=0.02,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = a_sub.copy()
            print(
                f"[{time.time():.2f}] Exp‑subgrad improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase F – Simulated annealing (short stochastic refinement)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sa_deadline = deadline - 1.0
        a_sa, val_sa = _simulated_annealing(best_seq, sa_deadline, target_sum)
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(
                f"[{time.time():.2f}] SA improved best = {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase G – Final polishing with very high β
    # --------------------------------------------------------------
    if time.time() + 1.0 < deadline:
        a_fin, val_fin = _adam_gradient_descent(
            best_seq,
            deadline,
            target_sum,
            max_beta=1e6,
            step_factor=0.5,
            gamma=0.995,
        )
        if val_fin < best_score:
            best_score = val_fin
            best_seq = a_fin.copy()
            print(f"[{time.time():.2f}] Final polish best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑processing: trim trailing near‑zero entries and return
    # --------------------------------------------------------------
    arr = np.array(best_seq, dtype=float)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]
    final_seq = arr.tolist()

    # optional final evaluation (harmless if evaluate_sequence is not defined)
    try:
        val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {val:.6f}")
    except Exception:
        pass

    return final_seq
