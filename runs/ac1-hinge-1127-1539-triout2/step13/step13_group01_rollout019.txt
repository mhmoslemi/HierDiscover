"""
propose_candidate
-----------------

Search for a low‑score non‑negative step sequence.

The routine combines three ideas:

1. **Parameterized families** – a short grid search over simple analytic
   shapes (exponential decay, power‑law decay, reversed power‑law, and
   Beta‑type (1‑x)^p).  Each candidate is normalised to the usual
   √(2 n) sum, scored with a fast FFT‑based convolution and the best
   vector is kept.

2. **Aggressive simulated‑annealing refinement** – starting from the
   best shape, a massive number of tiny mass‑transfer moves are applied.
   The convolution is updated in O(n) time per move, keeping the algorithm
   fast enough to run for most of the allocated time budget.

3. **Fallback & warm‑start** – the previous run’s best vector is kept in the
   global variable ``height_sequence_1`` and used as an additional start.
   If the search fails to find a better result a simple decreasing linear
   sequence is used as a safety net.

The function respects the supplied time budget (default 1000 s) and
returns a plain list of floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (same as the reference implementation)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) / (sum(a)²).  No normalisation is applied."""
    n = a.shape[0]
    max_b = _conv_fft(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0 , Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer (copied from the previous solution)
# ----------------------------------------------------------------------


def anneal_mass_transfer(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Simulated‑annealing that moves tiny amounts of mass from ‘bad’
    indices (large contribution to the current peak) to ‘good’ indices
    (small height).  The convolution is updated in O(n) for each move.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _conv_fft(a)                     # exact convolution a*a
    cur_val = _score(a)
    best_val = cur_val
    best_a = a.copy()

    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = cur_val * 0.9                    # initial temperature

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        temperature = max(0.0, T0 * (1.0 - np.sqrt(frac)))

        # current peak and its contributing indices
        k_max = int(np.argmax(b))
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        contrib = a[idx] * a[k_max - idx]
        if contrib.sum() > 0.0:
            donor = rng.choice(idx, p=contrib / contrib.sum())
        else:
            donor = rng.integers(lo, hi + 1)

        # receiver – favour small heights
        inv = 1.0 / (a + 1e-12)
        inv = inv / inv.sum()
        receiver = rng.choice(n, p=inv)

        if donor == receiver:
            continue

        max_delta = min(a[donor] * 0.08, target_sum * 0.01)
        if max_delta <= 0.0:
            continue
        scale = temperature / T0 if T0 > 0 else 0.0
        delta = rng.random() * max_delta * scale
        if delta <= 0.0:
            continue

        # ----- exact O(n) update of the convolution -----------------
        b_new = b.copy()
        # linear part: 2·a·Δ (donor removal, receiver addition)
        b_new[donor:donor + n] -= 2.0 * delta * a
        b_new[receiver:receiver + n] += 2.0 * delta * a
        # quadratic part (Δ²)
        b_new[2 * donor] += delta * delta
        b_new[2 * receiver] += delta * delta
        if donor != receiver:
            b_new[donor + receiver] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max() / (target_sum * target_sum))

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_val - cur_val) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a[donor] -= delta
            a[receiver] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Anneal improved best={best_val:.6f}")

        # occasional full recomputation to avoid drift
        if it % 4000 == 0:
            a = _project_to_simplex(a, target_sum)
            b = _conv_fft(a)
            cur_val = _score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}] Full recompute improved best={best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low benchmark score.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    start = time.time()
    deadline = start + budget_s - 0.5        # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n → finer granularity but slower)
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 18000
    elif budget_s >= 500:
        n = 12000
    else:
        n = 8000

    target_sum = np.sqrt(2.0 * n)            # the usual normalisation

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm start from the previous run (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        warm = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if warm.shape[0] != n:
            warm = _resize_sequence(warm, n)
        warm = warm * (target_sum / warm.sum())
        s = _score(warm)
        best_seq = warm.copy()
        best_score = s
        print(f"[{time.time():.2f}] Warm‑start score = {s:.6f}")

    # --------------------------------------------------------------
    # 1) Grid search over a few analytically‑motivated families
    # --------------------------------------------------------------
    rng = np.random.default_rng()

    # Exponential decay a_i = r**i
    eps_vals = np.logspace(-7, -2, num=200)            # epsilon = 1 - r
    for eps in eps_vals:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        a_raw = np.power(r, np.arange(n, dtype=np.float64))
        a = a_raw * (target_sum / a_raw.sum())
        s = _score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Exp eps={eps:.2e}  score={s:.6f}")

    # Power‑law decreasing a_i = (i+1)^(-p)
    p_vals = np.linspace(0.0, 0.8, num=200)           # p small – near‑uniform
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        if time.time() >= deadline:
            break
        a_raw = idx ** (-p)
        a = a_raw * (target_sum / a_raw.sum())
        s = _score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Power‑law p={p:.4f}  score={s:.6f}")

    # Beta‑type decreasing a_i = (1 - i/(n‑1))**p
    p_vals2 = np.linspace(0.0, 5.0, num=200)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    one_minus_x = 1.0 - x
    for p in p_vals2:
        if time.time() >= deadline:
            break
        a_raw = one_minus_x ** p
        a = a_raw * (target_sum / a_raw.sum())
        s = _score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Beta p={p:.4f}  score={s:.6f}")

    # A few random Dirichlet starts (to add diversity)
    for _ in range(5):
        if time.time() >= deadline:
            break
        a_raw = rng.random(n)
        a = a_raw * (target_sum / a_raw.sum())
        s = _score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Random start score={s:.6f}")

    # --------------------------------------------------------------
    # 2) Aggressive annealing refinement from the best candidate
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_ann, s_ann = anneal_mass_transfer(best_seq, target_sum, deadline)
        if s_ann < best_score:
            best_score = s_ann
            best_seq = a_ann.copy()

    # --------------------------------------------------------------
    if best_seq is None:               # should never happen, but safety net

        best_seq = best_seq * (target_sum / best_seq.sum())

    arr = np.array(best_seq, dtype=np.float64)
    nz = np.where(arr > 1e-12)[0]
    if nz.size:
        arr = arr[: nz[-1] + 1]

    final_seq = arr.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"[{time.time():.2f}] Final evaluated score = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print(f"[{time.time():.2f}] Evaluation error: {exc}")

    # store for the next run (warm start)
    globals()["height_sequence_1"] = final_seq
    return final_seq
