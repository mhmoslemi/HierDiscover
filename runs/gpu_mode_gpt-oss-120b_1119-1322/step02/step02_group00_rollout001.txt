"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1. **Layer‑norm** the input tensor `x` (float32) – this is cheap and done with
   PyTorch for numerical stability.
2. Cast to **fp16** and apply the 5 weight‑only linear projections
   (left, right, left_gate, right_gate, out_gate).  The projections are
   performed with `torch.nn.functional.linear`, which on CUDA uses cuBLAS
   Tensor‑Core GEMM and is already highly optimized.
3. If a mask is supplied it is broadcast over the hidden dimension and
   multiplied into `left` and `right` **before** gating – exactly as the
   reference implementation does.
4. Apply the three sigmoid gates element‑wise.
5. **Core N³ update** – compute for every hidden channel `h` and batch `b`

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   This is a batched matrix multiplication `C = A @ Bᵀ` where each
   `(b,h)` pair defines a matrix `A` (shape N×N) and a matrix `B` (shape N×N).
   The heavy arithmetic is off‑loaded to a custom Triton kernel
   `batched_gemm_out_kernel`.  The kernel works on fp16 inputs,
   accumulates in fp32, and uses Tensor‑Core `tl.dot` with 64‑×‑64‑×‑64 tiles.
6. Reshape the kernel output back to `[B, N, N, H]` (fp16), apply the hidden‑
   dimension LayerNorm, multiply by `out_gate`, and finally project back to
   the original dimension `D` with a bias‑free linear layer.
7. Cast the result to **float32** and return.

Only the cubic‑time matrix multiplication (step 5) is written in Triton;
all other inexpensive operations stay in PyTorch, which keeps the
implementation simple while still achieving a large speed‑up over the
pure‑PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,          # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,          # B: batch, col (k), row (n) strides
    stride_cb, stride_cm, stride_cn,          # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many independent (batch × hidden) matrices.

    Each matrix A has shape (M, K), each B has shape (N, K) and C = (M, N).
    All tensors are fp16; accumulation is fp32.
    """
    pid_batch = tl.program_id(2)               # which (batch * hidden) matrix
    pid_m = tl.program_id(0)                   # block row of C
    pid_n = tl.program_id(1)                   # block column of C

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Bounds checking
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (batch * hidden) matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # Pointers to the beginning of the tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        # Load tiles, applying masks for edges
        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate the matrix product (Tensor‑Core dot)
        acc += tl.dot(a, b)

    # Write the output tile
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)

            - mask:        torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights:     dict mapping weight names to torch.Tensor (float32)
            - config:      dict containing keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D] (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for everything else
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to fp16 (bias‑free linear layers)
    # ------------------------------------------------------------------
    w_left     = weights["left_proj.weight"].to(torch.float16)
    w_right    = weights["right_proj.weight"].to(torch.float16)
    w_lgate    = weights["left_gate.weight"].to(torch.float16)
    w_rgate    = weights["right_gate.weight"].to(torch.float16)
    w_outgate  = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out   = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left)    # [B, N, N, H]
    right = F.linear(x_h, w_right)

    # ------------------------------------------------------------------
    # 5) Compute sigmoid gates – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask has the proper dtype and shape
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # 8) Prepare tensors for the batched GEMM (A @ Bᵀ)
    #    Layout after permutation: [B, H, N, N] → reshape → [B*H, N, N]
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_t.view(-1, N, N)   # [B*H, N, N] (fp16)
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N] (fp16)

    total_matrices = left_flat.shape[0]   # B * H

    # Output buffer for the GEMM (fp16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 9) Triton kernel launch – batched A @ Bᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # blocks along M (rows)
        triton.cdiv(N, BLOCK_N),          # blocks along N (cols)
        total_matrices,                   # one program per (batch * hidden) matrix
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0),   # stride_ab  (batch stride)
        left_flat.stride(1),   # stride_am  (row stride)
        left_flat.stride(2),   # stride_ak  (col stride)
        right_flat.stride(0),  # stride_bb  (batch stride)
        right_flat.stride(2),  # stride_bk  (col stride in B, i.e. k‑dimension)
        right_flat.stride(1),  # stride_bn  (row stride in B, i.e. n‑dimension)
        out_flat.stride(0),    # stride_cb  (batch stride)
        out_flat.stride(1),    # stride_cm  (row stride)
        out_flat.stride(2),    # stride_cn  (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10) Reshape kernel output back to [B, N, N, H] (fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 11) Hidden‑dim LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original dimension D (fp16)
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 13) Cast to float32 for downstream consumption
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
