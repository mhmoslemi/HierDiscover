def _gate_and_mask_kernel(
    combined_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_c_batch, stride_c_i, stride_c_k, stride_c_c,   # combined strides
    stride_m_batch, stride_m_i, stride_m_k,
    stride_o_batch, stride_o_i, stride_o_k, stride_o_c,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute: out = (combined[..., :H] * mask[..., None]) * sigmoid(combined[..., H:]).
    combined shape: (B, N, N, 2*H)
    mask shape   : (B, N, N)   (broadcast over H)
    out shape    : (B, N, N, H)
    """
    pid_i = tl.program_id(0)   # tile over first N dimension (i)
    pid_k = tl.program_id(1)   # tile over second N dimension (k)
    pid_f = tl.program_id(2)   # flattened (batch, H-tile)

    # number of H‑tiles per batch
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    batch_id = pid_f // num_h_tiles
    h_tile_start = (pid_f % num_h_tiles) * BLOCK_H

    # offsets inside the tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
    offs_h = h_tile_start + tl.arange(0, BLOCK_H)

    # bounds
    mask_i = offs_i < N
    mask_k = offs_k < N
    mask_h = offs_h < H
    mask_ijkh = (mask_i[:, None, None] &
                 mask_k[None, :, None] &
                 mask_h[None, None, :])

    # -----------------------------------------------------------------
    # Load mask (broadcast over H)
    # NOTE: mask_ptr may point to a tensor of ones when no real mask is provided.
    mask_vals = tl.load(
        mask_ptr + batch_id * stride_m_batch
                + offs_i[:, None] * stride_m_i
                + offs_k[None, :] * stride_m_k,
        mask=mask_i[:, None] & mask_k[None, :],
        other=1.0,
    )  # (BLOCK_I, BLOCK_K)
    mask_vals = mask_vals[:, :, None]   # (BLOCK_I, BLOCK_K, 1)

    # -----------------------------------------------------------------
    # Base address for this (i,k) slice of the combined tensor.
    base_ptr = combined_ptr + batch_id * stride_c_batch \
               + offs_i[:, None, None] * stride_c_i \
               + offs_k[None, :, None] * stride_c_k

    # Projection part (first H channels)
    proj = tl.load(
        base_ptr + offs_h[None, None, :] * stride_c_c,
        mask=mask_ijkh,
        other=0.0,
    )

    # Gate logits (second H channels, offset by H)
    gate = tl.load(
        base_ptr + (H + offs_h[None, None, :]) * stride_c_c,
        mask=mask_ijkh,
        other=0.0,
    )

    # -----------------------------------------------------------------
    # Apply mask to projection
    proj = proj * mask_vals

    # Sigmoid gate
    gate = 1.0 / (1.0 + tl.exp(-gate))

    # Final gated output
    out = proj * gate

    # Store result
    out_ptr_base = out_ptr + batch_id * stride_o_batch \
                   + offs_i[:, None, None] * stride_o_i \
                   + offs_k[None, :, None] * stride_o_k \
                   + offs_h[None, None, :] * stride_o_c

    tl.store(out_ptr_base, out, mask=mask_ijkh)

    H = hidden_dim

    right = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    BLOCK_I = 64   # tile over first N dimension
    BLOCK_K = 64   # tile over second N dimension

    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) FP32
