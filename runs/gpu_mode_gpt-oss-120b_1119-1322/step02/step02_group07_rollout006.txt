```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Layernorm 0
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # fp32

    # Convert to fp16 for subsequent ops
    x_fp16 = x.half().contiguous()

    B, N, _, _ = x_fp16.shape
    M = B * N * N

    # Prepare combined weight for five linear transforms
    # We use torch.cat along dim=0
    # Ensure all weights are FP16
    w_left = weights["left_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right = weights["right_proj.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()

    combined_weight = torch.cat([w_left, w_left_gate, w_right, w_right_gate, w_out_gate], dim=0)

    # Flatten input for linear
    x_flat = x_fp16.view(-1, dim)  # shape (M, dim)

    # Single fused linear
    # Note: torch.nn.functional.linear expects weight shape (out_features, in_features)
    linear_out = torch.nn.functional.linear(x_flat, combined_weight)  # shape (M, 5*hidden), fp16

    # Split outputs
    out_split = linear_out.view(M, 5, hidden)  # (M, five, hidden)
    # Index: 0 left_proj, 1 left_gate_pre, 2 right_proj, 3 right_gate_pre, 4 out_gate_pre
    left_proj_flat = out_split[:, 0, :]
    left_gate_pre_flat = out_split[:, 1, :]
    right_proj_flat = out_split[:, 2, :]
    right_gate_pre_flat = out_split[:, 3, :]
    out_gate_pre_flat = out_split[:, 4, :]

    # Apply sigmoid to gates (still half)
    left_gate_flat = torch.sigmoid(left_gate_pre_flat)
    right_gate_flat = torch.sigmoid(right_gate_pre_flat)
    out_gate = torch.sigmoid(out_gate_pre_flat).to(torch.float32)  # keep fp32

    # Reshape to original 4D shape
    left = left_proj_flat.view(B, N, N, hidden)
    right = right_proj_flat.view(B, N, N, hidden)
    left_gate = left_gate_flat.view(B, N, N, hidden)
    right_gate = right_gate_flat.view(B, N, N, hidden)

    # Optional mask and gating fusion via Triton
    if not nomask:
        mask_fp16 = mask.to(dtype=torch.float16).contiguous()
        left, right = fuse_mask_gate(
            left, right, mask_fp16, left_gate, right_gate, hidden
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # Batched GEMM: left_out @ right_out.T across hidden dim
    # Reshape to (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    # Perform batch matmul: [B*hidden, N, N] * [B*hidden, N, N]^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16

    # Reshape back to (B, N, N, hidden) and convert to fp32
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # Second LayerNorm over hidden dimension (fp32)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate (fp32) and final projection
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # fp32

    return out
```

@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    total_elements: tl.int64,
    H: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (one per element of left but broadcast across H)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # Multiply
    l = l * lg * m
    r = r * rg * m

    # Store
    tl.store(out_left_ptr + offs, l, mask=mask)

"""
Outgoing TriMul (TriMul) – highly‑optimized forward pass.

Algorithm
---------
1.  Layer‑norm on the input (FP32).
2.  Cast to FP16 and flatten to (M, dim) where M = B·N·N.
3.  Fuse *five* linear projections (left, left_gate, right, right_gate, out_gate)
    into a single GEMM by concatenating the weight matrices.
4.  Split the fused result, apply sigmoid to the three gate tensors.
5.  Reshape the three tensor streams back to [B, N, N, hidden] format.
6.  Fuse optional binary mask, left‑gate and right‑gate with a small Triton kernel:
        left_out  = left  * left_gate  * mask
        right_out = right * right_gate * mask
7.  Batched GEMM over the sequence dimension:
        out[b,i,j,d] = Σ_k left_out[b,i,k,d] * right_out[b,j,k,d]
    (implemented as a strided‑batched bmm on FP16 tensors → Tensor‑Cores).
8.  Cast to FP32, second Layer‑Norm over the hidden dimension.
9.  Apply the output‑gate (FP32) and final linear projection back to `dim`.

The kernel in step‑6 is required by the problem statement; all heavy
matrix‑multiplications are delegated to cuBLAS for maximum throughput on an
H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel – mask + per‑channel gating fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,          # fp16 * left  (contiguous)
    right_ptr,         # fp16 * right (contiguous)
    mask_ptr,          # fp16 * mask  (B,N,N) contiguous
    left_gate_ptr,     # fp16 * left_gate (contiguous)
    right_gate_ptr,    # fp16 * right_gate (contiguous)
    out_left_ptr,      # fp16 * output left
    out_right_ptr,     # fp16 * output right
    total_elems: tl.int64,
    H: tl.constexpr,   # hidden dimension (known at compile time)
    BLOCK: tl.constexpr,
):
    """Fuse mask, left‑gate and right‑gate.
    Layout of the 4‑D tensors is (b,i,j,h) → offset = ((b*N + i)*N + j)*H + h.
    The mask has layout (b,i,j) → offset = ((b*N + i)*N + j).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load everything (zero for out‑of‑range threads)
    l  = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # mask index (broadcast over H)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # fuse three element‑wise multiplications
    l = l * lg * m
    r = r * rg * m

    # store results
    tl.store(out_left_ptr  + offs, l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


def _apply_mask_and_gate(left, right, mask, left_gate, right_gate, hidden):
    """Launch the Triton kernel for mask+gate fusion."""
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    # Allocate output buffers
    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Ensure all inputs are contiguous for the kernel
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate= right_gate.contiguous()
    mask_fp16 = mask.to(dtype=torch.float16).contiguous()

    _mask_gate_kernel[grid](
        left, right,
        mask_fp16,
        left_gate, right_gate,
        left_out, right_out,
        total,
        H=hidden,
        BLOCK=BLOCK,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [B, N, N, dim],  float32
            mask         : torch.Tensor, shape [B, N, N],      bool/float32
            weights      : dict of torch.Tensor model parameters
            config       : dict with keys
                           - "dim"        (int)
                           - "hidden_dim" (int)
                           - "nomask"     (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim], float32
    """
    # --------------------------------------------------------------
    # unpack input
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim   = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0. Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32

    # --------------------------------------------------------------
    # 1. Cast to FP16 and flatten for a *single* fused linear layer
    # --------------------------------------------------------------
    x_fp16 = x.half().contiguous()
    B, N, _, _ = x_fp16.shape
    M = B * N * N                     # number of (b,i,j) positions
    x_flat = x_fp16.view(-1, dim)    # (M, dim)

    # --------------------------------------------------------------
    # 2. Fuse the five linear maps (left, left_gate, right, right_gate,
    #    out_gate) into one GEMM.
    # --------------------------------------------------------------
    # Convert all weight tensors to FP16 once
    w_left       = weights["left_proj.weight"].half()
    w_left_gate  = weights["left_gate.weight"].half()
    w_right      = weights["right_proj.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate   = weights["out_gate.weight"].half()

    # Concatenate -> shape (5*hidden, dim)
    fused_weight = torch.cat(
        [w_left, w_left_gate, w_right, w_right_gate, w_out_gate], dim=0
    )  # (5*hidden, dim)

    # Single fused linear
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (M, 5*hidden) FP16

    # --------------------------------------------------------------
    # 3. Split fused output and apply sigmoid to the three gate tensors
    # --------------------------------------------------------------
    fused_out = fused_out.view(M, 5, hidden)  # (M, five, hidden)
    left_proj_flat      = fused_out[:, 0, :]  # (M, hidden)
    left_gate_pre_flat  = fused_out[:, 1, :]
    right_proj_flat     = fused_out[:, 2, :]
    right_gate_pre_flat = fused_out[:, 3, :]
    out_gate_pre_flat   = fused_out[:, 4, :]

    left_gate_flat  = torch.sigmoid(left_gate_pre_flat)   # FP16
    right_gate_flat = torch.sigmoid(right_gate_pre_flat)  # FP16
    out_gate        = torch.sigmoid(out_gate_pre_flat).to(torch.float32)  # keep FP32 for later

    # --------------------------------------------------------------
    # 4. Reshape back to 4‑D tensors
    # --------------------------------------------------------------
    left       = left_proj_flat.view(B, N, N, hidden)
    right      = right_proj_flat.view(B, N, N, hidden)
    left_gate  = left_gate_flat.view(B, N, N, hidden)
    right_gate = right_gate_flat.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 5. Optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if not nomask:
        left, right = _apply_mask_and_gate(
            left, right, mask, left_gate, right_gate, hidden
        )
    else:
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 6. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a strided‑batched bmm on FP16 → Tensor‑Cores.
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) – we need hidden as the batch dimension
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # (B*hidden, N, N) @ (B*hidden, N, N)^T
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # --------------------------------------------------------------
    # 7. Restore original layout and promote to FP32
    # --------------------------------------------------------------
    out = (
        out_mat_fp16.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 8. Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # FP32

    # --------------------------------------------------------------
    # 9. Output gate (FP32) and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # (dim, hidden) weight – FP32

    return out
