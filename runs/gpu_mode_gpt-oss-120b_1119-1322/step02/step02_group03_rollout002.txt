"""
TriMul (outgoing) – Triton‑accelerated forward pass
--------------------------------------------------
The implementation follows the “outgoing” TriMul described in AlphaFold‑3.
The heavy N³ pairwise update is performed by a custom Triton kernel that
computes, for every batch element and hidden channel *h*,

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

i.e. a batched GEMM `C = A @ Bᵀ` with `A` = left (M=N, K=N) and `B` = right (N,N).
All other operations (LayerNorm, linear projections, sigmoid‑gates,
optional mask, final projection) are carried out with PyTorch in FP16/F32
where appropriate.  The kernel works on FP16 inputs, accumulates in FP32,
and uses Tensor‑Core `tl.dot` for maximum throughput on H100.

The kernel is deliberately simple (no autotuning) but uses a fairly
large tile (128×128) that fits the 64 KB shared‑memory budget of a single
program on H100, resulting in a very low launch overhead for the N³
computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # dimensions of the GEMM (all equal to seq_len)
    stride_ab, stride_am, stride_ak,   # A: batch‑stride, row‑stride, col‑stride
    stride_bb, stride_bk, stride_bn,   # B: batch‑stride, col‑stride (k), row‑stride (n)
    stride_cb, stride_cm, stride_cn,   # C: batch‑stride, row‑stride, col‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM used in TriMul:
        C[b] = A[b] @ B[b]ᵀ
    where A and B have shape (M, K) and (N, K) respectively.
    The kernel processes FP16 data and accumulates in FP32.
    """
    pid_batch = tl.program_id(2)            # which (batch * hidden) matrix
    pid_m = tl.program_id(0)                # block‑row of C
    pid_n = tl.program_id(1)                # block‑col of C

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # Pointers to the current tile in A (M×K) and B (N×K)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K

        a = tl.load(
            a_tile_ptr + cur_k * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_k) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_k * stride_bk,
            mask=((offs_k + cur_k) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation (Tensor‑Core)
        acc += tl.dot(a, b)

    # Write the result
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
        - weights      : dict of model parameters (torch.Tensor)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates
    #   left_proj   = Linear(dim -> hidden_dim)
    #   left_gate   = Linear(dim -> hidden_dim) + sigmoid
    #   right_proj  = Linear(dim -> hidden_dim)
    #   right_gate  = Linear(dim -> hidden_dim) + sigmoid
    #   The two linear layers per side are concatenated so that `x`
    #   is read only once for each side.
    # ------------------------------------------------------------------
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*hidden_dim, dim)

    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*hidden_dim, dim)

    # Linear without bias: result shape (B, N, N, 2*hidden_dim)
    left_combined = F.linear(x_fp16, left_combined_w)   # FP16
    right_combined = F.linear(x_fp16, right_combined_w)

    # Split projection and gate parts
    left_proj = left_combined[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = left_combined[..., hidden_dim:].sigmoid()           # (B,N,N,H)

    right_proj = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:].sigmoid()

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)   # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ update – batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # B = batch size, H = hidden_dim
    # Rearrange to (B*H, N, N) so that each (batch,hidden) pair is a separate matrix
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_t.view(-1, N, N)                # (B*H, N, N)  FP16
    right_flat = right_t.view(-1, N, N)              # (B*H, N, N)  FP16
    total = left_flat.shape[0]                       # B * H

    # Output in FP32 (accumulation precision)
    out_flat = torch.empty((total, N, N), dtype=torch.float32, device=device)

    # Tile size – 128×128 tiles fit in 64 KB shared memory on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        total,                     # one program per (batch * hidden)
    )

    batched_gemm_out_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0),   # stride_ab (batch stride)
        left_flat.stride(1),   # stride_am (row stride)
        left_flat.stride(2),   # stride_ak (col stride)
        right_flat.stride(0),  # stride_bb (batch stride)
        right_flat.stride(2),  # stride_bk (col‑stride for Bᵀ, i.e. K dim)
        right_flat.stride(1),  # stride_bn (row‑stride for Bᵀ, i.e. N dim)
        out_flat.stride(0),    # stride_cb (batch stride)
        out_flat.stride(1),    # stride_cm (row stride)
        out_flat.stride(2),    # stride_cn (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original layout: (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # FP32

    # out_gate is computed from the *normalized* input `x`
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # (B,N,N,H) FP32
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) FP32

    return out
