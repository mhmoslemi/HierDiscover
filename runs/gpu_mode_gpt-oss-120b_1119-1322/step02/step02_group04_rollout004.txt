"""
custom_kernel
==============

Implementation of the *outgoing* Triangle‑Multiplicative Update (TriMul) used in
AlphaFold‑3. The heavy N³ “triangle multiplication” is performed entirely in a
custom Triton kernel, while the surrounding linear projections, layer‑norms and
sigmoid gates are executed with PyTorch (float16 for the compute‑intensive part).
All arithmetic is carried out in half‑precision where possible and the final
output is returned as float32.

Key ideas
---------
1. Fuse each linear projection with its gate by concatenating the weight
   matrices, then split and apply the sigmoid.
2. Apply the input mask (if present) before the gates, exactly as in the
   reference implementation.
3. Perform the core operation
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM using a Triton kernel.  The kernel processes the (B·H)
   independent N×N matrix multiplications in a single launch with 3‑D grid
   (i‑tile, j‑tile, batch‑head).  Accumulation is done in float32 for numerical
   stability.
4. Layer‑norm over the hidden dimension, element‑wise multiplication with the
   output gate (also via a tiny Triton kernel), and the final projection back
   to the original channel dimension.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # float32* (out_norm flattened)
    b_ptr,                # float32* (out_gate flattened)
    out_ptr,              # float32* (result)
    size,                 # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """Simple element‑wise multiplication."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # half*   (B*H, N, N) – left side
    right_ptr,         # half*   (B*H, N, N) – right side
    out_ptr,           # float*  (B*H, N, N) – result
    N,                 # int32 – sequence length
    BATCH,             # int32 – B*H (batch * hidden)
    # strides for left
    stride_left_batch, stride_left_i, stride_left_k,
    # strides for right
    stride_right_batch, stride_right_i, stride_right_k,
    # strides for out
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched triangle multiplication:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]   (for each hidden head)
    where b indexes the fused batch/hidden dimension (B*H).
    """
    pid_i = tl.program_id(0)          # tile along the i‑axis
    pid_j = tl.program_id(1)          # tile along the j‑axis
    pid_b = tl.program_id(2)          # fused batch/hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i_off = i_start + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    j_off = j_start + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    mask_i = i_off < N
    mask_j = j_off < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, head) entry
    # ------------------------------------------------------------------
    left_batch_ptr  = left_ptr  + pid_b * stride_left_batch
    right_batch_ptr = right_ptr + pid_b * stride_right_batch
    out_batch_ptr   = out_ptr   + pid_b * stride_out_batch

    # ------------------------------------------------------------------
    # Pointers to the start of the i‑rows and j‑columns (k‑dimension offset = 0)
    # ------------------------------------------------------------------
    left_row_base  = left_batch_ptr  + i_off[:, None] * stride_left_i   # (BLOCK_M, 1)
    right_col_base = right_batch_ptr + j_off[None, :] * stride_right_i  # (1, BLOCK_N)

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)        # (BLOCK_K,)
        mask_k = k_off < N

        # left tile: (i, k)
        left_tile = tl.load(
            left_row_base + k_off[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            dtype=tl.float16,
            other=0.0,
        )  # shape (BLOCK_M, BLOCK_K)

        # right tile: (k, j) – note the transpose (right column base + k)
        right_tile = tl.load(
            right_col_base + k_off[:, None] * stride_right_k,
            mask=mask_k[:, None] & mask_j[None, :],
            dtype=tl.float16,
            other=0.0,
        )  # shape (BLOCK_K, BLOCK_N)

        # Accumulate with FP32 precision
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptr_tile = out_batch_ptr + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptr_tile, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input Layer‑Norm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to half‑precision for the heavy compute
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse projection + gate for the left side
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)  # (H, C)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, C)

    left_comb = torch.nn.functional.linear(x_h, left_comb_w)    # [B,N,N,2H]
    left_proj = left_comb[..., :hidden_dim]                     # [B,N,N,H]
    left_gate_pre = left_comb[..., hidden_dim:]                # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Fuse projection + gate for the right side
    # ------------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    right_comb = torch.nn.functional.linear(x_h, right_comb_w)
    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 5. Output gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x_h, out_gate_w)    # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)                       # half

    # ------------------------------------------------------------------
    # 6. Optional masking (applied before the gates, as in the reference)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N) → (B,N,N,1) in half precision
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 7. Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)   # half
    right_gate = torch.sigmoid(right_gate_pre) # half

    left = left_proj * left_gate                # [B,N,N,H], half
    right = right_proj * right_gate             # [B,N,N,H], half

    # ------------------------------------------------------------------
    # 8. Triangle multiplication via custom Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch size, seq_len, _, hidden_dim

    # Re‑order to (B, H, N, N) -> (B*H, N, N) contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(-1, N, N)               # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)

    batch_h = left_flat.shape[0]                       # B * hidden_dim

    # Allocate output buffer in FP32 (required for later LayerNorm)
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # Strides (all tensors are contiguous)
    s_l_b, s_l_i, s_l_k = left_flat.stride()
    s_r_b, s_r_i, s_r_k = right_flat.stride()
    s_o_b, s_o_i, s_o_j = out_flat.stride()

    # Tile sizes – tuned for H100 (use 64‑64 tiles with a 32‑wide reduction)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        batch_h,                   # fused batch‑head dimension
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        batch_h,
        s_l_b,
        s_l_i,
        s_l_k,
        s_r_b,
        s_r_i,
        s_r_k,
        s_o_b,
        s_o_i,
        s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,H] float32

    # ------------------------------------------------------------------
    # 9. Hidden‑dim LayerNorm (float32)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,H], float32

    # ------------------------------------------------------------------
    # 10. Element‑wise gating (out_norm * out_gate) via Triton
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.contiguous().view(-1)                 # float32
    out_gate_flat = out_gate.to(torch.float32).contiguous().view(-1)  # float32
    gated_flat = torch.empty_like(out_norm_flat)

    total_elements = out_norm_flat.numel()
    BLOCK_SIZE = 65536  # 64 K elements per program
    grid_elem = (triton.cdiv(total_elements, BLOCK_SIZE),)

    elementwise_mul_kernel[grid_elem](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(B, N, N, hidden_dim)  # [B,N,N,H], float32

    # ------------------------------------------------------------------
    # 11. Final projection back to the original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    gated_h = gated.to(torch.float16)                     # half
    out_h = torch.nn.functional.linear(gated_h, to_out_w)  # [B,N,N,dim], half

    # Return as float32 (as required by the API)
    return out_h.to(torch.float32)
