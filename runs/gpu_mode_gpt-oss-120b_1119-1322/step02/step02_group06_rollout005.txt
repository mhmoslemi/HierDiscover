import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for boundaries
    m_mask = offs_m < M
    n_mask = offs_n < N

    # create accumulators in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # K dimension offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_id * stride_A_batch + (offs_m[:, None] * stride_A_m) + (offs_k[None, :] * stride_A_k)
        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # load B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + batch_id * stride_B_batch + (offs_k[:, None] * stride_B_k) + (offs_n[None, :] * stride_B_n)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # write back C
    c_ptrs = C_ptr + batch_id * stride_C_batch + (offs_m[:, None] * stride_C_m) + (offs_n[None, :] * stride_C_n)
    # cast back to fp16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # cast weights to half (or appropriate dtype). Keep bias weight same dtype half.
    def to_half(t): return t.to(torch.float16) if t is not None else None

    norm_weight = to_half(weights["norm.weight"])
    norm_bias = to_half(weights["norm.bias"])
    left_proj_weight = to_half(weights["left_proj.weight"])
    right_proj_weight = to_half(weights["right_proj.weight"])
    left_gate_weight = to_half(weights["left_gate.weight"])
    right_gate_weight = to_half(weights["right_gate.weight"])
    out_gate_weight = to_half(weights["out_gate.weight"])
    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias = to_half(weights["to_out_norm.bias"])
    to_out_weight = to_half(weights["to_out.weight"])

    # 1. LayerNorm on input (half)
    x = torch.nn.functional.layer_norm(
        input_tensor.half(),
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape (B,N,N,dim) half

    # flatten to (B*N*N, dim)
    x_flat = x.view(-1, dim)  # contiguous? Since view on half broadcast.

    # 2. Fused linear for left + left gate
    left_concat_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # (2*H, dim)
    left_concat = torch.nn.functional.linear(x_flat, left_concat_weight)  # (B*N*N, 2*H) half
    left_proj_raw = left_concat[:, :hidden_dim]  # (B*N*N, H)
    left_gate_raw = left_concat[:, hidden_dim:]  # (B*N*N, H)

    # 3. Fused linear for right + right gate
    right_concat_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)
    right_concat = torch.nn.functional.linear(x_flat, right_concat_weight)
    right_proj_raw = right_concat[:, :hidden_dim]
    right_gate_raw = right_concat[:, hidden_dim:]

    # 4. out_gate separate linear

def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_A_batch, stride_A_m, stride_A_k,
                          stride_B_batch, stride_B_k, stride_B_n,
                          stride_C_batch, stride_C_m, stride_C_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    m_mask = offs_m < M
    n_mask = offs_n < N

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # Load A tile

"""
TriMul (outgoing) – fast half‑precision implementation.

Algorithm
---------
1. Cast the input to fp16 and apply the first LayerNorm (dim‐axis).
2. Fuse the left/right projection + gate linear layers:
   * (proj | gate)·x  →  (2·hidden_dim, dim) weight matrix.
   * Split the result into projected values and gate logits.
3. (Optional) apply the pair‑wise mask to the projected tensors.
4. Apply sigmoid to the gate logits and multiply them with the projected tensors.
5. Core triangle‑multiplicative update:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM over the hidden dimension.
   We perform the GEMM with `torch.matmul` on fp16 tensors (Tensor‑Core
   accelerated) – the batch dimension is (B·hidden_dim).
6. Layer‑norm over the hidden dimension (`to_out_norm`).
7. Element‑wise multiplication with the output gate using a tiny Triton kernel.
8. Final linear projection back to `dim` and cast the result to fp32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    out_norm_ptr,   # fp16* (out_norm flattened)
    out_gate_ptr,   # fp16* (out_gate flattened)
    out_ptr,        # fp16* (result flattened)
    size,           # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(out_norm_ptr + offs, mask=mask)
    b = tl.load(out_gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) for the triangle update
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each batch index ``b``:
        C[b] = A[b] @ B[b]   where A[b] is (M, K) and B[b] is (K, N)
    All tensors are assumed to be in fp16; accumulation is performed in fp32.
    """
    batch_id = tl.program_id(0)          # batch (B * hidden_dim)
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = (A_ptr + batch_id * stride_A_batch
                  + offs_m[:, None] * stride_A_m
                  + offs_k[None, :] * stride_A_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B tile (K x N)
        b_ptrs = (B_ptr + batch_id * stride_B_batch
                  + offs_k[:, None] * stride_B_k
                  + offs_n[None, :] * stride_B_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # multiply‑accumulate
        acc += tl.dot(a, b)

    # store result (fp16)
    c_ptrs = (C_ptr + batch_id * stride_C_batch
              + offs_m[:, None] * stride_C_m
              + offs_n[None, :] * stride_C_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass – highly optimized for H100 with fp16/TensorCores.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], float32 (on CUDA)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be all‑ones
        - weights      : dict of model weights (float32)
        - config       : dict, must contain ``dim`` and ``hidden_dim``.
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper: cast tensors to fp16 once (weights are stored FP32)
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    norm_w = to_fp16(weights["norm.weight"])
    norm_b = to_fp16(weights["norm.bias"])

    left_proj_w = to_fp16(weights["left_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])

    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (keep everything in fp16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.half(),
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Flatten for the linear layers
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gate for left side
    # ------------------------------------------------------------------
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2·H, dim)
    left_concat = F.linear(x_flat, left_concat_w)                # (B·N·N, 2·H)
    left_proj_raw = left_concat[:, :hidden_dim]                  # (B·N·N, H)
    left_gate_raw = left_concat[:, hidden_dim:]                  # (B·N·N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Fused projection + gate for right side
    # ------------------------------------------------------------------
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2·H, dim)
    right_concat = F.linear(x_flat, right_concat_w)
    right_proj_raw = right_concat[:, :hidden_dim]
    right_gate_raw = right_concat[:, hidden_dim:]

    # ------------------------------------------------------------------
    # 5️⃣ Output gate (no bias)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(x_flat, out_gate_w)               # (B·N·N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to 4‑D tensors
    # ------------------------------------------------------------------
    left_proj = left_proj_raw.view(bs, N, N, hidden_dim)      # (B, N, N, H)
    right_proj = right_proj_raw.view(bs, N, N, hidden_dim)
    out_gate = out_gate_raw.view(bs, N, N, hidden_dim)

    if not nomask:
        # mask is [B, N, N]; broadcast to the hidden dimension
        mask_h = mask.to(torch.float16, non_blocking=True).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate_raw).view(bs, N, N, hidden_dim)
    left = left_proj * left_gate          # (B, N, N, H)  fp16
    right = right_proj * right_gate       # (B, N, N, H)  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Core triangle multiplication – batched GEMM over the hidden dim
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, N, i)   i‑axis
    # right requires (K, N) ordering → transpose the inner pair of axes
    right_perm = right.permute(0, 3, 2, 1)        # (B, H, k, j)

    # Compute batch size = B * hidden_dim
    batch_groups = bs * hidden_dim

    # Strides needed by the Triton kernel
    stride_A_batch = left_perm.stride(0)   # stride for the outer batch dimension
    stride_A_m = left_perm.stride(2)       # stride for the M (i) axis
    stride_A_k = left_perm.stride(3)       # stride for the K (k) axis

    stride_B_batch = right_perm.stride(0)  # batch stride
    stride_B_k = right_perm.stride(2)      # K axis stride
    stride_B_n = right_perm.stride(3)      # N (j) axis stride

    # Output tensor (same shape as left_perm/right_perm)
    out_mat = torch.empty_like(left_perm, dtype=torch.float16)

    # Choose tile sizes that fit well on H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_groups,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_kernel[grid](
        A_ptr=left_perm.data_ptr(),
        B_ptr=right_perm.data_ptr(),
        C_ptr=out_mat.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_A_batch=stride_A_batch,
        stride_A_m=stride_A_m,
        stride_A_k=stride_A_k,
        stride_B_batch=stride_B_batch,
        stride_B_k=stride_B_k,
        stride_B_n=stride_B_n,
        stride_C_batch=out_mat.stride(0),
        stride_C_m=out_mat.stride(2),
        stride_C_n=out_mat.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to the original layout: (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1)      # (B, N, N, H)  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Output‑side LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # (B, N, N, H)  fp16

    # ------------------------------------------------------------------
    # 🔟 Element‑wise multiply with the output gate (Triton)
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.contiguous().view(-1)      # fp16 1‑D view
    out_gate_flat = out_gate.contiguous().view(-1)      # fp16 1‑D view
    gated_flat = torch.empty_like(out_norm_flat)        # fp16

    total_elements = out_norm_flat.numel()
    BLOCK_SIZE = 65536
    grid_mul = (triton.cdiv(total_elements, BLOCK_SIZE),)
    elementwise_mul_kernel[grid_mul](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(bs, N, N, hidden_dim)       # (B, N, N, H)  fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to ``dim`` and cast to fp32
    # ------------------------------------------------------------------
    out_fp32 = F.linear(gated, to_out_w)                # (B, N, N, dim)  fp16
    return out_fp32.float()
