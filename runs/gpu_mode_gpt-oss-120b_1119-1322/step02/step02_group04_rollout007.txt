def batch_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col, # row stride for B_T (which is original row stride)
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute batch
    batch_idx = pid // ( (M + BLOCK_M - 1) // BLOCK_M * (N + BLOCK_N - 1)// BLOCK_N )
    pid = pid % ( (M + BLOCK_M - 1) // BLOCK_M * (N + BLOCK_N - 1)// BLOCK_N )
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n
    # compute ranges
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute k loops
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        # compute mask for boundaries
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]
        # pointer offsets
        a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        b_ptrs = b_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

    # store result

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3 style) using Triton.

Algorithm
---------
1. Layer‑norm over the input feature dimension (dim).
2. Compute left/right projections, their gates and the out‑gate (all in float32).
3. Apply optional mask, multiply by the gates and cast the tensors to float16.
4. Core “triangle multiplication”:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched GEMM where each (batch,head) pair is a dense N×N
   matrix multiply with the second operand transposed.  The multiplication
   is implemented in a custom Triton kernel that
   • uses FP16 arithmetic and FP32 accumulation,
   • processes tiles of 64×64 elements,
   • works directly on the original NHWC layout (no permute copies).
5. Layer‑norm over the hidden dimension (hidden_dim) in float32.
6. Element‑wise gating with the out‑gate (torch multiplication – a kernel
   is already used for step 4, satisfying the requirement).
7. Final linear projection back to the original channel size.

All heavy arithmetic (steps 3‑4) runs in half‑precision on Tensor‑cores,
while the light per‑element ops stay in float32 for numerical stability.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication with the second operand transposed.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr,                     # float16*  left tensor   (B,N,N,H)
    b_ptr,                     # float16*  right tensor  (B,N,N,H) – accessed transposed
    c_ptr,                     # float16*  output tensor (B,N,N,H)
    M, N, K,                   # matrix dimensions (all equal to seq_len)
    # strides for A (left)
    stride_a_batch, stride_a_row, stride_a_col, stride_a_head,
    # strides for B (right, accessed as Bᵀ)
    stride_b_batch, stride_b_row, stride_b_col, stride_b_head,
    # strides for C (output)
    stride_c_batch, stride_c_row, stride_c_col, stride_c_head,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HEAD_DIM: tl.constexpr,   # hidden_dim (number of heads per batch)
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decompose program id into (batch, head, tile_m, tile_n)
    # ------------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_matrix = num_tiles_m * num_tiles_n

    matrix_idx = pid // tiles_per_matrix                # 0 … B*H‑1
    tile_idx   = pid % tiles_per_matrix

    pid_m = tile_idx // num_tiles_n                     # tile row index
    pid_n = tile_idx %  num_tiles_n                     # tile col index

    batch_idx = matrix_idx // HEAD_DIM
    head_idx  = matrix_idx %  HEAD_DIM

    # ------------------------------------------------------------------
    # Base pointers for this (batch, head) matrix
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_idx * stride_a_batch + head_idx * stride_a_head
    b_base = b_ptr + batch_idx * stride_b_batch + head_idx * stride_b_head
    c_base = c_ptr + batch_idx * stride_c_batch + head_idx * stride_c_head

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i – rows of A / C
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j – cols of Bᵀ / C

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        a_mask = (offs_i < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_j < N)[None, :]

        # A tile  (M×K) – strides: row → stride_a_row, col → stride_a_col
        a_ptrs = a_base + offs_i[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Bᵀ tile (K×N) – strides: row → stride_b_row (original col), col → stride_b_col (original row)
        b_ptrs = b_base + offs_k[:, None] * stride_b_row + offs_j[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # FP16 dot product accumulated into FP32
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result (C) – cast back to FP16
    # ------------------------------------------------------------------
    c_mask = (offs_i < M)[:, None] & (offs_j < N)[None, :]
    c_ptrs = c_base + offs_i[:, None] * stride_c_row + offs_j[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float, optional)
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with at least ``dim`` and ``hidden_dim``.
                         May contain ``nomask`` (bool).
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (C = dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )   # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates (all in float32)
    # ------------------------------------------------------------------
    # left projection
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    # right projection
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # optional mask (float32)
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # gates
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # apply gates
    left  = left_proj * left_gate       # [B, N, N, H]
    right = right_proj * right_gate

    # cast to half for the heavy multiplication
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Batched triangle multiplication (A @ Bᵀ) via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left_h.shape       # B, seq_len, seq_len, hidden_dim
    assert H == hidden_dim, "hidden_dim mismatch"

    # Strides of the original (B,N,N,H) layout – no permute / copy
    stride_a_batch, stride_a_row, stride_a_col, stride_a_head = left_h.stride()
    stride_b_batch, stride_b_row, stride_b_col, stride_b_head = right_h.stride()
    # For the transposed view of B: row stride = original col stride, col stride = original row stride
    stride_b_row_T = stride_b_col
    stride_b_col_T = stride_b_row

    # Output tensor (same layout as left/right)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Tile sizes – 64 is a good trade‑off on H100 (fits TensorCores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute how many program IDs we need
    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    programs_per_matrix = tiles_m * tiles_n
    total_programs = B * hidden_dim * programs_per_matrix

    grid = (total_programs,)

    # Launch kernel
    tri_mul_matmul_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        M=N, N=N, K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_a_head=stride_a_head,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row_T,
        stride_b_col=stride_b_col_T,
        stride_b_head=stride_b_head,
        stride_c_batch=stride_a_batch,          # same as A
        stride_c_row=stride_a_row,
        stride_c_col=stride_a_col,
        stride_c_head=stride_a_head,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HEAD_DIM=hidden_dim,
    )
    # ------------------------------------------------------------------
    # 4️⃣ LayerNorm over hidden_dim (float32) and out‑gate gating
    # ------------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)   # [B, N, N, H]

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B, N, N, H]

    # Element‑wise gating (out_norm * out_gate)
    gated = out_norm * out_gate   # still float32

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]       # (dim, hidden_dim)
    out = torch.nn.functional.linear(gated, to_out_w)  # [B, N, N, dim] – float32

    return out
