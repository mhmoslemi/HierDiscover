import triton.language as tl

@triton.jit
def apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr,
    out_ptr,
    size,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    proj = tl.load(proj_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)

    # sigmoid

    out = proj * gate_sig * mask_val

@triton.jit
def gate_mask_fused_kernel(
    lproj_ptr, lgate_ptr, rproj_ptr, rgate_ptr, mask_ptr,
    lout_ptr, rout_ptr,
    size,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    lproj = tl.load(lproj_ptr + offs, mask=mask)
    lgate = tl.load(lgate_ptr + offs, mask=mask)
    rproj = tl.load(rproj_ptr + offs, mask=mask)
    rgate = tl.load(rgate_ptr + offs, mask=mask)
    m = tl.load(mask_ptr + offs, mask=mask)

    lgate_sig = 1.0/(1.0 + tl.exp(-lgate))
    rgate_sig = 1.0/(1.0 + tl.exp(-rgate))

    lout = lproj * lgate_sig * m
    rout = rproj * rgate_sig * m

    tl.store(lout_ptr + offs, lout, mask=mask)
    tl.store(rout_ptr + offs, rout, mask=mask)
```

import triton
import triton.language as tl

@triton.jit
def gate_mask_fused_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    # mask to avoid out of bounds
    active = offs < n_elements

    # load tensors
    lp = tl.load(left_proj_ptr + offs, mask=active, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=active, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    m  = tl.load(mask_ptr + offs, mask=active, other=1.0)

    # sigmoid in half precision
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    lo = lp * lg_sig * m
    ro = rp * rg_sig * m

    tl.store(out_left_ptr + offs, lo, mask=active)
    tl.store(out_right_ptr + offs, ro, mask=active)

    The heavy N³ contraction is performed with a batched cuBLAS GEMM

    hidden_dim = config["hidden_dim"]

        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],

    )  # (5*H, dim)

    fused_out = F.linear(x_flat, fused_weight)  # (B*N*N, 5*H)

    H = hidden_dim
    left_proj_f32 = fused_out[:, 0:H]
    left_gate_f32 = fused_out[:, H:2 * H]
    right_proj_f32 = fused_out[:, 2 * H:3 * H]
    right_gate_f32 = fused_out[:, 3 * H:4 * H]
    out_gate_f32 = fused_out[:, 4 * H:5 * H]

    left_gate_sig = torch.sigmoid(left_gate_f32)
    right_gate_sig = torch.sigmoid(right_gate_f32)
    out_gate_sig = torch.sigmoid(out_gate_f32)  # will be used later

    # 4️⃣ Cast projections & gates to fp16 for the cubic contraction
    left_proj = left_proj_f32.to(torch.float16)
    right_proj = right_proj_f32.to(torch.float16)
    left_gate = left_gate_sig.to(torch.float16)
    right_gate = right_gate_sig.to(torch.float16)

    # 5️⃣ Prepare mask (broadcast to feature dimension)
    if mask is None or mask.numel() == 0:
        mask_fp16 = torch.ones(B, N, N, 1, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)

    # 6️⃣ Fuse mask & gates into the projections with a tiny Triton kernel
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    total = left_proj.numel()
    BLOCK_SIZE = 1024  # max 1024 threads per block on H100
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    gate_mask_fused_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_fp16,
        left_out,
        right_out,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=32,
    )

    # 7️⃣ Batched matrix multiply: out[i,j] = Σ_k left[i,k]·right[j,k]
    #    Treat each (batch, head) pair as an independent matmul.
    #    left_out/right_out shape: [B, N, N, H]
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)      # (BH, N, N)
    right_flat = right_perm.view(B * H, N, N)    # (BH, N, N)

    # torch.bmm uses Tensor Cores for fp16 → fp16 → fp32 accumulation
    out_mat_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_mat_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton helper: apply a sigmoid gate to a half‑precision tensor.
# This kernel is deliberately tiny – it only computes
#      out = proj * sigmoid(gate)
# for a flattened view of the tensor.  The heavy cubic contraction
# is delegated to torch.bmm (cuBLAS) which is far more efficient on
# H100 TensorCores.
# --------------------------------------------------------------------------- #
@triton.jit
def _gate_fused_kernel(
    proj_ptr,               # half*   – projection (e.g. left or right)
    gate_ptr,               # half*   – pre‑gate values
    out_ptr,                # half*   – gated result
    n_elements,             # i32     – total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < n_elements

    # load the two operands
    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)

    # half‑precision sigmoid (1 / (1 + exp(-x)))
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # write gated tensor
    tl.store(out_ptr + offs, proj * gate_sig, mask=active)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul operator.

    Algorithm
    ---------
    1️⃣ Layer‑norm over the input feature dimension.
    2️⃣ Fuse the five linear maps (left/right projection, three gates) into one
       dense matmul (`F.linear`).  The result is split into the five tensors.
    3️⃣ Apply a sigmoid to the three gate tensors.
    4️⃣ Cast projections and gates to ``float16`` and fuse *projection × sigmoid*
       with a tiny Triton kernel.  An optional mask (broadcasted over the hidden
       dimension) is applied with a simple ``*`` after the kernel.
    5️⃣ Core cubic contraction (∑ₖ left[i,k]·right[j,k]) is performed as a batched
       matrix multiply:  ``torch.bmm(left, rightᵀ)``.  This uses cuBLAS Tensor‑Core
       kernels and is far faster than a hand‑written Triton GEMM on H100.
    6️⃣ Second layer‑norm over the hidden dimension, multiply by the output gate,
       and a final linear projection back to the original feature dimension.
    All heavy arithmetic stays on the GPU; only lightweight element‑wise work is
    handled by Triton.
    """
    # ----------------------------------------------------------------------- #
    # Unpack inputs
    # ----------------------------------------------------------------------- #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # also called c_z in the paper

    # ----------------------------------------------------------------------- #
    # 1️⃣ First LayerNorm (over the last feature axis)
    # ----------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                           # [B, N, N, dim]

    B, N, _, _ = x.shape

    # ----------------------------------------------------------------------- #
    # 2️⃣ Fuse the five linear maps into a single matmul
    # ----------------------------------------------------------------------- #
    x_flat = x.reshape(B * N * N, dim).contiguous()          # (B·N², dim)

    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )                                                       # (5·H, dim)

    # One dense projection – keep it in FP32 for numerical stability
    fused_out = F.linear(x_flat, fused_weight)              # (B·N², 5·H)

    H = hidden_dim
    left_proj_f32 = fused_out[:, 0:H]
    left_gate_f32 = fused_out[:, H:2 * H]
    right_proj_f32 = fused_out[:, 2 * H:3 * H]
    right_gate_f32 = fused_out[:, 3 * H:4 * H]
    out_gate_f32 = fused_out[:, 4 * H:5 * H]

    # ----------------------------------------------------------------------- #
    # 3️⃣ Sigmoid the gate tensors
    # ----------------------------------------------------------------------- #
    left_gate_sig = torch.sigmoid(left_gate_f32)
    right_gate_sig = torch.sigmoid(right_gate_f32)
    out_gate_sig = torch.sigmoid(out_gate_f32)   # used later

    # ----------------------------------------------------------------------- #
    # 4️⃣ Cast projections & gates to half‑precision for the cubic step
    # ----------------------------------------------------------------------- #
    left_proj = left_proj_f32.to(torch.float16)
    right_proj = right_proj_f32.to(torch.float16)
    left_gate = left_gate_sig.to(torch.float16)
    right_gate = right_gate_sig.to(torch.float16)

    # ----------------------------------------------------------------------- #
    # 5️⃣ Gate the projections with a tiny Triton kernel
    # ----------------------------------------------------------------------- #
    total_elems = left_proj.numel()                     # B·N·N·H
    BLOCK_SIZE = 1024                                   # max threads per block
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    left_gated = torch.empty_like(left_proj)
    right_gated = torch.empty_like(right_proj)

    # left side
    _gate_fused_kernel[grid](
        left_proj,
        left_gate,
        left_gated,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=32,
    )
    # right side
    _gate_fused_kernel[grid](
        right_proj,
        right_gate,
        right_gated,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=32,
    )

    # ----------------------------------------------------------------------- #
    # 6️⃣ Optional mask (broadcast over the hidden dimension)
    # ----------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        # mask shape: [B, N, N] → [B, N, N, 1] → half‑precision
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left_gated = left_gated * mask_fp16
        right_gated = right_gated * mask_fp16

    # ----------------------------------------------------------------------- #
    # 7️⃣ Core N³ contraction via batched GEMM (torch.bmm → cuBLAS TensorCore)
    # ----------------------------------------------------------------------- #
    # reshape to (B·H, N, N) where each slice is one independent matmul
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_gated.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)                  # (BH, N, N)
    right_flat = right_perm.view(B * H, N, N)                # (BH, N, N)

    # out_ij = Σ_k left[i,k] * right[j,k]  →  A @ Bᵀ per batch
    out_mat_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # convert to FP32 for the following LayerNorms
    out = out_mat_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # shape now: [B, N, N, H]

    # ----------------------------------------------------------------------- #
    # 8️⃣ Second LayerNorm, output‑gate, final linear projection
    # ----------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                           # still FP32

    # multiply by the output gate (cast to FP32)
    out_gate = out_gate_sig.to(out.dtype)
    out = out * out_gate

    # final linear back to original feature dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
```
