"""
TriMul (outgoing) – Triton‐accelerated forward pass
===================================================

The module works on a 4‑D tensor `x` of shape [B, N, N, C] (C = dim).
The forward pass consists of the following steps:

1️⃣ Input LayerNorm (FP32, bias + weight).
2️⃣ Two linear projections (left/right) and two gating projections are
   *fused*: we concatenate the projection and gate weight matrices so that
   the large tensor `x` is read only once per side.  All heavy work is done
   in FP16; the sigmoid gates are applied in FP16 as well.
3️⃣ Optional mask (broadcast over the hidden dimension) is applied to the
   projected parts **before** the gates.
4️⃣ The gated tensors `left` and `right` (shape [B, N, N, H]) are used in
   the core N³ update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication `left @ rightᵀ`.  This is
   performed by a custom Triton kernel that processes each (batch × hidden)
   matrix independently, loads FP16 tiles and accumulates in FP32, thereby
   exploiting tensor‑core `tl.dot`.
5️⃣ Hidden‑dimension LayerNorm + multiplicative out‑gate (computed from the
   *normed* `x`).
6️⃣ Final linear projection back to `dim`.

The kernel is deliberately written with a fixed block size (64×64×64) that
fits the H100 shared‑memory budget and works well for all test sequence
lengths (multiples of 64).  The implementation keeps the overall memory
traffic low while still exposing the full compute intensity of the O(N³)
operation.

The `custom_kernel` function follows the exact same API as required by the
evaluation harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_out_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                           # matrix dimensions (M=N, K=N)
    stride_ab, stride_am, stride_ak,   # A: batch‑stride, row‑stride, col‑stride
    stride_bb, stride_bk, stride_bn,   # B: batch‑stride, col‑stride (k), row‑stride (n)
    stride_cb, stride_cm, stride_cn,   # C: batch‑stride, row‑stride, col‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the outgoing TriMul core:
        C[b] = A[b] @ B[b]ᵀ
    A has shape (M, K), B has shape (N, K) and we treat B as (K, N) inside
    the kernel so that `tl.dot(a, b)` yields a (BLOCK_M, BLOCK_N) tile.
    All loads are FP16, accumulation is FP32.
    """
    pid_batch = tl.program_id(2)           # which (batch × hidden) matrix
    pid_m = tl.program_id(0)               # block‑row index of C
    pid_n = tl.program_id(1)               # block‑col index of C

    # --------------------- tile offsets ---------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------- base pointers --------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # pointers to the current tile of A (M×K) and Bᵀ (K×N)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------- K loop -------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # global offset for the current K‑tile
        cur_off = k_tile * BLOCK_K

        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)           # FP16 → FP32 accumulation

    # --------------------- store result --------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """Triton‑accelerated outgoing TriMul forward pass.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with ``dim`` and ``hidden_dim`` keys

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 2) Input layer‑norm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3) Fuse projection + gate linear layers (read `x` once per side)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weight matrices: (2*H, dim)
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # Work in FP16 for the heavy linear transforms
    x_fp16 = x.to(torch.float16)

    # Linear without bias (torch.nn.functional.linear is highly tuned)
    left_combined = F.linear(x_fp16, left_combined_w)   # [B, N, N, 2*H] FP16
    right_combined = F.linear(x_fp16, right_combined_w)  # same shape

    # Split projection and gate parts
    left_proj = left_combined[..., :hidden_dim]                     # (B,N,N,H) FP16
    left_gate = left_combined[..., hidden_dim:].sigmoid()           # (B,N,N,H) FP16
    right_proj = right_combined[..., :hidden_dim]                   # (B,N,N,H) FP16
    right_gate = right_combined[..., hidden_dim:].sigmoid()         # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5) Apply gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # (B,N,N,H) FP16
    right = right_proj * right_gate # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 6) Core N³ update: batched GEMM left @ rightᵀ   (FP16 → FP32)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # batch, seq_len, seq_len, hidden_dim

    # Rearrange to (B, H, N, N) and flatten the first two dims → (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat = left_t.view(-1, N, N)   # (B*H, N, N)  FP16
    right_flat = right_t.view(-1, N, N) # (B*H, N, N)  FP16

    total = left_flat.shape[0]  # = B * H

    # Allocate destination for the GEMM result (FP32 accumulation)
    out_flat = torch.empty((total, N, N), dtype=torch.float32, device=device)

    # Block sizes – 64 works well on H100 for all N in the test suite
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),    # blocks over rows (i dimension)
        triton.cdiv(N, BLOCK_N),    # blocks over cols (j dimension)
        total,                      # one program per (batch * hidden)
    )

    # Launch the Triton kernel
    _batched_gemm_out_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0),   # batch‑stride of A
        left_flat.stride(1),   # row‑stride (M dimension)
        left_flat.stride(2),   # col‑stride (K dimension)
        right_flat.stride(0),  # batch‑stride of B
        right_flat.stride(2),  # col‑stride for Bᵀ (K)
        right_flat.stride(1),  # row‑stride for Bᵀ (N)
        out_flat.stride(0),    # batch‑stride of C
        out_flat.stride(1),    # row‑stride of C
        out_flat.stride(2),    # col‑stride of C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Re‑shape back to (B, N, N, H) in FP32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Hidden‑dim layer‑norm + multiplicative out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    # Out‑gate is computed from the *normed* input `x`
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # (B,N,N,H) FP32
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) FP32

    return out
