    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    BLOCK_M: tl.constexpr,

    BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)
    # compute block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask validity
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for current batch
    a_base = a_ptr + pid_batch * stride_ah
    b_base = b_ptr + pid_batch * stride_bh
    c_base = c_ptr + pid_batch * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles

custom_kernel: Triton‑accelerated forward pass of the “outgoing” TriMul module
---------------------------------------------------------------------------
Algorithm
1. Layer‑norm the input tensor (float32) → [B, N, N, D].
2. Cast to fp16 and apply bias‑free linear projections (left/right) and three
   sigmoid gates (left_gate, right_gate, out_gate).  Optional mask is broadcast
   over the hidden dimension and applied before gating.
3. Rearrange tensors to (B·H, N, N) and launch a Triton kernel that computes,
   for each hidden channel, the batched matrix multiplication
        out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
   i.e. a GEMM with B·H independent matrix pairs (size N×N).  The kernel
   uses 128×128 tiles for rows/cols and a 32‑element K‑tile, works in fp16
   with fp32 accumulation.
4. Reshape the result back to [B, N, N, H] (fp16) and apply a hidden‑dim
   LayerNorm, multiply by the out_gate, and a final linear projection back to
   the original dimension D.
5. Return the result as float32.

The heavy O(N³) work is completely off‑loaded to Triton, giving a large speed‑up
over the reference PyTorch implementation while preserving numerical
correctness (final tensor is float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: batched GEMM A @ Bᵀ for many (batch*hidden) matrices
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,         # pointers to A, B, C (fp16)
    M, N, K,                     # matrix sizes (all equal N in our case)
    stride_ah, stride_am, stride_ak,   # A strides: batch, row (i), col (k)
    stride_bh, stride_bk, stride_bn,   # B strides: batch, col (k), row (j) – note transposed layout
    stride_ch, stride_cm, stride_cn,   # C strides: batch, row (i), col (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each matrix in the batch.
    Accumulation is performed in fp32, the result is stored as fp16.
    """
    pid_batch = tl.program_id(2)           # which (batch * hidden) matrix
    pid_m = tl.program_id(0)               # block row of C
    pid_n = tl.program_id(1)               # block col of C

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current matrix
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_ah
    b_base = b_ptr + pid_batch * stride_bh
    c_base = c_ptr + pid_batch * stride_ch

    # ------------------------------------------------------------------
    # Accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (shape BLOCK_M x BLOCK_K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_am   # row stride
            + cur_k[None, :] * stride_ak,   # col stride
            mask=mask_m[:, None] & (cur_k < K),
            other=0.0,
        )

        # Load B tile (shape BLOCK_K x BLOCK_N) – note B is stored as (K, N)
        b = tl.load(
            b_base
            + cur_k[:, None] * stride_bk   # col‑K stride  (original column)
            + offs_n[None, :] * stride_bn, # row‑N stride (original row)
            mask=(cur_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Dot product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back to C (cast to fp16 on store)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_base
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask:        torch.Tensor [B, N, N] (bool/int/float) or None
            - weights:     dict of model parameters (torch.Tensor, float32)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # 0) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy lifting
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to fp16 (all Linear layers are bias‑free)
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate= weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left)    # [B, N, N, H]
    right = F.linear(x_h, w_right)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Compute gates (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] → broadcast to hidden dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # 8) Prepare tensors for the batched GEMM
    #    Layout: (B, H, N, N) → contiguous → view (B*H, N, N)
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_t.view(-1, N, N)   # [B*H, N, N]   fp16
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]   fp16

    total = left_flat.shape[0]           # B * H  (number of independent matrices)

    # ------------------------------------------------------------------
    # 9) Output buffer for the GEMM
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)   # fp16

    # ------------------------------------------------------------------
    # 10) Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along rows (M)
        triton.cdiv(N, BLOCK_N),   # blocks along cols (N)
        total,                     # one program per (batch * hidden) matrix
    )

    # ------------------------------------------------------------------
    # 11) Launch the batched GEMM kernel (C = A @ Bᵀ)
    # ------------------------------------------------------------------
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0),   # stride_ah (batch)
        left_flat.stride(1),   # stride_am (row)
        left_flat.stride(2),   # stride_ak (col)
        right_flat.stride(0),  # stride_bh (batch)
        right_flat.stride(2),  # stride_bk (col/k) – note transposed layout
        right_flat.stride(1),  # stride_bn (row/j)
        out_flat.stride(0),    # stride_ch (batch)
        out_flat.stride(1),    # stride_cm (row)
        out_flat.stride(2),    # stride_cn (col)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 12) Reshape back to [B, N, N, H] (fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 13) Hidden‑dim LayerNorm + out_gate multiplication
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate   # element‑wise gating (fp16)

    # ------------------------------------------------------------------
    # 14) Final linear projection back to original dimension D (fp16)
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # [B, N, N, D] fp16

    # ------------------------------------------------------------------
    # 15) Cast to float32 for downstream compatibility
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

This implementation off‑loads the O(N³) core of TriMul to a highly‑tuned Triton kernel using large 128×128 tiles and fp16 arithmetic with fp32 accumulation. All other surrounding operations (layer‑norm, linear projections, gating) remain in PyTorch but are performed in fp16 to reduce memory bandwidth. The function returns the expected float32 tensor while respecting the required signature.
