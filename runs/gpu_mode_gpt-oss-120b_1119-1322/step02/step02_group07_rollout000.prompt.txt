import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Outgoing TriMul (Triangular Multiplicative Update)

Algorithm
---------
1. Input LayerNorm (FP32).
2. Linear projections -> left/right (FP32) and three gating linear layers.
3. Convert projections to fp16.
4. *Triton kernel* – fuse optional binary mask, left‑gate and right‑gate:
      left_out = left * mask * left_gate
      right_out = right * mask * right_gate
   (mask is broadcasted over the hidden dimension without materialising a
    full‑size mask tensor.)
5. Batched GEMM   out = left_out @ right_outᵀ  for every hidden channel.
   Implemented with torch.bmm on fp16 tensors ⇒ Tensor‑Core utilisation.
6. Reshape back to [B, N, N, hidden] and promote to fp32.
7. Second LayerNorm over the hidden dimension.
8. Apply the output gate (sigmoid) and final linear projection back to
   the model dimension.

The heavy computation (step 5) is performed by cuBLAS (torch.bmm) which
is far faster than a naïve Triton matmul for the batch sizes used in
AlphaFold3.  Step 4 provides at least one custom Triton kernel as
required by the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr,                # fp16 * left  (contiguous)
    right_ptr,               # fp16 * right (contiguous)
    mask_ptr,                # fp16 * mask  (B,N,N) contiguous, broadcasted
    left_gate_ptr,           # fp16 * left_gate  (contiguous)
    right_gate_ptr,          # fp16 * right_gate (contiguous)
    out_left_ptr,            # fp16 * output left
    out_right_ptr,           # fp16 * output right
    total_elements: tl.int64,
    H: tl.constexpr,         # hidden dimension (known at compile time)
    BLOCK: tl.constexpr,     # block size (threads per program)
):
    """Fuse mask and per‑channel gating.
    The tensors are flattened, i.e. layout order:
        (b, i, j, h) → offset = ((b*N + i)*N + j)*H + h
    The mask has shape (B, N, N) → offset_mask = ((b*N + i)*N + j)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load data (masked loads → zero for out‑of‑range threads)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast over H)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # Fuse the three element‑wise multiplications
    l = l * lg * m
    r = r * rg * m

    # Store results
    tl.store(out_left_ptr + offs, l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [B, N, N, dim],  float32
            mask         : torch.Tensor, shape [B, N, N],      float32/bool
            weights      : dict of torch.Tensor model parameters
            config       : dict with keys
                           - "dim"        (int)
                           - "hidden_dim" (int)
                           - "nomask"     (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim], float32
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (still FP32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep FP32 for later

    # ------------------------------------------------------------------
    # 4. Cast projections to fp16
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Optional mask + gating fusion via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask: convert to fp16 (0/1) and make contiguous
        mask_fp16 = mask.to(dtype=torch.float16).contiguous()
        # Ensure all inputs are contiguous for the kernel
        left_fp16 = left_fp16.contiguous()
        right_fp16 = right_fp16.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # Allocate fused outputs
        left_out = torch.empty_like(left_fp16)
        right_out = torch.empty_like(right_fp16)

        total_elems = left_fp16.numel()
        BLOCK = 1024  # threads per program
        grid = ( (total_elems + BLOCK - 1) // BLOCK, )

        _mask_gate_kernel[grid](
            left_fp16,
            right_fp16,
            mask_fp16,
            left_gate,
            right_gate,
            left_out,
            right_out,
            total_elems,
            H=hidden,
            BLOCK=BLOCK,
        )
        left_fp16 = left_out
        right_fp16 = right_out
    else:
        # No mask → only apply the per‑channel gates
        left_fp16 = left_fp16 * left_gate
        right_fp16 = right_fp16 * right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out = left @ rightᵀ   (fp16 → fp32 accumulator)
    # ------------------------------------------------------------------
    B, N, _, _ = left_fp16.shape  # B, N, N, hidden
    # reshape to (B*hidden, N, N)
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # torch.bmm uses Tensor Cores for fp16 inputs
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 7. Restore original layout and promote to FP32
    # ------------------------------------------------------------------
    out = out_mat_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
