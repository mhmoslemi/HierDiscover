import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Optimized outgoing TriMul kernel (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm over the input feature dimension (C = dim).
2. Fuse the two linear projections per side (proj + gate) into a single GEMM,
   then split the result and apply a sigmoid to the gate half.
3. Compute the out‑gate (separate linear + sigmoid).
4. Optionally mask the projected tensors, then apply left/right gates.
5. Perform the core “triangle multiplication”:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is realized as a large batched half‑precision GEMM:
   (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N).
6. Layer‑norm over the hidden dimension (H = hidden_dim) in float32.
7. Triton kernel for the cheap element‑wise product out_norm * out_gate.
8. Final half‑precision linear projection back to the original channel size.

All heavy tensor work uses float16 (Tensor‑core friendly) and cuBLAS.
Only the element‑wise gating (step 7) is written in Triton, satisfying the
requirement to include a custom kernel.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # float32*  (out_norm flattened)
    b_ptr,                # float32*  (out_gate flattened)
    out_ptr,              # float32*  (output flattened)
    size,                 # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """
    Simple element‑wise multiplication kernel.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool / 0‑1 float)
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension (C = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to float16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse left/right projections with their gates
    # ------------------------------------------------------------------
    # ----- left side -----
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)  # (H, C)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, C)

    left_comb = torch.nn.functional.linear(x_h, left_comb_w)     # [B,N,N,2H]
    left = left_comb[..., :hidden_dim]                          # [B,N,N,H]
    left_gate_pre = left_comb[..., hidden_dim:]                 # [B,N,N,H]
    left_gate = torch.sigmoid(left_gate_pre)                    # half

    # ----- right side -----
    right_proj_w = weights["right_proj.weight"].to(torch.float16)   # (H, C)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, C)

    right_comb = torch.nn.functional.linear(x_h, right_comb_w)     # [B,N,N,2H]
    right = right_comb[..., :hidden_dim]                           # [B,N,N,H]
    right_gate_pre = right_comb[..., hidden_dim:]                  # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_pre)                     # half

    # ----- out gate (separate) -----
    out_gate_w = weights["out_gate.weight"].to(torch.float16)       # (H, C)
    out_gate_pre = torch.nn.functional.linear(x_h, out_gate_w)    # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)                         # half

    # ------------------------------------------------------------------
    # 4. Apply optional mask and the left/right gates
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) → (B, N, N, 1) and cast to half
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triangle multiplication via batched half‑precision GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # (B, N, N, H)

    # (B, H, N, N) layout, then merge B and H for a single bmm batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)   # (B·H, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # Each head does: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. Output‑side LayerNorm over hidden_dim (still in float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H], float32

    # ------------------------------------------------------------------
    # 7. Element‑wise gating (out_norm * out_gate) using Triton
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.contiguous().view(-1)                     # float32
    out_gate_flat = out_gate.to(torch.float32).contiguous().view(-1)   # float32
    gated_flat = torch.empty_like(out_norm_flat)

    total_elements = out_norm_flat.numel()
    BLOCK_SIZE = 65536  # 64 K elements per program

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(B, N, N, hidden_dim)  # [B,N,N,H], float32

    # ------------------------------------------------------------------
    # 8. Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)

    # Cast gated to half‑precision for the final GEMM
    gated_h = gated.to(torch.float16)
    out_h = torch.nn.functional.linear(gated_h, to_out_w)  # [B,N,N,dim], half

    # Return in float32 as required by the API
    return out_h.to(torch.float32)
