```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_gate_mask_kernel(
    proj_ptr,            # [total, H] flattened
    gate_ptr,            # same shape
    mask_ptr,            # [total] flattened mask per (b,i,j) i.e., total/hidden_dim entries
    out_ptr,
    total_elements: tl.constexpr,  # total elements = B* N * N * H
    H: tl.constexpr,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)
    mask_off = off // H
    mask = tl.load(mask_ptr + mask_off, mask=off < total_elements, other=0.0)
    proj = tl.load(proj_ptr + off, mask=off < total_elements, other=0.0)
    gate = tl.load(gate_ptr + off, mask=off < total_elements, other=0.0)
    # assumes mask is broadcast to half
    out = proj * gate * mask
    tl.store(out_ptr + off, out, mask=off < total_elements)

def apply_gate_mask(proj, gate, mask):
    # proj, gate: FP16 tensors shape [B,N,N,H]
    # mask: FP16 tensor shape [B,N,N] (or None)
    B,N,N2,H = proj.shape
    if mask is None:
        return proj * gate
    # ensure contiguous
    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()
    # Flatten
    total = proj.numel()
    out = torch.empty_like(proj)
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    fused_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK=BLOCK,
    )
    return out
```

"""TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm (FP32) on the input tensor  x ∈ ℝ^{B×N×N×D}.
2. Cast x → FP16 and apply the five bias‑free linear layers
   (left/right projections, three gates).  All linear layers use
   torch.nn.functional.linear which internally calls cuBLAS.
3. Fuse the element‑wise mask (if present) with the two gating
   multiplications by a tiny Triton kernel.  The kernel works on
   the flattened tensor of size B·N·N·H and broadcasts a
   per‑pair mask across the hidden dimension H.
4. Compute the cubic update
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which equals a batched matrix‑multiply
        C = A @ Bᵀ   for each hidden channel.
   We reshape left/right to (B·H, N, N) and use the highly‑optimized
   torch.bmm (cuBLAS, Tensor‑cores) instead of a custom GEMM.
5. Reshape back to [B,N,N,H], apply a second layer‑norm on the
   hidden dimension (FP32), multiply by the output‑gate, and finally
   project back to the original dimension D.
6. Return the result as a FP32 tensor.

A Triton kernel is required (step 3) but the heavy O(N³) work is
delegated to cuBLAS via torch.bmm, which on an H100 gives a large
speed‑up over a naïve Triton GEMM implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) with mask broadcast.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    proj_ptr,               # fp16 pointer to projected tensor   [B,N,N,H] flattened
    gate_ptr,               # fp16 pointer to gate tensor         same shape
    mask_ptr,               # fp16 pointer to mask tensor         [B,N,N] flattened
    out_ptr,                # fp16 pointer to output tensor       same shape as proj/gate
    total_elements: tl.constexpr,   # B * N * N * H
    hidden_dim: tl.constexpr,       # H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion: out = proj * gate * mask   (mask broadcast over H)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    # Guard: compute only within valid range.
    mask = offs < total_elements

    # Load projected value and gate value (FP16).
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index: one mask value per (b,i,j) -> repeats every H entries.
    mask_offs = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_offs, mask=mask, other=0.0)

    # Fuse the three tensors.
    out = proj * gate * mask_val

    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj, gate, mask):
    """
    Fuse ``proj * gate`` and optional ``mask`` using Triton.

    Args:
        proj (torch.Tensor): [B,N,N,H] FP16
        gate (torch.Tensor): [B,N,N,H] FP16 (already after sigmoid)
        mask (torch.Tensor or None): [B,N,N] FP16 or None

    Returns:
        torch.Tensor: result of elementwise multiplication, same shape as proj.
    """
    if mask is None:
        return proj * gate

    # Ensure contiguous layout for flattening.
    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()

    total = proj.numel()                     # B*N*N*H
    H = proj.shape[-1]                       # hidden_dim
    out = torch.empty_like(proj)

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    fused_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data (tuple):
            - input_tensor (torch.Tensor): [B, N, N, D]  (float32)
            - mask (torch.Tensor or None): [B, N, N]   (float32 or bool)
            - weights (dict): model parameters (float32)
            - config (dict): must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor: [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D], float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Helper to promote weight tensors on‑the‑fly, avoiding extra memory.
    def _to_half(t):
        return t.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections and gates (bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_fp16, _to_half(weights["left_proj.weight"]))          # [B,N,N,H] fp16
    right_proj = F.linear(x_fp16, _to_half(weights["right_proj.weight"]))

    left_gate = F.linear(x_fp16, _to_half(weights["left_gate.weight"])).sigmoid()   # fp16
    right_gate = F.linear(x_fp16, _to_half(weights["right_gate.weight"])).sigmoid()
    out_gate = F.linear(x_fp16, _to_half(weights["out_gate.weight"])).sigmoid()    # fp16

    # ------------------------------------------------------------------
    # 4) Optional mask + element‑wise gating (fused Triton kernel)
    # ------------------------------------------------------------------
    if mask is not None:
        # Convert mask to FP16 and flatten for Triton kernel.
        mask_fp16 = mask.to(torch.float16)
    else:
        mask_fp16 = None

    left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)   # [B,N,N,H]
    right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 5) Core cubic update: batched GEMM A @ Bᵀ  (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Move hidden dimension to batch axis: [B, H, N, N]
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    # Collapse (B * H) into a single batch dimension.
    left_flat = left_t.view(-1, N, N)           # [B*H, N, N]  fp16
    right_flat = right_t.view(-1, N, N)        # [B*H, N, N]  fp16

    # Batched matrix‑multiply: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # fp16

    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm (float32) + out‑gate multiplication
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])   # [B,N,N,D] float32

    return out_fp32
```
