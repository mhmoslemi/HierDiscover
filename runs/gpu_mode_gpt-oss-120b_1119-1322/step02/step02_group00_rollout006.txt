"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated core update.
The kernel computes, for each hidden channel h,
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
which is a batched matrix multiplication A @ Bᵀ with A/B of shape (N,N).
All heavy arithmetic is done in fp16 (accumulated in fp32) while
layer‑norms and the final projection stay in float32 for numerical stability.
A small Triton kernel is used for the N³ GEMM; the remaining ops use
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B: batch, col (k), row (n) strides (Bᵀ layout)
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each (batch * hidden) matrix:
        C = A @ Bᵀ
    where A, B, C are fp16 tensors with fp32 accumulation.
    The tile sizes BLOCK_M/N/K are chosen at launch time.
    """
    pid_batch = tl.program_id(2)               # which (batch * hidden) matrix
    pid_m = tl.program_id(0)                    # block-row of C
    pid_n = tl.program_id(1)                    # block-col of C

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # Tile pointers (initial k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        # Load A and B tiles (apply bounds masks)
        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # Write the result tile back to C
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (torch.Tensor, float32)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the rest of the forward pass
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) FP16 weight copies (bias‑free linear layers)
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate= weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left = F.linear(x_h, w_left)   # [B, N, N, H]
    right = F.linear(x_h, w_right)

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Prepare tensors for the batched GEMM (A @ Bᵀ)
    #    Layout needed: [batch*hidden, N, N] where each matrix is fp16.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, seq_len, hidden_dim
    # Move hidden dim to second axis and make it contiguous for optimal access
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_t.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]

    total = left_flat.shape[0]           # B * H

    # ------------------------------------------------------------------
    # 9) Allocate output buffer (fp16)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 10) Choose tile sizes (simple heuristic)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    # ------------------------------------------------------------------
    # 11) Triton launch – compute C = A @ Bᵀ for every hidden channel
    # ------------------------------------------------------------------
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows of C
        triton.cdiv(N, BLOCK_N),   # cols of C
        total,                     # one program per (batch * hidden)
    )
    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0),   # stride_ab (batch stride)
        left_flat.stride(1),   # stride_am (row stride)
        left_flat.stride(2),   # stride_ak (col stride)
        right_flat.stride(0),  # stride_bb (batch stride)
        right_flat.stride(2),  # stride_bk (col‑k stride for Bᵀ)
        right_flat.stride(1),  # stride_bn (row‑n stride for Bᵀ)
        out_flat.stride(0),    # stride_cb (batch stride)
        out_flat.stride(1),    # stride_cm (row stride)
        out_flat.stride(2),    # stride_cn (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 12) Reshape back to [B, N, N, H] (fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 13) Hidden‑dim LayerNorm, multiply by out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 14) Final projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)          # [B, N, N, D]  fp16

    # ------------------------------------------------------------------
    # 15) Cast back to float32 (expected downstream type)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
