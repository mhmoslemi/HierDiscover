```python
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Steps:
    1. LayerNorm over channel dim (dim).
    2. Fuse proj+gate linear transforms for left/right using concatenated weights.
    3. Compute out_gate.
    4. Apply optional mask.
    5. Apply sigmoid gates.
    6. Compute triangle multiplication per hidden dim (batched GEMM in FP16).
    7. LayerNorm over hidden dim.
    8. Element‑wise multiply with out_gate (Triton kernel, FP16).
    9. Final linear projection back to original dim.
    All heavy computation in FP16 to exploit Tensor‑cores on H100.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1️⃣ Input LayerNorm in FP16
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    ).to(torch.float16)   # shape (B,N,N,dim)

    # flatten batch*seq*seq for linear
    x_flat = x.view(-1, dim)   # (B*N*N, dim)

    # 2️⃣ Left proj+gate fused linear
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2*H, dim)
    left_concat = torch.nn.functional.linear(x_flat, left_concat_w)  # (B*N*N, 2*H)
    left_proj_raw = left_concat[:, :hidden_dim]   # (B*N*N, H)
    left_gate_raw = left_concat[:, hidden_dim:]   # (B*N*N, H)

    # 2️⃣ Right proj+gate fused linear
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_concat = torch.nn.functional.linear(x_flat, right_concat_w)
    right_proj_raw = right_concat[:, :hidden_dim]
    right_gate_raw = right_concat[:, hidden_dim:]

    # 3️⃣ Output gate linear
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_raw = torch.nn.functional.linear(x_flat, out_gate_w)   # (B*N*N, H)

    # 4️⃣ Reshape to [B,N,N,H]
    left_proj = left_proj_raw.view(bs, N, N, hidden_dim)
    right_proj = right_proj_raw.view(bs, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    out_gate = torch.sigmoid(out_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)

    # optional mask apply before gating
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # 5️⃣ Gated left/right
    left = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # 6️⃣ Triangle multiplication (batched GEMM)
    # make batch dimension B*H; each GEMM (N,N) @ (N,N)^T
    # reshape to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)   # (B*H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # GEMM
    out_flat = torch.bmm(left_perm, right_perm.transpose(1, 2))   # (B*H,N,N) fp16
    # reshape back to (B,N,N,H)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B,N,N,H) fp16

    # 7️⃣ Output-side LayerNorm over hidden dimension (fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )   # (B,N,N,H) fp16

    # 8️⃣ Element‑wise multiplication with out_gate via Triton (still fp16)
    # flatten
    total = out_norm.numel()
    out_norm_flat = out_norm.view(-1)
    out_gate_flat = out_gate.view(-1)
    gated_flat = torch.empty_like(out_norm_flat)
    BLOCK_SIZE = 65536  # 64k
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    elementwise_mul_fp16_kernel[grid](
        out_norm_flat, out_gate_flat, gated_flat, total,
        BLOCK_SIZE=BLOCK_SIZE
    )
    gated = gated_flat.view(bs, N, N, hidden_dim)   # fp16

    # 9️⃣ Final linear projection back to dim (fp16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)   # (dim, H)
    output = torch.nn.functional.linear(gated, to_out_w)   # (B,N,N,dim) fp16

    # Cast to float32 as required by API
    return output.to(torch.float32)

"""
custom_kernel - Outgoing Triangle‑Multiplicative Update (TriMul)

The kernel implements the forward pass of the TriMul block from AlphaFold‑3
using a highly‑fused FP16 pipeline so that the bulk of the work runs on
Tensor‑cores of an H100.  The steps are:

1️⃣  Layer‑norm over the input channel dimension (dim) – FP16.
2️⃣  Fuse projection + gate linear layers for the *left* and *right*
    tensors by concatenating their weights and performing a single GEMM.
3️⃣  Compute the output‑gate linear projection.
4️⃣  (Optional) apply the pair‑wise mask to the left/right projections.
5️⃣  Apply sigmoid gates to the left/right projections.
6️⃣  **Core triangle multiplication** – for every hidden channel h,
    compute `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.
    This is realized as a batched FP16 GEMM (`torch.bmm`) over the
    (B·hidden) batch dimension.
7️⃣  Layer‑norm over the hidden dimension (hidden_dim) – FP16.
8️⃣  Element‑wise product `out_norm * out_gate` using a tiny Triton
    kernel (still FP16, no extra copy).
9️⃣  Final linear projection back to the original dimensionality (dim) –
    FP16, then cast the result to FP32 as required by the API.

All heavy arithmetic stays in FP16 to exploit Tensor‑cores; only the
final cast to FP32 is performed at the end.  The Triton kernel is only
used for the element‑wise multiplication (step 8) – the rest is handled
by highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (FP16) used for out_norm * out_gate
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_fp16_kernel(
    a_ptr,          # float16* (out_norm flattened)
    b_ptr,          # float16* (out_gate flattened)
    out_ptr,        # float16* (output flattened)
    size,           # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Outgoing Triangle‑Multiplicative Update (TriMul) forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N]   (bool/float)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict containing at least ['dim', 'hidden_dim', 'nomask']

    Returns
    -------
    torch.Tensor
        The processed tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣  Input LayerNorm (FP16)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias   = weights["norm.bias"].to(device).to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)                     # [B, N, N, dim]

    # -------------------------------------------------
    # Flatten for the first linear layers
    # -------------------------------------------------
    x_flat = x.view(-1, dim)                # (B·N·N, dim)

    # -------------------------------------------------
    # 2️⃣  Fused projection + gate for *left*
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2·H, dim)
    left_concat = torch.nn.functional.linear(x_flat, left_concat_w)               # (B·N·N, 2·H)
    left_proj_raw = left_concat[:, :hidden_dim]          # (B·N·N, H)
    left_gate_raw = left_concat[:, hidden_dim:]          # (B·N·N, H)

    # -------------------------------------------------
    # 2️⃣  Fused projection + gate for *right*
    # -------------------------------------------------
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2·H, dim)
    right_concat = torch.nn.functional.linear(x_flat, right_concat_w)
    right_proj_raw = right_concat[:, :hidden_dim]
    right_gate_raw = right_concat[:, hidden_dim:]

    # -------------------------------------------------
    # 3️⃣  Output gate linear (no bias)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_raw = torch.nn.functional.linear(x_flat, out_gate_w)    # (B·N·N, H)

    # -------------------------------------------------
    # 4️⃣  Reshape to [B, N, N, H] and apply optional mask
    # -------------------------------------------------
    left_proj = left_proj_raw.view(bs, N, N, hidden_dim)   # fp16
    right_proj = right_proj_raw.view(bs, N, N, hidden_dim) # fp16

    if not nomask:
        # Convert mask to FP16 and broadcast over the hidden dimension
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣  Sigmoid gates (FP16) and element‑wise gating
    # -------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)

    left  = left_proj * left_gate      # (B, N, N, H)  FP16
    right = right_proj * right_gate    # (B, N, N, H)  FP16

    # -------------------------------------------------
    # 6️⃣  Core triangle multiplication: batched GEMM in FP16
    # -------------------------------------------------
    # bring hidden dimension into the batch axis -> (B·H, N, N)
    left_batched  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # each batch: (N×N) @ (N×N)^T  -> (N×N)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))   # (B·H, N, N) FP16

    # reshape back to (B, N, N, H)
    out = out_batched.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)       # (B, N, N, H) FP16

    # -------------------------------------------------
    # 7️⃣  Output‑side LayerNorm over hidden dimension (FP16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )   # (B, N, N, H) FP16

    # -------------------------------------------------
    # 8️⃣  Element‑wise multiplication with out_gate via Triton (FP16)
    # -------------------------------------------------
    # Flatten tensors
    total = out_norm.numel()
    out_norm_flat = out_norm.contiguous().view(-1)          # FP16 1‑D
    out_gate_flat = out_gate.contiguous().view(-1)          # FP16 1‑D
    gated_flat = torch.empty_like(out_norm_flat)           # FP16 1‑D

    # Triton kernel launch (only on CUDA)
    if device.type == "cuda":
        BLOCK_SIZE = 65536  # 64 K elements per program
        grid = (triton.cdiv(total, BLOCK_SIZE),)
        elementwise_mul_fp16_kernel[grid](
            out_norm_flat,
            out_gate_flat,
            gated_flat,
            total,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # CPU fallback – pure PyTorch (still FP16)
        gated_flat = out_norm_flat * out_gate_flat

    gated = gated_flat.view(bs, N, N, hidden_dim)          # (B, N, N, H) FP16

    # -------------------------------------------------
    # 9️⃣  Final linear projection back to `dim` (FP16) and cast to FP32
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)   # (dim, H)
    output_fp16 = torch.nn.functional.linear(gated, to_out_w)          # (B, N, N, dim) FP16

    # Return FP32 as expected by the surrounding model
    return output_fp16.to(torch.float32)
