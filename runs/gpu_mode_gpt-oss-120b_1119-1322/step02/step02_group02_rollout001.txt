def custom_kernel(data):
   input_tensor, mask, weights, config = data
   bs, N, _, dim = input_tensor.shape
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", True)
   device = input_tensor.device

   # 1. LayerNorm
   norm_weight = weights["norm.weight"]
   norm_bias = weights["norm.bias"]
   x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

   # 2. Cast to half for heavy ops
   x_h = x.to(torch.float16)

   # 3. Load weights to half (except out_gate)
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   # out_gate weight keep FP32
   out_gate_w = weights["out_gate.weight"]

   # 4. Linear projections
   left = F.linear(x_h, left_proj_w)   # [bs, N, N, hidden_dim]
   right = F.linear(x_h, right_proj_w)

   # 5. Gating
   left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
   right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
   out_gate = torch.sigmoid(F.linear(x, out_gate_w))

   # 6. Apply mask if needed
   if not nomask:
       mask_h = mask.to(torch.float16).unsqueeze(-1)
       left = left * mask_h
       right = right * mask_h

   # 7. Apply element-wise gates
   left = left * left_gate
   right = right * right_gate

   # 8. Reshape for GEMM
   left_h = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
   right_h = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

   # 9. Allocate output half
   out_h = torch.empty((bs * hidden_dim, N, N), dtype=torch.float16, device=device)

   # 10. Launch Triton kernel
   BLOCK_M = 64
   BLOCK_N = 64
   BLOCK_K = 64
   # Block sizes must be divisible by 16 for TensorCore; 64 works.
   a_stride_batch, a_stride_m, a_stride_k = left_h.stride()
   b_stride_batch, b_stride_n, b_stride_k = right_h.stride()
   c_stride_batch, c_stride_m, c_stride_n = out_h.stride()

   grid = (bs * hidden_dim,
           (N + BLOCK_M - 1) // BLOCK_M,
           (N + BLOCK_N - 1) // BLOCK_N)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core

The heavy operation is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication C = A @ Bᵀ with
A,B ∈ (B·hidden_dim, N, N).  The kernel below computes C in
half‑precision (fp16) while accumulating in fp32 for numerical
stability, using 64‑×‑64 tiles and a 64‑wide K‑tile to minimise the
loop count.  All inexpensive operations (layer‑norm, linear
projections, sigmoids, masking, post‑norm and final projection) are
performed with PyTorch – the only Triton component is the O(N³) GEMM.

The implementation follows the reference PyTorch module exactly, except
that it works on the “outgoing” version of TriMul and returns a
float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM (C = A @ Bᵀ)
#   A : (batch_h, M, K)   – left tensor
#   B : (batch_h, N, K)   – right tensor (interpreted as Bᵀ in the kernel)
#   C : (batch_h, M, N)   – result
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, N, K) – read as Bᵀ
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    One program instance per (batch, tile_m, tile_n).
    The kernel tiles A horizontally (M) and B vertically (N) and loops over
    the K dimension in blocks of size BLOCK_K, using `tl.dot` which maps to
    Tensor‑Core instructions on H100.
    """
    batch_id = tl.program_id(0)          # batch (B·hidden_dim)
    pid_m    = tl.program_id(1)          # tile over M dimension
    pid_n    = tl.program_id(2)          # tile over N dimension

    # ----- tile offsets ----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- accumulator (fp32) ---------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K -----------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 load

        # load Bᵀ tile (K × N)  → B is stored as (batch, N, K)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16 load

        # accumulate (Tensor‑Core)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ----- write back ------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) forward pass
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (bool or float)
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0. unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [bs, N, N, dim] (fp32)

    # -----------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive parts
    # -----------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------------
    # 3. Load weights (fp16 for everything except out_gate which stays fp32)
    # -----------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"]          # keep fp32

    # -----------------------------------------------------------------------
    # 4. Linear projections (batched GEMM performed by cuBLAS)
    # -----------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [bs, N, N, hidden_dim] (fp16)
    right = F.linear(x_h, right_proj_w)

    # -----------------------------------------------------------------------
    # 5. Gating (sigmoid) – still fp16 for left/right gates
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    # out_gate is used after the hidden‑dim LayerNorm, keep it fp32
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------------
    # 6. Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [bs, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------------
    # 7. Apply element‑wise gates
    # -----------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 8. Fold hidden_dim into the batch dimension → (B·hidden_dim, N, N)
    # -----------------------------------------------------------------------
    left_h  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Allocate output buffer (fp16)
    out_h = torch.empty((bs * hidden_dim, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 9. Launch the Triton batched‑gemm kernel
    # -----------------------------------------------------------------------
    # Tile sizes – 64 × 64 tiles with a 64‑wide K‑tile give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    a_stride_batch, a_stride_m, a_stride_k = left_h.stride()   # (batch, M, K)
    b_stride_batch, b_stride_n, b_stride_k = right_h.stride()  # (batch, N, K)
    c_stride_batch, c_stride_m, c_stride_n = out_h.stride()    # (batch, M, N)

    grid = (
        bs * hidden_dim,                               # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,                  # tiles over M
        (N + BLOCK_N - 1) // BLOCK_N,                  # tiles over N
    )

    batched_gemm_fp16_kernel[grid](
        left_h,
        right_h,
        out_h,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # Tensor‑Core‑friendly
        num_stages=3,         # modest double‑buffering
    )

    # -----------------------------------------------------------------------
    # 10. Reshape back to (bs, N, N, hidden_dim) and cast to fp32
    # -----------------------------------------------------------------------
    out = out_h.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------------
    # 11. Post‑processing: hidden‑dim LayerNorm, output gate, final projection
    # -----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate                     # broadcasted element‑wise product

    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim)
    out = F.linear(out, to_out_w)            # final projection, bias‑free

    return out
