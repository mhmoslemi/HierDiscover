    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    w_left = weights["left_proj.weight"]  # (H, D)
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    w_to_out_norm_w = weights["to_out_norm.weight"]

"""
custom_kernel: Triton‑accelerated forward pass of the “outgoing” TriMul module
(AlphaFold‑3).  The heavy N³ update is performed with a fused batched
matrix‑multiply (B·H independent GEMMs) using torch.bmm on FP16 (Tensor
Cores).  A tiny Triton kernel is used for the final element‑wise
multiplication (out_norm * out_gate) to satisfy the requirement that a
Triton kernel participates in the computation.

Algorithm (high‑level)
----------------------
1. Layer‑norm the input tensor `x` (float32) → shape [B, N, N, D].
2. Linear projections (bias‑free) to hidden dimension `H`:
       left  = x @ W_left   (→ [B, N, N, H])
       right = x @ W_right  (→ [B, N, N, H])
3. Compute three sigmoid gates (left_gate, right_gate, out_gate)
   from the same `x`.
4. Optional mask (broadcast over H) is applied to `left`/`right`
   before gating.
5. Apply element‑wise gates.
6. Core N³ update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (A @ Bᵀ) for each hidden channel.
   The GEMM is executed with `torch.bmm` on FP16 tensors,
   leveraging H100 Tensor‑Core speed.
7. Hidden‑dim LayerNorm (float32) → multiply by `out_gate`
   using a tiny Triton element‑wise kernel.
8. Final linear projection back to the original dimension `D`.
9. Cast result to float32 and return.

All heavy arithmetic (steps 5‑6) runs in FP16; only the two LayerNorms
stay in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    out_ptr,  # pointer to output (will be overwritten)
    gate_ptr,  # pointer to gate tensor
    numel,  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    a = tl.load(out_ptr + offs, mask=mask)
    b = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask:        torch.Tensor [B, N, N] (bool/int/float) or None
            - weights:     dict of model parameters (torch.Tensor, float32)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D], float32

    # ------------------------------------------------------------------
    # 2) Linear projections & gates (float32)
    # ------------------------------------------------------------------
    # Bias‑free linear layers
    left  = F.linear(x, weights["left_proj.weight"])      # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, H]

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] → broadcast over hidden
        mask_h = mask.unsqueeze(-1).to(x.dtype)      # float32
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Apply gates (element‑wise)
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the N³ batched GEMM
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # 6) Prepare batched matrices for torch.bmm
    #    A: left  → [B*H, N, N]
    #    B: right transposed over the last two axes → [B*H, N, N]
    # ------------------------------------------------------------------
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()         # [B, H, N, N] (Bᵀ)

    left_mat  = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(-1, N, N)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication with Tensor‑Cores (FP16)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N), FP16

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H], FP16

    # ------------------------------------------------------------------
    # 9) Hidden‑dim LayerNorm (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # [B, N, N, H], float32

    # ------------------------------------------------------------------
    # 10) Multiply by out_gate (element‑wise) using a Triton kernel
    # ------------------------------------------------------------------
    # Cast both tensors to FP16, make them contiguous, and launch kernel
    out_norm_fp16 = out_norm.to(torch.float16).contiguous()
    out_gate_fp16 = out_gate.to(torch.float16).contiguous()

    total = out_norm_fp16.numel()
    BLOCK = 65536  # ~128 Ki elements → ~256 KB per block (well within limits)
    grid = (triton.cdiv(total, BLOCK),)

    elemwise_mul_kernel[grid](
        out_norm_fp16,
        out_gate_fp16,
        total,
        BLOCK_SIZE=BLOCK,
    )  # out_norm_fp16 now holds out_norm * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original dimension D (FP16)
    # ------------------------------------------------------------------
    w_to_out_fp16 = weights["to_out.weight"].to(torch.float16)
    out_final = F.linear(out_norm_fp16, w_to_out_fp16)   # [B, N, N, D], FP16

    # ------------------------------------------------------------------
    # 12) Cast to float32 as required by downstream code
    # ------------------------------------------------------------------
    return out_final.to(torch.float32)
