"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The forward pass consists of:
  1️⃣ Layer‑norm on the 4‑D input (float32).
  2️⃣ FP16 linear projections (left/right) and three gating linear layers.
  3️⃣ A tiny Triton kernel that fuses the element‑wise multiplication
     ``proj * gate * mask`` for the *left* and *right* tensors.
  4️⃣ The cubic “tri‑multiplicative” update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
     This is exactly a batched GEMM (A @ Bᵀ) for each hidden channel.
     We use a single `torch.bmm` call (cuBLAS) – it is far faster than a
     hand‑written Triton GEMM on H100.
  5️⃣ Hidden‑dim LayerNorm, out‑gate multiplication and a final linear
     projection back to the original dimension.

Only the element‑wise “gate‑and‑mask” step is implemented in Triton,
fulfilling the requirement that *part* of the work lives in a kernel
while keeping the heavy ∼N³ GEMM in cuBLAS for maximal speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) for the left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def mul_gate_mask_kernel(
    proj_ptr,          # FP16 projected tensor (left or right)          [total]
    gate_ptr,          # FP16 gate tensor (same shape)                [total]
    mask_ptr,          # FP16 mask, shape [B*N*N] (broadcasted on H)  [mask_len]
    out_ptr,           # FP16 output tensor (same shape as proj_ptr)  [total]
    total_elems: tl.constexpr,   # number of elements in proj/gate tensors
    hidden_dim: tl.constexpr,    # size of the hidden dimension (H)
    BLOCK_SIZE: tl.constexpr,    # number of threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load projected value and gate
    proj = tl.load(proj_ptr + offs, mask=valid, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # Compute index into the (broadcasted) mask:
    # mask is stored as a flat array of size B*N*N,
    # each hidden element repeats the same mask entry.
    mask_idx = offs // hidden_dim
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of torch.Tensor parameters (float32)
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input Layer‑Norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Helper – cast a weight once to FP16
    def _to_half(t):
        return t.to(torch.float16)

    # Linear projections (bias‑free)
    left = F.linear(x_fp16, _to_half(weights["left_proj.weight"]))   # [B,N,N,H] fp16
    right = F.linear(x_fp16, _to_half(weights["right_proj.weight"]))

    # Gating linear layers + sigmoid (still fp16)
    left_gate = F.linear(x_fp16, _to_half(weights["left_gate.weight"])).sigmoid()
    right_gate = F.linear(x_fp16, _to_half(weights["right_gate.weight"])).sigmoid()
    out_gate = F.linear(x_fp16, _to_half(weights["out_gate.weight"])).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional mask – flatten once for the Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        mask_flat = None
    else:
        # mask: [B,N,N] -> FP16 flat array of length B*N*N
        mask_flat = mask.to(torch.float16).reshape(-1).contiguous()

    # ------------------------------------------------------------------
    # 4) Fuse (proj * gate * mask) with a small Triton kernel
    # ------------------------------------------------------------------
    def _apply_gate_and_mask(proj, gate):
        """Apply ``proj = proj * gate * mask`` (mask optional)."""
        if mask_flat is None:
            # No mask → plain element‑wise multiplication (still FP16)
            return proj * gate
        total = proj.numel()
        out = torch.empty_like(proj)

        BLOCK_SIZE = 4096               # 4 K threads per program – a good trade‑off
        grid = (triton.cdiv(total, BLOCK_SIZE),)

        mul_gate_mask_kernel[grid](
            proj,
            gate,
            mask_flat,
            out,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        return out

    left = _apply_gate_and_mask(left, left_gate)   # [B,N,N,H] fp16
    right = _apply_gate_and_mask(right, right_gate)

    # ------------------------------------------------------------------
    # 5) Core N³ update – batched GEMM using cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # B, N, N, hidden_dim
    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_t.view(-1, N, N)           # [B*H, N, N]  fp16
    right_flat = right_t.view(-1, N, N)         # [B*H, N, N]  fp16

    # Batched matrix multiply: (left) @ (right)ᵀ   →  (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6) Hidden‑dim Layer‑Norm (float32) + out‑gate multiplication
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Multiply by the (sigmoid) out‑gate
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dimension D (float32)
    # ------------------------------------------------------------------
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
```
