import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: simple element‑wise multiplication (float32)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,            # float32* (first operand)
    b_ptr,            # float32* (second operand)
    out_ptr,          # float32* (output)
    size,             # total number of elements to process
    BLOCK_SIZE: tl.constexpr,
):
    """
    Multiply two 1‑D float32 buffers element‑wise.
    """
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold‑3) – highly‑optimized forward pass.

    Steps
    -----
    1. LayerNorm over the input feature dimension (dim) – FP32.
    2. Fuse the three linear projections (left_proj+gate, right_proj+gate,
       out_gate) into a single half‑precision GEMM, then split and apply a
       sigmoid to the gate portions.
    3. Apply optional mask and element‑wise gating (mask * left_gate,
       mask * right_gate) in one fused multiply per side.
    4. Core triangle multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
       Implemented as a batched half‑precision GEMM:
       (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N).
    5. LayerNorm on the hidden dimension (float32).
    6. Element‑wise product with the out‑gate using a tiny Triton kernel.
    7. Final half‑precision linear projection back to the original channel
       dimension; result cast to float32 before return.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to half for the heavy compute
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 3. Fused linear projections (left, right, out_gate) + sigmoid gates
    # ------------------------------------------------------------------
    # Gather all five weight matrices (no bias) and concatenate them.
    # Order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16) # (H, C)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16) # (H, C)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)   # (H, C)

    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # (5*H, C)

    # Single half‑precision GEMM: (B·N·N, C) @ (C, 5·H)ᵀ  →  (B·N·N, 5·H)
    combined = torch.nn.functional.linear(x_h, combined_weight)  # [B,N,N,5*H]

    # ------------------------------------------------------------------
    # 4. Slice the result and apply sigmoid to the gate halves
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = combined[..., 0:H]
    left_gate   = torch.sigmoid(combined[..., H:2*H])
    right_proj  = combined[..., 2*H:3*H]
    right_gate  = torch.sigmoid(combined[..., 3*H:4*H])
    out_gate    = torch.sigmoid(combined[..., 4*H:5*H])   # shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 5. Mask + left/right gating (fused multiply)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * (left_gate * mask_h)
        right = right_proj * (right_gate * mask_h)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Triangle multiplication (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # After the gating we have tensors of shape (B,N,N,H).  We need to
    # perform, for each head h, a matrix multiplication over the k dimension.
    # We reshape to (B*H, N, N) and use torch.bmm.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(bs * hidden_dim, N, N)          # (B·H, N, N)
    right_flat = right_perm.view(bs * hidden_dim, N, N).transpose(1, 2)  # (B·H, N, N)

    out_flat = torch.bmm(left_flat, right_flat)               # (B·H, N, N)

    # Restore the original layout (B,N,N,H)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 7. Output‑side LayerNorm (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # [B,N,N,H] float32

    # ------------------------------------------------------------------
    # 8. Element‑wise product with out_gate (Triton)
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)
    total_elems = out_norm.numel()
    BLOCK_SIZE = 65536  # tuneable; 64 K elements per program

    gated = torch.empty_like(out_norm)
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        out_norm.view(-1),
        out_gate_fp32.view(-1),
        gated.view(-1),
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # gated now contains out_norm * out_gate   (float32)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim` (half‑precision GEMM)
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    gated_h = gated.to(torch.float16)                           # (B,N,N,H)

    output_h = torch.nn.functional.linear(gated_h, to_out_weight)  # (B,N,N,dim) half

    # ------------------------------------------------------------------
    # 10. Cast back to float32 for the API contract
    # ------------------------------------------------------------------
    return output_h.to(torch.float32)
