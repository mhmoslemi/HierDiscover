```
@triton.jit
def proj_gate_kernel(
    x_ptr, mask_ptr,
    proj_w_ptr, gate_w_ptr,
    out_ptr,
    B, N, dim, hidden,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_batch, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = batch_hidden // hidden
    h = batch_hidden % hidden

    i0 = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i0 < N
    mask_j = j0 < N

    # offsets for x: need to read x at (b,i,j,dim)
    x_ptrs = x_ptr + b * stride_xb + i0[:, None] * stride_xi + j0[None, :] * stride_xj
    # shape (BLOCK_M, BLOCK_N) each will be used within loop over k
    # but we need to iterate over k dimension reading scalar each time

    # init accumulators
    acc_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(dim):
        # load weight for this hidden h
        w_proj = tl.load(proj_w_ptr + h * dim + k)  # scalar
        w_gate = tl.load(gate_w_ptr + h * dim + k)  # scalar

        # load x for this k
        x_k = tl.load(x_ptrs + k * stride_xd, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

        acc_proj += x_k * w_proj
        acc_gate += x_k * w_gate

    # apply sigmoid to gate
    gate = 1.0 / (1.0 + tl.exp(-acc_gate))
    out = acc_proj * gate

    # apply mask if present
    if mask_ptr is not None:
        mask_val = tl.load(mask_ptr + b * stride_mask_b + i0[:, None] * stride_mask_i + j0[None, :] * stride_mask_j, mask=mask_i[:, None] & mask_j[None, :], other=1.0)  # assume mask is 1 for keep, 0 for drop
        out = out * mask_val

    # write output to out_ptr
    out_ptrs = out_ptr + b * stride_out_batch + h * stride_out_h + i0[:, None] * stride_out_i + j0[None, :] * stride_out_j
    tl.store(out_ptrs, out.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

```
@triton.jit
def proj_gate_kernel(
    x_ptr,     # [B,N,N,dim] FP16
    proj_w_ptr, # [hidden, dim] FP16
    gate_w_ptr, # [hidden, dim] FP16
    out_ptr,   # [B, hidden, N, N] FP16
    B, N, dim, hidden,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = batch_hidden // hidden
    h = batch_hidden % hidden

    # offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulators
    acc_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over dim
    for d in range(dim):
        # load weights (scalar) for current hidden channel
        w_proj = tl.load(proj_w_ptr + h * dim + d, dtype=tl.float16).to(tl.float32)
        w_gate = tl.load(gate_w_ptr + h * dim + d, dtype=tl.float16).to(tl.float32)

        # compute pointer to x for this dim slice
        x_ptrs = x_ptr + b * stride_xb + offs_i[:, None] * stride_xi + offs_j[None, :] * stride_xj + d * stride_xd
        x = tl.load(x_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16).to(tl.float32)

        acc_proj += x * w_proj
        acc_gate += x * w_gate

    # apply sigmoid to gate
    gate = 1.0 / (1.0 + tl.exp(-acc_gate))

    out = acc_proj * gate

    # store out to out_ptr
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, out.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def proj_gate_kernel(
    x_ptr, proj_w_ptr, gate_w_ptr, out_ptr,
    B, N, dim, hidden,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = batch_hidden // hidden
    h = batch_hidden % hidden

    # offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulators in fp32
    acc_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for d in range(dim):
        # weight scalars
        w_proj = tl.load(proj_w_ptr + h * dim + d, dtype=tl.float16).to(tl.float32)
        w_gate = tl.load(gate_w_ptr + h * dim + d, dtype=tl.float16).to(tl.float32)

        # load x slice
        x_ptrs = x_ptr + b * stride_xb + offs_i[:, None] * stride_xi + offs_j[None, :] * stride_xj + d * stride_xd
        x = tl.load(x_ptrs,
                    mask=mask_i[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        acc_proj += x * w_proj
        acc_gate += x * w_gate

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-acc_gate))

    out = acc_proj * gate

    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             out.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
```

```
    x_h = x.to(torch.float16)

    left = torch.empty((bs, hidden, N, N), dtype=torch.float16, device=device)
    right = torch.empty((bs, hidden, N, N), dtype=torch.float16, device=device)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    BLOCK_N = 32

    stride_xb, stride_xi, stride_xj, stride_xd = x_h.stride()
    # Strides for left/right output
    stride_left_b, stride_left_h, stride_left_i, stride_left_j = left.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_j = right.stride()

    # Compute left
    grid = (

        (N + BLOCK_M - 1) // BLOCK_M,

    proj_gate_kernel[grid](

        left_proj_w,
        left_gate_w,
        left,
        B=bs,

        dim=dim,

        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_out_b=stride_left_b,
        stride_out_h=stride_left_h,
        stride_out_i=stride_left_i,
        stride_out_j=stride_left_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )

    proj_gate_kernel[grid](
        x_h,
        right_proj_w,
        right_gate_w,
        right,
        B=bs,
        N=N,
        dim=dim,
        hidden=hidden,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_out_b=stride_right_b,
        stride_out_h=stride_right_h,
        stride_out_i=stride_right_i,
        stride_out_j=stride_right_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )

    # Apply mask if needed
    if not nomask:
        # mask shape [bs, N, N] -> [bs, 1, N, N] broadcast over hidden
        mask_h = mask.to(torch.float16).unsqueeze(1)
        left = left * mask_h
        right = right * mask_h

    # Reshape for batch matrix multiplication
    left_h = left.view(bs * hidden, N, N)
    right_h = right.view(bs * hidden, N, N)

    # Perform batched GEMM using torch.bmm (fast cuBLAS)
    out_h = torch.bmm(left_h, right_h.transpose(1, 2))  # (bs*hidden, N, N) fp16

    # reshape back to (bs, N, N, hidden)
    out = out_h.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # Convert to FP32 for rest of pipeline
    out = out.to(torch.float32)

    # out_gate (gate on output channels)
    # Compute gating from original normalized x (FP32)
    out_gate_raw = F.linear(x, out_gate_w)  # [bs,N,N,hidden] fp32
    out_gate = torch.sigmoid(out_gate_raw)

    # Hidden-dim layernorm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)
    # Apply output gate
    out = out * out_gate

    # Final projection back to original dim
    to_out_w = weights["to_out.weight"]  # (dim, hidden)
    out = F.linear(out, to_out_w)  # shape (bs,N,N,dim)

    return out
```

```
"""
TriMul (outgoing) implementation with fused projection/gate kernels and cuBLAS batched GEMM.
- Normalizes input.
- Projects and gates the tensor with a fused Triton kernel that directly computes:
    left = (x·W_left_projᵀ) ⊙ sigmoid(x·W_left_gateᵀ) (and same for right).
- Optional boolean mask is applied after gating.
- The heavy N³ interaction is performed with a single torch.bmm (cuBLAS) which on H100
  exploits Tensor‑Cores.
- Post‑processing consists of a hidden‑dim LayerNorm, an output gate and the final
  linear projection back to the original channel dimension.
All heavy compute is done in FP16, accumulation in FP32, and final output in FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: fused projection + sigmoid gate (no mask)
#   out[b, h, i, j] = sum_k x[b,i,j,k] * w_proj[h,k] * sigmoid( sum_k x[b,i,j,k] * w_gate[h,k] )
# ---------------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr, proj_w_ptr, gate_w_ptr, out_ptr,
    B, N,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    DIM: tl.constexpr, HIDDEN: tl.constexpr
):
    """
    Computes projection and gated projection in one pass.
    Parameters
    ----------
    x_ptr : pointer to input tensor [B,N,N,D] (fp16)
    proj_w_ptr, gate_w_ptr : pointers to weight matrices [H, D] (fp16)
    out_ptr : pointer to output tensor [B,H,N,N] (fp16)
    """
    batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch and hidden indices
    b = batch_hidden // HIDDEN
    h = batch_hidden % HIDDEN

    # Tile offsets for the spatial dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds checking
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulators (FP32 for precision)
    acc_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the input channel dimension (DIM)
    for d in range(DIM):
        # Load scalar weights for this hidden channel and d‑th input feature
        w_proj = tl.load(proj_w_ptr + h * DIM + d, dtype=tl.float16).to(tl.float32)
        w_gate = tl.load(gate_w_ptr + h * DIM + d, dtype=tl.float16).to(tl.float32)

        # Load the slice x[b,i,j,d] for the whole tile
        x_ptrs = (x_ptr
                  + b * stride_xb
                  + offs_i[:, None] * stride_xi
                  + offs_j[None, :] * stride_xj
                  + d * stride_xd)
        x = tl.load(x_ptrs,
                    mask=mask_i[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        acc_proj += x * w_proj
        acc_gate += x * w_gate

    # Sigmoid gate (computed in FP32)
    gate = 1.0 / (1.0 + tl.exp(-acc_gate))

    # Final gated product
    out = acc_proj * gate

    # Store result
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs,
             out.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    stride_xb, stride_xi, stride_xj, stride_xd = x_h.stride()
    # strides of the output tensors (B,hidden,N,N)
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = left.stride()

    # Triton launch configuration – product BLOCK_M*BLOCK_N ≤ 1024
    BLOCK_M = 32
    BLOCK_N = 32

    grid = (
        bs * hidden,                     # one program per (batch, hidden) pair
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # left side
    proj_gate_kernel[grid](
        x_ptr=x_h,
        proj_w_ptr=left_proj_w,
        gate_w_ptr=left_gate_w,
        out_ptr=left,
        B=bs,
        N=N,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        DIM=dim,
        HIDDEN=hidden,
        num_warps=8,
    )

    # right side (same launch grid, different weights)
    proj_gate_kernel[grid](
        x_ptr=x_h,
        proj_w_ptr=right_proj_w,
        gate_w_ptr=right_gate_w,
        out_ptr=right,
        B=bs,
        N=N,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        DIM=dim,
        HIDDEN=hidden,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 3. Optional masking (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(1)   # (bs,1,N,N)
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4. Batched matrix multiplication: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented with torch.bmm (cuBLAS) which on H100 reaches Tensor‑Core peak.
    # --------------------------------------------------------------
    # reshape to (bs*hidden, N, N) – contiguous layout needed by torch.bmm
    left_h  = left.view(bs * hidden, N, N)
    right_h = right.view(bs * hidden, N, N)

    # torch.bmm operates on (B, M, K) @ (B, K, N)
    out_h = torch.bmm(left_h, right_h.transpose(1, 2))   # FP16 result, shape (bs*hidden,N,N)

    # reshape back to (bs,N,N,hidden) and convert to FP32 for the remaining ops
    out = out_h.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 5. Output gate (computed on the original FP32 tensor)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # (bs,N,N,hidden) FP32

    # --------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, gating, and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # final projection back to `dim` (FP32)

    return out

"""
TriMul (outgoing) implementation with fused projection/gate Triton kernels and
cuBLAS‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm on the last dimension (FP32).
2. Cast to FP16 and use a custom Triton kernel that, for each hidden channel,
   simultaneously computes the linear projection and its sigmoid gate:

       left  = (x·W_left_projᵀ) ⊙ sigmoid(x·W_left_gateᵀ)
       right = (x·W_right_projᵀ) ⊙ sigmoid(x·W_right_gateᵀ)

   The kernel works on tiles of the spatial dimensions (i,j) to keep the
   thread‑block size ≤1024 threads (BLOCK_M·BLOCK_N ≤ 1024) and accumulates
   in FP32 for numerical stability.
3. If a mask is supplied, broadcast it over the hidden channel and apply it
   after gating (still in FP16).
4. The heavy N³ interaction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed with `torch.bmm`, i.e. a cuBLAS‑backed batched matrix‑multiply
   that fully exploits Tensor‑Cores on H100.
5. Post‑process the result with a hidden‑dim LayerNorm, an output gate
   (computed on the original FP32 tensor), and a final linear projection back
   to the original channel dimension.
All intensive arithmetic is in FP16; reductions are accumulated in FP32,
and the final output is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: fused projection + sigmoid gate (mask applied later)
# ---------------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr, proj_w_ptr, gate_w_ptr, out_ptr,
    B, N,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    DIM: tl.constexpr, HIDDEN: tl.constexpr
):
    """
    Computes gated projection for a tile of the (i,j) plane.
    Arguments
    ---------
    x_ptr           : pointer to input tensor    [B,N,N,D] (fp16)
    proj_w_ptr      : pointer to weight matrix    [H,D]   (fp16) – projection
    gate_w_ptr      : pointer to weight matrix    [H,D]   (fp16) – gate
    out_ptr         : pointer to output tensor    [B,H,N,N] (fp16)
    B, N            : batch size, spatial size
    stride_*        : strides of the corresponding tensors
    BLOCK_M, BLOCK_N: tile sizes (must satisfy BLOCK_M*BLOCK_N ≤ 1024)
    DIM, HIDDEN     : compile‑time constants for the channel dimensions
    """
    batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch index and hidden‑channel index
    b = batch_hidden // HIDDEN
    h = batch_hidden % HIDDEN

    # Tile offsets (i‑ and j‑coordinates)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulators in FP32
    acc_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the input feature dimension (DIM)
    for d in range(DIM):
        # Load the scalar weights for the current hidden channel and d‑th input feature
        w_proj = tl.load(proj_w_ptr + h * DIM + d, dtype=tl.float16).to(tl.float32)
        w_gate = tl.load(gate_w_ptr + h * DIM + d, dtype=tl.float16).to(tl.float32)

        # Load x[b,i,j,d] for the whole tile
        x_ptrs = (x_ptr
                  + b * stride_xb
                  + offs_i[:, None] * stride_xi
                  + offs_j[None, :] * stride_xj
                  + d * stride_xd)
        x = tl.load(x_ptrs,
                    mask=mask_i[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        acc_proj += x * w_proj
        acc_gate += x * w_gate

    # Sigmoid on the gate accumulator (FP32)
    gate = 1.0 / (1.0 + tl.exp(-acc_gate))

    # Final gated output
    out = acc_proj * gate

    # Store the result (cast back to FP16)
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs,
             out.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ---------------------------------------------------------------------------
# Entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N]  (bool/float) – ignored when nomask=True
        - weights      : dict containing the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` (optionally ``nomask``)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0. Unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (FP32)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [bs, N, N, dim]  FP32

    # -----------------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑heavy part
    # -----------------------------------------------------------------------
    x_h = x.to(torch.float16)  # [bs, N, N, dim]  FP16

    # -----------------------------------------------------------------------
    # 3. Fused projection + gating for left/right tensors (FP16)
    # -----------------------------------------------------------------------
    left  = torch.empty((bs, hidden, N, N), dtype=torch.float16, device=device)
    right = torch.empty((bs, hidden, N, N), dtype=torch.float16, device=device)

    # weights (FP16)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # strides of the input tensor (B,N,N,dim)
    stride_xb, stride_xi, stride_xj, stride_xd = x_h.stride()
    # strides of the output tensors (B,hidden,N,N)
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = left.stride()

    # Triton launch configuration (BLOCK_M * BLOCK_N ≤ 1024)
    BLOCK_M = 32
    BLOCK_N = 32

    grid = (
        bs * hidden,                     # one program per (batch, hidden) pair
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # left side
    proj_gate_kernel[grid](
        x_ptr=x_h,
        proj_w_ptr=left_proj_w,
        gate_w_ptr=left_gate_w,
        out_ptr=left,
        B=bs,
        N=N,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        DIM=dim,
        HIDDEN=hidden,
        num_warps=8,
    )

    # right side (same launch grid, different weights)
    proj_gate_kernel[grid](
        x_ptr=x_h,
        proj_w_ptr=right_proj_w,
        gate_w_ptr=right_gate_w,
        out_ptr=right,
        B=bs,
        N=N,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        DIM=dim,
        HIDDEN=hidden,
        num_warps=8,
    )

    # -----------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(1)   # shape (bs,1,N,N)
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------------
    # 5. Batched N³ GEMM: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Performed with torch.bmm (cuBLAS) – on H100 this is Tensor‑Core bound.
    # -----------------------------------------------------------------------
    # reshape to (bs*hidden, N, N) for torch.bmm
    left_h  = left.view(bs * hidden, N, N)          # FP16
    right_h = right.view(bs * hidden, N, N)         # FP16

    out_h = torch.bmm(left_h, right_h.transpose(1, 2))   # FP16, shape (bs*hidden,N,N)

    # reshape back to (bs, N, N, hidden) and convert to FP32 for the rest
    out = out_h.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------------
    # 6. Output gate (computed on the FP32 tensor)
    # -----------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # (bs,N,N,hidden)

    # -----------------------------------------------------------------------
    # 7. Hidden‑dim LayerNorm, gating and final projection back to `dim`
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection (FP32)

    return out
```
