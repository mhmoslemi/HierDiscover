"""
Triton‐accelerated outgoing TriMul (Triangular Multiplicative Update).

Algorithm
---------
1️⃣ Layer‑norm over the input feature dimension.
2️⃣ Fuse the 5 linear maps (left_proj, left_gate, right_proj, right_gate, out_gate)
   into a single half‑precision matmul:
       X_flat (B·N²×dim) @ W_fused.T (dim×5·c_z) → (B·N²×5·c_z)
   The result is split into the 5 tensors; the three gate tensors are passed
   through a sigmoid.
3️⃣ Apply the optional binary mask and element‑wise fuse the gates with the
   projections (all in fp16, in‑place to minimise traffic).
4️⃣ Core cubic contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (A @ Bᵀ) for every head d.  A custom Triton kernel
   processes each (batch, head) pair as an independent N×N matmul, using
   fp16 Tensor‑Core multiplication with fp32 accumulation.
5️⃣ Second layer‑norm, out‑gate multiplication and final fp32 linear projection
   back to the original feature dimension.

All heavy arithmetic (step 4) runs on the GPU via Triton.  The remaining ops
are performed with PyTorch fused kernels in half‑precision for bandwidth
efficiency; the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ
    A : (batch, M, K)  – row‑major
    B : (batch, N, K)  – row‑major, accessed transposed as (K, N)
    C : (batch, M, N)
    All batches are independent.
    """
    batch_id = tl.program_id(0)      # batch = B * hidden_dim
    pid_m    = tl.program_id(1)      # tile‑row index
    pid_n    = tl.program_id(2)      # tile‑col index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile as transposed (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fp16 multiplication, fp32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write out result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype float32.
    """
    # --------------------------------------------------------------------- #
    # Unpack arguments – mask may be omitted (nomask case)
    # --------------------------------------------------------------------- #
    if len(data) == 4:
        inp, mask, weights, config = data
    else:  # fall back – no mask provided
        inp, weights, config = data
        mask = None

    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # often called c_z

    # --------------------------------------------------------------------- #
    # 1️⃣ LayerNorm over the last feature dimension (float32)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                          # [B, N, N, dim]

    B, N, _, _ = x.shape

    # --------------------------------------------------------------------- #
    # 2️⃣ Fuse the 5 linear maps into a single half‑precision matmul
    # --------------------------------------------------------------------- #
    x_flat = x.reshape(B * N * N, dim).contiguous().to(torch.float16)

    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )                                          # (5*hidden_dim, dim)
    fused_weight = fused_weight.to(torch.float16)

    fused_out = F.linear(x_flat, fused_weight)   # (B·N², 5·hidden_dim)  fp16

    H = hidden_dim
    left_proj_flat  = fused_out[:, 0:H]
    left_gate_flat  = fused_out[:, H:2 * H]
    right_proj_flat = fused_out[:, 2 * H:3 * H]
    right_gate_flat = fused_out[:, 3 * H:4 * H]
    out_gate_flat   = fused_out[:, 4 * H:5 * H]

    # Gate activations (still fp16)
    left_gate_flat  = torch.sigmoid(left_gate_flat)
    right_gate_flat = torch.sigmoid(right_gate_flat)
    out_gate_flat   = torch.sigmoid(out_gate_flat)

    # Reshape back to (B, N, N, H)
    left_proj = left_proj_flat.view(B, N, N, H)
    left_gate = left_gate_flat.view(B, N, N, H)
    right_proj = right_proj_flat.view(B, N, N, H)
    right_gate = right_gate_flat.view(B, N, N, H)
    out_gate = out_gate_flat.view(B, N, N, H)

    # --------------------------------------------------------------------- #
    # 3️⃣ Fuse mask (if any) and gate values with the projections – in‑place
    # --------------------------------------------------------------------- #
    # left = left_proj * left_gate   (in‑place to avoid extra buffers)
    left = left_proj
    left.mul_(left_gate)          # fp16
    right = right_proj
    right.mul_(right_gate)        # fp16

    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1).to(left.dtype)   # broadcast, fp16
        left.mul_(mask)
        right.mul_(mask)

    # --------------------------------------------------------------------- #
    # 4️⃣ Core cubic contraction via the batched GEMM kernel (A @ Bᵀ)
    # --------------------------------------------------------------------- #
    # Rearrange to (B, H, N, N) → (B·H, N, N) layout expected by the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)   # (B·H, N, N)  fp16
    right_mat = right_perm.view(B * H, N, N) # (B·H, N, N)  fp16

    # Output buffer – accumulate in float32 for numerical stability
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------------- #
    # Strides (batch‑major layout)
    # --------------------------------------------------------------------- #
    stride_am = left_mat.stride(1)   # row stride of A (M)
    stride_ak = left_mat.stride(2)   # col stride of A (K)
    stride_ab = left_mat.stride(0)   # batch stride of A

    stride_bk = right_mat.stride(2)  # K stride of B (since we load transposed)
    stride_bn = right_mat.stride(1)  # N stride of B (row)
    stride_bb = right_mat.stride(0)  # batch stride of B

    stride_cm = out_mat.stride(1)    # row stride of C (M)
    stride_cn = out_mat.stride(2)    # col stride of C (N)
    stride_cb = out_mat.stride(0)    # batch stride of C

    # --------------------------------------------------------------------- #
    # Kernel launch configuration (tuned for H100 TensorCores)
    # --------------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                   # batch dimension (B·H)
        (N + BLOCK_M - 1) // BLOCK_M,            # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,            # col tiles
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,          # double‑buffering for higher throughput
    )

    # --------------------------------------------------------------------- #
    # 5️⃣ Post‑processing: reshape, second LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------------- #
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # still float32

    # Apply out‑gate (cast to float32 to match `out`)
    out_gate = out_gate.to(out.dtype)
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
```
