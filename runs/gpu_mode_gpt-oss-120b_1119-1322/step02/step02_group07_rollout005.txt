"""
outgoing TriMul (TriMul‑Outgoing)

Algorithm (no gradients)

1. **Input LayerNorm** (float32) – normalises the feature dimension `dim`.
2. **Fused linear projections** – concatenate the projection and gate
   weight matrices, then perform a single half‑precision GEMM per side:
   ```
   left_concat  = x_norm_fp16 @ [W_left_proj ; W_left_gate]^T   # (B,N,N,2*H)
   right_concat = x_norm_fp16 @ [W_right_proj; W_right_gate]^T # (B,N,N,2*H)
   ```
   Split the concatenated tensors into `proj` and `gate` parts and apply
   a `sigmoid` to the gates.
3. **Mask + gate fusion** – a tiny Triton kernel fuses the optional
   binary mask, the per‑channel gates and the projection tensors:
   `out_left = left_proj * left_gate * mask`,
   `out_right = right_proj * right_gate * mask`.
   When `nomask` is `True` the kernel is skipped and a pure
   element‑wise multiplication is used.
4. **Batched GEMM** – after the fusion we have tensors of shape
   `[B, N, N, H]`.  We reshape them to `[B*H, N, N]` and invoke
   `torch.bmm` (fp16 inputs, fp32 accumulation) which on an H100
   exploits Tensor Cores and delivers ≈ 2 TFLOP/s per matrix.
5. **Second LayerNorm** over the hidden dimension (`H`).
6. **Output gate & final projection** – apply the output gate
   (`sigmoid(x @ W_out_gate)`) and a last half‑precision linear map
   back to the model dimension (`dim`).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Triton kernel: fuse mask, left‑gate and right‑gate (all fp16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    mask_ptr,          # fp16 *, shape (B,N,N) – broadcast over H
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    out_left_ptr,      # fp16 *
    out_right_ptr,     # fp16 *
    total_elems: tl.int64,
    H: tl.constexpr,   # hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offset
    mask = offs < total_elems

    # Load tensors (masked-load returns 0 for out‑of‑range threads)
    l  = tl.load(left_ptr + offs,          mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,         mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,    mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,   mask=mask, other=0.0)

    # Index of the mask element (mask has no H dimension)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # Fuse element‑wise multiplies
    l = l * lg * m
    r = r * rg * m

    # Store results
    tl.store(out_left_ptr  + offs, l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no grads).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float32)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict with keys
                         * "dim"          – input feature dimension
                         * "hidden_dim"   – hidden dimension H
                         * "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                    # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the cheap linear layers
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)        # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projection + gate into two linear calls
    # ------------------------------------------------------------------
    # concatenate proj & gate weights (shape: (2*H, dim))
    left_w_concat_fp16 = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)
    right_w_concat_fp16 = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # single fused matmul for each side
    left_concat = F.linear(x_fp16, left_w_concat_fp16)   # [B,N,N,2*H] fp16
    right_concat = F.linear(x_fp16, right_w_concat_fp16) # [B,N,N,2*H] fp16

    # split into projection & gate tensors
    left_proj = left_concat[..., :hidden]                # fp16
    left_gate = torch.sigmoid(left_concat[..., hidden:])# fp16
    right_proj = right_concat[..., :hidden]              # fp16
    right_gate = torch.sigmoid(right_concat[..., hidden:])# fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + per‑channel gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 (0/1) and make it contiguous
        mask_fp16 = mask.to(dtype=torch.float16, device=device).contiguous()

        # Allocate output buffers
        left_fused = torch.empty_like(left_proj)
        right_fused = torch.empty_like(right_proj)

        total_elements = left_proj.numel()
        BLOCK = 1024  # threads per program (tunable)
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            mask_fp16,
            left_gate,
            right_gate,
            left_fused,
            right_fused,
            total_elements,
            H=hidden,
            BLOCK=BLOCK,
        )
        left_fp16 = left_fused
        right_fp16 = right_fused
    else:
        # No mask → direct element‑wise multiplication (in‑place)
        left_fp16 = left_proj * left_gate
        right_fp16 = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out = left @ rightᵀ   (all fp16)
    # ------------------------------------------------------------------
    B, N, _, _ = left_fp16.shape               # B, N, N, H
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # torch.bmm uses Tensor Cores for fp16 → fp16 matmul with fp32 accumulation
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore the original layout and promote to FP32 for the next LayerNorm
    out = out_mat_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                            # [B,N,N,hidden] (fp32)

    # ------------------------------------------------------------------
    # 7️⃣ Output gate (sigmoid) – computed in FP16 for speed
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16, non_blocking=True))
    ).to(torch.float32)  # broadcastable to out (FP32)

    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim` (FP16 weight, FP32 activation)
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )                                            # [B,N,N,dim] (fp32)

    return out
