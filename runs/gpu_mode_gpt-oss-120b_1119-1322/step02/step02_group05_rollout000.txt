import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B.T for many small independent batches.
    A: (batch, M, K) contiguous row-major
    B: (batch, N, K) row-major (notice K is last dimension)
    C: (batch, M, N)
    All tensors are in FP16, accumulation in FP16.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base ptrs for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # Load B tile transposed (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        acc += tl.dot(a, b, dtype=tl.float16)

    # Store
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """Triton‑accelerated outgoing TriMul forward pass.

    The function follows the reference PyTorch implementation, but
    1) fuses the five linear maps (proj/gate/out_gate) into a single
       matmul;
    2) computes the cubic contraction `out[i,j] = Σ_k left[i,k]*right[j,k]`
       with a custom batched GEMM kernel that runs on TensorCores;
    3) keeps the heavy tensor arithmetic in FP16 for speed and casts
       back to FP32 only for the final LayerNorm and linear projection.
    """
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1. Input layer‑norm (FP32)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim] (float32)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------ #
    # 2. Fuse the five linear layers into a single matmul (FP16)
    # ------------------------------------------------------------------ #
    # flatten to (B*N*N, dim)
    x_flat = x.reshape(B * N * N, dim).contiguous()
    # concatenate weights (5*hidden, dim)
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*hidden, dim)

    # perform one matmul in FP16
    x_flat = x_flat.to(torch.float16)
    fused_out = F.linear(x_flat, fused_w)   # (B*N*N, 5*hidden)

    H = hidden_dim
    left_proj_f, left_gate_f, right_proj_f, right_gate_f, out_gate_f = torch.split(
        fused_out, H, dim=1
    )
    left_gate = torch.sigmoid(left_gate_f)
    right_gate = torch.sigmoid(right_gate_f)
    out_gate = torch.sigmoid(out_gate_f)

    # reshape back to [B,N,N,H] (FP16)
    left_proj = left_proj_f.view(B, N, N, H)
    left_gate = left_gate.view(B, N, N, H)
    right_proj = right_proj_f.view(B, N, N, H)
    right_gate = right_gate.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------ #
    # 3. Apply optional mask and fuse gates (still FP16)
    # ------------------------------------------------------------------ #
    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------ #
    # 4. Cubic contraction via batched GEMM kernel
    # ------------------------------------------------------------------ #
    # Transform to (B, H, N, N) → (B*H, N, N) layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)   # (BH, N, N)
    right_mat = right_perm.view(B * H, N, N) # (BH, N, N)

    # Output buffer in FP16 (to be cast later)
    out_mat = torch.empty(B * H, N, N, dtype=torch.float16, device=device)

    # Strides (in elements)
    # A
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    # B (row‑major (N,K))
    stride_bb = right_mat.stride(0)
    stride_bk = right_mat.stride(2)
    stride_bn = right_mat.stride(1)
    # C
    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Tile sizes – empirical sweet spot for H100 TensorCores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # batch
        (N + BLOCK_M - 1) // BLOCK_M,       # M tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # N tiles
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # Reshape back to original layout [B,N,N,H] (FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5. Second layer‑norm + out‑gate + final projection (FP32)
    # ------------------------------------------------------------------ #
    out = out.to(torch.float32)   # cast back to FP32 for stable LN
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Triton‑accelerated outgoing TriMul (Triangular Multiplicative Update).

Algorithm
---------
1️⃣ Layer‑norm over the input feature dimension (FP32).
2️⃣ Fuse the five linear maps (left_proj, left_gate, right_proj, right_gate,
   out_gate) into **one** matmul: (B·N², dim) @ (dim, 5·hidden)ᵀ → (B·N²,5·hidden).
   Split the result, apply sigmoid to the three gate tensors and reshape
   back to (B,N,N,hidden) (all FP16).
3️⃣ If a mask is supplied, fuse it together with the gates and the projections.
4️⃣ Core cubic contraction  
      out[i,j] = Σ_k left[i,k]·right[j,k]  
   is performed as a batched GEMM‑kernel (A @ Bᵀ) where each (batch,head)
   pair is an independent N×N matmul.  The kernel runs on TensorCores,
   uses FP16 arithmetic for both loads and accumulation and processes
   many tiles per launch.
5️⃣ Second LayerNorm on the hidden dimension, multiply by the out‑gate,
   and a final FP32 linear projection back to the original feature dimension.

All heavy tensor arithmetic (steps 2–4) stays in FP16; the final
LayerNorm / linear layers are executed in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of independent matrices.
    A : (batch, M, K)   row‑major
    B : (batch, N, K)   row‑major (accessed transposed inside)
    C : (batch, M, N)   row‑major
    All tensors are FP16; accumulation is performed in FP16.
    """
    batch_id = tl.program_id(0)          # batch dimension = B * hidden_dim
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over the K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )

        # Load B tile as transposed (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0)
        )

        # Tensor‑core dot product (FP16 accumulation)
        acc += tl.dot(a, b, dtype=tl.float16)

    # Write out the result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                     # (5·H, dim)

    # FP16 matmul
    fused_out = F.linear(x_flat.to(torch.float16), fused_weight)   # (B·N², 5·H)

    H = hidden_dim
    left_proj_f, left_gate_f, right_proj_f, right_gate_f, out_gate_f = torch.split(
        fused_out, H, dim=1
    )
    left_gate   = torch.sigmoid(left_gate_f)
    right_gate  = torch.sigmoid(right_gate_f)
    out_gate    = torch.sigmoid(out_gate_f)

    # reshape back to [B,N,N,H] (still FP16)
    left_proj  = left_proj_f.view(B, N, N, H)
    left_gate  = left_gate.view(B, N, N, H)
    right_proj = right_proj_f.view(B, N, N, H)
    right_gate = right_gate.view(B, N, N, H)
    out_gate   = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------ #
    # 3️⃣ Apply optional mask and fuse gates (still FP16)
    # ------------------------------------------------------------------ #
    if mask is not None and mask.numel() > 0:
        mask_f = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left  = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------ #
    # 4️⃣ Cubic contraction via batched GEMM (A @ Bᵀ) on TensorCores
    # ------------------------------------------------------------------ #
    # Permute to (B, H, N, N) → (B·H, N, N) layout expected by kernel
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * H, N, N)    # (BH, N, N)
    right_mat = right_perm.view(B * H, N, N)   # (BH, N, N)

    # Allocate output in FP16 (cast later)
    out_mat = torch.empty(B * H, N, N, dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bb = right_mat.stride(0)
    stride_bk = right_mat.stride(2)
    stride_bn = right_mat.stride(1)

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Tile sizes – tuned for H100 TensorCores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # col tiles
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # Reshape back to the original layout [B, N, N, H] (FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5️⃣ Second LayerNorm, out‑gate, final projection (FP32)
    # ------------------------------------------------------------------ #
    out = out.to(torch.float32)   # cast for stable LN
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])

    return out
```

This implementation keeps the memory‑intensive cubic contraction entirely on the GPU using a custom Triton batched‑GEMM kernel that exploits H100 Tensor‑Core FP16 arithmetic. The surrounding linear layers are fused where possible, and all heavy arithmetic stays in FP16 to minimise bandwidth, with a final cast to FP32 only for the second LayerNorm and the output projection.
