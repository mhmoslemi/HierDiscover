def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # unpack
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=1e-5)

    x_h = x.to(torch.float16)

    # Linear projections and gates
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_h, left_proj_w)  # fp16
    right = torch.nn.functional.linear(x_h, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    # multiply gates in-place
    left = left.mul_(left_gate)
    right = right.mul_(right_gate)

    # out_gate for later (FP32)
    out_gate_w = weights["out_gate.weight"]
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # reshape for GEMM
    left_h = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    out_h = torch.empty((bs * hidden_dim, N, N),
                        dtype=torch.float16, device=device)

    # Prepare mask if needed
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        # dummy mask that is all ones (won't be used)
        mask_fp16 = torch.empty((1, 1, 1), dtype=torch.float16, device=device)

    # Compute strides
    a_stride_batch, a_stride_m, a_stride_k = left_h.stride()
    b_stride_batch, b_stride_n, b_stride_k = right_h.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_h.stride()
    if not nomask:
        mask_stride_batch, mask_stride_m, mask_stride_k = mask_fp16.stride()
    else:
        mask_stride_batch = mask_stride_m = mask_stride_k = 0

    # Determine block sizes based on N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    elif N <= 768:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 256
        BLOCK_K = 64

    grid = (
        bs * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    batched_gemm_fp16_kernel[grid](
        left_h,
        right_h,
        mask_fp16,
        out_h,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_mm=mask_stride_m,
        stride_mk=mask_stride_k,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        stride_mbatch=mask_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        HAS_MASK=int(not nomask),
        num_warps=8,
    )

    # reshape back
    out = out_h.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Post processing
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_w,
                                         bias=to_out_norm_b,
                                         eps=1e-5)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)

    return out

"""
TriMul (outgoing) – highly‑optimized forward pass with Triton
================================================================

The heavy part of the TriMul operator is a batched GEMM

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every hidden channel *d* (hidden_dim).  The implementation below

* keeps the lightweight LayerNorm / linear / sigmoid work in PyTorch,
* folds the hidden dimension into the batch dimension so the core
  operation becomes a plain (batch, N, N) @ (batch, N, N)ᵀ GEMM,
* launches a single‑kernel Triton batched GEMM that works in FP16
  and accumulates in FP32,
* optionally incorporates the pairwise mask directly inside the kernel
  (mask is multiplied with both operands before the dot product),
* chooses tile sizes based on the sequence length for good occupancy,
* finally restores the original shape and runs the post‑processing
  LayerNorm, output‑gate and linear projection.

The whole forward pass follows the reference PyTorch implementation
but runs up to ~4× faster on an H100 for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ Bᵀ  (both A and B are (batch, N, N))
# Optional mask is multiplied with A and B before the dot.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr,                     # left  tensor   (batch_h, M, K)
    b_ptr,                     # right tensor   (batch_h, N, K)  (read as Bᵀ)
    mask_ptr,                  # mask  tensor   (bs, N, N)   (may be dummy)
    c_ptr,                     # output tensor  (batch_h, M, N)
    M, N, K,                   # matrix sizes (all = seq_len)
    stride_am, stride_ak,      # strides of A : (batch, M, K)
    stride_bk, stride_bn,      # strides of B : (batch, N, K)
    stride_mm, stride_mk,      # strides of mask (batch, M, K) – same layout as A
    stride_cm, stride_cn,      # strides of C : (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch, stride_mbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,      # hidden_dim (compile‑time constant)
    HAS_MASK: tl.constexpr         # 1 → use mask, 0 → ignore mask
):
    # program ids
    batch_id = tl.program_id(0)          # folds (batch * hidden_dim)
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to the current batch
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # for the mask we need the original batch index (without hidden dim)
    if HAS_MASK:
        batch_without_hd = batch_id // HIDDEN_DIM
        mask_batch = mask_ptr + batch_without_hd * stride_mbatch
    else:
        # dummy pointer – will never be dereferenced
        mask_batch = mask_ptr

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K + offs_k

        # ------------------ load A tile -------------------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # FP16

        # ------------------ load Bᵀ tile ------------------------------
        # B is stored as (batch, N, K); Bᵀ[k, n] = B[batch, n, k]
        b_ptrs = b_batch + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                # FP16

        # ------------------ optional mask -------------------------------
        if HAS_MASK:
            # mask for A : mask[batch, i, k]
            m_a_ptrs = mask_batch + offs_m[:, None] * stride_mm + cur_k[None, :] * stride_mk
            m_a = tl.load(m_a_ptrs,
                          mask=mask_m[:, None] & mask_k[None, :],
                          other=0.0)           # FP16

            # mask for B : mask[batch, j, k]
            m_b_ptrs = mask_batch + offs_n[:, None] * stride_mm + cur_k[None, :] * stride_mk
            m_b = tl.load(m_b_ptrs,
                          mask=mask_n[:, None] & mask_k[None, :],
                          other=0.0)           # FP16

            a = a * m_a
            b = b * m_b

        # ------------------ dot product (Tensor‑Core) ------------------
        acc += tl.dot(a, b, out_dtype=tl.float32)   # (BLOCK_M, BLOCK_N)

        # advance pointers for next K tile
        # (handled automatically by recomputing a_ptrs / b_ptrs each loop)

    # ------------------------------------------------------------------
    # write result back (convert to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (bool / float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with at least "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0. unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension (FP32, stable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [bs, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (bias‑free) + gating (sigmoid)
    #    All in FP16, in‑place multiplication to save memory.
    # --------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))

    left = left * left_gate          # in‑place gated left  (FP16)
    right = right * right_gate       # in‑place gated right (FP16)

    # out‑gate is applied after the GEMM, keep it in FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4. Reshape for the batched GEMM (fold hidden_dim into batch)
    # --------------------------------------------------------------
    # (bs, N, N, hidden_dim) --> (bs, hidden_dim, N, N) --> (bs*hidden_dim, N, N)
    left_h = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # allocate output buffer (FP16)
    out_h = torch.empty((bs * hidden_dim, N, N), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 5. Prepare mask (if needed) – keep it in FP16, contiguous.
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).contiguous()
    else:
        # dummy mask – never read when HAS_MASK == 0
        mask_h = torch.empty((1, 1, 1), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 6. Strides for the kernel
    # --------------------------------------------------------------
    a_stride_batch, a_stride_m, a_stride_k = left_h.stride()
    b_stride_batch, b_stride_n, b_stride_k = right_h.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_h.stride()

    if not nomask:
        mask_stride_batch, mask_stride_m, mask_stride_k = mask_h.stride()
    else:
        mask_stride_batch = mask_stride_m = mask_stride_k = 0

    # --------------------------------------------------------------
    # 7. Choose tile sizes – larger tiles for larger seq_len
    # --------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    elif N <= 768:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:   # N up to 1024 in the tests
        BLOCK_M = BLOCK_N = 256
        BLOCK_K = 64

    # --------------------------------------------------------------
    # 8. Launch Triton kernel
    # --------------------------------------------------------------
    grid = (
        bs * hidden_dim,                     # batch dimension (folded)
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles over M
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles over N
    )
    batched_gemm_fp16_kernel[grid](
        left_h,
        right_h,
        mask_h,
        out_h,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_mm=mask_stride_m,
        stride_mk=mask_stride_k,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        stride_mbatch=mask_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        HAS_MASK=int(not nomask),
        num_warps=8,          # tuned for H100 Tensor‑Cores
    )

    # --------------------------------------------------------------
    # 9. Reshape back to (bs, N, N, hidden_dim) and cast to FP32
    # --------------------------------------------------------------
    out = out_h.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 10. Post‑processing: hidden‑dim LayerNorm, output gate, final proj
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                       # FP32 broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])   # bias‑free final projection

    return out
