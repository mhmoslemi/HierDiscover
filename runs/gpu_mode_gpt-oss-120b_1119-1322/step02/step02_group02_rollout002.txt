"""
TriMul (outgoing) – fast Triton‑fused implementation
====================================================

Algorithm
---------
* **LayerNorm** of the input (`dim` channels) – stay in FP32 for numeric stability.
* **Fused linear projections**: the four linear layers that produce
  `left_proj`, `right_proj`, `left_gate`, `right_gate` and the output gate are
  merged into a single `F.linear` call with a weight matrix of shape
  `(5*hidden_dim, dim)`.  This cuts the amount of memory traffic on the large
  `(B·N·N, dim)` tensor by a factor of five.
* **Mask / gating** are applied element‑wise in FP16.
* **Batched GEMM** (`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`) is
  performed with a custom Triton kernel.  The hidden dimension is folded into
  the batch dimension (`batch_h = B * hidden_dim`).  The kernel computes a
  half‑precision matrix product with FP32 accumulation and stores the result
  directly as FP32 (`C = A @ Bᵀ`).
* **Post‑processing**: a second `LayerNorm` on the hidden dimension,
  multiplication with the (FP32) output‑gate, and the final bias‑free
  projection back to `dim` channels.

The heavy work (five 2‑D GEMMs + the batched GEMM) is therefore carried out by
high‑performance matrix–multiplication kernels; all remaining operations are
tiny.  The implementation respects the required signature
`custom_kernel((input, mask, weights, config)) → output`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    # strides for A (batch, M, K)
    stride_am, stride_ak,
    # strides for B (batch, K, N) – we read Bᵀ
    stride_bk, stride_bn,
    # strides for C (batch, M, N)
    stride_cm, stride_cn,
    # batch‑level strides
    stride_abatch, stride_bbatch, stride_cbatch,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched half‑precision GEMM
    C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, j, k]
    (i.e. B is read as Bᵀ).  Accumulation is FP32, result is stored as FP32.
    """
    batch_id = tl.program_id(0)   # combined (batch, hidden) index
    pid_m    = tl.program_id(1)   # tile on the M dimension
    pid_n    = tl.program_id(2)   # tile on the N dimension

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # per‑batch pointers (already include the base offset for the batch)
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # masks for out‑of‑bounds tiles
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load A[i, k] -------------------------------------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ---- load Bᵀ[k, j]  (i.e. B[j, k]) -------------------------------
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ---- accumulate ----------------------------------------------------
        acc += tl.dot(a, b, out_dtype=tl.float32)   # (BLOCK_M, BLOCK_N)

    # ---- write back C ----------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]  (float32)
        - mask         : torch.Tensor[bs, N, N]       (bool/float, optional)
        - weights      : dict of model parameters
        - config       : dict containing at least "dim", "hidden_dim",
                         optionally "nomask" (bool)

    Returns
    -------
    torch.Tensor[bs, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )                                   # [bs, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2. Fuse the five linear layers (left/right proj & gates + out_gate)
    # ------------------------------------------------------------------
    # Cast input once to FP16 – the heavy matrix‑multiplies are done in FP16
    x_h = x.to(torch.float16)

    # Collect and concatenate the five weight matrices (all shape (hidden_dim, dim))
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)

    # (5*hidden_dim, dim)
    all_W = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )

    # One fused linear: result shape [..., 5*hidden_dim]  (FP16)
    fused = F.linear(x_h, all_W)   # [bs, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 3. Slice the fused tensor
    # ------------------------------------------------------------------
    parts = torch.split(fused, hidden_dim, dim=-1)
    left      = parts[0]                     # [bs, N, N, hidden_dim]  FP16
    right     = parts[1]                     # FP16
    left_gate = torch.sigmoid(parts[2])      # FP16
    right_gate= torch.sigmoid(parts[3])      # FP16
    out_gate  = torch.sigmoid(parts[4])      # FP16 now, will be cast later

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [bs, N, N] -> [bs, N, N, 1] (FP16)
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the batched GEMM (fold hidden_dim into batch)
    # ------------------------------------------------------------------
    # From (bs, N, N, hidden_dim) -> (bs*hidden_dim, N, N), contiguous layout
    left_h  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (FP32 to avoid a final cast)
    out_h = torch.empty((bs * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Launch the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # strides of the three tensors (batch, M, K) / (batch, K, N) / (batch, M, N)
    a_stride_batch, a_stride_m, a_stride_k = left_h.stride()
    b_stride_batch, b_stride_n, b_stride_k = right_h.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_h.stride()

    grid = (
        bs * hidden_dim,                         # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_fp16_kernel[grid](
        left_h,
        right_h,
        out_h,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # tuned for H100 Tensor‑Cores
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to (bs, N, N, hidden_dim) – already FP32
    # ------------------------------------------------------------------
    out = (
        out_h.view(bs, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [bs, N, N, hidden_dim] in float32

    # ------------------------------------------------------------------
    # 9. Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                   # still FP32

    # out_gate was FP16 → cast to FP32 for the multiplication
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate

    # Final projection back to `dim` (bias‑free)
    to_out_w = weights["to_out.weight"]   # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)         # FP32 result [bs, N, N, dim]

    return out
