```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_fuse_kernel(left_ptr, right_ptr, mask_ptr, left_gate_ptr, right_gate_ptr,
                         total_elements, H: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < total_elements

    # load
    l = tl.load(left_ptr + offset, mask=mask, other=0.0)
    r = tl.load(right_ptr + offset, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offset, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offset, mask=mask, other=0.0)

    # compute mask index (broadcast over hidden)
    idx_mask = offset // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # fused multiplication
    l = l * m * lg
    r = r * m * rg

    # store back in-place
    tl.store(left_ptr + offset, l, mask=mask)
    tl.store(right_ptr + offset, r, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul (TriMul) forward pass.
    Performs layer‑norm, a fused linear‑projection+gate, optional mask‑gate fusion,
    a batched matrix‑multiply, second layer‑norm, output‑gate and final projection.
    The heavy element‑wise mask‑gate step is implemented in Triton.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (FP32)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # 2. Output gate (computed in FP32 before casting)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 3. Cast input to FP16 for the big GEMMs
    x_fp16 = x.half()

    # 4. Concatenate projection and gate weights (FP16)
    #   dims: (hidden, dim) each
    #   order: left_proj, left_gate, right_proj, right_gate
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"]
    ], dim=0).half()   # shape (4*hidden, dim)

    # 5. Single linear to produce both projections and gates
    combined = F.linear(x_fp16, w_concat)  # [B, N, N, 4*hidden]  (FP16)

    # 6. Split the combined tensor
    #   view as [B,N,N,4*hidden] then split
    split_chunks = torch.split(combined, hidden, dim=-1)
    left_proj, left_gate_raw, right_proj, right_gate_raw = split_chunks

    # 7. Apply sigmoid to gate tensors (still FP16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # 8. Fuse mask (if any) and gates into projection tensors
    if not nomask:
        # mask: (B,N,N) -> FP16 (0/1) broadcast over hidden
        mask_fp16 = mask.to(torch.float16).contiguous()
        # ensure contiguity for Triton kernel
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_proj.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            mask_fp16,
            left_gate,
            right_gate,
            total,
            H=hidden,
            BLOCK=BLOCK,
        )
    else:
        # No mask, just apply gates
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # 9. Batched GEMM: out = left @ right^T  (fp16)
    B, N, _, _ = left_proj.shape
    # shape (B*hidden, N, N)
    left_mat = left_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # torch.bmm uses Tensor Cores on fp16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # 10. Restore layout and promote to FP32
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).float()

    # 11. Second LayerNorm over hidden dim (FP32)
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # 12. Apply output gate (FP32) and final projection to dim
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
Outgoing TriMul (Tri‑multiplicative update) – fast forward implementation.

Algorithm
---------
1️⃣  Layer‑Norm over the model dimension (FP32).
2️⃣  Compute a *single* large FP16 linear layer that simultaneously produces
    left‑proj, left‑gate, right‑proj and right‑gate:
        combined = x_fp16 @ W_concat.T          (W_concat = [Lproj; Lgate; Rproj; Rgate])
    Split the result into the four tensors.
3️⃣  Apply sigmoid to the two gate tensors (still FP16).
4️⃣  Fuse optional binary mask, left‑gate and right‑gate with a tiny Triton
    kernel (in‑place on the projection tensors).  When no mask is present we
    simply use in‑place ``mul_``.
5️⃣  Batched matrix multiplication using Tensor‑Core FP16 GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    (implemented as ``torch.bmm`` on a view of shape ``(B*H, N, N)``).
6️⃣  Restore the original ``[B, N, N, H]`` layout and cast to FP32.
7️⃣  Second Layer‑Norm over the hidden dimension (FP32).
8️⃣  Multiply by the output gate (computed from the first‑norm tensor)
    and apply the final FP32 linear projection back to the model dimension.

Only the mask‑gate fusion is written in Triton – the rest uses highly
optimised PyTorch kernels (e.g. cuBLAS for the large mat‑mul).  This reduces
the number of separate GEMM calls from six to three and eliminates
intermediate copies, giving a large speed‑up on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                # fp16 * left  (B,N,N,H) contiguous
    right_ptr,               # fp16 * right (B,N,N,H) contiguous
    mask_ptr,                # fp16 * mask  (B,N,N)   contiguous
    left_gate_ptr,           # fp16 * left_gate  (B,N,N,H)
    right_gate_ptr,          # fp16 * right_gate (B,N,N,H)
    total_elems: tl.int64,   # B*N*N*H
    H: tl.constexpr,         # hidden dimension (known at compile time)
    BLOCK: tl.constexpr,     # threads per program
):
    """
    For each element (b,i,j,h):
        left  = left  * mask(b,i,j) * left_gate
        right = right * mask(b,i,j) * right_gate
    The operation is performed in‑place on the left/right tensors.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load the five tensors
    l = tl.load(left_ptr + offs,   mask=mask, other=0.0)
    r = tl.load(right_ptr + offs,  mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask is broadcasted over the hidden dimension
    idx_mask = offs // H                     # ((b,N,i)*N + j)
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # Fuse the three element‑wise multiplies
    l = l * lg * m
    r = r * rg * m

    # Store back in‑place
    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float, optional)
        - weights      : dict of model parameters (all on device)
        - config       : dict with keys
                         * "dim"        – model dimension
                         * "hidden_dim" – hidden dimension inside TriMul
                         * "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim] fp32

    # ------------------------------------------------------------------
    # 2️⃣  Output gate (computed in FP32, needed later)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden] fp32

    # ------------------------------------------------------------------
    # 3️⃣  Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 4️⃣  Concatenate four weight matrices (FP16) and perform ONE matmul
    # ------------------------------------------------------------------
    #   order: left_proj | left_gate | right_proj | right_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).half()   # shape (4*hidden, dim)

    # single Linear → (B,N,N,4*hidden)
    combined = F.linear(x_fp16, w_concat)   # fp16

    # ------------------------------------------------------------------
    # 5️⃣  Split into the four tensors
    # ------------------------------------------------------------------
    left_proj, left_gate_raw, right_proj, right_gate_raw = torch.split(
        combined, hidden, dim=-1
    )   # each view has shape [B,N,N,hidden] (fp16)

    # ------------------------------------------------------------------
    # 6️⃣  Apply sigmoid to the gate tensors (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)   # fp16
    right_gate = torch.sigmoid(right_gate_raw) # fp16

    # ------------------------------------------------------------------
    # 7️⃣  Fuse optional mask + gate into the projection tensors
    # ------------------------------------------------------------------
    if not nomask:
        # mask → FP16 (0/1) and make contiguous for Triton
        mask_fp16 = mask.to(torch.float16).contiguous()   # (B,N,N)

        # ensure all tensors are contiguous (required for Triton pointer arithmetic)
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left_proj.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            mask_fp16,
            left_gate,
            right_gate,
            total,
            H=hidden,
            BLOCK=BLOCK,
        )
    else:
        # No mask – only apply the per‑channel gates (in‑place, no extra memory)
        left_proj.mul_(left_gate)
        right_proj.mul_(right_gate)

    # ------------------------------------------------------------------
    # 8️⃣  Batched GEMM:   out = left @ rightᵀ   (FP16 → FP32 accumulator)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape                      # B, N, N, hidden
    # reshape to (B*hidden, N, N) – a copy is cheap compared to the GEMM cost
    left_mat  = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Torch's bmm uses Tensor‑Cores for fp16 inputs
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # ------------------------------------------------------------------
    # 9️⃣  Restore layout & up‑cast to FP32
    # ------------------------------------------------------------------
    out = (
        out_mat_fp16.view(B, hidden, N, N)      # (B, hidden, N, N)
        .permute(0, 2, 3, 1)                    # (B, N, N, hidden)
        .float()                                 # FP32 for the next LN
    )

    # ------------------------------------------------------------------
    # 🔟  Second LayerNorm over the hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B,N,N,hidden] fp32

    # ------------------------------------------------------------------
    # 1️⃣1️⃣  Apply output gate (FP32) and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise (FP32)
    out = F.linear(out, weights["to_out.weight"])  # final Linear → [B,N,N,dim] (FP32)

    return out
