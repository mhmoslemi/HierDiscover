"""
Triton‑accelerated forward pass for the *outgoing* TriMul operator.

Algorithm
---------
1. Layer‑norm over the last feature dimension.
2. Fuse the five linear maps (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) into a single matmul:  X_flat @ W_fusedᵀ.
   The three gate tensors are run through a sigmoid.
3. Apply the optional mask and fuse the gates with the projections.
4. Compute the cubic contraction
      out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   for every head *d* using a **batched GEMM** kernel that evaluates
   C = A @ Bᵀ with A,B ∈ (B·H,N,N).  The kernel is a classic tiled
   Tensor‑Core kernel (BLOCK_M×BLOCK_N tiles, K‑tiles of size BLOCK_K)
   and accumulates in FP32.
5. Second layer‑norm on the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to the original feature dimension.

The heavy N³ work is performed entirely on the GPU by the Triton kernel,
while all linear / norm / activation work is carried out with
high‑throughput torch kernels (mixed‑precision FP16 where possible).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ for many independent batches.
# A, B : (batch, M, K)   – row‑major
# C    : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # batch dimension (B·H)
    pid_m = tl.program_id(1)             # tile‑row index
    pid_n = tl.program_id(2)             # tile‑col index

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointer for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile transposed (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot (FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # Write the result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] containing the module output.
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # often denoted c_z

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (over the last feature dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                          # [B, N, N, dim]

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 3️⃣ fuse the five linear maps into a single matmul
    # ------------------------------------------------------------------
    # flatten -> (B·N², dim)
    x_flat = x.reshape(B * N * N, dim).contiguous()

    # concatenate weight matrices (each of shape hidden_dim × dim)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )                                          # (5·hidden_dim, dim)

    # use FP16 for the bulk of the work
    fused_weight = fused_weight.to(torch.float16)
    x_flat = x_flat.to(torch.float16)

    fused_out = F.linear(x_flat, fused_weight)  # (B·N², 5·hidden_dim)

    H = hidden_dim
    left_proj_flat  = fused_out[:, 0:H]
    left_gate_flat  = fused_out[:, H:2*H]
    right_proj_flat = fused_out[:, 2*H:3*H]
    right_gate_flat = fused_out[:, 3*H:4*H]
    out_gate_flat   = fused_out[:, 4*H:5*H]

    # apply sigmoid to gates
    left_gate_flat  = torch.sigmoid(left_gate_flat)
    right_gate_flat = torch.sigmoid(right_gate_flat)
    out_gate_flat   = torch.sigmoid(out_gate_flat)

    # reshape back to (B, N, N, H)
    left_proj = left_proj_flat.view(B, N, N, H)
    left_gate = left_gate_flat.view(B, N, N, H)
    right_proj = right_proj_flat.view(B, N, N, H)
    right_gate = right_gate_flat.view(B, N, N, H)
    out_gate   = out_gate_flat.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ fuse mask (if any) and gates with the projections
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ensure half precision for the contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ core N³ contraction via the batched GEMM kernel (A @ Bᵀ)
    # ------------------------------------------------------------------
    # bring head dimension to the second axis and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & head dimensions -> (B·H, N, N)
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # output buffer (FP32 accumulation)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # strides (row‑major layout)
    stride_am = left_mat.stride(1)     # stride for M dimension
    stride_ak = left_mat.stride(2)     # stride for K dimension
    stride_ab = left_mat.stride(0)     # batch stride

    stride_bn = right_mat.stride(1)    # N stride (row)
    stride_bk = right_mat.stride(2)    # K stride (col) – equals 1 after contiguity
    stride_bb = right_mat.stride(0)    # batch stride

    stride_cm = out_mat.stride(1)      # M stride
    stride_cn = out_mat.stride(2)      # N stride
    stride_cb = out_mat.stride(0)      # batch stride

    # tile sizes – tuned for H100 TensorCores (FP16×FP16→FP32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # batch dimension (B·H)
        (N + BLOCK_M - 1) // BLOCK_M,       # rows tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # cols tiles
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps ≈ 256 threads – good for 64×64 tiles
        num_stages=3,         # double‑buffering for latency hiding
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ second LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                          # still FP32

    out = out * out_gate.to(out.dtype)        # gating
    out = F.linear(out, weights["to_out.weight"])   # project back to dim

    return out
