"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
================================================================
The operator works on a tensor ``x`` of shape [B, N, N, C] and produces a
tensor of the same shape.  The computation consists of

1. Layer‑norm over the feature dimension C.
2. Five linear maps (left_proj, left_gate, right_proj, right_gate, out_gate)
   are fused into a single half‑precision GEMM:
        (B·N², C) @ (5·H, C)ᵀ → (B·N², 5·H)
   where ``H = hidden_dim``.
3. A tiny Triton kernel fuses the three sigmoid gates, the optional mask and
   the two projections, producing the tensors
        left  = left_proj * sigmoid(left_gate) * mask
        right = right_proj * sigmoid(right_gate) * mask
        out_gate = sigmoid(out_gate_proj)
   All tensors are kept in fp16.
4. The cubic contraction
        out[i,j] = Σₖ left[i,k] · right[j,k]
   is a batch of independent matrix multiplies (B·H) of shape N×N.  The
   contraction is performed with a highly‑optimized cuBLAS batched GEMM
   via ``torch.bmm``, which on an H100 is substantially faster than a
   custom Triton kernel for the typical N∈{256,…,1024}.
5. A second Layer‑norm on the hidden dimension, multiplication by
   ``out_gate`` and a final linear projection back to the original
   feature dimension.

Only step 3 is implemented in Triton (the rest uses highly‑optimized
torch kernels).  All heavy arithmetic is carried out in fp16, while the
final output is returned as fp32.

The implementation respects the “no‑mask’’ case by supplying a mask of
ones.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    fused_ptr,                 # (rows, 5*H)  – result of the fused linear
    mask_ptr,                  # (rows,)     – 1.0 for uncovered entries, 0.0/1.0 otherwise
    left_out_ptr,              # (rows*H,)   – gated * masked left projection
    right_out_ptr,             # (rows*H,)   – gated * masked right projection
    out_gate_ptr,              # (rows*H,)   – out‑gate after sigmoid
    rows,                      # = B * N * N
    hidden,                    # = H
    BLOCK_SIZE: tl.constexpr, # number of elements processed per program
):
    """
    For every pair (b,i,j) and every hidden index h compute

        left  = left_proj * sigmoid(left_gate) * mask
        right = right_proj * sigmoid(right_gate) * mask
        out_gate = sigmoid(out_gate_proj)

    All tensors are fp16.  The mask is broadcast across the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    # Global mask of valid elements
    valid = offs < rows * hidden

    # De‑compose the flat index into (row, hidden)
    row = offs // hidden          # which (b,i,j) entry
    h   = offs % hidden           # which hidden head

    # Offsets into the fused matrix (5*H columns per row)
    base = row * (5 * hidden)     # start of the row in fused_ptr

    # Load the five raw values
    left_proj   = tl.load(fused_ptr + base + h,                     mask=valid, other=0.0)
    left_gate   = tl.load(fused_ptr + base + hidden + h,          mask=valid, other=0.0)
    right_proj  = tl.load(fused_ptr + base + 2 * hidden + h,      mask=valid, other=0.0)
    right_gate  = tl.load(fused_ptr + base + 3 * hidden + h,      mask=valid, other=0.0)
    out_gate_v  = tl.load(fused_ptr + base + 4 * hidden + h,      mask=valid, other=0.0)

    # Sigmoid – the standard logistic function
    left_gate_s  = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_s = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate_s   = 1.0 / (1.0 + tl.exp(-out_gate_v))

    # Load the (b,i,j) mask and broadcast over hidden
    mask_val = tl.load(mask_ptr + row, mask=valid, other=1.0)

    # Apply mask and gates
    left  = left_proj  * left_gate_s  * mask_val
    right = right_proj * right_gate_s * mask_val

    # Store results
    tl.store(left_out_ptr  + offs, left,      mask=valid)
    tl.store(right_out_ptr + offs, right,     mask=valid)
    tl.store(out_gate_ptr  + offs, out_gate_s,mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] containing the module output.
    """
    # --------------------------------------------------------------- #
    # Unpack and extract configuration
    # --------------------------------------------------------------- #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden = config["hidden_dim"]   # often called c_z
    eps = 1e-5

    # --------------------------------------------------------------- #
    # 1️⃣ First LayerNorm (float32) – keep everything in fp32 for the LN
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B, N, N, dim]  (float32)

    B, N, _, _ = x.shape

    # --------------------------------------------------------------- #
    # 2️⃣ Fuse the five linear maps into a single half‑precision GEMM
    # --------------------------------------------------------------- #
    x_flat = x.reshape(B * N * N, dim).contiguous()
    x_flat = x_flat.to(torch.float16)

    # concatenate the five weight matrices (shape: (5*hidden, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )
    fused_weight = fused_weight.to(torch.float16)

    # single linear transformation
    fused_out = F.linear(x_flat, fused_weight)        # (B·N², 5·hidden)

    # --------------------------------------------------------------- #
    # 3️⃣ Fuse sigmoid, mask and gates with a tiny Triton kernel
    # --------------------------------------------------------------- #
    rows = B * N * N

    # mask handling – a tensor of shape (rows,) with 1.0 if no mask is given
    if mask is None or mask.numel() == 0:
        mask_fp16 = torch.ones(rows, dtype=torch.float16, device=device)
    else:
        # mask is [B, N, N]; broadcast across hidden dimension later
        mask_fp16 = mask.view(-1).to(torch.float16)

    # output buffers for left, right and out_gate (flattened)
    left_flat  = torch.empty(rows * hidden, dtype=torch.float16, device=device)
    right_flat = torch.empty(rows * hidden, dtype=torch.float16, device=device)
    out_gate_flat = torch.empty(rows * hidden, dtype=torch.float16, device=device)

    # Triton launch parameters
    BLOCK_SIZE = 1024                      # tuneable; works well on H100
    grid = ((rows * hidden + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    _gate_mask_fuse_kernel[grid](
        fused_out,                     # fused_ptr
        mask_fp16,                     # mask_ptr
        left_flat,                     # left_out_ptr
        right_flat,                    # right_out_ptr
        out_gate_flat,                 # out_gate_ptr
        rows,
        hidden,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
    )

    # reshape to 4‑D tensors
    left = left_flat.view(B, N, N, hidden)       # [B,N,N,H]
    right = right_flat.view(B, N, N, hidden)     # [B,N,N,H]
    out_gate = out_gate_flat.view(B, N, N, hidden)  # [B,N,N,H]

    # --------------------------------------------------------------- #
    # 4️⃣ Core N³ contraction: batch GEMM left @ rightᵀ
    #    (performed with torch.bmm, which on H100 is faster than a custom kernel)
    # --------------------------------------------------------------- #
    # bring the hidden dimension into the batch axis for the GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat = left_perm.view(B * hidden, N, N)           # (BH, N, N)
    right_mat = right_perm.view(B * hidden, N, N)         # (BH, N, N)

    # Batched matrix multiplication: (BH, N, N) @ (BH, N, N)ᵀ → (BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16 with fp32 acc

    # Bring back to the original layout
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out = out.to(torch.float32)          # layer‑norm expects float32

    # --------------------------------------------------------------- #
    # 5️⃣ Second LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                   # still fp32

    # apply out_gate (cast to fp32 for the multiplication)
    out = out * out_gate.to(out.dtype)

    # final linear projection back to the original feature dimension (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
