"""
custom_kernel
==============

Triton‑accelerated forward pass of the **outgoing** TriMul module (AlphaFold3).

Algorithm
---------

1. **Layer‑norm** over the last channel (`dim`) – performed in FP32 with
   `torch.nn.functional.layer_norm`.

2. **Fused projections + gating**  
   * Concatenate the projection and gate matrices (each `hidden_dim × dim`);
   * One `torch.nn.functional.linear` per side reads the large 4‑D tensor
     only once, producing a tensor of shape `[B, N, N, 2·hidden_dim]` in
     FP16.
   * Split into projection and gate, apply `sigmoid` to the gate,
     optionally multiply by the mask and finally element‑wise multiply
     projection × gate → `left`, `right` (FP16).

3. **Core N³ update** – a batched matrix multiplication  
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.  
   For each hidden dimension `h` we need the product `A @ Bᵀ` where
   `A = left[b,:, :,h]` and `B = right[b,:, :,h]`.  This is implemented
   by a custom Triton kernel that:
   * treats each pair `(batch, hidden)` as a separate GEMM,
   * works on FP16 inputs and accumulates in FP32,
   * loads tiles of size `BLOCK_M × BLOCK_K` and `BLOCK_K × BLOCK_N`,
   * uses `tl.dot` (Tensor‑core) for the inner product,
   * writes the FP32 result to `out`.

4. **Hidden‑dim layer‑norm + out‑gate** – FP32 layer‑norm over the
   `hidden_dim` channel, followed by element‑wise multiplication with
   `out_gate = sigmoid(Linear(x))`.

5. **Final projection** – a weight‑only Linear back to `dim`.

The heavy work (step 3) is entirely on GPU via Triton, while the lighter
operations stay in PyTorch.  All tensors stay on the device throughout,
and the final output is returned as float‑32.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM   C = A @ Bᵀ
#   A : (total, M, K)   FP16
#   B : (total, N, K)   FP16      (we access Bᵀ in the kernel)
#   C : (total, M, N)   FP32
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_out_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch‑stride, row‑stride, col‑stride
    stride_bb, stride_bk, stride_bn,   # B: batch‑stride, col‑stride (k), row‑stride (n) → Bᵀ
    stride_cb, stride_cm, stride_cn,   # C: batch‑stride, row‑stride, col‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM used by TriMul.
    Each program (grid element) handles one (batch × hidden) matrix.
    """
    pid_batch = tl.program_id(2)          # which (batch * hidden) matrix
    pid_m = tl.program_id(0)              # block row of C
    pid_n = tl.program_id(1)              # block col of C

    # ---------------------- tile offsets ----------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------- base pointers --------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # pointers to A[i,k] tile and Bᵀ[k,j] tile
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ---------------------- accumulation ----------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # ---------------------- write back -----------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        - mask         : torch.Tensor or None   [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (torch.Tensor)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm  (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B, N, N, dim], dtype float32

    # --------------------------------------------------------------
    # 2) Fuse projections + gates (FP16 heavy work)
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # concatenate proj & gate weights (no bias) → shape (2*hidden_dim, dim)
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)

    # fused linear: read `x` only once per side
    left_combined = F.linear(x_fp16, left_w)   # [B, N, N, 2*hidden_dim], FP16
    right_combined = F.linear(x_fp16, right_w)

    # split into projection and gate parts
    left_proj = left_combined[..., :hidden_dim]                      # (B,N,N,H)
    left_gate = left_combined[..., hidden_dim:].sigmoid()            # (B,N,N,H)

    right_proj = right_combined[..., :hidden_dim]
    right_gate = right_combined[..., hidden_dim:].sigmoid()

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask is [B,N,N]; expand to [B,N,N,1] and cast to fp16
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 4) Apply gating (still FP16)
    # --------------------------------------------------------------
    left = left_proj * left_gate      # (B,N,N,H)
    right = right_proj * right_gate   # (B,N,N,H)

    # --------------------------------------------------------------
    # 5) Core N³ update – batched GEMM (FP16 → FP32)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                    # B, seq_len, seq_len, hidden_dim

    # reshape to (B*H, N, N) – each hidden slice becomes an independent GEMM
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_t.view(-1, N, N)                # (B*H, N, N), FP16
    right_flat = right_t.view(-1, N, N)              # (B*H, N, N), FP16
    total = left_flat.shape[0]                       # B * H

    # output accumulator (FP32)
    out_flat = torch.empty((total, N, N), dtype=torch.float32, device=device)

    # Choose tile size: 128 works for multiples of 128, otherwise fall back to 64
    if N % 128 == 0:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 64
    BLOCK_K = 64

    # grid: (row_blocks, col_blocks, total_matrices)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total,
    )

    batched_gemm_out_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0),   # batch‑stride for A
        left_flat.stride(1),   # row‑stride for A (M)
        left_flat.stride(2),   # col‑stride for A (K)
        right_flat.stride(0),  # batch‑stride for B
        right_flat.stride(2),  # stride of K dimension in B (col‑stride for Bᵀ)
        right_flat.stride(1),  # stride of N dimension in B (row‑stride for Bᵀ)
        out_flat.stride(0),    # batch‑stride for C
        out_flat.stride(1),    # row‑stride for C (M)
        out_flat.stride(2),    # col‑stride for C (N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # well‑tuned for H100 Tensor‑core cores
    )

    # reshape back to (B, N, N, H)  – now in FP32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # FP32

    # out_gate = sigmoid(Linear(x))  – computed from the *original* x (float32)
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # (B,N,N,H)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7) Final projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim), FP32

    return out
```
