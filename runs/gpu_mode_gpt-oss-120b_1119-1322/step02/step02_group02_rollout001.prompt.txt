import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – highly‑optimized forward pass
================================================

The operator works on a 4‑D tensor *x* of shape [B, N, N, C] (C = dim).
The heavy part of the computation is a batched matrix multiplication

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every hidden channel *d* (hidden_dim).  This is an O(N³·hidden_dim)
operation.  The implementation follows the reference PyTorch code but
uses mixed‑precision (FP16) for the compute‑bound part and a custom
Triton kernel that performs the batched GEMM with Tensor‑Core friendly
half‑precision arithmetic.  All lightweight operations (layer‑norm,
linear projections, sigmoids, masking) are done with PyTorch.

Key ideas
---------
* Layer‑norm, linear projections and the three gates are kept in FP32
  for numerical stability and are cast to FP16 right before the GEMM.
* The hidden dimension is folded into the batch dimension
  (batch_h = B·hidden_dim) so that the core operation becomes a
  strided‑batched GEMM of shape (batch_h, N, N) @ (batch_h, N, N)ᵀ.
* A Triton kernel (`batched_gemm_fp16_kernel`) computes C = A @ Bᵀ
  using FP16 inputs and accumulates into FP32 (the fastest path on
  H100).  The kernel is tiled with BLOCK_M = BLOCK_N = 128 and
  BLOCK_K = 32 and runs with 8 warps → full Tensor‑Core utilisation.
* After the GEMM the result is reshaped back, post‑processed with a
  second LayerNorm, an output gate and a final projection to the original
  dimension.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ
#   A : (batch_h, M, K)   – left tensor   (M = N, K = N)
#   B : (batch_h, N, K)   – right tensor  (rows=j, cols=k)
#   C : (batch_h, M, N)   – output tensor (i, j)
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, N, K) – read as Bᵀ
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    One program instance per (batch, tile_m, tile_n).
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A tile (M × K) ----
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # FP16 load

        # ---- Load Bᵀ tile (K × N) ----
        # B is stored as (batch, N, K); Bᵀ[k, n] = B[batch, n, k]
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # FP16 load

        # ---- Accumulate (Tensor‑Core) ----
        acc += tl.dot(a, b, out_dtype=tl.float32)   # (BLOCK_M, BLOCK_N)

    # ---- Write back result (convert to FP16) ----
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict, must contain "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0. unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (FP32)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [bs, N, N, dim]  FP32

    # -----------------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑heavy part
    # -----------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------------
    # 3. Linear projections (bias‑free) – everything in FP16
    # -----------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left  = F.linear(x_h, left_proj_w)    # [bs, N, N, hidden_dim]  FP16
    right = F.linear(x_h, right_proj_w)

    # -----------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in FP16
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # keep FP32 for later

    # -----------------------------------------------------------------------
    # 5. Optional mask and element‑wise gates
    # -----------------------------------------------------------------------
    if not nomask:
        # mask: [bs, N, N] → [bs, N, N, 1] (FP16)
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_h
        right = right * mask_h

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 6. Fold hidden_dim into the batch dimension and prepare for GEMM
    # -----------------------------------------------------------------------
    # (bs, N, N, hidden_dim) → (bs, hidden_dim, N, N) → (batch_h, N, N)
    left_h  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Allocate output tensor (FP16)
    out_h = torch.empty((bs * hidden_dim, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 7. Launch the Triton batched‑GEMM kernel
    # -----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    a_stride_batch, a_stride_m, a_stride_k = left_h.stride()
    b_stride_batch, b_stride_n, b_stride_k = right_h.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_h.stride()

    grid = (
        bs * hidden_dim,                           # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_fp16_kernel[grid](
        left_h,
        right_h,
        out_h,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # tuned for H100 Tensor‑Cores
    )

    # -----------------------------------------------------------------------
    # 8. Reshape back to (bs, N, N, hidden_dim) and convert to FP32
    # -----------------------------------------------------------------------
    out = out_h.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------------
    # 9. Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # out_gate already in FP32 (shape [bs, N, N, hidden_dim])
    out = out * out_gate

    to_out_w = weights["to_out.weight"]           # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                 # final projection (bias‑free)

    return out
