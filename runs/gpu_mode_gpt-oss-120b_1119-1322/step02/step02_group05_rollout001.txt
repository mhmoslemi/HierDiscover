"""
Triton‑accelerated “outgoing” TriMul (Triangular Multiplicative Update).

Algorithm
---------
1️⃣ Layer‑norm on the input channel dimension.
2️⃣ Fuse the five linear maps (left‑proj, left‑gate, right‑proj, right‑gate,
   out‑gate) into a single matmul:   X_flat (B·N²×C) @
   W_fusedᵀ (C×5·H) → (B·N²×5·H).  Split the result and apply a sigmoid to the
   three gate tensors.
3️⃣ (Optional) mask the left/right projections and multiply by their gates.
4️⃣ Core cubic contraction   out[i,j] = Σₖ left[i,k]·right[j,k]   for every
   hidden head H.  This is a batched GEMM C = A @ Bᵀ with batch = B·H.
   The heavy lifting is done by a custom Triton kernel that uses
   TL‑dot (tensor‑core) with FP16 inputs and FP32 accumulation.
5️⃣ Second layer‑norm on the hidden dimension, multiply by the out‑gate and
   a final linear projection back to the original feature dimension.
All heavy pointwise work stays in FP16, the GEMM accumulates in FP32 and the
final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Batched GEMM kernel: C = A @ Bᵀ  (A,B,C : (batch, M, K), (batch, N, K), (batch, M, N))
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: row‑stride, col‑stride, batch‑stride
    stride_bk, stride_bn, stride_bb,   # B (accessed transposed): K‑stride, N‑stride, batch‑stride
    stride_cm, stride_cn, stride_cb,   # C: row‑stride, col‑stride, batch‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM where B is accessed as Bᵀ.
    Uses Tensor‑Core friendly blocks (M/N/K divisible by 16) and FP32
    accumulation.
    """
    batch_id = tl.program_id(0)          # batch = B * hidden_dim
    pid_m    = tl.program_id(1)          # tile‑row
    pid_n    = tl.program_id(2)          # tile‑col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------- #
    # K‑loop
    # -------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # A tile – (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ tile – (K × N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot (FP16 × FP16 → FP32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # -------------------------------------------------------------------- #
    # Store
    # -------------------------------------------------------------------- #
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry‑point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32).
    """
    torch.set_grad_enabled(False)                     # no autograd needed

    # --------------------------------------------------------------------- #
    # Unpack
    # --------------------------------------------------------------------- #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden = config["hidden_dim"]                      # called c_z in the paper
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1️⃣ Input layer‑norm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                                  # [B, N, N, dim]

    B, N, _, _ = x.shape

    # --------------------------------------------------------------------- #
    # 2️⃣ Fuse five linear maps into one matmul (FP16 for speed)
    # --------------------------------------------------------------------- #
    x_flat = x.reshape(B * N * N, dim).contiguous()
    # Concatenate the five weight matrices (shape: (5*hidden, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )
    fused_weight = fused_weight.to(torch.float16)
    x_flat = x_flat.to(torch.float16)

    # (B·N², 5·hidden)  –  no bias in any of the linear layers
    fused_out = F.linear(x_flat, fused_weight)        # FP16 → (B·N², 5·hidden)

    H = hidden
    # Slice the concatenated tensor
    left_proj_f = fused_out[:, 0:H]
    left_gate_f = fused_out[:, H:2 * H]
    right_proj_f = fused_out[:, 2 * H:3 * H]
    right_gate_f = fused_out[:, 3 * H:4 * H]
    out_gate_f = fused_out[:, 4 * H:5 * H]

    # Gate activations (still FP16)
    left_gate_f = torch.sigmoid(left_gate_f)
    right_gate_f = torch.sigmoid(right_gate_f)
    out_gate_f = torch.sigmoid(out_gate_f)

    # Reshape to (B, N, N, H)
    left_proj = left_proj_f.view(B, N, N, H)
    left_gate = left_gate_f.view(B, N, N, H)
    right_proj = right_proj_f.view(B, N, N, H)
    right_gate = right_gate_f.view(B, N, N, H)
    out_gate = out_gate_f.view(B, N, N, H)            # FP16

    # --------------------------------------------------------------------- #
    # 3️⃣ Optional mask + gate fusion (still FP16)
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> [B, N, N, 1] (FP16)
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------------------- #
    # 4️⃣ Core N³ contraction – batched GEMM (A @ Bᵀ)
    # -------------------------------------------------------------- #
    # Move hidden dimension to the 2nd axis and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Shape (B·H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Output buffer – we accumulate in FP32 (required for numerical stability)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ----------------------------------------------------------------- #
    # Strides (row‑major layout)
    # ----------------------------------------------------------------- #
    stride_am = left_mat.stride(1)        # row stride (M)
    stride_ak = left_mat.stride(2)        # col stride (K)
    stride_ab = left_mat.stride(0)        # batch stride

    stride_bk = right_mat.stride(2)       # K‑stride in transposed view (original col)
    stride_bn = right_mat.stride(1)       # N‑stride (original row)
    stride_bb = right_mat.stride(0)       # batch stride

    stride_cm = out_mat.stride(1)         # row stride (M)
    stride_cn = out_mat.stride(2)         # col stride (N)
    stride_cb = out_mat.stride(0)         # batch stride

    # ----------------------------------------------------------------- #
    # Tile sizes – tuned for FP16 TensorCores on H100
    # ----------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                   # batch axis
        (N + BLOCK_M - 1) // BLOCK_M,            # rows
        (N + BLOCK_N - 1) // BLOCK_N,            # cols
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                      # M
        N,                      # N
        N,                      # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,           # pipelined prefetch for higher throughput
    )

    # Back‑to original layout: [B, N, N, H] (still FP32)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5️⃣ Second LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                                  # FP32

    out_gate = out_gate.to(out.dtype)                 # FP32
    out = out * out_gate

    # Final linear projection back to original channel dimension (FP32)
    out = F.linear(out, weights["to_out.weight"])

    return out
