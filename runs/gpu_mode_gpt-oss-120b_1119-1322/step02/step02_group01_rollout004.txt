===================================================

The Tensor has shape [B, N, N, D] (float32).
The forward consists of:
1. Input LayerNorm (float32).
2. Three linear projections (left/right and three gates) are fused:
   - left_proj + left_gate, right_proj + right_gate are each computed
     with a *single* matmul (weights are concatenated along the output axis).
3. Optional binary mask is applied to the projected tensors.
4. Gating (sigmoid) and element‑wise multiplication (still FP16).
5. Core N³ update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM C = A @ Bᵀ for every hidden channel d.
   A dedicated Triton kernel launches one program per (batch * hidden) matrix.
   The kernel uses FP16 × FP16 → FP32 accumulation with a 64×64 tile
   (tuned for H100 Tensor‑cores) and writes the result back in FP16.
6. Hidden‑dim LayerNorm (float32), multiplication by the output‑gate,
   and a final linear projection back to the original dimension.

Only the cubic update (step 5) runs on Triton; everything else uses
high‑performance PyTorch kernels.  All heavy linear work is performed
in FP16 for speed, while the final LayerNorms are kept in FP32 for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (FP16×FP16 → FP32)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B: batch, col(k), row(n) strides
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel used in the TriMul “outgoing” update."""
    pid_batch = tl.program_id(2)          # which (batch * hidden) matrix
    pid_m = tl.program_id(0)              # block row of C
    pid_n = tl.program_id(1)              # block col of C

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this batch element
    # --------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # --------------------------------------------------------------
    # Pointers to the start of the current tile (k = 0)
    # --------------------------------------------------------------
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiling)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K

        # Load A tile (shape BLOCK_M × BLOCK_K)
        a = tl.load(
            a_tile_ptr + cur_k * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_k) < K),
            other=0.0,
        )
        # Load Bᵀ tile (shape BLOCK_K × BLOCK_N)
        b = tl.load(
            b_tile_ptr + cur_k * stride_bk,
            mask=((offs_k + cur_k) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result (FP32 → FP16)
    # --------------------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
            - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
            - weights      : dict of torch.Tensor (float32)
            - config       : dict with at least “dim” and “hidden_dim”

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D] float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for all heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Helper to cast a weight once
    def _to_half(t):
        return t.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse left/right projections with their respective gates
    # ------------------------------------------------------------------
    # concatenate (proj, gate) weights along the output feature axis
    w_left = torch.cat([_to_half(weights["left_proj.weight"]),
                        _to_half(weights["left_gate.weight"])], dim=0)   # (2*H, D)
    w_right = torch.cat([_to_half(weights["right_proj.weight"]),
                         _to_half(weights["right_gate.weight"])], dim=0)  # (2*H, D)

    # fused linear: result shape [B, N, N, 2*H]
    left_fused = F.linear(x_fp16, w_left)    # no bias
    right_fused = F.linear(x_fp16, w_right)

    # split back into (proj, gate) parts
    left_proj = left_fused[..., :hidden]       # [B, N, N, H]
    left_gate = left_fused[..., hidden:].sigmoid()   # [B, N, N, H]
    right_proj = right_fused[..., :hidden]
    right_gate = right_fused[..., hidden:].sigmoid()

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1]  (FP16)
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5) Apply gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    B, N, _, H = left.shape  # B, N, N, hidden

    # ------------------------------------------------------------------
    # 6) Out‑gate (computed from the normalized input)
    # ------------------------------------------------------------------
    out_gate = F.linear(x_fp16, _to_half(weights["out_gate.weight"])).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7) Tri‑multiplicative update via Triton (batched GEMM)
    # ------------------------------------------------------------------
    # Re‑order to (B, H, N, N) and flatten batch*hidden
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_t.view(-1, N, N)    # [B*H, N, N]  (FP16)
    right_flat = right_t.view(-1, N, N)

    total = left_flat.shape[0]           # B * H

    # Allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat)  # [B*H, N, N]

    # Choose tile size – 64×64 works for all tested seq‑lens and fits H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i dimension)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j dimension)
        total,                     # one program per (batch * hidden)
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K
        left_flat.stride(0),       # stride_ab (batch stride)
        left_flat.stride(1),       # stride_am (row stride)
        left_flat.stride(2),       # stride_ak (col stride)
        right_flat.stride(0),      # stride_bb (batch stride)
        right_flat.stride(2),      # stride_bk (col‑stride of B)
        right_flat.stride(1),      # stride_bn (row‑stride of B)
        out_flat.stride(0),        # stride_cb (batch stride)
        out_flat.stride(1),        # stride_cm (row stride)
        out_flat.stride(2),        # stride_cn (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, H] (still FP16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9) Hidden‑dim LayerNorm (float32), out‑gate multiplication,
    #    and final projection back to original dimension D
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # final projection (float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
