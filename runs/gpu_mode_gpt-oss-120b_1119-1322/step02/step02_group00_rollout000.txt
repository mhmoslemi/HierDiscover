"""
custom_kernel – Triton‑accelerated forward pass of the “outgoing” TriMul
module (AlphaFold‑3).

Algorithm (high‑level)
----------------------
1. Layer‑norm the input tensor (float32) → [B, N, N, D].
2. Cast to fp16 and apply four bias‑free linear projections (left, right)
   together with three sigmoid gates (left_gate, right_gate, out_gate).
3. Optional mask (broadcast over the hidden dimension) is applied to
   `left`/`right` before gating.
4. Gate the projected tensors element‑wise.
5. Core N³ update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ for every hidden
   channel *h*.  A custom Triton kernel (`batched_gemm_out_kernel`) computes
   all (B·H) products in fp16, accumulating into fp32.
6. Hidden‑dim layer‑norm + element‑wise multiplication by `out_gate`.
7. Final linear projection back to the original dimension D.
8. Cast the result back to float32 (the expected downstream dtype).

All heavy arithmetic (step 5) runs on the GPU; the surrounding
operations stay in PyTorch and are kept in fp16 for speed while the
final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,          # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,          # B: batch, col (k), row (n) strides
    stride_cb, stride_cm, stride_cn,          # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each (batch × hidden) matrix:
        C = A @ Bᵀ
    where A (M×K), B (N×K) and C (M×N) are fp16 tensors.
    Accumulation is in fp32 to preserve numeric quality.
    """
    pid_batch = tl.program_id(2)                # which (batch*hidden) matrix
    pid_m = tl.program_id(0)                    # block row of C
    pid_n = tl.program_id(1)                    # block col of C

    # Offsets of this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # Pointers to the start of the current tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # a, b are fp16; dot promotion -> fp32
        acc += tl.dot(a, b)

    # Write the result tile
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights:     dict mapping weight names to torch.Tensor (float32)
            - config:      dict with keys "dim" (D) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the bulk of the computation
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (bias‑free layers)
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate= weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left)    # [B, N, N, H]
    right = F.linear(x_h, w_right)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Compute sigmoid gates – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # (B, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 8) Rearrange for batched GEMM: (B*H, N, N)
    #    Layout required by the Triton kernel: [batch, rows, cols]
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_t.view(-1, N, N)   # [B*H, N, N]  fp16
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]  fp16

    total = left_flat.shape[0]          # B * H

    # Output buffer for the GEMM (fp16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 9) Triton kernel launch – compute left @ rightᵀ for each (b,h)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),  # blocks along M (row)
        triton.cdiv(N, BLOCK_N),  # blocks along N (col)
        total,                    # one program per (batch * hidden) matrix
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0),   # batch stride for A
        left_flat.stride(1),   # row stride for A
        left_flat.stride(2),   # col stride for A
        right_flat.stride(0),  # batch stride for B
        right_flat.stride(2),  # col stride (k) for B
        right_flat.stride(1),  # row stride (n) for B
        out_flat.stride(0),    # batch stride for C
        out_flat.stride(1),    # row stride for C
        out_flat.stride(2),    # col stride for C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10) Reshape back to [B, N, N, H] (fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 11) Hidden‑dim LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # [B, N, N, D]  fp16

    # ------------------------------------------------------------------
    # 13) Cast to float32 (the expected dtype for downstream code)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
