import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_fp16_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul kernel (AlphaFold‑3 style) - fused linear projections.
    Steps:
    1. Layer‑norm on input.
    2. Fuse left/right projections, gates, and out‑gate into a single
       half‑precision GEMM.
    3. Apply optional mask, compute left/right gated tensors.
    4. Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ.
    5. Layer‑norm on hidden dimension.
    6. Triton element‑wise multiplication with out‑gate.
    7. Final linear projection back to ``dim``.
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32 → float16 afterwards)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(dtype)  # [B,N,N,dim] half

    B, N, _, _ = x.shape
    M = B * N * N

    # ------------------------------------------------------------------
    # 2. Fused projection + gates + out‑gate (single GEMM)
    # ------------------------------------------------------------------
    # Gather all five weight matrices (no bias)
    w_left_proj = weights["left_proj.weight"].to(dtype)          # (H, dim)
    w_left_gate = weights["left_gate.weight"].to(dtype)          # (H, dim)
    w_right_proj = weights["right_proj.weight"].to(dtype)        # (H, dim)
    w_right_gate = weights["right_gate.weight"].to(dtype)        # (H, dim)
    w_out_gate = weights["out_gate.weight"].to(dtype)            # (H, dim)

    fused_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # shape (5*H, dim)

    # Flatten input for linear
    x_flat = x.view(M, dim)                     # (M, dim)

    # One large linear
    combined = torch.nn.functional.linear(x_flat, fused_weight)  # (M, 5*H)

    # Split
    H = hidden_dim
    left_proj = combined[:, 0:H]
    left_gate_pre = combined[:, H:2*H]
    right_proj = combined[:, 2*H:3*H]
    right_gate_pre = combined[:, 3*H:4*H]
    out_gate_pre = combined[:, 4*H:5*H]

    # Activation
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 3. Optional mask and gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (M, 1)
        mask_flat = mask.view(M, 1).to(dtype)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Triangle multiplication (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Reshape to (B, H, N, N)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # Merge batch and head dimensions
    left_b = left.view(B * H, N, N)
    right_b = right.view(B * H, N, N)

    out_b = torch.bmm(left_b, right_b.transpose(1, 2))               # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Output‑side LayerNorm over hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 6. Element‑wise multiplication w/ out_gate (Triton)
    # ------------------------------------------------------------------
    # Shape match out_norm (B,N,N,H) and out_gate (B,N,N,H)
    out_gate = out_gate.view(B, N, N, H)

    out_norm_flat = out_norm.view(-1)
    out_gate_flat = out_gate.view(-1)
    gated_flat = torch.empty_like(out_norm_flat)

    total_elem = out_norm_flat.numel()
    BLOCK = 65536
    grid = (triton.cdiv(total_elem, BLOCK),)

    elementwise_mul_fp16_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elem,
        BLOCK_SIZE=BLOCK,
    )
    gated = gated_flat.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(dtype)  # (dim, H)
    gated_flat = gated.view(M, H)
    out_half = torch.nn.functional.linear(gated_flat, to_out_w)  # (M, dim)
    out_float = out_half.view(B, N, N, dim).to(torch.float32)

    return out_float

Outgoing TriMul kernel (AlphaFold‑3 style) – fused linear projections.

Algorithm
---------
1️⃣  Layer‑norm on the input feature dimension (dtype=float32 → float16).
2️⃣  Fuse *all* linear projections/gates (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a **single** half‑precision GEMM.
3️⃣  Apply optional mask (only to the left/right projections) and multiply

4️⃣  Triangle multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    realised as a batched half‑precision GEMM
       (B·H, N, N) @ (B·H, N, N)ᵀ .
5️⃣  Layer‑norm over the hidden dimension (still in float16).
6️⃣  Element‑wise product with the out‑gate using a tiny Triton kernel.
7️⃣  Final half‑precision linear projection back to the original channel size.
8️⃣  Return float32 tensor of shape [B, N, N, dim].

All heavy work (matrix multiplications, layer‑norms) is performed in
float16 so that the H100 can fully exploit Tensor‑cores.  The Triton
kernel is required for the element‑wise multiplication step.

    w_left_gate = weights["left_gate.weight"].to(dtype_half)    # (H, dim)
    w_right_proj = weights["right_proj.weight"].to(dtype_half)  # (H, dim)
    w_right_gate = weights["right_gate.weight"].to(dtype_half)  # (H, dim)
    w_out_gate = weights["out_gate.weight"].to(dtype_half)      # (H, dim)

    # (5*H, dim) fused weight matrix
    fused_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )

    # Flatten input for the linear
    x_flat = x.view(M, dim)                                     # (M, dim)

    # One large linear: (M, dim) @ (dim, 5*H)ᵀ → (M, 5*H)
    combined = torch.nn.functional.linear(x_flat, fused_weight)   # (M, 5*H)

    H = hidden_dim
    left_proj      = combined[:, 0:H]
    left_gate_pre  = combined[:, H:2*H]
    right_proj     = combined[:, 2*H:3*H]
    right_gate_pre = combined[:, 3*H:4*H]
    out_gate_pre   = combined[:, 4*H:5*H]

    # Sigmoids for gates
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)                     # (M, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + gate application
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(M, 1).to(dtype_half)               # (M, 1)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    left  = left_proj * left_gate                               # (M, H)
    right = right_proj * right_gate                             # (M, H)

    # ------------------------------------------------------------------
    # 4️⃣ Triangle multiplication (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, H, N, N) → flatten batch & head dimensions
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_b  = left.view(B * H, N, N)                              # (B·H, N, N)
    right_b = right.view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))           # (B·H, N, N)

    # Restore (B, N, N, H)
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output‑side LayerNorm over hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_half)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_half)

    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B, N, N, H) half

    # ------------------------------------------------------------------
    # 6️⃣ Element‑wise multiplication with out_gate (Triton)
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)               # (B, N, N, H) half
    out_norm_flat = out_norm.view(-1)                 # 1‑D half
    out_gate_flat = out_gate.view(-1)                 # 1‑D half
    gated_flat = torch.empty_like(out_norm_flat)      # allocate result

    total_elems = out_norm_flat.numel()
    BLOCK = 65536                                      # ~64 K elements per program
    grid = (triton.cdiv(total_elems, BLOCK),)

    elementwise_mul_fp16_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )
    gated = gated_flat.view(B, N, N, H)               # (B, N, N, H) half

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(dtype_half)   # (dim, H)
    gated_flat = gated.view(M, H)                       # (M, H)
    out_half = torch.nn.functional.linear(gated_flat, to_out_w)  # (M, dim)

    # Return float32 as required by the API
    return out_half.view(B, N, N, dim).to(torch.float32)
