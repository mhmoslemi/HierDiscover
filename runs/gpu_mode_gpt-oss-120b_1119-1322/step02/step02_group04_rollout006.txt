    out_ptr, out_gate_ptr, out_fused_ptr,
    ln_weight_ptr, ln_bias_ptr,
    total_vectors, eps,
    hidden_dim: tl.constexpr):

    # guard
    if pid >= total_vectors:
        return

    off = pid * hidden_dim

    sum_f = tl.float32(0.0)
    sum_sq_f = tl.float32(0.0)
    for i in range(hidden_dim):
        val = tl.load(out_ptr + off + i, dtype=tl.float16)
        v = val.to(tl.float32)
        sum_f += v
        sum_sq_f += v * v

    mean = sum_f / hidden_dim
    var = sum_sq_f / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    for i in range(hidden_dim):

        gate = tl.load(out_gate_ptr + off + i, dtype=tl.float16)

        norm = norm * w.to(tl.float32) + b.to(tl.float32)
        gated = norm * gate.to(tl.float32)
        tl.store(out_fused_ptr + off + i, gated.to(tl.float16))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # input layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias)
    # cast to half
    x_h = x.to(torch.float16)

    # fuse linear transforms
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w], dim=0)  # shape (5*H, dim)

    combined = torch.nn.functional.linear(x_h, combined_weight)  # shape (...,5H)

    # slice
    H = hidden_dim
    left = combined[..., :H]
    left_gate_pre = combined[..., H:2*H]
    right = combined[..., 2*H:3*H]
    right_gate_pre = combined[..., 3*H:4*H]
    out_gate_pre = combined[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # apply mask if needed
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # apply gating
    left = left * left_gate
    right = right * right_gate

    # triangle multiplication
    B, N, _, _ = left.shape
    # permute to (B,H,N,N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    left_flat = left_perm.view(B*hidden_dim, N, N)
    right_flat = right_perm.view(B*hidden_dim, N, N)

    out = torch.bmm(left_flat, right_flat.transpose(1,2))
    # reshape back to (B,N,N,H)
    out = out.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)

    # Flatten for fused kernel
    total_vectors = B * N * N
    out_flat = out.view(total_vectors, hidden_dim)
    out_gate_flat = out_gate.view(total_vectors, hidden_dim)

    # fused layernorm + gate

    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

        out_flat,
        out_gate_flat,
        out_fused,
        to_out_norm_weight,
        to_out_norm_bias,
        total_vectors,
        1e-5,
        hidden_dim,
    )

    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    final = torch.nn.functional.linear(out_fused, to_out_w)   # (total_vectors, dim)
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

        sum_val += f
        sum_sq += f * f
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---- normalize, apply LN weight/bias, gate, and store ----
    for i in range(hidden_dim):
        val = tl.load(out_ptr + off + i, dtype=tl.float16)
        gate = tl.load(out_gate_ptr + off + i, dtype=tl.float16)
        w = tl.load(ln_weight_ptr + i, dtype=tl.float16)
        b = tl.load(ln_bias_ptr + i, dtype=tl.float16)

        norm = (val.to(tl.float32) - mean) * inv_std
        norm = norm * w.to(tl.float32) + b.to(tl.float32)
        gated = norm * gate.to(tl.float32)
        tl.store(out_fused_ptr + off + i, gated.to(tl.float16))

def custom_kernel(data):
    """Outgoing TriMul (AlphaFold‑3) with fused linear projection,
    fused hidden‑dim LayerNorm + gate (Triton) and batched GEMM.

    The implementation reduces the number of large matrix multiplications
    from three to one, fuses the hidden‑dim normalization and gating, and
    keeps the heavy arithmetic in float16 (tensor‑core friendly)."""
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)  ------------------------------------
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections (proj + gate) into a single GEMM --
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                       # half

    # weights (all half)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(torch.float16) # (H, C)
    right_gate_w = weights["right_gate.weight"].to(torch.float16) # (H, C)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)     # (H, C)

    # concat along output dimension (5H, C)
    fused_w = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)  # (5*H, C)

    # single linear = (B·N·N, C) @ (C, 5H)ᵀ  → (B·N·N, 5H)
    fused = torch.nn.functional.linear(x_h, fused_w)               # [B, N, N, 5H]

    # ------------------------------------------------------------------
    # 3. Slice projections/gates and apply mask/gates -----------------
    # ------------------------------------------------------------------
    H = hidden_dim
    # shape [..., 5H]
    left_proj = fused[..., :H]
    left_gate_pre = fused[..., H:2*H]
    right_proj = fused[..., 2*H:3*H]
    right_gate_pre = fused[..., 3*H:4*H]
    out_gate_pre = fused[..., 4*H:5*H]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)        # [B,N,N,H] half

    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate        # [B,N,N,H] half
    right = right_proj * right_gate     # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 4. Core “triangle multiplication”: out = Σₖ left[...,k] * right[...,k]
    #    realised as batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # batched matmul
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B·H, N, N)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Fuse hidden‑dim LayerNorm (to_out_norm) with the out‑gate.
    #    Kernel works on flattened view (B·N·N, H).
    # ------------------------------------------------------------------
    total_vectors = B * N * N
    out_vec = out.view(total_vectors, hidden_dim).contiguous()
    out_gate_vec = out_gate.view(total_vectors, hidden_dim).contiguous()

    # to_out_norm parameters
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)

    out_fused_vec = torch.empty_like(out_vec)  # half

    # launch Triton kernel (one program per vector)
    grid = (total_vectors,)
    fused_norm_gate_kernel[grid](
        out_vec,
        out_gate_vec,
        out_fused_vec,
        to_out_norm_w,
        to_out_norm_b,
        total_vectors,
        1e-5,
        hidden_dim,
    )

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final = torch.nn.functional.linear(out_fused_vec, to_out_w)  # (total_vectors, dim) half
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused hidden‑dim LayerNorm + out‑gate.
#   out       : (V, H)  float16   – result of the triangle multiplication
#   out_gate  : (V, H)  float16   – sigmoid(out_gate(x))
#   out_fused : (V, H)  float16   – (LayerNorm(out) * out_gate)
#   ln_w,ln_b : (H,)    float16   – weight & bias of the hidden‑dim LayerNorm
# The kernel works on a flattened view V = B·N·N.
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_kernel(
    out_ptr,                # float16*  [V, H]
    out_gate_ptr,           # float16*  [V, H]
    out_fused_ptr,          # float16*  [V, H] (output)
    ln_weight_ptr,          # float16*  [H]
    ln_bias_ptr,            # float16*  [H]
    num_vecs,               # int32    total number of vectors V
    eps,                    # float32  epsilon for LayerNorm
    hidden_dim: tl.constexpr,   # compile‑time hidden dimension H
):
    pid = tl.program_id(0)                     # one program per vector
    if pid >= num_vecs:
        return

    # ------------------------------------------------------------------
    # 1) compute mean and variance of the H‑element vector.
    #    Accumulate in float32 for numerical stability.
    # ------------------------------------------------------------------
    off = pid * hidden_dim                     # start offset of the vector
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)
    for i in range(hidden_dim):
        v = tl.load(out_ptr + off + i, dtype=tl.float16)   # half
        f = v.to(tl.float32)
        sum_val += f
        sum_sq  += f * f
    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) apply LayerNorm, multiply by the out‑gate and store.
    # ------------------------------------------------------------------
    for i in range(hidden_dim):
        val  = tl.load(out_ptr + off + i, dtype=tl.float16)
        gate = tl.load(out_gate_ptr + off + i, dtype=tl.float16)
        w    = tl.load(ln_weight_ptr + i, dtype=tl.float16)
        b    = tl.load(ln_bias_ptr + i, dtype=tl.float16)

        # (val - mean) / sqrt(var+eps)
        norm = (val.to(tl.float32) - mean) * inv_std
        # weight & bias of the hidden‑dim LayerNorm
        norm = norm * w.to(tl.float32) + b.to(tl.float32)
        # finally multiply by the sigmoid gate
        gated = norm * gate.to(tl.float32)

        tl.store(out_fused_ptr + off + i, gated.to(tl.float16))

    """Outgoing TriMul (AlphaFold‑3 style).

    1) Layer‑norm on the input channel dimension (float32).
    2) Fuse *all* linear projections (left/right proj + their gates + out‑gate)
       into a single GEMM → saves kernel launch overhead.
    3) Apply optional pair‑wise mask, then left/right element‑wise gates.
    4) Core “triangle multiplication’’:
         out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
       implemented as a batched half‑precision GEMM.
    5) Fuse the hidden‑dim LayerNorm (to_out_norm) with the out‑gate
       using a custom Triton kernel (single pass per (b,i,j) vector).
    6) Final linear projection back to the original channel size.
    7) Return float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # hidden dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                       # [B, N, N, C]  float32
    x_h = x.to(torch.float16)               # half for the compute‑intensive part

    # ------------------------------------------------------------------
    # 2) Fuse all linear projections (proj + gate) into one GEMM
    # ------------------------------------------------------------------
    # model weights (half precision)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(torch.float16) # (H, C)
    right_gate_w = weights["right_gate.weight"].to(torch.float16) # (H, C)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)   # (H, C)

    # concatenate → shape (5*H, C)
    fused_w = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)

    # single linear layer: (B·N·N, C) @ (C, 5H)ᵀ → (B·N·N, 5H)

    H = hidden_dim

    left_proj   = fused[..., :H]
    left_gate_p = fused[..., H:2*H]

    out_gate_p  = fused[..., 4*H:5*H]

    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)          # [B,N,N,H] half

    # ------------------------------------------------------------------

        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj * left_gate    # [B,N,N,H] half
    right = right_proj * right_gate # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 4) Triangle multiplication via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B, H, N, N) for the batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_perm.view(B * hidden_dim, N, N)    # (B·H, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Fuse hidden‑dim LayerNorm (to_out_norm) with the out‑gate
    # ------------------------------------------------------------------
    total_vecs = B * N * N
    out_vec       = out.view(total_vecs, hidden_dim).contiguous()          # (V,H) half
    out_gate_vec  = out_gate.view(total_vecs, hidden_dim).contiguous()    # (V,H) half

    # to_out_norm parameters (half)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)   # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)     # (H,)

    out_fused_vec = torch.empty_like(out_vec)  # (V,H) half

    # launch Triton kernel – one program per vector
    grid = (total_vecs,)
    fused_norm_gate_kernel[grid](
        out_vec,
        out_gate_vec,
        out_fused_vec,
        to_out_norm_w,
        to_out_norm_b,
        total_vecs,
        1e-5,
        hidden_dim=hidden_dim,          # constexpr argument
    )

    # ------------------------------------------------------------------
    # 6) Final linear projection back to original channel size
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final = torch.nn.functional.linear(out_fused_vec, to_out_w)   # (V, dim) half
    final = final.view(B, N, N, dim).to(torch.float32)            # float32 output

    return final
