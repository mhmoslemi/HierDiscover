def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Prepare weights half
    w_norm = weights["norm.weight"].half()
    b_norm = weights["norm.bias"].half()
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()
    w_to_out_norm = weights["to_out_norm.weight"].half()
    b_to_out_norm = weights["to_out_norm.bias"].half()
    w_to_out = weights["to_out.weight"].half()

    # 1. Input LayerNorm in FP16
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w_norm,
        bias=b_norm,
    ).half()

    # 2. Linear projections to hidden dim
    # flatten is handled by F.linear
    left = torch.nn.functional.linear(x_norm, w_left_proj)  # fp16
    right = torch.nn.functional.linear(x_norm, w_right_proj)

    # 3. Gating values and output gate
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, w_left_gate))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, w_right_gate))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, w_out_gate))

    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Ensure contiguity
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        # allocate outputs
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elems = left.numel()
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        fused_mask_gate_kernel[grid](
            left, right, mask_fp16,
            left_gate, right_gate,
            left_out, right_out,
            total_elems,
            H=hidden,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out
    else:
        left = left * left_gate
        right = right * right_gate
        if mask is not None:
            # if mask tensor provided but nomask True (i.e., maybe mask is all ones). If mask present but we ignore, fine.
            left = left * mask.unsqueeze(-1).to(torch.float16)
            right = right * mask.unsqueeze(-1).to(torch.float16)

    # 5. Batched GEMM: left @ right^T
    # Rearrange to (B, hidden, N, N)
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # matmul
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # back to (B, N, N, hidden)
    out = out.permute(0, 2, 3, 1).contiguous()

    # 6. Second layer norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    # Apply out_gate
    out = out * out_gate
    # Final projection
    out = torch.nn.functional.linear(out, w_to_out)

    # Cast to float32
    return out.to(torch.float32)
```

Add Triton kernel:

```
@triton.jit
def fused_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    total_elements: tl.int64,
    H: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load
    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0))

    # compute mask index
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=tl.float16(0))

    l = l * lg * m
    r = r * rg * m

    tl.store(out_left_ptr + offs, l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)

"""
Outgoing TriMul (Triangular Multiplicative Update) – fast half‑precision
implementation.

Algorithm
---------
1️⃣  LayerNorm on the input (half‑precision weights/biases).
2️⃣  Linear projections (FP16) → left / right tensors of shape
    [B, N, N, H]  (H = hidden_dim).
3️⃣  Compute three gating tensors (sigmoid of linear projections).
4️⃣  Fuse optional binary mask with left/right gating using a small
    Triton kernel (no materialised broadcasted mask tensor).
5️⃣  Batched matrix multiplication:
        out[b, h] = left[b, h] @ right[b, h].T   (shape B×H×N×N)
    Implemented with a single `torch.matmul` which utilizes cuBLAS
    Tensor‑Core kernels.
6️⃣  LayerNorm over the hidden dimension (FP16 weights/biases).
7️⃣  Apply the output gate (sigmoid) and a final linear projection back
    to the model dimension.
8️⃣  Return FP32 output.

All heavy arithmetic is performed in float16 to maximise H100 Tensor‑Core
throughput; only the final result is upcast to float32.  The Triton
kernel does the element‑wise mask‑gate fusion, satisfying the
requirement of having a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr,         # fp16 *
    right_ptr,        # fp16 *
    mask_ptr,         # fp16 *
    left_gate_ptr,    # fp16 *
    right_gate_ptr,   # fp16 *
    out_left_ptr,     # fp16 *
    out_right_ptr,    # fp16 *
    total_elems: tl.int64,
    H: tl.constexpr,          # hidden dimension
    BLOCK: tl.constexpr,      # threads per program
):
    """Fuse mask and per‑channel gating.

    Tensors are flattened in the order (b, i, j, h):
        offset = ((b*N + i)*N + j)*H + h
    The mask has shape (B, N, N) → offset_mask = ((b*N + i)*N + j).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load raw values (zero‑fill out‑of‑range threads)
    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Broadcast mask over the hidden dimension
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=tl.float16(0.0))

    # Fuse three element‑wise multiplies
    l = l * lg * m
    r = r * rg * m

    tl.store(out_left_ptr + offs,  l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [B, N, N, dim],   float32
            mask         : torch.Tensor, shape [B, N, N],      float32 or bool
            weights      : dict of torch.Tensor model parameters (FP32)
            config       : dict with keys
                           - "dim"         (int)
                           - "hidden_dim"  (int)
                           - "nomask"      (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast model weights to half‑precision once
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"].half()
    b_norm = weights["norm.bias"].half()
    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()
    w_to_out_norm = weights["to_out_norm.weight"].half()
    b_to_out_norm = weights["to_out_norm.bias"].half()
    w_to_out = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    # LayerNorm can work directly on fp16 tensors when its weight/bias are fp16.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w_norm,
        bias=b_norm,
    ).half()                      # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B, N, N, hidden]  (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x, w_left_proj)   # fp16
    right = F.linear(x, w_right_proj) # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Gating tensors (sigmoid of linear projections) – still FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))   # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # mask → FP16 and make contiguous
        mask_fp16 = mask.to(dtype=torch.float16).contiguous()
        # Ensure all tensors are contiguous for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elements = left.numel()
        BLOCK = 1024  # threads per program (tunable)
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        _mask_gate_kernel[grid](
            left, right,
            mask_fp16,
            left_gate, right_gate,
            left_out, right_out,
            total_elements,
            H=hidden,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out
    else:
        # No mask → just apply per‑channel gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out = left @ rightᵀ  (fp16)
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) so that torch.matmul treats the first
    # two dimensions as batch dimensions.
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous() # B, H, N, N

    # Matrix multiplication with Tensor‑Core support
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # B, H, N, N

    # Back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, w_to_out)   # still fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast to FP32 for downstream consumption
    # ------------------------------------------------------------------
    return out.to(torch.float32)
