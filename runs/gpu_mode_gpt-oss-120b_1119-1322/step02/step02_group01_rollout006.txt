custom_kernel – Triton‑accelerated “outgoing” TriMul forward pass
================================================================
We follow the reference PyTorch implementation and fuse the cubic
N³ update into a single batched GEMM kernel:

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
                = (left @ rightᵀ)   for each hidden slice h

All heavy work (the GEMM) is performed in FP16 with FP32 accumulation.
Block sizes (128×128×64) and 8 warps are chosen to fully exploit
Tensor‑cores on an H100.  The rest of the TriMul pipeline (LayerNorm,
linear projections, sigmoid gates and the final projection) uses
high‑performance PyTorch kernels.  The final result is cast back to
float32.

The kernel processes B·H independent matrix multiplies in a single
launch (grid‑dim‑2 = B·H).  Memory traffic is minimal (inputs/outputs are
FP16) and edge cases are masked.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   (A,B : [batch, N, N]  in FP16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B: batch, row (k), col (n) strides
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    """Batched FP16 GEMM (A @ Bᵀ) with FP32 accumulation."""
    pid_batch = tl.program_id(2)      # which (batch * hidden) matrix
    pid_m = tl.program_id(0)          # block row of C
    pid_n = tl.program_id(1)          # block column of C

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch*hidden) matrix
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # ------------------------------------------------------------------
    # Pointers to the start of the current tile (k = 0)
    # ------------------------------------------------------------------
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiling)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        cur_off = k_tile * BLOCK_K

        # Load tiles of A (shape BLOCK_M x BLOCK_K) and Bᵀ
        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation (Tensor‑core friendly)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back FP16 result (cast from FP32)
    # ------------------------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict with the model parameters (float32)
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – all heavy linear work stays in half precision
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    def _to_half(t):
        return t.to(torch.float16)

    # Linear projections (bias‑free)
    left = F.linear(x_fp16, _to_half(weights["left_proj.weight"]))   # [B,N,N,H]
    right = F.linear(x_fp16, _to_half(weights["right_proj.weight"]))

    # Gating logits → sigmoid (still FP16)
    left_gate = F.linear(x_fp16, _to_half(weights["left_gate.weight"])).sigmoid()
    right_gate = F.linear(x_fp16, _to_half(weights["right_gate.weight"])).sigmoid()
    out_gate = F.linear(x_fp16, _to_half(weights["out_gate.weight"])).sigmoid()

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] → [B,N,N,1] and cast to fp16
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, N, N, hidden_dim (H == hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Core cubic update using Triton (batched GEMM A @ Bᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then collapse the batch*hidden dimension
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(-1, N, N)    # [B*H, N, N]  fp16
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]  fp16

    total = left_flat.shape[0]           # B * H independent matrix multiplies

    # Allocate output buffer (fp16)
    out_flat = torch.empty_like(left_flat)  # [B*H, N, N]

    # ------------------------------------------------------------------
    # Kernel launch configuration (tuned for H100 Tensor‑cores)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i‑dimension)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j‑dimension)
        total,                     # one program per (batch * hidden)
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K
        left_flat.stride(0),       # stride_ab (batch stride)
        left_flat.stride(1),       # stride_am (row stride)
        left_flat.stride(2),       # stride_ak (col stride)
        right_flat.stride(0),      # stride_bb (batch stride)
        right_flat.stride(2),      # stride_bk (row‑stride for Bᵀ)
        right_flat.stride(1),      # stride_bn (col‑stride for Bᵀ)
        out_flat.stride(0),        # stride_cb (batch stride)
        out_flat.stride(1),        # stride_cm (row stride)
        out_flat.stride(2),        # stride_cn (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps → 256 threads per block
        num_stages=2,              # double‑buffering of tiles
    )

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, H] (still FP16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (float32), out‑gate scaling, final proj
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Multiply by the out‑gate (cast to float32)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final projection back to original dimension D (float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
