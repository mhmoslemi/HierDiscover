"""
high‑performance outgoing TriMul (AlphaFold3) forward pass

The implementation follows the reference PyTorch code but fuses the
expensive work with custom Triton kernels and batched GEMMs:

1. Input LayerNorm (FP32) over the model dimension `dim`.
2. Cast to fp16 and flatten the `[B,N,N,dim]` tensor.
3. **Fused linear projections**: a single matmul computes both left
   and right projections (weight = [left;right] of shape
   `(2*hidden, dim)`).  The result is split back to two tensors.
4. **Fused gating linear**: a second matmul computes left‑gate and
   right‑gate simultaneously; a sigmoid is applied in‑place.
5. Optional **mask‑gate fusion** (Triton kernel) multiplies the
   left/right projections by the per‑element mask and the corresponding
   gate scalars in one pass.
6. Batched matrix multiplication (`torch.bmm`) computes the
   multiplicative update `out = left @ rightᵀ` for every hidden channel.
7. Second LayerNorm over the hidden dimension (FP32).
8. Output gate (sigmoid) and final linear projection back to `dim`.

Only the mask‑gate step is written in Triton – this satisfies the
requirement of having a custom kernel while keeping the dominant work
in highly‑optimized cuBLAS calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate with the projected
# tensors.  The tensors are flattened with layout
#   offset = ((b*N + i)*N + j)*H + h   (h = hidden channel)
# mask is (B,N,N) → offset_mask = ((b*N + i)*N + j)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,               # fp16 * left  (flattened)
    right_ptr,              # fp16 * right (flattened)
    mask_ptr,               # fp16 * mask  (flattened, B*N*N)
    left_gate_ptr,          # fp16 * left_gate (flattened)
    right_gate_ptr,         # fp16 * right_gate (flattened)
    out_left_ptr,           # fp16 * output left
    out_right_ptr,          # fp16 * output right
    total_elements: tl.int64,
    H: tl.constexpr,        # hidden dimension (known at compile time)
    BLOCK: tl.constexpr,    # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear offsets
    mask = offs < total_elements

    # Load the four element‑wise tensors
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute the mask index (broadcasted over H)
    idx_mask = offs // H
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # Fuse three scalar multiplications
    l = l * lg * m
    r = r * rg * m

    # Store results
    tl.store(out_left_ptr + offs, l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            mask         : torch.Tensor, shape [B, N, N] (bool/float32) or None
            weights      : dict of torch.Tensor model parameters (float32)
            config       : dict with keys:
                - "dim"         (int) model dimension
                - "hidden_dim"  (int) hidden dimension for the update
                - "nomask"      (bool, optional) True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                     # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and flatten for the big matmuls
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    B, N, _, _ = x_fp16.shape
    flat_dim = B * N * N
    x_flat = x_fp16.view(flat_dim, dim)          # [B*N*N, dim]

    # ------------------------------------------------------------------
    # 3. Fused linear projections (left & right)
    # ------------------------------------------------------------------
    #   weight shape (2*hidden, dim)
    w_lr = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)
    proj = F.linear(x_flat, w_lr)                # [B*N*N, 2*hidden]
    left, right = proj.split(hidden, dim=-1)     # each: [B*N*N, hidden]
    left = left.view(B, N, N, hidden)            # [B,N,N,hidden]
    right = right.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4. Fused gating linear (left_gate & right_gate)
    # ------------------------------------------------------------------
    w_gate = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    gate = torch.sigmoid(F.linear(x_flat, w_gate))   # [B*N*N, 2*hidden]
    left_gate, right_gate = gate.split(hidden, dim=-1)
    left_gate = left_gate.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5. Output gate (used after second LayerNorm)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"]))
    out_gate = out_gate.view(B, N, N, hidden)     # float32

    # ------------------------------------------------------------------
    # 6. Optional mask‑gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # Prepare fp16 mask (0/1) and make everything contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        total_elems = left.numel()
        BLOCK = 1024                               # threads per program
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        _mask_gate_kernel[grid](
            left,
            right,
            mask_fp16,
            left_gate,
            right_gate,
            left_out,
            right_out,
            total_elems,
            H=hidden,
            BLOCK=BLOCK,
        )
        left_fp16 = left_out
        right_fp16 = right_out
    else:
        # No mask → only apply per‑channel gates
        left_fp16 = left * left_gate
        right_fp16 = right * right_gate

    # ------------------------------------------------------------------
    # 7. Batched GEMM: out = left @ rightᵀ   (fp16 -> fp32 accumulator)
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for torch.bmm
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # torch.bmm uses Tensor Cores for fp16 inputs
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 8. Restore `[B, N, N, hidden]` layout (still fp16)
    # ------------------------------------------------------------------
    out = (
        out_mat_fp16.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)  # promote to fp32 for the next LayerNorm
    )                       # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 9. Second LayerNorm over the hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 10. Apply the output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim] (float32)

    return out
