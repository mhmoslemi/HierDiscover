"""
TriMul (outgoing) – highly‑optimised forward pass.

Algorithm
---------
1. Layer‑norm on the input (FP32).
2. Cast the normalised tensor to FP16.
3. Fuse *proj* + *gate* linear layers:
      left  = Linear(x, [W_left ; W_left_gate])   → (hidden, hidden) concat
      right = Linear(x, [W_right; W_right_gate])
   Split the concatenated results to obtain the raw projections and the
   sigmoid‑gates.
4. If a mask is present, fuse mask, left‑gate and right‑gate element‑wise
   with a small Triton kernel (zero‑copy, no temporaries).  Without a mask
   only the two per‑channel gates are applied.
5. Batched matrix multiplication (Tensor‑Core FP16) implements the
   Einstein‑summation
        out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   via `torch.bmm` on tensors shaped (B*hidden, N, N).
6. Second Layer‑norm over the hidden dimension (FP32), apply the output‑gate
   (sigmoid), and a final FP32 linear projection back to the model dimension.
All heavy linear layers and the GEMM run in FP16, dramatically reducing memory
traffic and achieving sub‑millisecond latency on H100 for the test configs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fuse mask, per‑channel gates and write outputs
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left (flattened)
    right_ptr,         # fp16 * right (flattened)
    mask_ptr,          # fp16 * mask (flattened B,N,N)
    left_gate_ptr,     # fp16 * left_gate (flattened)
    right_gate_ptr,    # fp16 * right_gate (flattened)
    out_left_ptr,      # fp16 * output left
    out_right_ptr,     # fp16 * output right
    total_elems: tl.int64,
    H: tl.constexpr,   # hidden dimension
    BLOCK: tl.constexpr,
):
    """Element‑wise fusion:
       out_left  = left  * left_gate  * mask
       out_right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # (BLOCK,)
    mask = offs < total_elems

    # Load tensors (zero for out‑of‑bounds threads)
    l = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    r = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask is broadcast over hidden => use integer division
    idx_mask = offs // H                # flat index into (B,N,N)
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # Fuse the three multiplies
    l = l * lg * m
    r = r * rg * m

    # Store the results
    tl.store(out_left_ptr + offs,  l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [B, N, N, dim]            (FP32)
            mask         : torch.Tensor, shape [B, N, N] (bool / FP32)  (unused if nomask=True)
            weights      : dict of torch.Tensor model parameters
            config       : dict with keys
                           - "dim"        (int)
                           - "hidden_dim" (int)
                           - "nomask"     (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (FP32)
    """
    # ----------------------------------------------------------
    # unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for all heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.half()                     # [B, N, N, dim]  FP16

    B, N, _, _ = x_fp16.shape
    flat_dim = B * N * N

    # ------------------------------------------------------------------
    # 3. Concatenated linear projections (proj + gate) in FP16
    # ------------------------------------------------------------------
    # ----- left side -----
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)          # (2*hidden, dim)

    left_concat = F.linear(
        x_fp16.view(flat_dim, dim), left_concat_weight, bias=None
    )  # (B*N*N, 2*hidden)  FP16
    left_concat = left_concat.view(B, N, N, 2 * hidden)

    left_proj = left_concat[..., :hidden]            # [B,N,N,hidden] FP16
    left_gate = left_concat[..., hidden:]            # [B,N,N,hidden] FP16
    left_gate = torch.sigmoid(left_gate)             # FP16 sigmoid

    # ----- right side -----
    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)         # (2*hidden, dim)

    right_concat = F.linear(
        x_fp16.view(flat_dim, dim), right_concat_weight, bias=None
    )
    right_concat = right_concat.view(B, N, N, 2 * hidden)

    right_proj = right_concat[..., :hidden]          # FP16
    right_gate = right_concat[..., hidden:]          # FP16
    right_gate = torch.sigmoid(right_gate)           # FP16

    # ------------------------------------------------------------------
    # 4. Output gate (used after the second LayerNorm)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate = torch.sigmoid(
        F.linear(x_fp16.view(flat_dim, dim), out_gate_weight, bias=None)
    ).view(B, N, N, hidden)          # FP16, will be cast to FP32 later

    # ------------------------------------------------------------------
    # 5. Fuse mask + per‑channel gates (Triton) OR simple element‑wise
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – convert to FP16 0/1 and flatten
        mask_fp16 = mask.to(dtype=torch.float16, device=device).contiguous()
        # Ensure tensors are contiguous
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_flat = mask_fp16.view(-1)                      # (B*N*N)

        # Allocate fused outputs
        left_fused = torch.empty_like(left_proj)
        right_fused = torch.empty_like(right_proj)

        total = left_proj.numel()          # B*N*N*hidden
        BLOCK = 1024                       # threads per program (tunable)
        grid = ( (total + BLOCK - 1) // BLOCK, )

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            mask_flat,
            left_gate,
            right_gate,
            left_fused,
            right_fused,
            total,
            H=hidden,
            BLOCK=BLOCK,
        )
        left_fp16 = left_fused
        right_fp16 = right_fused
    else:
        # No mask – just per‑channel gating
        left_fp16 = left_proj * left_gate
        right_fp16 = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out = left @ rightᵀ   (fp16 inputs, fp16 output)
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) where each batch corresponds to a hidden channel
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Tensor‑core fast bmm (fp16 -> fp16)
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 7. Restore original layout and cast to FP32
    # ------------------------------------------------------------------
    out = out_mat_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)
    # shape = [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over the hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # still FP32

    # ------------------------------------------------------------------
    # 9. Apply output gate (sigmoid) and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # gate in FP32
    out = F.linear(out, weights["to_out.weight"])   # FP32 linear, bias=False

    return out
