"""
TriMul (outgoing) forward pass – Triton‑accelerated cubic contraction.

Algorithm
---------
1. Layer‑norm over the input channels.
2. Fuse the five linear maps (left/right projections, three gates) into a
   single half‑precision matmul:  (B·N²×dim) @ (dim×5·hidden_dim) → (B·N²×5·hidden_dim).
   The result is split and the three gate tensors are passed through a sigmoid.
3. Apply the (optional) pair mask and fuse the gates with the projections.
   All tensors are kept in fp16 for the heavy‑weight N³ contraction.
4. Compute the contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h]
   for every head *h* using a custom batched GEMM kernel.  The kernel works on
   the original (B,N,N,hidden) layout – no extra permutation or copy is needed.
5. Second layer‑norm on the hidden dimension, scale by the out‑gate and a final
   linear projection back to the original feature dimension.

Only the cubic contraction (step 4) is performed in Triton; the remaining
operations rely on highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) over (batch, hidden) × (N × N) matrices.
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                                 # matrix dimensions
    stride_am, stride_ak, stride_ab, stride_ah,   # A: row, col, batch, hidden strides
    stride_bk, stride_bn, stride_bb, stride_bh,   # B: col (k), row (n), batch, hidden strides
    stride_cm, stride_cn, stride_cb, stride_ch,   # C: row, col, batch, hidden strides
    HIDDEN_DIM: tl.constexpr,                # number of heads (hidden_dim)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many independent batches.
    A has shape (batch, M, K, hidden), B has shape (batch, N, K, hidden)
    (the hidden dimension is folded into the batch dimension by the launch grid).
    The kernel receives a *combined* batch index `pid = b * H + h`.
    """
    pid = tl.program_id(0)          # combined batch (batch * hidden)
    pid_m = tl.program_id(1)        # tile row index
    pid_n = tl.program_id(2)        # tile col index

    # ---- decode combined batch id into (batch, head) ---------------------------------
    head = pid % HIDDEN_DIM
    batch = pid // HIDDEN_DIM

    # Base pointers for the (batch, head) pair
    a_base = a_ptr + batch * stride_ab + head * stride_ah
    b_base = b_ptr + batch * stride_bb + head * stride_bh
    c_base = c_ptr + batch * stride_cb + head * stride_ch

    # ---- tile offsets -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the K dimension ----------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile (transposed view K × N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M,K)·(K,BLOCK_N) → (BLOCK_M,BLOCK_N)

    # ---- store result -----------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point: custom Triton‑accelerated TriMul forward.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N]   (optional, can be None)
        - weights_dict : mapping of the required weight tensors
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------- #
    # Unpack inputs
    # ----------------------------------------------------------------------- #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # often called c_z

    # ----------------------------------------------------------------------- #
    # 1️⃣ LayerNorm over the last feature dimension (float32)
    # ----------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]  (float32)

    B, N, _, _ = x.shape

    # ----------------------------------------------------------------------- #
    # 2️⃣ Fuse the five linear maps into a single half‑precision matmul
    # ----------------------------------------------------------------------- #
    x_flat = x.reshape(B * N * N, dim).contiguous()        # (B·N², dim)

    # Concatenate the weight matrices (shape: (5·hidden, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )   # (5·hidden, dim)

    # Cast to fp16 for speed
    fused_weight = fused_weight.to(torch.float16)
    x_flat = x_flat.to(torch.float16)

    fused_out = F.linear(x_flat, fused_weight)          # (B·N², 5·hidden)
    H = hidden_dim

    # Split and apply sigmoid to the three gate parts
    left_proj_flat  = fused_out[:, 0:H]
    left_gate_flat  = fused_out[:, H:2 * H]
    right_proj_flat = fused_out[:, 2 * H:3 * H]
    right_gate_flat = fused_out[:, 3 * H:4 * H]
    out_gate_flat   = fused_out[:, 4 * H:5 * H]

    left_gate_flat = torch.sigmoid(left_gate_flat)
    right_gate_flat = torch.sigmoid(right_gate_flat)
    out_gate_flat = torch.sigmoid(out_gate_flat)

    # Reshape back to (B, N, N, H)
    left_proj = left_proj_flat.view(B, N, N, H)
    left_gate = left_gate_flat.view(B, N, N, H)
    right_proj = right_proj_flat.view(B, N, N, H)
    right_gate = right_gate_flat.view(B, N, N, H)
    out_gate = out_gate_flat.view(B, N, N, H)

    # ----------------------------------------------------------------------- #
    # 3️⃣ Apply optional mask and fuse gates with the projections
    # ----------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1], fp16
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure fp16 (they already are)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----------------------------------------------------------------------- #
    # 4️⃣ Core N³ contraction via a custom batched GEMM kernel (A @ Bᵀ)
    # ----------------------------------------------------------------------- #
    # Output: [B, N, N, H]  (float32 accumulator)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides for the left, right and output tensors
    # left / right have shape (B, N, N, H)  → strides: (N*N*H, N*H, H, 1)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    stride_am = s_left[1]          # row‑stride i
    stride_ak = s_left[2]          # col‑stride k
    stride_ab = s_left[0]          # batch‑stride (B)
    stride_ah = s_left[3]          # hidden‑stride (always 1)

    stride_bk = s_right[2]         # k‑stride (same as left)
    stride_bn = s_right[1]         # row‑stride j
    stride_bb = s_right[0]         # batch‑stride (B)
    stride_bh = s_right[3]         # hidden‑stride (1)

    stride_cm = s_out[1]           # row‑stride i for output
    stride_cn = s_out[2]           # col‑stride j for output
    stride_cb = s_out[0]           # batch‑stride (B)
    stride_ch = s_out[3]           # hidden‑stride (1)

    # Tile sizes – a good starting point for H100 tensor‑cores.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                              # combined (batch, head)
        (N + BLOCK_M - 1) // BLOCK_M,                # tiles over rows
        (N + BLOCK_N - 1) // BLOCK_N,                # tiles over cols
    )

    _batched_gemm_kernel[grid](
        left,
        right,
        out,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_cb,
        stride_ch,
        hidden_dim,            # compile‑time constant for the kernel
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ----------------------------------------------------------------------- #
    # 5️⃣ Second LayerNorm, out‑gate and final projection
    # ----------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )   # still float32

    out_gate = out_gate.to(out.dtype)
    out = out * out_gate

    # Final projection back to the original feature dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
