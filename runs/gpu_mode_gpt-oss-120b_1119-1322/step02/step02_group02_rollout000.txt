"""
TriMul (outgoing) – fast forward.
================================

The heavy part of the outgoing TriMul operator is a batched matrix
multiplication

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]                (1)

for each hidden channel *d* (hidden_dim).  This can be written as a
batched GEMM `C = A @ Bᵀ` with `A = left`, `B = right`.  The implementation
below follows the reference PyTorch model but

* keeps the LayerNorms, linear projections and the three gates in FP32
  (numerical stability),
* casts the gated tensors to **FP16** only for the GEMM,
* launches a hand‑written Triton kernel that computes the GEMM in FP16
  with FP32 accumulation and writes the result directly in FP32 (no extra
  cast after the kernel),
* finishes the forward pass with a second LayerNorm, an output‑gate and a
  final linear projection back to the original dimension.

All operations are fused as much as possible while still keeping the
high‑throughput Triton kernel for the compute‑bound (1) step.  The kernel
uses `BLOCK_M = BLOCK_N = 128` and `BLOCK_K = 64`, 8 warps per program
instance and works on H100 Tensor‑Core hardware.

The function `custom_kernel` matches the required signature:
    (input, mask, weights, config) → output
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM with FP16 inputs and FP32 accumulation.
#   A : (batch_h, M, K)   – left tensor   (M = N, K = N)
#   B : (batch_h, N, K)   – right tensor  (rows=j, cols=k)
#   C : (batch_h, M, N)   – output tensor (i, j)
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, N, K) – read as Bᵀ
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    One program instance per (batch, tile_m, tile_n).
    The accumulator is FP32, the output is stored as FP32.
    """
    batch_id = tl.program_id(0)          # batch dimension
    pid_m    = tl.program_id(1)          # tile along M
    pid_n    = tl.program_id(2)          # tile along N

    # --------------------------------------------------------------------
    # Compute offsets for the current tile
    # --------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch (stride in elements)
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A tile (M × K) -------------------------------------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                 # fp16 load

        # ---- Load Bᵀ tile (K × N) -----------------------------------------------
        # B is stored as (batch, N, K); we need Bᵀ[k, n] = B[batch, n, k]
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                 # fp16 load

        # ---- Accumulate -----------------------------------------------------------
        acc += tl.dot(a, b, out_dtype=tl.float32)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------------
    # Write back the result as FP32
    # --------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (bool/float, may be None)
        - weights      : dict of torch.Tensors (model parameters, FP32)
        - config       : dict, must contain "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0. unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [bs, N, N, dim]  FP32

    # -----------------------------------------------------------------------
    # 2. Linear projections (FP32) -------------------------------------------
    #    left = x @ left_proj.Wᵀ    (bias‑free)
    #    right = x @ right_proj.Wᵀ
    # -----------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_w)        # [bs, N, N, hidden_dim]  FP32
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------------
    # 3. Optional mask (broadcast to hidden dim)
    # -----------------------------------------------------------------------
    if not nomask:
        # mask shape: [bs, N, N] → [bs, N, N, 1]
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # FP32
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------------
    # 4. Gating (sigmoid) – still FP32
    # -----------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [bs,N,N,hidden_dim] FP32
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # FP32, will be used later

    # -----------------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 6. Cast to FP16 for the compute‑heavy GEMM
    # -----------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # -----------------------------------------------------------------------
    # 7. Fold hidden_dim into the batch dimension → (bs*hidden_dim, N, N)
    # -----------------------------------------------------------------------
    #   left_h : [bs, N, N, hidden_dim] → [bs*hidden_dim, N, N]
    left_h  = left_h.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Allocate output for the GEMM (FP32, no extra cast required)
    out_h = torch.empty((bs * hidden_dim, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------------
    # 8. Launch the Triton batched GEMM kernel
    # -----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64                     # larger K tile reduces loop overhead

    # Strides in *elements* (not bytes)
    a_stride_batch, a_stride_m, a_stride_k = left_h.stride()   # (batch, M, K)
    b_stride_batch, b_stride_n, b_stride_k = right_h.stride()  # (batch, N, K)
    c_stride_batch, c_stride_m, c_stride_n = out_h.stride()   # (batch, M, N)

    grid = (
        bs * hidden_dim,                         # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,            # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,            # tiles along N (j)
    )

    batched_gemm_fp16_kernel[grid](
        left_h,
        right_h,
        out_h,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # tuned for H100 Tensor‑Cores
    )

    # -----------------------------------------------------------------------
    # 9. Reshape back to [bs, N, N, hidden_dim] (FP32)
    # -----------------------------------------------------------------------
    out = out_h.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [bs,N,N,hidden_dim]

    # -----------------------------------------------------------------------
    # 10. Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # Apply the output gate (still FP32)
    out = out * out_gate

    # Final linear projection back to original dimension
    to_out_w = weights["to_out.weight"]          # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # FP32 result [bs,N,N,dim]

    return out
