"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The forward consists of three stages:

1️⃣ **Pre‑processing (PyTorch)**  
   • Layer‑norm on the input.  
   • A single fused linear layer that computes the two projections
     (left / right) **and** the three gating projections (left_gate,
     right_gate, out_gate) in one `torch.nn.functional.linear`.  This
     saves five separate matmuls.  
   • Optional mask is applied only to the two projected tensors.  
   • Sigmoid gates are applied (still fp16).

2️⃣ **Core cubic update (Triton)**  
   For each batch element and each hidden channel we need

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   which is exactly a batched matrix‑multiply `C = A @ Bᵀ` where
   `A` and `B` are `N×N` fp16 matrices and the accumulation is done in
   fp32 for numerical stability.  The kernel processes **B·H** independent
   GEMMs in parallel with a tile size (64 × 64) and K‑tile = 64 – a
   configuration that fully utilizes H100 Tensor‑cores while keeping the
   shared‑memory footprint modest.

3️⃣ **Post‑processing (PyTorch)**  
   • Layer‑norm over the hidden dimension (fp32).  
   • Multiplication by `out_gate`.  
   • Final linear projection back to the original channel dimension.

Only the GEMM (step 2) is written in Triton; the remaining operations are
performed with highly‑optimized PyTorch kernels.  The implementation
supports both masked / un‑masked inputs and works for all configurations
used in the tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_ab, stride_am, stride_ak,   # A batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B batch, col(k), row(n) strides
    stride_cb, stride_cm, stride_cn,   # C batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ
    A : [batch, M, K]   (fp16)
    B : [batch, N, K]   (fp16)  -- logically we load it as Bᵀ with strides
    C : [batch, M, N]   (fp16, accumulates in fp32)
    """
    pid_batch = tl.program_id(2)               # which (batch*hidden) matrix
    pid_m = tl.program_id(0)                   # block row of C
    pid_n = tl.program_id(1)                   # block col of C

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this batch element
    # --------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # --------------------------------------------------------------
    # Pointers to the start of the current tile (k = 0)
    # --------------------------------------------------------------
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiling)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result (cast to fp16)
    # --------------------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse all five linear layers (proj + three gates) into one matmul
    # ------------------------------------------------------------------
    # Weights have shape (out_features, in_features) i.e. (hidden_dim, dim)
    # Concatenate them along the output dimension.
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [5*hidden_dim, dim]

    # Perform a single fused linear projection
    fused = F.linear(x_fp16, concat_weight)  # [B, N, N, 5*hidden_dim] (fp16)

    # Split the concatenated result
    h = hidden_dim
    left = fused[..., :h]               # [B, N, N, h]
    right = fused[..., h : 2 * h]       # [B, N, N, h]
    left_gate = fused[..., 2 * h : 3 * h].sigmoid()   # fp16
    right_gate = fused[..., 3 * h : 4 * h].sigmoid()
    out_gate = fused[..., 4 * h :].sigmoid()         # fp16, kept for later

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)          # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis and make the matrix
    # layout contiguous (row‑major) for maximal Tensor‑core efficiency.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (fp16)
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N] (fp16)

    left_flat = left_t.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_t.view(-1, N, N)

    total = left_flat.shape[0]          # B * H

    # Output buffer for the cubic update (fp16)
    out_flat = torch.empty_like(left_flat)  # [B*H, N, N]

    # ------------------------------------------------------------------
    # 7) Triton kernel launch (batched GEMM A @ Bᵀ)
    # ------------------------------------------------------------------
    # Tile sizes – tuned for the H100 Tensor‑cores.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i‑dimension)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j‑dimension)
        total,                     # one program per (batch*hidden) matrix
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K
        left_flat.stride(0),       # stride_ab (batch stride)
        left_flat.stride(1),       # stride_am (row stride)
        left_flat.stride(2),       # stride_ak (col stride)
        right_flat.stride(0),      # stride_bb
        right_flat.stride(2),      # stride_bk (col‑stride for Bᵀ)
        right_flat.stride(1),      # stride_bn (row‑stride for Bᵀ)
        out_flat.stride(0),        # stride_cb
        out_flat.stride(1),        # stride_cm
        out_flat.stride(2),        # stride_cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to the original layout
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 9) Hidden‑dim LayerNorm (float32) and final projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # out_gate was computed in fp16; bring it to fp32 before multiplication
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear projection back to dim (float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
```

This implementation fuses the five inexpensive linear layers into a single matmul, uses a larger (128 × 128) tile size for the cubic GEMM, and keeps most data in FP16 while accumulating in FP32 for accuracy. It satisfies the required API (`custom_kernel`) and runs entirely on the GPU (H100) with Triton‑accelerated core computation.
