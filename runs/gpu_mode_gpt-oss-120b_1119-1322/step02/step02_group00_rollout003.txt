Triton‐accelerated **outgoing** TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. LayerNorm (FP32) → [B,N,N,D].
2. Cast to fp16 and apply bias‑free linear projections:
       left  = x @ W_left   (→ [B,N,N,H])

4. Optional mask (broadcast over H) is multiplied into left/right *before*

5. Apply element‑wise gates → gated left/right.
6. Core N³ update:

8. Cast to float32 (as required by downstream code).

    # mask out‑of‑bounds threads
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------ base pointers -------------------------------------------------
    a_base = a_ptr + pid_batch * batch_stride_a
    b_base = b_ptr + pid_batch * batch_stride_b
    c_base = c_ptr + pid_batch * batch_stride_c

    # pointers to the start of the current tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * row_stride_a + offs_k[None, :] * col_stride_a
    # Bᵀ on‑the‑fly: we need B[n, k] -> tile of shape (K, BLOCK_N)
    b_tile_ptr = b_base + offs_k[:, None] * col_stride_b + offs_n[None, :] * row_stride_b

    # ------------ accumulation --------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K

        # load A tile (block_m x block_k)
        a = tl.load(
            a_tile_ptr + cur_k * col_stride_a,
            mask=mask_m[:, None] & ((offs_k + cur_k) < K),
            other=0.0,
        )
        # load Bᵀ tile (block_k x block_n) – note the transposition in the pointer arithmetic
        b = tl.load(
            b_tile_ptr + cur_k * col_stride_b,
            mask=((offs_k + cur_k) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # half‑precision dot, accumulated into fp32
        acc += tl.dot(a, b)

    # ------------ write back ----------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * row_stride_c + offs_n[None, :] * col_stride_c
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D]   (float32)
            - mask:        torch.Tensor [B, N, N] or None (bool/int/float)
            - weights:     dict of model parameters (float32 tensors)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all weight tensors to fp16 (all linear layers are bias‑free)
    # ------------------------------------------------------------------
    w_left     = weights["left_proj.weight"].to(torch.float16)
    w_right    = weights["right_proj.weight"].to(torch.float16)
    w_lgate    = weights["left_gate.weight"].to(torch.float16)
    w_rgate    = weights["right_gate.weight"].to(torch.float16)
    w_outgate  = weights["out_gate.weight"].to(torch.float16)
    w_outnorm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_outnorm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out   = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    # Shape: [B, N, N, hidden_dim]
    left  = F.linear(x_h, w_left)
    right = F.linear(x_h, w_right)

    # ------------------------------------------------------------------
    # 5) Gate values – sigmoid after linear projection
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, N, N] → broadcast to hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # batch, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # 8) Rearrange for batched GEMM: (B*H, N, N)
    #    Layout after .contiguous() is row‑major -> (batch, row, col)
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat = left_t.view(-1, N, N)   # [B*H, N, N]  fp16
    right_flat = right_t.view(-1, N, N)  # same shape

    total_matrices = left_flat.shape[0]   # B * H

    # ------------------------------------------------------------------
    # 9) Allocate output buffer (fp16)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)   # [B*H, N, N]

    # ------------------------------------------------------------------
    # 10) Triton kernel launch – batched A @ Bᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (rows of C)
        triton.cdiv(N, BLOCK_N),   # blocks along N (cols of C)
        total_matrices,            # one program per (batch * hidden) matrix
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.stride(0),   # batch stride for A
        left_flat.stride(1),   # row stride for A (M dimension)
        left_flat.stride(2),   # col stride for A (K dimension)
        right_flat.stride(0),  # batch stride for B
        right_flat.stride(1),  # row stride for B (N dimension)
        right_flat.stride(2),  # col stride for B (K dimension)
        out_flat.stride(0),    # batch stride for C
        out_flat.stride(1),    # row stride for C (M dimension)
        out_flat.stride(2),    # col stride for C (N dimension)
        N,                     # M
        N,                     # N
        N,                     # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 11) Reshape back to [B, N, N, H] (fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 12) Hidden‑dim LayerNorm + out_gate multiplication
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_outnorm_w,
        bias=w_outnorm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 13) Final projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # still fp16, shape [B, N, N, D]

    # ------------------------------------------------------------------
    # 14) Cast back to float32 (matching the reference implementation)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
