"""
custom_kernel
--------------

Implements the *outgoing* TriMul update (TriMul) used in AlphaFold‑3 and related
protein‑structure models.  The forward pass consists of

1.  Input LayerNorm (FP32).
2.  Two linear projections → ``left`` and ``right``  (dim → hidden_dim).
3.  Three gating linear layers → ``left_gate``, ``right_gate``, ``out_gate``
    (sigmoid activations, ``out_gate`` kept in FP32, the others cast to FP16).
4.  **Triton kernel** fuses the optional binary mask (shape ``[B,N,N]``) and
    the per‑channel gates:
       ``left_out  = left  * mask * left_gate``
       ``right_out = right * mask * right_gate``
    The kernel works on the flattened tensor (``B,N,N,hidden``) and therefore
    avoids materialising a full broadcasted mask.
5.  Batched matrix multiplication (FP16 inputs, FP32 accumulator) computes
       ``out_{b,i,j,:} = Σ_k left_out[b,i,k,:] * right_out[b,j,k,:]``.
    This is realised as a large batched GEMM via ``torch.bmm`` which uses
    Tensor‑cores on H100.
6.  Second LayerNorm over the hidden dimension (FP32).
7.  Apply the output gate and a final linear projection back to ``dim``.

Only the mask‑&‑gate fusion is performed in Triton – the rest of the
computation relies on PyTorch’s highly‑optimised kernels.  The function
accepts a tuple ``(input_tensor, mask, weights, config)`` and returns a
float‑32 tensor of shape ``[B, N, N, dim]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr,                # fp16 * left  (contiguous)
    right_ptr,               # fp16 * right (contiguous)
    mask_ptr,                # fp16 * mask  (B,N,N) contiguous, broadcasted
    left_gate_ptr,           # fp16 * left_gate  (contiguous)
    right_gate_ptr,          # fp16 * right_gate (contiguous)
    out_left_ptr,            # fp16 * output left
    out_right_ptr,           # fp16 * output right
    total_elements: tl.int64,
    H: tl.constexpr,         # hidden dimension (known at compile time)
    BLOCK: tl.constexpr,     # threads per program
):
    """Fuse mask (optional) and gating for ``left`` / ``right`` tensors.

    The tensors are flattened in row‑major order:
        offset = ((b*N + i)*N + j)*H + h
    ``mask`` has shape (B,N,N) → ``offset_mask = ((b*N + i)*N + j)``.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear offsets
    mask = offs < total_elements

    # ------------------------------------------------------------------
    # Load the four operand tensors (masked load → zero outside range)
    # ------------------------------------------------------------------
    l  = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Load mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    idx_mask = offs // H                                 # drop the hidden index
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Fuse three element‑wise multiplications
    # ------------------------------------------------------------------
    l = l * lg * m
    r = r * rg * m

    # ------------------------------------------------------------------
    # Store the results
    # ------------------------------------------------------------------
    tl.store(out_left_ptr  + offs, l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        ``(input_tensor, mask, weights, config)`` where
        * ``input_tensor`` : torch.Tensor, shape ``[B, N, N, dim]``, dtype float32
        * ``mask``         : torch.Tensor, shape ``[B, N, N]`` (float32 or bool)
        * ``weights``      : dict of torch.Tensor model parameters
        * ``config``       : dict with keys ``dim``, ``hidden_dim`` and optional
                             ``nomask`` (bool).  ``nomask=True`` signals that the
                             mask should be ignored.

    Returns
    -------
    torch.Tensor
        Tensor of shape ``[B, N, N, dim]``, dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (still FP32)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # keep FP32

    # Cast gates to FP16 for the fused kernel (mask and gate are applied after
    # the projection, which is also cast to FP16)
    left_gate  = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Cast projections to FP16 (the heavy N³ computation is done in FP16)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Fuse optional mask and per‑channel gates via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # Mask -> FP16 (0 / 1) and ensure contiguity
        mask_fp16 = mask.to(dtype=torch.float16, device=left_fp16.device).contiguous()
        # All tensors must be contiguous for the kernel
        left_fp16   = left_fp16.contiguous()
        right_fp16  = right_fp16.contiguous()
        left_gate   = left_gate.contiguous()
        right_gate  = right_gate.contiguous()

        # Allocate outputs
        left_out  = torch.empty_like(left_fp16)
        right_out = torch.empty_like(right_fp16)

        total_elems = left_fp16.numel()
        BLOCK = 1024                      # threads per program (tune if needed)
        grid  = ( (total_elems + BLOCK - 1) // BLOCK, )  # 1‑D grid

        _mask_gate_kernel[grid](
            left_fp16,
            right_fp16,
            mask_fp16,
            left_gate,
            right_gate,
            left_out,
            right_out,
            total_elems,
            H=hidden_dim,
            BLOCK=BLOCK,
        )
        left_fp16  = left_out
        right_fp16 = right_out
    else:
        # No mask → only apply the per‑channel gates
        left_fp16  = left_fp16 * left_gate
        right_fp16 = right_fp16 * right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out = left @ rightᵀ  (fp16 inputs, fp32 accumulation)
    # ------------------------------------------------------------------
    B, N, _, _ = left_fp16.shape               # (B, N, N, hidden)
    # reshape to (B*hidden, N, N) for torch.bmm
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm automatically uses Tensor‑Cores for fp16 inputs
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 7. Restore original layout and promote to FP32
    # ------------------------------------------------------------------
    out = out_mat_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over the hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
