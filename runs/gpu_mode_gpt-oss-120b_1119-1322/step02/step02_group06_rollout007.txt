@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offset start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute masks
        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_head * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)  # fp16

        # Load B tile (transposed): original B shape (batch, N, K) = (batch, N, N)

        # dot product
        acc += tl.dot(a, b)  # Returns (BLOCK_M, BLOCK_N) in fp32

    # store result

```python
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: batched matrix multiplication (left_fused @ right_fused^T)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    left_ptr,  # fp16*  shape (batch_head, M, K)
    right_ptr, # fp16*  shape (batch_head, N, K) (will be read transposed)
    out_ptr,   # fp16*  shape (batch_head, M, N)
    batch_head: tl.constexpr,
    N: tl.constexpr,
    stride_l_batch: tl.constexpr, stride_l_m: tl.constexpr, stride_l_k: tl.constexpr,
    stride_r_batch: tl.constexpr, stride_r_k: tl.constexpr, stride_r_n: tl.constexpr,
    stride_o_batch: tl.constexpr, stride_o_m: tl.constexpr, stride_o_n: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    m_mask = offs_m < N
    n_mask = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < N

        # A tile (batch, m, k)
        a_ptrs = left_ptr \
            + batch * stride_l_batch \
            + offs_m[:, None] * stride_l_m \
            + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=m_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # B tile (batch, k, n)  — we read right matrix transposed
        b_ptrs = right_ptr \
            + batch * stride_r_batch \
            + offs_k[:, None] * stride_r_k \
            + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & n_mask[None, :],
                    other=0.0)

        # matrix multiply on the tile, accumulate in fp32
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write result as fp16
    c_ptrs = out_ptr \
        + batch * stride_o_batch \
        + offs_m[:, None] * stride_o_m \
        + offs_n[None, :] * stride_o_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=m_mask[:, None] & n_mask[None, :])

# ----------------------------------------------------------------------
# Kernel 2: elementwise multiplication (fp32 * fp32) used for out_norm * out_gate
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    size,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass with fused linear projections and a custom
    batched GEMM kernel. Steps:
      1. Layer‑norm the input.
      2. One large linear projection computes left/right projections, gates
         and the output‑gate, reducing reads of the input tensor.
      3. Apply the optional pair‑wise mask and sigmoid gates.
      4. Fuse projection + gate to obtain `left_fused` and `right_fused`.
      5. Compute the core triangle multiplication
         out[b,i,j,h] = Σₖ left_fused[b,i,k,h] * right_fused[b,j,k,h]
         using a Triton batched‑gemm kernel written in FP16 with FP32
         accumulation.
      6. Layer‑norm over the hidden dimension, multiply by the output‑gate,
         and finally linearly project back to the original model dimension.
    The implementation keeps the final tensor in FP32 to match the reference.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Concatenated linear projection (FP16)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                # [B, N, N, dim]
    M = bs * N * N
    x_flat = x_fp16.view(M, dim)                # (M, dim)

    # concatenate weight matrices for left_proj, left_gate, right_proj,
    # right_gate and out_gate (bias=False everywhere)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # shape (5*hidden_dim, dim)
    w_concat = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(torch.float16)

    # one big matmul
    raw = torch.nn.functional.linear(x_flat, w_concat)  # (M, 5*hidden_dim)

    H = hidden_dim
    left_proj_raw = raw[:, :H]            # (M, H)
    left_gate_raw  = raw[:, H:2 * H]      # (M, H)
    right_proj_raw = raw[:, 2 * H:3 * H]  # (M, H)
    right_gate_raw = raw[:, 3 * H:4 * H]  # (M, H)
    out_gate_raw   = raw[:, 4 * H:]       # (M, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)
        mask_flat = mask_fp16.view(M, 1)                     # (M,1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw).to(torch.float16)

    left_fused  = left_proj_raw * left_gate   # (M, H)
    right_fused = right_proj_raw * right_gate # (M, H)

    # ------------------------------------------------------------------
    # 5️⃣ Core triangle multiplication with a custom Triton kernel
    #    left_fused and right_fused are reshaped to (B, H, N, N)
    # ------------------------------------------------------------------
    # reshape & permute to (B, H, N, N)
    left_fused = left_fused.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()
    right_fused = right_fused.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()

    batch_head = bs * hidden_dim  # B * H
    # flatten the leading batch*head dimension for the kernel
    left_flat = left_fused.reshape(batch_head, N, N)
    right_flat = right_fused.reshape(batch_head, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16, shape (batch_head, N, N)

    # strides (expressed in elements, not bytes)
    stride_l_batch = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    # right is accessed transposed inside the kernel
    stride_r_batch = right_flat.stride(0)
    stride_r_k = right_flat.stride(2)   # column stride (originally last dim)
    stride_r_n = right_flat.stride(1)   # row stride (originally second dim)

    stride_o_batch = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # launch config – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_head,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_head,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output side LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # (B, N, N, H) FP32

    # ------------------------------------------------------------------
    # 7️⃣ Multiply by output gate (Triton kernel)
    # ------------------------------------------------------------------
    # flatten for kernel
    out_norm_flat = out_norm.view(-1)
    out_gate_flat = out_gate.view(-1).to(torch.float32)
    gated_flat = torch.empty_like(out_norm_flat)

    total = out_norm_flat.numel()
    BLOCK_SIZE = 65536
    grid_mul = (triton.cdiv(total, BLOCK_SIZE),)
    elementwise_mul_kernel[grid_mul](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(bs, N, N, hidden_dim)  # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim` (FP32)
    # ------------------------------------------------------------------
    out_final = torch.nn.functional.linear(gated, weights["to_out.weight"])  # (B,N,N,dim)

    return out_final
```

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1 – batched GEMM for the triangle multiplication
# Computes out[b,h,i,j] = Σₖ left_fused[b,h,i,k] * right_fused[b,h,j,k]
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    left_ptr,         # fp16*  (B*H, M, K)  where M=N and K=N
    right_ptr,        # fp16*  (B*H, N, K)  (will be read transposed)
    out_ptr,          # fp16*  (B*H, M, N)
    batch_head: tl.constexpr,
    N: tl.constexpr,
    stride_l_batch: tl.constexpr, stride_l_m: tl.constexpr, stride_l_k: tl.constexpr,
    stride_r_batch: tl.constexpr, stride_r_k: tl.constexpr, stride_r_n: tl.constexpr,
    stride_o_batch: tl.constexpr, stride_o_m: tl.constexpr, stride_o_n: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)                     # which (B*H) pair we are processing
    pid_m = tl.program_id(1)                     # tile row   (i)
    pid_n = tl.program_id(2)                     # tile col   (j)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32 (full‑precision reduction)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summed‑over sequence index)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile : shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr \
            + batch * stride_l_batch \
            + offs_m[:, None] * stride_l_m \
            + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile (transposed) : shape (BLOCK_K, BLOCK_N)
        # we read right as (k, n) because we need right[j,k] = right_T[k,j]
        b_ptrs = right_ptr \
            + batch * stride_r_batch \
            + offs_k[:, None] * stride_r_k \
            + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # matrix‑multiply the tile (fp16 * fp16 → fp32)
        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result as fp16
    # ------------------------------------------------------------------
    c_ptrs = out_ptr \
        + batch * stride_o_batch \
        + offs_m[:, None] * stride_o_m \
        + offs_n[None, :] * stride_o_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Kernel 2 – element‑wise multiplication (used for out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,        # fp32*
    b_ptr,        # fp32*
    out_ptr,      # fp32*
    size,         # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing Triangle‑Multiplicative Update (TriMul) forward pass.

    1️⃣  Layer‑norm the input.
    2️⃣  Fuse the five linear projections (left‑proj, left‑gate,
        right‑proj, right‑gate, out‑gate) into a single matmul to
        minimise reads of the large 4‑D tensor.
    3️⃣  Apply the optional pair‑wise mask and sigmoid gates.
    4️⃣  Fuse projection + gate → `left_fused` / `right_fused`.
    5️⃣  Core triplet multiplication:
        out[b,i,j,h] = Σₖ left_fused[b,i,k,h] * right_fused[b,j,k,h]
        performed by a custom batched GEMM Triton kernel (fp16 → fp32
        accumulation).
    6️⃣  Layer‑norm over the hidden dimension (fp32).
    7️⃣  Element‑wise multiply by the output gate (Triton).
    8️⃣  Final linear projection back to the original model dimension.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                          # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Concatenated linear projection (convert to fp16)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)               # [B, N, N, dim]  fp16
    M = bs * N * N
    x_flat = x_fp16.view(M, dim)               # (M, dim)

    # concatenate all five weight matrices (bias=False everywhere)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                        # (5*H, dim)

    # single big matmul
    raw = torch.nn.functional.linear(x_flat, w_concat)   # (M, 5*H)

    H = hidden_dim
    left_proj_raw = raw[:, :H]                 # (M, H)
    left_gate_raw = raw[:, H:2 * H]
    right_proj_raw = raw[:, 2 * H:3 * H]
    right_gate_raw = raw[:, 3 * H:4 * H]
    out_gate_raw = raw[:, 4 * H:]              # (M, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)
        mask_flat = mask_fp16.view(M, 1)                     # (M,1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).to(torch.float16)
    out_gate = torch.sigmoid(out_gate_raw).to(torch.float16)

    left_fused = left_proj_raw * left_gate      # (M, H)
    right_fused = right_proj_raw * right_gate   # (M, H)

    # ------------------------------------------------------------------
    # 5️⃣ Core triangle multiplication (custom Triton kernel)
    #    reshape to (B, H, N, N) and flatten the leading (B*H) axis
    # ------------------------------------------------------------------
    left_fused = left_fused.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()
    right_fused = right_fused.view(bs, N, N, H).permute(0, 3, 1, 2).contiguous()

    batch_head = bs * hidden_dim                # B * H
    left_flat = left_fused.reshape(batch_head, N, N)
    right_flat = right_fused.reshape(batch_head, N, N)

    out_flat = torch.empty_like(left_flat)      # fp16, (B*H, N, N)

    # strides (in elements, not bytes)
    stride_l_batch = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_k = right_flat.stride(2)   # column stride (original last dim)
    stride_r_n = right_flat.stride(1)   # row stride (original second dim)

    stride_o_batch = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # launch parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_head,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_head,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H) – fp16
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output‑side LayerNorm (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                          # (B, N, N, H)  fp32

    # ------------------------------------------------------------------
    # 7️⃣ Multiply by output gate (Triton elementwise kernel)
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.view(-1)
    out_gate_flat = out_gate.view(-1).to(torch.float32)
    gated_flat = torch.empty_like(out_norm_flat)

    total = out_norm_flat.numel()
    BLOCK_SIZE = 65536
    grid_mul = (triton.cdiv(total, BLOCK_SIZE),)
    elementwise_mul_kernel[grid_mul](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(bs, N, N, hidden_dim)   # (B, N, N, H) fp32

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to the model dimension
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(gated, weights["to_out.weight"])  # (B, N, N, dim)

    return output
