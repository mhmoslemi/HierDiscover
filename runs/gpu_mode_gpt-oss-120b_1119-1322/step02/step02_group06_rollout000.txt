"""
TriMul (outgoing) – fast forward pass.

The kernel fuses the three linear projections (left‑proj/gate, right‑proj/gate,
out‑gate) into a single matmul, uses FP16 for the heavy tensor algebra,
applies the optional pairwise mask, then performs the triangle‐multiplication
as a batched GEMM over the hidden dimension.  A tiny Triton kernel is kept for
the final element‑wise gate‑multiplication (required by the problem statement),
while the rest is performed with highly‑optimized cuBLAS/Torch kernels.
The implementation works for any of the test configurations listed in the
statement and returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: simple element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # float32* – out_norm (flattened)
    b_ptr,          # float32* – out_gate (flattened)
    out_ptr,        # float32* – result (flattened)
    size,           # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass – fused, mixed‑precision implementation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:         torch.Tensor [B, N, N] (bool/float)
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing "dim", "hidden_dim", and "nomask" flag

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ Input LayerNorm (over channel `dim`)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape: [B, N, N, dim] (FP32)

    # -------------------------------------------------
    # 2️⃣ Cast to FP16 for the massive linear work
    # -------------------------------------------------
    x_h = x.to(torch.float16).view(-1, dim)   # (B·N·N, dim)

    # -------------------------------------------------
    # 3️⃣ Fuse all linear projections into one matmul
    #    Order: left_proj, left_gate, right_proj, right_gate, out_gate
    # -------------------------------------------------
    # Collect the weights (all are stored as (out_features, in_features))
    left_proj_w  = weights["left_proj.weight"]      # (H, dim)
    left_gate_w  = weights["left_gate.weight"]      # (H, dim)
    right_proj_w = weights["right_proj.weight"]     # (H, dim)
    right_gate_w = weights["right_gate.weight"]     # (H, dim)
    out_gate_w   = weights["out_gate.weight"]       # (H, dim)

    # Concatenate on the output‑feature axis → (5*H, dim)
    concat_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(torch.float16)

    # One fused linear
    y = torch.nn.functional.linear(x_h, concat_w)   # (B·N·N, 5*H)

    # -------------------------------------------------
    # 4️⃣ Split results and reshape back to 4‑D tensors
    # -------------------------------------------------
    # each slice has shape (B·N·N, H)
    splits = torch.split(y, hidden_dim, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = splits

    # Restore [B, N, N, H] layout – keep in FP16
    left_proj = left_proj_raw.view(bs, N, N, hidden_dim)
    left_gate = left_gate_raw.view(bs, N, N, hidden_dim)
    right_proj = right_proj_raw.view(bs, N, N, hidden_dim)
    right_gate = right_gate_raw.view(bs, N, N, hidden_dim)
    out_gate   = out_gate_raw.view(bs, N, N, hidden_dim)

    # -------------------------------------------------
    # 5️⃣ Optional pair‑wise mask (applied only to the proj terms)
    # -------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to fp16 and broadcast
        mask_h = mask.to(torch.float16).unsqueeze(-1)           # (B, N, N, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -------------------------------------------------
    # 6️⃣ Sigmoid gates and element‑wise gating
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)       # (B, N, N, H) FP16
    right_gate = torch.sigmoid(right_gate)     # (B, N, N, H) FP16
    out_gate   = torch.sigmoid(out_gate)       # (B, N, N, H) FP16

    left  = left_proj * left_gate              # FP16
    right = right_proj * right_gate            # FP16

    # -------------------------------------------------
    # 7️⃣ Core triangle multiplication (batched GEMM over hidden dim)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # -------------------------------------------------
    # Bring hidden dim to batch axis for an efficient BMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)  # (B*H, N, N)

    # Batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original ordering → (B, N, N, H) in FP16
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -------------------------------------------------
    # 8️⃣ Output‑side LayerNorm (over hidden_dim) – performed in FP32
    # -------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B, N, N, H) FP32

    # -------------------------------------------------
    # 9️⃣ Element‑wise multiplication with out_gate (Triton kernel)
    # -------------------------------------------------
    out_norm_flat = out_norm.contiguous().view(-1)                     # 1‑D FP32
    out_gate_flat = out_gate.to(torch.float32).contiguous().view(-1)   # 1‑D FP32
    gated_flat = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK_SIZE = 65536                     # reasonable launch size for H100
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(bs, N, N, hidden_dim)   # (B, N, N, H) FP32

    # -------------------------------------------------
    # 🔟 Final linear projection back to the original `dim`
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"]           # (dim, hidden_dim) – FP32
    output = torch.nn.functional.linear(gated, to_out_w)   # (B, N, N, dim) FP32

    return output
