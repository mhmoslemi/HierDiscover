    device = x.device

    norm_b = weights["norm.bias"]

    # Cast to half
    x_h = x.to(torch.float16)

    # 2. Combined weight
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Concatenate
    combined_w = torch.cat([left_proj_w, left_gate_w,
                           right_proj_w, right_gate_w,
                           out_gate_w], dim=0).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Linear
    combined = torch.nn.functional.linear(x_h, combined_w)  # shape [B, N, N, 5*hidden_dim]

    # Split
    h = hidden_dim
    left_proj = combined[..., :h]
    left_gate_pre = combined[..., h:2*h]
    right_proj = combined[..., 2*h:3*h]
    right_gate_pre = combined[..., 3*h:4*h]
    out_gate_pre = combined[..., 4*h:5*h]

    # Sigmoids
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # reshape: (B, H, N, N)

    right_perm = right.permute(0, 3, 1, 2).contiguous()

"""
Optimized outgoing Triangle‑Multiplicative‑Update (TriMul) kernel.

Algorithm
---------
1. Layer‑norm the input tensor (dtype half) – weight & bias are cast to half.
2. Fuse the four linear projections (left/right + their gates) and the out‑gate
   into a single GEMM:  x_h @ W_comb (W_comb has shape (5·hidden_dim, dim)).
   Slice the result into left_proj, left_gate, right_proj, right_gate and out_gate.
3. Apply the optional mask, then element‑wise sigmoid‑gating.
4. Form the gated tensors  left = left_proj * left_gate,
   right = right_proj * right_gate (still half‑precision).
5. Triangle multiplication:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   Implemented as a batched half‑precision GEMM:
     (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N).
6. Layer‑norm over the hidden dimension (half‑precision).
7. Fuse the final element‑wise product out_norm * out_gate with a tiny
   Triton kernel that works entirely in float16.
8. Linear projection back to the original channel dimension (half‑precision).
9. Return the result in float32 as required by the API.

All heavy compute stays in float16 to maximise Tensor‑core utilisation.
Only two kernel launches are needed: the elementwise multiply (step 7) and
the cheap‑memory kernel for the mask/gating could be added if desired.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for element‑wise half‑precision multiplication
# ----------------------------------------------------------------------
@triton.jit
def _elementwise_mul_half_kernel(
    a_ptr,                # half*   (first operand, flattened)
    b_ptr,                # half*   (second operand, flattened)
    out_ptr,              # half*   (output, flattened)
    size,                 # i64     (total number of elements)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] (bool / 0‑1 float) or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep half‑precision)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim], dtype matches input (float32→float32)

    # Cast to float16 for all subsequent heavy work
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections (left/right + gates + out_gate)
    # ------------------------------------------------------------------
    # Gather and cast all weights once
    left_proj_w   = weights["left_proj.weight"]   # (hidden_dim, dim)
    left_gate_w   = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"]  # (hidden_dim, dim)
    right_gate_w  = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_w    = weights["out_gate.weight"]    # (hidden_dim, dim)

    combined_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)                      # shape (5·hidden_dim, dim)

    # One big GEMM: (B·N·N, dim) @ (dim, 5·hidden_dim) → (B·N·N, 5·hidden_dim)
    combined = torch.nn.functional.linear(x_h, combined_w)  # [B, N, N, 5·H]

    # ------------------------------------------------------------------
    # 3. Slice the fused output and apply sigmoid gates
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj      = combined[..., :H]                 # [B,N,N,H]
    left_gate_pre  = combined[..., H:2 * H]            # [B,N,N,H]
    right_proj     = combined[..., 2 * H:3 * H]        # [B,N,N,H]
    right_gate_pre = combined[..., 3 * H:4 * H]        # [B,N,N,H]
    out_gate_pre   = combined[..., 4 * H:5 * H]        # [B,N,N,H]

    left_gate  = torch.sigmoid(left_gate_pre)   # half
    right_gate = torch.sigmoid(right_gate_pre)  # half
    out_gate   = torch.sigmoid(out_gate_pre)    # half

    # ------------------------------------------------------------------
    # 4. Optional mask and gating
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask -> half and broadcast over the hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj  * left_gate    # [B,N,N,H]
    right = right_proj * right_gate   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5. Triangle multiplication (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, H, N, N) –> (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left_perm.view(B * H, N, N)          # (B·H, N, N)
    right_flat = right_perm.view(B * H, N, N)         # (B·H, N, N)

    # Core operation: left @ rightᵀ   (tensor‑core friendly, FP16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output‑side LayerNorm (half‑precision)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B,N,N,H], half

    # ------------------------------------------------------------------
    # 7. Element‑wise multiplication with out_gate (Triton kernel)
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.view(-1)            # half
    out_gate_flat = out_gate.view(-1)            # half
    gated_flat = torch.empty_like(out_norm_flat) # half

    total_elts = out_norm_flat.numel()
    BLOCK_SIZE = 65536  # 64 K elements per program
    grid = (triton.cdiv(total_elts, BLOCK_SIZE),)

    _elementwise_mul_half_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elts,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(B, N, N, H)  # [B,N,N,H], half

    # ------------------------------------------------------------------
    # 8. Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    out_h = torch.nn.functional.linear(gated, to_out_w)    # [B,N,N,dim], half

    # Return in float32 as specified by the external API
    return out_h.to(torch.float32)
