"""
custom_kernel – Triton‑accelerated forward of the “outgoing” TriMul module.

Algorithm
---------
1. Layer‑norm the input tensor (float32) across the feature axis `dim`.
2. Cast to FP16 and apply three bias‑free linear projections:
      * left_proj  → (B, N, N, H)
      * right_proj → (B, N, N, H)
      * gate logits (left_gate, right_gate, out_gate) → sigmoid
3. Optional element‑wise mask (broadcast over the hidden dim).
4. Apply the left / right gates.
5. Core cubic update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply `left @ rightᵀ` for each hidden slice.
   This is performed by a fused Triton kernel (`batched_gemm_out_kernel`)
   using FP16 values with FP32 accumulation (Tensor‑core friendly).
6. Reshape the result to (B, N, N, H) (still FP16).
7. Layer‑norm over the hidden dimension (float32), scale by `out_gate`,
   and a final bias‑free linear projection back to the original `dim`.
8. Return the final tensor in float32.

The heavy N³ work is therefore confined to a single Triton GEMM kernel,
while the remaining cheap ops stay in highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   (A,B : [batch, N, N]  in FP16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_ab, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B: batch, row(k), col(j) strides
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    """Batched FP16 GEMM C = A @ Bᵀ with FP32 accumulation."""
    pid_batch = tl.program_id(2)                # which (batch * hidden) matrix
    pid_m = tl.program_id(0)                    # block row of C (i)
    pid_n = tl.program_id(1)                    # block col of C (j)

    # ------------------------------------------------------------------
    # Compute tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this matrix slice
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # ------------------------------------------------------------------
    # Pointers to the start of the current tile (k = 0)
    # ------------------------------------------------------------------
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32 (Tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (tiling)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot of two (BLOCK_M×BLOCK_K) and (BLOCK_K×BLOCK_N) tiles → FP32
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back results (cast to FP16)
    # ------------------------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operation.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, L, L, D] (float32)
            - mask         : torch.Tensor [B, L, L] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, L, L, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, L, L, D]  float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Helper to cast weight tensors once
    def to_half(t):
        return t.to(torch.float16)

    # Linear projections (bias‑free)
    left = F.linear(x_fp16, to_half(weights["left_proj.weight"]))   # [B, L, L, H]
    right = F.linear(x_fp16, to_half(weights["right_proj.weight"]))

    # Gating logits → sigmoid (still FP16)
    left_gate = F.linear(x_fp16, to_half(weights["left_gate.weight"])).sigmoid()
    right_gate = F.linear(x_fp16, to_half(weights["right_gate.weight"])).sigmoid()
    out_gate = F.linear(x_fp16, to_half(weights["out_gate.weight"])).sigmoid()

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, L, L] → [B, L, L, 1] and cast to FP16
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply left/right gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, L, L, hidden_dim (H == hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Core cubic update via Triton batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    # --------------------------------------------------------------
    #   Rearrange to (B, H, N, N) and flatten the first two dims
    # --------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(-1, N, N)    # [B*H, N, N]   FP16
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]   FP16

    total = left_flat.shape[0]  # B * H independent GEMMs

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # Kernel launch configuration (tuned for large N on H100)
    # --------------------------------------------------------------
    # Basic block sizes – good for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i‑axis)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j‑axis)
        total,                     # one program per (batch * hidden)
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K
        left_flat.stride(0),       # stride_ab (batch stride)
        left_flat.stride(1),       # stride_am (row stride)
        left_flat.stride(2),       # stride_ak (col stride)
        right_flat.stride(0),      # stride_bb (batch stride)
        right_flat.stride(2),      # stride_bk (row‑stride for Bᵀ)
        right_flat.stride(1),      # stride_bn (col‑stride for Bᵀ)
        out_flat.stride(0),        # stride_cb (batch stride)
        out_flat.stride(1),        # stride_cm (row stride)
        out_flat.stride(2),        # stride_cn (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps → 256 threads per block
        num_stages=2,              # double‑buffering of tiles
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, H] (still FP16)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (float32), out‑gate scaling, final projection
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out‑gate (cast to float32)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final projection back to original dimension D (float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
