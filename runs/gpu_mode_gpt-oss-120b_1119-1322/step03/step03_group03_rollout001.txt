"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated core update.
The kernel computes, for each hidden channel h,
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
which is a batched matrix multiplication A @ Bᵀ with A/B of shape (N,N).
All heavy arithmetic (the N³ GEMM) is done in fp16 with fp32 accumulation.
LayerNorms, gating and the final projection stay in float32/half for numerical
stability.  The routine works with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch‑stride, row‑stride, col‑stride
    stride_bb, stride_bk, stride_bn,   # B (to be transposed): batch‑stride, k‑stride, n‑stride
    stride_cb, stride_cm, stride_cn,   # C: batch‑stride, row‑stride, col‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for every (batch * hidden) matrix.
    A, B, C are fp16 tensors; accumulation is fp32.
    """
    pid_batch = tl.program_id(2)          # index over (batch * hidden) matrices
    pid_m = tl.program_id(0)               # block‑row of C
    pid_n = tl.program_id(1)               # block‑col of C

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # pointers to the first tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        # Load A and B tiles with boundary guards
        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

    # write output tile
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    """

    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data

    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------

        BLOCK_K = 64

        BLOCK_K = 64

    # ------------------------------------------------------------------

        triton.cdiv(N, BLOCK_M),   # rows of C

    )

        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0),   # stride_ab (batch stride)
        left_flat.stride(1),   # stride_am (row stride)
        left_flat.stride(2),   # stride_ak (col stride)
        right_flat.stride(0),  # stride_bb (batch stride)
        right_flat.stride(2),  # stride_bk (col‑k stride for Bᵀ)
        right_flat.stride(1),  # stride_bn (row‑n stride for Bᵀ)
        out_flat.stride(0),    # stride_cb (batch stride)
        out_flat.stride(1),    # stride_cm (row stride)
        out_flat.stride(2),    # stride_cn (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 12) Reshape back to original 4‑D layout [B, N, N, H] (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 13) Hidden‑dim LayerNorm (float32 for stability) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 14) Final projection back to the original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)          # [B, N, N, D]  fp16

    # ------------------------------------------------------------------
    # 15) Convert to float32 (expected downstream type) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
