"""
custom_kernel – Triton‑accelerated “outgoing” TriMul forward pass
================================================================
The module follows the reference PyTorch implementation:

    1. Layer‑norm over the input channels (dim).
    2. Two bias‑free linear projections  → left, right  (dim → hidden_dim).
    3. Optional mask multiplication (broadcasted over the hidden dim).
    4. Gating – three sigmoids (left_gate, right_gate, out_gate) – all in FP16.
    5. Element‑wise gating of left/right.
    6. Core cubic update:
           out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
       which is exactly a batched matrix‑multiply
           C = A @ Bᵀ
       for every hidden slice h.  This N×N × N multiplication is
       performed by a single Triton kernel (FP16 × FP16 → FP32 accumulation).
    7. Hidden‑dim LayerNorm, out‑gate scaling and final linear projection back
       to the original channel dimension (dim).  The result is returned as FP32.

All heavy arithmetic (steps 2‑6) is carried out in FP16, whilst the two
LayerNorms and the final projection use FP32 for numerical stability.
The kernel is tuned for H100 Tensor‑cores (BLOCK_M=N_BLOCK=128, BLOCK_K=64,
8 warps, 2 stages).  The code works for any batch size ≤ 2, any
sequence length from the test‑suite, and both masked / un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : [batch, N, N]   fp16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_out_kernel(
    a_ptr,            # ptr to A (fp16)
    b_ptr,            # ptr to B (fp16)
    c_ptr,            # ptr to C (fp16)
    M, N, K,          # matrix dimensions (all equal to seq_len)
    stride_ab,        # batch stride for A
    stride_am,        # row stride for A  (i‑dimension)
    stride_ak,        # col stride for A  (k‑dimension)
    stride_bb,        # batch stride for B
    stride_bk,        # row stride for Bᵀ (k‑dimension)
    stride_bn,        # col stride for Bᵀ (n‑dimension)
    stride_cb,        # batch stride for C
    stride_cm,        # row stride for C (i‑dimension)
    stride_cn,        # col stride for C (j‑dimension)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    One program (pid) handles a whole (batch*hidden) matrix.
    """
    pid_batch = tl.program_id(2)          # which independent matrix
    pid_m = tl.program_id(0)              # block row of C
    pid_n = tl.program_id(1)              # block col of C

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # pointers to the start of the current tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K
        cur_k = offs_k + cur_off

        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & (cur_k < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=(cur_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # FP16 dot → FP32 acc

    # write back FP16 result (cast)
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask        : torch.Tensor [B, N, N] or None
            - weights     : dict of model parameters (float32)
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Helper to cast a weight to half‑precision on the correct device
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # Linear projections (bias‑free) → [B, N, N, H]
    left = F.linear(x_fp16, to_half(weights["left_proj.weight"]))
    right = F.linear(x_fp16, to_half(weights["right_proj.weight"]))

    # Gating logits → sigmoid (still FP16) → [B, N, N, H]
    left_gate = F.linear(x_fp16, to_half(weights["left_gate.weight"])).sigmoid()
    right_gate = F.linear(x_fp16, to_half(weights["right_gate.weight"])).sigmoid()
    out_gate = F.linear(x_fp16, to_half(weights["out_gate.weight"])).sigmoid()

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] → [B, N, N, 1]  (FP16)
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape   # H == hidden_dim

    # ------------------------------------------------------------------
    # 5️⃣ Core cubic update – batched GEMM (A @ Bᵀ) using Triton
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → flatten batch*hidden dimension
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(-1, N, N)    # [B*H, N, N]  fp16
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]  fp16

    total_mats = left_flat.shape[0]      # B * H independent GEMMs

    out_flat = torch.empty_like(left_flat)  # allocate fp16 output

    # ------------------------------------------------------------------
    # Kernel launch configuration (tuned for H100)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along i‑dimension
        triton.cdiv(N, BLOCK_N),   # blocks along j‑dimension
        total_mats,                # one program per (batch * hidden)
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K
        left_flat.stride(0),       # stride_ab  (batch stride)
        left_flat.stride(1),       # stride_am  (row stride)
        left_flat.stride(2),       # stride_ak  (col stride)
        right_flat.stride(0),      # stride_bb  (batch stride)
        right_flat.stride(2),      # stride_bk  (row stride for Bᵀ)
        right_flat.stride(1),      # stride_bn  (col stride for Bᵀ)
        out_flat.stride(0),        # stride_cb  (batch stride)
        out_flat.stride(1),        # stride_cm  (row stride)
        out_flat.stride(2),        # stride_cn  (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 256 threads per block
        num_stages=2,              # double‑buffer
    )

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, H] (still FP16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out‑gate (convert gate to FP32 first)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear projection back to original dim
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
```
