"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated batched GEMM.

The heavy O(N³) contraction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
is a batched matrix multiply C = A @ Bᵀ where A and B have shape (N, N)
for every hidden channel h and every batch element.
All linear layers, layer‑norms and gating are performed with PyTorch;
the N³ contraction is off‑loaded to a custom Triton kernel that works in
fp16 (accumulated in fp32) and writes the result back in fp16.
The final projection and the cast back to float32 complete the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_kernel(
    a_ptr,                     # left  [total, M, K]  fp16
    b_ptr,                     # right [total, N, K]  fp16 (will be read as Bᵀ)
    c_ptr,                     # out   [total, M, N]  fp32 accumulator, stored as fp16
    M, N, K,                   # matrix dimensions (all equal to seq_len)
    stride_ab, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B (used as Bᵀ): batch, k, n strides
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each (batch * hidden) matrix:
        C = A @ Bᵀ
    with A, B in fp16 and fp32 accumulation.
    """
    pid_batch = tl.program_id(2)   # which (batch * hidden) matrix
    pid_m = tl.program_id(0)       # block row of C
    pid_n = tl.program_id(1)       # block col of C

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # tile pointers for the first K‑tile
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # current offset in the K dimension
        cur_off = k_tile * BLOCK_K

        # load tiles (masked)
        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product: fp16 inputs, fp32 accumulation
        acc += tl.dot(a, b)

    # write the tile back
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    # 4) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left)   # [B, N, N, H]
    right = F.linear(x_h, w_right)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional masking (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6) Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    left  = left  * left_gate   # element‑wise
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Prepare tensors for the batched GEMM: reshape to [total, N, N]
    #    where total = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, seq_len, hidden_dim
    # move hidden dim in front of the two spatial axes for contiguous layout
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_t.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]

    total = left_flat.shape[0]          # = B * hidden_dim

    # ------------------------------------------------------------------
    # 8) Allocate output buffer (fp16)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)   # fp16, same shape as left_flat

    # ------------------------------------------------------------------
    # 9) Choose tile sizes (heuristic based on sequence length)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    # ------------------------------------------------------------------
    # 10) Launch Triton kernel: C = A @ Bᵀ  (batch * hidden) times
    # ------------------------------------------------------------------
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows of C
        triton.cdiv(N, BLOCK_N),   # cols of C
        total,                     # one program per (batch * hidden)
    )
    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0),   # stride_ab  (batch stride)
        left_flat.stride(1),   # stride_am  (row stride)
        left_flat.stride(2),   # stride_ak  (col stride)
        right_flat.stride(0),  # stride_bb  (batch stride)
        right_flat.stride(2),  # stride_bk  (col‑k stride for Bᵀ)
        right_flat.stride(1),  # stride_bn  (row‑n stride for Bᵀ)
        out_flat.stride(0),    # stride_cb  (batch stride)
        out_flat.stride(1),    # stride_cm  (row stride)
        out_flat.stride(2),    # stride_cn  (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 11) Reshape back to the original layout [B, N, N, H] (fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 12) Hidden‑dim LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate               # broadcasted element‑wise
    out = F.linear(out, w_to_out)      # [B, N, N, D]  fp16

    # ------------------------------------------------------------------
    # 13) Cast back to float32 (expected downstream dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
