"""
TriMul outgoing update – Triton‐accelerated core

The heavy part of the AlphaFold3 TriMul forward pass is the
einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of H (hidden_dim) independent matrix multiplications
left_d @ right_d.T for d = 0 … hidden_dim‑1.
The kernel below computes this operation on the GPU.
Each program instance (a Triton “thread”) evaluates a tile of
size (BLOCK_M, BLOCK_N, H_BLOCK) producing a vector of H_BLOCK
channel values for every (i, j) pair in the tile.
The accumulated result is stored in FP16 and later processed with
layer‑norms and linear projections (performed in PyTorch).

Key ideas
* Tile the sequence dimensions (i, j) with BLOCK_M×BLOCK_N threads.
* Loop over the reduction dimension k in blocks of size BLOCK_K.
* For each k‑tile load the (i,k,*) and (j,k,*) vectors (size H_BLOCK)
  in FP16, cast to FP32 for accumulation.
* Perform an outer‑product for each k element:
        acc[i,j,:] += left[i,k,:] * right[j,k,:]
* Mask out‑of‑bounds indices (sequence length and hidden‑dim tail).
* Launch the kernel for each hidden‑dim chunk (H_offset) because H may
  be larger than the compile‑time H_BLOCK.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    H_offset,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, H_BLOCK: tl.constexpr
):
    """
    Compute for a batch element b:
        out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]
    for a tile (i ∈ BLOCK_M, j ∈ BLOCK_N, h ∈ H_BLOCK).
    """
    pid_i = tl.program_id(0)      # tile index over i
    pid_j = tl.program_id(1)      # tile index over j
    pid_b = tl.program_id(2)      # batch index

    # absolute indices inside the full tensor
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = H_offset + tl.arange(0, H_BLOCK)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator for the tile (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N, H_BLOCK), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks of BLOCK_K
    # ------------------------------------------------------------------
    for base_k in range(0, N, BLOCK_K):
        # vector of k indices for the current tile (size BLOCK_K)
        k_idx = base_k + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ------------------------------------------------------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K, H_BLOCK)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + k_idx[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_d
        )
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        left_tile = tl.load(left_ptrs, mask=mask_left, other=tl.float16(0)).to(tl.float32)

        # ------------------------------------------------------------------
        # Load right tile: shape (BLOCK_N, BLOCK_K, H_BLOCK)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + offs_j[:, None, None] * stride_right_i
            + k_idx[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_d
        )
        mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        right_tile = tl.load(right_ptrs, mask=mask_right, other=tl.float16(0)).to(tl.float32)

        # ------------------------------------------------------------------
        # Accumulate outer‑product over the inner k dimension
        # ------------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            # left_vec: (BLOCK_M, H_BLOCK), right_vec: (BLOCK_N, H_BLOCK)
            left_vec = left_tile[:, kk, :]          # M × H
            right_vec = right_tile[:, kk, :]        # N × H
            # outer product broadcast → (M, N, H)
            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # ------------------------------------------------------------------
    # Store the result back to global memory (FP16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_d
    )
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]  (optional)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and ``nomask``)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])     # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy reduction kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Allocate output buffer for the tiled multiplication
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Launch the Triton kernel in hidden‑dim chunks
    # ------------------------------------------------------------------
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 16      # rows per tile
    BLOCK_N = 16      # cols per tile
    BLOCK_K = 16      # reduction tile size
    H_BLOCK = 32      # hidden‑dim sub‑tile (compile‑time constant)

    # Grid dimensions: (i‑tiles, j‑tiles, batch)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs,
    )
    for h_start in range(0, hidden_dim, H_BLOCK):
        trimul_outgoing_kernel[grid](
            left_ptr=left.data_ptr(),
            right_ptr=right.data_ptr(),
            out_ptr=out.data_ptr(),
            B=bs,
            N=N,
            H=hidden_dim,
            stride_left_b=stride_left[0],
            stride_left_i=stride_left[1],
            stride_left_k=stride_left[2],
            stride_left_d=stride_left[3],
            stride_right_b=stride_right[0],
            stride_right_i=stride_right[1],
            stride_right_k=stride_right[2],
            stride_right_d=stride_right[3],
            stride_out_b=stride_out[0],
            stride_out_i=stride_out[1],
            stride_out_j=stride_out[2],
            stride_out_d=stride_out[3],
            H_offset=h_start,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            H_BLOCK=H_BLOCK,
        )

    # ------------------------------------------------------------------
    # Post‑processing in FP32 (layer‑norm, gating, final projection)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    # Layer‑norm on the hidden dimension
    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the output gate
    out_fp32 = out_fp32 * out_gate

    # Final linear projection back to the original dim
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
