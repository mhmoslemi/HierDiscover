def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    bh_idx = tl.program_id(axis=0)  # B*H
    block_i = tl.program_id(axis=1)
    block_j = tl.program_id(axis=2)

    # Compute batch and hidden channel
    b = bh_idx // H
    h = bh_idx % H

    # Pointers base
    left_ptr = left_ptr + b * stride_l_b + h * stride_l_h
    right_ptr = right_ptr + b * stride_r_b + h * stride_r_h
    out_ptr = out_ptr + b * stride_o_b + h * stride_o_h

    i_offsets = block_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offsets = block_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        k_base = k * BLOCK_K

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5
    device = input_tensor.device
    # Input dims
    B, N, _, C = input_tensor.shape
    # LayerNorm on input (dim)
    # Compute mean and var across last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    # Apply weight & bias
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask = mask.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask
    else:
        mask = None
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute tri-mul using Triton kernel
    # Prepare output tensor
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # compute strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # choose block sizes based on N size
    # pick 128 for i/j if N>=256 else 64...
    if N <= 256:
        BLOCK_I = 64
        BLOCK_J = 64
        BLOCK_K = 32
    else:
        BLOCK_I = 128
        BLOCK_J = 128
        BLOCK_K = 32

    # grid
    grid = (B * hidden_dim, triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))
    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K
    )
    # Now out is shape [B,N,N,hidden_dim]

    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = ((out - out_mean) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - out_mean) * torch.rsqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # multiply by out_gate
    out_norm = out_norm * out_gate

    # final linear projection
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out_final

used in AlphaFold‑3 and related protein‑structure models.  The
computation consists of:

1. Layer‑norm on the input ``x`` (shape [B, N, N, C]).
2. Linear projections to a hidden dimension *H* (``left`` and ``right``).
3. Gated masking of the projections.
4. A 4‑D “tensor‑multiply”:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is equivalent to a batched matrix multiplication of size
   (N × N) for every hidden channel *h* (i.e. ``left`` @ ``right``ᵀ).
   The heavy inner product is performed by a custom Triton kernel
   that tiles the ``i``, ``j`` and reduction ``k`` dimensions.
5. Layer‑norm on the hidden result, gating with ``out_gate`` and a
   final linear projection back to the original channel dimension.

Only the “tensor‑multiply” (step 4) is written in Triton – the rest
uses regular PyTorch ops for simplicity and numerical stability.
The kernel processes a single (batch, hidden‑channel) pair at a time
and tiles the output with configurable block sizes (default 128×128
output tiles, 32‑element reduction tiles).  Boundary handling is
performed with masks, and all accumulations are done in float32.
The final output is returned as a float32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *Pointer* to the projected & gated left tensor*
    right_ptr,              # *Pointer* to the projected & gated right tensor*
    out_ptr,                # *Pointer* to the output tensor
    B, N, H,                # batch, sequence length, hidden dimension
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # strides for out
    BLOCK_I: tl.constexpr,  # tile size in the i (row) dimension
    BLOCK_J: tl.constexpr,  # tile size in the j (col) dimension
    BLOCK_K: tl.constexpr,  # tile size in the reduction (k) dimension
):
    """
    Compute:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    for a single (b, h) pair.  The kernel is tiled over the i‑j plane
    (BLOCK_I × BLOCK_J) and reduces over k in blocks of size BLOCK_K.
    """
    # -------------------------------------------------------------
    # 1) Identify the (batch, hidden) pair handled by this program.
    # -------------------------------------------------------------
    bh_idx = tl.program_id(axis=0)               # 0 … (B*H)-1
    pid_i = tl.program_id(axis=1)                # tile in i‑direction
    pid_j = tl.program_id(axis=2)                # tile in j‑direction

    b = bh_idx // H                               # batch index
    h = bh_idx % H                                # hidden‑channel index

    # Base pointers for this (b, h) slice
    left = left_ptr + b * stride_l_b + h * stride_l_h
    right = right_ptr + b * stride_r_b + h * stride_r_h
    out = out_ptr + b * stride_o_b + h * stride_o_h

    # -------------------------------------------------------------
    # 2) Compute tile offsets
    # -------------------------------------------------------------
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Masks for out‑of‑bounds rows / columns
    i_mask = i_base < N
    j_mask = j_base < N
    # Only allocate accumulator for the valid tile size (full BLOCK_I×BLOCK_J)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # -------------------------------------------------------------
    # 3) Loop over the reduction dimension k in BLOCK_K chunks.
    # -------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K + tl.arange(0, BLOCK_K)

        # Boundary masks for k
        k_mask = k_base < N

        # ---------------------------------------------------------
        # Load left[i, k]  -> shape (BLOCK_I, BLOCK_K)
        # ---------------------------------------------------------
        #   left offset = i*stride_l_i + k*stride_l_k
        left_ptrs = left + i_base[:, None] * stride_l_i + k_base[None, :] * stride_l_k
        left_mask = i_mask[:, None] & k_mask[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ---------------------------------------------------------
        # Load right[j, k] transposed to shape (BLOCK_K, BLOCK_J)
        #   we want right_T[k, j] so that we can call tl.dot(left, right_T)
        # ---------------------------------------------------------
        #   right offset = k*stride_r_j + j*stride_r_k   (note the swapped strides)
        right_ptrs = right + k_base[:, None] * stride_r_j + j_base[None, :] * stride_r_k
        right_mask = k_mask[:, None] & j_mask[None, :]
        right_tile_T = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ---------------------------------------------------------
        # Accumulate: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        # ---------------------------------------------------------
        acc += tl.dot(left_tile, right_tile_T)   # sum over the K dimension

    # -------------------------------------------------------------
    # 4) Write the accumulated tile back to global memory.
    # -------------------------------------------------------------
    out_ptrs = out + i_base[:, None] * stride_o_i + j_base[None, :] * stride_o_j
    store_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool) or None
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with at least keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]           # C  (input channel dimension)
    hidden_dim = config["hidden_dim"]   # H (projected hidden dimension)
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (over last dim = dim)
    # ------------------------------------------------------------------
    # Compute mean / variance per (b, i, j)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    # Apply learned scale (weight) and bias
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float and the same device
        mask = mask.to(dtype).unsqueeze(-1)   # shape [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))   # later used

    # Apply gates element‑wise
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Tri‑multiplicative update (heavy part) – Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape = [B, N, N, H]

    # Allocate output tensor for the hidden result
    out_hidden = torch.empty_like(left)  # shape [B, N, N, H]

    # Ensure tensors are contiguous (required for correct strides)
    left = left.contiguous()
    right = right.contiguous()
    out_hidden = out_hidden.contiguous()

    # Strides are in element units (not bytes) – PyTorch returns exactly that.
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_hidden.stride()

    # Choose tile sizes – they work well for N up to 1024 on H100.
    # Small N benefits from smaller tiles to reduce idle threads.
    if N <= 256:
        BLOCK_I = 64
        BLOCK_J = 64
        BLOCK_K = 32
    else:
        BLOCK_I = 128
        BLOCK_J = 128
        BLOCK_K = 32

    # Grid: (batch * hidden, num_i_tiles, num_j_tiles)
    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left, right, out_hidden,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) LayerNorm on the hidden result (over hidden_dim)
    # ------------------------------------------------------------------
    out_mean = out_hidden.mean(dim=-1, keepdim=True)
    out_var = ((out_hidden - out_mean) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out_hidden - out_mean) * torch.rsqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # 7) Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    # Final linear: hidden_dim -> dim (no bias)
    out = F.linear(out_norm, weights["to_out.weight"])

    return out
```
