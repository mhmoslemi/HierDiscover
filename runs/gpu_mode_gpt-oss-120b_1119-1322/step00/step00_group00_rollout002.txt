def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # represents combined batch*head index

    # Derive batch and head indices
    b = pid_z // H
    h = pid_z % H

    # Compute offsets for i and j
    i_start = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create mask for ok indices
    i_mask = i_start < N
    j_mask = j_start < N

    # Offsets into left and right memory
    # left index: (b,i,k,h)
    # right index: (b,j,k,h)

    # Offsets for left and right for each tile at k dimension will be computed inside loop.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute batch and head index
    b = pid_z // H
    h = pid_z % H

    # compute start indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks for valid i,j
    mask_i = i < N
    mask_j = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # pointers for left (i,k)
        left_ptrs = left_ptr + b * stride_l_b + (i[:, None] * stride_l_i) + (k[None, :] * stride_l_k) + h * stride_l_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # pointers for right transposed (k,j)
        right_ptrs = right_ptr + b * stride_r_b + (k[:, None] * stride_r_k) + (j[None, :] * stride_r_j) + h * stride_r_h
        right_t = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product over k dimension
        acc += tl.dot(left, right_t)

    # write out
    out_ptrs = out_ptr + b * stride_o_b + (i[:, None] * stride_o_i) + (j[None, :] * stride_o_j) + h * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Outgoing TriMul implementation for AlphaFold‑3 style pairwise update.

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel that tiles the (i,j) plane
with BLOCK_M×BLOCK_N and reduces over k in blocks of size BLOCK_K.
All surrounding LayerNorms, linear projections and gating are
handled with PyTorch for readability. The kernel works on
float32 data (the accumulator is float32) and supports arbitrary
batch size B, sequence length N and hidden dimension H.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: computes the 3‑D tensor contraction.
# -------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # tensor pointers
    B, N, H,                                     # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # strides for left   (b,i,k,h)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # strides for right  (b,j,k,h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # strides for out    (b,i,j,h)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # tile index along i
    pid_j = tl.program_id(1)          # tile index along j
    pid_z = tl.program_id(2)          # combined batch‑head index

    # ---------------------------------------------------------
    # Derive batch (b) and head (h) from pid_z
    # ---------------------------------------------------------
    b = pid_z // H
    h = pid_z % H

    # ---------------------------------------------------------
    # Compute the start indices for the i, j and k axes
    # ---------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over the reduction dimension k
    # ---------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left:   shape (BLOCK_M, BLOCK_K)  -> indices (i, k)
        left_offset = (
            b * stride_l_b
            + (i[:, None] * stride_l_i)
            + (k[None, :] * stride_l_k)
            + h * stride_l_h
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right transposed: we need (k, j) so we index k first then j
        # shape (BLOCK_K, BLOCK_N)
        right_offset = (
            b * stride_r_b
            + (k[:, None] * stride_r_k)
            + (j[None, :] * stride_r_j)
            + h * stride_r_h
        )
        right_t = tl.load(right_ptr + right_offset,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # dot accumulation over k
        acc += tl.dot(left, right_t)

    # ---------------------------------------------------------
    # Store the result
    # ---------------------------------------------------------
    out_offset = (
        b * stride_o_b
        + (i[:, None] * stride_o_i)
        + (j[None, :] * stride_o_j)
        + h * stride_o_h
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------
# Entry point used by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (bool or 0/1)
            - weights: dict of model parameters (see code)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------------------------------------------------------
    # 1) LayerNorm on the input (dim = C)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------------------------------------------------------
    # 2) Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ---------------------------------------------------------
    if mask is not None:
        mask_ex = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left = left * mask_ex
        right = right * mask_ex

    # ---------------------------------------------------------
    # 4) Gating (sigmoid of linear maps)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5) Allocate output buffer for the contraction
    # ---------------------------------------------------------
    out = torch.empty_like(left)   # shape [B,N,N,H], dtype float32

    # Ensure contiguous storage (required for stride calculations)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # ---------------------------------------------------------
    # 6) Launch Triton kernel for the O(N³) reduction
    # ---------------------------------------------------------
    B, N, _, H = left.shape
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tiling parameters – empirically good for H100 and the given sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * H,                            # batch‑head dimension
    )

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------
    # 7) Final LayerNorm over the hidden dimension
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------------------------------------------------------
    # 8) Apply the output gate and final linear projection
    # ---------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
