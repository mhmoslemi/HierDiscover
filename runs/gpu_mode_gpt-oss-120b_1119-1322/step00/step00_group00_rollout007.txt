```
@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, D, eps,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_D: tl.constexpr
):
    # program ID along rows
    row_id = tl.program_id(0)
    total_rows = B * N * N
    if row_id >= total_rows:
        return

    # Compute b,i,j
    b = row_id // (N * N)
    rem = row_id % (N * N)
    i = rem // N
    j = rem % N

    # Pointers base
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # Each thread handles one d index in a tile of size BLOCK_D
    # Use thread index across block dimension
    idx = tl.arange(0, BLOCK_D)
    # Compute sum and sumsq
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for offset in range(0, D, BLOCK_D):
        d = offset + idx
        mask = d < D
        # Load x
        x = tl.load(x_base + d * stride_xd, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.where(mask, x, 0.0)
        sum_sq += tl.where(mask, x * x, 0.0)
    # Reduce across block
    sum_val = tl.sum(sum_val, axis=0)
    sum_sq = tl.sum(sum_sq, axis=0)

    # Compute mean and var
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass to write output
    for offset in range(0, D, BLOCK_D):
        d = offset + idx
        mask = d < D
        x = tl.load(x_base + d * stride_xd, mask=mask, other=0.0).to(tl.float32)
        norm = (x - mean) * inv_std
        w = tl.load(weight_ptr + d, mask=mask).to(tl.float32) if weight_ptr != tl.nullptr else 1.0
        b_ = tl.load(bias_ptr + d, mask=mask).to(tl.float32) if bias_ptr != tl.nullptr else 0.0
        out = norm * w + b_
        tl.store(out_base + d * stride_od, out, mask=mask)

    # 8. Final linear projection

"""
TriMul (outgoing) forward pass implemented with a custom Triton LayerNorm kernel.
Algorithm
---------
1. Layer‑norm the input 4‑D tensor over the last channel dimension (dim).
2. Project the normalized tensor to a hidden dimension (hidden_dim) with three
   linear layers (left_proj, right_proj, gate_proj) – all implemented with
   torch.nn.functional.linear (highly‑optimised GEMM on GPU).
3. Apply the optional pair mask to the left/right projections.
4. Compute element‑wise sigmoid gates (left_gate, right_gate) and multiply.
5. Compute the core TriMul operation:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which is exactly a batched matrix multiplication:
        left_perm  = left.permute(0,3,1,2)   # (B,hidden,N,N)
        right_perm = right.permute(0,3,2,1)  # (B,hidden,N,N)
        out = torch.matmul(left_perm, right_perm)   # (B,hidden,N,N)
   reshape back to (B,N,N,hidden).
6. Layer‑norm the result over the hidden dimension (again with the Triton kernel).
7. Apply the output gate (sigmoid of a linear projection) and a final linear
   projection back to the original channel dimension.
All heavy linear & matmul work is handled by PyTorch's tuned kernels;
the Triton kernel fuses the mean/variance reduction and the affine
LayerNorm, reducing memory traffic and demonstrating a custom GPU kernel.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (mean/variance reduction + affine transform)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const float*   input  (B,N,N,D)
    out_ptr,              # float*         output (B,N,N,D)
    weight_ptr,           # const float*   gamma  (D,)
    bias_ptr,             # const float*   beta   (D,)
    B, N, eps,            # int32 scalars: batch, seq_len, epsilon
    stride_xb, stride_xi, stride_xj, stride_xd,   # int64 strides for input
    stride_ob, stride_oi, stride_oj, stride_od,   # int64 strides for output
    DIM: tl.constexpr,    # channel dimension (known at compile time)
    BLOCK_D: tl.constexpr # threads per block (<= 1024, multiple of 32)
):
    pid = tl.program_id(0)                      # one program per (b,i,j) row
    total_rows = B * N * N
    if pid >= total_rows:
        return

    # ----------- decode (b,i,j) from pid ----------
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # pointers to the start of this row (no d offset yet)
    x_row_ptr = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_row_ptr = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # each thread gets a unique lane id in [0, BLOCK_D)
    lane_id = tl.arange(0, BLOCK_D)

    # -------------------------------------------------
    # 1) Compute sum and sum of squares over the channel dimension
    # -------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    # loop over the channel dimension in chunks of BLOCK_D
    for offs in range(0, DIM, BLOCK_D):
        d = offs + lane_id                          # absolute channel index
        mask = d < DIM
        # load a single element per thread (float32)
        x = tl.load(x_row_ptr + d * stride_xd, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.where(mask, x, 0.0)
        sum_sq  += tl.where(mask, x * x, 0.0)

    # block‑wide reduction: every thread receives the total sum / sumsq
    sum_val = tl.sum(sum_val, axis=0)
    sum_sq  = tl.sum(sum_sq, axis=0)

    # mean and variance (scalar, broadcast to all threads)
    mean = sum_val / DIM
    var  = sum_sq / DIM - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # -------------------------------------------------
    # 2) Write normalized + affine transformed output
    # -------------------------------------------------
    for offs in range(0, DIM, BLOCK_D):
        d = offs + lane_id
        mask = d < DIM
        x = tl.load(x_row_ptr + d * stride_xd, mask=mask, other=0.0).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (x - mean) * inv_std

        # affine parameters (gamma, beta)
        gamma = tl.load(weight_ptr + d, mask=mask).to(tl.float32)
        beta  = tl.load(bias_ptr   + d, mask=mask).to(tl.float32)

        out = norm * gamma + beta
        tl.store(out_row_ptr + d * stride_od, out, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension using the Triton kernel above.
    Arguments:
        x      – (B, N, N, D) tensor (float32)
        weight – (D,)   gamma
        bias   – (D,)   beta
        eps    – small constant for numerical stability
    Returns:
        Tensor of the same shape and dtype as `x`.
    """
    B, N, N2, D = x.shape
    assert N == N2, "input must be a square pair tensor"
    total_rows = B * N * N

    out = torch.empty_like(x)

    # strides in element units (PyTorch gives them in number of elements)
    sx_b, sx_i, sx_j, sx_d = x.stride()
    so_b, so_i, so_j, so_d = out.stride()

    # Ensure weight / bias are contiguous 1‑D tensors
    weight = weight.contiguous()
    bias = bias.contiguous()

    BLOCK_D = 128                     # fits most channel sizes (max 384 in tests)

    layernorm_kernel[(total_rows,)](
        x, out,
        weight, bias,
        B, N, eps,
        sx_b, sx_i, sx_j, sx_d,
        so_b, so_i, so_j, so_d,
        DIM=D,
        BLOCK_D=BLOCK_D,
        num_warps=4,                 # 4 warps per block works well on H100
    )
    return out


# ----------------------------------------------------------------------
# Entry point for the TriMul outgoing operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operation.
    data = (input_tensor, mask, weights, config)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = layernorm_triton(input_tensor, norm_weight, norm_bias, eps=eps)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – using highly‑optimised torch kernels
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Optional pair mask (shape [B, N, N])
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask broadcasted to hidden dimension
        mask_f = mask.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul operation via batched matrix multiplication
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape for batched GEMM: (B, hidden, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose the k‑axis

    # (B, hidden, N, N) = (B, hidden, N, N) @ (B, hidden, N, N)
    out = torch.matmul(left_perm, right_perm)           # (B, hidden, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (Triton) over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)        # (B, N, N, dim)

    return out
