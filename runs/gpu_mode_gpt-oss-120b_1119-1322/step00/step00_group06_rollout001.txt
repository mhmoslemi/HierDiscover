```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm on input
    # Input normalization
    # Using torch.nn.functional.layer_norm
    # param: normalized_shape = (dim,)
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    
    # Linear projects
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])   # shape [bs,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask if provided (and not nomask)
    if mask is not None and mask.numel() > 0 and not config.get("nomask", True):
        mask = mask.unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Prepare for triton kernel
    # Permute to (bs, hidden_dim, N, N) and cast to fp16
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch and hidden_dim
    B_H = bs * hidden_dim
    left_flat = left_perm.view(B_H, N, N).to(torch.float16)
    right_flat = right_perm.view(B_H, N, N).to(torch.float16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = ( (N + BLOCK_N - 1) // BLOCK_N,
             (N + BLOCK_M - 1) // BLOCK_M,
             B_H )
    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        M=N, N=N, K=N,
        stride_az=left_flat.stride(0), stride_am=left_flat.stride(1), stride_ak=left_flat.stride(2),
        stride_bz=right_flat.stride(0), stride_bk=right_flat.stride(1), stride_bn=right_flat.stride(2),
        stride_cz=out_flat.stride(0), stride_cm=out_flat.stride(1), stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    # reshape result
    out = out_flat.view(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape [bs,N,N,hidden_dim]
    # convert to float32 for next steps
    out = out.to(torch.float32)

    # to_out_norm (layernorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # multiply by out_gate
    out = out * out_gate

    # final linear to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module. Implements the heavy
    pairwise multiplication (einsum over k) using a custom Triton kernel
    that performs a batched matrix multiplication per hidden channel.
    The remaining linear and layer‑norm parts are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # input shape
    bs, N, _, _ = input_tensor.shape

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2. Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Gating values (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # 4. Optional mask
    if mask is not None and not config.get("nomask", True):
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 5. Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 6. Triton batched GEMM: out = left @ right^T over the middle dimension
    #    rearrange tensors to (B, hidden_dim, N, N) -> (Z, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    Z = bs * hidden_dim
    # flatten batch and hidden dim
    left_flat = left_perm.view(Z, N, N).to(torch.float16)
    right_flat = right_perm.view(Z, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # kernel launch parameters – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_m (rows)
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_n (cols)
        Z
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_az=left_flat.stride(0),
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bz=right_flat.stride(0),
        stride_bk=right_flat.stride(2),   # row stride = original column stride (k)
        stride_bn=right_flat.stride(1),   # col stride = original row stride (j)
        stride_cz=out_flat.stride(0),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2
    )

    # 7. Reshape back to [B,N,N,hidden_dim] (float32)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # 8. Output LayerNorm (over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # 9. Multiply by out_gate
    out = out * out_gate

    # 10. Final linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass.

The heavy operation is the 4‑dimensional einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We rewrite this as a batch of matrix multiplications:
    (B*hidden_dim) independent GEMMs of size (N,N) where
    left  -> (Z, M, K)   = (Z, N, N)
    right -> (Z, K, N)   = (Z, N, N) (transposed on‑the‑fly)

The GEMMs are executed by a custom Triton kernel (fp16 inputs,
fp32 accumulation).  All other linear layers, layer‑norms and
sigmoids are performed with PyTorch which already uses highly
optimized kernels.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (all == seq_len)
    stride_az, stride_am, stride_ak,   # A: (Z, M, K)
    stride_bz, stride_bk, stride_bn,   # B: (Z, K, N)  (transposed on‑the‑fly)
    stride_cz, stride_cm, stride_cn,   # C: (Z, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM with fp32 accumulation."""
    pid_z = tl.program_id(2)               # batch*hidden dimension
    pid_m = tl.program_id(0)               # row block
    pid_n = tl.program_id(1)               # col block

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the current tile of A and B
    a_ptrs = A_ptr + pid_z * stride_az \
                     + offs_m[:, None] * stride_am \
                     + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_z * stride_bz \
                     + offs_k[:, None] * stride_bk \
                     + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0, dtype=tl.float16)
        acc += tl.dot(a, b)      # (BLOCK_M,BLOCK_K)·(BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # write result back
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_z * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    hidden_dim = config["hidden_dim"]

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape -> [B, N, N, 1] to broadcast over hidden dim
        mask = mask.to(x.dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the Triton GEMM
    #    (B, N, N, hidden) -> (Z, N, N) where Z = B * hidden)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    Z = bs * hidden_dim

    left_flat = left_perm.view(Z, N, N).to(torch.float16)
    right_flat = right_perm.view(Z, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 7. Launch Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # pid_m  (rows)
        (N + BLOCK_N - 1) // BLOCK_N,   # pid_n  (cols)
        Z                               # pid_z  (batch*hidden)
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_az=left_flat.stride(0),
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bz=right_flat.stride(0),
        stride_bk=right_flat.stride(2),   # row (k) stride = original column stride
        stride_bn=right_flat.stride(1),   # col (j) stride = original row stride
        stride_cz=out_flat.stride(0),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 10. Apply out‑gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
