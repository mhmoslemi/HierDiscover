@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_akb, # for a: [M,N,K] -> stride for M (batch dim), N, K
    stride_bk, stride_bm, stride_bn, # for b: [M,K,N] => stride for M, K, N
    stride_cm, stride_cn, stride_ck,  # for c: [M,N,N] => we can treat stride_m (M), stride_n (N), stride_n2 (N) same as ... 
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for out of bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + (offs_m[:, None] * stride_am
                         + offs_k[None, :] * stride_ak)  # shape (BLOCK_M, BLOCK_K)
        b_ptrs = b_ptr + (offs_k[:, None] * stride_bk
                         + offs_n[None, :] * stride_bn)  # shape (BLOCK_K, BLOCK_N)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16) if x.dtype == torch.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(2)  # which batch (M index)
    pid_i = tl.program_id(0)  # block row (i)
    pid_j = tl.program_id(1)  # block col (j)

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load from A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch * stride_am + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load from B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + batch * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
    C[m,i,j] = sum_k A[m,i,k] * B[m,k,j]
    where A: (M,N,K), B: (M,K,N), C: (M,N,N).
    """
    batch = tl.program_id(2)          # M dimension = batch * hidden_dim
    pid_i = tl.program_id(0)          # output i-block
    pid_j = tl.program_id(1)          # output j-block

    # Compute the absolute coordinates of the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[m, i, k]
        a_ptrs = a_ptr + batch * stride_am + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B[m, k, j]
        b_ptrs = b_ptr + batch * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store to C[m, i, j]
    c_ptrs = c_ptr + batch * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """TriMul “outgoing” forward pass.

    This implementation follows the reference PyTorch module but offloads the
    core N×N×N tensor contraction (einsum
    `... i k d, ... j k d -> ... i j d`) to a fused Triton kernel. The kernel
    treats each hidden dimension as an independent batch and computes
    `C = A @ Bᵀ` where A and B are the left/right projected tensors.
    """
    # Unpack
    x, mask, weights, config = data
    # x: [B, N, N, dim]
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on channel dimension -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections (no bias) -----
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Mask (optional) -----
    # config may contain "nomask": True if there is no mask
    if not config.get("nomask", False) and mask is not None:
        # Convert mask to float and broadcast
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Gating -----
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Core tri-mul contraction via Triton -----
    # reshape to (M, N, N) where M = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, hidden, N, N]  (i,k)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous()     # [B, hidden, N, N]  (k,j)

    M = B * hidden_dim
    # flatten the batch*hidden_dim dimension
    A = left_perm.view(M, N, N)          # (M, i, k)
    B_mat = right_perm_T.view(M, N, N)   # (M, k, j)

    # Cast to fp16 for the heavy matmul, keep accumulator in fp32
    A_fp16 = A.to(torch.float16)
    B_fp16 = B_mat.to(torch.float16)

    # Output placeholder (fp32)
    C = torch.empty((M, N, N), dtype=torch.float32, device=x.device)

    # Strides (after .contiguous())
    stride_am, stride_an, stride_ak = A_fp16.stride()   # (M, N, K) strides
    stride_bm, stride_bk, stride_bn = B_fp16.stride()
    stride_cm, stride_cn, stride_ck = C.stride()

    # Choose block sizes (tuned for H100)
    # Block sizes are compile‑time constants; we pick modest values that work for all test configs.
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32

    # Grid: (tiles_i, tiles_j, M)
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)

    # Launch kernel
    batched_matmul_kernel[grid](
        A_fp16,
        B_fp16,
        C,
        M, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- Output LayerNorm & final linear -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

# -*- coding: utf-8 -*-
"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over its channel dimension.
2. Project the normalized tensor with two linear layers (left/right) – no bias.
3. (Optionally) apply a binary mask to the projected tensors.
4. Compute three sigmoidal gates from the same normalized tensor and
   element‑wise multiply them into left/right.
5. The core operation is the einsum:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
       out = left @ rightᵀ
   for each (batch * hidden_dim) matrix of shape (N,N).  This is
   implemented as a custom Triton kernel that treats every hidden
   dimension as an independent batch, loops over the K dimension in
   blocks and accumulates in FP32.
6. Reshape the result back to [B,N,N,hidden_dim], apply a second
   LayerNorm, multiply by the output gate and finally project back to
   the original channel dimension.

The kernel runs on an H100 (Trition 3.3.1) and uses FP16 arithmetic for
the heavy N³ work while keeping FP32 accumulators for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C[m,i,j] = Σ_k A[m,i,k] * B[m,k,j]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute many independent N×N matrix products.

    A  : (M, N, K)
    B  : (M, K, N)
    C  : (M, N, N)   where   C[m,i,j] = Σ_k A[m,i,k] * B[m,k,j]
    """
    batch = tl.program_id(2)                # index over the "M" dimension
    pid_i = tl.program_id(0)                # tile along output row (i)
    pid_j = tl.program_id(1)                # tile along output column (j)

    # ------------------------------------------------------------------
    # Block coordinates
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A[m,i,k]  (shape BLOCK_M×BLOCK_K)
        a_ptrs = a_ptr + batch * stride_am \
                 + offs_i[:, None] * stride_an \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B[m,k,j]  (shape BLOCK_K×BLOCK_N)
        b_ptrs = b_ptr + batch * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # FP16·FP16 → FP32 accumulation

    # ------------------------------------------------------------------
    # Store the tile to C[m,i,j]
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch * stride_cm \
             + offs_i[:, None] * stride_cn \
             + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor (bool/float) shape [B, N, N] or None
        * weights_dict : mapping from string keys to torch.Tensor weights
        * config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Layer‑norm on the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (nomask → ignore mask)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    #   left : (B, hidden, N, N)  – rows = i, cols = k
    #   right_T : (B, hidden, N, N) – rows = k, cols = j
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, i, k)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous()      # (B, H, k, j)

    M = B * hidden_dim                                          # total independent matrices

    A = left_perm.view(M, N, N)                                 # (M, i, k)
    B_mat = right_perm_T.view(M, N, N)                          # (M, k, j)

    # Use FP16 for the heavy multiplication (memory‑bandwidth reduction)
    A_fp16 = A.to(torch.float16)
    B_fp16 = B_mat.to(torch.float16)

    # Output placeholder (FP32 accumulator)
    C = torch.empty((M, N, N), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------
    # Strides (contiguous tensors guarantee the expected layout)
    # ------------------------------------------------------------------
    stride_am, stride_an, stride_ak = A_fp16.stride()   # (M,N,K)
    stride_bm, stride_bk, stride_bn = B_fp16.stride()   # (M,K,N)
    stride_cm, stride_cn, stride_ck = C.stride()        # (M,N,N)

    # ------------------------------------------------------------------
    # Tile size selection – works for all test configurations
    # ------------------------------------------------------------------
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32

    # Grid: (tiles_i, tiles_j, batch‑matrix count)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M,
    )

    # ------------------------------------------------------------------
    # Launch the Triton kernel
    # ------------------------------------------------------------------
    batched_matmul_kernel[grid](
        A_fp16,
        B_fp16,
        C,
        M,
        N,
        N,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original 4‑D layout
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
