"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The heavy O(N³) step:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is computed as a batched matrix multiply per (batch, hidden) pair
(left @ right.T).  The kernel loads tiles of the two (N×N) matrices,
accumulates in FP32 and writes the result back in FP32.  All other
operations (LayerNorm, linear projections, gating) are performed with
native PyTorch functions.  The implementation works for the shapes
used in AlphaFold‑3 (B∈{1,2}, N≤1024, hidden_dim=128) and respects the
required masks and weight tensors.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # runtime sizes
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right.T for each (batch, hidden) pair."""
    pid_bc   = tl.program_id(0)   # batch*hidden index
    pid_tile = tl.program_id(1)   # tiled i/j index

    # Decode batch and hidden channel
    b = pid_bc // D
    d = pid_bc % D

    # Tile layout
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    i0 = (pid_tile // num_tiles_j) * BLOCK_M
    j0 = (pid_tile %  num_tiles_j) * BLOCK_N

    # Offsets in each dimension
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Base pointers for this (b,d)
    left_base  = left_ptr  + b * stride_left_batch  + d * stride_left_d
    right_base = right_ptr + b * stride_right_batch + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_batch   + d * stride_out_d

    # Pointers to the current tile:
    #   A: (M,K) from left  -> left[i,k]
    #   B: (K,N) from right -> right[j,k] (transposed load)
    a_ptrs = left_base + (offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
    b_ptrs = right_base + (offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)  # (M,K)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)  # (K,N)
        acc += tl.dot(a, b)   # (M,N) accumulation in FP32

        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # Write the result
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights:     dict of torch.Tensors for all linear / LN parameters
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Heavy O(N³) contraction via Triton kernel
    # -----------------------------------------------------------------
    # Cast to half for bandwidth‑limited kernel
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device)

    # Tensor strides (in elements, not bytes)
    lsb, lsi, lsk, lsd = left_h.stride()
    rsb, rsi, rsk, rsd = right_h.stride()
    osb, osi, osj, osd = out.stride()

    # Tile sizes tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch * hidden_dim) x (num_i_tiles * num_j_tiles)
    num_i_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * hidden_dim, num_i_tiles * num_j_tiles)

    trimul_matmul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        lsb,
        lsi,
        lsk,
        lsd,
        rsb,
        rsi,
        rsk,
        rsd,
        osb,
        osi,
        osj,
        osd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + gating + final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
