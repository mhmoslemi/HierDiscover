"""
TriMul (outgoing) – fast forward implementation.

The module receives a 4‑D tensor x ∈ ℝ^{B×N×N×C}.  The forward pass consists of
1) LayerNorm over the last dimension C.
2) Two linear projections (left/right) from C → H (hidden_dim) and a mask
   multiplication.
3) Three gating projections (left_gate, right_gate, out_gate) from C → H,
   followed by a sigmoid and element‑wise gating of the projected tensors.
4) A “tri‑multiplicative’’ reduction:
        out_{b,i,j,h} = Σ_{k=0}^{N‑1} left_{b,i,k,h} * right_{b,j,k,h}
   which is exactly a batched matrix multiplication
        C = A @ B^{T}   with A,B ∈ ℝ^{(B·H)×N×N}.
   The reduction is performed by a custom Triton kernel that:
   – tiles the output with BLOCK_M × BLOCK_N blocks,
   – tiles the reduction over K with BLOCK_K,
   – loads tiles in fp16, uses tl.dot (Tensor‑core) for the inner product,
   – accumulates in fp32 and writes the result as fp32.
5) LayerNorm over the hidden dimension, out‑gate multiplication and a final
   linear projection back to the original dimension C.

All heavy arithmetic (the N³ reduction) is executed on the GPU via Triton;
the remaining simple per‑element ops use PyTorch for ease of implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication C = A @ Bᵀ for tensors of shape
        A : (BATCH, M, K)
        B : (BATCH, N, K)   (interpreted as (BATCH, K, N) inside the kernel)
    The result C has shape (BATCH, M, N).

    All loads are masked to avoid out‑of‑bounds accesses.
    """
    pid_batch = tl.program_id(0)   # which batch (flattened B·H)
    pid_m = tl.program_id(1)       # row tile index
    pid_n = tl.program_id(2)       # column tile index

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the output tile
    c_ptrs = (
        C_ptr
        + pid_batch * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_Ab
            + offs_m[:, None] * stride_Am
            + offs_k[None, :] * stride_Ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # B tile: (BLOCK_K, BLOCK_N)   (B is stored as (K, N))
        b_ptrs = (
            B_ptr
            + pid_batch * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # Tensor‑core dot product, result accumulated in fp32
        acc += tl.dot(a, b)

    # -------------------- write back --------------------
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, C] (float32, on CUDA)
            mask         : torch.Tensor [B, N, N]   (bool/float, on CUDA)
            weights      : dict of torch.Tensor containing all model weights
            config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor [B, N, N, C] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension C
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (left / right) and optional mask
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    if not nomask:
        # mask shape [B, N, N] → broadcast to hidden dim
        mask_ex = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_ex
        right = right * mask_ex

    # ------------------------------------------------------------------
    # 3) Gating projections + sigmoid
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 4) Tri‑multiplicative reduction via a fused Triton kernel
    #    Compute C = A @ Bᵀ where A and B are (B·H, N, N).
    # ------------------------------------------------------------------
    #   reshape to (B, H, N, N) → (B·H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N)
    right = right.view(B * hidden_dim, N, N)

    # cast to half for the kernel (Tensor‑core friendly)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # output buffer (fp32 accumulation)
    out_h = torch.empty(
        (B * hidden_dim, N, N), dtype=torch.float32, device=device
    )

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_h.stride()
    stride_Bb, stride_Bk, stride_Bn = right_h.stride()
    stride_Cb, stride_Cm, stride_Cn = out_h.stride()

    # Tile sizes – tuned for H100 Tensor‑core (fits in registers)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B * hidden_dim,
        N,
        N,
        N,                     # K == N
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5) Output LayerNorm over hidden dimension, out‑gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]          # (C, H)

    # LayerNorm over hidden_dim
    out = (out - out.mean(dim=-1, keepdim=True)) / torch.sqrt(
        out.var(dim=-1, unbiased=False, keepdim=True) + eps
    )
    out = out * to_out_norm_weight + to_out_norm_bias

    # Apply out_gate (shape matches)
    out = out * out_gate

    # Final linear projection back to original dimension C
    out = F.linear(out, to_out_weight)   # [B, N, N, C]

    return out
