@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_row, stride_a_col, stride_a_h,
    stride_b_batch, stride_b_row, stride_b_col, stride_b_h,
    stride_c_batch, stride_c_row, stride_c_col, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_batch = B * H
    # Compute block indices
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    # batch index
    batch_idx = pid // (num_blocks_i * num_blocks_j)
    block_idx = pid % (num_blocks_i * num_blocks_j)
    block_i = block_idx // num_blocks_j
    block_j = block_idx % num_blocks_j
    # Compute starting indices
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute base pointers
    # Determine original batch and d
    b = batch_idx // H
    d = batch_idx % H
    # Starting pointer for this batch+hidden slice
    a_batch_ptr = a_ptr + b * stride_a_batch + d * stride_a_h
    b_batch_ptr = b_ptr + b * stride_b_batch + d * stride_b_h
    c_batch_ptr = c_ptr + b * stride_c_batch + d * stride_c_h

    # Offsets for i and j indices
    offs_a = i[:, None] * stride_a_row + tl.arange(0, BLOCK_K)[None, :] * stride_a_col
    offs_b = tl.arange(0, BLOCK_K)[:, None] * stride_b_row + j[None, :] * stride_b_col
    # Offsets for output
    offs_c = i[:, None] * stride_c_row + j[None, :] * stride_c_col

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        a = tl.load(a_batch_ptr + offs_a, mask=(i < N) & (k + tl.arange(0, BLOCK_K) < N), other=0.0)
        b = tl.load(b_batch_ptr + offs_b, mask=(j < N) & (k + tl.arange(0, BLOCK_K) < N), other=0.0)
        # Cast to float32 if needed
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        # increment pointers for next k block
        a_batch_ptr += BLOCK_K * stride_a_col
        b_batch_ptr += BLOCK_K * stride_b_row
    # Write result
    tl.store(c_batch_ptr + offs_c, acc.to(tl.float16), mask=(i < N) & (j < N))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # LayerNorm on last dim
    # Get weight and bias for norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])   # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    # Apply mask if present
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)
        left_proj = left_proj * mask
        right_proj = right_proj * mask
    # Multiply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # Optionally convert to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    # allocate output
    out_fp16 = torch.empty_like(left_fp16)
    # Compute strides
    B, N, _, _ = left_fp16.shape
    stride_a_batch = left_fp16.stride(0)
    stride_a_row = left_fp16.stride(1)
    stride_a_col = left_fp16.stride(2)
    stride_a_h = left_fp16.stride(3)  # should be 1
    stride_b_batch = right_fp16.stride(0)
    stride_b_row = right_fp16.stride(2)  # row k dimension (original third axis)
    stride_b_col = right_fp16.stride(1)  # column j dimension (original second axis)
    stride_b_h = right_fp16.stride(3)
    stride_c_batch = out_fp16.stride(0)
    stride_c_row = out_fp16.stride(1)
    stride_c_col = out_fp16.stride(2)
    stride_c_h = out_fp16.stride(3)
    # Launch kernel
    total_batch = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (total_batch * ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N),)
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_a_batch, stride_a_row, stride_a_col, stride_a_h,
        stride_b_batch, stride_b_row, stride_b_col, stride_b_h,
        stride_c_batch, stride_c_row, stride_c_col, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4 # maybe
    )
    # Convert out to float32 for following layernorm etc.
    out = out_fp16.to(torch.float32)
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    # Apply output gate
    out = out * out_gate
    # Finally linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_batch = B * H
    # compute block indices
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_i * num_blocks_j)
    block_idx = pid % (num_blocks_i * num_blocks_j)
    block_i = block_idx // num_blocks_j
    block_j = block_idx % num_blocks_j

    # compute batch and hidden index
    b = batch_idx // H
    h = batch_idx % H

    # compute base pointers for this slice
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # row / column offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, N, BLOCK_K):
        # indices for k tile
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptrs = a_base + offs_i[:, None] * stride_a_row + offs_k[None, :] * stride_a_col

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor `x` over its last dimension.
2. Compute left/right projections and their element‑wise sigmoid gates:
       left  = (x @ W_left)  * sigmoid(x @ W_left_gate)
       right = (x @ W_right) * sigmoid(x @ W_right_gate)
   Optionally mask the projected tensors (mask shape [B,N,N]).
3. Compute the core triangular multiplication
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix product per hidden channel:
       out_d = left_d @ right_dᵀ   for each d ∈ [0, hidden_dim).
   The heavy Σ‑over‑k reduction is performed by a custom Triton kernel
   that treats the (batch, hidden) pair as a fused batch dimension.
4. Apply a second LayerNorm over the hidden dimension, an output gate and
   the final linear projection back to `dim`.

The kernel runs in FP32 (loads and accumulates) but can be easily switched
to FP16 if needed.  It processes tiles of size (BLOCK_M, BLOCK_N) and loops
over the shared K dimension in tiles of size BLOCK_K.  The total number of
programs launched equals (batch * hidden_dim) × ceil(N/BLOCK_M) × ceil(N/BLOCK_N).

Only the O(N³·hidden_dim) triangular multiplication is executed on the GPU
via Triton; the remaining cheap operations (LayerNorm, Linear, Sigmoid) use
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, N, H,                           # batch, sequence length, hidden_dim
    stride_a_batch, stride_a_row, stride_a_col, stride_a_h,
    stride_b_batch, stride_b_row, stride_b_col, stride_b_h,
    stride_c_batch, stride_c_row, stride_c_col, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k a[b,i,k,h] * b[b,j,k,h] (h = hidden channel)."""
    pid = tl.program_id(0)

    total_batch = B * H
    # number of tiles in each spatial dimension
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)

    # ---- decode program id into (batch_h, i_tile, j_tile) ----
    batch_h_idx = pid // (num_blocks_i * num_blocks_j)
    tile_idx = pid % (num_blocks_i * num_blocks_j)
    tile_i = tile_idx // num_blocks_j
    tile_j = tile_idx % num_blocks_j

    # original batch and hidden channel
    b_idx = batch_h_idx // H            # 0 <= b_idx < B
    h_idx = batch_h_idx % H             # 0 <= h_idx < H

    # base pointers for this (b,h) slice
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # i and j coordinates for this tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the K dimension (the shared dimension) ----
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current K‑tile
        a_ptrs = a_base + offs_i[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        b_ptrs = b_base + offs_k[:, None] * stride_b_row + offs_j[None, :] * stride_b_col

        # masked loads (out‑of‑bounds entries are zero)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # accumulation: acc[i,j] += Σ_k a[i,k] * b[k,j]
        acc += tl.dot(a, b)

    # write result back to C
    c_ptrs = c_base + offs_i[:, None] * stride_c_row + offs_j[None, :] * stride_c_col
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask:        torch.Tensor of shape [B, N, N] or None
            - weights:     dict of torch.Tensor model parameters
            - config:      dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] on the same device.
    """
    # -------------------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # -------------------------------------------------------------------------
    # 1) Input layer‑norm
    # -------------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------------------
    # 2) Linear projections + gates
    # -------------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply optional mask (mask can be None or a bool/float tensor)
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 3) Core triangular multiplication (Triton)
    # -------------------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape
    # Allocate output tensor
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Compute strides for the (B,N,N,hidden) tensor (contiguous, last dim fastest)
    stride_a_batch = left.stride(0)   # N * N * hidden
    stride_a_row   = left.stride(1)   # N * hidden   (i dimension)
    stride_a_col   = left.stride(2)   # hidden       (k dimension)
    stride_a_hdim  = left.stride(3)   # 1

    # For the right tensor we need the K‑dimension as the “row” (k) and
    # the J‑dimension as the “column” (j) because we want rightᵀ.
    stride_b_batch = right.stride(0)                     # same as left
    stride_b_row   = right.stride(2)                     # hidden (k)
    stride_b_col   = right.stride(1)                     # N * hidden (j)
    stride_b_hdim  = right.stride(3)                     # 1

    # Output strides (same layout as inputs)
    stride_c_batch = out.stride(0)
    stride_c_row   = out.stride(1)
    stride_c_col   = out.stride(2)
    stride_c_hdim  = out.stride(3)

    # Tile sizes – chosen to fit comfortably in shared memory on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Number of program instances
    total_batch = B * hidden_dim
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_batch * grid_i * grid_j,)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_a_hdim,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_b_hdim,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        stride_c_hdim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # good occupancy on H100
        num_stages=3,
    )

    # -------------------------------------------------------------------------
    # 4) Second layer‑norm, output gate and final linear projection
    # -------------------------------------------------------------------------
    # Normalize over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the learned output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
