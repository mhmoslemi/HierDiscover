def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; # In config.

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight=weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply layernorm
    # Use torch.nn.functional.layer_norm: but dimension param should be dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape B,N,N,H
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gating values
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Call Triton GEMM kernel
    # Flatten tensors contiguously; ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape

    out_mul = torch.empty_like(left) # [B,N,N,H] float32

    # Compute strides
    # Input pointer strides using .stride()
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

"""
TriMul (outgoing) – Triton‐accelerated implementation
====================================================

This file provides a drop‑in replacement for the PyTorch TriMul module
used in AlphaFold‑3 and related protein‑structure models.  The heavy
pairwise “einsum”  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is implemented as a batched GEMM kernel in Triton that operates on
the hidden dimension `hidden_dim` (treated as an extra batch axis).
All other operations (LayerNorm, Linear projections, gating and the
final Linear projection) are performed with plain PyTorch for simplicity.

The kernel processes a tile of the output matrix (i‑tile × j‑tile) for a
single batch `b` and a single hidden channel `h`.  The computation is
performed in float32 for numerical stability while the memory traffic
remains low thanks to tiling (BLOCK_M = BLOCK_N = BLOCK_K = 64).  The
kernel works for any batch size ≤ 2, sequence length up to 1024 and
hidden dimensions 128/384.

The public entry point is `custom_kernel(data)`, where `data` is a tuple

    (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

and the returned tensor has shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: one hidden channel per program instance
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # *float32   (B,N,N,hidden)
    right_ptr,              # *float32   (B,N,N,hidden)
    out_ptr,                # *float32   (B,N,N,hidden)

    B, N, H,                # grid shape parameters
    # strides of left tensor
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides of right tensor
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    # strides of output tensor
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k reduction dimension
):
    """Compute one tile of the TriMul reduction for a single hidden channel."""
    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # decode program id into (batch, hidden, i_tile, j_tile)
    b = pid // (H * num_i * num_j)
    tmp = pid % (H * num_i * num_j)
    h = tmp // (num_i * num_j)
    tmp = tmp % (num_i * num_j)
    i_tile = tmp // num_j
    j_tile = tmp % num_j

    # linear offsets for the tile
    i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundary handling
    mask_i = i < N
    mask_j = j < N

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    for k_tile in range(0, N, BLOCK_K):
        k = k_tile + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---- load left[i, k, h] -------------------------------------------------
        # layout: (B, N_i, N_k, H)
        left_offset = (
            b * stride_left_batch
            + i[:, None] * stride_left_i
            + k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # (BLOCK_M, BLOCK_K)

        # ---- load right_T[k, j, h]  (i.e. right[j, k, h] transposed) -------------
        # we want a matrix (BLOCK_K, BLOCK_N) where first axis is k
        right_offset = (
            b * stride_right_batch
            + k[:, None] * stride_right_k
            + j[None, :] * stride_right_i   # j uses the stride of the second dim (i)
            + h * stride_right_h
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)   # (BLOCK_K, BLOCK_N)

        # ---- dot product ---------------------------------------------------------
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write output tile
    # ------------------------------------------------------------------
    out_offset = (
        b * stride_out_batch
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptr + out_offset,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )
    # end kernel


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Arguments:
        data: tuple (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (all are 2‑D or 1‑D as in the reference)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()  # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5) Apply gating to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul multiplication – Triton GEMM over the k dimension
    # ------------------------------------------------------------------
    # Ensure contiguity for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape

    # Allocate output tensor (same dtype as inputs)
    out_mul = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Strides (in elements, not bytes) – taken directly from PyTorch tensors
    # ------------------------------------------------------------------
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out_mul.stride()

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Number of tiles in i/j dimensions
    num_i_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N

    # Total number of programs = B * H * num_i_tiles * num_j_tiles
    total_programs = B * H * num_i_tiles * num_j_tiles

    grid = (total_programs,)

    # Launch the Triton kernel
    trimul_outgoing_kernel[grid](
        left,
        right,
        out_mul,
        B,
        N,
        H,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_batch,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,    # tuned for H100
    )
    # Synchronize to guarantee completion before proceeding
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7) Post‑multiplication LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_mul,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8) Apply the output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
```
