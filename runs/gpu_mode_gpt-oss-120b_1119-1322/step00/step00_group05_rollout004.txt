import torch
import triton
import triton.language as tl

# Triton kernel definition
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # column block
    pid_b = tl.program_id(2)  # batch block (b*hidden)

    # Compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # Compute pointers + masks
        a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)
        # dot product across K
        acc += tl.dot(a, b)
    # Store result
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the outgoing TriMul module.
    ...
    """
    import torch
    import triton
    import triton.language as tl
    
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    # weights loading
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    
    # linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)
    
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate
    
    # apply mask if provided
    if mask is not None:
        mask = mask.to(left.dtype)  # ensure same dtype
        mask = mask.unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask
    
    # Convert to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    
    B, N, _, _ = left.shape
    # rearrange to (B, hidden_dim, N, N) and transpose right on last two axes
    left_perm = left.permute(0, 3, 1, 2)    # (B, H, N, N) where axes (i,k)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N) where axes (k,j)
    
    # flatten batch and hidden_dim
    batch_hidden = B * hidden_dim
    left_flat = left_perm.reshape(batch_hidden, N, N)
    right_flat = right_perm.reshape(batch_hidden, N, N)
    
    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)
    
    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_hidden)
    
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    
    # reshape back to original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # (B,N,N,hidden_dim)
    
    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    
    out = out * out_gate
    
    # final linear projection to original dim
    out = torch.nn.functional.linear(out, to_out_weight)
    
    return out

"""
custom_kernel: Triton‑accelerated forward pass for the outgoing TriMul module.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a fused batched fp16 GEMM written in Triton.
All other operations (LayerNorms, linear projections, gating and the final
projection) are carried out with PyTorch – they are cheap compared with the
contraction. The kernel works for any batch size (≤2), sequence length (up to
1024) and hidden dimension (≤384). The final output is returned as float32.
"""

import torch
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the core contraction.
# ---------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all == seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)      # row‑tile of C
    pid_n = tl.program_id(1)      # col‑tile of C
    pid_b = tl.program_id(2)      # batch × hidden index (flattened)

    # ------------------- offsets within the block -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- loop over K dimension --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # pointers into A and B (including batch offset)
        a_ptrs = a_ptr + pid_b * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_b * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # load as fp16, promote to fp32 for the dot
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # block‑dot (M×K) × (K×N) → (M×N)
        acc += tl.dot(a, b)

    # ------------------- write result ----------------------------
    c_ptrs = c_ptr + pid_b * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    # store as fp16 (the rest of the model works in fp32)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ---------------------------------------------------------------
# Entry point used by the evaluator
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul (outgoing) implementation.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:        torch.Tensor [B, N, N] or None
            - weights:     dict with all linear/LayerNorm parameters
            - config:      dict with 'dim' and 'hidden_dim'
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    import torch
    import triton

    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure same dtype and broadcast correctly
        mask = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Move the heavy contraction to Triton (batched fp16 GEMM)
    # ------------------------------------------------------------------
    # cast to fp16 for Tensor‑Core efficiency
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # permute to (B, hidden, N, N).  For `right` we need the K‑axis first.
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, k, j)

    # flatten (B, hidden) → batch for the kernel
    batch_hidden = B * hidden_dim
    left_flat = left_perm.reshape(batch_hidden, N, N)
    right_flat = right_perm.reshape(batch_hidden, N, N)

    # allocate output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_hidden               # batch × hidden
    )

    # launch the Triton kernel
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                         # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout and promote to fp32 for the rest
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # → [B,N,N,dim]

    return out
