    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # get dtype

    # Layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Left proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # shape (B,N,N,hidden_dim)
    # Right proj
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present and not nomask config param can be determined by mask shape
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    BLOCK_N = 64

@triton.jit
def trimul_outgoing_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute batch and hidden index
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # Compute offset base for this batch+hidden
    # Compute pointer offsets: each offset is in elements
    # Base pointer offsets:
    offs_batch = batch_idx * stride_l_b + hidden_idx * stride_l_h

    # Offsets for i/j/k within block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Make sure i/j valid
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to left and right subtiles
        # left[batch_idx, offs_i, offs_k, hidden_idx]
        a_ptrs = left_ptr + (offs_i[:, None] * stride_l_i
                               + offs_k[None, :] * stride_l_k
                               + offs_batch)
        # right[batch_idx, offs_j, offs_k, hidden_idx] but we need transposed: use (offs_k, offs_j)
        b_ptrs = right_ptr + (offs_k[:, None] * stride_r_k
                               + offs_j[None, :] * stride_r_j

        # load with mask

        # Compute dot product
        acc += tl.dot(a, b)

    # Write out result
    c_ptrs = out_ptr + (offs_i[:, None] * stride_o_i
                         + offs_j[None, :] * stride_o_j
                         + batch_idx * stride_o_b
                         + hidden_idx * stride_o_h)

import torch
import triton
import triton.language as tl

@triton.jit
def tr_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    using a tiled GEMM implementation in Triton.
    left and right are half‑precision, accumulator is float32, output is float32.
    """
    pid_i = tl.program_id(0)  # block along M dimension (i)
    pid_j = tl.program_id(1)  # block along N dimension (j)
    pid_bh = tl.program_id(2) # combined batch * hidden channel

    # Decode batch and hidden ids
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # Base offsets for this batch+hidden slice
    left_base = batch_idx * stride_l_b + hidden_idx * stride_l_h
    right_base = batch_idx * stride_r_b + hidden_idx * stride_r_h
    out_base = batch_idx * stride_o_b + hidden_idx * stride_o_h

    # Compute the 2‑D coordinates within the M‑N tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: left[batch, i, k, hidden]
        a_ptrs = left_ptr + left_base \
                 + offs_i[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: right[batch, j, k, hidden] → we need (k, j) layout
        b_ptrs = right_ptr + right_base \
                 + offs_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Half‑precision matmul, accumulate in FP32
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = out_ptr + out_base \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    """
    Triton implementation of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape (B, N, N, dim) in float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm over the last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask
    if not config.get("nomask", True) and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply gates
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # Prepare output tensor
    batch, N, _, _ = left.shape
    out = torch.empty(batch, N, N, hidden_dim,
                      dtype=torch.float32, device=left.device)

    # Strides for the 4‑D tensors (B,N,N,H) – they are contiguous in memory
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tiling parameters – chosen to fit H100 shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch * hidden_dim,
    )

    tr_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        batch,
        N,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Gated output
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) kernel for AlphaFold3‑style pairwise updates.

Algorithm
---------
1. Layer‑norm the [B, N, N, C] input.
2. Project to a hidden dimension H using two bias‑free linear layers
   (left_proj, right_proj) and apply an optional mask.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized input and multiply them element‑wise with the projected tensors.
4. The core operation is
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM over the sequence dimension N for every hidden
   channel.  This reduction is implemented in Triton (half‑precision inputs,
   FP32 accumulation) and written to a float‑32 output.
5. Layer‑norm the intermediate result over H, apply the out_gate and a final
   linear projection back to the original channel dimension C.

The heavy N³ reduction (step 4) is the only part written in Triton; all other
operations use highly‑optimized PyTorch primitives.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tr_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute the outgoing triangle multiplicative update.
    left, right : (B, N, N, H) in fp16
    out         : (B, N, N, H) in fp32
    """
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_bh = tl.program_id(2)  # combined batch * hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden‑channel indices
    # --------------------------------------------------------------
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # Base offsets for this (batch, hidden) slice
    left_base  = batch_idx * stride_l_b + hidden_idx * stride_l_h
    right_base = batch_idx * stride_r_b + hidden_idx * stride_r_h
    out_base   = batch_idx * stride_o_b + hidden_idx * stride_o_h

    # --------------------------------------------------------------
    # Tile coordinates within the M×N plane
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile from left  : left[batch, i, k, hidden]
        a_ptrs = left_ptr + left_base \
                 + offs_i[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load a tile from right : right[batch, j, k, hidden] → need (k, j) layout
        b_ptrs = right_ptr + right_base \
                 + offs_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Half‑precision dot product, accumulate in FP32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the result back to global memory (FP32)
    # --------------------------------------------------------------
    c_ptrs = out_ptr + out_base \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton kernel
    for the N³ reduction.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              input_tensor  – [B, N, N, C] (float32)
              mask_tensor   – [B, N, N]  (bool/int or None)
              weights_dict  – mapping described in the prompt
              config_dict   – must contain ``dim`` and ``hidden_dim`` and optional
                               ``nomask`` flag

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating scalars (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates and cast to half‑precision for the kernel
    left  = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Core N³ reduction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty(B, N, N, hidden_dim,
                      dtype=torch.float32, device=left.device)

    # Strides for contiguous tensors of shape (B, N, N, H)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – chosen to fit H100 shared memory and provide good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim,                   # batch × hidden
    )

    tr_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # element‑wise gating

    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # back to dim

    return out
