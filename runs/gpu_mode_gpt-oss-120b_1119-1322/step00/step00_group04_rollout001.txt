def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # strengths: define dtype and device
    device = input_tensor.device

    # LayerNorm: (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # input may be float32; use float32 layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projs: we can use torch.nn.functional.linear
    left_proj_weight = weights["left_proj.weight"] # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight) # shape (B,N,N,hidden_dim) - x shape (B,N,N,dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask 
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(dtype=left.dtype)  # broadcast
        left = left * mask_f
        right = right * mask_f

    # gate linear ops
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Fuse multiplication left = left * left_gate ; right = right * right_gate using Triton kernel

    # Flatten left, left_gate
    left_f = left.reshape(-1)
    left_gate_f = left_gate.reshape(-1)
    left_out_f = torch.empty_like(left_f)
    right_f = right.reshape(-1)
    right_gate_f = right_gate.reshape(-1)
    right_out_f = torch.empty_like(right_f)

    # Triton kernel
    import triton
    import triton.language as tl

    @triton.jit
    def mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
        pid = tl.program_id(0)
        block_start = pid * BLOCK_SIZE
        offs = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offs < size
        a = tl.load(a_ptr + offs, mask=mask, other=0.0)
        b = tl.load(b_ptr + offs, mask=mask, other=0.0)
        out = a * b
        tl.store(out_ptr + offs, out, mask=mask)

    size = left_f.numel()
    grid = lambda meta: (triton.cdiv(size, meta['BLOCK_SIZE']),)
    mul_kernel[grid](left_f, left_gate_f, left_out_f, size, BLOCK_SIZE=1024)
    mul_kernel[grid](right_f, right_gate_f, right_out_f, size, BLOCK_SIZE=1024)

    # reshape back
    left = left_out_f.view_as(left)
    right = right_out_f.view_as(right)

    # Now compute out via batched matrix multiplication.
    # We want left: (B,N,N,H) & right: (B,N,N,H)
    # Permute to (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    # Reshape to (B*H, N, N)
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)
    # Compute out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous() # shape (B,N,N,H)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate scaling: out_gate is shape (B,N,N,H). However out_gate we computed earlier is shape (B,N,N,hidden_dim) because we used linear. So out = out * out_gate.

    out = out * out_gate

    # to_out linear to get final dim
    to_out_weight = weights["to_out.weight"]
    # torch.nn.functional.linear works with last dimension
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton+PyTorch implementation of the “outgoing” TriMul operator.
    The kernel fuses element‑wise multiplication of the projected tensors
    with their learned gates. The expensive triple‑wise contraction
    ``out_{i,j} = Σ_k left_{i,k} * right_{j,k}`` is performed with a
    batched GEMM (left @ rightᵀ) which lets cuBLAS handle the N³ work
    efficiently on an H100.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # torch.nn.functional.linear expects (…, in_features)
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)  # (B, N, N, hidden_dim)

    # ---------- Optional mask ----------
    if mask is not None:
        # broadcast mask over hidden_dim
        mask_ = mask.unsqueeze(-1).to(dtype=left.dtype)
        left = left * mask_
        right = right * mask_
    else:
        # no mask – nothing to do
        pass

    # ---------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ---------- Fuse projection * gate with Triton ----------
    # reshape to 1‑D tensors for the kernel (contiguous)
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)

    left_fused = torch.empty_like(left_flat)
    right_fused = torch.empty_like(right_flat)

    size = left_flat.numel()
    # Kernel launch configuration
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(size, meta['BLOCK_SIZE']),)

    elementwise_mul_kernel[grid](left_flat, left_gate_flat, left_fused, size, BLOCK_SIZE=BLOCK)
    elementwise_mul_kernel[grid](right_flat, right_gate_flat, right_fused, size, BLOCK_SIZE=BLOCK)

    # reshape back
    left = left_fused.view_as(left)   # (B, N, N, hidden_dim)
    right = right_fused.view_as(right)

    # ---------- Triple multiplication via batched GEMM ----------
    # bring hidden dimension to the batch axis: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # out_mat[i] = left_mat[i] @ right_mat[i].T  -> shape (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # (B, N, N, dim)

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The module performs:
1) LayerNorm over the input pairwise tensor.
2) Linear projections to a hidden dimension (left/right).
3) Optional masking of the projected tensors.
4) Gated element‑wise multiplication (left * sigmoid(left_gate),
   right * sigmoid(right_gate)).  The element‑wise multiplication is
   fused in a tiny Triton kernel.
5) The cubic contraction
        out_{i,j} = Σ_k left_{i,k} * right_{j,k}
   is realised as a batched GEMM (left @ rightᵀ) after reshaping the
   hidden dimension into the batch axis, letting cuBLAS handle the
   N³ work efficiently on an H100.
6) LayerNorm + out‑gate scaling on the hidden tensor.
7) Final linear projection back to the original feature dimension.

Only the element‑wise multiplication (step 4) is executed in Triton;
the heavy N³ work uses the highly‑optimised cuBLAS GEMM.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, size,
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise a * b → out for 1‑D flattened tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with all Linear/LayerNorm parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_broadcast = mask.unsqueeze(-1).to(dtype=left.dtype)  # (B, N, N, 1)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4) Compute gating values (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Fuse projection * gate using a Triton element‑wise kernel
    # ------------------------------------------------------------------
    # Flatten tensors to 1‑D contiguous buffers for the kernel.
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)

    left_fused = torch.empty_like(left_flat)
    right_fused = torch.empty_like(right_flat)

    total_elems = left_flat.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total_elems, meta["BLOCK_SIZE"]),)

    # left = left * left_gate
    _elementwise_mul_kernel[grid](
        left_flat,
        left_gate_flat,
        left_fused,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )
    # right = right * right_gate
    _elementwise_mul_kernel[grid](
        right_flat,
        right_gate_flat,
        right_fused,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # Restore original shape (B, N, N, hidden_dim)
    left = left_fused.view_as(left)
    right = right_fused.view_as(right)

    # ------------------------------------------------------------------
    # 6) Cubic contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Put hidden dimension into the batch axis: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # Merge batch and hidden dimensions for a single batched GEMM
    left_mat = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # out_mat[i] = left_mat[i] @ right_mat[i].T   → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 8) Apply out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate  # (B, N, N, hidden_dim)

    to_out_weight = weights["to_out.weight"]  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # (B, N, N, dim)

    return out
