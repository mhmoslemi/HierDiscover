@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_hb = tl.program_id(axis=2)

    # compute batch and hidden
    b = pid_hb // H
    h = pid_hb % H

    # compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers base for left, right, out
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for k block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        # left: (i, k) shape (M,K)
        left_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # right transposed: (k, j) shape (K,N)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

        # load
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)  # left (M,K) * right (K,N) -> (M,N)
    
    # store
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # Norm: compute mean and variance across dim
    # Using torch.nn.functional.layer_norm with weight and bias
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections
    # Input flatten shape
    x_flat = x.view(-1, dim)
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(bs, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(bs, N, N, hidden_dim)

    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [bs,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["left_gate.weight"])).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["right_gate.weight"])).view(bs, N, N, hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["out_gate.weight"])).view(bs, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # Compute core tri-mul using triton
    # left, right float32
    # Allocate output
    out = torch.empty_like(left)  # shape [bs,N,N,hidden_dim] (float32)
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute strides
    # left_strides = left.stride()
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # grid dimensions
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    # import triton
    import triton
    import triton.language as tl

    @triton.jit
    def tri_mul_outgoing_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid_m = tl.program_id(axis=0)
        pid_n = tl.program_id(axis=1)
        pid_hb = tl.program_id(axis=2)

        # derive batch and hidden index
        b = pid_hb // H
        h = pid_hb % H

        # compute start offset indices
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_m = offs_m < N
        mask_n = offs_n < N

        # base pointers for this batch and hidden
        left_base = left_ptr + b * stride_l_b + h * stride_l_h
        right_base = right_ptr + b * stride_r_b + h * stride_r_h
        out_base = out_ptr + b * stride_o_b + h * stride_o_h

        # accumulator in float32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over K dimension
        for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left pointers (M x K)
            left_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
            # right pointers transposed (K x N)
            right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j

            # load blocks with masking
            left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
            right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            # dot product accumulation
            acc += tl.dot(left, right)

        # store result
        out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
        tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    # launch kernel
    tri_mul_outgoing_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M, BLOCK_N, BLOCK_K,
    )

    # now out: shape [bs,N,N,hidden_dim]

    # apply to_out_norm (LayerNorm) and gating and final linear
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # apply out_gate elementwise
    out = out * out_gate

    # final linear to dim
    out_flat = out.view(-1, hidden_dim)
    out_proj = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).view(bs, N, N, dim)

    return out_proj

    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    # if mask is None or nomask True, make mask all ones
    if mask is None or config.get("nomask", False):
        mask = torch.ones(B, N, N, dtype=input_tensor.dtype, device=input_tensor.device)
    # reshape mask for broadcasting
    mask = mask.unsqueeze(-1)  # [B,N,N,1]

    # layernorm on dim dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # flatten for linear
    x_flat = x.view(-1, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left = left * mask
    right = right * mask

    left_gate = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden_dim)

    right = right * right_gate

"""
TriMul (outgoing) kernel for AlphaFold3‑style pair updates.

Algorithm
---------
1. Layer‑norm the input tensor over the last feature dimension.
2. Apply four linear projections (left, right, three gates) without bias.
3. Mask the left/right projections (if a mask is supplied) and
   multiply them element‑wise with their respective gates.
4. Core operation: for each batch b and hidden channel h,
   compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a matrix multiply L_h @ R_hᵀ per hidden channel.
   This is implemented with a fused Triton kernel that tiles the
   i/j dimensions (BLOCK_M·BLOCK_N) and iterates over k in blocks
   (BLOCK_K).  The kernel works on float32 data for numerical
   exactness.
5. Layer‑norm the hidden result, multiply with the out‑gate,
   and apply a final linear projection back to the original feature
   dimension.

The kernel fuses the O(N³) reduction across the sequence dimension and
is tiled to keep the working set in registers/shared memory,
yielding a large speed‑up over a naïve PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,                     # const float*  [B, N, N, H]
    right_ptr,                    # const float*  [B, N, N, H]
    out_ptr,                      #       float*  [B, N, N, H]
    B, N, H,                      # problem sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,        # tile size along i
    BLOCK_N: tl.constexpr,        # tile size along j
    BLOCK_K: tl.constexpr,        # reduction tile size
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The kernel processes one (batch, hidden) pair per program
    (grid axis‑2) and tiles the (i,j) plane using a 2‑D grid.
    """
    pid_m = tl.program_id(axis=0)      # tile row index (i)
    pid_n = tl.program_id(axis=1)      # tile column index (j)
    pid_hb = tl.program_id(axis=2)     # fused batch & hidden index

    # Decode batch and hidden channel
    b = pid_hb // H
    h = pid_hb % H

    # -------------------- tile coordinates --------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  (BLOCK_M x BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right transposed: we need right[k, j] (BLOCK_K x BLOCK_N)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate the inner product over k
        acc += tl.dot(left, right)       # (BLOCK_M, BLOCK_N)

    # -------------------- store result --------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation implemented with
    a custom Triton kernel.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32)
        - mask         : torch.Tensor [B, N, N]    (float32 or bool)
        - weights      : dict of torch.Tensor containing all model weights
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # --------------------------------------------------------------
    # Optional mask handling
    # --------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=device)
    # Broadcast mask to the feature dimension
    mask = mask.unsqueeze(-1)                # [B, N, N, 1]

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # Flatten the (B,N,N) leading dimensions for a batched linear
    x_flat = x.view(-1, dim)                     # [(B*N*N), dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    left = left * mask
    right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul (custom Triton kernel)
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # shape [B, N, N, hidden_dim], float32

    # Tile sizes – tuned for H100, can be adjusted if needed
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in elements (PyTorch uses row‑major layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Grid dimensions: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, out‑gate, and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # Final projection back to the original dimension
    out = F.linear(out.view(-1, hidden_dim), weights["to_out.weight"]).view(B, N, N, dim)

    return out
