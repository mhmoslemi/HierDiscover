"""
TriMul (outgoing) – fast Triton implementation.

Algorithm
---------
1. Layer‑normalize the input over the last dimension (dim).
2. Linear projections (`left_proj`, `right_proj`) and three gating
   projections (`left_gate`, `right_gate`, `out_gate`) are computed with a
   single torch.nn.functional.linear call on the flattened (B·N·N, dim) tensor.
3. Apply the optional mask and the element‑wise sigmoid gates:
       left  = (x·W_left)   * left_gate   * mask
       right = (x·W_right)  * right_gate  * mask
4. The expensive “einsum”  
   `out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}`
   is a batched matrix multiplication `L_h @ R_hᵀ` for each hidden channel *h*.
   This is performed by a tiled GEMM kernel written in Triton.
   The kernel works on the combined batch–hidden axis (B·H) and computes
   blocks of size (BLOCK_M × BLOCK_N) with an inner‑tile size BLOCK_K.
5. A layer‑norm over the hidden dimension (`to_out_norm`) and the final
   linear projection (`to_out`) are executed with PyTorch.
   The result has shape `[B, N, N, dim]`.

Only the O(N³·H) part is accelerated with Triton; all other operations use
high‑performance PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched (over batch & hidden) matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,                # *const* fp16 left tensor   [B,N,N,H]
    right_ptr,               # *const* fp16 right tensor  [B,N,N,H]
    out_ptr,                 # *mut*   fp16 output tensor [B,N,N,H]
    B, N, H,                 # problem size
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time constants
    BLOCK_M: tl.constexpr,   # rows of output block
    BLOCK_N: tl.constexpr,   # cols of output block
    BLOCK_K: tl.constexpr,   # reduction tile size
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for every hidden channel `h`. The kernel iterates over the reduction
    dimension `k` in tiles of size BLOCK_K and accumulates in float32.
    """
    pid_bh = tl.program_id(0)          # combined batch * hidden index
    pid_i  = tl.program_id(1)          # block row
    pid_j  = tl.program_id(2)          # block col

    # --------------------------------------------------------------
    # Decode batch and hidden indices
    # --------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh %  H

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --------------------------------------------------------------
    # Offsets for the block of rows/cols we are computing
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load a tile from left (shape BLOCK_M x BLOCK_K) ----
        a_ptrs = left_base + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load a tile from rightᵀ (shape BLOCK_K x BLOCK_N) ----
        # rightᵀ_{k,j} = right_{j,k}
        b_ptrs = right_base + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ---- Accumulate dot product ----
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the result
    # --------------------------------------------------------------
    c_ptrs = out_base + \
             offs_i[:, None] * stride_out_i + \
             offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data: tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch Tensor of shape [B, N, N, dim]
        - mask: torch Tensor of shape [B, N, N] (or None)
        - weights: dict of torch Tensors containing the model weights
        - config: dict with keys:
            * dim         – input feature dimension
            * hidden_dim  – hidden dimension used inside TriMul
            * nomask      – bool, True → ignore mask
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", True) and mask is not None

    # ------------------------------------------------------------------
    # Layer‑norm on the input (last dimension)
    # ------------------------------------------------------------------
    # weights["norm.weight"] and ["norm.bias"] have shape [dim]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B,N,N,dim]

    B, N, _, _ = x.shape
    # ------------------------------------------------------------------
    # Linear projections and gates (all without bias)
    # ------------------------------------------------------------------
    # Helper: flatten batch & spatial dimensions for a single matmul call
    x_flat = x.reshape(-1, dim)                     # (B·N·N, dim)

    # Projections
    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim))
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim))
    out_gate   = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim))

    # ------------------------------------------------------------------
    # Apply mask and element‑wise gates
    # ------------------------------------------------------------------
    if use_mask:
        # mask: [B,N,N] → [B,N,N,1] with same dtype as the tensors
        mask_val = mask.to(x.dtype).unsqueeze(-1)   # broadcast over hidden
        left  = left * mask_val * left_gate
        right = right * mask_val * right_gate
    else:
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to float16 for the heavy einsum
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output buffer (fp16) for the Triton kernel
    out_fp16 = torch.empty_like(left_fp16, device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Prepare strides (in elements) for the kernel
    # ------------------------------------------------------------------
    sL_b, sL_i, sL_k, sL_h = left_fp16.stride()
    sR_b, sR_i, sR_k, sR_h = right_fp16.stride()
    sO_b, sO_i, sO_j, sO_h = out_fp16.stride()

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * hidden_dim,                     # batch * hidden
            triton.cdiv(N, BLOCK_M),           # rows
            triton.cdiv(N, BLOCK_N))           # cols

    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_i, sR_k, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # hint for H100 (tuned empirically)
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Continue the forward pass in float32
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # Output layer‑norm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out_flat = out.reshape(-1, hidden_dim)          # (B·N·N, hidden_dim)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
