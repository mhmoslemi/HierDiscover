@triton.jit
def apply_gate_mask_kernel(
    out_ptr,
    a_ptr,
    gate_ptr,
    mask_ptr,
    B,
    N,
    H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_a_b, stride_a_i, stride_a_j, stride_a_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid_bh = tl.program_id(0)  # combined batch*head dimension
    pid_i = tl.program_id(1)  # tile of i dimension
    pid_j = tl.program_id(2)  # tile of j dimension

    # Compute actual batch and head indices:
    # bh = pid_bh
    b = pid_bh // H
    h = pid_bh % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Load inputs

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch code but
* fuses the element‑wise mask·gate·projection step with a custom Triton kernel,
* performs the heavy O(N³·C_hidden) reduction as a batched matrix multiply
  (left @ rightᵀ) in FP16 for maximum throughput on H100,
* finishes with LayerNorm, output gating and a final linear projection.

All tensors are assumed to be on the CUDA device.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fuse mask, gate and projection (element‑wise multiply)
#   out = proj * gate * mask
#   proj : [B, N, N, H]  (float32)
#   gate : [B, N, N, H]  (float32)
#   mask : [B, N, N]     (float32, broadcast over H)
#   out  : [B, N, N, H]  (float32)
# -------------------------------------------------------------------------
@triton.jit
def _gate_mask_fused_kernel(
    out_ptr, proj_ptr, gate_ptr, mask_ptr,
    B, N, H,
    # strides of out   (B,N,N,H)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # strides of proj  (B,N,N,H)
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    # strides of gate  (B,N,N,H)
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    # strides of mask  (B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    # program ids: 0 -> combined batch*head, 1 -> i tile, 2 -> j tile
    pid_bh = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_j  = tl.program_id(2)

    # decode batch index b and hidden index h from the flattened dimension
    b = pid_bh // H
    h = pid_bh % H

    # offsets inside the i/j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # ----- load proj -----
    proj = tl.load(
        proj_ptr
        + b * stride_proj_b
        + h * stride_proj_h
        + offs_i[:, None] * stride_proj_i
        + offs_j[None, :] * stride_proj_j,
        mask=mask_ij,
        other=0.0,
    )
    # ----- load gate -----
    gate = tl.load(
        gate_ptr
        + b * stride_gate_b
        + h * stride_gate_h
        + offs_i[:, None] * stride_gate_i
        + offs_j[None, :] * stride_gate_j,
        mask=mask_ij,
        other=0.0,
    )
    # ----- load mask (broadcast over H) -----
    mask = tl.load(
        mask_ptr
        + b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j,
        mask=mask_ij,
        other=0.0,
    )
    # fused multiply
    out = proj * gate * mask

    # ----- store result -----
    tl.store(
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_ij,
    )

# -------------------------------------------------------------------------
def _apply_gate_and_mask(proj, gate, mask):
    """
    proj, gate : [B, N, N, H] (float32)
    mask       : [B, N, N] (float32) – already on the same device
    returns proj * gate * mask (float32) using Triton.
    """
    B, N, _, H = proj.shape
    BLOCK_M = 64
    BLOCK_N = 64

    out = torch.empty_like(proj)

    # strides
    stride_out = out.stride()
    stride_proj = proj.stride()
    stride_gate = gate.stride()
    stride_mask = mask.stride()

    # launch config
    total_bh = B * H
    grid = (
        total_bh,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _gate_mask_fused_kernel[grid](
        out,
        proj,
        gate,
        mask,
        B,
        N,
        H,
        # out strides
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        # proj strides
        stride_proj[0],
        stride_proj[1],
        stride_proj[2],
        stride_proj[3],
        # gate strides
        stride_gate[0],
        stride_gate[1],
        stride_gate[2],
        stride_gate[3],
        # mask strides
        stride_mask[0],
        stride_mask[1],
        stride_mask[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )
    # make sure the kernel is finished before we use `out`
    torch.cuda.synchronize()
    return out

# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor (or None), weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # unpack config & weights
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]   # called C_hidden in the paper

    # LayerNorm on the input channel dimension
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Projection weights (no bias)
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    # Gating weights (no bias)
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    # Output LayerNorm + final projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (float32)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)        # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid) – also float32
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 4. Mask handling (broadcasted over hidden dimension)
    # -----------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        # a tensor of ones – shape (B,N,N)
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        # ensure float32 for the kernel
        mask = mask_tensor.to(dtype=dtype)

    # -----------------------------------------------------------------
    # 5. Fuse mask + gate + projection with Triton
    # -----------------------------------------------------------------
    left = _apply_gate_and_mask(left, left_gate, mask)   # [B,N,N,hidden_dim]
    right = _apply_gate_and_mask(right, right_gate, mask)

    # -----------------------------------------------------------------
    # 6. Heavy reduction:  out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
    #    Implemented as batched matmul (left @ rightᵀ) per hidden channel.
    #    Computed in FP16 for speed, then cast back to FP32.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape

    # Cast to half‑precision for the matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Bring the hidden dimension next to batch for a single large batch matmul
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Merge B and H
    left_mat = left_perm.view(-1, N, N)      # [(B*H), N, N]
    right_mat = right_perm.view(-1, N, N)    # [(B*H), N, N]

    # Batched matrix multiplication: each row i uses the transpose of right on dim K
    out_mat_h = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [(B*H), N, N]

    # Restore original layout and cast back to fp32
    out_h = out_mat_h.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, gating and final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )                                          # [B, N, N, hidden_dim]

    out = out * out_gate                      # element‑wise gating

    # Final projection back to original channel dimension
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return out
