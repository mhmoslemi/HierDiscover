    left_ptr, right_ptr, out_ptr,
    B, N, H,

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b,i,j,h] = sum_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h]
    using a fused GEMM per (b,h) pair.
    """
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile col
    pid_task = tl.program_id(2)  # flattened (batch, hidden) index

    # Decode batch and hidden indices
    batch = pid_task // H
    h = pid_task % H

    # Starting offsets of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Compute base pointers for this (batch, h) slice
    left_base = left_ptr + batch * stride_left_b + h * stride_left_h
    right_base = right_ptr + batch * stride_right_b + h * stride_right_h
    out_base = out_ptr + batch * stride_out_b + h * stride_out_h

    # Offsets within current tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: left[batch, i, k, h]
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B_T tile (right transposed): right[batch, j, k, h] -> we need B_T[k, j]
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot‑product accumulation
        acc += tl.dot(a, b)   # a: (M, K), b: (K, N) → (M, N)

    # Store result as FP16
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

                      right: torch.Tensor,
                      out: torch.Tensor,
                      BLOCK_M=128,
                      BLOCK_N=128,
                      BLOCK_K=64):

    assert left.shape == right.shape == out.shape
    B, N, N2, H = left.shape
    assert N == N2

    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    num_tasks = B * H

             (N + BLOCK_N - 1) // BLOCK_N,

        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    right = right.contiguous()

    out = torch.empty_like(left)   # (B,N,N,hidden) in fp16

# -*- coding: utf-8 -*-
"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out[b,d] = left[b,d] @ right[b,d].T
for every batch and hidden‑dimension channel.
The kernel evaluates the above for all (b,d) in parallel using fp16
arithmetic (Tensor‑core friendly) and accumulates in fp32.
All other operations (LayerNorms, linear projections, gating) are done
with regular PyTorch primitives in fp16 for speed.
The final result is returned as float32.
"""

import torch
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum contraction.
# ------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    (b,h) is flattened into a single task dimension.
    All tensors are expected to be in FP16; accumulation is in FP32.
    """
    pid_i = tl.program_id(0)          # tile row  (i)
    pid_j = tl.program_id(1)          # tile col  (j)
    pid_task = tl.program_id(2)       # flattened (batch, hidden) index

    # Derive batch and hidden indices from the flattened task id
    batch = pid_task // H
    h = pid_task % H

    # Tile start coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Base pointers for this (batch, h) slice
    left_base = left_ptr + batch * stride_left_b + h * stride_left_h
    right_base = right_ptr + batch * stride_right_b + h * stride_right_h
    out_base = out_ptr + batch * stride_out_b + h * stride_out_h

    # Offsets inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,h]  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i +
                              offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0))

        # right[b,j,k,h] but we need its transpose w.r.t. (k,j)
        # load right_T[k,j] where k is the row dimension
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k +
                               offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0))

        # Accumulate dot product (a: MxK, b: KxN) → (M,N)
        acc += tl.dot(a, b)

    # Store the result back as FP16
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i +
                           offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_outgoing(left: torch.Tensor,
                      right: torch.Tensor,
                      out: torch.Tensor,
                      BLOCK_M: int = 64,
                      BLOCK_N: int = 64,
                      BLOCK_K: int = 64):
    """
    Simple wrapper that launches ``trimul_outgoing_kernel``.
    left/right/out must have shape [B, N, N, H] and be contiguous FP16 tensors.
    """
    B, N, N2, H = left.shape
    assert N == N2, "left/right must be square in the last two dimensions"
    assert right.shape == left.shape == out.shape

    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Number of independent GEMM tasks = B * H
    num_tasks = B * H

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            num_tasks)

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# ------------------------------------------------------------
# Entry‑point used by the evaluator.
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (bool/int) or None
        - weights      : dict of torch.Tensors, all in float32
        - config       : dict containing ``dim``, ``hidden_dim`` and a flag ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast everything to FP16 for the heavy computation path
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------
    # Input LayerNorm : normalizes over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_w, bias=norm_b, eps=1e-5
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N); broadcast over hidden dimension
        mask_f = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Ensure contiguous layout before invoking Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # Contraction via Triton (the O(N³) core)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # FP16 buffer: (B,N,N,hidden)
    _tri_mul_outgoing(left, right, out)

    # ------------------------------------------------------------------
    # Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # (B,N,N,dim)

    # Return in FP32 (the usual dtype for downstream code)
    return out.to(torch.float32)
