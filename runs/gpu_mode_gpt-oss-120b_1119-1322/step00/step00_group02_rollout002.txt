    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)  # shape [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    if not nomask:

        left = left * mask

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    right_gate = F.linear(x, right_gate_weight).sigmoid()

def trimul_einsum_kernel(A, B, C,
                         M, N, K,
                         stride_Ab, stride_Am, stride_An,
                         stride_Bb, stride_Bm, stride_Bn,
                         stride_Cb, stride_Cm, stride_Cn,

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The kernel computes the heavy part of the module:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for every batch b and hidden dimension h.
We treat (batch * hidden_dim) as a fused batch dimension and perform a
batched matrix multiplication A @ Bᵀ using Triton, which is the bottleneck
(N³·hidden_dim operations). All other operations (LayerNorm, linear
projections, sigmoids, gating, final LayerNorm and linear) are performed
with PyTorch which already provides fast GPU kernels.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimension up to 128 (or larger) and returns a float32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_einsum_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_An,
    stride_Bb, stride_Bm, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel computing C = A @ Bᵀ.

    A, B, C are 3‑D tensors with shape (batch, M, K), (batch, K, N) and
    (batch, M, N) respectively.  The batch dimension is the fused
    (batch * hidden_dim) dimension of the TriMul op.
    """
    batch_id = tl.program_id(0)      # fused batch * hidden_dim
    pid_m    = tl.program_id(1)      # tile row index
    pid_n    = tl.program_id(2)      # tile col index

    # ---------- tile coordinates ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- K‑loop ----------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = A + batch_id * stride_Ab \
                 + offs_m[:, None] * stride_Am \
                 + offs_k[None, :] * stride_An
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B block (already transposed for the matmul): (BLOCK_K, BLOCK_N)
        b_ptrs = B + batch_id * stride_Bb \
                 + offs_k[:, None] * stride_Bm \
                 + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate block matrix product
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # ---------- write back ----------
    c_ptrs = C + batch_id * stride_Cb \
             + offs_m[:, None] * stride_Cm \
             + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul forward pass (outgoing version).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights      : dict of model parameters (see code)
        - config       : dict with keys "dim", "hidden_dim", "nomask", optionally "eps"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32) – the result of the TriMul module.
    """
    # ---------------- unpack inputs ----------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # ---------------- LayerNorm on input ----------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, D]

    # ---------------- Linear projections ----------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ---------------- optional mask ----------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ---------------- Gates (sigmoids) ----------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = F.linear(x, left_gate_w).sigmoid()   # [B,N,N,H]
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------- Prepare for Triton GEMM -------------------
    # bring hidden dimension into the batch axis
    left_ph  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_ph = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # right needs to be transposed on the two N dims for the A @ Bᵀ operation
    right_T = right_ph.transpose(-2, -1).contiguous()  # [B, H, N, N]

    # flatten (batch, hidden) -> fused batch dimension
    fused_batch = bs * hidden_dim
    left_flat  = left_ph.view(fused_batch, seq_len, seq_len)        # [BH, N, N]
    right_flat = right_T.view(fused_batch, seq_len, seq_len)       # [BH, N, N]

    # allocate output tensor for the GEMM result
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------- Launch Triton kernel -------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        fused_batch,
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
    )

    _trimul_einsum_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                # M
        seq_len,                # N
        seq_len,                # K (summation dimension)
        left_flat.stride(0),    # stride batch for A
        left_flat.stride(1),    # stride row    for A (M)
        left_flat.stride(2),    # stride col    for A (K)
        right_flat.stride(0),   # stride batch for B
        right_flat.stride(1),   # stride row    for B (K)
        right_flat.stride(2),   # stride col    for B (N)
        out_flat.stride(0),     # stride batch for C
        out_flat.stride(1),     # stride row    for C (M)
        out_flat.stride(2),     # stride col    for C (N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- Reshape back to original layout -------------------
    out_per_head = out_flat.view(bs, hidden_dim, seq_len, seq_len)  # [B, H, N, N]
    out = out_per_head.permute(0, 2, 3, 1).contiguous()             # [B, N, N, H]

    # ------------------- to_out_norm + gating -------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate   # apply the output gate

    # ------------------- final linear projection -------------------
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)   # shape [B, N, N, D]

    return out
