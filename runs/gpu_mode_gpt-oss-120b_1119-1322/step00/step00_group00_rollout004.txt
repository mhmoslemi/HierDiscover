@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    # derive batch and hidden index
    batch = pid_bh // H
    hidden = pid_bh % H

    # Compute start indices
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    # accumulation accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension
    for k_offset in range(0, N, BLOCK_K):
        # compute offsets and masks for valid range
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        cur_i = i_start + tl.arange(0, BLOCK_I)
        cur_j = j_start + tl.arange(0, BLOCK_J)

        # mask for out-of-bound
        mask_i = cur_i < N
        mask_j = cur_j < N
        mask_k = cur_k < N

        # compute pointers for left (i,k) element
        left_ptrs = left_ptr + batch * stride_left_b + hidden * stride_left_h + cur_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        # load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # compute pointers for right transposed (k,j) (since we need right[j,k])
        right_ptrs = right_ptr + batch * stride_right_b + hidden * stride_right_h + cur_k[:, None] * stride_right_k + cur_j[None, :] * stride_right_i
        # mask for k and j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product contributions
        acc += tl.dot(left, right)  # left (BLOCK_I, BLOCK_K) dot right (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
    # store results
    out_ptrs = out_ptr + batch * stride_out_b + hidden * stride_out_h + cur_i[:, None] * stride_out_i + cur_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul "outgoing" forward kernel (AlphaFold3).

The module works on a 4‑D tensor X ∈ ℝ^{B×N×N×C}
and computes

    X̂ = LayerNorm(X)
    L = (X̂·Wₗ) ⊙ mask ⊙ σ(X̂·Wₗᵍ)          # left projection + gate
    R = (X̂·Wᵣ) ⊙ mask ⊙ σ(X̂·Wᵣᵍ)          # right projection + gate
    Oₖ[i,j] = Σₖ L[i,k]·R[j,k]   (for each hidden channel k)
    O = LayerNorm(O) ⊙ σ(X̂·Wₒᵍ) · Wₒ

The heavy O(N³·H) reduction (the “einsum”) is implemented in a
hand‑written Triton kernel.  The kernel treats each (batch, hidden)
pair as an independent matrix multiplication:

    (N×N) ← (N×K) × (K×N)  with K = N

where the second factor is accessed in transposed order so that the
summation index K can be reduced inside the kernel.  The kernel works
in FP16 for the two inputs (to reduce memory traffic) and accumulates
in FP32, producing a FP32 output that is subsequently processed with
LayerNorm and the final linear projection using normal PyTorch code.

The implementation is fully self‑contained and supports the various
test shapes listed in the problem description.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # half*   [B, N, N, H]
    right_ptr,         # half*   [B, N, N, H]
    out_ptr,           # float*  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute one (batch, hidden) slice of the outgoing TriMul einsum.

    For a given hidden channel h and batch b we evaluate
        out[i, j] = Σₖ left[i, k] * right[j, k]
    where left and right have shape (N, N) for the selected slice.
    """
    pid_i = tl.program_id(0)        # tile row
    pid_j = tl.program_id(1)        # tile col
    pid_bh = tl.program_id(2)       # combined batch & hidden dimension

    # ---- decode batch & hidden index ---------------------------------
    batch = pid_bh // H
    hidden = pid_bh % H

    # ---- tile start coordinates ---------------------------------------
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # ---- accumulator (FP32) -------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ---- loop over the reduction dimension K ---------------------------
    for k_base in range(0, N, BLOCK_K):
        # indices inside the current tile
        cur_i = i_start + tl.arange(0, BLOCK_I)
        cur_j = j_start + tl.arange(0, BLOCK_J)
        cur_k = k_base + tl.arange(0, BLOCK_K)

        # out‑of‑bounds masks
        mask_i = cur_i < N
        mask_j = cur_j < N
        mask_k = cur_k < N

        # ----- load left[i, k] ------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + hidden * stride_left_h
            + cur_i[:, None] * stride_left_i
            + cur_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0, dtype=tl.float16)
        left = left.to(tl.float32)   # promote for accumulation

        # ----- load right[j, k] as transposed view (k, j) --------------------
        # We need a tile of shape (BLOCK_K, BLOCK_J) where the first dim is k.
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + hidden * stride_right_h
            + cur_k[:, None] * stride_right_k   # k dimension
            + cur_j[None, :] * stride_right_i   # j dimension (original i stride)
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0, dtype=tl.float16)
        right = right.to(tl.float32)

        # ----- rank‑K dot product (BLOCK_I×BLOCK_K)·(BLOCK_K×BLOCK_J) --------
        acc += tl.dot(left, right)   # result shape (BLOCK_I, BLOCK_J)

    # ---- store the computed tile -----------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + hidden * stride_out_h
        + i_start + tl.arange(0, BLOCK_I)[:, None] * stride_out_i
        + j_start + tl.arange(0, BLOCK_J)[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "Input must be square in the two middle dimensions"
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # Handle mask (may be omitted)
    # --------------------------------------------------------------------- #
    if mask_tensor is None:
        mask = torch.ones((bs, N, N), dtype=input_tensor.dtype,
                          device=input_tensor.device)
    else:
        mask = mask_tensor.to(input_tensor.dtype)
    mask = mask.unsqueeze(-1)                     # [B, N, N, 1]

    # --------------------------------------------------------------------- #
    # Load all weight tensors
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Input layer‑norm
    # --------------------------------------------------------------------- #
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=eps)

    # --------------------------------------------------------------------- #
    # Linear projections (dim → hidden_dim) – no bias
    # --------------------------------------------------------------------- #
    left_proj = F.linear(x_norm, left_proj_weight)    # [B,N,N,H]
    right_proj = F.linear(x_norm, right_proj_weight)

    # --------------------------------------------------------------------- #
    # Gating (sigmoid) – also dim → hidden_dim
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # --------------------------------------------------------------------- #
    # Apply mask and gates (still in FP32)
    # --------------------------------------------------------------------- #
    left = left_proj * mask * left_gate
    right = right_proj * mask * right_gate

    # Cast to half for the heavy reduction (reduces memory traffic)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # --------------------------------------------------------------------- #
    # Allocate output tensor for the einsum result (float32)
    # --------------------------------------------------------------------- #
    out = torch.empty((bs, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=input_tensor.device)

    # --------------------------------------------------------------------- #
    # Strides for Triton (in elements, not bytes)
    # --------------------------------------------------------------------- #
    s_lb, s_li, s_lk, s_lh = left_h.stride()
    s_rb, s_ri, s_rk, s_rh = right_h.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # --------------------------------------------------------------------- #
    # Kernel launch configuration
    # --------------------------------------------------------------------- #
    BLOCK_I = 64   # rows per tile
    BLOCK_J = 64   # cols per tile
    BLOCK_K = 32   # reduction tile size

    grid = (
        ( (N + BLOCK_I - 1) // BLOCK_I,
          (N + BLOCK_J - 1) // BLOCK_J,
          bs * hidden_dim )   # one program per (batch, hidden) slice
    )

    # Triton kernel launch
    tri_mul_outgoing_kernel[grid](
        left_h, right_h, out,
        bs, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4  # 4 warps works well for the tested sizes
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------------- #
    # Output LayerNorm, gate, and final linear projection (hidden_dim → dim)
    # --------------------------------------------------------------------- #
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    out_norm = out_norm * out_gate
    final_out = F.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return final_out
