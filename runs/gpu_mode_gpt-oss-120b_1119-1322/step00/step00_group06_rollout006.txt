    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,

    IS_BF16: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for each batch
    a_base = a_ptr + batch_idx * stride_ab
    b_base = b_ptr + batch_idx * stride_bb
    c_base = c_ptr + batch_idx * stride_cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak

def custom_kernel(data):
    input, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # get weight tensors. Might be on GPU already.

    # compute layer norm: use torch.nn.functional.layer_norm
    # However we need to pass weight and bias: weights["norm.weight"] shape [dim], weights["norm.bias"] shape [dim].
    # Use torch.nn.functional.layer_norm(input, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5). Might need dtype.

    x = torch.nn.functional.layer_norm(input, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Compute left_proj = linear (no bias)
    left_proj = torch.nn.functional.linear(x, weights['left_proj.weight'])  # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x, weights['right_proj.weight'])
    left_gate_pre = torch.nn.functional.linear(x, weights['left_gate.weight'])
    right_gate_pre = torch.nn.functional.linear(x, weights['right_gate.weight'])
    out_gate_pre = torch.nn.functional.linear(x, weights['out_gate.weight'])

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)  # ensure same dtype
        left = left_proj * mask * left_gate
        right = right_proj * mask * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Convert to half for compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for batched gemm
    B, N, _, H = left.shape
    # permute to [B, H, N, N] then contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to [B*H, N, N]
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)  # shape [BH,N,N] float32

    # Determine strides
    stride_a_batch = left_mat.stride(0)  # Should be N*N

    stride_a_k = left_mat.stride(2)  # stride for columns (k)

        stride_ak=stride_a_k,
        stride_ab=stride_a_batch,
        stride_bk=1,            # B transposed stride for K (original col)
        stride_bn=N,            # B transposed stride for N (original row)

        BLOCK_N=64,

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,H

    # multiply out_gate after gating (out_gate is shape B,N,N,H)
    out_norm = out_norm * out_gate

    # final projection
    out_final = torch.nn.functional.linear(out_norm, weights['to_out.weight'])  # shape B,N,N,dim

    return out_final

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # a: (BM, BK), b: (BK, BN) => (BM, BN)

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-enhanced forward for the outgoing TriMul operator.
    1. Apply LayerNorm to the input.
    2. Compute linear projections for left/right and their gates.
    3. Fuse mask and gate via element‑wise multiplication.
    4. Compute the N³ sum with a custom batched GEMM kernel that treats each
       (batch, hidden) pair as an independent GEMM.
    5. Apply a second LayerNorm, out‑gate, and final linear projection.
    """
    # unpack the tuple
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # --- 1. LayerNorm over the feature dim ---
    # torch.nn.functional.layer_norm expects normalized shape and optional weight/bias
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # --- 2. Linear projections and gates (no bias) ---
    # Projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    # --- 3. Apply mask (if present) and gates ---
    # mask may be bool or float; convert to same dtype as projections
    if mask is not None and mask.numel() > 0:
        # expand mask to last dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
        left = left_proj * mask_exp * left_gate
        right = right_proj * mask_exp * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
    # Cast to lower precision for the heavy GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # --- 4. Core N³ contraction via batched GEMM kernel ---
    # Rearrange to (B*H, N, N) where H = hidden_dim
    # Permute to bring hidden dim to batch axis, then flatten batches
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # Output buffer (float32 accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # Strides (already contiguous)
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)   # row stride (M)
    stride_ak = left_mat.stride(2)   # col stride (K)
    stride_b_batch = right_mat.stride(0)
    # For B we treat the K dimension as the original K (=col) and N as rows
    stride_bk = right_mat.stride(2)  # original K stride = 1
    stride_bn = right_mat.stride(1)  # original N stride = N
    # Output strides
    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)    # row stride (M)
    stride_cn = out_mat.stride(2)    # col stride (N)

    # Tile sizes – tuned for H100 FP16 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                 # M
        N,                 # N
        N,                 # K
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # --- 5. Second LayerNorm over hidden dim ---
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out gate
    out = out * out_gate
    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
custom_kernel: Triton‑accelerated forward pass for the outgoing TriMul operator.

Algorithm
---------
1. Apply LayerNorm over the last (feature) dimension.
2. Compute three linear projections without bias:
   * left_proj, right_proj   – shape [B, N, N, H]
   * left_gate, right_gate, out_gate – same shape, followed by sigmoid.
3. Fuse the (optional) mask and the gate values with element‑wise mul:
   left  = left_proj  * left_gate  * mask
   right = right_proj * right_gate * mask
   (mask is broadcasted on the hidden dimension.)
4. Perform the core cubic contraction
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   via a custom Triton batched GEMM kernel.  Each (b,h) pair is treated as an
   independent GEMM of size N×N, so the kernel receives a batch size of B*H.
5. Layer‑norm the result over the hidden dimension, multiply by the out‑gate,
   and finish with a linear projection back to the original feature dimension.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel (A @ Bᵀ) used by TriMul.

    Arguments
    ---------
    a_ptr, b_ptr, c_ptr : pointers to the input / output tensors.
    M, N, K             : matrix dimensions.
    stride_*            : runtime strides for the tensors.
    BLOCK_*             : compile‑time tile sizes.
    """
    batch_id = tl.program_id(0)          # batch = B * hidden_dim
    pid_m    = tl.program_id(1)          # tile row index
    pid_n    = tl.program_id(2)          # tile col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load B tile as transposed (K × N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # a:(BM,BK), b:(BK,BN) → (BM,BN)

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] containing the module output.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # 1️⃣  LayerNorm over the feature dimension
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 2️⃣  Linear projections (no bias) and gates
    # --------------------------------------------------------------------- #
    # projections
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # gates – followed by sigmoid
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------------- #
    # 3️⃣  Apply mask (if present) and fuse gates
    # --------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        # mask shape [B, N, N] → broadcast to hidden dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left_proj  * mask_exp * left_gate
        right = right_proj * mask_exp * right_gate
    else:
        left  = left_proj  * left_gate
        right = right_proj * right_gate

    # Cast to FP16 for the heavy contraction
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 4️⃣  Core N³ contraction via a custom batched GEMM kernel
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape               # H == hidden_dim
    # bring hidden dim into the batch axis for a simple (B*H, N, N) layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * H, N, N)      # (BH, N, N)
    right_mat = right_perm.view(B * H, N, N)     # (BH, N, N)

    # output accumulator (float32)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # strides for the contiguous tensors
    stride_a_batch = left_mat.stride(0)    # BH stride
    stride_am = left_mat.stride(1)         # row stride (M)
    stride_ak = left_mat.stride(2)         # col stride (K)

    stride_b_batch = right_mat.stride(0)   # BH stride
    # treat B as transposed: K‑dimension stride = original col‑stride (1)
    stride_bk = right_mat.stride(2)        # =1
    # N‑dimension stride = original row‑stride (=N)
    stride_bn = right_mat.stride(1)        # =N

    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)          # row stride (M)
    stride_cn = out_mat.stride(2)          # col stride (N)

    # Tile sizes – tuned for H100 FP16 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5️⃣  Second LayerNorm, out‑gate, and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
