    # config
    dim = config["dim"]; hidden_dim = config["hidden_dim"]

    # Layernorm: using torch.nn.functional.layer_norm
    # epsilon maybe 1e-5, e.g., default.

    norm_bias = weights["norm.bias"]

    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B,N,N,dim]

    # Compute left_proj and right_proj
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # shape (hidden_dim, dim). Use dense linear: left = torch.einsum('...d,hd->...h', x, left_proj_weight) or torch.nn.functional.linear(x, left_proj_weight)
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if mask is not None:
        # mask shape [B,N,N]; unsqueeze -1
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))  # shape [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute out via our Triton kernel

    # Get pointers (contiguous). Ensure contiguous layout for left,right.
    left = left.contiguous()
    right = right.contiguous()
    # allocate output

    # Launch kernel
    # Compute stride values
    stride_left = left.stride()  # (stride_B, stride_i, stride_k, stride_h)

    stride_left_b, stride_left_i, stride_left_k, stride_left_h = stride_left
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()

        stride_right_b=stride_right_b,

        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,

import torch, triton, triton.language as tl

@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # compute batch and head
    batch = pid0 // H
    head = pid0 % H

    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # mask for out of bound
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M,K)
        left_ptrs = left_ptr + batch * stride_left_b + head * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (K,N) - note that we need k as rows, j as columns
        right_ptrs = right_ptr + batch * stride_right_b + head * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)  # shape (M,N)

    # store output
    out_ptrs = out_ptr + batch * stride_out_b + head * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm over the last dimension (dim).
2. Linear projections to a hidden dimension (`hidden_dim`) for the
   left and right tensors.
3. Optional mask applied to the projected tensors.
4. Gating networks (sigmoid‑activated linear layers) applied to the
   projected tensors.
5. Core triangular multiplication:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden channel and is
   implemented in a custom Triton kernel to fuse the loads, the
   inner‑product reduction and the stores.
6. Layer‑norm over the hidden dimension, multiplication by an output
   gate and a final linear projection back to `dim`.

Only the heavy O(N³·hidden_dim) contraction (step 5) is written in
Triton; everything else uses optimized PyTorch primitives.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes the einsum
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,          # *ptr, left tensor   [B, N, N, H]
    right_ptr,         # *ptr, right tensor  [B, N, N, H]
    out_ptr,           # *ptr, output tensor [B, N, N, H]
    B, N, H,           # problem size
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)   # combined batch * head dimension
    pid1 = tl.program_id(1)   # i‑tile index
    pid2 = tl.program_id(2)   # j‑tile index

    # Decode batch and head (hidden) from pid0
    batch = pid0 // H
    head  = pid0 %  H

    # Tile start offsets
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # Offsets within the tile (1‑D vectors)
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the (M×N) block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in tiles
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K)   <-- left[b, i, k, h]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + head  * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # Load right tile: shape (BLOCK_K, BLOCK_N)   <-- right[b, k, j, h]
        # (notice the transpose w.r.t. the original layout)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + head  * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i   # stride_right_i == stride over the second N axis
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # Accumulate dot‑product: (M×K) • (K×N) -> (M×N)
        # ------------------------------------------------------------------
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Store the result block
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + head  * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask        : torch.Tensor [B, N, N] or None
        - weights     : dict of torch.Tensors (see docstring above)
        - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    torch.backends.cuda.matmul.allow_tf32 = False  # deterministic FP32 matmuls

    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (over the last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid‑activated linear layers)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core triangular multiplication via Triton
    # ------------------------------------------------------------------
    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    batch, N_seq, _, _ = left.shape
    B = batch
    N = N_seq
    H = hidden_dim

    # Allocate output buffer
    out = torch.empty_like(left)  # [B, N, N, hidden_dim]

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tunable tile sizes – balanced for typical H100 workloads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions: (batch*head, i‑tiles, j‑tiles)
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch the kernel
    tri_mul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,   # the stride for the second N axis (j dimension)
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
