@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Ab, stride_Ak, stride_Ac,
                        stride_Bk, stride_Bn, stride_Bc,
                        stride_Cc, stride_Cn,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer offsets base for this batch
    a_ptrs = A_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Ak + tl.arange(0, BLOCK_K)[None, :] * stride_Ac
    b_ptrs = B_ptr + batch_id * stride_Bc + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A block (M block, K block)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)
        # load B block (K block, N block)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Bk
    # store result
    c_ptrs = C_ptr + batch_id * stride_Cc + offs_m[:, None] * stride_Cc + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16) if C_ptr.dtype == tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # get e.g. eps for layernorm
    eps = 1e-6

    # LN on x
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask if present
    if mask is not None:
        mask_u = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_u
        right = right * mask_u

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Reshape for batch matmul
    B, N, _, _ = left.shape
    batch = B * hidden_dim

    # Permute and view
    left_mat = left.permute(0,3,1,2).contiguous().view(batch, N, N)   # shape [B*H, M, K]
    right_mat = right.permute(0,3,2,1).contiguous().view(batch, N, N)  # shape [B*H, K, N] (transposed)

    # Convert to half for speed (optional)
    left_mat_h = left_mat.to(torch.float16)
    right_mat_h = right_mat.to(torch.float16)

    # Allocate output
    out_mat_h = torch.empty_like(left_mat_h)  # same shape [batch, N, N] dtype half

    # Define kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_mat_h, right_mat_h, out_mat_h,
        N, N, N,
        left_mat_h.stride(0), left_mat_h.stride(1), left_mat_h.stride(2),
        right_mat_h.stride(0), right_mat_h.stride(1), right_mat_h.stride(2),
        out_mat_h.stride(0), out_mat_h.stride(1), out_mat_h.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Cast back to float32
    out_mat = out_mat_h.to(torch.float32)

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final
```

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the Triangle Multiplicative Update (TriMul) operator.
    The heavy ``... i k d, ... j k d`` einsum is realised as a batch of
    N×N matrix multiplications (one per hidden channel) using a custom
    Triton GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # -------------------------------------------------
    # Linear projections
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # Optional masking (mask shape: [B, N, N])
    # -------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # -------------------------------------------------
    # Gating
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Batch GEMM (the O(N³) part) via Triton
    # -------------------------------------------------
    B, N, _, _ = left.shape   # left.shape == (B,N,N,hidden_dim)
    batch = B * hidden_dim

    # Rearrange tensors so that the hidden dimension becomes the batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)   # (batch, M, K)
    # right needs to be transposed on the two N axes before flattening
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(batch, N, N)  # (batch, K, N)

    # Cast to fp16 for the kernel (weights are still fp32)
    left_mat_h = left_mat.to(torch.float16)
    right_mat_h = right_mat.to(torch.float16)

    out_mat = torch.empty(batch, N, N, dtype=torch.float32, device=left.device)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_mat_h, right_mat_h, out_mat,
        N, N, N,
        left_mat_h.stride(0), left_mat_h.stride(1), left_mat_h.stride(2),
        right_mat_h.stride(0), right_mat_h.stride(1), right_mat_h.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # Output LayerNorm, gating and final projection
    # -------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate

    # Final linear to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
custom_kernel
==============

Forward pass of the Triangle Multiplicative Update (TriMul) used in AlphaFold‑3
and related models.  The heavy O(N³) einsum

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is rewritten as a batch of N×N matrix multiplications (one per hidden
channel) and executed with a custom Triton GEMM kernel.  All other
operations (LayerNorms, linear projections, sigmoidal gates) are carried
out with regular PyTorch for simplicity.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024,
input dimension `dim` (≤384) and hidden dimension `hidden_dim` (commonly 128).
Both masked and un‑masked inputs are supported.  The final result is a
float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batch GEMM:  C[b, m, n] = Σₖ A[b, m, k] * B[b, k, n]

    Parameters
    ----------
    A, B, C : pointers to the three tensors (float16 for A/B, float32 for C)
    M, N, K : integer dimensions
    stride_* : strides of the tensors (in element units)
    BLOCK_* : compile‑time tiling sizes
    """
    batch_id = tl.program_id(0)          # which (B·hidden_dim) matrix
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block column

    # Base pointers for the current batch
    a_batch = A + batch_id * stride_Ab
    b_batch = B + batch_id * stride_Bb
    c_batch = C + batch_id * stride_Cb

    # Offsets inside the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 (higher precision than the fp16 inputs)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A and B
        a_ptrs = a_batch + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 values
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16 values

        # Accumulate the dot product; use fp32 accumulator
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write the result back to C
    c_ptrs = c_batch + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the `TriMul` module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (see skeleton)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on linear maps of the *pre‑norm* tensor)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batch GEMM (the O(N³) part) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape               # left.shape == (B, N, N, hidden_dim)
    batch = B * hidden_dim                 # each hidden channel is an independent GEMM

    # Move hidden dimension to the batch axis and reshape to 2‑D matrices
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)   # (batch, M, K)
    # Right must be transposed on the two N‑axes before flattening
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(batch, N, N) # (batch, K, N)

    # Cast to fp16 for the kernel (still keep high‑precision accumulation)
    left_mat_h = left_mat.to(torch.float16)
    right_mat_h = right_mat.to(torch.float16)

    # Output buffer (float32 so that the final result needs no further casting)
    out_mat = torch.empty(batch, N, N, dtype=torch.float32, device=left.device)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_mat_h,
        right_mat_h,
        out_mat,
        N, N, N,
        left_mat_h.stride(0), left_mat_h.stride(1), left_mat_h.stride(2),
        right_mat_h.stride(0), right_mat_h.stride(1), right_mat_h.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original four‑dimensional layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
