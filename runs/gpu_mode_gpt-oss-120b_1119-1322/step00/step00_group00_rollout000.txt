def custom_kernel(data):
    x, mask, weights, config = data  # x: [bs, N, N, dim], mask: maybe shape, could be None if config "nomask".
    B,N,N_,dim = x.shape
    assert N == N_
    hidden_dim = config['hidden_dim']

    # LayerNorm for dim (with weight & bias)
    # Compute mean and variance over last dim
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    # x_norm = (x - mean) * rsqrt(var + eps) * norm_weight + norm_bias
    # Use torch.nn.functional.layer_norm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (bs,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if not nomask (mask None or config['nomask']==True)
    if not config.get('nomask', True):
        # mask shape (bs,N,N)
        # Convert mask to float and unsqueeze -1
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # broadcast over hidden_dim
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output
    out_fp16 = torch.empty_like(left_fp16)  # shape same as left

    # Get strides
    # left.shape: (B,N,N,H)
    stride_b, stride_i, stride_j, stride_h = left_fp16.stride()
    # but we need stride_i real and stride_k (second N) = stride_j? Actually we treat k dimension as second N after i dimension.
    # In indexing for left[i,k] we used stride_i for i offset and stride_j for k offset (H-case). But left shape (B,N,N,H)
    # left[i,k] corresponds to dims (B, i, k, h) -> stride_i = stride[1], stride_k = stride[2].
    stride_i = left_fp16.stride(1)  # B stride
    stride_k = left_fp16.stride(2)  # k stride
    stride_b = left_fp16.stride(0)

    # We don't need stride_h because we include hidden index via pointer offset hid (since stride_h=1).

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * hidden_dim)
    triton_trimul_matmul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

import torch
import triton
import triton.language as tl

# Triton kernel for batched per-head matrix multiplication
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_hb = tl.program_id(2)  # batch * hidden id

    # derive batch and hidden indices
    b = pid_hb // H
    h = pid_hb % H

    # offsets within N
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over reduction dim K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for the current tile
        a_ptrs = left_ptr + b * stride_b + h \
                 + offs_i[:, None] * stride_i \
                 + offs_k[None, :] * stride_k
        b_ptrs = right_ptr + b * stride_b + h \
                 + offs_j[:, None] * stride_i \
                 + offs_k[None, :] * stride_k

        # Load with mask
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write out
    c_ptrs = out_ptr + b * stride_b + h \
             + offs_i[:, None] * stride_i \
             + offs_j[None, :] * stride_k
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_ij)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    The heavy part – the einsum `out[i,j] = Σ_k left[i,k] * right[j,k]`
    – is performed by a custom Triton kernel that treats the hidden
    dimension as an independent batch dimension and computes a batched
    matrix‑multiply for each head.

    The surrounding LayerNorms, linear projections and gates are
    carried out with PyTorch because they are cheap compared to the
    O(N³·C) work.
    """
    # unpack inputs
    x, mask, weights, config = data
    # Expected shapes
    # x: [B, N, N, dim]
    B, N, N2, dim = x.shape
    assert N == N2, "pairwise dimensions must match"
    hidden_dim = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension ``dim``
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections to the hidden dimension
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating as in the reference)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid(linear) per hidden dim) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_norm, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy O(N³·C) tensor contraction – Triton kernel
    # ------------------------------------------------------------------
    # Cast to fp16 for speed; Triton kernel accumulates in fp32.
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output buffer (same dtype and shape as left/right)
    out_fp16 = torch.empty_like(left_fp16)

    # Strides for the [B, N, N, H] layout (contiguous)
    stride_b = left_fp16.stride(0)
    stride_i = left_fp16.stride(1)   # stride for the first N dimension (i)
    stride_k = left_fp16.stride(2)   # stride for the second N dimension (k)

    # Tunable block sizes – chosen to give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Synchronize to make sure the kernel finished before we use the result.
    torch.cuda.synchronize()

    # Cast back to fp32 for the rest of the pipeline.
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass accelerated with Triton.

The expensive part of the module is the Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]      (d = hidden_dim)
which is a batch of hidden‑dimension‑wise matrix multiplications:
    out_d = left_d @ right_d.T   for each hidden slice d.

The kernel below treats the hidden dimension as an extra batch dimension
and launches a standard tiled GEMM for each (batch, hidden) pair.
All surrounding LayerNorms, linear projections and sigmoidal gates are
performed with PyTorch – they are cheap compared to the O(N³·C) contraction.

The kernel works on half‑precision tensors (fp16) and accumulates in fp32
to keep numerical quality.  The final tensor is cast back to fp32 before
the output LayerNorm and the final linear projection.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,                # const *fp16   [B, N, N, H]
    right_ptr,               # const *fp16   [B, N, N, H]
    out_ptr,                 # *fp16         [B, N, N, H]
    B, N, H,                 # int32 scalar
    stride_b, stride_i, stride_k,   # int64 strides of the (B,N,N,H) layout
    BLOCK_M: tl.constexpr,  # tile size for the output i dimension
    BLOCK_N: tl.constexpr,  # tile size for the output j dimension
    BLOCK_K: tl.constexpr,  # tile size for the reduction k dimension
):
    """
    Compute, for a single hidden slice `h` and batch `b`:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The kernel is tiled in i (BLOCK_M) and j (BLOCK_N) and
    reduces over k in steps of BLOCK_K.  Hidden slices are encoded
    in program_id(2) together with the batch index.
    """

    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)
    pid_hb = tl.program_id(2) # batch * hidden index

    # ----------------------------------------------------------------------
    # Decode batch index and hidden index from pid_hb
    # ----------------------------------------------------------------------
    b = pid_hb // H          # batch id 0 .. B-1
    h = pid_hb % H           # hidden slice id 0 .. H-1

    # ----------------------------------------------------------------------
    # Offsets for the current tile
    # ----------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (higher precision for the sum over k)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ----------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the current k‑tile of left and right.
        # left: index order (b, i, k, h)
        a_ptrs = left_ptr + b * stride_b + h \
                 + offs_i[:, None] * stride_i \
                 + offs_k[None, :] * stride_k

        # right: index order (b, j, k, h) – note the first N dimension is `j`
        b_ptrs = right_ptr + b * stride_b + h \
                 + offs_j[:, None] * stride_i \
                 + offs_k[None, :] * stride_k

        # Load tiles; out‑of‑bounds entries are masked to zero
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Standard GEMM tile multiplication
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ----------------------------------------------------------------------
    # Write the result tile to global memory
    # ----------------------------------------------------------------------
    c_ptrs = out_ptr + b * stride_b + h \
             + offs_i[:, None] * stride_i \
             + offs_j[None, :] * stride_k
    mask_ij = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_ij)

    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy pairwise contraction – Triton kernel
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)   # same shape [B,N,N,H]

    # Strides of a contiguous (B,N,N,H) tensor
    stride_b = left_fp16.stride(0)   # B stride
    stride_i = left_fp16.stride(1)   # first N stride (i)
    stride_k = left_fp16.stride(2)   # second N stride (k)

    # Tunable tile sizes – chosen for good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles over i, tiles over j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on
    torch.cuda.synchronize()

    # Back to fp32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm and final gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final projection back to original ``dim``
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
