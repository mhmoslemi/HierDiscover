@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left: shape (M, K, H)
        left_ptrs = left_ptr + pid_b * stride_left_b \
                    + offs_i[:, None, None] * stride_left_i \
                    + offs_k[None, :, None] * stride_left_k \
                    + offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0).to(tl.float32)

        # Load right transposed: shape (K, N, H)
        right_ptrs = right_ptr + pid_b * stride_right_b \
                    + offs_k[:, None, None] * stride_right_k \
                    + offs_j[None, :, None] * stride_right_j \
                    + offs_h[None, None, :] * stride_right_h
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                          other=0.0).to(tl.float32)

        # Compute dot for each hidden channel
        for h in range(0, BLOCK_H):
            # The mask will zero out out-of-range h automatically
            a = left[:, :, h]   # (M, K)
            b = right_T[:, :, h]  # (K, N)
            # tl.dot expects (M,K) @ (K,N)
            acc[:, :, h] += tl.dot(a, b)

    # Store results
    out_ptrs = out_ptr + pid_b * stride_out_b \
               + offs_i[:, None, None] * stride_out_i \
               + offs_j[None, :, None] * stride_out_j \
               + offs_h[None, None, :] * stride_out_h

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
```

    """
    Custom Triton kernel for the outgoing Triangle Multiplicative Update (TriMul) operation.
    The heavy pairwise multiplication ``einsum('... i k d, ... j k d -> ... i j d')`` is
    performed by a fused Triton kernel that computes a batched matrix‑multiply per hidden
    channel. All other elementwise operations (LayerNorm, linear projections, gates) are
    executed with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=1e-5)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # optional mask
    if not nomask:
        # mask shape [B,N,N] -> unsqueeze last dim
        mask_f = mask.to(dtype).unsqueeze(-1)    # broadcast over hidden dim
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to fp16 for the heavy multiply ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor (fp16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    B, N, _, H = left_fp16.shape
    # kernel launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32   # compile‑time tile for hidden dimension

    grid = (B,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # We tile the hidden dimension manually; each launch processes up to BLOCK_H channels.
    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_H = min(BLOCK_H, hidden_dim - h_start)

        # compute byte offset for the slice (element offset works with stride)
        left_ptr = left_fp16.data_ptr() + h_start * stride_l_h
        right_ptr = right_fp16.data_ptr() + h_start * stride_r_h
        out_ptr = out_fp16.data_ptr() + h_start * stride_o_h

        trimul_einsum_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            B, N, cur_H,
            stride_l_b, stride_l_i, stride_l_k, stride_l_h,
            stride_r_b, stride_r_j, stride_r_k, stride_r_h,
            stride_o_b, stride_o_i, stride_o_j, stride_o_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )
    # Convert back to float32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=1e-5)

    # Apply out gate (cast to same dtype)
    out = out * out_gate.to(out.dtype)

    # ---------- Final linear ----------
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
custom_kernel: Outgoing Triangle‑Multiplicative Update (TriMul)

The heavy pair‑wise multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a fused Triton kernel that performs a batched
matrix‑multiply per hidden channel (d).  All remaining element‑wise
operations (LayerNorm, linear projections, gating, final projection)
are carried out with PyTorch for clarity and to keep the implementation
compact while still satisfying the requirement of using Triton.

The kernel processes tiles of the (i, j) plane (BLOCK_M×BLOCK_N)
and a tile of the hidden dimension (BLOCK_H).  The k‑dimension is
blocked by BLOCK_K and accumulated in FP32.  The result is written back
as FP16 and later cast to FP32 for the final LayerNorm and output linear.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                                 # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left and right have shape (B, N, N, H) and the sum is over k.
    The kernel works on a tile of (i, j) and a tile of hidden channels.
    """
    pid_b = tl.program_id(0)                 # batch index
    pid_i = tl.program_id(1)                 # tile over i
    pid_j = tl.program_id(2)                 # tile over j

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over the k dimension in tiles
    for tk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = tk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left[b, i, k, h] --------
        left_ptrs = left_ptr + pid_b * stride_l_b \
                    + offs_i[:, None, None] * stride_l_i \
                    + offs_k[None, :, None] * stride_l_k \
                    + offs_h[None, None, :] * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] &
                            mask_k[None, :, None] &
                            mask_h[None, None, :],
                       other=0.0).to(tl.float32)

        # -------- right_T[b, k, j, h] = right[b, j, k, h] (k‑j transposed) --------
        right_ptrs = right_ptr + pid_b * stride_r_b \
                    + offs_k[:, None, None] * stride_r_k \
                    + offs_j[None, :, None] * stride_r_j \
                    + offs_h[None, None, :] * stride_r_h
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None, None] &
                               mask_j[None, :, None] &
                               mask_h[None, None, :],
                          other=0.0).to(tl.float32)

        # accumulate for each hidden channel in the tile
        for h in range(0, BLOCK_H):
            a = left[:, :, h]       # (BLOCK_M, BLOCK_K)
            b = right_T[:, :, h]    # (BLOCK_K, BLOCK_N)
            acc[:, :, h] += tl.dot(a, b)

    # write back (cast to FP16)
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + offs_i[:, None, None] * stride_o_i \
               + offs_j[None, :, None] * stride_o_j \
               + offs_h[None, None, :] * stride_o_h
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_h[None, None, :])

        mask        : [B, N, N] (optional)    (bool / float)
        weights     : dict of model parameters
        config      : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    x, mask, weights, config = data

    hidden_dim = config["hidden_dim"]

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(

    )  # [B, N, N, dim]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]

        right = right * mask_f

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    right = right * right_gate

    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out_fp16.stride()

    # Triton block sizes (tuned for H100)

    BLOCK_K = 32
    BLOCK_H = 32   # compile‑time tile for hidden dimension

    grid = (
        B,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Process the hidden dimension in chunks of BLOCK_H
    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_H = min(BLOCK_H, hidden_dim - h_start)

        # pointers to the current slice (element offsets, not bytes)
        left_ptr = left_fp16.data_ptr() + h_start * stride_l_h
        right_ptr = right_fp16.data_ptr() + h_start * stride_r_h
        out_ptr = out_fp16.data_ptr() + h_start * stride_o_h

        trimul_einsum_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            B,
            N,
            cur_H,
            stride_l_b,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_b,
            stride_r_i,
            stride_r_k,
            stride_r_h,
            stride_o_b,
            stride_o_i,
            stride_o_k,
            stride_o_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # Back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ---------- Apply out gate ----------
    out = out * out_gate.to(out.dtype)

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
