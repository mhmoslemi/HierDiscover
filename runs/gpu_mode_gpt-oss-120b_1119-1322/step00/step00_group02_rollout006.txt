```
    B, N, H,

    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,

):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = tl.arange(0, VEC)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, VEC), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for k_idx in range(num_k):
        k_base = k_idx * BLOCK_K
        k = k_base + offs_k
        mask_k = k < N

        # load left (M,K,V)
        left_ptrs = left_ptr \
            + pid_b * stride_lb \
            + offs_i[:, None, None] * stride_li \
            + k[None, :, None] * stride_lk \
            + offs_d[None, None, :] * stride_ld
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # load right transposed (K,N,V) where K is rows, N is columns

            + offs_d[None, None, :] * stride_rd

        # Compute dot for each hidden subvector
        for d in range(VEC):
```

```python
def custom_kernel(data):
    """
    Fast implementation of the outgoing TriMul forward pass.

    Steps:
    1. Layer‑norm the input tensor.
    2. Compute left/right projections and their gates.
    3. Apply (optional) attention mask and gates.
    4. Convert the gated tensors to fp16 and run a fused Triton kernel that
       evaluates   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
       The kernel tiles the i and j dimensions (32×32), tiles the reduction
       over k (64), and processes the hidden dimension in groups of 8
       using `tl.dot`.  Accumulation is performed in fp32 for numerical
       stability.
    5. Apply a second LayerNorm, the output gate, and the final linear
       projection back to the original model dimension.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Apply gates and cast to fp16 for the heavy kernel
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton kernel for the einsum
    # ------------------------------------------------------------------
    def tri_mul_triton(left_fp16, right_fp16):
        B, N, _, H = left_fp16.shape
        out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_K = 64
        VEC = 8   # H must be divisible by VEC (128 -> 16 groups)

        grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

        tri_mul_kernel[grid](
            left_fp16,
            right_fp16,
            out_fp32,
            B, N, H,
            left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
            right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
            out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2), out_fp32.stride(3),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            VEC=VEC,
        )
        return out_fp32

    # ------------------------------------------------------------------
    # 7. Triton kernel definition
    # ------------------------------------------------------------------
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr, VEC: tl.constexpr
    ):
        pid_b = tl.program_id(2)
        pid_i = tl.program_id(0)
        pid_j = tl.program_id(1)

        # --------------------------------------------------------------
        # coordinate ranges
        # --------------------------------------------------------------
        offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)
        offs_d = tl.arange(0, VEC)

        mask_i = offs_i < N
        mask_j = offs_j < N

        # accumulator in fp32
        acc = tl.zeros((BLOCK_M, BLOCK_N, VEC), dtype=tl.float32)

        num_k = tl.cdiv(N, BLOCK_K)

        for k_tile in range(num_k):
            k_offset = k_tile * BLOCK_K + offs_k
            mask_k = k_offset < N

            # left: shape (M, K, V)
            left_ptrs = left_ptr \
                + pid_b * stride_lb \
                + offs_i[:, None, None] * stride_li \
                + k_offset[None, :, None] * stride_lk \
                + offs_d[None, None, :] * stride_ld
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None, None] & mask_k[None, :, None],
                           other=0.0)  # fp16

            # right (transposed K × N × V)
            right_ptrs = right_ptr \
                + pid_b * stride_rb \
                + k_offset[:, None, None] * stride_rk \
                + offs_j[None, :, None] * stride_rj \
                + offs_d[None, None, :] * stride_rd
            right = tl.load(right_ptrs,
                            mask=mask_k[:, None, None] & mask_j[None, :, None],
                            other=0.0)  # fp16

            # compute a small matrix‑multiply for each hidden sub‑vector
            for d in range(VEC):
                # (M, K) @ (K, N) -> (M, N)
                prod = tl.dot(left[:, :, d], right[:, :, d])  # fp16
                acc[:, :, d] += prod.to(tl.float32)

        # --------------------------------------------------------------
        # store result
        # --------------------------------------------------------------
        out_ptrs = out_ptr \
            + pid_b * stride_ob \
            + offs_i[:, None, None] * stride_oi \
            + offs_j[None, :, None] * stride_oj \
            + offs_d[None, None, :] * stride_od
        tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])

    # ------------------------------------------------------------------
    # 8. Run the kernel
    # ------------------------------------------------------------------
    out = tri_mul_triton(left, right)             # [B,N,N,hidden_dim] fp32

    # ------------------------------------------------------------------
    # 9. Second LayerNorm, output gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)            # -> [B,N,N,dim]

    return out
```

"""
TriMul (outgoing) – fast forward implementation.

The forward pass consists of:
1. Layer‑norm on the 4‑D input (B,N,N,dim).
2. Two bias‑free linear projections (left/right) and three gating
   projections (left_gate, right_gate, out_gate).  The gates are
   sigmoided and applied after an optional attention mask.
3. The heavy tensor contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is executed by a custom Triton kernel.  The kernel tiles the i‑ and
   j‑dimensions (32×32 threads per block), tiles the reduction over k
   (64 elements), and processes the hidden dimension in groups of 8
   (VEC).  Each (i,j) thread accumulates a VEC‑wide vector in fp32.
4. A second Layer‑norm, the output gate, and a final bias‑free linear
   projection back to the original dimension produce the module output.

The implementation keeps the intermediate tensors in fp16 to reduce
memory bandwidth while accumulating in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # fp16*  [B,N,N,H] (i,k,d)
    right_ptr,                    # fp16*  [B,N,N,H] (j,k,d) – we load transposed as (k,j,d)
    out_ptr,                      # fp32*  [B,N,N,H] (i,j,d)
    B, N, H,                      # runtime constants
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,        # tile size for i
    BLOCK_N: tl.constexpr,        # tile size for j
    BLOCK_K: tl.constexpr,        # tile size for k
    VEC: tl.constexpr,            # hidden dimension vectorisation factor
    H_CONST: tl.constexpr,        # hidden dimension (compile‑time)
):
    pid_b = tl.program_id(2)                # batch index
    pid_i = tl.program_id(0)                # block row index
    pid_j = tl.program_id(1)                # block col index

    # ------------------------------------------------------------------
    # Thread‑level coordinates
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # (M,)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # (N,)
    offs_k = tl.arange(0, BLOCK_K)                     # (K,)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Number of hidden blocks (H is a compile‑time constant)
    NUM_H_BLOCKS = H_CONST // VEC

    # ------------------------------------------------------------------
    # Loop over hidden blocks
    # ------------------------------------------------------------------
    for hb in range(NUM_H_BLOCKS):
        # hidden offsets for this block
        offs_d = tl.arange(0, VEC) + hb * VEC          # (V,)

        # accumulator in fp32 for this hidden block
        acc = tl.zeros((BLOCK_M, BLOCK_N, VEC), dtype=tl.float32)

        # ----------------------------------------------------------------
        # Reduction over k (tiled)
        # ----------------------------------------------------------------
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k_tiles):
            k_base = k_tile * BLOCK_K
            k = k_base + offs_k
            mask_k = k < N

            # ------------------------------------------------------------
            # Load a (M, K, V) tile from left  →  left[i,k,d]
            # ------------------------------------------------------------
            left_ptrs = left_ptr + \
                pid_b * stride_lb + \
                offs_i[:, None, None] * stride_li + \
                k[None, :, None] * stride_lk + \
                offs_d[None, None, :] * stride_ld
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None],
                other=0.0,
            )                     # fp16

            # ------------------------------------------------------------
            # Load a (K, N, V) tile from right → right[k,j,d]
            # (note the transposition: k is the leading dimension)
            # ------------------------------------------------------------
            right_ptrs = right_ptr + \
                pid_b * stride_rb + \
                k[:, None, None] * stride_rk + \
                offs_j[None, :, None] * stride_rj + \
                offs_d[None, None, :] * stride_rd
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None],
                other=0.0,
            )                     # fp16

            # ------------------------------------------------------------
            # Small GEMM for each hidden sub‑vector (size V)
            # ------------------------------------------------------------
            for d in range(VEC):
                # (M, K) @ (K, N) -> (M, N)
                prod = tl.dot(left[:, :, d], right[:, :, d])   # fp16 result
                acc[:, :, d] += prod.to(tl.float32)

        # ------------------------------------------------------------
        # Store the completed hidden block
        # ------------------------------------------------------------
        out_ptrs = out_ptr + \
            pid_b * stride_ob + \
            offs_i[:, None, None] * stride_oi + \
            offs_j[None, :, None] * stride_oj + \
            offs_d[None, None, :] * stride_od
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None],
        )


# ----------------------------------------------------------------------
# Helper that runs the above kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    left_fp16/right_fp16 : [B, N, N, H]  (float16)
    Returns:
        out : [B, N, N, H]  (float32)
    """
    B, N, _, H = left_fp16.shape
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=left_fp16.device)

    # Tuning parameters – they were chosen to respect the 1024‑thread limit
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    VEC = 8                     # H must be divisible by VEC (128 → 16 blocks)

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2), out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        VEC=VEC,
        H_CONST=H,                     # compile‑time hidden size
    )
    return out_fp32


# ----------------------------------------------------------------------
# Public entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton kernel.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (see key list in prompt)
        - config       : dict with at least 'dim', 'hidden_dim', optional 'nomask'
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack input
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-6  # LayerNorm epsilon (matches the reference)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # [hidden, dim]
    right_proj_weight = weights["right_proj.weight"]        # [hidden, dim]

    left = F.linear(x, left_proj_weight)    # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Apply gates and cast to fp16 (ready for the heavy kernel)
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton kernel – the O(N³) contraction
    # ------------------------------------------------------------------
    out = tri_mul_triton(left, right)          # [B,N,N,hidden] float32

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                      # broadcasted element‑wise

    to_out_weight = weights["to_out.weight"]   # [dim, hidden]
    out = F.linear(out, to_out_weight)        # back to [B,N,N,dim]

    return out
