def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if needed
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # B,N,N,1
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Move hidden dim to leading axis for matmul
    B, N, _, _ = left.shape
    left_h = left.permute(0, 3, 1, 2).contiguous()
    right_h = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*hidden dimension
    outer = B * hidden_dim

    right_flat = right_h.view(outer, N, N)

    left_flat_fp16 = left_flat.to(torch.float16)
    right_flat_fp16 = right_flat.to(torch.float16)

    # Allocate output
    out_flat_fp16 = torch.empty_like(left_flat_fp16)

    # Call Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

        a_ptr=left_flat_fp16,
        b_ptr=right_flat_fp16,
        c_ptr=out_flat_fp16,
        M=N,
        N=N,
        K=N,
        stride_a_outer=left_flat_fp16.stride(0),
        stride_am=left_flat_fp16.stride(1),
        stride_ak=left_flat_fp16.stride(2),
        stride_b_outer=right_flat_fp16.stride(0),
        stride_bk=right_flat_fp16.stride(2),  # transposed K stride
        stride_bn=right_flat_fp16.stride(1),  # transposed N stride
        stride_c_outer=out_flat_fp16.stride(0),
        stride_cm=out_flat_fp16.stride(1),
        stride_cn=out_flat_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Wait for kernel completion
    torch.cuda.synchronize()

    # Convert back to float32 and reshape
    out_h = out_flat_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(out_h, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out gate
    out_normed = out_normed * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_normed, to_out_weight)

    return output

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Project x to a hidden dimension H with two bias‑free linear layers:
       left  = x @ W_left   (shape B,N,N,H)
       right = x @ W_right  (shape B,N,N,H)
3. (Optional) Apply the pairwise mask M ∈ {0,1}^{B×N×N}.
4. Compute three sigmoid gates from x and multiply them element‑wise:
       left  = left  * sigmoid(x @ W_left_gate)
       right = right * sigmoid(x @ W_right_gate)
   also compute out_gate = sigmoid(x @ W_out_gate).
5. Rearrange to (B,H,N,N) and flatten the outer dimensions:
       L = left.permute(0,3,1,2).contiguous()  -> (B,H,N,N)
       R = right.permute(0,3,1,2).contiguous() -> (B,H,N,N)
       Lf = L.view(B*H, N, N)   # (outer, M, K)
       Rf = R.view(B*H, N, N)   # (outer, J, K)
6. Compute the contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
       out_d = Lf_d @ Rf_dᵀ   for each hidden slice d.
   The heavy O(N³·H) work is performed by a custom Triton kernel that
   carries out the batched mat‑mul for all (B·H) matrices in parallel,
   using FP16 loads and FP32 accumulation.
7. Reshape the result back to (B,N,N,H), apply a second LayerNorm,
   multiply by out_gate, and project back to the original dimension C.
8. Return the final tensor of shape (B,N,N,C) (float32).

The implementation only requires a single Triton kernel for the
core N³ contraction; all other ops are done with PyTorch (which is
negligible compared to the N³ work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_matmul_outgoing(
    a_ptr, b_ptr, c_ptr,
    # matrix sizes
    M, N, K,
    # strides (outer, row, col)
    stride_a_outer, stride_am, stride_ak,
    stride_b_outer, stride_bk, stride_bn,
    stride_c_outer, stride_cm, stride_cn,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ
    A : (outer, M, K)      (FP16)
    B : (outer, N, K)      (FP16)  (interpreted as transposed)
    C : (outer, M, N)      (FP16)
    The kernel processes one (outer) batch element per program,
    tiled by (BLOCK_M, BLOCK_N) with a reduction over K in tiles of BLOCK_K.
    """
    pid_outer = tl.program_id(0)   # outer dimension = B * hidden_dim
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # Compute absolute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the block in A and B
    a_ptrs = a_ptr + pid_outer * stride_a_outer \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_outer * stride_b_outer \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator (float32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute current K‑offset
        k_offset = k_iter * BLOCK_K
        # Bounds checks for the K dimension
        cur_k_valid = (offs_k + k_offset) < K

        # Load A and B tiles (FP16) with masks
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & cur_k_valid[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=cur_k_valid[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate dot product (inner product over K)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back C (FP16) with masking for partial tiles
    c_ptrs = c_ptr + pid_outer * stride_c_outer \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict of torch.Tensor model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        output: torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B, N, N, H)
    right = F.linear(x, right_proj_weight)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. (Optional) mask
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape is (B, N, N); broadcast to (B, N, N, 1)
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()   # (B,N,N,H)
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the N³ contraction
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dimension to the second axis for easy flattening
    left_h = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    outer = B * hidden_dim
    left_flat = left_h.view(outer, N, N)   # (outer, M=N, K=N)
    right_flat = right_h.view(outer, N, N)  # (outer, J=N, K=N)

    # Cast to FP16 for the heavy computation (still accurate enough)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output buffer
    out_fp16 = torch.empty_like(left_fp16)  # (outer, N, N), FP16

    # ------------------------------------------------------------------
    # 6. Triton batched mat‑mul: out = left @ rightᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        outer,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _triton_matmul_outgoing[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_fp16,
        M=N,
        N=N,
        K=N,
        stride_a_outer=left_fp16.stride(0),
        stride_am=left_fp16.stride(1),
        stride_ak=left_fp16.stride(2),
        stride_b_outer=right_fp16.stride(0),
        # B is transposed inside the kernel (K ↔ N)
        stride_bk=right_fp16.stride(2),  # original K‑stride becomes row‑stride
        stride_bn=right_fp16.stride(1),  # original N‑stride becomes column‑stride
        stride_c_outer=out_fp16.stride(0),
        stride_cm=out_fp16.stride(1),
        stride_cn=out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on
    torch.cuda.synchronize()

    # Convert back to FP32 and restore original layout
    out_h = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = F.layer_norm(
        out_h,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # (B,N,N,H)

    # Apply the output gate
    out_normed = out_normed * out_gate

    # Final linear back to dim
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_normed, to_out_weight)   # (B,N,N,C)

    return output
