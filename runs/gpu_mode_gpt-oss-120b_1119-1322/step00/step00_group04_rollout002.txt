def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # unpack dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    epsilon = 1e-5

    # Device
    device = input_tensor.device

    # LayerNorm on x
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean and var
    # using torch functionality: F.layer_norm? Use torch.nn.functional.layer_norm which supports weight and bias.
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=epsilon,
    )

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]

    # Compute proj using torch.nn.functional.linear
    # shape: (B,N,N,hidden_dim)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # linear expects input shape (..., dim) output shape (..., hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if not nomask (mask may be None)
    # mask shape: [B,N,N], can be bool/int
    if mask is not None:
        mask = mask.unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # Compute gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates to left and right
    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Reshape for batched matmul
    B, N, _, H = left_fp16.shape
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # (B*H, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # (B*H, N, N)

    # Compute tri-mul via kernel
    out_flat_fp16 = triangular_mul_kernel_launch(left_flat, right_flat)  # shape (B*H,N,N)
    # reshape back
    out_hidden_fp16 = out_flat_fp16.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # Cast back to float32
    out_hidden = out_hidden_fp16.to(torch.float32)

    # to_out_norm: LayerNorm over hidden dimension H
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden_norm = torch.nn.functional.layer_norm(
        out_hidden,

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_batch_a, stride_am, stride_ak,
    stride_batch_b, stride_bn, stride_bk,
    stride_batch_c, stride_cm, stride_cn,
    # const block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute number of blocks
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # total per batch
    num_blocks_batch = num_blocks_m * num_blocks_n

    batch = pid // num_blocks_batch
    pid_in_batch = pid % num_blocks_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_batch_ptr = a_ptr + batch * stride_batch_a
    b_batch_ptr = b_ptr + batch * stride_batch_b
    c_batch_ptr = c_ptr + batch * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B: shape (BLOCK_N, BLOCK_K)
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bn) + (offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # multiply-accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton implementation of AlphaFold3 outgoing TriMul forward pass.
    ...
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    epsilon = 1e-5

    # Input shape
    batch, N, _, _ = input_tensor.shape    # [B,N,N,dim]

    # LayerNorm on last dim
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=epsilon
    )
    
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask
    if not config.get("nomask", False):
        # mask shape [B,N,N]; expand to last dim
        mask = mask.unsqueeze(-1).to(left.dtype)   # match dtype
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for the heavy multiplication
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Reshape to (B*hidden, N, N)
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # Triton kernel: compute C = A @ B^T (sum over K)
    out_flat_fp16 = _tri_mul_fp16(left_flat, right_flat)

    # Restore original shape [B, N, N, hidden]
    out_hidden_fp16 = out_flat_fp16.view(batch, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # Convert back to fp32 for the remaining layers
    out_hidden = out_hidden_fp16.to(torch.float32)

    # to_out_norm (LayerNorm over hidden)
    out_hidden_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=epsilon
    )

    # Multiply by out_gate
    out_hidden_norm = out_hidden_norm * out_gate

    # Final projection back to dim
    out = torch.nn.functional.linear(out_hidden_norm, weights["to_out.weight"])

    return out

TriMul (outgoing) – Triton‑accelerated forward pass

This implementation follows the reference PyTorch TriMul module used in AlphaFold‑3.
The expensive “tri‑multiplicative” part

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed with a custom Triton kernel that fuses the three‑dimensional
matrix multiplication across the hidden dimension (C_hidden).  All linear
projections, LayerNorms and sigmoid gates are carried out in PyTorch (float32)
and the large inner product is executed in float16 with FP32 accumulation
(TF32/Tensor‑core friendly).  The kernel computes a batched GEMM
C = A @ Bᵀ for each hidden channel, re‑shaping the tensors so that the hidden
dimension is folded into the batch dimension.

The overall algorithm:
1. Layer‑norm the input over the feature axis (dim).
2. Linear projections → left, right (no bias).
3. Optional mask (broadcast on the hidden axis) applied to the projections.
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the same
   normalized input.
5. Apply left/right gates element‑wise.
6. Cast left/right to float16, fold the hidden dimension into the batch
   dimension and invoke the Triton kernel to obtain the tri‑multiplicative term
   of shape (B, N, N, hidden).
7. Cast the result back to float32, apply a second LayerNorm, multiply by
   out_gate and finally project back to the original dimensionality.

The Triton kernel is a standard tiled GEMM (BLOCK_M × BLOCK_N × BLOCK_K)
with FP16 inputs and FP32 accumulation, suitable for H100 Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (sum over K)
# A : (batch, M, K)   – float16
# B : (batch, N, K)   – float16
# C : (batch, M, N)   – float16
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                         # problem size
    stride_batch_a, stride_am, stride_ak,   # A strides (batch, M, K)
    stride_batch_b, stride_bn, stride_bk,   # B strides (batch, N, K)
    stride_batch_c, stride_cm, stride_cn,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) Compute batch, block‑row (m) and block‑col (n) indices
    # ------------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # ------------------------------------------------------------------
    # 2) Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_ptr = a_ptr + batch * stride_batch_a
    b_batch_ptr = b_ptr + batch * stride_batch_b
    c_batch_ptr = c_ptr + batch * stride_batch_c

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr \
               + (offs_m[:, None] * stride_am) \
               + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile: (BLOCK_N, BLOCK_K)  (note the “row” dimension is N)
        b_ptrs = b_batch_ptr \
               + (offs_n[:, None] * stride_bn) \
               + (offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4) Write the result tile
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr \
           + (offs_m[:, None] * stride_cm) \
           + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel; works on float16 tensors.
# ----------------------------------------------------------------------
def _tri_mul_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (B, N, N) using the Triton kernel.
    A, B – float16, same batch size, same N.
    Returns C of the same shape and dtype.
    """
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K  # square matrices per hidden channel

    C = torch.empty((batch, N, N), dtype=A.dtype, device=A.device)

    # Strides are given in element counts (not bytes)
    stride_batch_a, stride_am, stride_ak = A.stride()
    stride_batch_b, stride_bn, stride_bk = B.stride()
    stride_batch_c, stride_cm, stride_cn = C.stride()

    # Tunable block sizes – 64 works well on H100 for FP16/TensorCore GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = tl.cdiv(N, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    total_blocks = batch * grid_m * grid_n
    grid = (total_blocks,)

    tri_mul_kernel[grid](
        a_ptr=A,
        b_ptr=B,
        c_ptr=C,
        M=N,
        N=N,
        K=K,
        stride_batch_a=stride_batch_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_batch_b=stride_batch_b,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_batch_c=stride_batch_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough for FP16 TensorCore work‑group
    )
    return C

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (float32 or bool, optional)
        - weights      : dict of torch.Tensor with all model parameters
        - config       : dict containing `dim`, `hidden_dim`, `nomask`

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) First LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)       # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to float16 and fold hidden dim into the batch dimension
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape   # H == hidden_dim
    # reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # ------------------------------------------------------------------
    # 7) Tri‑multiplicative product via Triton (C = A @ Bᵀ)
    # ------------------------------------------------------------------
    out_flat_fp16 = _tri_mul_fp16(left_flat, right_flat)   # [B*H, N, N]

    # ------------------------------------------------------------------
    # 8) Restore original layout: [B, N, N, hidden]
    # ------------------------------------------------------------------
    out_hidden_fp16 = out_flat_fp16.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # Back to float32 for the remaining (LayerNorm, projection)
    out_hidden = out_hidden_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 10) Multiply by the output gate
    # ------------------------------------------------------------------
    out_hidden = out_hidden * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out_hidden, weights["to_out.weight"])   # [B, N, N, dim]

    return out
