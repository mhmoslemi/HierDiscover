"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor `x` with the provided LN weights.
2. Compute four linear projections of `x` (left/right) and three gating
   projections (left_gate, right_gate, out_gate).  All linear layers are
   bias‑free, the gating layers are followed by a sigmoid.
3. Apply the (optional) pairwise mask to the left/right tensors and
   multiply by the corresponding gates.
4. Compute the core contraction
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   This is a batch of matrix‑multiplications:
        left(b,d)  : (N,N)   → (B*H, N, N)
        right_T(b,d): (N,N)   → (B*H, N, N)  (transpose K,N)
        out(b,d)   : (N,N)   ← left @ right_T
   The heavy O(N³·H) work is performed by a custom Triton GEMM kernel.
5. Reshape the result back to [B, N, N, H] and apply a second
   Layer‑norm (hidden dimension) followed by the output gate.
6. Final linear projection back to the original dimension `dim`.

Only the batched GEMM (step 4) is implemented in Triton; the surrounding
operations use PyTorch for simplicity and ease of handling the
layer‑norm/gate parameters.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton batched GEMM: C[batch, M, N] = A[batch, M, K] @ B[batch, K, N]
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    A : (batch, M, K)  row‑major
    B : (batch, K, N)  row‑major
    C : (batch, M, N)  row‑major
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds loads
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers with the correct batch offset
    a_ptrs = (
        A_ptr
        + pid_batch * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_ptrs = (
        B_ptr
        + pid_batch * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                 # (M,K)·(K,N) → (M,N)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [B, N, N] (may be ignored)
        - weights_dict : mapping str → torch.Tensor containing all model weights
        - config_dict  : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (dimension = dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (bias‑free) and gating logits
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])        # [B,N,N,hidden_dim]

    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction using Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # Collapse batch × hidden_dim into a single GEMM batch dimension
    left_batch = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, N, N)

    # Right‑hand side must be transposed on the K‑axis for the contraction
    right_T = right_batch.permute(0, 2, 1).contiguous()   # (B*H, K=N, N)

    # Allocate output
    out_batch = torch.empty_like(left_batch)           # (B*H, N, N)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        out_batch.shape[0],                                     # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    _batched_gemm_kernel[grid](
        left_batch,
        right_T,
        out_batch,
        N,                     # M
        N,                     # N
        N,                     # K
        left_batch.stride(0),  # stride_a_batch
        left_batch.stride(1),  # stride_a_m
        left_batch.stride(2),  # stride_a_k
        right_T.stride(0),     # stride_b_batch
        right_T.stride(1),     # stride_b_k (K dimension)
        right_T.stride(2),     # stride_b_n (N dimension)
        out_batch.stride(0),   # stride_c_batch
        out_batch.stride(1),   # stride_c_m
        out_batch.stride(2),   # stride_c_n
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Make sure the kernel has finished before we continue
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, hidden_dim]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Hidden‑dim layer‑norm, output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,hidden_dim]

    out_norm = out_norm * out_gate                        # gating
    # Final linear: hidden_dim → dim
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
```
