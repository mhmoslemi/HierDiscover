"""
TriMul “outgoing” kernel (AlphaFold3)

Given a 4‑D tensor X ∈ ℝ^{B×N×N×C}:
1. Layer‑norm over the last channel (C).
2. Two linear projections → left, right ∈ ℝ^{B×N×N×H}.
3. Optional pairwise mask is applied.
4. Gated element‑wise (sigmoid) multiplications.
5. Core contraction:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   i.e. a batched matrix multiplication L @ Rᵀ for every hidden channel h.
   This is implemented with a specialised Triton kernel that fuses the
   K‑loop and avoids materialising large intermediate tensors.
6. Layer‑norm + gating on the hidden dimension.
7. Final linear projection back to the original channel size.

The kernel processes one (batch, hidden) pair per program‑id (3‑D grid).
For each tile (M×N) of the output it loads (M×K) and (K×N) blocks of
the two inputs (half‑precision), performs a dot‑product accumulation in
float32, and writes the result in float32.  The rest of the pipeline
(LayerNorm, Linear, Sigmoid) is performed with standard PyTorch
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from triton import cdiv


@triton.jit
def triton_trimul_kernel(
    left_ptr,                     # *float16   [B,N,N,H]
    right_ptr,                    # *float16   [B,N,N,H]
    out_ptr,                      # *float32   [B,N,N,H]
    B, N, H,                      # scalar runtime args
    # strides for left tensor
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right tensor
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    # strides for output tensor
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)                  # block row
    pid_n = tl.program_id(1)                  # block column
    pid_z = tl.program_id(2)                  # combined batch * hidden index

    # decompose combined index into batch and hidden channel
    batch_idx = pid_z // H
    hidden_idx = pid_z % H

    # offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along the reduction (k) dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for tk in range(num_k_tiles):
        k_base = tk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (M x K) ------------------------------------
        left_ptrs = (
            left_ptr
            + batch_idx * stride_left_batch
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + hidden_idx * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # ---- load right tile (K x N) after implicit transpose ------------
        # right is stored as [batch, i, k, h]; we want (k, j) order.
        right_ptrs = (
            right_ptr
            + batch_idx * stride_right_batch
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
            + hidden_idx * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # ---- fused multiply‑add -----------------------------------------
        acc += tl.dot(left, right)   # dot yields float32 because acc is float32

    # ---- write the result ------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_idx * stride_out_batch
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + hidden_idx * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Entry point used by the evaluator.

    Args:
        data: tuple (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------ #
    # helper to fetch a weight (bias may be None)
    # ------------------------------------------------------------------ #
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------ #
    # 1. LayerNorm over the input channel dimension
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=1e-5,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias) → left/right  [B,N,N,hidden]
    # ------------------------------------------------------------------ #
    left = F.linear(x, w("left_proj.weight"))
    right = F.linear(x, w("right_proj.weight"))

    # ------------------------------------------------------------------ #
    # 3. Apply optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------ #
    if mask_tensor is None:
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)
    mask = mask.unsqueeze(-1)                       # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------ #
    # 4. Compute gates (sigmoid of linear layers)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    # ------------------------------------------------------------------ #
    # 5. Apply gates to the projected tensors
    # ------------------------------------------------------------------ #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 6. Cast to float16 for the heavy O(N³) contraction
    # ------------------------------------------------------------------ #
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Output buffer for the contraction (float32)
    out = torch.empty(
        (bs, seq_len, seq_len, hidden_dim),
        dtype=torch.float32,
        device=device,
    )

    # ------------------------------------------------------------------ #
    # 7. Launch Triton kernel that computes out_{b,i,j,h} = Σ_k left * right
    # ------------------------------------------------------------------ #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for the three tensors (in elements, not bytes)
    s_left = left_h.stride()    # (B, i, k, h)
    s_right = right_h.stride()
    s_out = out.stride()

    grid = (
        cdiv(seq_len, BLOCK_M),
        cdiv(seq_len, BLOCK_N),
        bs * hidden_dim,
    )

    triton_trimul_kernel[grid](
        left_h,
        right_h,
        out,
        bs,
        seq_len,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------ #
    # 8. Hidden‑dim layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=1e-5,
    )
    out = out * out_gate                         # element‑wise gating
    out = F.linear(out, w("to_out.weight"))      # back to dim

    return out
