    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    device = x.device

    # Cast input to float16 for compute
    x = x.to(torch.float16)

    # LayerNorm
    eps = 1e-5
    # cast weights and biases to half
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Flatten for linear
    x_flat = x_norm.reshape(-1, dim)  # shape [B*N*N, dim]

    # Projections
    left = torch.nn.functional.linear(x_flat, left_proj_w)  # shape [B*N*N, hidden_dim]
    right = torch.nn.functional.linear(x_flat, right_proj_w)

    # Reshape back
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # Apply mask if needed
    if config.get("nomask", False) or mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)
    mask_expanded = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_expanded
    right = right * mask_expanded

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)

    # Core TriMul via Triton
    out_intermediate = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Compute strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out_intermediate.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

        left,
        right,
        out_intermediate,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Now out_intermediate shape [B,N,N,hidden_dim] dtype float32
    # to_out_norm
    eps = 1e-5
    out_mean = out_intermediate.mean(-1, keepdim=True)
    out_var = out_intermediate.var(-1, unbiased=False, keepdim=True)
    out_norm = (out_intermediate - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply out_gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection
    out_norm_flat = out_norm.view(-1, hidden_dim)
    out = torch.nn.functional.linear(out_norm_flat, to_out_w)  # shape [B*N*N, dim]
    out = out.view(B, N, N, dim)

    return out

"""
TriMul outgoing implementation (AlphaFold3 style).

The forward pass consists of:
1. Layer‑norm over the last channel dimension.
2. Linear projections to a hidden dimension (left/right) and three
   gating projections (left_gate, right_gate, out_gate).
3. Optional element‑wise mask on the projected tensors.
4. Gating (element‑wise multiplication with sigmoid outputs).
5. Core “tri‑multiplicative” update:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This O(N³) step is implemented as a batched matrix multiplication in
   Triton.  The batch and hidden indices are fused into a single program‑id,
   yielding a kernel launched with grid (B*H, ceil(N/M), ceil(N/N)).
6. Layer‑norm on the hidden result, gated with out_gate, and final projection
   back to the original channel dimension.

All heavy arithmetic (step 5) runs in half‑precision for bandwidth while
accumulating in FP32 for numerical stability.  The rest of the pipeline
uses regular PyTorch ops (highly optimized on the H100).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the tri‑multiplicative update (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # fp16 *  [B, N, N, H]
    right_ptr,              # fp16 *  [B, N, N, H]
    out_ptr,                # fp32 *  [B, N, N, H]
    B, N, H,                # runtime sizes
    # strides of left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides of right
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides of out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute for each (b, h):
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    using a tiled mat‑mul (i‑k) · (k‑j) → (i‑j).
    """
    pid0 = tl.program_id(0)          # fused batch*hidden axis
    pid_i = tl.program_id(1)          # tile over i (rows)
    pid_j = tl.program_id(2)          # tile over j (cols)

    # --------------------------------------------------------------
    # decode batch and hidden index from pid0
    # --------------------------------------------------------------
    batch = pid0 // H
    hidden = pid0 % H

    # --------------------------------------------------------------
    # offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)      # [M]
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)      # [N]
    # base pointer offsets (batch and hidden dimension)
    base_left  = batch * stride_left_b  + hidden * stride_left_h
    base_right = batch * stride_right_b + hidden * stride_right_h
    base_out   = batch * stride_out_b   + hidden * stride_out_h

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)                     # [K]

        # bounds checks
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # ----------------------------------------------------------
        # load left tile : shape (M, K)
        # left layout: (batch, i, k, hidden)
        # ----------------------------------------------------------
        left_ptrs = left_ptr + base_left \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----------------------------------------------------------
        # load right tile (transposed) : shape (K, N)
        # right layout: (batch, j, k, hidden)  -> we need (k, j)
        # ----------------------------------------------------------
        right_ptrs = right_ptr + base_right \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate dot product over K
        acc += tl.dot(left, right)      # (M,N)

    # --------------------------------------------------------------
    # store the result (FP32)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + base_out \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # configuration
    # ------------------------------------------------------------------
    dim = config["dim"]                 # input/output channel dimension
    hidden_dim = config["hidden_dim"]   # internal hidden dimension
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape   # input shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # move everything to fp16 for the heavy compute (except final output)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias   = weights["norm.bias"].to(torch.float16, device=device)

    mean = x.mean(-1, keepdim=True)
    var  = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gating projections
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16, device=device)   # [H, dim]
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16, device=device)

    # Flatten to (B*N*N, dim) for fast matmul
    x_flat = x_norm.view(-1, dim)      # (B*N*N, dim)

    # Projections to hidden dimension
    left  = F.linear(x_flat, left_proj_w)   # (B*N*N, H)
    right = F.linear(x_flat, right_proj_w)
    left  = left.view(B, N, N, hidden_dim).contiguous()
    right = right.view(B, N, N, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # Optional mask (shape [B,N,N])
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16, device=device)
    mask_expanded = mask_tensor.unsqueeze(-1)               # [B,N,N,1]
    left  = left * mask_expanded
    right = right * mask_expanded

    # ------------------------------------------------------------------
    # Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul multiplication (Triton)
    # ------------------------------------------------------------------
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in element units)
    s_left  = left.stride()          # (b, i, k, h)
    s_right = right.stride()
    s_out   = out_hidden.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_kernel[grid](
        left, right, out_hidden,
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, num_stages=2
    )

    # ------------------------------------------------------------------
    # Post‑processing on the hidden result
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32, device=device)   # [H]
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32, device=device)

    # LayerNorm over hidden dimension
    mean_h = out_hidden.mean(-1, keepdim=True)
    var_h  = out_hidden.var(-1, unbiased=False, keepdim=True)
    out_norm = (out_hidden - mean_h) / torch.sqrt(var_h + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply out gate (already in fp16 → cast)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final projection back to original dimension
    to_out_w = weights["to_out.weight"].to(torch.float32, device=device)   # [dim, H]
    out_norm_flat = out_norm.view(-1, hidden_dim)      # (B*N*N, H)
    out = F.linear(out_norm_flat, to_out_w)            # (B*N*N, dim)
    out = out.view(B, N, N, dim)                      # [B,N,N,dim]

    return out
