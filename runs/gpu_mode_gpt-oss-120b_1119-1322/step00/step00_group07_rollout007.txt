    x_norm_h = x_norm.half()

    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    right_proj = torch.nn.functional.linear(x_norm_h, right_proj_w, bias=None)

    # Gate values
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, left_gate_w, bias=None))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, right_gate_w, bias=None))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, out_gate_w, bias=None))

    device = input_tensor.device
    dim = config["dim"]

    nomask = config.get("nomask", True)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            (dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"])
    x_norm_h = x_norm.half()

    # Project and gate weights
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"].half()

    # Projections
    left_proj = torch.nn.functional.linear(x_norm_h, left_proj_w)
    right_proj = torch.nn.functional.linear(x_norm_h, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, left_gate_w))

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B,
    N,
    H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Block start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds in i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block A (shape M x K)
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load right block B transposed (shape K x N): we need rows K (k) and cols N (j)
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Rank-2 dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM kernel.
The heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is executed in a batched GEMM kernel that processes each (batch, hidden) pair
as an independent GEMM of shape (N, N) @ (N, N)ᵀ.  All other ops (layer‑norm,
linear projections, gating, masking) are performed with PyTorch (mixed‑precision)
while the matrix‑multiply core is accelerated by the custom kernel.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,        # *fp16 left  [B, N, N, H]
    right_ptr,       # *fp16 right [B, N, N, H]
    out_ptr,         # *fp32 out   [B, N, N, H]
    B: tl.int32, N: tl.int32, H: tl.int32,
    # left strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # right strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # out strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM:  C[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h]   (B = right)"""
    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)
    pid_bh = tl.program_id(2)         # combined (batch, hidden)

    # decode batch & hidden index
    b = pid_bh // H
    h = pid_bh % H

    # offsets within the N dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load left block A[ i , k ] -----
        a_ptrs = (left_ptr
                  + b * stride_l_b
                  + h * stride_l_h
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                # fp16
        a = a.to(tl.float32)

        # ----- Load right block Bᵀ[ k , j ] -----
        # note: B is stored as (b, j, k, h); we load it transposed so that
        # its shape is (BLOCK_K, BLOCK_N) → dot(A,B) yields (i,j)
        b_ptrs = (right_ptr
                  + b * stride_r_b
                  + h * stride_r_h
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                # fp16
        b = b.to(tl.float32)

        # ----- Rank‑2 dot (M×K)·(K×N) -----
        acc += tl.dot(a, b)

    # ----- Store result -----
    c_ptrs = (out_ptr
              + b * stride_o_b
              + h * stride_o_h
              + offs_i[:, None] * stride_o_i
              + offs_j[None, :] * stride_o_j)
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]  (bool/float, may be None)
        - weights_dict : mapping of weight names → torch.Tensor
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input layer‑norm (float32 for stability)
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_norm_h = x_norm.half()                     # mixed‑precision for the rest

    # -----------------------------------------------------------------
    # 2) Linear projections + gates (all in fp16)
    # -----------------------------------------------------------------
    # projection weights
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    # gate weights
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # projections
    left_proj = torch.nn.functional.linear(x_norm_h, left_proj_w)      # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x_norm_h, right_proj_w)

    # gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, out_gate_w))  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3) Apply mask (if any) and gate multiplicatively
    # -----------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        # mask broadcasted over the hidden dimension
        mask_h = mask_tensor.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_h
        right = right_proj * right_gate * mask_h
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    # -----------------------------------------------------------------
    # 4) Batched GEMM via Triton
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑blocks
        B * H,                              # (batch, hidden) pairs
    )

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                       # efficient utilization on H100
    )

    # -----------------------------------------------------------------
    # 5) Output normalization, gating and final projection
    # -----------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden dim)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # out_gate is still fp16 → cast to fp32 for the multiplication
    out = out * out_gate.float()

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].half()
    out = torch.nn.functional.linear(out.half(), to_out_w, bias=None).float()

    return out
