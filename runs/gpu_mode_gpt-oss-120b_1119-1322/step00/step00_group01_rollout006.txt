```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)  # combined batch*head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # compute batch and head indices
    b = pid_bh // H
    h = pid_bh % H
    
    # compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    
    # compute offsets for pointers including row/col
    off_i = tl.arange(0, BLOCK_M)
    off_j = tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)
    
    # compute masks for out-of-bounds
    mask_i = i_start + off_i < N
    mask_j = j_start + off_j < N

    # accumulate result
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute strides for left and right pointers for given batch b and head h
    # base address for this batch+head
    left_base = left_ptr + b * stride_l_b + h * stride_l_d
    right_base = right_ptr + b * stride_r_b + h * stride_r_d

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + off_k
        k_valid = cur_k < N
        
        # left tile: rows i, cols k
        left_ptrs = left_base + (i_start + off_i[:, None]) * stride_l_i + (cur_k[None, :]) * stride_l_k
        left_tile = tl.load(left_ptrs, mask = mask_i[:, None] & k_valid[None, :], other=0.0)
        
        # right tile: rows j, cols k -> we need shape (K, N) after transpose
        # load as (K, N) by swapping indices in pointer expression
        right_ptrs = right_base + (cur_k[:, None]) * stride_r_k + (j_start + off_j[None, :]) * stride_r_i
        right_tile = tl.load(right_ptrs, mask = k_valid[:, None] & mask_j[None, :], other=0.0)
        
        # dot product
        acc += tl.dot(left_tile, right_tile)  # left_tile: (M,K), right_tile: (K,N)
    
    # write back
    out_base = out_ptr + b * stride_o_b + h * stride_o_d
    out_ptrs = out_base + (i_start + off_i[:, None]) * stride_o_i + (j_start + off_j[None, :]) * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask = mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # convert to proper dtype
    dtype = input_tensor.dtype
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    assert dim == config["dim"], "dim mismatch"
    
    # LayerNorm
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(dim=-1, keepdim=True)) / torch.sqrt(input_tensor.var(dim=-1, unbiased=False, keepdim=True) + eps)
    x = x * norm_weight + norm_bias
    
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # Apply mask if provided
    if mask is not None and mask.numel() > 0:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask
    else:
        # mask of ones
        pass
    
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    
    # Compute out via triton
    out = tri_mul_triton(left, right)  # returns fp16
    
    # Change back to float32 for later
    out = out.to(torch.float32)
    
    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = (out - out.mean(dim=-1, keepdim=True)) / torch.sqrt(out.var(dim=-1, unbiased=False, keepdim=True) + eps)
    out = out * to_out_norm_weight + to_out_norm_bias
    
    # Multiply out_gate
    out_gate = out_gate.to(out.dtype)
    out = out * out_gate
    
    # Linear to out dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    
    return out

def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                   stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for boundary
    mask_i = i_start + offs_i < N
    mask_j = j_start + offs_j < N

    # base pointers for this batch and head
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # load left tile (M,K)
        left_ptrs = left_base + (i_start + offs_i[:, None]) * stride_l_i + (cur_k[None, :]) * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

"""
TriMul (outgoing) forward implementation.

The kernel computes the heavy N‑³ reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for every batch b and hidden channel h.
It is expressed as a batched matrix multiplication L_h @ R_hᵀ for each
hidden dimension and is implemented with a custom Triton kernel that
uses tf32/FP16 tensor‑core dot products. The surrounding LayerNorm,
linear projections and gating are performed with PyTorch for simplicity.

Supported configurations:
    B ∈ {1,2}, N ∈ {128,256,512,1024}, dim ∈ {128,384},
    hidden_dim ∈ {128,384,768}
The kernel works for both masked and un‑masked inputs.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched (B*H) matrix‑multiplication L @ Rᵀ
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_bh = tl.program_id(0)      # combined batch * hidden dim
    pid_i  = tl.program_id(1)      # tile over output rows  (i)
    pid_j  = tl.program_id(2)      # tile over output cols (j)

    # ------------------------------------------------------------------- #
    # decode batch and hidden‑channel index
    # ------------------------------------------------------------------- #
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------- #
    # tile origin in the output matrix
    # ------------------------------------------------------------------- #
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ------------------------------------------------------------------- #
    # index vectors within a tile
    # ------------------------------------------------------------------- #
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks
    mask_i = i_start + offs_i < N
    mask_j = j_start + offs_j < N

    # ------------------------------------------------------------------- #
    # base pointers that already include batch and hidden‑channel offset
    # ------------------------------------------------------------------- #
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulate in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # reduction over the sequence dimension K
    # ------------------------------------------------------------------- #
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # left tile: shape (M, K)   -> rows i, columns k
        left_ptrs = left_base + \
                    (i_start + offs_i[:, None]) * stride_l_i + \
                    (cur_k[None, :])            * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile transposed on‑the‑fly: shape (K, N) -> rows k, columns j
        right_ptrs = right_base + \
                     (cur_k[:, None])            * stride_r_k + \
                     (j_start + offs_j[None, :]) * stride_r_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------- #
    # write back (cast to FP16 for storage)
    # ------------------------------------------------------------------- #
    out_ptrs = out_base + \
               (i_start + offs_i[:, None]) * stride_o_i + \
               (j_start + offs_j[None, :]) * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Helper that launches the Triton kernel
# --------------------------------------------------------------------------- #
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 64) -> torch.Tensor:
    """
    Triton‑backed implementation of the N³ reduction.
    left / right : [B, N, N, H]  (dtype = float16 or float32)
    Returns       : tensor of the same shape and dtype.
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,               # a reasonable choice for H100
        num_stages=2
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype = float32)
    """
    # ------------------------------------------------------------------- #
    # unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------- #
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, normalized_shape=(dim,),
                     weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        mask = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------- #
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Cast to FP16 for the heavy N³ reduction (tensor‑cores)
    # ------------------------------------------------------------------- #
    left_fp  = left.to(torch.float16)
    right_fp = right.to(torch.float16)

    # ------------------------------------------------------------------- #
    # N³ reduction via Triton
    # ------------------------------------------------------------------- #
    out_fp = tri_mul_triton(left_fp, right_fp,
                            BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)   # fp16 output

    # Back to FP32 for the remaining LayerNorm + projection
    out = out_fp.to(torch.float32)

    # ------------------------------------------------------------------- #
    # LayerNorm over hidden_dim
    # ------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------- #
    # Apply the final output gate and linear projection back to dim
    # ------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
