    stride_right_b, stride_right_i, stride_right_k, stride_right_h,

    num_j = tl.cdiv(N, BLOCK_N)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # out = kernel(left, right)
    out = trimul_outgoing_kernel_launcher(left, right) # Return shape B,N,N,hidden_dim

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    # final linear to output dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

TriMul (outgoing) forward implementation using Triton.
----------------------------------------------------
The heavy part of the module is the Einstein‑summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batch of hidden_dim independent matrix
multiplications `left[..., :, :, d] @ right[..., :, :, d].T`.
The kernel tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and
the hidden dimension (BLOCK_D) and loops over the reduction axis k
in blocks (BLOCK_K).  All other operations (LayerNorms, linear
projections, sigmoid gates) are performed with PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N³ hidden‑dim multiplication
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,        # tensors
    B, N, H,                             # runtime shapes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute out = Σ_k left[..., i,k,d] * right[..., j,k,d]
    left, right: [B,N,N,H]   (contiguous, row‑major)
    out         : [B,N,N,H]
    Tiles:
        i  -> BLOCK_M
        j  -> BLOCK_N
        d  -> BLOCK_D
    Reduction k is blocked with size BLOCK_K.
    """
    # ------------------------------------------------------------------
    # 1) program IDs and block offsets
    # ------------------------------------------------------------------
    pid_ij = tl.program_id(0)            # flattened (i_tile, j_tile)
    pid_bd = tl.program_id(1)            # flattened (batch, d_tile)

    # number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_d = tl.cdiv(H, BLOCK_D)

    # i‑tile and j‑tile
    i_tile = pid_ij // num_j
    j_tile = pid_ij %  num_j
    i_off = i_tile * BLOCK_M
    j_off = j_tile * BLOCK_N

    # batch and d‑tile
    batch = pid_bd // num_d
    d_tile = pid_bd %  num_d
    d_off = d_tile * BLOCK_D

    # ------------------------------------------------------------------
    # 2) index vectors
    # ------------------------------------------------------------------
    i_idx = i_off + tl.arange(0, BLOCK_M)
    j_idx = j_off + tl.arange(0, BLOCK_N)
    d_idx = d_off + tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds (tail handling)
    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_d = d_idx < H

    # accumulator for the tile: [BLOCK_M, BLOCK_N, BLOCK_D]
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) reduction over k
    # ------------------------------------------------------------------
    k = 0
    while k < N:
        k_idx = k + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ------------------------------------------------------------------
        # Load left tile  -> shape (M, K, D)
        # left[b, i, k, d]
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + batch * stride_left_b \
            + i_idx[:, None, None] * stride_left_i \
            + k_idx[None, :, None] * stride_left_k \
            + d_idx[None, None, :] * stride_left_h
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )  # (M, K, D)

        # ------------------------------------------------------------------
        # Load right tile (transposed) -> shape (K, N, D)
        # right[b, j, k, d]  (we want k first so that we can dot)
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + batch * stride_right_b \
            + k_idx[:, None, None] * stride_right_k \
            + j_idx[None, :, None] * stride_right_i \
            + d_idx[None, None, :] * stride_right_h
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )  # (K, N, D)

        # ------------------------------------------------------------------
        # For each d in the block compute a small GEMM
        # ------------------------------------------------------------------
        for db in range(0, BLOCK_D):
            # Skip completely out‑of‑range d
            if not mask_d[db]:
                continue
            left_slice  = left[:, :, db]   # (M, K)
            right_slice = right[:, :, db]  # (K, N)
            acc[:, :, db] += tl.dot(left_slice, right_slice)   # (M, N)

        k += BLOCK_K

    # ------------------------------------------------------------------
    # 4) write back the result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + batch * stride_out_b \
        + i_idx[:, None, None] * stride_out_i \
        + j_idx[None, :, None] * stride_out_j \
        + d_idx[None, None, :] * stride_out_h
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def trimul_outgoing_kernel_launcher(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left, right : [B, N, N, H] (float32, contiguous)
    Returns out : [B, N, N, H]
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 16   # hidden dimension is split into blocks of 16

    # Compute launch grid
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_d = (H + BLOCK_D - 1) // BLOCK_D
    grid = (num_i * num_j, B * num_d)   # (i*j tiles, batch*hidden‑tiles)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul layer.
    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N] or None  (bool / 0‑1)
        - weights      : dict of tensors (see problem statement)
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    output : torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]  # [hidden_dim, dim]

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, N, N] -> unsqueeze for broadcasting
        mask_ = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5) Main N³ multiplication via Triton
    # ------------------------------------------------------------------
    out = trimul_outgoing_kernel_launcher(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]   # [dim, hidden_dim]
    out = F.linear(out, to_out_weight)        # back to [B, N, N, dim]

    return out
