def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, D, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_bd = tl.program_id(0) # 0..B*D-1
    pid_m = tl.program_id(1) # M tile index
    pid_n = tl.program_id(2) # N tile index

    # compute b and d
    b = pid_bd // D
    d = pid_bd % D

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for valid indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to output
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left (A) block shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + b*stride_left_b + d*stride_left_d + offs_m[:, None]*stride_left_i + offs_k[None, :]*stride_left_k

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ (B,N,N,dim).
2. Linear projections (no bias) → left, right ∈ (B,N,N,hidden_dim).
3. Optional mask (broadcasted on the channel dimension) is applied.
4. Gating: left_gate, right_gate, out_gate = sigmoid(Linear(x)).
   left ← left * left_gate,  right ← right * right_gate.
5. Core operation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of shape
   (B·hidden_dim, N, N) where each slice computes
       left_slice @ right_sliceᵀ .
   The computation is performed by a custom Triton kernel that
   tiles the M×N output (i, j) and reduces over K (the k‑axis).
6. Layer‑norm on the hidden dimension, multiply by out_gate,
   and final linear projection back to dim.
The Triton kernel works in float32 (can be changed to fp16 for extra speed)
and is launched with a 3‑D grid:
   (B·hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N)).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batch of matmul C = A @ Bᵀ  (A,B shape: (B,N,N,hd))
# ----------------------------------------------------------------------
@triton.jit
def _trmul_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    B, D, N,           # scalar sizes
    # strides for left tensor (B,N,N,D)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right tensor (B,N,N,D)
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for out tensor (B,N,N,D)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_bd = tl.program_id(0)      # combined batch * hidden_dim index
    pid_m  = tl.program_id(1)      # i‑tile
    pid_n  = tl.program_id(2)      # j‑tile

    # Decode batch and hidden dimension from pid_bd
    b = pid_bd // D
    d = pid_bd % D

    # Offsets for the i and j tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointer to the output tile
    out_ptrs = (out_ptr + b * stride_out_b
                + d * stride_out_d
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M x K)
        a_ptrs = (left_ptr + b * stride_left_b
                  + d * stride_left_d
                  + offs_m[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block (K x N) – note B is accessed as (k, j) because we need Bᵀ
        b_ptrs = (right_ptr + b * stride_right_b
                  + d * stride_right_d
                  + offs_k[:, None] * stride_right_k
                  + offs_n[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate dot product over the K‑axis
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Store the result tile
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused batched matmul using the kernel above
# ----------------------------------------------------------------------
def _trmul_forward(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 64) -> torch.Tensor:
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    left/right: (B,N,N,D) float32
    Returns out of the same shape/dtype.
    """
    assert left.shape == right.shape
    B, N, _, D = left.shape

    out = torch.empty((B, N, N, D), dtype=left.dtype, device=left.device)

    # Strides in element units (not bytes)
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    grid = (
        B * D,                                 # batch × hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N           # j‑tiles
    )
    _trmul_kernel[grid](
        left, right, out,
        B, D, N,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # reasonable for H100
    )
    # Ensure the kernel has finished before we read the tensor
    torch.cuda.synchronize()
    return out


# ----------------------------------------------------------------------
# Simple LayerNorm (no bias handling edge‑cases)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension.
    weight/bias have shape (C,).
    """
    # Compute per‑instance mean and variance
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward – the heavy N³ reduction is off‑loaded
    to a Triton kernel. All surrounding linear/gate and LayerNorm
    operations are performed with native PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Configuration & weight extraction
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over `dim`
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias)
    left_proj_weight  = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]         # (hidden_dim, dim)

    # Gating projections (no bias)
    left_gate_weight  = weights["left_gate.weight"]          # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]         # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]           # (hidden_dim, dim)

    # Output head
    to_out_norm_weight = weights["to_out_norm.weight"]       # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]         # (hidden_dim,)
    to_out_weight      = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor, norm_weight, norm_bias, eps=eps)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        # all‑ones mask → no effect
        mask = torch.ones_like(input_tensor[..., 0], dtype=input_tensor.dtype, device=input_tensor.device)
    mask_expanded = mask.unsqueeze(-1)   # (B,N,N,1)

    left = left * mask_expanded
    right = right * mask_expanded

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction via Triton
    # ------------------------------------------------------------------
    # The kernel works best in float32; you may switch to fp16 by casting
    # left/right to half and adjusting BLOCK sizes as needed.
    out = _trmul_forward(left, right,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6. Output head LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = _layer_norm(out, to_out_norm_weight, to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
