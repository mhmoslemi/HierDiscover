    # LayerNorm
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [..., hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

    right_k = right_k.permute(0, 3, 1, 2).contiguous()

import triton
import triton.language as tl

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H, B,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_hidden = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden index
    b = batch_hidden // H
    h = batch_hidden % H

    # Coordinates
    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    # Pointers to the start of this batch and hidden slice
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Define ranges
    row_offset = row_start + tl.arange(0, BLOCK_M)
    col_offset = col_start + tl.arange(0, BLOCK_N)
    # Mask out-of-bounds rows/cols
    row_mask = row_offset < N
    col_mask = col_offset < N

    # Loop over k dimension
    num_k_blocks = (N + BLOCK_K - 1) // BLOCK_K
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left tile (M x K)
        left_ptrs = left_base + row_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=row_mask[:, None] & k_mask[None, :], other=0.0, dtype=tl.float16)

        # Load right tile (N x K) (j dimension corresponds to col_offset)
        right_ptrs = right_base + col_offset[:, None] * stride_r_i + k_offset[None, :] * stride_r_k
        right_tile = tl.load(right_ptrs, mask=col_mask[:, None] & k_mask[None, :], other=0.0, dtype=tl.float16)

        # Compute dot product, left (M,K) * right^T (K,N) (trans_b=True => treat right as (K,N))
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # Store result
    out_ptrs = out_base + row_offset[:, None] * stride_o_i + col_offset[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=row_mask[:, None] & col_mask[None, :])

"""
TriMul (outgoing) forward pass implemented with Torch + a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension.
2. Linear projections:
   * left  = x·W_left   (shape [B,N,N,H])
   * right = x·W_right  (shape [B,N,N,H])
3. Compute three gating vectors with sigmoid:
   left_gate , right_gate , out_gate  (shape [B,N,N,H])
4. Apply optional mask and gates:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
5. Perform the core O(N³) “tri‑multiplicative” update:
   out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiply for every hidden channel `h`.
   It is implemented by a fused Triton kernel that iterates over the
   reduction dimension `k` in blocks.
6. Layer‑norm the result over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
All operations are performed in float32 (the matmul kernel accumulates in
float32 even when the input tensors are half‑precision).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H, B,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for a fixed batch `b` and hidden channel `h`:

        out[i, j] = Σ_k left[i, k] * right[j, k]

    Parameters
    ----------
    left_ptr, right_ptr, out_ptr : pointers to the tensors.
    N, H, B                     : problem sizes.
    stride_*                    : tensor strides (in elements).
    BLOCK_M, BLOCK_N, BLOCK_K   : compile‑time tile sizes.
    """
    # 3‑D grid: (batch_hidden, block_i, block_j)
    batch_hidden = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Decode batch and hidden indices
    b = batch_hidden // H
    h = batch_hidden % H

    # Starting indices of the output tile
    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Range vectors
    rows = row_start + tl.arange(0, BLOCK_M)
    cols = col_start + tl.arange(0, BLOCK_N)
    row_mask = rows < N
    col_mask = cols < N

    # Loop over the reduction dimension `k`
    num_k_blocks = (N + BLOCK_K - 1) // BLOCK_K
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        ks = k_start + tl.arange(0, BLOCK_K)
        k_mask = ks < N

        # Load a tile of `left`  (shape BLOCK_M x BLOCK_K)
        left_offsets = rows[:, None] * stride_l_i + ks[None, :] * stride_l_k
        left_tile = tl.load(left_base + left_offsets,
                            mask=row_mask[:, None] & k_mask[None, :],
                            other=0.0, dtype=tl.float16)

        # Load a tile of `right` (shape BLOCK_N x BLOCK_K)
        # Note: we store it as (N, K) and use trans_b=True in tl.dot
        right_offsets = cols[:, None] * stride_r_i + ks[None, :] * stride_r_k
        right_tile = tl.load(right_base + right_offsets,
                             mask=col_mask[:, None] & k_mask[None, :],
                             other=0.0, dtype=tl.float16)

        # GEMM: (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # Write the final tile to global memory
    out_offsets = rows[:, None] * stride_o_i + cols[None, :] * stride_o_j
    tl.store(out_base + out_offsets,
             acc,
             mask=row_mask[:, None] & col_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper to fetch weights (they may be stored as Parameter or Tensor)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim is the channel dimension)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, _w("left_proj.weight"))   # [B,N,N,H]
    right = torch.nn.functional.linear(x_norm, _w("right_proj.weight"))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Gating vectors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, _w("left_gate.weight")))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, _w("right_gate.weight")))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, _w("out_gate.weight")))

    # ------------------------------------------------------------------
    # Optional mask handling (mask may be None or all‑ones)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy O(N³) kernel (if possible)
    # ------------------------------------------------------------------
    if dtype in (torch.float16, torch.bfloat16):
        left_k = left.to(torch.float16)
        right_k = right.to(torch.float16)
    else:
        left_k = left
        right_k = right

    # Rearrange to [B, H, N, N] for a batched GEMM per hidden channel
    left_k = left_k.permute(0, 3, 1, 2).contiguous()
    right_k = right_k.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_k.shape

    # Output buffer (accumulate in float32)
    out_k = torch.empty_like(left_k, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per tile
    BLOCK_N = 64   # cols per tile
    BLOCK_K = 64   # reduction block size

    grid = (B * H,                # batch * hidden
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trimul_kernel[grid](
        left_k, right_k, out_k,
        N, H, B,
        left_k.stride(0), left_k.stride(1), left_k.stride(2), left_k.stride(3),
        right_k.stride(0), right_k.stride(1), right_k.stride(2), right_k.stride(3),
        out_k.stride(0), out_k.stride(1), out_k.stride(2), out_k.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to the original layout: [B, N, N, H]
    out = out_k.permute(0, 2, 3, 1).contiguous()   # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Gated output and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, _w("to_out.weight"))

    return out
