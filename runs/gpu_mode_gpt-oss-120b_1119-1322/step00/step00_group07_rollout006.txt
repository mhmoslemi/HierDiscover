```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, dim_check = input_tensor.shape
    assert dim_check == dim
    device = input_tensor.device

    eps = 1e-5

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)   # shape [bs,N,N,dim]
    # cast to half
    x_norm = x_norm.to(torch.float16)

    # Linear Proj
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply mask if present
    if mask is not None:
        mask = mask.to(torch.float16).unsqueeze(-1)   # shape [bs,N,N,1] broadcast to hidden_dim
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # Flatten batch and hidden_dim into a single batch dimension for Triton kernel
    left = left.view(bs * hidden_dim, N, N)   # shape: [batch_h, M, K]
    right = right.view(bs * hidden_dim, N, N) # shape: [batch_h, N, N]

    # Run Triton batched GEMM
    out = triton_matmul(left, right, N, N, N)  # returns same shape as left: [batch_h, N, N] in half

    # Reshape back to [bs, N, N, hidden_dim]
    out = out.view(bs, N, N, hidden_dim)

    # Convert back to float32 for post-processing
    out = out.to(torch.float32)

    # to_out_norm: layernorm over hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # multiply by out_gate (which is still half; convert to float32)
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate

    # linear to final dim
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [bs,N,N,dim]
```

"""
custom_kernel – TriMul (outgoing) forward pass

Algorithm
---------
1. Layer‑norm the input tensor `x` on the last dimension.
2. Linear projections (left/right) and three gating projections are computed
   with bias‑free linear layers.  Gating vectors are passed through a sigmoid.
3. Apply optional mask and element‑wise gates:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (mask is broadcast over the hidden dimension)
4. Fuse the hidden dimension into a batch dimension
        batch_h = bs * hidden_dim
   Reshape `left` / `right` to (batch_h, N, N) such that each
   hidden‑channel slice becomes an independent matrix.
5. Compute the core "tri‑multiplicative" update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched GEMM:
        out_batch = left_batch @ right_batchᵀ
   A custom Triton kernel (`batched_gemm_kernel`) performs this
   half‑precision matmul (accumulating in FP32) for all batches.
6. Reshape the result back to (bs, N, N, hidden_dim) and convert to FP32.
7. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and finally project back to the original dimension with `to_out`.

The heavy O(N³·hidden) work is performed on the GPU by the Triton kernel,
while light‑weight linear / norm / sigmoid work is done with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ  where
#   A : (batch_h, M, K)   – left  tensor   (M=N, K=N)
#   B : (batch_h, N, K)   – right tensor   (rows=j, cols=k)
#   C : (batch_h, M, N)   – output tensor (i,j)
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, N, K)   – we read Bᵀ by swapping indices
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # Starting coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A tile (M×K) ----
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load Bᵀ tile (K×N) ----
        # B is stored as (batch, N, K); we need element Bᵀ[k, n] = B[batch, n, k].
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ---- Accumulate ----
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ---- Write back result (convert to FP16) ----
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    """
    TriMul (outgoing) forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict, must contain "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0. unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (float32 for numerical stability)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [bs, N, N, dim]  FP32

    # -----------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – keep in FP32 for now

    right_proj_w = weights["right_proj.weight"]

    out_gate_w   = weights["out_gate.weight"]
