@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    pid0 = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # Compute batch and hidden indices
    b = pid0 // H
    h = pid0 % H
    
    # Compute start indices for i and j
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    
    # Range within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    
    # Masks for out-of-bound i/j
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # Load left tile (i,k)
        left_ptrs = left_ptr + b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        
        # Load right tile transposed (k,j)
        right_ptrs = right_ptr + b * stride_right_b + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j + h * stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        
        # Perform dot product accumulation
        acc += tl.dot(left, right)
        
    # Store result
    out_ptrs = out_ptr + b * stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,D).
2. Linear projections → left, right  (shape B,N,N,H) and three sigmoidal gates.
3. Optional pair‑wise mask is applied before gating.
4. Core “tri‑mul” operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for every hidden head (H) and batch,
   implemented in a custom Triton kernel that tiles the (i,j) output plane
   (BLOCK_M × BLOCK_N) and iterates over K‑tiles (BLOCK_K).  The kernel uses
   `tl.dot` for the inner product and accumulates in FP32.
5. Layer‑norm over the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original channel size D.

Only the cubic “tri‑mul” step is hand‑written in Triton; all other
operations use native PyTorch kernels (highly tuned cuBLAS/cuDNN).
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = Σ_k left[...,k,h] * right[...,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Kernel computes   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid0 = tl.program_id(0)   # combined batch * head index
    pid_m = tl.program_id(1)   # i‑tile
    pid_n = tl.program_id(2)   # j‑tile

    # ------------------------------------------------------------------
    # Resolve batch (b) and hidden‑head (h) from pid0
    # ------------------------------------------------------------------
    b = pid0 // H
    h = pid0 % H

    # Starting coordinates of the current tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Offsets inside the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Bounds masks for the (i,j) dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in BLOCK_K sized chunks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_block in range(0, num_k_blocks):
        k_start = k_block * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left tile   (i,k)  → shape [BLOCK_M, BLOCK_K]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # FP32 (default)

        # ------------------------------------------------------------------
        # Load right tile transposed (k,j)  → shape [BLOCK_K, BLOCK_N]
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Accumulate dot product over the K‑dimension
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the result tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_outgoing(left: torch.Tensor, right: torch.Tensor,
                     block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Compute the outgoing triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left, right : torch.Tensor
        Shape (B, N, N, H). Must be contiguous and on CUDA.
    block_m, block_n, block_k : int
        Tiling sizes for the kernel.

    Returns
    -------
    out : torch.Tensor
        Shape (B, N, N, H), same dtype and device as the inputs.
    """
    assert left.is_cuda and right.is_cuda, "Tensors must be on CUDA"
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape

    # Allocate output
    out = torch.empty_like(left)

    # Ensure contiguous memory layout (required for correct strides)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Extract strides (in element counts)
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_k, s_out_h = out.stride()

    # Triton grid: (batch * head, i‑tiles, j‑tiles)
    grid = (
        B * H,
        (N + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
    )

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_i,   # j‑dimension uses the same stride as the first spatial axis
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_k,
        s_out_h,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D]
        - mask         : torch.Tensor of shape [B, N, N] (binary, optional)
        - weights      : dict of pre‑loaded weight tensors
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        The transformed tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract weight tensors (all are on the same device as the input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm over the channel dimension D
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → shape (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)      # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional pair‑wise mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates (mask has already been applied)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core triangular multiplicative update (Triton kernel)
    # ------------------------------------------------------------------
    # The kernel works efficiently for the allowed shapes (B∈{1,2}, N up to 1024,
    # hidden_dim up to 128) on an H100.
    out = tri_mul_outgoing(left, right, block_m=64, block_n=64, block_k=64)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm over hidden_dim, then gate, then final linear proj
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # → [B,N,N,dim]

    return out
