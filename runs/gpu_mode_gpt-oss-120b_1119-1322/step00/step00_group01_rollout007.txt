def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # unpack weights
    # ensure device same
    # layernorm and linear using pt

    # compute x_norm
    dim = config["dim"]; hidden_dim = config["hidden_dim"]

    # fetch epsilon maybe config? we'll set eps=1e-5.

    x = input_tensor

    # LayerNorm across last dim
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    # x_norm = (x - mean) / sqrt(var+eps) * norm_weight + norm_bias
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_expanded
        right = right * mask_expanded

    # Gates
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Now compute "out" via Triton.
    B, N, _, _ = left.shape  # Actually left shape [B,N,N,H]
    H = hidden_dim

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    # Output buffer
    out_agg = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # get strides, convert to int64
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_agg.stride()

    # compute block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64  # block along reduction axis
    BLOCK_D = 8

    # compute grid
    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    num_d_blocks = (H + BLOCK_D - 1) // BLOCK_D
    num_batch = B
    # combine batch and d blocks into first grid dimension
    grid = (num_batch * num_d_blocks, num_i_blocks, num_j_blocks)

    # launching kernel
    triton_outgoing_kernel[grid](
        left, right, out_agg,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    # Now out_agg now contains sum over K.

    # Apply to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out_agg, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

"""
TriMul (outgoing) implementation.

The forward pass consists of three stages:

1.  PyTorch (fast, memory‑bound) part:
    * Input LayerNorm.
    * Linear projections `left_proj`, `right_proj`.
    * Optional binary mask multiplication.
    * Gated linear layers `left_gate`, `right_gate`, `out_gate`.

2.  Triton kernel:
    Computes the core O(N³·H) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    for each batch `b` and a block of the hidden dimension `d`.
    The kernel treats the hidden dimension as an extra broadcasted
    dimension so a single `tl.dot` call evaluates the reduction for
    `BLOCK_D` channels at once.  Results are accumulated in FP32 and
    written back in FP16.

3.  PyTorch post‑processing:
    * Hidden‑dim LayerNorm (`to_out_norm`).
    * Output gate (`out_gate`).
    * Final linear projection back to the original dimension.

The kernel is deliberately kept simple and uses block sizes that
fit well on an H100 (BLOCK_M=32, BLOCK_N=32, BLOCK_K=64, BLOCK_D=8).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,               # *fp16   [B, N, N, H]
    right_ptr,              # *fp16   [B, N, N, H]
    out_ptr,                # *fp16   [B, N, N, H]
    B, N, H,                # runtime integers
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int64 strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # int64 strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # int64 strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    left : (B, N, N, H) – indexed as (b, i, k, d)
    right: (B, N, N, H) – indexed as (b, j, k, d)
    The kernel iterates over `k` in blocks of size BLOCK_K and
    accumulates into FP32 registers.
    """
    pid_batch_d = tl.program_id(0)   # combined batch * hidden‑block id
    pid_i = tl.program_id(1)         # block row (i)
    pid_j = tl.program_id(2)         # block col (j)

    # ---- batch & hidden‑dim block -------------------------------------------------
    num_d_blocks = tl.cdiv(H, BLOCK_D)
    batch_id = pid_batch_d // num_d_blocks
    d_block_id = pid_batch_d % num_d_blocks

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    d_start = d_block_id * BLOCK_D

    # offsets for each dimension
    i_off = i_start + tl.arange(0, BLOCK_M)
    j_off = j_start + tl.arange(0, BLOCK_N)
    d_off = d_start + tl.arange(0, BLOCK_D)

    # out‑of‑range masks
    mask_i = i_off < N
    mask_j = j_off < N
    mask_d = d_off < H

    # accumulator in FP32: (BLOCK_D, BLOCK_M, BLOCK_N)
    acc = tl.zeros((BLOCK_D, BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over k --------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # -------- left : shape (D, M, K) -----------------------------------------
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_b
            + d_off[:, None, None] * stride_l_h
            + i_off[None, :, None] * stride_l_i
            + k_off[None, None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=(mask_d[:, None, None] & mask_i[None, :, None] & mask_k[None, None, :]),
            other=0.0,
        )  # fp16

        # -------- right : shape (D, K, N) ----------------------------------------
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_b
            + d_off[:, None, None] * stride_r_h
            + k_off[:, None, None] * stride_r_k
            + j_off[None, None, :] * stride_r_j
        )
        right = tl.load(
            right_ptrs,
            mask=(mask_d[:, None, None] & mask_k[:, None, None] & mask_j[None, None, :]),
            other=0.0,
        )  # fp16

        # dot: (D,M,K) @ (D,K,N) -> (D,M,N)
        acc += tl.dot(left, right)  # result is FP32

    # ---- write back ---------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_b
        + i_off[None, :, None] * stride_o_i
        + j_off[None, None, :] * stride_o_j
        + d_off[:, None, None] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=(mask_d[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :]),
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        * input   – Tensor[B, N, N, dim] (float32)
        * mask    – Tensor[B, N, N]   (bool/float) or ``None``.
        * weights – dict with the following keys (all torch.Tensors on the same device):
            - "norm.weight", "norm.bias"
            - "left_proj.weight", "right_proj.weight"
            - "left_gate.weight", "right_gate.weight", "out_gate.weight"
            - "to_out_norm.weight", "to_out_norm.bias"
            - "to_out.weight"
        * config  – dict containing at least ``"dim"`` and ``"hidden_dim"``.
    Returns
    -------
    Tensor[B, N, N, dim]  – the result of the TriMul forward pass (float32).
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ----------------------------------------------------------------------
    # 1) Pre‑processing in PyTorch (LayerNorm, Linear, gating, masking)
    # ----------------------------------------------------------------------
    # Input LayerNorm
    x = F.layer_norm(inp, (dim,), weight=weights["norm.weight"],
                     bias=weights["norm.bias"], eps=eps)

    # Linear projections to hidden dimension (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply optional mask (mask is broadcast over the hidden dimension)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to FP16 for the heavy reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # H == hidden_dim

    # ----------------------------------------------------------------------
    # 2) Triton kernel: contraction over the third axis
    # ----------------------------------------------------------------------
    # Output buffer (FP16, will be cast back to FP32 later)
    out_hidden = torch.empty((B, N, N, H), dtype=torch.float16,
                             device=inp.device)

    # Gather stride information (in element counts, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out_hidden.stride()

    # Block sizes (tuned for H100, FP16)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D = 8

    # Grid dimensions
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_d = (H + BLOCK_D - 1) // BLOCK_D
    grid = (B * grid_d, grid_i, grid_j)

    # Launch kernel
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        H,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_j,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Convert back to FP32 for the remaining lightweight ops
    out_hidden_f32 = out_hidden.to(torch.float32)

    # ----------------------------------------------------------------------
    # 3) Post‑processing (hidden‑dim LayerNorm, output gate, final linear)
    # ----------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate  # apply output gate

    # Final projection back to original dimension
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
