"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The module implements the forward pass of the “Triangle Multiplicative
Update (outgoing)” operation used in AlphaFold‑3.  The only heavy
computation is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
`left @ rightᵀ` for every hidden channel.  This kernel fuses the
N³ reduction across the sequence dimension and runs completely on the
GPU (float‑32 accumulation, float‑16/32 input).  All other operations
(LayerNorm, linear projections, sigmoid gates) are performed with
native PyTorch calls.

The kernel works on a tile of size (BLOCK_M × BLOCK_N) of the output
matrix for a single `(batch, hidden)` pair.  The third program‑id axis
covers `batch * hidden_dim` independent GEMM tasks, each computing

    C = A @ Bᵀ   with   A = left[b,:, :, h]   (N × N)
                     B = right[b,:, :, h]  (N × N)

The implementation uses explicit stride handling so it works for any
contiguous layout of the input tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core einsum
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # strides for left (batch, i, k, hidden)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (batch, j, k, hidden)
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    # strides for out  (batch, i, j, hidden)
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    # problem size
    N, H,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    for a tile (BLOCK_M × BLOCK_N) of (i, j) and a single (b, h) pair.
    """
    pid_c = tl.program_id(2)                     # index over (batch, hidden)
    batch = pid_c // H
    hidden = pid_c % H

    # base pointers for the current (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_l_batch + hidden * stride_l_h
    right_base = right_ptr + batch * stride_r_batch + hidden * stride_r_h
    out_base   = out_ptr   + batch * stride_o_batch + hidden * stride_o_h

    pid_m = tl.program_id(0)                     # tile over i (output row)
    pid_n = tl.program_id(1)                     # tile over j (output col)

    # linear offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows/cols (handles N not multiple of BLOCK_*)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_out = mask_i[:, None] & mask_j[None, :]

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_idx in range(num_k):
        offs_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load A = left[b, i, k, h] -----
        a_ptrs = (left_base
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- load Bᵀ = right[b, j, k, h] (transposed view) -----
        # we need Bᵀ[k, j] = right[b, j, k, h]
        b_ptrs = (right_base
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ----- accumulate -----
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                 # input / output channel dimension
    hidden_dim = config["hidden_dim"]   # inner hidden dimension
    eps = 1e-5                          # LayerNorm epsilon (default)

    # ------------------------------------------------------------------
    # weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input (dim)
    # ------------------------------------------------------------------
    x = input_tensor
    # mean & var over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias          # broadcast over last dim

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x_norm, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # cast mask to same dtype and broadcast
        mask_exp = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections on the *original* normed input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core Triton kernel: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    Equivalent to batched GEMM left @ rightᵀ for each hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # [B,N,N,hidden_dim]

    # ensure contiguity – Triton works with arbitrary strides, but contiguous
    # layout is faster for the block loads
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # stride information (int64)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – tuned for H100 (float32 compute bound)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim                    # (batch, hidden) tasks
    )

    _trimul_gemm_kernel[grid](
        left, right, out,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,                     # 8 warps gives good occupancy on H100
    )

    # ------------------------------------------------------------------
    # 6️⃣ LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # 7️⃣ Apply final gating and output linear projection
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate
    output = F.linear(out_gated, to_out_weight)   # [B,N,N,dim] , no bias

    return output
