    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile index
    pid_n = tl.program_id(1)  # j tile index
    pid_bd = tl.program_id(2)  # batch*D index

    batch = pid_bd // D
    d = pid_bd % D

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N

"""
TriMul (outgoing) forward pass.

The heavy part of the TriMul block is the 4‑D einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for each hidden dimension d:
    out_d = left_d @ right_d.T      (left/right: [B,N,N] → [B,N,N] per d)

A fused Triton kernel computes this N×N×hidden_dim GEMM for each batch
without materialising the large intermediate tensor, while the surrounding
layer‑norms, linear projections and gating are handled in PyTorch for
readability.

The kernel works on blocks (BLOCK_M × BLOCK_N) and iterates over the
reduction dimension K in tiles of size BLOCK_K.  It supports arbitrary
sequence length N (up to 1024 in the tests) and hidden dimension D
(typically 128).  All operations are performed in float32; you may cast
inputs to fp16 before the kernel call for extra speed if desired.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, D,
    # strides (in element units, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a tiled GEMM.  The third index (k) is reduced; the fourth
    dimension d is fused into the grid (one block per (b,d) pair).
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_bd = tl.program_id(2)         # combined batch & hidden dim

    # recover batch and hidden index from the combined dimension
    batch = pid_bd // D
    d = pid_bd % D

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, d) pair
    left_base  = left_ptr  + batch * stride_left_b  + d * stride_left_d
    right_base = right_ptr + batch * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + batch * stride_out_b   + d * stride_out_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles of BLOCK_K
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for the k dimension (handles tail)
        mask_k = offs_k < N

        # pointers for the current tile
        a_ptrs = left_base \
                 + offs_m[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        b_ptrs = right_base \
                 + offs_k[:, None] * stride_right_k \
                 + offs_n[None, :] * stride_right_i

        # load tiles (apply masks)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate the dot product (a: M×K, b: K×N)
        acc += tl.dot(a, b)

    # write back the result
    out_ptrs = out_base \
               + offs_m[:, None] * stride_out_i \
               + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a Triton kernel for the core
    reduction.  The API matches the problem statement.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper: LayerNorm using provided weights (no bias if not given)
    # ------------------------------------------------------------------
    def _layer_norm(x, weight, bias, eps=1e-5):
        return F.layer_norm(x, (x.shape[-1],), weight=weight, bias=bias, eps=eps)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    # Allocate output tensor for the reduction
    out = torch.empty_like(left)   # shape [B,N,N,hidden_dim], dtype matches left/right

    # Ensure tensors are contiguous for correct stride calculation
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides (in elements) for the 4‑D tensors
    sL = left.stride()   # (b,i,k,d)
    sR = right.stride()
    sO = out.stride()

    # Block sizes – tuned for H100 and the typical N range
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles over i, tiles over j, batches × hidden_dim)
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    # Launch the Triton kernel
    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        # problem dimensions
        B=bs,
        N=seq_len,
        D=hidden_dim,
        # strides
        stride_left_b=sL[0],
        stride_left_i=sL[1],
        stride_left_k=sL[2],
        stride_left_d=sL[3],
        stride_right_b=sR[0],
        stride_right_i=sR[1],
        stride_right_k=sR[2],
        stride_right_d=sR[3],
        stride_out_b=sO[0],
        stride_out_i=sO[1],
        stride_out_j=sO[2],
        stride_out_d=sO[3],
        # block sizes (compile‑time constants)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # 4 warps gives good occupancy for these sizes
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # Final linear: hidden_dim → dim
    output = F.linear(out, weights["to_out.weight"])

    return output
