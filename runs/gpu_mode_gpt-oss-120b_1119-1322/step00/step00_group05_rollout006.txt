def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch
    pid_i = tl.program_id(1)  # i
    pid_j = tl.program_id(2)  # j

    # bounds check (in case extra threads)
    mask = (pid_i < N) & (pid_j < N) & (pid_b < B)
    if not mask:
        return

    i = pid_i
    j = pid_j

    # accumulator
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    k = 0

        # accumulate

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Linear projections (left/right) and three gating projections are applied
   using the supplied weight tensors (no bias terms).
3. Optional mask (shape [B,N,N]) is applied to the projected values.
4. Gates are applied element‑wise (sigmoid).
5. Core “tri‑multiplicative” update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   This is performed by a Triton kernel where each thread computes the
   full hidden‑dim vector `d` for a single (b,i,j) triplet and loops over
   the sequence dimension `k`.  Half‑precision is used for the loads and the
   accumulation is performed in float32 to preserve numerical stability.
6. The result is layer‑normed again, multiplied by an output gate and finally
   projected back to the original dimension.

The kernel purposefully computes the O(N³·hidden_dim) contraction while
keeping memory traffic low by loading 16‑bit vectors and accumulating in
32‑bit registers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel for the O(N³) contraction
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # fp16*  [B,N,N,H]
    right_ptr,         # fp16*  [B,N,N,H]
    out_ptr,           # fp16*  [B,N,N,H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_H: tl.constexpr,            # hidden dimension (compile‑time)
):
    # program ids: (batch, i, j)
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # bounds – grid size equals N so the check is only for safety
    if pid_b >= B or pid_i >= N or pid_j >= N:
        return

    i = pid_i
    j = pid_j

    # accumulator for the hidden dimension
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # loop over k dimension
    k = 0
    while k < N:
        # compute base pointers for the current (i,k) and (j,k)
        left_offset  = pid_b * stride_l_b + i * stride_l_i + k * stride_l_k
        right_offset = pid_b * stride_r_b + j * stride_r_i + k * stride_r_k

        # -----------------------------------------------------------------
        # Load a vector of length H for left[i,k,:] and right[j,k,:]
        # -----------------------------------------------------------------
        offs_h = tl.arange(0, BLOCK_H)
        left_vec  = tl.load(left_ptr + left_offset  + offs_h * stride_l_h,
                            dtype=tl.float16)
        right_vec = tl.load(right_ptr + right_offset + offs_h * stride_r_h,
                            dtype=tl.float16)

        # accumulate in float32 to keep precision
        acc += (left_vec * right_vec).to(tl.float32)

        k += 1

    # store the result back (back to fp16)
    out_offset = pid_b * stride_o_b + i * stride_o_i + j * stride_o_j
    tl.store(out_ptr + out_offset + offs_h * stride_o_h,
             acc.to(tl.float16))


# -------------------------------------------------------------
# Entry point called by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input layer‑norm
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]    # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)                # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)              # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 3) Optional mask (mask is [B,N,N]; broadcast on last dim)
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid of linear projection)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel for the TriMul contraction
    # -----------------------------------------------------------------
    # Cast to FP16 for the kernel (reduces memory traffic)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)  # same shape [B,N,N,hidden_dim]

    # launch grid: one thread per (b,i,j) triplet
    grid = (bs, seq_len, seq_len)

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        seq_len,
        hidden_dim,
        # strides for left
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        # strides for right
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        # strides for out
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_H=hidden_dim,
        num_warps=4,
    )

    # back to FP32 for the following LayerNorm + linear
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output layer‑norm, gating and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]           # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                 # final shape [B,N,N,dim]

    return out
