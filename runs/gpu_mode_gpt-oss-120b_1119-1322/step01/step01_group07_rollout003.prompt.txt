```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
Triton‐accelerated outgoing TriMul (Triangular Multiplicative Update).

The module follows the reference PyTorch implementation from AlphaFold3:
1. Layer‑norm on the input pair representation.
2. Two linear projections → left/right.
3. Optional binary mask is applied to left/right.
4. Three gating linear layers (sigmoid) are computed from the normalized input.
5. left/right are multiplied elementwise with their gates.
6. The core “einsum”  out[i,j] = Σ_k left[i,k] * right[j,k]  is performed as a
   batched matrix multiplication C = A @ Bᵀ where A and B have shape
   (batch·hidden_dim, seq_len, seq_len).  This heavy part is implemented with a
   custom Triton kernel that tiles the matmul (BLOCK_M × BLOCK_N) and loops over
   K‑tiles (BLOCK_K).  Half‑precision (fp16) is used for the GEMM while the
   accumulator is fp32.
7. The result is reshaped back to [B, N, N, hidden_dim] (fp32).
8. A second LayerNorm, gating with out_gate, and a final linear projection
   back to the original channel dimension are applied.

The kernel is deliberately simple, using `tl.dot` for the inner product.
All other operations are expressed with PyTorch for simplicity and to keep
the code self‑contained.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A : (batch, M, K)   fp16
# B : (batch, N, K)   fp16   (note the K‑dimension is the last one, we load Bᵀ)
# C : (batch, M, N)   fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bn, stride_bk,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)              # tile of rows of A / C
    pid_n = tl.program_id(2)              # tile of cols of Bᵀ / C

    # ----- tile coordinates -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ----- base pointers for this batch -----
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    # ----- pointers to the current tile -----
    # A tile : (BLOCK_M, BLOCK_K)
    a_tile_ptr = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # B tile : we need Bᵀ → load as (BLOCK_K, BLOCK_N)
    b_tile_ptr = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ----- accumulation in fp32 -----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension in blocks -----
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_idx in range(0, num_k_blocks):
        # Load A block
        a = tl.load(
            a_tile_ptr + k_idx * BLOCK_K * stride_ak,
            mask=(offs_m[:, None] < M) & ((offs_k[None, :] + k_idx * BLOCK_K) < K),
            other=0.0,
        )
        # Load B block (as Bᵀ)
        b = tl.load(
            b_tile_ptr + k_idx * BLOCK_K * stride_bk,
            mask=((offs_k[:, None] + k_idx * BLOCK_K) < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # Inner product over K
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ----- write back result (fp16) -----
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_store)


def _batched_gemm_fp16(A: torch.Tensor,
                       B: torch.Tensor,
                       BLOCK_M: int = 128,
                       BLOCK_N: int = 128,
                       BLOCK_K: int = 32) -> torch.Tensor:
    """
    Wrapper for the Triton kernel.
    A: (batch, M, K)  fp16
    B: (batch, N, K)  fp16   (the GEMM computes A @ Bᵀ)
    Returns C: (batch, M, N) fp16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in number of elements (not bytes)
    a_stride_batch = A.stride(0)
    a_stride_m = A.stride(1)
    a_stride_k = A.stride(2)

    b_stride_batch = B.stride(0)
    b_stride_n = B.stride(1)
    b_stride_k = B.stride(2)

    c_stride_batch = C.stride(0)
    c_stride_m = C.stride(1)
    c_stride_n = C.stride(2)

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        a_ptr=A,
        b_ptr=B,
        c_ptr=C,
        batch_stride_a=a_stride_batch,
        batch_stride_b=b_stride_batch,
        batch_stride_c=c_stride_batch,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bn=b_stride_n,
        stride_bk=b_stride_k,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        M=M,
        N=N,
        K=K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   torch.float32
            mask         : [B, N, N]        torch.float32 or bool
            weights      : dict of model parameters (torch tensors)
            config       : dict with keys:
                            - "dim"          (int)
                            - "hidden_dim"   (int)
                            - "nomask"       (bool, optional)

    Returns:
        out : [B, N, N, dim]  torch.float32
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional binary mask (broadcast on the channel axis)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Cast to fp16 for the heavy multiplication
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape   # B = batch size, N = sequence length

    # --------------------------------------------------------------
    # 7. Reshape to (B*hidden_dim, N, N) and run batched GEMM
    #    out = left @ rightᵀ  (for each hidden channel)
    # --------------------------------------------------------------
    left_reshaped = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_fp16 = _batched_gemm_fp16(
        left_reshaped,
        right_reshaped,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )

    # --------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim] and promote to fp32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 9. Output LayerNorm
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 10. Apply out gate and final linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
