import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation (AlphaFold‑3) in a highly
optimized fashion.  The heavy compute – a batched matrix multiply that
sums over the sequence dimension – is off‑loaded to a Triton kernel.
All other steps (LayerNorm, linear projections, sigmoid gates, masking,
and the final output projection) are performed with dense PyTorch
operations.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the feature dimension.
2. Project `x` to a hidden dimension (`left_proj`, `right_proj`) and
   apply optional pairwise masking.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) and
   multiply them into the projected tensors.
4. The core operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched GEMM where for every hidden channel `d` we need
   `left[..., :, :] @ right[..., :, :].T`.  This is performed by a
   custom Triton kernel `batch_matmul_kernel`.
5. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate and finally project back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM of A @ Bᵀ   (A,B shape = [BH, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    A_ptr,               # const float*   [BH, M, K]
    B_ptr,               # const float*   [BH, N, K]   (will be read transposed)
    C_ptr,               #       float*   [BH, M, N]
    batch_head,          # int32: total number of (batch * hidden) matrices
    M, N, K,             # int32: matrix dimensions (M = N = seq_len)
    stride_ah, stride_am, stride_ak,    # int32: strides for A (batch, row, col)
    stride_bh, stride_bn, stride_bk,    # int32: strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,    # int32: strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs:
    #   axis 0 -> tile over output matrix (M x N)
    #   axis 1 -> each independent batch-head (batch * hidden)
    # ------------------------------------------------------------------
    pid = tl.program_id(axis=0)
    bh  = tl.program_id(axis=1)

    # ------------------------------------------------------------------
    # Block coordinates
    # ------------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid %  num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch‑head
    a_base = A_ptr + bh * stride_ah
    b_base = B_ptr + bh * stride_bh
    c_base = C_ptr + bh * stride_ch

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summed sequence dimension)
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile:  (BLOCK_M, BLOCK_K)   rows = offs_m, cols = offs_k
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ tile: (BLOCK_K, BLOCK_N)
        #   Bᵀ[k, n] = B[n, k]  → row index = k (original col), col = n (original row)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot‑product over the K dimension; broadcasting yields a (BLOCK_M, BLOCK_N) tile
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back C
    # ------------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Python helper that launches the Triton kernel
# ----------------------------------------------------------------------
def batch_matmul(A: torch.Tensor, B: torch.Tensor,
                 BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for A,B of shape [BH, N, N] using Triton.
    Returns C with the same shape and dtype as A (assumed float32).
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    BH, N, K = A.shape
    assert N == K, "Only square matrices are required for TriMul"

    # Output tensor
    C = torch.empty_like(A)

    # Grid definition: one program per (M‑tile, N‑tile) * BH
    grid_x = ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)
    grid = (grid_x, BH)   # (tiles, batch‑head)

    # Launch kernel
    batch_matmul_kernel[grid](
        A,
        B,
        C,
        BH,
        N, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # reasonable default for H100
    )
    return C

# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional masking (mask shape = [bs, N, N])
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)               # [bs, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating of the projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul – batched GEMM left @ rightᵀ over the sequence dim.
    #    We reshape to combine batch and hidden dimensions into a single
    #    leading dimension so that a single Triton launch can handle all
    #    (batch × hidden) matrix products.
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim] --> [B, hidden_dim, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # flatten batch*hidden -> BH
    A = left_perm.view(B * H, N, N)
    B_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiply (A @ Bᵀ)
    out_mat = batch_matmul(A, B_mat)  # shape [BH, N, N]

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
