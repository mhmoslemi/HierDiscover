    mask_n = offs_n < N

    b = b_ptr + batch_id * stride_bbatch
    c = c_ptr + batch_id * stride_cbatch

"""
custom_kernel – TriMul (outgoing) forward pass
================================================
The implementation follows the reference PyTorch module but moves the
computational bottleneck (the N×N batched matrix multiplication over
the hidden dimension) to a tiny Triton kernel.  The steps are

1. Layer‑Norm over the last dimension.
2. Four bias‑free linear projections (left/right and three gates) and
   element‑wise sigmoid on the gates.
3. Optional mask and element‑wise gating.
4. Cast the gated tensors to FP16.
5. Re‑order the tensors so the hidden dimension becomes part of the
   batch dimension:
        (B,N,N,H) → (B·H,N,N)   –  row‑major layout for both operands.
6. Batched GEMM C = A @ Bᵀ where A = left, B = right (already transposed
   by a cheap ``permute``).  The GEMM is performed by a small Triton kernel
   that works on FP16 inputs and accumulates in FP32 (Tensor‑Core friendly).
7. Reshape the result back to (B,N,N,H) and convert to FP32.
8. Layer‑Norm over the hidden dimension, multiply by the output gate,
   and finally project back to the original dimension with a bias‑free
   linear layer.

The kernel is deliberately simple (tiling M/N = 128, K = 32) which
fits comfortably in the H100 shared‑memory budget and exploits Tensor
Cores via ``tl.dot``.  For very small ``seq_len`` the kernel is still
faster than a naïve Python/Einsum implementation, while the rest of the
pipeline is handled by highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton batched GEMM kernel (row‑major A and B)
# ---------------------------------------------------------------------------
@triton.jit
def gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix sizes
    stride_am, stride_ak,                # A: (batch, M, K)
    stride_bk, stride_bn,                # B: (batch, K, N)
    stride_cm, stride_cn,                # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -----------------------------------------------------------------------
    # Block coordinates
    # -----------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------------
    # Pointers to the start of the current batch
    # -----------------------------------------------------------------------
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # -----------------------------------------------------------------------
    # Accumulator in FP32 (Tensor‑Core friendly)
    # -----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # Loop over the K dimension
    # -----------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K×N) – B is already transposed, so we read it as‑is
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate the tile product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------------
    # Write back the result (convert to FP16)
    # -----------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ---------------------------------------------------------------------------
# Main entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor[bs, N, N]        (bool/float, CUDA)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict, must contain "dim", "hidden_dim" and optional
                         "nomask" (bool, default True)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0️⃣ unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (float32)
    # -----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [bs, N, N, dim]  FP32

    # -----------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) – still in FP32
    # -----------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [bs, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # 3️⃣ Gating projections + sigmoid
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------------
    # 4️⃣ Optional mask + element‑wise gating
    # -----------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5️⃣ Cast to FP16 – the heavy GEMM works in half precision
    # -----------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # -----------------------------------------------------------------------
    # 6️⃣ Fold the hidden dimension into the batch dimension and
    #     prepare row‑major matrices for the GEMM.
    #
    #     left : (bs, N, N, H) → (bs, H, N, N) → (B·H, N, N)
    #     right_T = right transposed on the last two axes so that the gemm
    #               becomes a standard batched matmul.
    # -----------------------------------------------------------------------
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()   # (bs, H, N, N)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()  # (bs, H, N, N) = B_T

    batch_h = bs * hidden_dim
    left_flat  = left_perm.view(batch_h, N, N)   # (batch_h, M=N, K=N)
    right_flat = right_perm.view(batch_h, N, N)  # (batch_h, K=N, N)

    # -----------------------------------------------------------------------
    # 7️⃣ Allocate output tensor for the GEMM (FP16)
    # -----------------------------------------------------------------------
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 8️⃣ Triton GEMM launch
    # -----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides in *elements* (not bytes) – obtained from the contiguous tensors
    a_stride_batch, a_stride_m, a_stride_k = left_flat.stride()
    b_stride_batch, b_stride_k, b_stride_n = right_flat.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_flat.stride()

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps works well on H100
        num_stages=3,
    )

    # -----------------------------------------------------------------------
    # 9️⃣ Reshape back to (bs, N, N, hidden_dim) and convert to FP32
    # -----------------------------------------------------------------------
    out_h = out_flat.view(bs, hidden_dim, N, N)
    out_h = out_h.permute(0, 2, 3, 1).contiguous()   # [bs, N, N, hidden_dim]
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------------
    # 🔟 Post‑processing: LayerNorm over hidden_dim, out‑gate, final linear
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])   # (dim, hidden_dim) → [bs,N,N,dim]

    return out
