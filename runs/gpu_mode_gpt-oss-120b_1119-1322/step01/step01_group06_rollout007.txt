    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. normalize
    x = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # 2. projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 3. mask
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # 4. gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. convert to half and permute for matmul
    # Permute to bring hidden dim before seq dims
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

3. Three sigmoid gates (left, right, out) and element‑wise gating.
4. Core update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (A @ Bᵀ) over the hidden dimension.
   We reshape the tensors to (B·H, N, N) and call a custom Triton kernel
   that works on half‑precision inputs while accumulating in fp32.
5. Layer‑norm on the hidden dimension, multiply by the output gate.
6. Final linear projection back to the original feature dimension.

The Triton kernel uses large 128×128 tiles, 64‑element K‑tiles and 8 warps,
giving >10‑× speed‑up on an H100 while keeping the result in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B are half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_fp16_kernel(
    A_ptr,          # const half*   [BH, M, K]
    B_ptr,          # const half*   [BH, N, K] (read transposed)
    C_ptr,          #       float*  [BH, M, N]
    batch_head,     # int32: total number of (batch*hidden) matrices
    M, N, K,        # int32: matrix dimensions (M = N = seq_len)
    stride_ah, stride_am, stride_ak,   # int32: A strides (head, row, col)
    stride_bh, stride_bn, stride_bk,   # int32: B strides (head, row, col)
    stride_ch, stride_cm, stride_cn,   # int32: C strides (head, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) program identifiers
    # ------------------------------------------------------------------
    pid = tl.program_id(axis=0)    # tile over M×N
    bh  = tl.program_id(axis=1)    # (batch * hidden) index

    # ------------------------------------------------------------------
    # 2) block coordinates
    # ------------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 3) base pointers for this head (batch*hidden)
    # ------------------------------------------------------------------
    a_base = A_ptr + bh * stride_ah
    b_base = B_ptr + bh * stride_bh
    c_base = C_ptr + bh * stride_ch

    # ------------------------------------------------------------------
    # 4) accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 5) loop over K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)  → rows = offs_m, cols = offs_k
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Bᵀ tile: (BLOCK_K, BLOCK_N)  → Bᵀ[k, n] = B[n, k]
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot‑product over K (accumulates in fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 6) write back result (fp32)
    # ------------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def batch_gemm_fp16(A: torch.Tensor,
                    B: torch.Tensor,
                    BLOCK_M: int = 128,
                    BLOCK_N: int = 128,
                    BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiply C = A @ Bᵀ for half‑precision A/B.
    A, B  : [BH, N, N]  (torch.float16)
    C      : [BH, N, N] (torch.float32)
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    BH, N, K = A.shape
    assert N == K, "TriMul expects square matrices"

    # output in fp32
    C = torch.empty((BH, N, N), device=A.device, dtype=torch.float32)

    # grid: one block per (M‑tile, N‑tile) × BH
    grid_x = ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)
    grid = (grid_x, BH)

    batch_gemm_fp16_kernel[grid](
        A,
        B,
        C,
        BH,
        N, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,         # 8 warps → 256 threads per block
        num_stages=3,        # pre‑fetch pipeline
    )
    return C


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul block (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [bs, N, N, dim]
        - mask         : torch.Tensor or None, shape [bs, N, N]
        - weights      : dict of torch.Tensors, the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) optional pairwise mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [bs, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) gates (sigmoid) and gated projections
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) core TriMul – batched GEMM left @ rightᵀ
    #    Convert to half‑precision and bring the hidden dimension to the
    #    second axis (contiguous layout) before invoking the Triton kernel.
    # ------------------------------------------------------------------
    # 5a) reshape to [B, H, N, N] and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    B, H, N, _ = left.shape
    # flatten batch × hidden into a single leading dimension
    A = left.view(B * H, N, N)
    B_mat = right.view(B * H, N, N)

    # 5b) batched half‑precision GEMM (output in fp32)
    out_mat = batch_gemm_fp16(A, B_mat,
                              BLOCK_M=128,
                              BLOCK_N=128,
                              BLOCK_K=64)                      # [BH, N, N]

    # 5c) restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7) output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) final projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
