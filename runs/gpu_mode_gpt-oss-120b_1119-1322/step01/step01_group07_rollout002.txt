8️⃣ Compute the output gate (`out_gate`) from the original normalized input,
   apply it, and finish with a linear projection back to the original
   dimension D.

Only the heavy N×N matrix multiplication (step 6) is written in Triton;
all other operations stay in PyTorch.  The fused projection/gate reduces the
number of cuBLAS calls and improves bandwidth, while the kernel uses
tensor‑core‑friendly tile sizes (128×128×32) for the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (FP16 inputs, FP32 accumulator)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bn, stride_bk,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)        # which batch (B*H)
    pid_m = tl.program_id(1)            # row tile of A / C
    pid_n = tl.program_id(2)            # col tile of Bᵀ / C

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    # pointers to the tiles
    a_tile_ptr = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # load B as transposed: we need Bᵀ, so we read (k, n) order
    b_tile_ptr = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_idx in range(num_k_blocks):
        # global K offset
        cur_k = k_idx * BLOCK_K

        # masked loads
        a = tl.load(
            a_tile_ptr + cur_k * stride_ak,
            mask=(offs_m[:, None] < M) & ((offs_k[None, :] + cur_k) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_k * stride_bk,
            mask=((offs_k[:, None] + cur_k) < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write back result as FP16
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_store)


def _batched_gemm_fp16(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 32,
) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    A: (batch, M, K)   FP16
    B: (batch, N, K)   FP16   (kernel computes A @ Bᵀ)
    Returns C: (batch, M, N) FP16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides (in elements)
    a_batch_stride = A.stride(0)
    a_m_stride = A.stride(1)
    a_k_stride = A.stride(2)

    b_batch_stride = B.stride(0)
    b_n_stride = B.stride(1)
    b_k_stride = B.stride(2)

    c_batch_stride = C.stride(0)
    c_m_stride = C.stride(1)
    c_n_stride = C.stride(2)

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        a_ptr=A,
        b_ptr=B,
        c_ptr=C,
        batch_stride_a=a_batch_stride,
        batch_stride_b=b_batch_stride,
        batch_stride_c=c_batch_stride,
        stride_am=a_m_stride,
        stride_ak=a_k_stride,
        stride_bn=b_n_stride,
        stride_bk=b_k_stride,
        stride_cm=c_m_stride,
        stride_cn=c_n_stride,
        M=M,
        N=N,
        K=K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Public entry point expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, D]  (float32)
            mask         : torch.Tensor [B, N, N]    (bool or float)
            weights      : dict of model parameters (torch tensors)
            config       : dict with keys:
                           - "dim" (int)
                           - "hidden_dim" (int)
                           - "nomask" (bool, optional)
    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  FP32

    # --------------------------------------------------------------
    # 2. Fused projection + gating for the left branch
    #    (proj | gate) weight concatenation saves one cuBLAS call.
    # --------------------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )  # (2*H, D)
    left_combined = F.linear(x, left_combined_weight)  # [B, N, N, 2*H]

    left = left_combined[..., :hidden_dim]                     # proj
    left_gate_pre = left_combined[..., hidden_dim:]            # gate pre‑sigmoid
    left_gate = torch.sigmoid(left_gate_pre)                  # [B,N,N,H]

    # --------------------------------------------------------------
    # 3. Fused projection + gating for the right branch
    # --------------------------------------------------------------
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )  # (2*H, D)
    right_combined = F.linear(x, right_combined_weight)  # [B,N,N,2*H]

    right = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------------------------------------------------
    # 4. Optional mask (broadcast on channel dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Cast to FP16 for the heavy matmul
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape   # batch size, sequence length

    # --------------------------------------------------------------
    # 7. Reshape to (B*H, N, N) and run the Triton batched GEMM
    #    out = left @ rightᵀ  (for each hidden channel)
    # --------------------------------------------------------------
    # Permute to bring hidden dimension next to batch then collapse.
    left_reshaped = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Choose tile sizes based on N (simple heuristic)
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32

    out_fp16 = _batched_gemm_fp16(
        left_reshaped,
        right_reshaped,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B*H, N, N] FP16

    # --------------------------------------------------------------
    # 8. Reshape back to [B, N, N, H] and convert to FP32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 9. Output LayerNorm
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,H]

    # --------------------------------------------------------------
    # 10. Compute out_gate, apply it, and final linear projection back to D
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,D]

    return out
