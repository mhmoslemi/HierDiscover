"""
custom_kernel
==============

Implements the *outgoing* TriMul operation (AlphaFold‑3) in a fast
mixed‑precision fashion.

Algorithm
---------
1. Layer‑normalize the input (float32 → half) with the provided
   `norm.*` parameters.
2. Project the normalized tensor to the hidden dimension with two
   linear layers (`left_proj`, `right_proj`) – all work in FP16.
3. Optional pair‑wise mask is applied (broadcast over the hidden dim).
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`),
   also in FP16, and multiply them into the projected tensors.
5. Core operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM `A @ Bᵀ` where
   `A = left` and `B = right`.  This is carried out by a custom Triton
   kernel (`batch_matmul_kernel`) that operates on FP16 data and
   accumulates in FP32.
6. Second LayerNorm over the hidden dimension, output‑gate multiplication,
   and final linear projection back to `dim`.  The last two steps are
   performed in FP16 and the result is promoted to float32 for the
   caller.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ   (A,B: [BH, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    A_ptr,                     # const __fp16*  [BH, M, K]
    B_ptr,                     # const __fp16*  [BH, N, K]   (read transposed)
    C_ptr,                     #       __fp16*  [BH, M, N]
    BH, M, N, K,               # int32 (batch*head, rows, cols, shared dim)
    stride_ah, stride_am, stride_ak,   # strides for A
    stride_bh, stride_bn, stride_bk,   # strides for B (row‑major)
    stride_ch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)   # tile id over output matrix
    bh  = tl.program_id(axis=1)   # which (batch * hidden) matrix

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid %  num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this matrix
    a_base = A_ptr + bh * stride_ah
    b_base = B_ptr + bh * stride_bh
    c_base = C_ptr + bh * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K (the summed sequence dimension)
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile  (M × K)
        a_ptrs = a_base \
                 + (offs_m[:, None] * stride_am) \
                 + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load Bᵀ tile  (K × N)  -> Bᵀ[k, n] = B[n, k]
        b_ptrs = b_base \
                 + (offs_k[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result (down‑cast to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_base \
             + (offs_m[:, None] * stride_cm) \
             + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def batch_matmul(A: torch.Tensor,
                 B: torch.Tensor,
                 *,
                 BLOCK_M=128,
                 BLOCK_N=128,
                 BLOCK_K=64) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ where A,B have shape
    [BH, N, N] and dtype torch.half.  Returns C of the same shape/dtype.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    BH, N, K = A.shape
    assert N == K, "TriMul requires square matrices"

    C = torch.empty_like(A)

    # Grid: one program per (tile_m * tile_n) per batch-head
    grid_x = ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)
    grid = (grid_x, BH)

    batch_matmul_kernel[grid](
        A,
        B,
        C,
        BH,
        N,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # H100 is best with 8 warps per block
        num_stages=3,     # pipeline depth
    )
    return C

# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (or None)
        - weights      : dict of torch.Tensors containing all linear and
                         LayerNorm parameters (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast everything to FP16 for the heavy parts (weights are copied once)
    # ------------------------------------------------------------------
    dtype = torch.float16
    x = x.to(dtype)

    # Linear weights (no bias)
    w_left_proj   = weights["left_proj.weight"].to(dtype)
    w_right_proj  = weights["right_proj.weight"].to(dtype)
    w_left_gate   = weights["left_gate.weight"].to(dtype)
    w_right_gate  = weights["right_gate.weight"].to(dtype)
    w_out_gate    = weights["out_gate.weight"].to(dtype)
    w_to_out_norm = weights["to_out_norm.weight"].to(dtype)
    b_to_out_norm = weights["to_out_norm.bias"].to(dtype)
    w_to_out      = weights["to_out.weight"].to(dtype)

    # LayerNorm parameters (keep FP32 for numerical stability; PyTorch
    # will up‑cast them automatically when the input is FP16)
    ln_weight = weights["norm.weight"]
    ln_bias   = weights["norm.bias"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=ln_weight, bias=ln_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dim (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x, w_left_proj)          # [B,N,N,hidden]
    right = F.linear(x, w_right_proj)         # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → [B,N,N,1] → FP16
        mask_f = mask.to(dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – all FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x, w_out_gate))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul: batched GEMM left @ rightᵀ over the sequence dim
    #    Transform tensors to shape [B*hidden, N, N] (contiguous)
    # ------------------------------------------------------------------
    #   [B,N,N,hidden] → [B,hidden,N,N] → view -> [BH,N,N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)
    Bmat = right_perm.view(B * H, N, N)

    # Batched GEMM (FP16 inputs, FP32 accumulation)
    out_mat = batch_matmul(A, Bmat)   # [BH, N, N] FP16

    # Reshape back to original layout: [B,N,N,hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden,),
                       weight=w_to_out_norm,
                       bias=b_to_out_norm)

    # ------------------------------------------------------------------
    # 7) Output gating (still FP16)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to `dim` (FP16) and cast to FP32
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # [B,N,N,dim] in FP16
    out = out.to(torch.float32)

    return out
