@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_h,  # hidden stride
    stride_bm, stride_bk, stride_bn,
    stride_c_m, stride_c_n, stride_c_k,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_m = tl.program_id(2)  # combined batch idx

    # compute actual batch (b) and hidden (h)
    b_idx = pid_m // hidden_dim
    h_idx = pid_m % hidden_dim

    # compute offsets for i and j tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointer offsets
    # pointer for A: a_ptr + b_idx * stride_am + h_idx * stride_h
    a_base = a_ptr + b_idx * stride_am + h_idx * stride_h
    # pointer for B
    b_base = b_ptr + b_idx * stride_bm + h_idx * stride_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # fp16->fp32
    # store result
    c_ptrs = c_ptr + pid_m * stride_c_m + offs_i[:, None] * stride_c_n + offs_j[None, :] * stride_c_k

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel( # arguments
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_h,        # hidden stride
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_m = tl.program_id(2)

    # derive batch and hidden indexes
    batch_id = pid_m // hidden_dim
    hidden_id = pid_m - batch_id * hidden_dim   # pid_m % hidden_dim

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_id * stride_am + hidden_id * stride_h
    b_base = b_ptr + batch_id * stride_bm + hidden_id * stride_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # fp16*fp16 -> fp32

    # store result
    c_ptrs = c_ptr + pid_m * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_ck

```python
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul.

    The heavy N³ work is performed by a custom batched GEMM kernel that
    computes, for each hidden channel,
        out[i,j] = Σ_k left[i,k] * right[j,k]
    with FP16 matrix multiplication and FP32 accumulation.
    All other operations (layer‑norms, linear projections, gating) are
    done with native torch kernels.
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for GEMM ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ---------- Prepare dimensions ----------
    M = B * hidden_dim       # number of independent matrix multiplications
    K = N                    # sum dimension

    # Strides for left (A)
    stride_am, stride_an, stride_ak, stride_h = left_fp16.stride()
    # Strides for right (B)
    stride_bm, stride_bk, stride_bn, _ = right_fp16.stride()

    # Output tensor (FP32 accumulator)
    C = torch.empty((M, N, N), dtype=torch.float32, device=device)

    stride_cm, stride_cn, stride_ck = C.stride()

    # Tile size selection (empirically good for all test sizes)
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32  # fits nicely into TensorCores

    # Grid launch
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M,
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        C,
        M,
        N,
        K,
        stride_am,
        stride_an,
        stride_ak,
        stride_h,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,hidden)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # ---------- Output gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The N³ work is a batched matrix multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h]
for every hidden channel *h*.  We treat (batch * hidden) as an
independent matrix‑batch and compute each product with a custom
Triton kernel that uses FP16 matrix multiplication (Tensor Cores) and
FP32 accumulation for numerical stability.  All surrounding operations
(LayerNorm, linear projections, gating) are performed with native
torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    a_ptr,                     # left  tensor (FP16)
    b_ptr,                     # right tensor (FP16)
    c_ptr,                     # output (FP32)
    M, N, K,                   # shapes: M = B*H, N = seq_len, K = seq_len
    stride_am, stride_an, stride_ak,     # A strides: (batch, i, k)
    stride_h,                                # stride of hidden axis (==1 for our layout)
    stride_bm, stride_bk, stride_bn,     # B strides: (batch, k, j)
    stride_cm, stride_cn, stride_ck,     # C strides: (batch, i, j)
    hidden_dim,                # hidden dimension (runtime int)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM C[m,i,j] = Σ_k A[m,i,k] * B[m,k,j]"""

    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_m = tl.program_id(2)          # combined (batch * hidden)

    # Derive batch index and hidden index from the combined dimension
    batch_id  = pid_m // hidden_dim
    hidden_id = pid_m - batch_id * hidden_dim   # == pid_m % hidden_dim

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the specific (batch, hidden) slice
    a_base = a_ptr + batch_id * stride_am + hidden_id * stride_h
    b_base = b_ptr + batch_id * stride_bm + hidden_id * stride_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile  (i,k)
        a_ptrs = a_base + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile  (k,j)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Tensor‑core dot product (FP16 × FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # Store the computed tile
    c_ptrs = c_ptr + pid_m * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor   : torch.Tensor (bool/float) shape [B, N, N] or None
        * weights_dict  : mapping from string keys to torch.Tensor weights
        * config_dict   : contains ``dim``, ``hidden_dim`` and a ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (skip if nomask=True)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)        # (B,N,N,1)
        left  = left  * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoids) – element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy N³ work
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Free the FP32 intermediates (optional, reduces peak memory)
    del left, right

    # ------------------------------------------------------------------
    # 6. Prepare batched GEMM parameters
    # ------------------------------------------------------------------
    M = B * hidden_dim               # number of independent matrices
    K = N

    # Strides of the original layout (B,N,N,hidden) – contiguous in H
    stride_am, stride_an, stride_ak, stride_h = left_fp16.stride()   # (B,N,N,H)
    stride_bm, stride_bk, stride_bn, _ = right_fp16.stride()

    # Output accumulator (FP32) in contiguous layout (M,N,N)
    C = torch.empty((M, N, N), dtype=torch.float32, device=device)
    stride_cm, stride_cn, stride_ck = C.stride()

    # ------------------------------------------------------------------
    # 7. Tile size selection – works well for all test configurations
    # ------------------------------------------------------------------
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 64   # larger K‑tile reduces the loop count

    # Grid: (tiles_i, tiles_j, batch‑hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M,
    )

    # ------------------------------------------------------------------
    # 8. Launch the Triton kernel
    # ------------------------------------------------------------------
    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        C,
        M,
        N,
        K,
        stride_am,
        stride_an,
        stride_ak,
        stride_h,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 10. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
