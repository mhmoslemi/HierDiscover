"""
TriMul (outgoing) – highly‑optimised forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the channel dimension of the 4‑D tensor.
2️⃣  Linear projections → *left_proj*, *right_proj* (no bias).
3️⃣  Linear gates → *left_gate*, *right_gate*, *out_gate* (sigmoid).
4️⃣  Fuse element‑wise mask (if any), gating and FP16 casting with a
    Triton kernel.  The kernel writes two contiguous FP16 buffers:
        *L*:   left_proj * left_gate * mask   (shape M×N×N)
        *R*:   right_proj * right_gate * mask (shape M×N×N)
    where M = batch × hidden_dim.
5️⃣  Batched matrix multiplication on the FP16 buffers using
    `torch.bmm`.  This is a cuBLAS call that fully exploits Tensor‑Cores.
6️⃣  Reshape the result back to [B,N,N,hidden], convert to FP32,
    apply a second LayerNorm, multiply by *out_gate* and finally a
    linear projection back to the original channel dimension.
The Triton kernel is the only custom GPU kernel and handles the heavy
memory‑bound pre‑processing; the actual O(N³) work is performed by the
highly‑optimised cuBLAS batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def pack_buffers_kernel(
    # input tensors
    left_proj_ptr, left_gate_ptr, mask_ptr,
    right_proj_ptr, right_gate_ptr,
    # output buffers (FP16)
    left_out_ptr, right_out_ptr,
    # run‑time constants
    B, N, H,
    # strides of the 4‑D inputs
    lp_b, lp_i, lp_k, lp_h,
    lg_b, lg_i, lg_k, lg_h,
    rp_b, rp_i, rp_k, rp_h,
    rg_b, rg_i, rg_k, rg_h,
    # strides of the mask (if used)
    m_b, m_i, m_k,
    # compile‑time constants
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    """Pack the projected tensors + gates (and optional mask) into FP16
    contiguous buffers ready for a batched GEMM.

    The logical batch id = pid = batch_idx * H + hidden_idx.
    """
    pid = tl.program_id(2)                     # 0 … (B*H‑1)
    batch_idx = pid // H
    hidden_idx = pid % H

    # Base pointers for this (batch,hidden) slice
    lp_base = left_proj_ptr + batch_idx * lp_b + hidden_idx * lp_h
    lg_base = left_gate_ptr + batch_idx * lg_b + hidden_idx * lg_h
    rp_base = right_proj_ptr + batch_idx * rp_b + hidden_idx * rp_h
    rg_base = right_gate_ptr + batch_idx * rg_b + hidden_idx * rg_h
    if USE_MASK:
        m_base = mask_ptr + batch_idx * m_b
    # --------------------------------------------------------------------
    # Tile coordinates in the N‑by‑N plane
    # --------------------------------------------------------------------
    pid_i = tl.program_id(0)                   # row tile
    pid_k = tl.program_id(1)                   # column tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_i = offs_i < N
    mask_k = offs_k < N
    # --------------------------------------------------------------------
    # Load left_proj, left_gate (and mask) → apply gating → cast to fp16
    # --------------------------------------------------------------------
    a_ptrs = lp_base + offs_i[:, None] * lp_i + offs_k[None, :] * lp_k
    g_ptrs = lg_base + offs_i[:, None] * lg_i + offs_k[None, :] * lg_k
    a = tl.load(a_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0)
    g = tl.load(g_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0)
    if USE_MASK:
        mask_ptrs = m_base + offs_i[:, None] * m_i + offs_k[None, :] * m_k
        m = tl.load(mask_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        a = a * m
    a = a * g                                 # apply gate
    a = a.to(tl.float16)                      # FP16 for the GEMM

    # --------------------------------------------------------------------
    # Store left buffer (contiguous M×N×N)
    # --------------------------------------------------------------------
    out_offset = pid * (N * N) + offs_i[:, None] * N + offs_k[None, :]
    tl.store(left_out_ptr + out_offset, a,
             mask=mask_i[:, None] & mask_k[None, :])

    # --------------------------------------------------------------------
    # Load right_proj, right_gate (and mask) → apply gating → cast to fp16
    # --------------------------------------------------------------------
    b_ptrs = rp_base + offs_i[:, None] * rp_i + offs_k[None, :] * rp_k
    h_ptrs = rg_base + offs_i[:, None] * rg_i + offs_k[None, :] * rg_k
    b = tl.load(b_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0)
    h = tl.load(h_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0)
    if USE_MASK:
        b = b * m
    b = b * h                                 # apply gate
    b = b.to(tl.float16)

    # --------------------------------------------------------------------
    # Store right buffer (contiguous M×N×N)
    # --------------------------------------------------------------------
    tl.store(right_out_ptr + out_offset, b,
             mask=mask_i[:, None] & mask_k[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor (bool/float) shape [B, N, N] or None
        * weights_dict : mapping from string keys to torch.Tensor weights
        * config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # ------------------------------------------------------------------
    # Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]  FP32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if use_mask:
        mask = mask.to(x.dtype)               # [B,N,N]  FP32

    # ------------------------------------------------------------------
    # Allocate FP16 buffers for the batched GEMM
    # ------------------------------------------------------------------
    M = B * hidden_dim                            # total independent matrices
    left_buf  = torch.empty((M, N, N), dtype=torch.float16, device=x.device)
    right_buf = torch.empty((M, N, N), dtype=torch.float16, device=x.device)

    # ------------------------------------------------------------------
    # Strides (all tensors are contiguous in memory after the F.linear calls)
    # ------------------------------------------------------------------
    lp_b, lp_i, lp_k, lp_h = left_proj.stride()
    lg_b, lg_i, lg_k, lg_h = left_gate.stride()
    rp_b, rp_i, rp_k, rp_h = right_proj.stride()
    rg_b, rg_i, rg_k, rg_h = right_gate.stride()
    if use_mask:
        m_b, m_i, m_k = mask.stride()
    else:
        # Dummy values – they will never be used because USE_MASK=0
        m_b = m_i = m_k = 0

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_I = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,   # tiles along i
        (N + BLOCK_K - 1) // BLOCK_K,   # tiles along k
        M,                              # one program per (batch, hidden)
    )

    # ------------------------------------------------------------------
    # Pack the buffers (fuse mask, gating and cast to FP16)
    # ------------------------------------------------------------------
    pack_buffers_kernel[grid](
        left_proj, left_gate, mask if use_mask else torch.empty(0, device=x.device),
        right_proj, right_gate,
        left_buf, right_buf,
        B, N, hidden_dim,
        lp_b, lp_i, lp_k, lp_h,
        lg_b, lg_i, lg_k, lg_h,
        rp_b, rp_i, rp_k, rp_h,
        rg_b, rg_i, rg_k, rg_h,
        m_b, m_i, m_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        USE_MASK=1 if use_mask else 0,
    )

    # ------------------------------------------------------------------
    # Batched GEMM: out = left_buf @ right_bufᵀ   (FP16 inputs → FP16 out)
    # ------------------------------------------------------------------
    # Note: torch.bmm will use cuBLAS and Tensor‑Core acceleration.
    out_fp16 = torch.bmm(left_buf, right_buf.transpose(1, 2))

    # ------------------------------------------------------------------
    # Reshape back to [B,N,N,hidden] (now FP16) and cast to FP32
    # ------------------------------------------------------------------
    out = (
        out_fp16
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
