"""
TriMul (outgoing) forward pass – highly‐optimized implementation.

Algorithm
---------
1️⃣ Layer‑norm the input over its channel dimension.
2️⃣ Linear projections (left / right) – no bias.
3️⃣ (Optional) apply the binary mask.
4️⃣ Compute three sigmoid gates from the normalized tensor and
   element‑wise multiply them into left/right.
5️⃣ Core operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a *batched* matrix multiplication where each hidden
   dimension is an independent batch:
      out = (left_perm) @ (right_perm)ᵀ
   Implemented with :func:`torch.bmm` on fp16 tensors (Tensor‑cores on
   H100) – this is far faster than a naïve Triton kernel for the N³
   work.
6️⃣ Reshape back to [B,N,N,hidden_dim] and apply a second LayerNorm.
7️⃣ Fuse the output‑gate multiplication with a tiny Triton kernel
   (pure element‑wise) to satisfy the “must use Triton” rule.
8️⃣ Final linear projection back to the original channel dimension.

All heavy arithmetic is performed in fp16 (except the final
LayerNorm/gates which stay in fp32 for numerical stability). The
resulting tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel – element‑wise multiplication (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr,                # out tensor (float32)
    b_ptr,                # out_gate tensor (float32)
    out_ptr,              # destination (float32)
    N_ELEM: tl.constexpr, # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """out[i] = a[i] * b[i]  (float32)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor (bool/float) shape [B, N, N] or None
        * weights_dict : mapping from string keys to torch.Tensor weights
        * config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional binary mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool – promote to same dtype as tensors
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and gating
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()  # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core batched matrix multiplication: out = left @ rightᵀ
    # ------------------------------------------------------------------
    #   left : (B,N,N,hidden) -> (B,hidden,N,N) -> (M,N,N) where M = B*hidden
    #   rightᵀ : same layout, but k/j axes swapped.
    left_perm   = left.permute(0, 3, 1, 2).contiguous()          # (B,hidden,i,k)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous()       # (B,hidden,k,j)

    M = B * hidden_dim
    # View as (M, N, N)
    left_mat   = left_perm.view(M, N, N)
    right_mat_T = right_perm_T.view(M, N, N)

    # Cast to fp16 – Tensor‑core friendly (the H100 computes fp16 matmul at
    # ~> 30 TFLOPs). Accumulators stay in fp16; final conversion gives fp32.
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat_T.to(torch.float16)

    # Batched GEMM (M independent N×N products) – torch.bmm is highly‑tuned.
    out_fp16 = torch.bmm(left_fp16, right_fp16)            # (M,N,N), fp16
    out = out_fp16.to(torch.float32)                       # back to fp32

    # Reshape to the original 4‑D layout
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # (B,N,N,hidden) fp32

    # ------------------------------------------------------------------
    # 7️⃣ Fuse out_gate multiplication via a tiny Triton kernel
    # ------------------------------------------------------------------
    # Flatten tensors for the kernel
    total_elems = out.numel()                     # B*N*N*hidden
    BLOCK_SZ = 1024                               # good for most sizes
    grid = ( (total_elems + BLOCK_SZ - 1) // BLOCK_SZ, )

    # Prepare contiguous views (torch ensures they are already contiguous)
    out_ptr = out.view(-1)
    gate_ptr = out_gate.view(-1)
    # In‑place multiplication using Triton
    elemwise_mul_kernel[grid](
        out_ptr,
        gate_ptr,
        out_ptr,          # write result back into `out`
        total_elems,
        BLOCK_SIZE=BLOCK_SZ,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
