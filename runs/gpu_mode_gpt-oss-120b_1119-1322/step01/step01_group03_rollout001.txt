"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x`  (shape [B,N,N,D]).
2. Linear projections (`left_proj`, `right_proj`) and three gating
   linear layers (`left_gate`, `right_gate`, `out_gate`).  All
   linear layers are bias‑free.
3. Optional mask (shape [B,N,N]) is broadcast and applied to the
   projected tensors before gating.
4. Apply the sigmoid gates.
5. Compute the “tri‑multiplicative” update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   for every hidden channel `h`.  This is a batched matrix‑multiply
   A @ Bᵀ.  The heavy‑weight part is performed by a custom Triton
   kernel that works in **float16** (halving memory traffic) while
   accumulating in **float32** for numerical stability.
6. Hidden‑dimension LayerNorm, multiply by `out_gate` and project back
   to the original dimension with `to_out`.

The Triton kernel processes all `(batch * hidden_dim)` matrices in a
single launch, using a 2‑level tiling (BLOCK_M×BLOCK_N tiles with
BLOCK_K‑depth) that maps naturally to H100 Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_fp16_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers to left, right, output
    M, N, K,                        # matrix dimensions
    stride_ab, stride_am, stride_ak,  # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,  # B: batch, col (k), row (n) strides
    stride_cb, stride_cm, stride_cn,  # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent (batch*head) matrices.

    A, B are stored as FP16; accumulation is in FP32.
    """
    pid_batch = tl.program_id(2)          # which (batch * hidden) matrix
    pid_m = tl.program_id(0)              # block row of C
    pid_n = tl.program_id(1)              # block col of C

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this batch element
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # Pointers to the start of the (i,j) tile, k = 0
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        # --------------------------------------------------------------
        # Load A tile (FP16) – mask out-of‑bounds with zeros
        # --------------------------------------------------------------
        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=tl.float16(0.0),
        )
        # --------------------------------------------------------------
        # Load Bᵀ tile (FP16)
        # --------------------------------------------------------------
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )
        # --------------------------------------------------------------
        # Partial dot product – result is FP16, cast to FP32 before accu.
        # --------------------------------------------------------------
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back result (FP32) with proper masking
    # --------------------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask:         torch.Tensor [B, N, N] (bool/int/float) or None
            - weights:      dict of model parameters (torch.Tensor)
            - config:       dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2) Linear projections & gates (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3) Mask (broadcast) – if mask is a tensor, otherwise skip
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Apply gates (still float32)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # batch, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # 5) Cast to FP16 and launch Triton kernel for the N³ update
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, H, N, N) -> flatten first two dims -> (B*H, N, N)
    left_t = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right_fp16.permute(0, 3, 1, 2).contiguous()

    left_flat = left_t.view(-1, N, N)   # [B*H, N, N]  (FP16)
    right_flat = right_t.view(-1, N, N)

    total = left_flat.shape[0]          # B * H

    out_flat = torch.empty((total, N, N), dtype=torch.float32, device=device)

    # Tuned block sizes for H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M
        triton.cdiv(N, BLOCK_N),   # blocks along N
        total,                     # one program per (batch * hidden) matrix
    )

    batched_gemm_out_fp16_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_ab=left_flat.stride(0),   # batch stride
        stride_am=left_flat.stride(1),   # row stride (M)
        stride_ak=left_flat.stride(2),   # col stride (K)
        stride_bb=right_flat.stride(0),  # batch stride
        stride_bk=right_flat.stride(2),  # col stride for B (will be used as K)
        stride_bn=right_flat.stride(1),  # row stride for B (will be used as N)
        stride_cb=out_flat.stride(0),    # batch stride
        stride_cm=out_flat.stride(1),    # row stride (M)
        stride_cn=out_flat.stride(2),    # col stride (N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7) Hidden‑dim LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
