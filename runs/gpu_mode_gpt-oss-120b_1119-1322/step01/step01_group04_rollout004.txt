```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base ptr for this batch
    a_batch = a_ptr + batch_id * batch_stride_a
    b_batch = b_ptr + batch_id * batch_stride_b
    c_batch = c_ptr + batch_id * batch_stride_c

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load A tile
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (B already transposed)
        b = tl.load(
            b_batch + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product, accumulate in fp16 (fast)
        acc += tl.dot(a, b, out_dtype=tl.float16)
    # write back
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # load weights
    ...
    # LN
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Project
    left = F.linear(x, left_proj_weight)   # shape B,N,N,H
    right = F.linear(x, right_proj_weight)

    # mask if present
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute and reshape for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] R^T

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    # allocate output
    out_flat = torch.empty((batch_dim, N, N), dtype=torch.float16, device=input_tensor.device)

    # calculate strides
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bn = right_flat.stride(2)

        BLOCK_N = 128

def tri_mul_gemm_kernel(
    a_ptr,            # left   [batch, M, K]  (fp16)
    b_ptr,            # right  [batch, K, N]  (fp16, already transposed)
    c_ptr,            # out    [batch, M, N] (fp16)
    batch_stride_a,   # stride to next batch in A (elements)
    stride_am, stride_ak,
    batch_stride_b,
    stride_bk, stride_bn,
    batch_stride_c,
    stride_cm, stride_cn,
    M, N, K,          # all equal to sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B for many independent batches (FP16)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_base = a_ptr + batch_id * batch_stride_a
    b_base = b_ptr + batch_id * batch_stride_b
    c_base = c_ptr + batch_id * batch_stride_c

    # Accumulator in FP16 (fast on H100)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_base + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile [BLOCK_K, BLOCK_N] (B already transposed)
        b = tl.load(
            b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot product for the tile
        acc += tl.dot(a, b, out_dtype=tl.float16)

    # ------------------------------------------------------------------
    # Write the result tile
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entrypoint called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C]  (float32)
        - mask         : torch.Tensor[B, N, N]    (float32/bool) or None
        - weights      : dict of torch.Tensor model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Model parameters (all stored as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm   (over channel dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden]

    # Apply the element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Tri‑multiplicative contraction (batched GEMM)
    #    left : [B,N,N,hidden] → [B, hidden, N, N] (i,k)
    #    right: [B,N,N,hidden] → [B, hidden, N, N] (j,k) → transposed to (k,j)
    # ------------------------------------------------------------------
    # Cast to FP16 before the heavy GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute so that hidden becomes the second axis (batch‑wise GEMM)
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, hidden, i, k]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, k, j] (already transposed)

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    # Collapse batch and hidden dimensions
    left_flat = left_perm.view(batch_dim, N, N)   # [BH, M=i, K=k]
    right_flat = right_perm.view(batch_dim, N, N) # [BH, K=k, N=j]

    # Output buffer (FP16, will be cast back later)
    out_flat = torch.empty((batch_dim, N, N), dtype=torch.float16, device=device)

    # Strides in element counts (required by Triton)
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # over M (i)
    stride_ak = left_flat.stride(2)   # over K (k)

    batch_stride_b = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # over K (k)
    stride_bn = right_flat.stride(2)  # over N (j)

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # over M (i)
    stride_cn = out_flat.stride(2)    # over N (j)

    # Launch grid: one program per (batch_dim, tile_m, tile_n)
    grid = (
        batch_dim,
        (N + 128 - 1) // 128,  # M‑tiles (upper bound, kernel will mask tails)
        (N + 128 - 1) // 128,  # N‑tiles
    )

    # Triton will select the best autotuned configuration.
    tri_mul_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=batch_stride_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=batch_stride_b,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        batch_stride_c=batch_stride_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        # NOTE: the following compile‑time arguments are overridden by
        # the autotune configs; we keep a dummy default to keep the
        # signature complete.
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )

    # Reshape back to the original layout and cast to float32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]
    out = out.float()

    # ------------------------------------------------------------------
    # 6) Final post‑processing
    # ------------------------------------------------------------------
    # hidden‑dim layer‑norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Apply the output gate (cast gate to the same dtype for the mul)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to original channel dimension
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
