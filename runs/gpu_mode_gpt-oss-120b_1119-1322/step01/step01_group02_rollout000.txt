"""
custom_kernel – Triton‑accelerated forward pass for the **outgoing** TriMul
operator (AlphaFold‑3 style).

Algorithm
---------
1.  Layer‑norm over the last feature dimension (dim).
2.  Linear projections (no bias) for left/right tensors and three gating
    projections → sigmoid.
3.  Fuse gates and optional mask:   left  = left_proj  * left_gate  * mask,
                                     right = right_proj * right_gate * mask.
4.  Core cubic contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM A @ Bᵀ for every hidden channel *h*.  We treat the
   hidden dimension as part of the batch (batch = B·H) and launch a single
   Triton kernel that performs many independent GEMMs.  Input tensors are
   cast to bfloat16, the dot product uses Tensor‑Core (bf16) and the
   accumulation is performed in float32.
5.  Layer‑norm over the hidden dimension, multiply by the out‑gate,
    final linear projection back to the original feature dimension.

The kernel implements the O(N³·H) contraction; all other operations use
high‑performance PyTorch primitives.  The implementation works for
any batch size ≤2, sequence length up to 1024, and both masked / unmasked
inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: (M,K)   strides (row,col,batch)
    stride_bk, stride_bn, stride_bb,   # B: (K,N)   strides (col,row,batch)
    stride_cm, stride_cn, stride_cb,   # C: (M,N)   strides (row,col,batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel: for each batch idx compute
           C = A @ B   where A: M×K, B: K×N   (B is stored transposed)
       The kernel is launched with a 3‑D grid:
           dim0 – batch (B·H)
           dim1 – tile‑row (M / BLOCK_M)
           dim2 – tile‑col (N / BLOCK_N)
    """
    batch_id = tl.program_id(0)          # index over B·H
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile col

    # ----- tile start offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- base pointers for this batch --------------------------------------
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K in tiles ------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # load A tile (M×K)  – dtype bf16 or fp16, promoted to FP32 on dot
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load B tile as (K×N) (transposed view of original B)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # FP32 accumulation

    # ----- write back ---------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype float32).
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension
    hidden_dim = config["hidden_dim"]  # projection dimension (H)

    # --------------------------------------------------------------------- #
    # 1️⃣  LayerNorm on input feature dimension
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 2️⃣  Linear projections (no bias) and gating values
    # --------------------------------------------------------------------- #
    # Linear projections to hidden_dim (weight shapes: hidden_dim x dim)
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Gate projections → sigmoid, shape same as left/right
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------------- #
    # 3️⃣  Apply optional mask and fuse gates
    # --------------------------------------------------------------------- #
    if mask_tensor is not None and mask_tensor.numel() > 0:
        # mask [B,N,N] → broadcast to hidden dim
        mask_exp = mask_tensor.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left  = left_proj  * mask_exp * left_gate
        right = right_proj * mask_exp * right_gate
    else:
        left  = left_proj  * left_gate
        right = right_proj * right_gate

    # Cast to bfloat16 for the heavy contraction (memory‑bandwidth friendly)
    left  = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    # --------------------------------------------------------------------- #
    # 4️⃣  Core N³ contraction via batched GEMM kernel
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape                # H == hidden_dim
    # Rearrange so hidden dimension becomes part of the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * H, N, N)     # (BH, N, N)
    right_mat = right_perm.view(B * H, N, N)    # (BH, N, N)

    # Output accumulator (FP32)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ----- strides for contiguous (BH,N,N) tensors -------------------------
    # A (left_mat) strides
    stride_a_batch = left_mat.stride(0)      # batch stride = N * N
    stride_am = left_mat.stride(1)           # row stride (M) = N
    stride_ak = left_mat.stride(2)           # col stride (K) = 1

    # B (right_mat) strides – we treat B as transposed (K,N)
    stride_b_batch = right_mat.stride(0)     # batch stride = N * N
    stride_bk = right_mat.stride(2)          # col‑stride for K dim = 1
    stride_bn = right_mat.stride(1)          # row‑stride for N dim = N

    # C (out_mat) strides
    stride_c_batch = out_mat.stride(0)       # batch stride = N * N
    stride_cm = out_mat.stride(1)            # row stride (M) = N
    stride_cn = out_mat.stride(2)            # col stride (N) = 1

    # ----- tile sizes (tuned for H100, adapt to N) -------------------------
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32

    # Grid launch: (batch, tiles_M, tiles_N)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Triton kernel launch
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 performs well with 8 warps
    )

    # Reshape back to original [B, N, N, H] layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------------- #
    # 5️⃣  Second LayerNorm, out‑gate, final projection back to dim
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure final dtype is float32 (as required)
    return out.float()
