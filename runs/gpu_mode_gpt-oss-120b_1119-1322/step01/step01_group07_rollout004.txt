"""
Triton‑accelerated outgoing TriMul (Triangular Multiplicative Update).

The implementation follows the reference AlphaFold3 forward pass:

1. Layer‑Norm over the input pair representation (dim‑wise).
2. Two linear projections → left/right (dim → hidden_dim).
3. A single Triton kernel fuses:
   * optional binary mask (broadcast on the channel dimension),
   * element‑wise gating with sigmoid‑gates (left_gate, right_gate).
   The kernel works on flattened tensors so the mask is applied only once
   per (b,i,j) location and the result is written back in‑place.
4. The heavy “einsum” is realised as a batched matrix‑multiply
   `out = left @ rightᵀ`.  By reshaping to (B*hidden_dim, N, N) we can
   rely on cuBLAS (`torch.bmm`) which runs on Tensor‑Cores and gives
   near‑peak FP16 throughput.
5. A second Layer‑Norm over the hidden dimension, out‑gate gating,
   and a final linear projection back to the original dim.

Only the mask‑gate fusion is performed in Triton – all other operations
are expressed with PyTorch for simplicity and to keep the kernel launch
overhead minimal.  The kernel processes the tensors in FP16; the rest of
the model works in FP32, matching the reference implementation.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into left/right.
# The tensors are flattened as 1‑D contiguous buffers.
#   left  : [B, N, N, H]   (fp16)
#   right : [B, N, N, H]   (fp16)
#   l_gate: [B, N, N, H]   (fp16)
#   r_gate: [B, N, N, H]   (fp16)
#   mask  : [B, N, N]      (fp16) – broadcasted over H
#   out   : left/right are overwritten in‑place.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, lgate_ptr, rgate_ptr, mask_ptr,
    total_len,            # total number of fp16 elements in left/right (B*N*N*H)
    hidden_dim,           # H (compile‑time constant for better codegen)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)               # [BLOCK_SIZE]

    # ------------------------------------------------------------------
    # mask_idx = element_idx // H  (because mask is broadcast on channel)
    # ------------------------------------------------------------------
    mask_idx = offs // hidden_dim

    # Load mask (float16).  For out‑of‑bounds we return 1.0 → no effect.
    mask = tl.load(
        mask_ptr + mask_idx,
        mask=mask_idx < (total_len // hidden_dim),
        other=1.0,
    )

    # Load left/right and their gates; out‑of‑bounds values are zero (won't be used)
    l = tl.load(left_ptr + offs, mask=offs < total_len, other=0.0)
    lg = tl.load(lgate_ptr + offs, mask=offs < total_len, other=0.0)
    r = tl.load(right_ptr + offs, mask=offs < total_len, other=0.0)
    rg = tl.load(rgate_ptr + offs, mask=offs < total_len, other=0.0)

    # Fuse mask and gates
    l_out = l * lg * mask
    r_out = r * rg * mask

    # Write back in‑place
    tl.store(left_ptr + offs, l_out, mask=offs < total_len)
    tl.store(right_ptr + offs, r_out, mask=offs < total_len)


def _apply_mask_and_gate(left, right, l_gate, r_gate, mask, hidden_dim):
    """
    Apply `mask * left_gate` and `mask * right_gate` in‑place using Triton.
    All tensors must be contiguous FP16.  `mask` is a 1‑D FP16 tensor of
    shape (B*N*N,) – broadcasted over the channel dimension.
    """
    total_len = left.numel()                     # B * N * N * H
    # Choose a block size that fits the hardware (1024 is the max threads per block on H100)
    BLOCK = 1024
    num_blocks = math.ceil(total_len / BLOCK)

    _mask_gate_kernel[(num_blocks,)](
        left.data_ptr(),
        right.data_ptr(),
        l_gate.data_ptr(),
        r_gate.data_ptr(),
        mask.data_ptr(),
        total_len,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # No return – operation is in‑place


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N]       (float32 or bool)
            weights      : dict of model parameters (torch tensors)
            config       : dict with keys:
                            - "dim"        (int)
                            - "hidden_dim" (int)
                            - "nomask"     (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (float32 → fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values (float32) – sigmoid(linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # used later, stays float32

    # ------------------------------------------------------------------
    # 4. Convert to fp16 for the heavy computation
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    lgate_fp16 = left_gate.to(torch.float16).contiguous()
    rgate_fp16 = right_gate.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 5. Prepare mask (broadcasted over hidden_dim) – fp16 1‑D buffer
    # ------------------------------------------------------------------
    B, N, _, _ = left_fp16.shape
    if nomask:
        mask_fp16 = torch.ones(B * N * N, dtype=torch.float16, device=left_fp16.device)
    else:
        # mask may be bool or float; ensure float16 and flatten
        mask_fp16 = mask.to(dtype=torch.float16).contiguous().view(-1)

    # ------------------------------------------------------------------
    # 6. Fuse mask + gates into left/right (in‑place) via Triton
    # ------------------------------------------------------------------
    _apply_mask_and_gate(left_fp16, right_fp16, lgate_fp16, rgate_fp16, mask_fp16, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Batched matrix multiplication: out = left @ rightᵀ
    #    We reshape to (B*H, N, N) and rely on cuBLAS (torch.bmm) – fast on H100.
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched gemm (FP16 inputs, FP32 accumulation is automatic on Tensor Cores)
    out_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim] and promote to FP32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 10. Apply out_gate (float32) and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
