"""
custom_kernel – Highly‑optimized TriMul (outgoing) forward pass

Algorithm
---------
1.  Layer‑norm over the last dimension of the input.
2.  Four bias‑free linear projections (left/right) and three gating
    projections (left_gate, right_gate, out_gate) are computed with
    torch.nn.functional.linear (FP32) and the gates are passed through
    a sigmoid.
3.  Optional binary mask is applied (broadcast over the hidden dim)
    and the three gates are fused element‑wise into the projected tensors.
4.  The heavy triangular update
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix multiplication: for each hidden channel an
    (N×N) @ (N×N)ᵀ product is required.
    The hidden dimension is folded into the batch dimension
        batch_h = bs * hidden_dim
    and a custom Triton kernel performs the batched GEMM in FP16
    (accumulating in FP32).  The kernel is optimised for H100 with
    BLOCK_M=N_BLOCK=128, BLOCK_N=128, BLOCK_K=64 and 8 warps.
5.  The GEMM result is reshaped back to (bs, N, N, hidden_dim) and
    converted to FP32.
6.  Post‑processing: Layer‑norm over the hidden dimension, multiplication
    by the out_gate, and a final bias‑free linear projection back to the
    original dimension.

Only the compute‑dominant GEMM is written in Triton; all light‑weight
operations (layer‑norm, linear projections, sigmoid, gating) are done
with highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ
#   A : (batch_h, M, K) , B : (batch_h, N, K) (stored as N×K)
#   C : (batch_h, M, N)
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A strides (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B strides (batch, N, K) – note B is stored as (N,K)
    stride_cm, stride_cn, stride_cb,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch pointers
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # mask out‑of‑bounds tiles
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (tensor‑cores still use FP16 inputs)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----------- Load A tile (M × K) -------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----------- Load Bᵀ tile (K × N) -------------
        # B is stored as (N, K). We need Bᵀ[k, n] = B[n, k].
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ----------- Accumulate -----------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ----------- Write back (convert to FP16) ----------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------------------
# Main entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]  (float32)
        - mask         : torch.Tensor[bs, N, N]       (bool or float), may be ignored
        - weights      : dict of tensors (model parameters)
        - config       : dict containing at least "dim" and "hidden_dim"
                         and optional "nomask" (bool)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -------------------------------------------------------------------
    # 0. unpack
    # -------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (FP32)
    # -------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – keep FP32 for now
    # -------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left  = F.linear(x, left_proj_w)   # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # -------------------------------------------------------------------
    # 3. Gating vectors (sigmoid)
    # -------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -------------------------------------------------------------------
    # 4. Apply mask (if any) and element‑wise gates
    # -------------------------------------------------------------------
    if not nomask:
        # mask broadcasted on hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [bs, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5. Cast to half‑precision for the heavy GEMM
    # -------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # -------------------------------------------------------------------
    # 6. Fold hidden_dim into the batch dimension (bs*hidden_dim, N, N)
    # -------------------------------------------------------------------
    # permute from (bs, N, N, hidden_dim) -> (bs, hidden_dim, N, N)
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim

    left_flat  = left_h.view(batch_h, N, N)   # (batch_h, M=N, K=N)
    right_flat = right_h.view(batch_h, N, N)  # (batch_h, N, K=N)

    # output buffer for GEMM
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # -------------------------------------------------------------------
    # 7. Launch Triton batched GEMM kernel
    # -------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # fits H100 tensor‑core tiles

    # strides in elements
    a_stride_batch, a_stride_m, a_stride_k = left_flat.stride()
    b_stride_batch, b_stride_n, b_stride_k = right_flat.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_flat.stride()

    grid = (
        batch_h,                                    # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,               # M tiles
        (N + BLOCK_N - 1) // BLOCK_N,               # N tiles
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_ab=a_stride_batch,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_bb=b_stride_batch,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_cb=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # empirical value for H100
    )

    # -------------------------------------------------------------------
    # 8. Reshape back to (bs, N, N, hidden_dim) and FP32
    # -------------------------------------------------------------------
    out_h = out_flat.view(bs, hidden_dim, N, N)          # (bs, hidden, N, N)
    out_h = out_h.permute(0, 2, 3, 1).contiguous()      # (bs, N, N, hidden)
    out = out_h.to(torch.float32)                       # FP32 for the remaining ops

    # -------------------------------------------------------------------
    # 9. Post‑processing: LayerNorm over hidden_dim, out‑gate, final proj
    # -------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # out_gate was computed in FP32 already
    out = out * out_gate

    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # bias‑free final projection

    return out
