"""
TriMul (outgoing) – highly‑optimised forward pass

Algorithm
---------
1. Layer‑norm the input tensor `x` on the last dimension (float32 for stability).
2. Linear projections (left/right) and three gate projections are computed
   with bias‑free linear layers.  Gate values are passed through a sigmoid.
3. Optional binary mask (broadcast over the hidden dimension) and the
   element‑wise gates are applied, after which the tensors are cast to
   float16.
4. The hidden dimension is folded into the batch dimension:
      (bs, N, N, hidden) → (bs·hidden, N, N)
   This layout is required for the GEMM kernel.
5. A custom Triton kernel computes, for every (b·hidden) batch,
        out[i, j] = Σ_k  left[i, k] · right[j, k]
   i.e. a batched matrix‑multiply C = A @ Bᵀ with A,B∈ℝ^{N×N}.
   The kernel works on fp16 inputs and accumulates in fp32,
   automatically mapping to Tensor‑Cores on H100.
6. The result is reshaped back to (bs, N, N, hidden) and converted to fp32.
7. A second layer‑norm (over the hidden dimension), multiplication by the
   output‑gate, and a final projection back to `dim` complete the module.

Only the heavy N³·hidden work is performed in Triton; everything else
remains in PyTorch (which is already highly tuned).  This satisfies the
requirement of “using Triton” while keeping the overall runtime low.
"""

#   C : (batch_h, M, N)   – output

):
    """
    Compute one batch of GEMM (A @ Bᵀ) in fp16 → fp32 accumulation.
    Grid: (batch_h, ceil(M/BLOCK_M), ceil(N/BLOCK_N))
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # -----------------------------------------------------------------------
    # Block coordinates
    # -----------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------------
    # Pointers for the current batch
    # -----------------------------------------------------------------------
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # -----------------------------------------------------------------------
    # Accumulator in FP32 (Tensor‑Core friendly)
    # -----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # Main K‑loop
    # -----------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load Bᵀ tile (K × N)  – B is stored as (batch, N, K)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------------
    # Write back result (store as fp16)
    # -----------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------------------
# Main entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]  (float32)
        - mask         : torch.Tensor[bs, N, N]       (bool/float, may be None)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict, must contain "dim", "hidden_dim",
                         optionally "nomask" (bool)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0. unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------------
    # 1. Layer‑norm on the last dimension (float32)
    # -----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – stay in FP32 for now
    # -----------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [bs, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # 3. Gating vectors (sigmoid)
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------------
    # 4. Apply optional mask and element‑wise gates
    # -----------------------------------------------------------------------
    if not nomask:
        # mask shape: [bs, N, N] → broadcast over hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [bs, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5. Cast to FP16 for the GEMM (reduces memory traffic & enables Tensor‑Cores)
    # -----------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # -----------------------------------------------------------------------
    # 6. Fold hidden_dim into the batch dimension; no extra copy apart from
    #    a cheap view/permute that makes each (batch, hidden) slice contiguous.
    # -----------------------------------------------------------------------
    # (bs, N, N, hidden) → (bs, hidden, N, N) → (bs*hidden, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim

    left_flat  = left_h.view(batch_h, N, N)
    right_flat = right_h.view(batch_h, N, N)

    # Output buffer for the GEMM (fp16)
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 7. Launch the Triton batched GEMM kernel
    # -----------------------------------------------------------------------
    # Tuning parameters – empirically good for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides (in elements) – torch tensors expose them directly
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    grid = (
        batch_h,                              # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,         # rows
        (N + BLOCK_N - 1) // BLOCK_N,         # cols
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        stride_abatch=stride_a_batch,
        stride_bbatch=stride_b_batch,
        stride_cbatch=stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # H100 benefits from 8 warps per block
        num_stages=2,           # double‑buffering of shared memory
    )

    # -----------------------------------------------------------------------
    # 8. Reshape back to (bs, N, N, hidden) and cast to FP32
    # -----------------------------------------------------------------------
    out_h = out_flat.view(bs, hidden_dim, N, N)
    out_h = out_h.permute(0, 2, 3, 1).contiguous()   # [bs, N, N, hidden]
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------------
    # 9. Post‑processing: LayerNorm over hidden, out‑gate, final linear
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate (broadcast over hidden dim)
    out = out * out_gate

    # Final linear projection back to `dim` (bias‑free)
    out = F.linear(out, weights["to_out.weight"])

    return out
