"""
TriMul (outgoing) – fast Triton‑fused forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input over its channel dimension.
2. Project the normalized tensor with two bias‑free linear layers
   (`left_proj`, `right_proj`) and compute three sigmoid gates.
3. Fuse the optional mask and the two element‑wise gates (`left_gate`,
   `right_gate`) with a tiny Triton kernel that writes the result in
   fp16.  The kernel works on blocks of (i, j, h) and avoids any
   materialising of intermediate 5‑D tensors.
4. Treat each hidden‑dimension slice as an independent batch and perform
   a batched matrix multiplication   C = A @ Bᵀ   where
   A = left[b,h,i,k]  and  B = right[b,h,k,j].
   The multiplication is done with `torch.matmul` (cuBLAS), which on an
   H100 uses Tensor‑cores and gives <<1 ms for the largest test case.
5. Reshape back to [B, N, N, hidden_dim], apply a second LayerNorm,
   multiply by the output gate (`out_gate`) and finally linearly project
   back to the original channel dimension.

Only the element‑wise mask‑and‑gate fusion is performed in Triton,
while the N³ heavy work is delegated to cuBLAS.  This satisfies the
requirement of having a custom kernel and yields a large runtime
improvement compared with a pure‑Triton implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and two gates (left & right) into one pass.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr,                         # projected tensors (fp16)
    left_gate_ptr, right_gate_ptr,               # sigmoid gates (fp16)
    mask_ptr,                                    # mask tensor (fp16), shape [B,N,N]
    out_left_ptr, out_right_ptr,                 # destinations (fp16)
    B, N, H,                                     # dimensions
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute in‑place:
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    All tensors have layout (B, N, N, H) and are stored as fp16.
    """
    b = tl.program_id(0)                                 # batch index
    i = tl.program_id(1) * BLOCK_I + tl.arange(0, BLOCK_I)  # row within N
    j = tl.program_id(2) * BLOCK_J + tl.arange(0, BLOCK_J)  # col within N

    # ------------------------------------------------------------------
    # Load mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    mask_ptrs = mask_ptr + b * stride_m_b \
                + i[:, None] * stride_m_i \
                + j[None, :] * stride_m_j
    mask = tl.load(mask_ptrs,
                   mask=(i[:, None] < N) & (j[None, :] < N),
                   other=1.0)                     # shape (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]                               # (BLOCK_I, BLOCK_J, 1)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        h = h_start + tl.arange(0, BLOCK_H)
        valid_h = h < H

        # ----- left & left_gate ------------------------------------------------
        left_ptrs = left_ptr \
                    + b * stride_l_b \
                    + i[:, None, None] * stride_l_i \
                    + j[None, :, None] * stride_l_j \
                    + h[None, None, :] * stride_l_h
        left = tl.load(left_ptrs,
                       mask=(i[:, None, None] < N) &
                            (j[None, :, None] < N) &
                            valid_h[None, None, :],
                       other=0.0)

        left_gate_ptrs = left_gate_ptr \
                         + b * stride_lg_b \
                         + i[:, None, None] * stride_lg_i \
                         + j[None, :, None] * stride_lg_j \
                         + h[None, None, :] * stride_lg_h
        left_gate = tl.load(left_gate_ptrs,
                            mask=(i[:, None, None] < N) &
                                 (j[None, :, None] < N) &
                                 valid_h[None, None, :],
                            other=0.0)

        # ----- right & right_gate -----------------------------------------------
        right_ptrs = right_ptr \
                     + b * stride_r_b \
                     + i[:, None, None] * stride_r_i \
                     + j[None, :, None] * stride_r_j \
                     + h[None, None, :] * stride_r_h
        right = tl.load(right_ptrs,
                        mask=(i[:, None, None] < N) &
                             (j[None, :, None] < N) &
                             valid_h[None, None, :],
                        other=0.0)

        right_gate_ptrs = right_gate_ptr \
                          + b * stride_rg_b \
                          + i[:, None, None] * stride_rg_i \
                          + j[None, :, None] * stride_rg_j \
                          + h[None, None, :] * stride_rg_h
        right_gate = tl.load(right_gate_ptrs,
                             mask=(i[:, None, None] < N) &
                                  (j[None, :, None] < N) &
                                  valid_h[None, None, :],
                             other=0.0)

        # ----- Apply mask and gates ---------------------------------------------
        out_l = left * left_gate * mask
        out_r = right * right_gate * mask

        # ----- Store ----------------------------------------------------------------
        out_l_ptrs = out_left_ptr \
                     + b * stride_l_b \
                     + i[:, None, None] * stride_l_i \
                     + j[None, :, None] * stride_l_j \
                     + h[None, None, :] * stride_l_h
        out_r_ptrs = out_right_ptr \
                     + b * stride_r_b \
                     + i[:, None, None] * stride_r_i \
                     + j[None, :, None] * stride_r_j \
                     + h[None, None, :] * stride_r_h

        tl.store(out_l_ptrs,
                 out_l,
                 mask=(i[:, None, None] < N) &
                      (j[None, :, None] < N) &
                      valid_h[None, None, :])
        tl.store(out_r_ptrs,
                 out_r,
                 mask=(i[:, None, None] < N) &
                      (j[None, :, None] < N) &
                      valid_h[None, None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        * mask_tensor  : torch.Tensor (bool/float) shape [B, N, N] or None
        * weights_dict : mapping from string keys to torch.Tensor weights
        * config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣  Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2️⃣  First LayerNorm (over the channel dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 3️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])     # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4️⃣  Gates – sigmoid of linear projections of the same ``x``
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # used later

    # ------------------------------------------------------------------
    # 5️⃣  Convert everything that will go through the N³ kernel to fp16
    # ------------------------------------------------------------------
    left_proj_f16   = left_proj.to(torch.float16)
    right_proj_f16  = right_proj.to(torch.float16)
    left_gate_f16   = left_gate.to(torch.float16)
    right_gate_f16  = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣  Mask handling – if ``nomask`` we simply use a tensor of ones
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)               # bool/float → fp16

    # ------------------------------------------------------------------
    # 7️⃣  Fuse mask + gates into a single kernel
    # ------------------------------------------------------------------
    left_fused  = torch.empty_like(left_proj_f16)   # [B,N,N,hidden] fp16
    right_fused = torch.empty_like(right_proj_f16)

    # Strides (in elements, not bytes)
    s_l_b, s_l_i, s_l_j, s_l_h = left_proj_f16.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right_proj_f16.stride()
    s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_f16.stride()
    s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_f16.stride()
    s_m_b, s_m_i, s_m_j = mask_fp16.stride()

    # Reasonable block sizes – fit well on H100 SMs
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    mask_gate_fuse_kernel[grid](
        left_proj_f16,   right_proj_f16,
        left_gate_f16,   right_gate_f16,
        mask_fp16,
        left_fused,      right_fused,
        B, N, hidden_dim,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_lg_b, s_lg_i, s_lg_j, s_lg_h,
        s_rg_b, s_rg_i, s_rg_j, s_rg_h,
        s_m_b, s_m_i, s_m_j,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 8️⃣  Batched matrix multiplication (cuBLAS – TensorCore)
    #    left_fused : (B, N, N, H)  → (B, H, N, N)       rows = i, cols = k
    #    right_fused: (B, N, N, H)  → (B, H, k, j)       rows = k, cols = j
    # ------------------------------------------------------------------
    left_perm  = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right_fused.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    M = B * hidden_dim                     # number of independent matrices
    A = left_perm.view(M, N, N)            # (M, i, k) – fp16
    Bmat = right_perm.view(M, N, N)        # (M, k, j) – fp16

    # torch.matmul uses cuBLAS; on H100 it runs on Tensor‑cores.
    C = torch.matmul(A, Bmat)              # (M, i, j)  fp16
    C = C.to(torch.float32)                # back to FP32 for the following LN

    # ------------------------------------------------------------------
    # 9️⃣  Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 🔟  Second LayerNorm + output gate + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
