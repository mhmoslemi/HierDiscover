```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
custom_kernel – TriMul (outgoing) forward pass

Algorithm
---------
1. Layer‑norm the input tensor `x` on the last dimension.
2. Linear projections (left/right) and three gating projections are computed
   with bias‑free linear layers.  Gating vectors are passed through a sigmoid.
3. Apply optional mask and element‑wise gates:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (mask is broadcast over the hidden dimension)
4. Fuse the hidden dimension into a batch dimension
        batch_h = bs * hidden_dim
   Reshape `left` / `right` to (batch_h, N, N) such that each
   hidden‑channel slice becomes an independent matrix.
5. Compute the core "tri‑multiplicative" update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched GEMM:
        out_batch = left_batch @ right_batchᵀ
   A custom Triton kernel (`batched_gemm_kernel`) performs this
   half‑precision matmul (accumulating in FP32) for all batches.
6. Reshape the result back to (bs, N, N, hidden_dim) and convert to FP32.
7. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and finally project back to the original dimension with `to_out`.

The heavy O(N³·hidden) work is performed on the GPU by the Triton kernel,
while light‑weight linear / norm / sigmoid work is done with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ  where
#   A : (batch_h, M, K)   – left  tensor   (M=N, K=N)
#   B : (batch_h, N, K)   – right tensor   (rows=j, cols=k)
#   C : (batch_h, M, N)   – output tensor (i,j)
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, N, K)   – we read Bᵀ by swapping indices
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # Starting coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A tile (M×K) ----
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load Bᵀ tile (K×N) ----
        # B is stored as (batch, N, K); we need element Bᵀ[k, n] = B[batch, n, k].
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ---- Accumulate ----
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ---- Write back result (convert to FP16) ----
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------------------
# Main entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict, must contain "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # 0. unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (float32 for numerical stability)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [bs, N, N, dim]  FP32

    # -----------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – keep in FP32 for now
    # -----------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left  = F.linear(x, left_proj_w)   # [bs, N, N, hidden_dim]  FP32
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------------
    # 3. Gating vectors (sigmoid, FP32)
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # same shape
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # used after the GEMM

    # -----------------------------------------------------------------------
    # 4. Optional mask and gate application
    # -----------------------------------------------------------------------
    # mask is either a full‑ones tensor (nomask) or a random 0/1 mask.
    if not config.get("nomask", True):
        # bring mask to same dtype, broadcast over hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [bs, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5. Cast to half‑precision for the heavy GEMM
    # -----------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # -----------------------------------------------------------------------
    # 6. Fold hidden_dim into the batch dimension and prepare tensors for Triton
    # -----------------------------------------------------------------------
    # shape currently: (bs, N, N, hidden_dim)
    # permute to (bs, hidden_dim, N, N) → (bs*hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim

    left_flat  = left_h.view(batch_h, N, N)
    right_flat = right_h.view(batch_h, N, N)

    # Allocate output tensor for GEMM (half‑precision)
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 7. Launch Triton kernel
    # -----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # strides (in elements) for the three tensors
    a_stride_batch, a_stride_m, a_stride_k = left_flat.stride()
    b_stride_batch, b_stride_n, b_stride_k = right_flat.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_flat.stride()

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # heuristic value for H100
    )

    # -----------------------------------------------------------------------
    # 8. Reshape back to (bs, N, N, hidden_dim) and convert to FP32
    # -----------------------------------------------------------------------
    out_h = out_flat.view(bs, hidden_dim, N, N)
    out_h = out_h.permute(0, 2, 3, 1).contiguous()   # [bs, N, N, hidden_dim]
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------------
    # 9. Post‑processing: LayerNorm over hidden_dim, out‑gate, final linear
    # -----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out_gate_f = out_gate.to(torch.float32)
    out = out * out_gate_f

    to_out_w = weights["to_out.weight"]          # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # final projection, bias‑free

    return out
```
