```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
TriMul (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor X of shape [B, N, N, C].
2. Apply linear projections (no bias) to obtain
   left and right tensors of shape [B, N, N, H] where H = hidden_dim.
3. (Optional) mask the projected tensors.
4. Compute three sigmoided gates from X and multiply them element‑wise:
      left  = left  * left_gate   (and mask)
      right = right * right_gate  (and mask)
5. Compute the costly contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of H independent matrix‑multiplications per batch.
   We fuse it into a custom Triton kernel for speed.
6. Layer‑norm the result on the hidden dimension, apply the out‑gate,
   and a final linear projection back to the original channel dimension C.

The Triton kernel fuses the N³ inner product for every hidden channel,
processing many (batch × hidden) GEMMs simultaneously.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_tri_mul_kernel(
    a_ptr,                # ptr to left tensor   [B*H, M, K]  (M=N, K=N)
    b_ptr,                # ptr to right tensor  [B*H, K, N]  (K=N, N=N)   (right is already transposed)
    c_ptr,                # ptr to output tensor [B*H, M, N]
    batch_stride_a,       # stride to the next batch in A
    stride_am, stride_ak, # strides for M and K in A
    batch_stride_b,       # stride to the next batch in B
    stride_bk, stride_bn, # strides for K and N in B
    batch_stride_c,       # stride to the next batch in C
    stride_cm, stride_cn, # strides for M and N in C
    M, N, K,              # dimensions (all equal to seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B for many independent batches.
    A : [batch, M, K]  (fp16)
    B : [batch, K, N]  (fp16)
    C : [batch, M, N]  (fp32 accumulator, stored as fp32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Compute offsets for the current block (i.e., tile)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks for M and N dimensions
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * batch_stride_a
    b_base = b_ptr + batch_id * batch_stride_b
    c_base = c_ptr + batch_id * batch_stride_c

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        # Global k indices for this tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Build pointer offsets
        a_ptrs = a_base + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        b_ptrs = b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Masks for the K dimension (handle tail tiles)
        mask_k = k_offset < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Compute the dot product for this tile; accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing version) using a fused Triton kernel
    for the N³ contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, C] (float32)
            mask:        torch.Tensor [B, N, N] or None (float32/bool)
            weights:     dict of model parameters (float32)
            config:      dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (unsqueeze to broadcast)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. TriMul contraction via Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute so that hidden dimension becomes the second axis.
    # For the right tensor we also transpose the last two axes so that
    # the K dimension aligns with the left tensor's K.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (M=i, K=k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (K=k, N=j)

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    # Collapse batch and hidden into a single GEMM batch dimension
    left_flat = left_perm.view(batch_dim, N, N)   # [BH, M, K]
    right_flat = right_perm.view(batch_dim, N, N) # [BH, K, N]

    # Allocate output (fp32)
    out_flat = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # stride over M dimension (i)
    stride_ak = left_flat.stride(2)   # stride over K dimension (k)

    batch_stride_b = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # stride over K dimension (k)
    stride_bn = right_flat.stride(2)  # stride over N dimension (j)

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # stride over M (i)
    stride_cn = out_flat.stride(2)    # stride over N (j)

    # Triton block sizes – tuned for H100 fp16 matmul
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_gemm_tri_mul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=batch_stride_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=batch_stride_b,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        batch_stride_c=batch_stride_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # H100 has many cores; 4 warps per block works well
    )

    # Reshape back to [B, H, N, N] and then to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Final post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                     # gated output
    out = F.linear(out, to_out_weight)       # final projection back to dim

    return out
```
