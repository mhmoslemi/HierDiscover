import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing triangle multiplicative update (TriMul) forward pass.
    Implements the AlphaFold‑3 “outgoing” TriMul using fused
    linear projections (project+gate), a single einsum for the core
    O(N³·H) multiplication, and a tiny Triton kernel for the
    element‑wise out‑gate multiplication.
    All heavy compute is in FP16, final output is FP32.
    """
    # Unpack data
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input layer‑norm over channel dim (C = dim)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias
    )   # [B,N,N,dim] float32

    # -------------------------------------------------
    # 2. Cast to fp16 for linear / matmul work
    # -------------------------------------------------
    x_h = x.to(torch.float16).view(-1, dim)   # (B*N*N, dim)

    # -------------------------------------------------
    # 3. Combined linear projections (proj + gate)
    # -------------------------------------------------
    # left:  proj + left_gate
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]],
        dim=0
    ).to(torch.float16)   # (2*H, dim)
    left_concat = torch.nn.functional.linear(x_h, left_concat_w) # (B*N*N, 2*H)
    left_proj_raw = left_concat[:, :hidden_dim]
    left_gate_raw = left_concat[:, hidden_dim:]

    # right: proj + right_gate
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]],
        dim=0
    ).to(torch.float16)   # (2*H, dim)
    right_concat = torch.nn.functional.linear(x_h, right_concat_w)
    right_proj_raw = right_concat[:, :hidden_dim]
    right_gate_raw = right_concat[:, hidden_dim:]

    # out_gate (separate)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    out_gate_raw = torch.nn.functional.linear(x_h, out_gate_w)   # (B*N*N, H)

    # -------------------------------------------------
    # 4. Reshape to [B,N,N,H] and apply mask (if any)
    # -------------------------------------------------
    left_proj = left_proj_raw.view(bs, N, N, hidden_dim)
    right_proj = right_proj_raw.view(bs, N, N, hidden_dim)

    if not nomask:
        # mask shape [B,N,N], unsqueeze to broadcast over H
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -------------------------------------------------
    # 5. Sigmoid gates and element‑wise gating
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    out_gate = torch.sigmoid(out_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6. Core triangle multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    performed as a batched fp16 matmul over the hidden dim.
    # -------------------------------------------------
    # Bring hidden dim to batch axis for a single torch.matmul call.
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_perm = right.permute(0, 3, 1, 2) # (B, H, N, K)

    # matmul treats the last two dims as matrices.
    # we need (i,k) · (j,k) -> sum over k, i.e., left @ rightᵀ
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N)

    # reshape back to original order
    out = out.permute(0, 2, 3, 1)   # (B, N, N, H)    fp16

    # -------------------------------------------------
    # 7. Output‑side LayerNorm (hidden_dim)
    # -------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )   # (B,N,N,H)  fp32

    # -------------------------------------------------
    # 8. Element‑wise multiplication with out_gate (Triton)
    # -------------------------------------------------
    out_norm_flat = out_norm.contiguous().reshape(-1)            # fp32 1‑D
    out_gate_flat = out_gate.to(torch.float32).contiguous().reshape(-1)  # fp32 1‑D
    gated_flat = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK_SIZE = 65536
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE
    )
    gated = gated_flat.view(bs, N, N, hidden_dim)   # fp32

    # -------------------------------------------------
    # 9. Final linear projection back to original dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    output = torch.nn.functional.linear(gated, to_out_w)   # (B,N,N,dim)

    return output

# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # float32* (out_norm flattened)
    b_ptr,          # float32* (out_gate flattened)
    out_ptr,        # float32* (output flattened)
    size,           # total number of elements
    BLOCK_SIZE: tl.constexpr,
):

    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    b = tl.load(b_ptr + offs, mask=mask)

    Outgoing Triangle‑Multiplicative Update (TriMul) forward pass.

    The implementation follows the AlphaFold‑3 TriMul module but fuses
    several linear layers to minimise kernel launches and memory traffic.
    Heavy work is performed in FP16, the final result is returned in FP32.

    Steps
    -----

    2️⃣ Fuse ``proj`` + ``gate`` linear transforms for the left and right

       performed as a batched FP16 matmul over the hidden dimension.

    7️⃣ Element‑wise multiplication with the output gate (via Triton).
    8️⃣ Final linear projection back to the original `dim`.
    """

    # -------------------------------------------------

    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ Input LayerNorm over channel dimension
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # -------------------------------------------------
    x_h = x.to(torch.float16).view(-1, dim)  # (B·N·N, dim)

    # -------------------------------------------------
    # 3️⃣ Fused linear projections (proj + gate) for left/right
    # -------------------------------------------------
    # left: [proj | gate] → (2·H, dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    left_concat = torch.nn.functional.linear(x_h, left_concat_w)  # (B·N·N, 2·H)
    left_proj_raw = left_concat[:, :hidden_dim]
    left_gate_raw = left_concat[:, hidden_dim:]

    # right: [proj | gate] → (2·H, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat = torch.nn.functional.linear(x_h, right_concat_w)
    right_proj_raw = right_concat[:, :hidden_dim]
    right_gate_raw = right_concat[:, hidden_dim:]

    # out‑gate (separate, shape H)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_raw = torch.nn.functional.linear(x_h, out_gate_w)  # (B·N·N, H)

    # -------------------------------------------------
    # 4️⃣ Reshape back to [B, N, N, H] and apply optional mask
    # -------------------------------------------------
    left_proj = left_proj_raw.view(bs, N, N, hidden_dim)
    right_proj = right_proj_raw.view(bs, N, N, hidden_dim)

    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -------------------------------------------------
    # 5️⃣ Sigmoid gates and element‑wise gating
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)
    out_gate = torch.sigmoid(out_gate_raw).view(bs, N, N, hidden_dim).to(torch.float16)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Core triangle multiplication (batched GEMM over hidden dim)
    # -------------------------------------------------
    # Bring hidden dimension to a batch axis: (B, H, N, K)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_perm = right.permute(0, 3, 1, 2) # (B, H, N, K)

    # Perform (N×K) @ (N×K)ᵀ for each (B,H) pair
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N)

    # Back to original ordering
    out = out.permute(0, 2, 3, 1)   # (B, N, N, H)   fp16

    # -------------------------------------------------
    # 7️⃣ Output‑side LayerNorm (hidden_dim) – fp32
    # -------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B, N, N, H)  fp32

    # -------------------------------------------------
    # 8️⃣ Element‑wise multiplication with out_gate (Triton)
    # -------------------------------------------------
    out_norm_flat = out_norm.contiguous().view(-1)          # fp32 1‑D
    out_gate_flat = out_gate.to(torch.float32).contiguous().view(-1)  # fp32 1‑D
    gated_flat = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK_SIZE = 65536
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(bs, N, N, hidden_dim)   # (B, N, N, H)  fp32

    # -------------------------------------------------
    # 9️⃣ Final linear projection back to original dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"]                # (dim, hidden_dim)
    output = torch.nn.functional.linear(gated, to_out_w)   # (B, N, N, dim)

    return output
```
