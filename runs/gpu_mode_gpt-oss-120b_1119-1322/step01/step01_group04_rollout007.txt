def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,  # not used maybe
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * batch_stride_a
    b_base = b_ptr + batch_id * batch_stride_b
    c_base = c_ptr + batch_id * batch_stride_c

    # Split M dimension in half
    offs_m_1 = offs_m
    offs_m_2 = offs_m + BLOCK_M // 2

    # Actually if BLOCK_M divisible by 2.

    a_ptrs_1 = a_base + offs_m_1[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    a_ptrs_2 = a_base + offs_m_2[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # p for B
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulators
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):

        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

```
@triton.jit
def batch_gemm_tri_mul_kernel(
    a_ptr,
    b_ptr,
    c_ptr,
    batch_stride_a,
    hidden_stride_a,
    stride_am,
    stride_ak,
    batch_stride_b,
    hidden_stride_b,
    stride_bk,
    stride_bn,
    batch_stride_c,
    hidden_stride_c,
    stride_cm,
    stride_cn,
    M,
    N,
    K,
    HIDDEN_DIM: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Compute ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute which batch and hidden channel
    h = batch_id % HIDDEN_DIM
    b = batch_id // HIDDEN_DIM

    # pointers to the start of this (b,h) matrix
    a_base = a_ptr + b * batch_stride_a + h * hidden_stride_a
    b_base = b_ptr + b * batch_stride_b + h * hidden_stride_b
    c_base = c_ptr + b * batch_stride_c + h * hidden_stride_c

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # split M into two halves to improve register usage
    half_m = BLOCK_M // 2
    offs_m1 = offs_m
    offs_m2 = offs_m + half_m

    mask_m1 = mask_m
    mask_m2 = offs_m2 < M

    acc1 = tl.zeros((half_m, BLOCK_N), dtype=tl.float32)
    acc2 = tl.zeros((half_m, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        a_ptrs1 = a_base + offs_m1[:, None] * stride_am + k_offset[None, :] * stride_ak
        a_ptrs2 = a_base + offs_m2[:, None] * stride_am + k_offset[None, :] * stride_ak
        b_ptrs = b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a1 = tl.load(a_ptrs1,
                     mask=mask_m1[:, None] & mask_k[None, :],
                     other=0.0)
        a2 = tl.load(a_ptrs2,
                     mask=mask_m2[:, None] & mask_k[None, :],
                     other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc1 += tl.dot(a1, b, out_dtype=tl.float32)
        acc2 += tl.dot(a2, b, out_dtype=tl.float32)

    # combine halves
    acc = tl.concatenate([acc1, acc2], axis=0)

    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) – fast Triton implementation

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) → L,R ∈ ℝ^{B×N×N×H}.
3. Optional mask (broadcast on the last dim) and sigmoidal gates.
   L ← L·mask·σ(L_gate) ,   R ← R·mask·σ(R_gate)
4. Contraction (the costly O(N³·H) part):
        out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
   This is a batch of H independent matrix‑multiplications per batch.
   We fuse the whole batch (B·H) inside a single Triton kernel.
   The kernel works on the original memory layout (no permute / copy) and
   computes many 128×128 tiles with Tensor‑Core‑accelerated `tl.dot`.
5. Layer‑norm over the hidden dimension, apply the output gate,
   and a final linear projection back to C.

The kernel avoids any materialised transposes: it treats the hidden
dimension as a batched axis and computes the transpose of the right
operand on‑the‑fly by swapping the two sequence strides.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_tri_mul_kernel(
    a_ptr,                     # left  tensor  [B, N, N, H]  (fp16)
    b_ptr,                     # right tensor  [B, N, N, H]  (fp16), transposed on‑fly
    c_ptr,                     # output tensor [B, N, N, H]  (fp32)
    batch_stride_a,            # stride to next batch (B dimension) of A
    hidden_stride_a,           # stride to next hidden channel inside a batch   (usually 1)
    stride_am, stride_ak,      # strides for the i‑ (M) and k‑ (K) dimensions of A
    batch_stride_b,            # stride to next batch (B dimension) of B
    hidden_stride_b,           # stride to next hidden channel inside a batch   (usually 1)
    stride_bk, stride_bn,      # strides for the k‑ (K) and j‑ (N) dimensions of B (transposed)
    batch_stride_c,            # stride to next batch (B dimension) of C
    hidden_stride_c,           # stride to next hidden channel inside a batch   (usually 1)
    stride_cm, stride_cn,      # strides for the i‑ (M) and j‑ (N) dimensions of C
    M, N, K,                   # sequence length (all equal)
    HIDDEN_DIM: tl.constexpr, # number of hidden channels (== H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Identify the (batch, hidden) pair handled by this program instance
    # ------------------------------------------------------------------
    batch_id = tl.program_id(0)                     # 0‑th axis = B·H
    pid_m = tl.program_id(1)                        # tile over i (rows)
    pid_n = tl.program_id(2)                        # tile over j (cols)

    # hidden index h and real batch index b
    h = batch_id % HIDDEN_DIM
    b = batch_id // HIDDEN_DIM

    # base pointers for this (b,h) matrix
    a_base = a_ptr + b * batch_stride_a + h * hidden_stride_a
    b_base = b_ptr + b * batch_stride_b + h * hidden_stride_b
    c_base = c_ptr + b * batch_stride_c + h * hidden_stride_c

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows   i
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols   j
    mask_m = offs_m < M
    mask_n = offs_n < N

    # split the M‑tile in two halves – this matches the classic
    # Triton matmul pattern and gives a better register balance
    half_m = BLOCK_M // 2
    offs_m1 = offs_m
    offs_m2 = offs_m + half_m
    mask_m1 = mask_m
    mask_m2 = offs_m2 < M

    # accumulators (fp32 for numerical stability)
    acc1 = tl.zeros((half_m, BLOCK_N), dtype=tl.float32)
    acc2 = tl.zeros((half_m, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in tiles
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)        # k‑indices for this tile
        mask_k = k_offset < K

        # pointers for the left operand (two halves of the M‑tile)
        a_ptrs1 = a_base + offs_m1[:, None] * stride_am + k_offset[None, :] * stride_ak
        a_ptrs2 = a_base + offs_m2[:, None] * stride_am + k_offset[None, :] * stride_ak

        # pointer for the right operand (transposed on‑the‑fly)
        b_ptrs  = b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load with masks – out‑of‑bounds elements are zeroed
        a1 = tl.load(a_ptrs1,
                     mask=mask_m1[:, None] & mask_k[None, :],
                     other=0.0)
        a2 = tl.load(a_ptrs2,
                     mask=mask_m2[:, None] & mask_k[None, :],
                     other=0.0)
        b  = tl.load(b_ptrs,
                     mask=mask_k[:, None] & mask_n[None, :],
                     other=0.0)

        # Tensor‑core dot product, accumulate in fp32
        acc1 += tl.dot(a1, b, out_dtype=tl.float32)
        acc2 += tl.dot(a2, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write the two halves back to C
    # ------------------------------------------------------------------
    c_ptrs1 = c_base + offs_m1[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs1,
             acc1,
             mask=mask_m1[:, None] & mask_n[None, :])

    c_ptrs2 = c_base + offs_m2[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs2,
             acc2,
             mask=mask_m2[:, None] & mask_n[None, :])

# Entry‑point used by the evaluator
# ----------------------------------------------------------------------

    """

        input_tensor : torch.Tensor  [B, N, N, C]   (float32)
        mask         : torch.Tensor  [B, N, N]   (float32/bool) or None
        weights      : dict of model parameters (float32)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # Load weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Layer‑norm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H] , float32
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) – cast to fp16 for the heavy GEMM
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # keep fp32 for final scaling

    # Apply gates
    left = (left * left_gate).to(torch.float16)   # fp16 from now on
    right = (right * right_gate).to(torch.float16)

    # --------------------------------------------------------------
    # 5. TriMul contraction – fused Triton kernel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B = batch size, N = seq_len
    # Prepare output buffer (fp32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in *elements* (not bytes)
    a_stride = left.stride()
    b_stride = right.stride()
    c_stride = out.stride()

    # A (left) strides
    batch_stride_a = a_stride[0]          # stride from one batch to the next (B dim)
    hidden_stride_a = a_stride[3]         # stride to the next hidden channel (usually 1)
    stride_am = a_stride[1]               # stride over i  (M dimension)
    stride_ak = a_stride[2]               # stride over k  (K dimension)

    # B (right) strides – we treat it as transposed (k, j)
    batch_stride_b = b_stride[0]
    hidden_stride_b = b_stride[3]
    stride_bk = b_stride[2]               # stride over k (row of transposed B)
    stride_bn = b_stride[1]               # stride over j (col of transposed B)

    # C (output) strides
    batch_stride_c = c_stride[0]
    hidden_stride_c = c_stride[3]
    stride_cm = c_stride[1]               # stride over i
    stride_cn = c_stride[2]               # stride over j

    # Tile sizes – tuned for H100 FP16 Tensor‑Core matmul
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                      # one program per (batch, hidden) pair
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_gemm_tri_mul_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        batch_stride_a=batch_stride_a,
        hidden_stride_a=hidden_stride_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=batch_stride_b,
        hidden_stride_b=hidden_stride_b,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        batch_stride_c=batch_stride_c,
        hidden_stride_c=hidden_stride_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # H100 benefits from 8 warps per block
        num_stages=2,       # double‑buffering of loads
    )

    # --------------------------------------------------------------
    # 6. Final post‑processing (layer‑norm, out‑gate, final projection)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate   # out_gate is fp32, broadcast over last dim
    out = F.linear(out, to_out_weight)   # [B, N, N, C]

    return out
```
