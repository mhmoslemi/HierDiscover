"""
custom_kernel: Triton‑accelerated "outgoing" TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` (float32) → cast to float16.
2. Linear projections (left/right) and three gates (left_gate, right_gate,
   out_gate) are computed in float16 (bias‑free linear layers).
3. Optional mask (bool/float) is broadcast and applied together with the
   gates (still in float16) → final tensors `left` and `right`
   of shape [B, N, N, H] (H = hidden_dim).
4. Core N³ update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is a batched GEMM (A @ Bᵀ) executed by a custom Triton kernel.
   The kernel works directly on the original memory layout
   (no permute / view copies) and uses float16 arithmetic with
   float32 accumulation.
5. The result (`out`) is cast back to float32, layer‑normed over the
   hidden dimension, multiplied by `out_gate` and projected back to the
   original dimension with a final linear layer.
All heavy work (step 4) is done inside Triton; the remaining steps use
PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with A @ Bᵀ, handling the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_ah,           # A: batch‑stride, hidden‑stride
    stride_am, stride_ak,           # A: row‑stride (i), col‑stride (k)
    stride_bb, stride_bh,           # B: batch‑stride, hidden‑stride
    stride_bk, stride_bn,           # B: col‑stride (k), row‑stride (j) – we load Bᵀ
    stride_cb, stride_ch,           # C: batch‑stride, hidden‑stride
    stride_cm, stride_cn,           # C: row‑stride (i), col‑stride (j)
    HIDDEN_DIM: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(2)                # combined (batch, hidden)
    pid_m = tl.program_id(0)              # tile row index
    pid_n = tl.program_id(1)              # tile col index

    # ------------------------------------------------------------------
    # Compute batch index and hidden‑channel index from pid
    # ------------------------------------------------------------------
    batch_idx = pid // HIDDEN_DIM
    hid_idx   = pid % HIDDEN_DIM

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) matrix
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_idx * stride_ab + hid_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_bb + hid_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cb + hid_idx * stride_ch

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the first tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K + offs_k

        # Load a tile of A (left)
        a = tl.load(
            a_tile_ptr + (k_tile * BLOCK_K) * stride_ak,
            mask=mask_m[:, None] & (cur_k < K),
            other=tl.float16(0.0),
        )
        # Load a tile of Bᵀ (right, transposed on‑the‑fly)
        b = tl.load(
            b_tile_ptr + (k_tile * BLOCK_K) * stride_bk,
            mask=(cur_k[:, None] < K) & mask_n[None, :],
            other=tl.float16(0.0),
        )
        acc += tl.dot(a, b)   # FP16·FP16 → FP32 accumulation
    # ------------------------------------------------------------------
    # Write the result tile
    # ------------------------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
            - weights      : dict of model parameters (torch.Tensor)
            - config       : dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  (float32)
    x_h = x.half()  # cast to FP16 for the heavy linear work

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to FP16 (biases stay FP32)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # final projection

    # ------------------------------------------------------------------
    # 3) Linear projections + gates (all FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)          # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)

    left_gate = F.linear(x_h, left_gate_w).sigmoid()
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask (if any) and the element‑wise gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * left_gate * mask_h
        right = right * right_gate * mask_h
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ update – batched GEMM A @ Bᵀ via Triton
    #    Output tensor `out_h` has shape [B, N, N, H] and dtype FP16.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    out_h = torch.empty_like(left)   # allocate FP16 output

    # Block sizes chosen for the H100 Tensor‑Core pipeline
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # blocks over i (rows)
        triton.cdiv(N, BLOCK_N),          # blocks over j (cols)
        B * hidden_dim,                   # one program per (batch, hidden) matrix
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out_h,
        N,                     # M
        N,                     # N
        N,                     # K
        left.stride(0),        # stride_ab (batch stride for A)
        left.stride(3),        # stride_ah (hidden stride for A)
        left.stride(1),        # stride_am (row stride for A, i dimension)
        left.stride(2),        # stride_ak (col stride for A, k dimension)
        right.stride(0),       # stride_bb (batch stride for B)
        right.stride(3),       # stride_bh (hidden stride for B)
        right.stride(2),       # stride_bk (col stride for B, k dimension)
        right.stride(1),       # stride_bn (row stride for B, j dimension – transposed)
        out_h.stride(0),       # stride_cb (batch stride for C)
        out_h.stride(3),       # stride_ch (hidden stride for C)
        out_h.stride(1),       # stride_cm (row stride for C, i dimension)
        out_h.stride(2),       # stride_cn (col stride for C, j dimension)
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 6) Hidden‑dim layer‑norm + out_gate (now in FP32)
    # ------------------------------------------------------------------
    out = out_h.float()                       # cast back to FP32 for LN
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.float()               # broadcast mul

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dim (FP32 output)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)               # weight is FP16, but torch up‑casts

    return out
