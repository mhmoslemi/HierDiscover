def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # 1. LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16 for linear projections
    x_fp16 = x.to(torch.float16)

    # Convert weights to fp16 once
    left_proj_weight_fp16 = weights["left_proj.weight"].to(torch.float16)
    left_gate_weight_fp16 = weights["left_gate.weight"].to(torch.float16)
    right_proj_weight_fp16 = weights["right_proj.weight"].to(torch.float16)
    right_gate_weight_fp16 = weights["right_gate.weight"].to(torch.float16)

    # Concatenate per side
    left_weights_cat = torch.cat([left_proj_weight_fp16, left_gate_weight_fp16], dim=0)  # (2*hidden, dim)
    right_weights_cat = torch.cat([right_proj_weight_fp16, right_gate_weight_fp16], dim=0)

    # Linear projections + gates
    left_cat = F.linear(x_fp16, left_weights_cat)  # shape [B,N,N,2*hidden]
    right_cat = F.linear(x_fp16, right_weights_cat)

    left_proj = left_cat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])

    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # permute to [B, hidden, N, N] and make contiguous
    left_proj_perm = left_proj.permute(0,3,1,2).contiguous()
    left_gate_perm = left_gate.permute(0,3,1,2).contiguous()
    right_proj_perm = right_proj.permute(0,3,1,2).contiguous()
    right_gate_perm = right_gate.permute(0,3,1,2).contiguous()

    B, H, N, _ = left_proj_perm.shape

    # Flatten batch*hidden
    BH = B * H
    A_proj = left_proj_perm.view(BH, N, N)
    A_gate = left_gate_perm.view(BH, N, N)
    B_proj = right_proj_perm.view(BH, N, N)
    B_gate = right_gate_perm.view(BH, N, N)

    # mask handling
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16)
        HAS_MASK = True
    else:
        mask_fp16 = torch.empty(1, device=device, dtype=torch.float16)  # dummy
        HAS_MASK = False

    # Run kernel
    out_mat = batch_matmul_gate(
        A_proj, A_gate, B_proj, B_gate, mask_fp16,
        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32, HAS_MASK=HAS_MASK
    )  # shape [BH,N,N] dtype float32

    # reshape back
    out_hidden = out_mat.view(B, H, N, N).permute(0,2,3,1)  # [B,N,N,hidden]

    # 2nd LayerNorm
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # output gate
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    out_hidden = out_hidden * out_gate

    # final projection to dim
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) – fast Triton implementation
================================================

This file contains a highly‑optimised forward implementation of the
*outgoing* triangle multiplicative update (TriMul) used in AlphaFold‑3.
The heavy computation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply (A @ Bᵀ) over the sequence dimension.
A custom Triton kernel fuses the linear projections, sigmoid gates and
optional pairwise mask directly into the GEMM, thus removing extra
copies of the (very large) tensors and cutting memory traffic.
The remaining operations (LayerNorms, final linear projection and the
output gate) are performed with regular PyTorch calls.

Key ideas
---------
* Cast the input to **float16** for the three linear projections and the
  two gate logits – this reduces memory bandwidth by 2× while keeping
  the final output in float32.
* Concatenate the projection‑ and gate‑weights for each side so that a
  single `F.linear` produces both the projected values and the pre‑sigmoid
  gate logits; the sigmoid is applied afterwards.
* The Triton kernel receives the projected tensors, the gate tensors and
  the optional mask, multiplies the gates (and mask) on‑the‑fly and
  accumulates the dot‑product in float32.
* The kernel works on a flattened “batch‑head’’ dimension
  (batch × hidden) and uses 128‑×‑128 tiles with a 32‑wide K‑loop,
  which fits well in the L2 of an H100. 8 warps are used per program.

The function `custom_kernel` follows the signature required by the
challenge and returns a tensor of shape `[bs, seq_len, seq_len, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with on‑the‑fly gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_gate_kernel(
    A_ptr, A_gate_ptr,               # left projection and gate       [BH, M, K]
    B_ptr, B_gate_ptr,               # right projection and gate      [BH, N, K]
    mask_ptr, C_ptr,                 # mask (optional) and output     [BH, M, N]
    batch_head, head_dim,            # total batch‑head, hidden dim
    M, N, K,                         # matrix dimensions (M=N=seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_a_gate_batch,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_b_gate_batch,
    stride_mask_batch, stride_mask_m, stride_mask_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(axis=0)          # tile over output (M×N)
    bh = tl.program_id(axis=1)           # flattened batch‑head index

    # --------------------------------------------------------------
    # Compute tile indices
    # --------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid %  num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this batch‑head
    # --------------------------------------------------------------
    a_base = A_ptr        + bh * stride_a_batch
    a_gate_base = A_gate_ptr + bh * stride_a_gate_batch
    b_base = B_ptr        + bh * stride_b_batch
    b_gate_base = B_gate_ptr + bh * stride_b_gate_batch
    c_base = C_ptr        + bh * stride_c_batch

    if HAS_MASK:
        batch_idx = bh // head_dim               # original batch id
        mask_base = mask_ptr + batch_idx * stride_mask_batch
    else:
        mask_base = tl.zeros((1,), dtype=tl.float16)   # dummy, never used

    # --------------------------------------------------------------
    # Accumulator (float32 for precision)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑loop: Σ_k  (A * A_gate * mask) · (B * B_gate * mask)ᵀ
    # --------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- left side -------------------------------------------------
        a_ptrs = a_base + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        a_gate_ptrs = a_gate_base + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a_gate = tl.load(a_gate_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0)

        a = a * a_gate
        if HAS_MASK:
            mask_a_ptrs = mask_base + (offs_m[:, None] * stride_mask_m) + (offs_k[None, :] * stride_mask_k)
            mask_a = tl.load(mask_a_ptrs,
                             mask=mask_m[:, None] & mask_k[None, :],
                             other=1.0)
            a = a * mask_a

        # ----- right side (transposed) ------------------------------------
        b_ptrs = b_base + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        b_gate_ptrs = b_gate_base + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b_gate = tl.load(b_gate_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)

        b = b * b_gate
        if HAS_MASK:
            # mask for the (j,k) entry – note the swap of axes
            mask_b_ptrs = mask_base + (offs_n[:, None] * stride_mask_m) + (offs_k[None, :] * stride_mask_k)
            mask_b = tl.load(mask_b_ptrs,
                             mask=mask_n[:, None] & mask_k[None, :],
                             other=1.0)
            b = b * mask_b

        # ----- dot product -------------------------------------------------
        acc += tl.dot(a, b)                     # fp16·fp16 → fp32 accumulation

    # --------------------------------------------------------------
    # Write back the tile
    # --------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def batch_matmul_gate(
    A_proj: torch.Tensor, A_gate: torch.Tensor,
    B_proj: torch.Tensor, B_gate: torch.Tensor,
    mask: torch.Tensor,
    head_dim: int,
    BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32,
    HAS_MASK: bool = False,
) -> torch.Tensor:
    """
    Compute C = (A_proj * A_gate * mask) @ (B_proj * B_gate * mask)ᵀ
    where all tensors have shape [BH, N, N] (BH = batch * hidden_dim).
    The result C has shape [BH, N, N] and dtype float32.
    """
    BH, N, K = A_proj.shape
    assert N == K, "TriMul expects square sequence dimensions"

    C = torch.empty_like(A_proj, dtype=torch.float32)

    # Strides (int32) – note that PyTorch stride is in elements, not bytes
    stride_a_batch = A_proj.stride(0)
    stride_a_m = A_proj.stride(1)
    stride_a_k = A_proj.stride(2)

    stride_a_gate_batch = A_gate.stride(0)

    stride_b_batch = B_proj.stride(0)
    stride_b_n = B_proj.stride(1)
    stride_b_k = B_proj.stride(2)

    stride_b_gate_batch = B_gate.stride(0)

    if HAS_MASK:
        stride_mask_batch = mask.stride(0)
        stride_mask_m = mask.stride(1)
        stride_mask_k = mask.stride(2)
    else:
        # dummy values – never used when HAS_MASK=False
        stride_mask_batch = stride_mask_m = stride_mask_k = 0

    stride_c_batch = C.stride(0)
    stride_c_m = C.stride(1)
    stride_c_n = C.stride(2)

    grid_x = ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)
    grid = (grid_x, BH)   # (tiles over M×N, batch‑head)

    batch_matmul_gate_kernel[grid](
        A_proj,
        A_gate,
        B_proj,
        B_gate,
        mask,
        C,
        BH,
        head_dim,
        N, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_a_gate_batch,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_b_gate_batch,
        stride_mask_batch, stride_mask_m, stride_mask_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
        num_warps=8,
    )
    return C

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. First LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections + sigmoid gates (mixed‑precision)
    #    * Concatenate projection & gate weights → a single matmul per side
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # left side (proj + gate)
    left_w_proj = weights["left_proj.weight"].to(torch.float16)
    left_w_gate = weights["left_gate.weight"].to(torch.float16)
    left_w_cat = torch.cat([left_w_proj, left_w_gate], dim=0)          # (2*hidden, dim)

    left_cat = F.linear(x_fp16, left_w_cat)                            # [..., 2*hidden]
    left_proj = left_cat[..., :hidden_dim]                             # (..., hidden)
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])              # (..., hidden)

    # right side (proj + gate)
    right_w_proj = weights["right_proj.weight"].to(torch.float16)
    right_w_gate = weights["right_gate.weight"].to(torch.float16)
    right_w_cat = torch.cat([right_w_proj, right_w_gate], dim=0)

    right_cat = F.linear(x_fp16, right_w_cat)
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 3. Re‑order tensors to [batch, hidden, seq, seq] (contiguous)
    # ------------------------------------------------------------------
    # shape after permute: [B, hidden, N, N] – now contiguous in memory
    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_proj.shape
    BH = B * H

    # Flatten batch × hidden → leading dimension for the kernel
    A_proj = left_proj.view(BH, N, N)
    A_gate = left_gate.view(BH, N, N)
    B_proj = right_proj.view(BH, N, N)
    B_gate = right_gate.view(BH, N, N)

    # ------------------------------------------------------------------
    # 4. Optional pairwise mask (float16).  If nomask==True we skip it.
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).contiguous()
        HAS_MASK = True
    else:
        mask_fp16 = torch.empty(1, device=device, dtype=torch.float16)  # dummy
        HAS_MASK = False

    # ------------------------------------------------------------------
    # 5. Core TriMul – fused batched GEMM (A @ Bᵀ) with gates & mask
    # ------------------------------------------------------------------
    out_hidden = batch_matmul_gate(
        A_proj, A_gate,
        B_proj, B_gate,
        mask_fp16,
        head_dim=hidden_dim,
        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32,
        HAS_MASK=HAS_MASK,
    )                                     # shape [BH, N, N], dtype float32

    # Restore original layout: [B, N, N, hidden]
    out_hidden = out_hidden.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Output gating (sigmoid) and final linear projection back to dim
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden]
    out_hidden = out_hidden * out_gate

    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```
