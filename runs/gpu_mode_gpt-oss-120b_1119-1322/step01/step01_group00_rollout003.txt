    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    pid_b = tl.program_id(0)  # B dimension? We could have grid (B, ceil(N/BLOCK_I), ceil(N/BLOCK_J))

    j_start = pid_j * BLOCK_J

"""
TriMul “outgoing” kernel – highly optimized Triton implementation.

Algorithm
---------

1. **Input layer‑norm** (C dimension) – executed with torch.nn.functional.layer_norm.
2. **Linear projections** (left / right) and three **sigmoid gates**
   (left_gate, right_gate, out_gate) – performed in fp16 with torch.nn.functional.linear.
3. **Mask** (if present) and gate application are fused with the projections.
4. **Core triangle multiplication**  
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  
   This is a batch of H independent N×N matrix‑multiplications.  
   We implement it with a custom Triton kernel that:
   * works on fp16 inputs, accumulates in fp32,
   * processes a tile of the output (BLOCK_I×BLOCK_J) for a small
     hidden‑dimension tile (TILE_H) at a time,
   * loops over K‑tiles (BLOCK_K) and uses `tl.dot` for the inner
     matrix‑multiply of each hidden slice.
5. **Output layer‑norm** over the hidden dimension – torch.nn.functional.layer_norm.
6. **Element‑wise gating** with `out_gate` – a tiny Triton kernel.
7. **Final linear projection** back to the original channel dimension (float32).

The heavy N³·C work is therefore performed entirely on the GPU with
minimal data movement and fp16×fp16→fp32 arithmetic, giving a large
speed‑up over the naïve implementation.
"""

import math
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # float32* (out_norm flattened)
    b_ptr,                # float32* (out_gate flattened)
    out_ptr,              # float32* (output flattened)
    size,                 # total number of elements
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: outgoing triangle multiplication
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,   # fp16*  (shape [B,N,N,H])
    right_ptr,  # fp16*  (shape [B,N,N,H])
    out_ptr,    # fp32*  (shape [B,N,N,H])
    B: tl.constexpr,           # batch size
    N: tl.constexpr,           # sequence length
    H: tl.constexpr,           # hidden dimension
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,     # tile size for output i‑axis
    BLOCK_J: tl.constexpr,     # tile size for output j‑axis
    BLOCK_K: tl.constexpr,     # tile size for reduction axis k
    TILE_H: tl.constexpr        # tile size for hidden dimension
):
    # 0‑th program id encodes (batch, hidden‑tile) pair.
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # --------------------------------------------------------------
    # Decode batch index and hidden‑tile index
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, TILE_H)
    b = pid0 // num_h_tiles                     # batch index
    tile_h = pid0 % num_h_tiles                  # hidden‑tile index
    h_start = tile_h * TILE_H

    # --------------------------------------------------------------
    # Offsets for the output tile
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_h = h_start + tl.arange(0, TILE_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator: (BLOCK_I, BLOCK_J, TILE_H) in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, TILE_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left tile (i, k, h) -------
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + offs_i[:, None, None] * stride_l_i \
            + offs_k[None, :, None] * stride_l_k \
            + offs_h[None, None, :] * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0).to(tl.float32)   # fp16 → fp32

        # ------- load right tile transposed (k, j, h) -------
        # we need right[j,k,h] -> (k,j,h) for dot product
        right_ptrs = right_ptr \
            + b * stride_r_b \
            + offs_k[:, None, None] * stride_r_k \
            + offs_j[None, :, None] * stride_r_i \
            + offs_h[None, None, :] * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                        other=0.0).to(tl.float32)

        # ------- inner product per hidden slice -------
        # unrolled over TILE_H (compile‑time constant)
        for dh in range(TILE_H):
            # Note: out‑of‑range hidden indices are zero‑padded by the masks above
            acc[:, :, dh] += tl.dot(left[:, :, dh], right[:, :, dh])   # (BLOCK_I, BLOCK_J)

    # --------------------------------------------------------------
    # Store the result (fp32) with proper masking
    # --------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + offs_i[:, None, None] * stride_o_i \
        + offs_j[None, :, None] * stride_o_j \
        + offs_h[None, None, :] * stride_o_h
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/0‑1) – may be ignored when ``nomask`` is True
        - weights      : dict of model weights (float32 tensors)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension (C = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (left / right) → hidden_dim
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Three sigmoid gates (left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Apply gates (element‑wise)
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6. Triangle multiplication (core N³·hidden work) – Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B, N, N, hidden_dim
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides (in elements) for left/right/out tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Triton launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    TILE_H  = 8                     # hidden‑tile size (must divide hidden_dim reasonably)

    num_h_tiles = (H + TILE_H - 1) // TILE_H
    grid = (
        B * num_h_tiles,                          # program_id(0) – batch * hidden‑tile
        (N + BLOCK_I - 1) // BLOCK_I,              # program_id(1) – i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J               # program_id(2) – j‑tiles
    )
    tri_mul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, TILE_H=TILE_H,
        num_warps=4,  # empirical choice for H100
    )

    # ------------------------------------------------------------------
    # 7. Output‑side LayerNorm over hidden_dim (float32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )   # [B, N, N, hidden_dim] (float32)

    # ------------------------------------------------------------------
    # 8. Gating with out_gate – Triton elementwise kernel
    # ------------------------------------------------------------------
    # Flatten tensors for 1‑D kernel launch
    out_norm_flat = out_norm.view(-1)
    out_gate_flat = out_gate.to(torch.float32).view(-1)
    gated_flat    = torch.empty_like(out_norm_flat)

    total_elements = out_norm_flat.numel()
    BLOCK_SIZE = 65536                     # about 64 Ki elements per program
    grid_elem = (triton.cdiv(total_elements, BLOCK_SIZE),)

    elementwise_mul_kernel[grid_elem](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE
    )
    gated = gated_flat.view(B, N, N, hidden_dim)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9. Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    output = torch.nn.functional.linear(gated, to_out_w)   # [B, N, N, dim] (float32)

    return output
