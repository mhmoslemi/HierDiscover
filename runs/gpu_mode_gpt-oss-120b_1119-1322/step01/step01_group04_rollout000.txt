"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) → left/right ∈ ℝ^{B×N×N×H}.
3. Optional mask (broadcasted) and three sigmoid gates are applied.
4. The costly contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix‑multiply of shape (B·H, N, N) @ (B·H, N, N)ᵀ.
   A custom Triton kernel performs this GEMM in fp16 with fp32 accumulation.
5. Layer‑norm on the hidden dimension, gating, and final linear projection
   back to the original channel dimension C.

The kernel is tuned for H100 (fp16 matrix‑multiply, 8 warps, 4 stages) and
uses dynamic block sizes (64 or 128) depending on the sequence length.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                # ptr to left  tensor [B·H, M, K]  (fp16)
    b_ptr,                # ptr to right tensor [B·H, K, N]  (fp16, K‑major)
    c_ptr,                # ptr to output tensor [B·H, M, N] (fp32)
    batch_stride_a,       # stride between consecutive batches in A
    stride_am, stride_ak, # strides for M and K in A
    batch_stride_b,       # stride between consecutive batches in B
    stride_bk, stride_bn, # strides for K and N in B
    batch_stride_c,       # stride between consecutive batches in C
    stride_cm, stride_cn, # strides for M and N in C
    M, N, K,              # matrix dimensions (all equal to seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # Program IDs: (batch, block_m, block_n)
    # ---------------------------------------------------------------
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ------------------------------ tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------ base pointers --------------------
    a_base = a_ptr + batch_id * batch_stride_a
    b_base = b_ptr + batch_id * batch_stride_b
    c_base = c_ptr + batch_id * batch_stride_c

    # ------------------------------ accumulator -----------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------ K‑loop ---------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        # Global k indices for this tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current tile of A and B
        a_ptrs = a_base + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        b_ptrs = b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Masks for the tails of M/K/N
        mask_k = k_offset < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate the dot product (fp16 × fp16 → fp32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------ store result --------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass using a fused Triton kernel for the
    N³ contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor[B, N, N, C] (float32)
            mask:         torch.Tensor[B, N, N] (bool/float) or None
            weights: dict of model parameters (float32)
            config:  dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]          # C
    hidden_dim = config["hidden_dim"]   # H
    eps        = 1e-5

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]  float32
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]  float32

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool; bring to same dtype as the data
        mask_ = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    # Apply gates
    left  = (left  * left_gate).to(torch.float16)   # fp16, masked & gated
    right = (right * right_gate).to(torch.float16)  # fp16, masked & gated

    # ------------------------------------------------------------------
    # 5. Triton‑fused contraction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    i.e. a batched GEMM of (B·H, N, N) @ (B·H, N, N)^T
    # ------------------------------------------------------------------
    # Rearrange tensors so that the hidden dimension becomes the batch axis.
    # left : (B, N, N, H) → (B, H, N, N)   (i,k)
    # right: (B, N, N, H) → (B, H, N, N)   (k,j)  – we transpose the last two dims.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape
    batch_dim   = B * H

    # Collapse batch and hidden dimensions -> independent GEMMs
    left_flat  = left_perm.view(batch_dim, N, N)   # [BH, M, K]
    right_flat = right_perm.view(batch_dim, N, N)  # [BH, K, N]

    # Allocate output (float32 accumulator)
    out_flat = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------- strides (elements) ----------------
    batch_stride_a = left_flat.stride(0)
    stride_am      = left_flat.stride(1)   # M‑major stride (= N)
    stride_ak      = left_flat.stride(2)   # K‑major stride (= 1)

    batch_stride_b = right_flat.stride(0)
    stride_bk      = right_flat.stride(1)  # K‑major stride (= N)
    stride_bn      = right_flat.stride(2)  # N‑major stride (= 1)

    batch_stride_c = out_flat.stride(0)
    stride_cm      = out_flat.stride(1)    # M‑major stride (= N)
    stride_cn      = out_flat.stride(2)    # N‑major stride (= 1)

    # ------------------------------- block size selection ---------------
    # Small sequence lengths benefit from 64‑tile, larger ones from 128‑tile.
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel
    tri_mul_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=batch_stride_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=batch_stride_b,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        batch_stride_c=batch_stride_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # H100 benefits from 8 warps per block
        num_stages=4,     # prefetch two stages ahead
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Final post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                    # gated hidden output
    out = F.linear(out, to_out_weight)      # back to original channel dim C

    return out
