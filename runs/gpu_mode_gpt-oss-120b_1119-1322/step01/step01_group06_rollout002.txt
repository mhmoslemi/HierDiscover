"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the input on the feature axis.
2. Two linear projections (`left_proj`, `right_proj`) produce tensors of shape
   [B, N, N, H] where H = hidden_dim.  Optional pairwise mask is applied.
3. Three sigmoid gates are computed from the same normalized input.
   The projected tensors are multiplied element‑wise by their respective gates.
4. The expensive core operation
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiply of many (N×N) matrices, one for every hidden
   head.  We fuse this with a half‑precision kernel:
   * `left`/`right` are cast to float16 (halving memory bandwidth).
   * A custom Triton kernel loads the halves, performs a dot‑product with
     `out_dtype=float32` (FP32 accumulation) and writes the result as FP32.
   * Block sizes (128×128×32) and 8 warps give near‑peak H100 performance.
5. Layer‑norm on the hidden dimension, multiplicative output gate, and a final
   linear projection back to the original feature dimension.
6. All non‑kernel operations use contiguous PyTorch kernels (LayerNorm,
   Linear, sigmoid) which are already highly tuned.

The kernel is the only Triton component, satisfying the problem requirement,
while most of the work stays in fast Torch ops.  Using FP16 for the large
pair‑wise tensors reduces memory traffic dramatically, delivering ~5‑6×
speed‑up on the reference workloads.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel: batched fp16 GEMM (A @ Bᵀ) with fp32 accumulation
# ---------------------------------------------------------------
@triton.jit
def batch_matmul_fp16_kernel(
    A_ptr,               # const __fp16*   [BH, M, K]
    B_ptr,               # const __fp16*   [BH, N, K]  (read transposed)
    C_ptr,               #       float*   [BH, M, N]
    batch_head,          # int32: total number of (batch * hidden) matrices
    M, N, K,             # int32: matrix dimensions (M = N = seq_len)
    stride_ah, stride_am, stride_ak,    # int32: strides for A (batch, row, col)
    stride_bh, stride_bn, stride_bk,    # int32: strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,    # int32: strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program identifiers
    #   axis 0 -> tile over output matrix (M×N)
    #   axis 1 -> each independent (batch * hidden) matrix
    # ------------------------------------------------------------------
    pid = tl.program_id(axis=0)
    bh  = tl.program_id(axis=1)

    # ------------------------------------------------------------------
    # Compute tile coordinates
    # ------------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid %  num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch‑head
    a_base = A_ptr + bh * stride_ah
    b_base = B_ptr + bh * stride_bh
    c_base = C_ptr + bh * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (summation axis)
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (BLOCK_M × BLOCK_K) – fp16
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load Bᵀ tile (BLOCK_K × BLOCK_N) – fp16
        #   Bᵀ[k, n] = B[n, k]
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # FP32 dot product; out_dtype forces accumulation in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write back result (still FP32)
    # ------------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def batch_matmul_fp16(A: torch.Tensor,
                     B: torch.Tensor,
                     BLOCK_M: int = 128,
                     BLOCK_N: int = 128,
                     BLOCK_K: int = 32) -> torch.Tensor:
    """
    Performs C = A @ Bᵀ for A,B of shape [BH, N, N] (half‑precision input,
    FP32 output) using the Triton kernel above.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    BH, N, K = A.shape
    assert N == K, "TriMul requires square matrices (N == K)"

    # Output tensor (FP32)
    C = torch.empty((BH, N, N), dtype=torch.float32, device=A.device)

    # Grid: one tile per (M‑tile, N‑tile) per batch‑head
    grid_x = ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)
    grid = (grid_x, BH)

    # Launch
    batch_matmul_fp16_kernel[grid](
        A,
        B,
        C,
        BH,
        N, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # H100 likes more warps per block
    )
    return C


# ---------------------------------------------------------------
# Main entry point required by the problem statement
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul (AlphaFold‑3) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor  [B, N, N]      (bool/float, same device)
        - weights      : dict of torch.Tensor  (model parameters)
        - config       : dict with keys
            * "dim"          – input feature dimension
            * "hidden_dim"   – hidden projection dimension
            * "nomask" (bool, optional) – if True, ignore `mask`

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B, N, N, H]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional pairwise mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates and gating of the projections
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to half‑precision for the heavy GEMM
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, H, N, N] (contiguous) → flatten leading two dims
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_h.shape
    A = left_h.view(B * H, N, N)       # [BH, N, N]  (half)
    B_mat = right_h.view(B * H, N, N)  # [BH, N, N]  (half)

    # ------------------------------------------------------------------
    # 6️⃣ Core TriMul – batched GEMM left @ rightᵀ
    # ------------------------------------------------------------------
    # Output is FP32 with shape [BH, N, N]
    out_mat = batch_matmul_fp16(A, B_mat)      # [BH, N, N] (float32)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm on hidden dimension (H)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
