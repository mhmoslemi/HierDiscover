"""
Optimized outgoing TriMul kernel (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm the input tensor *x* over the feature dimension (dim).
2️⃣ Linear projections → *left* , *right* (no bias) into the hidden dimension.
3️⃣ Optional pair‑wise mask multiplication.
4️⃣ Compute three sigmoid gates (left_gate, right_gate, out_gate) and
   apply them element‑wise to *left* and *right*.
5️⃣ Core operation (outgoing TriMul) is a batched GEMM
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix‑multiply
        C = A @ Bᵀ   with A = left , B = right .
   The GEMM is performed by a custom Triton kernel that works in
   FP16 (Tensor‑Core) while accumulating in FP16 → the final tensor is
   cast back to FP32.
6️⃣ Second LayerNorm on the hidden dimension, multiplication by *out_gate*.
7️⃣ Linear projection back to the original feature dimension.

All heavy arithmetic (steps 2‑5, 7) is done in half‑precision,
only the final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B shape = [BH, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    A_ptr,           # const half*   [BH, M, K]
    B_ptr,           # const half*   [BH, N, K]  (read transposed)
    C_ptr,           #       half*   [BH, M, N]
    batch_head,      # int32: total number of (batch * hidden) matrices
    M, N, K,         # int32: matrix dimensions (M = N = seq_len)
    stride_ah, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bn, stride_bk,   # strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Half‑precision batched GEMM (A @ Bᵀ).

    Each program processes a tile of size BLOCK_M×BLOCK_N of one
    matrix pair (identified by `bh = program_id(1)`).  The K‑dimension
    is traversed in strides of BLOCK_K using Tensor‑Core `tl.dot`.
    """
    pid = tl.program_id(axis=0)          # tile over (M,N)
    bh  = tl.program_id(axis=1)          # which (batch,head) matrix

    # --------------------------------------------------------------
    # Block coordinates
    # --------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid %  num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for this batch‑head
    # ------------------------------------------------------------------
    a_base = A_ptr + bh * stride_ah
    b_base = B_ptr + bh * stride_bh
    c_base = C_ptr + bh * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (FP16)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Bᵀ tile (K × N)  →  B[n, k]
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Tensor‑Core dot product on the K dimension
        acc += tl.dot(a, b)          # result is FP16

    # ------------------------------------------------------------------
    # Write back C
    # ------------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def _batch_gemm(A: torch.Tensor,
                B: torch.Tensor,
                BLOCK_M: int = 128,
                BLOCK_N: int = 128,
                BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for A,B ∈ [BH, N, N] in FP16.
    Returns C with same shape and dtype as A.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    BH, N, K = A.shape
    assert N == K, "TriMul expects square matrices"

    C = torch.empty_like(A)

    # Grid: one program per output tile × per (batch*head)
    grid_x = (tl.cdiv(N, BLOCK_M)) * (tl.cdiv(N, BLOCK_N))
    grid = (grid_x, BH)   # (tiles, batch‑head)

    _batch_gemm_kernel[grid](
        A,
        B,
        C,
        BH,
        N, N, K,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(1), B.stride(2),   # B: batch, row, col
        C.stride(0), C.stride(1), C.stride(2),   # C: batch, row, col
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per program (fits H100 well)
        num_stages=3,         # shared‑memory pipeline depth
    )
    return C

# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask : torch.Tensor of shape [B, N, N]  (or None)
        - weights : dict of torch.Tensors containing all linear‑layer
                    weights and LayerNorm parameters
        - config : dict with at least ``dim`` and ``hidden_dim`` keys.
                   Optional ``nomask`` (bool) controls whether ``mask`` is
                   applied.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype = float32)
    """
    # ----------------------- unpack -----------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------- constants ---------------------------------
    # Convert all model parameters to half‑precision (Tensor‑Core friendly)
    def _h(t): return t.to(torch.float16, non_blocking=True)

    norm_weight = _h(weights["norm.weight"])
    norm_bias   = _h(weights["norm.bias"])
    left_proj_weight  = _h(weights["left_proj.weight"])
    right_proj_weight = _h(weights["right_proj.weight"])
    left_gate_weight  = _h(weights["left_gate.weight"])
    right_gate_weight = _h(weights["right_gate.weight"])
    out_gate_weight   = _h(weights["out_gate.weight"])
    to_out_norm_weight = _h(weights["to_out_norm.weight"])
    to_out_norm_bias   = _h(weights["to_out_norm.bias"])
    to_out_weight      = _h(weights["to_out.weight"])

    # ----------------------- 1️⃣ LayerNorm on input --------------------
    # NOTE: LayerNorm works on FP16 as long as weight/bias are the same dtype.
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)          # all subsequent ops in half

    # ----------------------- 2️⃣ Linear projections --------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)  # same shape

    # ----------------------- 3️⃣ Optional masking ----------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ----------------------- 4️⃣ Gating (sigmoid) ----------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------- 5️⃣ Core TriMul (batched GEMM) ------------
    # reshape to [B, hidden, N, N] → flatten batch*hidden for one launch
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)          # [BH, N, N]
    B_mat = right_perm.view(B * H, N, N)     # [BH, N, N]

    # Batched half‑precision GEMM (A @ Bᵀ)
    out_mat = _batch_gemm(A, B_mat)         # [BH, N, N]  (FP16)

    # Restore the original layout [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ----------------------- 6️⃣ Second LayerNorm (hidden) ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ----------------------- 7️⃣ Output gating -----------------------
    out = out * out_gate

    # ----------------------- 8️⃣ Final projection --------------------
    out = F.linear(out, to_out_weight)      # back to [B,N,N,dim] (FP16)

    # Return in FP32 to match the reference implementation
    return out.to(torch.float32)
