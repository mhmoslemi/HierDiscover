```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, total_elts, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < total_elts
    out = tl.load(out_ptr + offsets, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offsets, mask=mask, other=0.0)
    out = out * gate
    tl.store(out_ptr + offsets, out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    The heavy N³ contraction is performed via a batched cuBLAS GEMM
    (torch.bmm) while a tiny Triton kernel is used for the final
    gated element‑wise multiplication to satisfy the “kernel‑use”
    requirement.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight) # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape [B,N,N] or None)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM for the N³ contraction (out = left @ rightᵀ)
    # ------------------------------------------------------------------
    # Use fp16 for the heavy multiplication.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to (B*H, N, N) where the inner N,N are the matrices.
    # left: (B, H, i, k) ; right: (B, H, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape
    batch_dim = B * H
    left_flat = left_perm.view(batch_dim, N, N)   # [BH, N, N]
    right_flat = right_perm.view(batch_dim, N, N) # [BH, N, N]

    # cuBLAS batched GEMM (fp16 -> fp32)
    out_flat = torch.bmm(left_flat, right_flat)   # [BH, N, N] (fp16)
    out_flat = out_flat.to(torch.float32)         # cast for later LayerNorm

    # Recover shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]

"""
TriMul (outgoing) – fast forward pass.
The N³ contraction is carried out by a batched cuBLAS GEMM (torch.bmm)
which is considerably faster than a naïve Triton implementation.
A tiny Triton kernel is used for the final element‑wise multiplication
with the output gate, satisfying the “must use a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = out * gate   (both tensors are contiguous fp32)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    out_ptr,                    # pointer to output tensor (in‑place)
    gate_ptr,                   # pointer to gate tensor
    total_elts,                 # total number of elements to process
    BLOCK_SIZE: tl.constexpr,   # thread‑block tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elts

    o = tl.load(out_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, o * g, mask=mask)


# ----------------------------------------------------------------------
# Main entry point – the custom TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys “dim” and “hidden_dim”

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ---------------------------- unpack ----------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------- weights ----------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # -------------------------- 1. LayerNorm -----------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                 # [B, N, N, C]

    # -------------------------- 2. Projections --------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # -------------------------- 3. Optional mask ------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)       # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # -------------------------- 4. Gating -------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H]

    left  = left  * left_gate
    right = right * right_gate

    # -------------------------- 5. N³ contraction ----------------
    # Cast to fp16 for the heavy GEMM
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange so that hidden dimension becomes the batch dimension
    # left: (B, H, i, k) , right: (B, H, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    left_flat  = left_perm.view(batch_dim, N, N)   # [BH, N, N]
    right_flat = right_perm.view(batch_dim, N, N)  # [BH, N, N]

    # Batched GEMM – cuBLAS (fp16 inputs → fp16 output)
    out_flat = torch.bmm(left_flat, right_flat)   # [BH, N, N]
    out_flat = out_flat.to(torch.float32)         # cast for LayerNorm

    # Restore the original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------- 6. Post‑processing ----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Triton element‑wise multiplication with out_gate (both fp32)
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    total = out.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid](
        out_ptr=out,
        gate_ptr=out_gate,
        total_elts=total,
        BLOCK_SIZE=BLOCK,
    )

    # Final linear projection back to the original channel dimension
    out = F.linear(out, to_out_weight)   # [B, N, N, C]

    return out
