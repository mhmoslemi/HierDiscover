"""
custom_kernel – TriMul (outgoing) forward pass

Algorithm
---------
1. Layer‑norm ``x`` on the last dimension (FP32 for stability).
2. Linear projections (left/right) and three gating projections (bias‑free).
3. Apply optional mask and element‑wise sigmoid gates.
4. Cast the gated tensors to FP16 and fold the hidden dimension into the
   batch dimension → shapes (B*H, N, N).
5. Transpose the RHS so that the inner‑most dimension (N) is contiguous.
   This yields optimal coalesced loads for the GEMM.
6. Batched GEMM C = A @ Bᵀ where A and B are the left/right tensors.
   The compute is performed by a custom Triton kernel that:
   * tiles the M, N dimensions (128×128) and the K dimension (64),
   * loads tiles in FP16 and accumulates in FP32 using ``tl.dot`` (Tensor‑Cores),
   * writes the result back in FP16.
7. Reshape the result back to (B, N, N, H) and convert to FP32.
8. Layer‑norm over the hidden dimension, multiply by an output gate,
   and a final bias‑free linear projection back to the original feature dimension.
9. Return the FP32 output tensor.

The heavy O(B·H·N³) work is executed entirely inside the Triton kernel,
while the cheap linear / norm / sigmoid work stays in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (batch, M, K)  – left tensor   (M=N, K=N)
#   B : (batch, K, N)  – right tensor   (already transposed)
#   C : (batch, M, N)  – result
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, K, N)
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ---------------------------------------------------------------
    # Program IDs  : 0 → batch, 1 → block‑row (M), 2 → block‑col (N)
    # ---------------------------------------------------------------
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A tile (M × K) ----
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- Load Bᵀ tile (K × N) ----
        # B is stored as (batch, K, N) → contiguous across N
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ---- Accumulate using Tensor‑Core dot ----
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ---------------------------------------------------------------
    # Write back C (store as FP16)
    # ---------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (bool/float) – may be all‑ones
        - weights      : dict of torch.Tensor parameters
        - config       : dict containing at least "dim", "hidden_dim" and
                         optional "nomask" (bool)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # -------------------------------------------------------------------
    # 0. Unpack
    # -------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -------------------------------------------------------------------
    # 1. LayerNorm on the feature dimension (FP32)
    # -------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [bs, N, N, dim] FP32

    # -------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – still FP32 for now
    # -------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_w)    # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # -------------------------------------------------------------------
    # 3. Gating projections (sigmoid)
    # -------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -------------------------------------------------------------------
    # 4. Optional mask and gating
    # -------------------------------------------------------------------
    if not config.get("nomask", True):

        left  = left * mask_f
        right = right * mask_f

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5. Cast to FP16 and fold hidden_dim into batch dimension
    # -------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    left_h = left_h.permute(0, 3, 1, 2).contiguous()          # (bs, H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous()        # (bs, H, N, N)

    batch_h = bs * hidden_dim

    left_flat = left_h.view(batch_h, N, N)      # (B*H, N, N) – A
    # Transpose RHS so that the inner‑most dimension (N) is contiguous:
    right_flat = right_h.permute(0, 1, 3, 2).contiguous().view(batch_h, N, N)   # (B*H, K, N)

    # -------------------------------------------------------------------
    # 6. Allocate output for the GEMM
    # -------------------------------------------------------------------
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # -------------------------------------------------------------------
    # 7. Launch the Triton GEMM kernel
    # -------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64                     # larger K‑tile → fewer loop iterations

    b_stride_batch, b_stride_k, b_stride_n = right_flat.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_flat.stride()

    grid = (
        batch_h,                           # dim‑0 = batch
        (N + BLOCK_M - 1) // BLOCK_M,      # dim‑1 = tile rows
        (N + BLOCK_N - 1) // BLOCK_N,      # dim‑2 = tile cols
    )

    batched_gemm_kernel[grid](

        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,   # stride for K dimension (contiguous after transpose)
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # H100 benefits from more warps per block
    )

    # -------------------------------------------------------------------
    # 8. Restore original layout and cast back to FP32
    # -------------------------------------------------------------------
    out_h = out_flat.view(bs, hidden_dim, N, N)                # (bs, H, N, N)
    out_h = out_h.permute(0, 2, 3, 1).contiguous()             # (bs, N, N, H)
