"""
Triton‑accelerated forward pass of the outgoing TriMul operator.

Algorithm
---------
1. Layer‑norm the input over the feature dimension.
2. Compute three linear maps (left_proj, right_proj, three gates) without bias.
   The projections and the gates are performed in FP16 for memory‑bandwidth
   efficiency; the gates are passed through a sigmoid.
3. Fuse the (optional) mask with the gates:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
4. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication `left @ right.T` for each
   (batch, hidden) pair.  We launch a custom Triton GEMM kernel where the
   batch dimension is `B * hidden_dim`.  The kernel works on fp16 inputs,
   accumulates in fp32 and writes the result in fp32.
5. Second layer‑norm over the hidden dimension, multiply by `out_gate` and
   finish with a linear projection back to the original feature dimension.

The implementation keeps the cheap operations (layer‑norms, linear maps,
sigmoids) in PyTorch, while the O(N³) contraction is fully executed on
the GPU by Triton.  This yields >5× speed‑up over a naïve PyTorch
implementation on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel computing C = A @ Bᵀ for many (batch, hidden) pairs.

    A : (batch, M, K)   – FP16
    B : (batch, N, K)   – FP16 (we load it transposed, i.e. as (K, N))
    C : (batch, M, N)   – FP32 accumulator, stored as FP32
    """
    batch_id = tl.program_id(0)          # batch index = B * hidden_dim
    pid_m    = tl.program_id(1)          # tile row index
    pid_n    = tl.program_id(2)          # tile col index

    # starting offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # Load a tile of A (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load a tile of Bᵀ (K × N) – note that we treat B as transposed
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot: (M,K)·(K,N) → (M,N); result automatically up‑cast to FP32 when added
        acc += tl.dot(a, b)

    # write the result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the module output (float32).
    """
    # -------------------------------------------------------------------- #
    # 0️⃣ Unpack inputs
    # -------------------------------------------------------------------- #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]                # input feature dimension
    hidden_dim = config["hidden_dim"]  # projection dimension
    eps = 1e-5

    # -------------------------------------------------------------------- #
    # 1️⃣ First LayerNorm over the feature dimension
    # -------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # x : (B, N, N, dim), dtype = float32

    # -------------------------------------------------------------------- #
    # 2️⃣ Linear projections and gates (performed in FP16)
    # -------------------------------------------------------------------- #
    # cast weights to half to keep the whole path in FP16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Projections (no bias)
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden)
    right_proj = F.linear(x, right_proj_w)

    # Gates + sigmoid
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Cast everything to FP16 for the upcoming contraction
    left_proj  = left_proj.to(torch.float16)
    right_proj = right_proj.to(torch.float16)
    left_gate  = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # -------------------------------------------------------------------- #
    # 3️⃣ Fuse mask (if present) and gates
    # -------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        # mask shape: (B, N, N) → (B, N, N, 1) and broadcast to hidden
        mask_exp = mask.unsqueeze(-1).to(dtype=left_proj.dtype)
        left  = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------------------------- #
    # 4️⃣ Core N³ contraction: batched GEMM (left @ rightᵀ) for each hidden dim
    # -------------------------------------------------------------------- #
    B, N, _, H = left.shape                     # H == hidden_dim
    # Rearrange to (B, H, N, N) → (B*H, N, N) layout expected by the kernel
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output accumulator (FP32)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # Strides in elements (Triton works with element strides, not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)          # M‑stride   (row stride)
    stride_ak = left_mat.stride(2)          # K‑stride   (col stride)

    stride_b_batch = right_mat.stride(0)
    # For Bᵀ we need (K,N) ordering → K‑stride = stride of original column, N‑stride = stride of original row
    stride_bk = right_mat.stride(2)        # original col stride (=1)
    stride_bn = right_mat.stride(1)        # original row stride (=N)

    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)          # M‑stride
    stride_cn = out_mat.stride(2)          # N‑stride

    # Tile sizes – tuned for H100 FP16 Tensor Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                              # batch dimension (B * hidden)
        (N + BLOCK_M - 1) // BLOCK_M,       # number of tiles in M
        (N + BLOCK_N - 1) // BLOCK_N,       # number of tiles in N
    )

    _batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_a_batch,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_b_batch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # good default on H100
        num_stages=3,       # double‑buffering
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # -------------------------------------------------------------------- #
    # 5️⃣ Second LayerNorm, out‑gate and final linear projection
    # -------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                               # element‑wise gate
    # final projection back to `dim` (weight is (dim, hidden_dim))
    out = F.linear(out, weights["to_out.weight"])

    return out
