import torch.nn.functional as F
import triton
import triton.language as tl

def batched_gemm_out_kernel(...):

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; device = input_tensor.device

    # layernorm input (float32)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)

    # Cast to half for remaining ops
    x_h = x.to(torch.float16)

    # cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections, no bias
    left = F.linear(x_h, left_proj_w)  # shape [B,N,N,H]
    right = F.linear(x_h, right_proj_w)

    # Compute gate activations
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # optional mask
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # shape [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # apply gates
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape

    # Permute to (B,H,N,N) for GEMM
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*hidden dimension
    left_flat = left_t.view(-1, N, N)
    right_flat = right_t.view(-1, N, N)

    total = left_flat.shape[0]  # B*H

    out_flat = torch.empty_like(left_flat)  # half

    # configure block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total,
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # hidden dim layer norm

"""
custom_kernel: Triton‑accelerated “outgoing” TriMul (AlphaFold‑3).

Algorithm (high‑level)
----------------------
1. Layer‑norm the input tensor `x` (float32) → shape [B,N,N,D].
2. Cast to fp16 and apply bias‑free linear projections:
       left  = x @ W_left   (→ [B,N,N,H])
       right = x @ W_right  (→ [B,N,N,H])
   Compute three sigmoid gates (left_gate, right_gate, out_gate) with
   the same projected dimension *H*.
3. Optional mask (broadcast over H) is applied to `left`/`right` before gating.
4. Gate the projected tensors element‑wise.
5. Core N³ update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `A @ Bᵀ` for each hidden
   channel *h*.  A dedicated Triton kernel (`batched_gemm_out_kernel`)
   processes all `B·H` matrices in fp16, accumulating into fp32.
6. Hidden‑dim layer‑norm, multiply by `out_gate`, final linear projection
   back to the original dimension *D*.
7. Cast the result back to float32 and return.

All heavy arithmetic (step 5) runs on the GPU in fp16, giving a large
speed‑up over the reference PyTorch implementation while preserving
numerical correctness (final tensor is float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,          # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,          # B: batch, col (k), row (n) strides
    stride_cb, stride_cm, stride_cn,          # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each batch element (including hidden channel) the product
        C = A @ Bᵀ
    where A, B, C have shape (M, K), (N, K), (M, N) respectively.
    All tensors are fp16; accumulation uses fp32.
    """
    pid_batch = tl.program_id(2)               # which (batch * hidden) matrix
    pid_m = tl.program_id(0)                    # block-row of C
    pid_n = tl.program_id(1)                    # block-col of C

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this matrix
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # pointers to the start of the current tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # a, b are fp16; dot produces fp16 which is promoted to fp32 when added
        acc += tl.dot(a, b)

    # write the tile back
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask:        torch.Tensor [B, N, N] (bool/int/float) or None
            - weights:     dict of model parameters (torch.Tensor, float32)
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D] float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all remaining ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to fp16 (bias‑free linear layers)
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate= weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, w_left)   # [B, N, N, H]
    right = F.linear(x_h, w_right)

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w_lgate))
    right_gate = torch.sigmoid(F.linear(x_h, w_rgate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_outgate))

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, N, N] → broadcast to hidden dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7) Apply gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # 8) Prepare tensors for batched GEMM: (B*H, N, N)
    #    Layout: [batch, hidden, i, k] → contiguous -> view
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left_t.view(-1, N, N)   # [B*H, N, N]  fp16
    right_flat = right_t.view(-1, N, N)

    total = left_flat.shape[0]   # B * H

    # output buffer for the matmul (fp16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 9) Triton kernel launch – batched A @ Bᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (row)
        triton.cdiv(N, BLOCK_N),   # blocks along N (col)
        total,                     # one program per (batch * hidden) matrix
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0),   # stride_ab (batch stride)
        left_flat.stride(1),   # stride_am (row stride)
        left_flat.stride(2),   # stride_ak (col stride)
        right_flat.stride(0),  # stride_bb (batch stride)
        right_flat.stride(2),  # stride_bk (col stride for B)
        right_flat.stride(1),  # stride_bn (row stride for B)
        out_flat.stride(0),    # stride_cb (batch stride)
        out_flat.stride(1),    # stride_cm (row stride)
        out_flat.stride(2),    # stride_cn (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10) Reshape back to [B, N, N, H] (fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 11) Hidden‑dim LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)          # [B, N, N, D] fp16

    # ------------------------------------------------------------------
    # 13) Cast to float32 for output (as expected by downstream code)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
