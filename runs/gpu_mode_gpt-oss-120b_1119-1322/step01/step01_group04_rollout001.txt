"""
TriMul (outgoing) – highly‑optimized forward pass.

The heavy part of the module is the contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batch of (batch × hidden_dim) independent GEMMs of size N×N.
This implementation:
  * Applies LayerNorm, linear projections and gating in PyTorch (they are cheap).
  * Packs the (batch, hidden) dimensions and permutes the tensors so that each
    GEMM is stored contiguously.
  * Executes the batched GEMM with a fused Triton kernel that uses Tensor‑Core
    `tl.dot`, mixed‑precision (FP16 inputs, FP32 accumulation) and an
    `@triton.autotune` search over several block sizes / warp counts.
  * Performs the final LayerNorm, output gating and linear projection back to
    the original channel dimension in PyTorch.

The kernel works for any batch size ≤2, sequence length up to 1024, and hidden
dimension 128 (as required by the test suite).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32, "num_warps": 4},
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32, "num_warps": 8},
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 64, "BLOCK_K": 32, "num_warps": 4},
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 64, "BLOCK_K": 32, "num_warps": 8},
            num_stages=2,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def batch_gemm_tri_mul_kernel(
    a_ptr,                # left tensor   [B*H, M, K]  (M=N, K=N)
    b_ptr,                # right tensor  [B*H, K, N]  (right already transposed)
    c_ptr,                # output tensor [B*H, M, N]
    batch_stride_a,       # stride from one (B*H) matrix to the next in A
    stride_am, stride_ak,  # strides for M and K inside A
    batch_stride_b,       # stride from one (B*H) matrix to the next in B
    stride_bk, stride_bn,  # strides for K and N inside B
    batch_stride_c,       # stride from one (B*H) matrix to the next in C
    stride_cm, stride_cn,  # strides for M and N inside C
    M, N, K,                # all equal to seq_len
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B for many independent batches.
    Each batch corresponds to one (batch, hidden) pair of the TriMul.
    A : [batch, M, K]  (FP16)
    B : [batch, K, N]  (FP16)
    C : [batch, M, N]  (FP32 accumulator, stored as FP32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * batch_stride_a
    b_base = b_ptr + batch_id * batch_stride_b
    c_base = c_ptr + batch_id * batch_stride_c

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_base + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        b_ptrs = b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn

        mask_k = k_offset < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass with a fused Triton kernel for the
    N³ contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor – torch.Tensor of shape [B, N, N, C] (float32)
            mask         – torch.Tensor of shape [B, N, N] or None
            weights      – dict of model parameters (float32 tensors)
            config       – dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. TriMul contraction – fused batched GEMM in Triton
    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy matrix multiplication
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Bring the hidden dimension to the second axis and make the right operand transposed.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]   (M=i, K=k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]   (K=k, N=j)

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    # Collapse batch and hidden into a single GEMM batch dimension.
    left_flat = left_perm.view(batch_dim, N, N)   # [BH, M, K]
    right_flat = right_perm.view(batch_dim, N, N) # [BH, K, N]

    # Allocate output (FP32 accumulator)
    out_flat = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides (in elements)
    # ------------------------------------------------------------------
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # stride over M (i)
    stride_ak = left_flat.stride(2)   # stride over K (k)

    batch_stride_b = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # stride over K (k)
    stride_bn = right_flat.stride(2)  # stride over N (j)

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # stride over M (i)
    stride_cn = out_flat.stride(2)    # stride over N (j)

    # ------------------------------------------------------------------
    # Launch the Triton kernel
    # ------------------------------------------------------------------
    grid = (
        batch_dim,
        (N + 127) // 128,   # max BLOCK_M candidate is 256, but 128 is safe upper bound for grid calc
        (N + 127) // 128,
    )
    # The kernel will pick the best BLOCK_M/BLOCK_N/BLOCK_K via autotune.
    batch_gemm_tri_mul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=batch_stride_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=batch_stride_b,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        batch_stride_c=batch_stride_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        # Compile‑time constants (will be overwritten by autotune)
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                # element‑wise gated output
    out = F.linear(out, to_out_weight)  # [B, N, N, C]

    return out
