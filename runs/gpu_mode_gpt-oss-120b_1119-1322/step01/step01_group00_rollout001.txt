    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_b = tl.program_id(axis=2) # batch

    # Offsets for block
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointers for each k block

"""
TriMul (outgoing) – fast forward implementation.

The kernel fuses several cheap operations and replaces the
O(N³) “triangle‑multiplication’’ with a custom Triton batched GEMM.
Key optimisations:
*   Left/right projections and their sigmoid gates are computed in a
    single linear call per side by concatenating the weights
    (2 × hidden_dim output channels).  This cuts the number of
    costly matmul‑like linear layers from 5 → 3.
*   The core `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
    is realised as a batched half‑precision matrix multiplication
    `left @ rightᵀ`.  A Triton kernel processes each (batch × head)
    matrix pair, accumulating in FP32 and writing back FP16.
*   An element‑wise kernel multiplies the output‑side LayerNorm result
    with the “out‑gate’’ tensor – this demonstrates a non‑trivial Triton
    kernel as required.
*   All heavy work (projections, triangle multiplication, final
    projection) is performed in FP16 to exploit H100 Tensor‑cores,
    whilst LayerNorms and the final output are kept in FP32 for
    numerical stability.

The function `custom_kernel` follows the signature required by the
test harness and returns a float‑32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel 1: batched half‑precision GEMM (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                # half*  (left,   shape = [B*H, M, K])
    b_ptr,                # half*  (rightᵀ, shape = [B*H, K, N])
    c_ptr,                # half*  (output, shape = [B*H, M, N])
    M, N, K,              # matrix dimensions
    stride_am, stride_ak, # strides of A (row‑major)
    stride_bk, stride_bn, # strides of B (row‑major, after transposition)
    stride_cm, stride_cn, # strides of C
    batch_stride_a,       # stride to next batch element of A
    batch_stride_b,       # stride to next batch element of B
    batch_stride_c,       # stride to next batch element of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for many independent (M×K)·(K×N) products."""
    pid_m = tl.program_id(0)          # block index in M dimension
    pid_n = tl.program_id(1)          # block index in N dimension
    pid_b = tl.program_id(2)          # batch · head index

    # ------------------------ offsets -------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds loads / stores
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (tiling)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with batch offset
        a_ptrs = a_ptr + pid_b * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Load tiles (mask‑out‑of‑bounds with 0.0)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # FP16 dot → cast to FP32 and accumulate
        acc += tl.float32(tl.dot(a, b))

    # Write back – cast accumulator to FP16
    c = tl.float16(acc)
    c_ptrs = c_ptr + pid_b * batch_stride_c \
            + offs_m[:, None] * stride_cm \
            + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton kernel 2: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # float32* (out_norm flattened)
    b_ptr,                # float32* (out_gate flattened)
    out_ptr,              # float32* (result)
    size,                 # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool or 0/1 float)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict with at least ``dim`` and ``hidden_dim``.
                         May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (over the channel dimension ``dim``)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for all heavy linear work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear proj + gate for the *left* side (concatenated)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    # concat along output dimension: (2*hidden_dim, dim)
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    # single linear call
    left_concat = torch.nn.functional.linear(x_h, left_concat_w)          # [B, N, N, 2*hidden]
    left = left_concat[..., :hidden_dim]                                 # proj
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])              # gate

    # --------------------------------------------------------------
    # 4. Linear proj + gate for the *right* side (concatenated)
    # --------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_concat = torch.nn.functional.linear(x_h, right_concat_w)        # [B, N, N, 2*hidden]
    right = right_concat[..., :hidden_dim]                                # proj
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])            # gate

    # --------------------------------------------------------------
    # 5. Out‑gate (separate linear + sigmoid)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"]
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, out_gate_w.to(torch.float16))
    )   # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 6. Optional mask (applied only to the projections, not to gates)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 7. Apply the left/right gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8. Triangle multiplication (batched GEMM) via Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape            # B, N, N, hidden
    # reshape to (B*hidden, N, N) and create the transposed version for right
    left_perm  = left.permute(0, 3, 1, 2).contiguous()          # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()         # (B, hidden, N, N) == rightᵀ

    left_flat  = left_perm.view(B * hidden_dim, N, N)           # (BH, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)          # (BH, N, N)

    out_flat = torch.empty_like(left_flat)                      # FP16 output buffer

    # strides needed by the kernel
    stride_am = left_flat.stride(1)     # stride for M dimension
    stride_ak = left_flat.stride(2)     # stride for K dimension
    stride_bk = right_flat.stride(1)    # stride for K dimension (after transpose)
    stride_bn = right_flat.stride(2)    # stride for N dimension
    stride_cm = out_flat.stride(1)      # stride for M dimension in C
    stride_cn = out_flat.stride(2)      # stride for N dimension in C

    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_flat.stride(0)
    batch_stride_c = out_flat.stride(0)

    # Blocking configuration – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # blocks in M
        (N + BLOCK_N - 1) // BLOCK_N,          # blocks in N
        B * hidden_dim,                         # number of independent GEMMs
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 9. Output‑side LayerNorm over ``hidden_dim``
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_fp32 = out.to(torch.float32)   # cast before LayerNorm for stability
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )                                   # (B, N, N, hidden)  float32

    # --------------------------------------------------------------
    # 10. Element‑wise multiplication with out_gate (Triton)
    # --------------------------------------------------------------
    # Flatten for the kernel
    out_norm_flat = out_norm.contiguous().view(-1)                # FP32
    out_gate_flat = out_gate.to(torch.float32).contiguous().view(-1)  # FP32
    gated_flat = torch.empty_like(out_norm_flat)

    total_elem = out_norm_flat.numel()
    BLOCK_SIZE = 65536                               # ~64 Ki elements per program
    grid_elem = (triton.cdiv(total_elem, BLOCK_SIZE),)
    elementwise_mul_kernel[grid_elem](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elem,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    gated = gated_flat.view(B, N, N, hidden_dim)     # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 11. Final projection back to original ``dim`` (FP16 -> FP32)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    gated_h = gated.to(torch.float16)
    output_h = torch.nn.functional.linear(gated_h, to_out_w)  # (B, N, N, dim) FP16
    output = output_h.to(torch.float32)

    return output
