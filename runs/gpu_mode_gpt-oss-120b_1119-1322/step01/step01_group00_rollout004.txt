def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])
    # convert to fp16
    x_h = x.to(dtype=torch.float16)

    B,N,N,_ = x_h.shape

    # combine projection weights
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)

    # reshape input
    x_flat = x_h.view(-1, dim)  # (B*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, combined_weight)
    proj = proj.view(B, N, N, 5*hidden_dim)

    # split slices
    left = proj[..., 0:hidden_dim]
    right = proj[..., hidden_dim:2*hidden_dim]
    left_gate_lin = proj[..., 2*hidden_dim:3*hidden_dim]
    right_gate_lin = proj[..., 3*hidden_dim:4*hidden_dim]
    out_gate_lin = proj[..., 4*hidden_dim:5*hidden_dim]

    # apply mask on left/right before gating if needed
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_h
        right = right * mask_h

    # compute gates
    left_gate = torch.sigmoid(left_gate_lin)
    right_gate = torch.sigmoid(right_gate_lin)
    out_gate = torch.sigmoid(out_gate_lin)

        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # elementwise multiply with out_gate (fp32) using triton
    out_norm_flat = out_norm.view(-1).contiguous()
    out_gate_flat = out_gate.to(torch.float32).view(-1).contiguous()
    gated_flat = torch.empty_like(out_norm_flat)

    total = out_norm_flat.numel()
    BLOCK_SIZE = 65536
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    gated = gated_flat.view(B,N,N,hidden_dim)

    # final linear projection
    output = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return output

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, B, H,
    stride_lb, stride_lh, stride_li, stride_lk,
    stride_rb, stride_rh, stride_ri, stride_rj,
    stride_ob, stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_head = tl.program_id(0) # combined batch and head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid_head // H
    h = pid_head % H

    # compute offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range indices
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for outputs
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile
        # left_ptr: (b, h, i, k)
        a_ptrs = left_ptr + b*stride_lb + h*stride_lh + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (transposed): treat right as (b, h, k, j)
        b_ptrs = right_ptr + b*stride_rb + h*stride_rh + offs_k[:, None] * stride_ri + offs_j[None, :] * stride_rj

"""
TriMul (outgoing) – high‑performance Triton implementation

The forward pass consists of:
1. Layer‑Norm over the channel dimension `dim`.
2. A *single* fused linear projection that computes the left/right
   projections as well as the three sigmoid gates in one matmul.  This
   halves the amount of memory traffic over the input tensor.
3. Optional mask application (broadcast on the hidden dimension).
4. Element‑wise gating of the left/right projections.
5. Core “triangle multiplication’’:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of `hidden_dim` independent N×N matrix‑multiplications.
   It is implemented as a tiled GEMM in Triton (one program per (batch,
   hidden) pair) – essentially the same kernel used in Triton‑based
   attention implementations.  All arithmetic is performed in FP16 with
   FP32 accumulation.
6. Layer‑Norm over the hidden dimension.
7. Element‑wise multiplication with the output gate (again a tiny Triton kernel).
8. Final linear projection back to the original `dim` (FP32).

All heavy work is done on the GPU, only small utility kernels are used
for element‑wise fusions.  The final output has shape
[batch, seq_len, seq_len, dim] and dtype float32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # float32*  (out_norm flattened)
    b_ptr,                # float32*  (out_gate flattened)
    out_ptr,              # float32*  (output flattened)
    size,                 # total number of elements
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: batched “triangle” GEMM  (left @ rightᵀ) per hidden slice
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,   # float16*   (B, H, N, N) – left projection
    right_ptr,  # float16*   (B, H, N, N) – right projection
    out_ptr,    # float16*   (B, H, N, N) – accumulator (output)
    N, B, H,    # sizes (int)
    # left strides
    stride_lb, stride_lh, stride_li, stride_lk,
    # right strides (original layout, we will read it transposed)
    stride_rb, stride_rh, stride_ri, stride_rj,
    # out strides
    stride_ob, stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    #  Program IDs
    # ------------------------------------------------------------------
    pid_head = tl.program_id(0)          # combined batch * head index
    pid_i   = tl.program_id(1)          # tile row
    pid_j   = tl.program_id(2)          # tile col

    b = pid_head // H                    # batch index
    h = pid_head %  H                    # hidden‑slice index

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over the reduction (k) dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        #  Load a tile of left  : shape (BLOCK_M, BLOCK_K)
        #  left layout = (b, h, i, k)
        # ------------------------------------------------------------------
        a_ptrs = left_ptr + b*stride_lb + h*stride_lh \
                 + offs_i[:, None] * stride_li \
                 + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ------------------------------------------------------------------
        #  Load a tile of right transposed : shape (BLOCK_K, BLOCK_N)
        #  right layout = (b, h, k, j)   (note the transpose)
        # ------------------------------------------------------------------
        b_ptrs = right_ptr + b*stride_rb + h*stride_rh \
                 + offs_k[:, None] * stride_ri \
                 + offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ------------------------------------------------------------------
        #  Accumulate
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)        # a,b are FP16, acc is FP32

    # ----------------------------------------------------------------------
    #  Store the result (convert back to FP16)
    # ----------------------------------------------------------------------
    c_ptrs = out_ptr + b*stride_ob + h*stride_oh \
             + offs_i[:, None] * stride_oi \
             + offs_j[None, :] * stride_oj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

        weights["left_gate.weight"],          # (hidden_dim, dim)
        weights["right_gate.weight"],         # (hidden_dim, dim)
        weights["out_gate.weight"]            # (hidden_dim, dim)
    ], dim=0).to(torch.float16)               # (5*hidden_dim, dim)

    # flatten input to (B*N*N, dim) for a single matmul
    x_flat = x_h.view(-1, dim)                # (B*N*N, dim)

    proj_flat = torch.nn.functional.linear(x_flat, combined_proj_w)   # (B*N*N, 5*hidden_dim)

    # reshape back to 4‑D tensor
    proj = proj_flat.view(B, N, N, 5 * hidden_dim)   # (B,N,N,5*hidden_dim)

    # split the concatenated result
    left          = proj[..., 0                :hidden_dim]
    right         = proj[..., hidden_dim       :2*hidden_dim]
    left_gate_lin = proj[..., 2*hidden_dim     :3*hidden_dim]
    right_gate_lin= proj[..., 3*hidden_dim     :4*hidden_dim]
    out_gate_lin  = proj[..., 4*hidden_dim     :5*hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left  * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_lin)
    right_gate = torch.sigmoid(right_gate_lin)
    out_gate   = torch.sigmoid(out_gate_lin)

    # ------------------------------------------------------------------
    # 5️⃣ Apply left/right gates (element‑wise)
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Triangle multiplication (batched GEMM) – Triton kernel
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) – contiguous layout required by the kernel
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    out_perm = torch.empty_like(left_perm)               # (B, H, N, N), FP16

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                       # one program per (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,        # tile rows
        (N + BLOCK_N - 1) // BLOCK_N         # tile cols
    )

    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N, B, hidden_dim,
        # left strides
        left_perm.stride(0), left_perm.stride(1), left_perm.stride(2), left_perm.stride(3),
        # right strides (original layout, will be accessed transposed)
        right_perm.stride(0), right_perm.stride(1), right_perm.stride(2), right_perm.stride(3),
        # out strides
        out_perm.stride(0), out_perm.stride(1), out_perm.stride(2), out_perm.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Re‑order back to the original layout (B, N, N, hidden_dim)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden_dim)  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Convert to FP32 for the following LayerNorm
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 8️⃣ Output‑side LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )                                           # (B,N,N,hidden_dim)  FP32

    # ------------------------------------------------------------------
    # 9️⃣ Element‑wise multiplication with the out‑gate (Triton)
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.view(-1).contiguous()
    out_gate_flat = out_gate.to(torch.float32).view(-1).contiguous()
    gated_flat    = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK_SIZE = 65536
    grid_gate = (triton.cdiv(total_elems, BLOCK_SIZE),)

    elementwise_mul_kernel[grid_gate](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE
    )
    gated = gated_flat.view(B, N, N, hidden_dim)   # (B,N,N,hidden_dim)  FP32

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(gated, weights["to_out.weight"])   # (B,N,N,dim)

    return output
```
