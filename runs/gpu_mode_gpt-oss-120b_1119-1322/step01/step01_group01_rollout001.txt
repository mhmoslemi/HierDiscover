"""
custom_kernel – TriMul (outgoing) forward pass
------------------------------------------------
Implements the “outgoing” TriMul operation from AlphaFold3.
The heavy N³·hidden work is performed by a fused Triton batched GEMM
that computes, for each hidden channel d,

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are the linearly projected and gated pair‑wise
representations of the input tensor x ∈ ℝ^{B×N×N×C}.  The kernel
operates on FP16 data (accumulating in FP32) and is launched with
large 2‑D tiles (BLOCK_M×BLOCK_N) and a moderate K‑tile (BLOCK_K).
All cheap element‑wise work (layer‑norm, linear projections,
sigmoid gating, masking) is done with native PyTorch in FP32 or
FP16, and the final projection back to the original C‑dimensional
space is performed after a second layer‑norm and an output gate.

The implementation follows the reference PyTorch code but fuses the
core triple‑multiplicative update, yielding a large speed‑up on
H100 GPUs.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ
#   A : (batch_h, M, K)   – left  tensor  (M=N, K=N)
#   B : (batch_h, N, K)   – right tensor  (rows=j, cols=k)
#   C : (batch_h, M, N)   – output tensor (i,j)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, N, K)   – we read Bᵀ by swapping indices
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for TriMul (A @ Bᵀ)."""
    batch_id = tl.program_id(0)          # which (hidden) batch we are processing
    pid_m    = tl.program_id(1)          # block row index
    pid_n    = tl.program_id(2)          # block col index

    # Compute the absolute coordinates of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current batch
    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    # Boundary masks for the current tile
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (float32 for numerical accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load Bᵀ tile (K × N) – B is stored as (batch, N, K)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_N)

    # Write the result back (store as float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Main entry point required by the problem statement
# --------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
            - mask         : torch.Tensor[bs, N, N] (bool or float)
            - weights      : dict of torch.Tensors (model parameters)
            - config       : dict containing at least 'dim' and 'hidden_dim',
                             optionally 'nomask' (bool)
    Returns:
        torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [bs, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – keep in FP32 for now
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    left  = F.linear(x, left_proj_w)    # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gating vectors (sigmoid, FP32)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4. Optional mask + element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # Broadcast mask over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to half‑precision for the heavy GEMM
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 6. Fold hidden_dim into the batch dimension and reshape for GEMM
    # ------------------------------------------------------------------
    # (bs, N, N, hidden_dim) --> (bs, hidden_dim, N, N) --> (batch_h, N, N)
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim                     # total number of GEMM “batches”

    left_flat  = left_h.view(batch_h, N, N)       # (batch_h, M, K) where M=K=N
    right_flat = right_h.view(batch_h, N, N)      # (batch_h, N, K)

    # Allocate output for the GEMM (half‑precision)
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7. Launch the Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes – chosen for H100 (tuned empirically)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides in element units
    a_stride_batch, a_stride_m, a_stride_k = left_flat.stride()   # (batch_h, N, N) → (N*N, N, 1)
    b_stride_batch, b_stride_n, b_stride_k = right_flat.stride()
    c_stride_batch, c_stride_m, c_stride_n = out_flat.stride()

    # Grid: (batch_h, num_blocks_M, num_blocks_N)
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bk=b_stride_k,
        stride_bn=b_stride_n,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        stride_abatch=a_stride_batch,
        stride_bbatch=b_stride_batch,
        stride_cbatch=c_stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # H100 friendly
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to (bs, N, N, hidden_dim) and cast to FP32
    # ------------------------------------------------------------------
    out_h = out_flat.view(bs, hidden_dim, N, N)          # (bs, hidden_dim, N, N)
    out_h = out_h.permute(0, 2, 3, 1).contiguous()      # (bs, N, N, hidden_dim)
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Post‑processing: hidden‑dim layer‑norm, output gate, final proj.
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"]          # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # final projection, bias‑free

    return out
