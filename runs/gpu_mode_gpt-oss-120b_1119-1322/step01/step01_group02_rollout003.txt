    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

"""
Triton‑accelerated forward pass of the outgoing TriMul (Triangular Multiplicative Update)
operator.

Algorithm
---------
1. Layer‑norm over the input feature dimension.
2. Fuse the three linear projections (left projection, left gate, right projection,
   right gate, out gate) into a *single* matmul:   X_flat (B·N²×dim) @
   W_fusedᵀ (dim×5·hidden) → (B·N²×5·hidden).  The result is split and the gate
   tensors are passed through a sigmoid.
3. Apply the optional mask and element‑wise fuse the gates with the projections.
4. Compute the cubic contraction
      out[i,j] = Σ_k left[i,k]·right[j,k]
   for every head (hidden dimension) using a custom batched GEMM kernel.  Each
   (batch, head) pair is treated as an independent N×N matmul with the right
   operand transposed, so the kernel receives a batch size of B·hidden.
5. Second layer‑norm on the hidden dimension, multiply by the out‑gate, and
   a final linear projection back to the original feature dimension.

All heavy‑weight tensor operations (the N³ contraction) are executed on the GPU
via Triton, while the remaining linear layers are fused into a single
torch.nn.functional.linear call to minimise memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel performing C = A @ Bᵀ for many independent batches.
    A, B, C have shapes (batch, M, K), (batch, N, K), (batch, M, N) respectively.
    The kernel expects A and B in row‑major layout; B is accessed transposed
    (K, N) so that we obtain the required contraction.
    """
    batch_id = tl.program_id(0)          # batch = B * hidden_dim
    pid_m    = tl.program_id(1)          # tile‑row index
    pid_n    = tl.program_id(2)          # tile‑col index

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (tiles of size BLOCK_K)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # Load A tile   (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile as transposed (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write out the result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] containing the module output.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]            # often called c_z

    # --------------------------------------------------------------------- #
    # 1️⃣ LayerNorm over the last feature dimension
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                            # [B, N, N, dim]

    B, N, _, _ = x.shape

    # --------------------------------------------------------------------- #
    # 2️⃣ Fuse the five linear maps (proj/gate/out_gate) into one matmul
    # --------------------------------------------------------------------- #
    # Flatten to (B·N², dim) for the matmul
    x_flat = x.reshape(B * N * N, dim).contiguous()

    # Concatenate the five weight matrices (shape: (5·hidden, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )                                            # (5*hidden, dim)

    # Cast to FP16 for speed; keep everything on the same device
    fused_weight = fused_weight.to(torch.float16)
    x_flat = x_flat.to(torch.float16)

    # Single linear pass
    fused_out = F.linear(x_flat, fused_weight)   # (B·N², 5·hidden)

    # Split the concatenated result
    H = hidden_dim
    left_proj_flat  = fused_out[:, 0:H]
    left_gate_flat  = fused_out[:, H:2 * H]
    right_proj_flat = fused_out[:, 2 * H:3 * H]
    right_gate_flat = fused_out[:, 3 * H:4 * H]
    out_gate_flat   = fused_out[:, 4 * H:5 * H]

    # Apply sigmoid to the three gate tensors
    left_gate_flat = torch.sigmoid(left_gate_flat)
    right_gate_flat = torch.sigmoid(right_gate_flat)
    out_gate_flat = torch.sigmoid(out_gate_flat)

    # Reshape back to (B, N, N, H)
    left_proj = left_proj_flat.view(B, N, N, H)
    left_gate = left_gate_flat.view(B, N, N, H)
    right_proj = right_proj_flat.view(B, N, N, H)
    right_gate = right_gate_flat.view(B, N, N, H)
    out_gate = out_gate_flat.view(B, N, N, H)

    # --------------------------------------------------------------------- #
    # 3️⃣ Fuse mask (if present) and gate values with the projections
    # --------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Cast to FP16 for the cubic contraction (already fp16)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 4️⃣ Core N³ contraction via a custom batched GEMM kernel (A @ Bᵀ)
    # --------------------------------------------------------------------- #
    # Rearrange to (B, H, N, N) → (B·H, N, N) layout required by the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)              # (BH, N, N)
    right_mat = right_perm.view(B * H, N, N)            # (BH, N, N)

    # Output buffer (accumulate in FP32)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # -------------------------- stride bookkeeping ----------------------- #
    stride_a_batch = left_mat.stride(0)    # batch stride (BH)
    stride_am = left_mat.stride(1)         # row stride (M)
    stride_ak = left_mat.stride(2)         # col stride (K)

    stride_b_batch = right_mat.stride(0)   # batch stride (BH)
    stride_bk = right_mat.stride(2)        # =1  (K dimension for transposed view)
    stride_bn = right_mat.stride(1)        # = N  (N dimension)

    stride_c_batch = out_mat.stride(0)     # batch stride
    stride_cm = out_mat.stride(1)          # row stride (M)
    stride_cn = out_mat.stride(2)          # col stride (N)

    # Tile sizes – tuned for H100 FP16 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                   # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,            # rows
        (N + BLOCK_N - 1) // BLOCK_N,            # cols
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to the original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5️⃣ Second LayerNorm, out‑gate, and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                            # still float32

    # Align dtype of out_gate (currently fp16) with out
    out_gate = out_gate.to(out.dtype)
    out = out * out_gate

    # Final linear projection back to the original feature dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
