    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)

    right_gate = torch.sigmoid(right_gate_pre)

"""
Optimized outgoing TriMul kernel.

Algorithm:
1. LayerNorm over feature dim.
2. Fuse left and left_gate linear projections (2*hidden_dim output), and similarly right/right_gate.
3. Compute gates with sigmoid, apply optional mask, multiply gated projections.
4. Perform batched GEMM for triangle multiplication using half precision.
5. LayerNorm over hidden dimension.
6. Triton kernel for elementwise gating with out_gate.
7. Final linear projection back to original dim.

All heavy ops use half precision and PyTorch cuBLAS kernels; only the elementwise
multiply is implemented in Triton.
"""
import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor [B, N, N, C] float32
        - mask: torch.Tensor [B, N, N] (bool or 0/1 float)
        - weights: dict of model parameters (float32)
        - config: dict containing at least ``dim`` and ``hidden_dim``.
                  May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )   # [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2. Cast to float16 for the heavy compute
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse left & left_gate (and right & right_gate) linear projections
    # ------------------------------------------------------------------
    # Left side
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)  # (H, dim)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, dim)

    left_comb = torch.nn.functional.linear(x_h, left_comb_w)   # [B,N,N,2H]
    left = left_comb[..., :hidden_dim]                        # [B,N,N,H]
    left_gate_pre = left_comb[..., hidden_dim:]               # [B,N,N,H]

    left_gate = torch.sigmoid(left_gate_pre)                  # half

    # Right side
    right_proj_w = weights["right_proj.weight"].to(torch.float16)   # (H, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, dim)

    right_comb = torch.nn.functional.linear(x_h, right_comb_w)   # [B,N,N,2H]
    right = right_comb[..., :hidden_dim]                        # [B,N,N,H]
    right_gate_pre = right_comb[..., hidden_dim:]               # [B,N,N,H]

    right_gate = torch.sigmoid(right_gate_pre)                  # half

    # ------------------------------------------------------------------
    # 4. Compute out_gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    out_gate_pre = torch.nn.functional.linear(x_h, out_gate_w) # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)                     # half

    # ------------------------------------------------------------------
    # 5. Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N) -> (B,N,N,1)
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Triangle multiplication via batched GEMM (half precision)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B,N,N,H

    # Permute to (B, H, N, N) and merge batch & head dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # GEMM: each head computes left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7. Output side LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )   # [B, N, N, H], float32

    # ------------------------------------------------------------------
    # 8. Element‑wise gating (out_norm * out_gate) – Triton kernel
    # ------------------------------------------------------------------
    # Flatten tensors for the kernel
    out_norm_flat = out_norm.contiguous().view(-1)               # float32
    out_gate_flat = out_gate.to(torch.float32).contiguous().view(-1)  # float32
    gated_flat = torch.empty_like(out_norm_flat)

    total = out_norm_flat.numel()
    BLOCK_SIZE = 65536  # 64K elements per program

    grid = (triton.cdiv(total, BLOCK_SIZE), )
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total,
        BLOCK_SIZE=BLOCK_SIZE
    )
    gated = gated_flat.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)

    # Cast gated to half for the final linear
    gated_h = gated.to(torch.float16)
    output_h = torch.nn.functional.linear(gated_h, to_out_w)  # [B,N,N,dim], half

    # Cast back to float32 as required
    output = output_h.to(torch.float32)

    return output

Optimized outgoing TriMul kernel (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm over the input feature dimension (C = dim).
2. Fuse the two linear projections per side (proj + gate) into a single GEMM,
   then split the result and apply a sigmoid to the gate half.
3. Compute the out‑gate (separate linear + sigmoid).
4. Optionally mask the projected tensors, then apply left/right gates.
5. Perform the core “triangle multiplication”:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is realized as a large batched half‑precision GEMM:
   (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N).
6. Layer‑norm over the hidden dimension (H = hidden_dim) in float32.
7. Triton kernel for the cheap element‑wise product out_norm * out_gate.
8. Final half‑precision linear projection back to the original channel size.

All heavy tensor work uses float16 (Tensor‑core friendly) and cuBLAS.
Only the element‑wise gating (step 7) is written in Triton, satisfying the
requirement to include a custom kernel.
"""

import triton.language as tl

    b_ptr,                # float32*  (out_gate flattened)
