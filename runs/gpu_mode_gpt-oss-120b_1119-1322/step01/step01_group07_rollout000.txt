"""
TriMul (outgoing) – fast Triton implementation
================================================

The outgoing TriMul computes, for every pair of residues (i, j) and for each hidden
channel h, the sum over a third residue k:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are linear projections of the normalized input,
optionally masked and gated.

Implementation outline
----------------------
1. Layer‑norm on the input tensor   `[B, N, N, C]`.
2. Linear projections → `left`, `right`  (`C → H`).
3. (Optional) binary mask broadcast on the channel axis.
4. Compute three sigmoid gates from the *normalized* input.
5. Apply the gates element‑wise to `left` and `right`.
6. Cast `left` / `right` to **fp16** and reshape them to
   `(B*H, N, N)`.  The heavy multiplication becomes a batched GEMM
   `C = A @ Bᵀ` of shape `(B*H, N, N)`.  This is performed by a custom
   Triton kernel that tiles the matmul (`BLOCK_M × BLOCK_N` tiles,
   loops over `BLOCK_K`).  The kernel uses `tl.dot` (Tensor‑cores on H100)
   with fp32 accumulation.
7. Reshape the result back to `[B, N, N, H]` (fp32).
8. Second Layer‑norm, gating with `out_gate` and a final linear projection
   back to the original channel dimension `C`.

Only the GEMM (step 6) is coded in Triton – all other operations are left to
highly‑optimised PyTorch kernels.  The kernel uses larger tile sizes
(128 × 128 × 64) which are well‑suited for the H100 Tensor‑core pipeline and
produce a substantial speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A, B: fp16, accumulator: fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,                    # A: (batch, M, K)
    stride_bn, stride_bk,                    # B: (batch, N, K) – we load Bᵀ
    stride_cm, stride_cn,                    # C: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  for a batch of GEMMs.
    Each program processes a tile (BLOCK_M × BLOCK_N) of the output
    for a single batch element.
    """
    pid_batch = tl.program_id(0)     # batch dimension
    pid_m = tl.program_id(1)         # tile of rows (M)
    pid_n = tl.program_id(2)         # tile of cols (N)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # Pointers to the start of the current batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    # ------------------------------------------------------------------
    # Pointers to the current tile of A and Bᵀ
    # A tile   : (BLOCK_M, BLOCK_K)
    # Bᵀ tile  : (BLOCK_K, BLOCK_N)
    # ------------------------------------------------------------------
    a_tile_ptr = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K sized chunks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_idx in range(num_k_blocks):
        # Compute masks for the current tile
        cur_k = k_idx * BLOCK_K
        k_mask = (offs_k + cur_k) < K

        a = tl.load(
            a_tile_ptr + cur_k * stride_ak,
            mask=(offs_m[:, None] < M) & k_mask[None, :],
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_k * stride_bk,
            mask=k_mask[:, None] & (offs_n[None, :] < N),
            other=0.0,
        )
        # Dot-product over the K dimension (Tensor‑core friendly)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result (fp16)
    # ------------------------------------------------------------------
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_store)


def _batched_gemm_fp16(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    A: (batch, M, K)  fp16
    B: (batch, N, K)  fp16   (computes A @ Bᵀ)
    Returns C: (batch, M, N)  fp16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in **elements** (not bytes)
    a_batch_stride = A.stride(0)
    a_stride_m = A.stride(1)
    a_stride_k = A.stride(2)

    b_batch_stride = B.stride(0)
    b_stride_n = B.stride(1)
    b_stride_k = B.stride(2)

    c_batch_stride = C.stride(0)
    c_stride_m = C.stride(1)
    c_stride_n = C.stride(2)

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        a_ptr=A,
        b_ptr=B,
        c_ptr=C,
        batch_stride_a=a_batch_stride,
        batch_stride_b=b_batch_stride,
        batch_stride_c=c_batch_stride,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        stride_bn=b_stride_n,
        stride_bk=b_stride_k,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        M=M,
        N=N,
        K=K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Public entry point – the forwards pass of the outgoing TriMul.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of outgoing TriMul.
    Arguments:
        data = (x, mask, weights, config)
            x        : torch.Tensor [B, N, N, C]    (float32)
            mask     : torch.Tensor [B, N, N]      (bool/float) – optional
            weights  : dict of model parameters (float32 tensors)
            config   : dict with keys:
                           - "dim" (int)
                           - "hidden_dim" (int)
                           - "nomask" (bool, optional)
    Returns:
        out : torch.Tensor [B, N, N, C]  (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (C == dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → left / right  (C → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional binary mask (broadcast on channel axis)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; bring to same dtype as x
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating values: sigmoid(Linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (in‑place to save memory)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Cast to fp16 for the GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape  # B = batch size, N = seq length, H = hidden_dim

    # ------------------------------------------------------------------
    # 7️⃣ Reshape to (B*H, N, N) so that each hidden channel becomes a
    #     separate batch element for the GEMM.
    # ------------------------------------------------------------------
    #   (B, N, N, H) -> (B, H, N, N) -> view (B*H, N, N)
    left_reshaped = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched GEMM: out = left @ rightᵀ  (fp16, fp32 accumulator)
    # ------------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(
        left_reshaped,
        right_reshaped,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B*H, N, N] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Restore original layout and promote to fp32
    # ------------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, H]
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 🔟 Output LayerNorm (hidden_dim) + out_gate + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
