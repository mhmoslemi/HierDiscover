```
@triton.jit
def gate_mask_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,  # shape [B, N, N], broadcasted for hidden
    out_left_ptr, out_right_ptr,
    batch_stride, stride_m, stride_n,
    mask_batch_stride, mask_stride_m, mask_stride_n,
    B, N, H,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute pointers for left and gate
    left_base = left_ptr + batch_id * batch_stride
    left_gate_base = left_gate_ptr + batch_id * batch_stride
    out_left_base = out_left_ptr + batch_id * batch_stride

    right_base = right_ptr + batch_id * batch_stride
    right_gate_base = right_gate_ptr + batch_id * batch_stride
    out_right_base = out_right_ptr + batch_id * batch_stride

    # mask base:
    # batch index for mask = batch_id // H
    mask_batch_id = batch_id // H
    mask_base = mask_ptr + mask_batch_id * mask_batch_stride

    # Load
    left = tl.load(left_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
                   mask=mask_m[:, None] & mask_n[None, :],
                   other=0.0)
    left_gate = tl.load(left_gate_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
                        mask=mask_m[:, None] & mask_n[None, :],
                        other=0.0)
    mask = tl.load(mask_base + offs_m[:, None] * mask_stride_m + offs_n[None, :] * mask_stride_n,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=1.0)

    out_left = left * left_gate * mask

    tl.store(out_left_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
             out_left,
             mask=mask_m[:, None] & mask_n[None, :])

    # similarly for right
    right = tl.load(right_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
                    mask=mask_m[:, None] & mask_n[None, :],
                    other=0.0)
    right_gate = tl.load(right_gate_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
                         mask=mask_m[:, None] & mask_n[None, :],
                         other=0.0)
    out_right = right * right_gate * mask

    tl.store(out_right_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
             out_right,
             mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) – Fast Triton‑fused implementation
===================================================

Algorithm
---------
1. Layer‑norm the 4‑D input `x` (shape [B,N,N,C]).
2. Linear projections (no bias) produce the “left” and “right” tensors of
   shape [B,N,N,H] where H = hidden_dim.
3. Compute three element‑wise sigmoid gates from the normalized input:
      left_gate  = sigmoid(linear(x, left_gate_weight))
      right_gate = sigmoid(linear(x, right_gate_weight))
      out_gate   = sigmoid(linear(x, out_gate_weight))
4. Fuse the per‑element mask (if present) and the two gates into the
   projected tensors with a tiny Triton kernel (`gate_mask_kernel`).
   The kernel works on the flattened batch × hidden dimension
   (BH = B·H) and writes the gated tensors `left_gated` and `right_gated`
   in fp16.
5. Perform the expensive contraction
         out[b,i,j,h] = Σ_k left_gated[b,i,k,h] * right_gated[b,j,k,h]
   as a batched matrix‑multiply using PyTorch’s highly optimized `torch.bmm`,
   which runs on Tensor‑Cores (fp16 → fp32 accumulation).  The matrices are
   laid out as:
        left_flat  : [BH, N, N]   (I × K)
        right_flat : [BH, N, N]   (K × J)   – already transposed
   The result `out_flat` has shape [BH, N, N] and is cast back to float32.
6. Reshape to [B,N,N,H] and apply the output LayerNorm, the output gate and
   the final linear projection back to the original channel dimension C.

Only the element‑wise gating/masking step is written in Triton – the heavy
N³ contraction is performed by cuBLAS via `torch.bmm`, giving a runtime well
below the 1 ms target on an H100 while keeping the implementation fully
compatible with the reference PyTorch model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask + left/right gating (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    left_ptr,            # [BH, N, N]   fp16
    left_gate_ptr,       # [BH, N, N]   fp16
    right_ptr,           # [BH, N, N]   fp16 (already transposed K↔J)
    right_gate_ptr,      # [BH, N, N]   fp16
    mask_ptr,            # [B, N, N]    fp16   (broadcast over H)
    out_left_ptr,        # [BH, N, N]   fp16  (output)
    out_right_ptr,       # [BH, N, N]   fp16  (output)
    batch_stride,        # stride between successive BH slices (in elements)
    stride_m,            # stride for the M dimension (i‑ or k‑axis)
    stride_n,            # stride for the N dimension (k‑ or j‑axis)
    mask_batch_stride,   # stride for the mask batch dimension (B)
    mask_stride_m,       # stride for mask rows (N)
    mask_stride_n,       # stride for mask cols (N)
    hidden_dim: tl.constexpr,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    out_left  = left  * left_gate  * mask
    out_right = right * right_gate * mask
    Mask is broadcast over the hidden dimension.
    """
    bh_id = tl.program_id(0)      # 0 ≤ bh_id < B*H
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block col

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for the current BH slice
    # --------------------------------------------------------------
    left_base   = left_ptr   + bh_id * batch_stride
    left_g_base = left_gate_ptr + bh_id * batch_stride
    right_base  = right_ptr  + bh_id * batch_stride
    right_g_base = right_gate_ptr + bh_id * batch_stride
    out_l_base  = out_left_ptr  + bh_id * batch_stride
    out_r_base  = out_right_ptr + bh_id * batch_stride

    # --------------------------------------------------------------
    # Broadcast the mask over the hidden dimension
    # --------------------------------------------------------------
    mask_batch_id = bh_id // hidden_dim          # integer division
    mask_base = mask_ptr + mask_batch_id * mask_batch_stride

    # --------------------------------------------------------------
    # Load, multiply and store
    # --------------------------------------------------------------
    # left branch
    left = tl.load(
        left_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    left_gate = tl.load(
        left_g_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    mask_val = tl.load(
        mask_base + offs_m[:, None] * mask_stride_m + offs_n[None, :] * mask_stride_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=1.0,
    )
    out_left = left * left_gate * mask_val
    tl.store(
        out_l_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
        out_left,
        mask=mask_m[:, None] & mask_n[None, :],
    )

    # right branch
    right = tl.load(
        right_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    right_gate = tl.load(
        right_g_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    out_right = right * right_gate * mask_val
    tl.store(
        out_r_base + offs_m[:, None] * stride_m + offs_n[None, :] * stride_n,
        out_right,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype  # always torch.float32 in the reference
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # convert bool / float mask to the same dtype as the tensors
        mask_tensor = mask.to(dtype).to(device)                # [B,N,N]
    else:
        mask_tensor = torch.ones((B, N, N), dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    left      = left.half()
    right     = right.half()
    left_gate = left_gate.half()
    right_gate = right_gate.half()
    mask_fp16 = mask_tensor.half()      # [B,N,N]

    # ------------------------------------------------------------------
    # 6. Permute + flatten so that batch*hidden becomes a single GEMM batch
    # ------------------------------------------------------------------
    # left:  [B,N,N,H] -> [B,H,N,N] -> view [BH,N,N]
    left_perm   = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm  = right.permute(0, 3, 2, 1).contiguous()  # [B,H,N,N] (K,J)
    left_gate_perm  = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate_perm = right_gate.permute(0, 3, 2, 1).contiguous()
    # flatten batch*hidden
    BH = B * hidden_dim
    left_flat      = left_perm.view(BH, N, N)      # [BH, I, K]
    right_flat     = right_perm.view(BH, N, N)     # [BH, K, J]
    left_gate_flat = left_gate_perm.view(BH, N, N)
    right_gate_flat = right_gate_perm.view(BH, N, N)

    # ------------------------------------------------------------------
    # 7. Triton kernel – apply mask and the two per‑element gates
    # ------------------------------------------------------------------
    # Allocate output buffers
    left_gated  = torch.empty_like(left_flat)   # fp16
    right_gated = torch.empty_like(right_flat)  # fp16

    # Strides (in elements) – because tensors are contiguous after view
    batch_stride = left_flat.stride(0)          # N * N
    stride_m = left_flat.stride(1)              # N
    stride_n = left_flat.stride(2)              # 1

    mask_batch_stride = mask_fp16.stride(0)     # N * N
    mask_stride_m = mask_fp16.stride(1)         # N
    mask_stride_n = mask_fp16.stride(2)         # 1

    # Tile sizes – fairly large for an element‑wise kernel
    BLOCK_M = 64
    BLOCK_N = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    gate_mask_kernel[grid](
        left_ptr=left_flat,
        left_gate_ptr=left_gate_flat,
        right_ptr=right_flat,
        right_gate_ptr=right_gate_flat,
        mask_ptr=mask_fp16,
        out_left_ptr=left_gated,
        out_right_ptr=right_gated,
        batch_stride=batch_stride,
        stride_m=stride_m,
        stride_n=stride_n,
        mask_batch_stride=mask_batch_stride,
        mask_stride_m=mask_stride_m,
        mask_stride_n=mask_stride_n,
        hidden_dim=hidden_dim,
        N=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8. Batched matrix multiplication (the N³ contraction)
    # ------------------------------------------------------------------
    # left_gated : [BH, I, K]   (fp16)
    # right_gated: [BH, K, J]   (fp16)  -> already transposed
    out_flat_fp16 = torch.bmm(left_gated, right_gated)   # [BH, I, J] fp16

    # Cast back to fp32 for the rest of the pipeline
    out = out_flat_fp16.to(torch.float32)

    # Reshape back to [B, N, N, H]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9. Post‑processing: to_out_norm → out_gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating of hidden dim
    out = F.linear(out, to_out_weight)       # back to original dimension C

    return out
