def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr
):
   pid_z = tl.program_id(2)  # batch index
   pid_m = tl.program_id(0)  # row tile idx or block_i
   pid_n = tl.program_id(1)  # col tile idx

   # Block tiling across M dimension:
   # use group scheduling
   num_block_m = tl.cdiv(N, BLOCK_M)
   num_block_n = tl.cdiv(N, BLOCK_N)
   # optionally with group m schedule for better load balancing:
   # compute pid = pid_m * num_block_n + pid_n; reorder? But we already have separate axes.

   # compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask
   mask_m = offs_m < N
   mask_n = offs_n < N

   # accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # iterate over K
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < K

       a_ptrs = a_ptr + pid_z * stride_am + offs_m[:, None] * stride_an + offs_k[None, :] * stride_ak

       # Dot product
       acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
   # end loop

   # Store result
   c_ptrs = c_ptr + pid_z * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_ck

def custom_kernel(data):
    x, mask, weights, config = data

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if (not nomask) and mask is not None:
        mask_float = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_float
        right = right * mask_float

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # Permute for GEMM (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    M = B * hidden_dim

    A = left_perm.view(M, N, N)
    B_mat = right_perm_T.view(M, N, N)

    # Cast to half for kernel
    A = A.to(torch.float16)
    B_mat = B_mat.to(torch.float16)

    # Allocate output
    C = torch.empty((M, N, N), dtype=torch.float32, device=x.device)

    # Strides
    stride_am, stride_an, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bn = B_mat.stride()
    stride_cm, stride_cn, stride_ck = C.stride()

    # Choose block sizes
    # Use 128 or 64 for M,N dims.
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M,
    )

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        M,
        N,
        N,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape to original layout
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – fast forward pass.

The heavy N³ work is a batched matrix multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where left/right ∈ ℝ^{B×N×N×H}.  By folding the hidden dimension H into the
batch dimension we obtain M = B·H independent matrix products of shape
(N,N).  The custom Triton kernel computes these products in FP16 with FP32
accumulators, while the surrounding linear layers, LayerNorms and gates stay
in PyTorch for simplicity.

The kernel layout
    a_ptr : (M, N, K)   – rows i, columns k
    b_ptr : (M, K, N)   – rows k, columns j
    c_ptr : (M, N, N)   – rows i, columns j

Each program instance (pid_z, pid_i, pid_j) processes one tile
BLOCK_M×BLOCK_N of a single (M‑th) matrix product, looping over K in
BLOCK_K‑sized chunks and accumulating with `tl.dot`.  The result is stored
as FP32 and later reshaped back to [B,N,N,H] and passed through the final
LayerNorm, gating and projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C[m,i,j] = Σ_k A[m,i,k] * B[m,k,j]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute many independent N×N matrix products.

    A : (M, N, K)   – rows i, columns k
    B : (M, K, N)   – rows k, columns j
    C : (M, N, N)   – rows i, columns j
    """
    pid_z = tl.program_id(2)          # which independent matrix
    pid_i = tl.program_id(0)          # tile row index
    pid_j = tl.program_id(1)          # tile column index

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[m,i,k]  → shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_z * stride_am \
                 + offs_i[:, None] * stride_an \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B[m,k,j]  → shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_z * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Matrix multiply of the tile
        acc += tl.dot(a, b)   # FP16·FP16 → FP32

    # ------------------------------------------------------------------
    # Write back the tile
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_cm \
             + offs_i[:, None] * stride_cn \
             + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor (bool/float) shape [B, N, N] or None
        * weights_dict : dict of torch.Tensor weights
        * config_dict  : contains ``dim``, ``hidden_dim`` and ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional binary mask (applied before gating)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates (still in FP32) then downcast to FP16 for the heavy GEMM
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    #    left : (B, N, N, H) -> (B, H, N, N) -> (M, N, N)  where M = B*H
    #    right_T : (B, N, N, H) -> (B, H, N, N) (k,j order) -> (M, N, N)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, i, k)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous()      # (B, H, k, j)

    M = B * hidden_dim                                         # total independent matrices

    A = left_perm.view(M, N, N)   # (M, i, k)
    B_mat = right_perm_T.view(M, N, N)   # (M, k, j)

    # Cast to FP16 for the kernel (already FP16 but ensure)
    A = A.to(torch.float16)
    B_mat = B_mat.to(torch.float16)

    # Output placeholder (FP32 accumulator)
    C = torch.empty((M, N, N), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------
    # 6. Strides for the contiguous layout
    # ------------------------------------------------------------------
    stride_am, stride_an, stride_ak = A.stride()   # (M, N, K)
    stride_bm, stride_bk, stride_bn = B_mat.stride()
    stride_cm, stride_cn, stride_ck = C.stride()

    # ------------------------------------------------------------------
    # 7. Tile size selection (covers all test configs)
    # ------------------------------------------------------------------
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        M,                              # independent matrices
    )

    # ------------------------------------------------------------------
    # 8. Launch the Triton kernel
    # ------------------------------------------------------------------
    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        M,
        N,
        N,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 10. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
