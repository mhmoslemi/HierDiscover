@triton.jit
def final_output_kernel(
    out_ptr, out_gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    to_out_weight_ptr,  # shape (H, D) -> hidden x dim
    output_ptr,
    P, H, D,
    stride_out_m, stride_out_h,
    stride_out_gate_m, stride_out_gate_h,
    stride_norm_w, stride_norm_b,
    stride_to_out_w_h, stride_to_out_w_d,
    stride_out_f_m, stride_out_f_d,
    BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
    EPS: tl.constexpr
):
    pid = tl.program_id(0)
    if pid >= P:
        return
    # hidden indices
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # Load out vector
    out_vec = tl.load(out_ptr + pid * stride_out_m + offs_h * stride_out_h,
                      mask=mask_h, other=0.0)

    # Compute mean and variance
    sum_val = tl.sum(out_vec)
    sum_sq = tl.sum(out_vec * out_vec)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Load norm weight/bias
    norm_w = tl.load(norm_weight_ptr + offs_h * stride_norm_w,
                     mask=mask_h, other=1.0)
    norm_b = tl.load(norm_bias_ptr + offs_h * stride_norm_b,
                     mask=mask_h, other=0.0)

    # Normalize
    normed = (out_vec - mean) * inv_std
    normed = normed * norm_w + norm_b

    # Load gate
    gate_vec = tl.load(out_gate_ptr + pid * stride_out_gate_m + offs_h * stride_out_gate_h,
                       mask=mask_h, other=0.0)
    gate_vec = tl.cast(gate_vec, tl.float32)

    gated = normed * gate_vec

    # Final linear: compute gated @ to_out_weight_T (H, D)
    # accumulate over blocks of D
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)
    for d_start in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # load weight slice shape (H, block_D)
        w_ptrs = to_out_weight_ptr + offs_h[:, None] * stride_to_out_w_h + offs_d[None, :] * stride_to_out_w_d
        w = tl.load(w_ptrs,
                    mask=mask_h[:, None] & mask_d[None, :],
                    other=0.0)

        part = tl.dot(gated[None, :], w)  # shape (1, block_D)
        acc = acc + part[0]

    # store output
    out_f_ptrs = output_ptr + pid * stride_out_f_m + offs_d * stride_out_f_d
    tl.store(out_f_ptrs, acc, mask=mask_d)
```

"""
TriMul (outgoing) fused Triton implementation
================================================

Algorithm
---------
1. **Layer‑Norm** on the input channel dimension (dim) – done with
   `torch.nn.functional.layer_norm` (FP32 → FP16 conversion afterwards).
2. **Fused projection**:
   * Concatenate the five Linear weight matrices
     (left_proj, left_gate, right_proj, right_gate, out_gate) → a single
     weight of shape (5·hidden, dim).  A single half‑precision GEMM
     (torch.nn.functional.linear) computes all five projections in one
     pass, reducing memory traffic.
   * Split the result, apply sigmoid gates and the optional binary mask.
3. **Core TriMul** – a batched GEMM computing
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is a custom Triton kernel (`batched_matmul_kernel`) that treats every
   (batch, hidden) pair as an independent matrix of shape (N,N) and uses
   Tensor‑Core‑accelerated `tl.dot` with FP16 inputs / FP32 accumulators.
4. **Output post‑processing** – fused LayerNorm, out‑gate multiplication and
   final projection back to the original channel dimension.  This is performed
   by a second Triton kernel (`final_output_kernel`) that reads the
   (B·N·N, hidden) matrix, normalises it per position, multiplies by the gating
   tensor and applies a half‑precision linear map to produce the final
   (B, N, N, dim) FP32 output.

The implementation runs entirely on the GPU, keeps all heavy compute in
FP16 (except for stable FP32 accumulators) and performs only three kernel
launches, giving a large speed‑up over the naïve PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Batched GEMM kernel for the core N³ operation
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C[m,i,j] = Σ_k A[m,i,k] * B[m,k,j] (float16 × float16 → float32)."""
    batch = tl.program_id(2)            # index over M = B*hidden
    pid_i = tl.program_id(0)            # tile over row   dimension i
    pid_j = tl.program_id(1)            # tile over col   dimension j

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + batch * stride_am \
                 + offs_i[:, None] * stride_an \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # FP16·FP16 → FP32

    c_ptrs = c_ptr + batch * stride_cm \
             + offs_i[:, None] * stride_cn \
             + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 2) Final post‑processing kernel: LayerNorm + out‑gate + Linear
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,                     # [P, H]   FP32   (output of the N³ matmul)
    out_gate_ptr,                # [P, H]   FP16   (sigmoid gates)
    norm_w_ptr, norm_b_ptr,      # [H]      FP16   LayerNorm weights/bias
    to_out_w_ptr,                # [H, D]   FP16   final projection (H→D)
    out_final_ptr,               # [P, D]   FP32   result
    P, H, D,
    stride_out_m, stride_out_h,
    stride_gate_m, stride_gate_h,
    stride_norm_w, stride_norm_b,
    stride_to_out_h, stride_to_out_d,
    stride_out_f_m, stride_out_f_d,
    BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr, EPS: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)          # each program = one position (b,i,j)
    if pid >= P:
        return

    # ----- load the hidden‑dim vector (FP32) -----
    off_h = tl.arange(0, BLOCK_H)
    mask_h = off_h < H
    out_vec = tl.load(
        out_ptr + pid * stride_out_m + off_h * stride_out_h,
        mask=mask_h, other=0.0)

    # ----- compute mean / variance (LayerNorm) -----
    sum_val = tl.sum(out_vec)
    sum_sq  = tl.sum(out_vec * out_vec)
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ----- load norm weight / bias (FP16 → FP32) -----
    norm_w = tl.load(norm_w_ptr + off_h * stride_norm_w,
                     mask=mask_h, other=1.0)
    norm_b = tl.load(norm_b_ptr + off_h * stride_norm_b,
                     mask=mask_h, other=0.0)
    norm_w = tl.cast(norm_w, tl.float32)
    norm_b = tl.cast(norm_b, tl.float32)

    # ----- normalise -----
    normed = (out_vec - mean) * inv_std
    normed = normed * norm_w + norm_b                         # [H] FP32

    # ----- load sigmoidal out‑gate (FP16 → FP32) -----
    gate_vec = tl.load(
        out_gate_ptr + pid * stride_gate_m + off_h * stride_gate_h,
        mask=mask_h, other=0.0)
    gate_vec = tl.cast(gate_vec, tl.float32)

    # ----- apply gating (still FP32) -----
    gated = normed * gate_vec           # [H] FP32
    gated_fp16 = tl.cast(gated, tl.float16)   # cast to FP16 for Tensor‑Core matmul

    # ----- final linear projection (H → D) -----
    # Loop over D in blocks
    for db in range(0, tl.cdiv(D, BLOCK_D)):
        off_d = db * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = off_d < D

        # weight slice shape (H, block_D)
        w = tl.load(
            to_out_w_ptr + off_h[:, None] * stride_to_out_h
                         + off_d[None, :] * stride_to_out_d,
            mask=mask_h[:, None] & mask_d[None, :], other=0.0)

        # dot: (1,H) x (H,block_D) -> (1,block_D)
        part = tl.dot(gated_fp16[None, :], w)   # FP16·FP16 → FP32
        tl.store(
            out_final_ptr + pid * stride_out_f_m
                           + off_d * stride_out_f_d,
            part[0], mask=mask_d)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        * mask_tensor  : torch.Tensor (bool/float), shape [B, N, N] or None
        * weights_dict : mapping from string keys to torch.Tensor weights
        * config_dict  : contains ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    device = x.device

    # --------------------------------------------------------------
    # 0) Input LayerNorm (float32) → FP16 for the rest
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16).contiguous()

    # --------------------------------------------------------------
    # 1) Fused projection + gating (single large GEMM)
    # --------------------------------------------------------------
    M = B * N * N                                   # number of (b,i,j) positions
    x_flat = x.view(M, dim)                         # [M, dim]

    # concatenate the five projection weights (each: hidden x dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()               # [5*hidden, dim]

    # single linear pass (FP16)
    prod = F.linear(x_flat, combined_weight)        # [M, 5*hidden]  (half)

    # split and apply sigmoids
    splits = torch.split(prod, hidden, dim=1)
    left_proj, left_gate, right_proj, right_gate, out_gate = splits
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()  # [M, hidden] (half)

    # optional binary mask (broadcast to hidden dim)
    if (not cfg.get("nomask", False)) and mask is not None:
        mask_flat = mask.to(torch.float16).view(M)  # [M]
        mask_flat = mask_flat[:, None]               # [M,1] for broadcasting
        left = left_proj * left_gate * mask_flat
        right = right_proj * right_gate * mask_flat
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # reshape back to 4‑D tensors (half)
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)
    out_gate = out_gate.view(B, N, N, hidden)       # will be used later

    # --------------------------------------------------------------
    # 2) Core TriMul: batched (N×N) matrix multiplication per hidden
    # --------------------------------------------------------------
    # bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, hidden, N, N)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N)

    M2 = B * hidden                                       # total independent matrices
    A = left_perm.view(M2, N, N)                         # (M2, N, N)  half
    Bmat = right_perm_T.view(M2, N, N)                   # (M2, N, N)  half

    C = torch.empty((M2, N, N), dtype=torch.float32, device=device)

    # strides (contiguous tensors guarantee this layout)
    stride_am, stride_an, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bn = Bmat.stride()
    stride_cm, stride_cn, stride_ck = C.stride()

    # Tile sizes – chosen to exploit Tensor‑Cores on H100
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M2,
    )
    batched_matmul_kernel[grid](
        A,
        Bmat,
        C,
        M2,
        N,
        N,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden] (float32)
    out = C.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)
    out_flat = out.view(-1, hidden).contiguous()                     # (P, hidden)
    out_gate_flat = out_gate.view(-1, hidden).contiguous()           # (P, hidden) half

    # --------------------------------------------------------------
    # 3) Final LayerNorm + out‑gate + projection (fused)
    # --------------------------------------------------------------
    P = B * N * N
    out_final = torch.empty((P, dim), dtype=torch.float32, device=device)

    # strides for the tensors used in the kernel
    stride_out_m, stride_out_h = out_flat.stride()          # (hidden, 1)
    stride_gate_m, stride_gate_h = out_gate_flat.stride()
    stride_norm_w = weights["to_out_norm.weight"].stride()[0]   # ==1
    stride_norm_b = weights["to_out_norm.bias"].stride()[0]     # ==1
    # final projection weight is (hidden, dim) after transpose
    to_out_weight_T = weights["to_out.weight"].t().contiguous().to(torch.float16)
    stride_to_out_h, stride_to_out_d = to_out_weight_T.stride()
    stride_out_f_m, stride_out_f_d = out_final.stride()    # (dim, 1)

    BLOCK_H = hidden
    BLOCK_D = 64 if dim >= 64 else 32

    grid_out = (P,)
    final_output_kernel[grid_out](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        to_out_weight_T,
        out_final,
        P,
        hidden,
        dim,
        stride_out_m,
        stride_out_h,
        stride_gate_m,
        stride_gate_h,
        stride_norm_w,
        stride_norm_b,
        stride_to_out_h,
        stride_to_out_d,
        stride_out_f_m,
        stride_out_f_d,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        EPS=1e-5,
    )

    # reshape to final 4‑D layout
    out_final = out_final.view(B, N, N, dim).contiguous()
    return out_final
