"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (dim).
2. Two bias‑free linear projections → left, right  (shape [B,N,N,H]).
3. Optional binary mask applied to the projections.
4. Compute three sigmoid gates from the normalized input and
   multiply them into left/right and later into the output.
5. Core operation:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication of H·B independent N×N
   products.  It is executed by a custom Triton kernel that
   * stores A/B in FP16,
   * accumulates in FP32,
   * uses 2‑stage double‑buffering,
   * 8 warps per block (256 threads) and flexible tile sizes.
6. Reshape, second LayerNorm, output‑gate and final linear projection
   back to the original channel dimension.

The kernel dominates the runtime (∼N³ work) and has been tuned for the
H100 (TensorCore‑enabled FP16 matmul).  All other ops use the highly‑
optimized PyTorch implementations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: many independent N×N matmuls (A @ Bᵀ)
# A : (M, N, K)   – rows = i, cols = k
# B : (M, K, N)   – rows = k, cols = j   (already transposed)
# C : (M, N, N)   – result
# M = B * hidden_dim,   N = K = seq_len
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    batch_id = tl.program_id(2)          # independent matrices (M)
    pid_i    = tl.program_id(0)          # tile row   (i)
    pid_j    = tl.program_id(1)          # tile col   (j)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Double‑buffering: load the first K‑tile of A and B
    # ------------------------------------------------------------------
    k_blk = tl.arange(0, BLOCK_K)
    offs_k = k_blk
    mask_k = offs_k < K

    a_ptrs = a_ptr + batch_id * stride_am \
                 + offs_i[:, None] * stride_an \
                 + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bn

    a = tl.load(a_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0)          # FP16
    b = tl.load(b_ptrs,
                mask=mask_k[:, None] & mask_j[None, :],
                other=0.0)          # FP16

    # ------------------------------------------------------------------
    # Loop over K tiles
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Prefetch next tile (unless this is the last iteration)
        next_k = offs_k + BLOCK_K
        mask_next_k = next_k < K

        a_next_ptrs = a_ptr + batch_id * stride_am \
                          + offs_i[:, None] * stride_an \
                          + next_k[None, :] * stride_ak
        b_next_ptrs = b_ptr + batch_id * stride_bm \
                          + next_k[:, None] * stride_bk \
                          + offs_j[None, :] * stride_bn

        a_next = tl.load(a_next_ptrs,
                         mask=mask_i[:, None] & mask_next_k[None, :],
                         other=0.0)
        b_next = tl.load(b_next_ptrs,
                         mask=mask_next_k[:, None] & mask_j[None, :],
                         other=0.0)

        # Compute on the current tile
        acc += tl.dot(a, b)          # FP16·FP16→FP32

        # Advance to next tile
        a = a_next
        b = b_next
        offs_k = next_k

    # ------------------------------------------------------------------
    # Write out the tile
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cm \
                 + offs_i[:, None] * stride_cn \
                 + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor  : torch.BoolTensor / torch.FloatTensor shape [B, N, N] or None
        * weights_dict : mapping from string → torch.Tensor (all parameters)
        * config_dict  : must contain ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]  (float32)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (binary 0/1)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask broadcasted to the channel dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)    # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()   # [B,N,N,hidden]

    left  = left * left_gate
    right = right * right_gate

    # Cast projections to fp16 before the heavy N³ work
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM
    #    left   : (B, hidden, i, k)  → (M, N, K)
    #    rightᵀ : (B, hidden, k, j)  → (M, K, N)
    # --------------------------------------------------------------
    #   M = B * hidden   (treated as independent batch dimension)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    M = B * hidden

    A = left_perm.view(M, N, N)      # (M, i, k)  FP16
    B_mat = right_perm.view(M, N, N) # (M, k, j)  FP16

    # Output accumulator (FP32) – final shape (M, N, N)
    C = torch.empty((M, N, N), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------
    # Strides (contiguous tensors guarantee the layout)
    # ------------------------------------------------------------------
    stride_am, stride_an, stride_ak = A.stride()      # (M,N,K)
    stride_bm, stride_bk, stride_bn = B_mat.stride()  # (M,K,N)
    stride_cm, stride_cn, stride_ck = C.stride()      # (M,N,N)

    # ------------------------------------------------------------------
    # Tile size selection – works for all test configurations
    # ------------------------------------------------------------------
    # For very small N we favour 64‑tile to increase occupancy.
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
    BLOCK_K = 32                     # multiple of 32 → good TensorCore utilization

    # Grid dimensions: (tiles_i, tiles_j, batch‑matrices)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M,
    )

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    batched_gemm_kernel[grid](
        A,
        B_mat,
        C,
        M,
        N,
        N,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block (good for H100)
        num_stages=3,         # double‑buffering depth
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to 4‑D layout [B,N,N,hidden]
    # --------------------------------------------------------------
    out = C.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # float32

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B,N,N,dim] (float32)

    return out
