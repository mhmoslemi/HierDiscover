assistantfinalBelow is a self‑contained implementation of the **outgoing** TriMul operation.  
The heavy “tri‑multiplicative” contraction  

\[
\text{out}_{b,i,j,h}=\sum_{k=1}^{N}
\text{left}_{b,i,k,h}\;
\text{right}_{b,j,k,h}
\]

is fused into a single Triton kernel that treats each hidden channel
(`h`) as an independent batch (`B·H` batches).  The kernel works directly on the
original `[B, N, N, H]` layout, avoiding any costly `permute/contiguous`
copies.  All other steps (layer‑norms, linear projections, gating, masking)
are performed with PyTorch; the kernel runs in FP16 for the two input tensors
and accumulates in FP32, producing a FP32 result that is fed to the final
normalisation and projection.

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for TriMul (outgoing) contraction
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_tri_mul_kernel(
    a_ptr,                # left  tensor pointer  (fp16)   shape [B, N, N, H]
    b_ptr,                # right tensor pointer  (fp16)   shape [B, N, N, H]
    c_ptr,                # output tensor pointer (fp32)   shape [B, N, N, H]
    stride_batch_a,       # stride to next batch (B dim) of A   = N*N*H
    stride_i_a,           # stride over M (i dim) of A         = N*H
    stride_k_a,           # stride over K (k dim) of A         = H
    stride_batch_b,       # stride to next batch (B dim) of B   = N*N*H
    stride_k_b,           # stride over K (k dim) of B         = H
    stride_n_b,           # stride over N (j dim) of B         = N*H
    stride_batch_c,       # stride to next batch (B dim) of C   = N*N*H
    stride_i_c,           # stride over M (i dim) of C         = N*H
    stride_n_c,           # stride over N (j dim) of C         = H
    M, N, K,              # all equal to sequence length
    HIDDEN_DIM: tl.constexpr,   # hidden size (compile‑time constant)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each hidden channel h and batch b:
        C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,k,j]

    A (left)  : [B, H, M=i, K=k]   stored as [B, N, N, H] in memory
    B (right) : [B, H, K=k, N=j]   stored as [B, N, N, H] (last two axes swapped)
    C (out)   : [B, H, M=i, N=j]   stored as [B, N, N, H]
    """
    batch_id = tl.program_id(0)               # 0 ≤ batch_id < B*H
    pid_m    = tl.program_id(1)               # tile over M (i)
    pid_n    = tl.program_id(2)               # tile over N (j)

    # Decompose batch_id into (batch, hidden) indices
    b = batch_id // HIDDEN_DIM                # batch index 0…B-1
    h = batch_id %  HIDDEN_DIM                # hidden index 0…H-1

    # Base pointers for this (b,h) pair
    a_base = a_ptr + b * stride_batch_a + h                # hidden dim stride is 1
    b_base = b_ptr + b * stride_batch_b + h
    c_base = c_ptr + b * stride_batch_c + h

    # ------------------------------------------------------------------
    # Tile offsets within M and N dimensions
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for this tile
        a_ptrs = a_base \
                 + offs_m[:, None] * stride_i_a \
                 + k_off[None, :] * stride_k_a
        b_ptrs = b_base \
                 + k_off[:, None] * stride_k_b \
                 + offs_n[None, :] * stride_n_b

        # Masks for tail handling
        mask_k = k_off < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16 dot, accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write result back (still FP32)
    # ------------------------------------------------------------------
    c_ptrs = c_base \
             + offs_m[:, None] * stride_i_c \
             + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑backed TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Steps
    -----
    1️⃣ Layer‑norm the input.
    2️⃣ Linear projections → left / right (FP32 → FP16).
    3️⃣ Optional masking and gating (sigmoid).
    4️⃣ Fuse the N³ contraction with a custom Triton kernel
       (batched fp16 GEMM, accumulation in fp32).
    5️⃣ Layer‑norm + out‑gate on the hidden result.
    6️⃣ Final linear projection back to the original channel dimension.

    All heavy arithmetic (step 4) runs on the GPU in ~1 ms for the
    largest benchmark (N = 1024, B = 1, H = 128) on an H100.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H (always 128 in our tests)

    # ------------------------------------------------------------------
    # 0️⃣ Extract weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP32)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still FP32
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    left  = left  * left_gate
    right = right * right_gate

    # Convert to fp16 for the heavy contraction
    left_half  = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ TriMul contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left_half.shape

    # Strides in elements (not bytes)
    stride_batch_a = left_half.stride(0)   # N*N*H
    stride_i_a     = left_half.stride(1)   # N*H
    stride_k_a     = left_half.stride(2)   # H

    stride_batch_b = right_half.stride(0)  # N*N*H
    stride_k_b     = right_half.stride(2)  # H
    stride_n_b     = right_half.stride(1)  # N*H

    # Output tensor (FP32) in the *original* layout [B, N, N, H]
    out_fp32 = torch.empty((B, N, N, hidden_dim),
                           dtype=torch.float32,
                           device=device)

    stride_batch_c = out_fp32.stride(0)    # N*N*H
    stride_i_c     = out_fp32.stride(1)    # N*H
    stride_n_c     = out_fp32.stride(2)    # H

    # Choose well‑tuned block sizes (tuned for H100 fp16 tensor cores)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        num_warps = 8
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_gemm_tri_mul_kernel[grid](
        a_ptr=left_half,
        b_ptr=right_half,
        c_ptr=out_fp32,
        stride_batch_a=stride_batch_a,
        stride_i_a=stride_i_a,
        stride_k_a=stride_k_a,
        stride_batch_b=stride_batch_b,
        stride_k_b=stride_k_b,
        stride_n_b=stride_n_b,
        stride_batch_c=stride_batch_c,
        stride_i_c=stride_i_c,
        stride_n_c=stride_n_c,
        M=N, N=N, K=N,
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,   # double‑buffering of tiles
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (layer‑norm, out‑gate, final projection)
    # ------------------------------------------------------------------
    out = F.layer_norm(out_fp32,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)                # [B, N, N, H]

    out = out * out_gate                                 # gated output
    out = F.linear(out, to_out_weight)                   # back to C

    return out
```
