"""
custom_kernel: Triton‑accelerated forward pass for the *outgoing* TriMul operator.

Algorithm
---------
1. Layer‑norm `x` over the feature dimension (dim).
2. Linear projections (no bias):
      left_proj  = x @ W_left   ∈ [B,N,N,H]
      right_proj = x @ W_right  ∈ [B,N,N,H]
   Gates (sigmoid):
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
3. Fuse optional mask, projection and gates:
      left  = left_proj  * left_gate  * mask
      right = right_proj * right_gate * mask
   (`mask` is broadcasted on the hidden dimension.)
4. Core cubic contraction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (A @ Bᵀ) where each (b,h) pair is an independent
   N×N matrix multiply. A custom Triton kernel does the heavy work.
5. Layer‑norm the result over the hidden dimension, multiply by `out_gate`,
   and finish with a linear projection back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: (M,N) layout + batch stride
    stride_bk, stride_bn, stride_bb,   # B: (K,N) layout (transposed) + batch stride
    stride_cm, stride_cn, stride_cb,   # C: (M,N) layout + batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel used by TriMul.

    Computes for each batch:
        C = A @ Bᵀ   where A,B ∈ (M,N) and Bᵀ ∈ (K,N) with K=M=N.
    All arithmetic is performed in FP32; inputs are FP16.
    """
    batch_id = tl.program_id(0)          # batch = B * hidden_dim
    pid_m    = tl.program_id(1)          # tile row index
    pid_n    = tl.program_id(2)          # tile column index

    # tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch‑specific base pointers
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile as transposed (K × N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # write‑back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] containing the module output.
    """
    # ------------------------------------------------------------------ #
    # 1️⃣  Unpack inputs
    # ------------------------------------------------------------------ #
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 2️⃣  First LayerNorm over the feature dimension
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------ #
    # 3️⃣  Linear projections (no bias) and gate values (sigmoid)
    # ------------------------------------------------------------------ #
    left_proj  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------ #
    # 4️⃣  Fuse mask and gates
    # ------------------------------------------------------------------ #
    if mask is not None and mask.numel() > 0:
        # mask: [B,N,N] → [B,N,N,1]  (broadcast on hidden dim)
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left  = left_proj  * left_gate  * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left  = left_proj  * left_gate
        right = right_proj * right_gate

    # Cast to FP16 for the D=H contraction (Tensor Cores)
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 5️⃣  Core N³ contraction via a custom batched GEMM kernel
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape  # H == hidden_dim

    # bring hidden dim into the batch axis – (B, H, N, N) → (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * H, N, N)   # (BH, N, N)  fp16
    right_mat = right_perm.view(B * H, N, N)  # (BH, N, N)  fp16

    # Output accumulator (float32)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ---- strides (all tensors are contiguous) ---------------------------------
    # A (left_mat)
    stride_am = left_mat.stride(1)   # row stride   = N
    stride_ak = left_mat.stride(2)   # col stride   = 1
    stride_ab = left_mat.stride(0)   # batch stride = N * N

    # B (right_mat) – we treat it as transposed (K,N)
    stride_bk = right_mat.stride(2)  # K‑stride = 1
    stride_bn = right_mat.stride(1)  # N‑stride = N
    stride_bb = right_mat.stride(0)  # batch stride = N * N

    # C (out_mat)
    stride_cm = out_mat.stride(1)    # row stride   = N
    stride_cn = out_mat.stride(2)    # col stride   = 1
    stride_cb = out_mat.stride(0)    # batch stride = N * N

    # ---- tile sizes (tuned for H100 FP16 TensorCores) -------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                         # batch dimension (B * hidden_dim)
        (N + BLOCK_M - 1) // BLOCK_M, # tile rows
        (N + BLOCK_N - 1) // BLOCK_N, # tile cols
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to the original [B, N, N, H] layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------ #
    # 6️⃣  Second LayerNorm, out‑gate, and final projection back to dim
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
