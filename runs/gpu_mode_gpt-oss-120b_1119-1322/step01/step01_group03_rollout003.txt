    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total,
    )

    batched_gemm_out_half_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,  # M
        N,  # N
        N,  # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    # cast to fp32 for layernorm
    out = out.to(torch.float32)

    # hidden dim layer norm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    # out_gate multiplication
    out = out * out_gate

    # final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a Triton‑accelerated N³ update.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B,N,N,D] (float32).
2. Linear projections (left/right) and three gating linear layers (left_gate,
   right_gate, out_gate).  All linear layers are bias‑free.
3. Optional mask (B,N,N) is broadcast and applied to the projected tensors
   before gating.
4. Apply sigmoid gates:
        left  = left  * left_gate
        right = right * right_gate
   After gating the tensors are cast to float16.
5. Core “tri‑multiplicative” update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ for every hidden
   channel h.  It is performed by the Triton kernel
   `batched_gemm_out_half_kernel` which works on FP16 inputs and
   accumulates in FP32.
6. Cast the result back to FP32, apply a hidden‑dim layer‑norm,
   multiply by `out_gate`, and finally project back to D with a linear
   layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_out_half_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    a_batch_stride, a_row_stride, a_col_stride,
    b_batch_stride, b_col_stride, b_row_stride,
    c_batch_stride, c_row_stride, c_col_stride,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many (batch*hidden) matrices.

    A, B, C are viewed as (total, M, K), (total, N, K) and (total, M, N).
    All tensors are stored row‑major.  A and B are FP16, accumulation is FP32.
    """
    pid_batch = tl.program_id(2)                # which (batch * hidden) matrix
    pid_m = tl.program_id(0)                    # block row of C
    pid_n = tl.program_id(1)                    # block column of C

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this matrix
    a_base = a_ptr + pid_batch * a_batch_stride
    b_base = b_ptr + pid_batch * b_batch_stride
    c_base = c_ptr + pid_batch * c_batch_stride

    # pointers to the current tile (k = 0)
    a_tile_ptr = a_base + offs_m[:, None] * a_row_stride + tl.arange(0, BLOCK_K)[None, :] * a_col_stride
    b_tile_ptr = b_base + tl.arange(0, BLOCK_K)[:, None] * b_col_stride + offs_n[None, :] * b_row_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        # load A tile
        a = tl.load(
            a_tile_ptr + cur_k * a_col_stride,
            mask=mask_m[:, None] & ((tl.arange(0, BLOCK_K) + cur_k) < K),
            other=0.0,
            dtype=tl.float16,
        )
        # load Bᵀ tile (B is accessed as [k, n])
        b = tl.load(
            b_tile_ptr + cur_k * b_col_stride,
            mask=((tl.arange(0, BLOCK_K) + cur_k) < K)[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # dot‑product accumulation (half‑precision multiply, fp32 accumulate)
        acc += tl.dot(a, b)                     # acc is fp32

    # write result (cast back to fp16 for storage)
    c_tile_ptr = c_base + offs_m[:, None] * c_row_stride + offs_n[None, :] * c_col_stride
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask:         torch.Tensor [B, N, N] (bool/int/float) or None
            - weights:      dict of model parameters (torch.Tensor, float32)
            - config:       dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gating linear layers (float32)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()   # [B,N,N,H]
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] → [B,N,N,1] → same dtype as x
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Apply gates (still float32)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 and launch the Triton N³ kernel
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) where the hidden dim is merged into the batch axis
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_h.shape
    left_flat  = left_h.view(-1, N, N)    # [B*H, N, N]
    right_flat = right_h.view(-1, N, N)
    total = left_flat.shape[0]            # B * H

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton grid/block configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 32   # reduction tile size

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i dimension)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j dimension)
        total,                     # one program per (batch * hidden) matrix
    )

    batched_gemm_out_half_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0),   # a_batch_stride
        left_flat.stride(1),   # a_row_stride
        left_flat.stride(2),   # a_col_stride
        right_flat.stride(0),  # b_batch_stride
        right_flat.stride(2),  # b_col_stride
        right_flat.stride(1),  # b_row_stride
        out_flat.stride(0),    # c_batch_stride
        out_flat.stride(1),    # c_row_stride
        out_flat.stride(2),    # c_col_stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H] and cast to FP32 for the remaining ops
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Hidden‑dim LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # element‑wise scale (both float32)

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
