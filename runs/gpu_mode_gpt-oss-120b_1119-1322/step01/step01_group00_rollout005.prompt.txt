import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The implementation follows the reference PyTorch TriMul module:

1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections (left, right) → hidden_dim, optional mask application.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Apply the left/right gates to the projected tensors.
5. Perform the core “triangle multiplication”:
   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h] .
   This is a batch of H independent matrix‑multiplications of size N×N.
   We fuse the H matrices into a single torch.bmm call with batch size B*H.
6. Cast back to float32 and Layer‑norm over the hidden dimension.
7. Multiply the result with the out‑gate – this element‑wise product is
   carried out by a tiny Triton kernel.
8. Final linear projection back to the original channel dimension.

Only the element‑wise gating (step 7) is performed with Triton; everything
else uses highly‑optimized torch/cublas primitives.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # float32*  (out_norm flattened)
    b_ptr,                # float32*  (out_gate flattened)
    out_ptr,              # float32*  (output flattened)
    size,                 # total number of elements
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    # offsets for this program instance
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)   # load out_norm
    b = tl.load(b_ptr + offs, mask=mask)   # load out_gate
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool or 0/1 float)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension (C = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to float16 for the heavy linear / matmul work (half‑precision)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right) → hidden_dim
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is (B, N, N); make it (B, N, N, 1) and cast to half
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Compute the three sigmoid gates from the *original* normalized input
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Apply left/right gates (element‑wise)
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6. Triangle multiplication: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    #   For each hidden slice h we need: out_h = left_h @ right_h.T
    #   By folding the hidden index into the batch dimension we can use
    #   torch.bmm which is highly optimized (uses cuBLAS/PTX kernels).
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # reshape to (B*hidden_dim, N, N)
    left_flat  = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # right needs to be transposed on the last two axes for the “i‑k * j‑k” pattern
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7. Convert to float32 for the remaining LayerNorms
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Output‑side LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )   # [B, N, N, hidden_dim] (float32)

    # ------------------------------------------------------------------
    # 9. Gating with out_gate (element‑wise) – Triton kernel
    # ------------------------------------------------------------------
    # Both tensors are contiguous 1‑D views for the kernel
    out_norm_flat = out_norm.contiguous().view(-1)                      # float32
    out_gate_flat = out_gate.to(torch.float32).contiguous().view(-1)    # float32
    gated_flat    = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK_SIZE = 65536  # ~64 Ki elements per program – fits in shared memory

    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)  # 1‑D grid
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE
    )
    gated = gated_flat.view(B, N, N, hidden_dim)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 10. Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]     # (dim, hidden_dim)
    output = torch.nn.functional.linear(gated, to_out_w)   # [B, N, N, dim]

    return output
