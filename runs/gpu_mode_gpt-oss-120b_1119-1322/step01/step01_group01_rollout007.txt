@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_batch = a_ptr + batch_id * stride_abatch
    b_batch = b_ptr + batch_id * stride_bbatch
    c_batch = c_ptr + batch_id * stride_cbatch

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    x_ptr,  # input x_norm shape (bs, N, N, dim)
    left_proj_w_ptr, right_proj_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    mask_ptr,  # optional mask shape (bs,N,N) maybe None or always passed
    left_out_ptr, right_out_ptr, out_gate_ptr,
    BATCH_H, N, DIM,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_d,
    stride_left_proj_w_hidden, stride_left_proj_w_dim,
    ... similarly for other weight pointers
    stride_mask_batch, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr = 32? due to dim
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    hidden_idx = batch_h_id % hidden_dim

    right_proj_w_row = ...
    left_gate_w_row = ...
    right_gate_w_row = ...
    out_gate_w_row = ...

    # Cast weights to float32 maybe for better compute.
    left_proj_w_row = left_proj_w_row.to(tl.float32)
    ...

    mask_tile = tl.load(mask_ptr + bs_idx*stride_mask_batch + offs_i[:,None]*stride_mask_i + offs_j[None,:]*stride_mask_j,
                        mask=(offs_i[:,None] < N) & (offs_j[None,:] < N),
                        other=1.0).to(tl.float32)  # mask default 1

    # Compute scaling factor: mask_f = mask_tile

    # Set accumulators for left_proj, right_proj, left_gate, right_gate, out_gate
    left_acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    right_acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    right_gate_acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    out_gate_acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        w_left_proj = left_proj_w_row[offs_d]  # (BLOCK_D)

        w_left_gate = ...
        w_right_gate = ...

    left_proj_w_ptr, right_proj_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    mask_ptr,
    BATCH_H: tl.constexpr,
    N: tl.constexpr,
    DIM: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_dim,
    stride_out_batch, stride_out_i, stride_out_j,
    stride_w_hidden, stride_w_dim,
    stride_mask_batch, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, HAS_MASK: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute indices
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute batch and hidden indices
    hidden_idx = batch_id % HIDDEN_DIM
    bs_idx = batch_id // HIDDEN_DIM

    # Load weight rows
    dim_range = tl.arange(0, DIM)
    lp_w = tl.load(left_proj_w_ptr + hidden_idx * stride_w_hidden + dim_range * stride_w_dim, mask=dim_range < DIM).to(tl.float32)
    rp_w = tl.load(right_proj_w_ptr + hidden_idx * stride_w_hidden + dim_range * stride_w_dim, mask=dim_range < DIM).to(tl.float32)
    lg_w = tl.load(left_gate_w_ptr + hidden_idx * stride_w_hidden + dim_range * stride_w_dim, mask=dim_range < DIM).to(tl.float32)
    rg_w = tl.load(right_gate_w_ptr + hidden_idx * stride_w_hidden + dim_range * stride_w_dim, mask=dim_range < DIM).to(tl.float32)
    og_w = tl.load(out_gate_w_ptr + hidden_idx * stride_w_hidden + dim_range * stride_w_dim, mask=dim_range < DIM).to(tl.float32)

    # Load x tile
    # Compose pointer: (bs, i, j, d)
    x_ptrs = x_ptr + bs_idx * stride_x_batch + offs_i[:, None, None] * stride_x_i + offs_j[None, :, None] * stride_x_j + dim_range[None, None, :] * stride_x_dim

    # Load with mask for out-of-range: mask_i and mask_j already and dim_range< DIM (true)

                mask=mask_i[:, None, None] & mask_j[None, :, None] & (dim_range[None, None, :] < DIM),

    left_proj = tl.sum(x * lp_w[None, None, :], axis=2)
    right_proj = tl.sum(x * rp_w[None, None, :], axis=2)
    left_gate_logit = tl.sum(x * lg_w[None, None, :], axis=2)
    right_gate_logit = tl.sum(x * rg_w[None, None, :], axis=2)
    out_gate_logit = tl.sum(x * og_w[None, None, :], axis=2)

    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_logit))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_logit))

        mask_tile = tl.load(mask_ptr + bs_idx * stride_mask_batch + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j,

    tl.store(out_gate_ptrs, out_gate_fp16, mask=store_mask)

"""
custom_kernel – TriMul (outgoing) forward pass

Algorithm
---------
1. Layer‑norm the input tensor on the last dimension (float‑32 for stability).
2. Fuse all five linear projections (left/right projections and three gates) into a
   single matrix multiplication:  X   [B·N·N, C]  @  Wᵀ [C, 5·H]  →  Y [B·N·N, 5·H] .
   This reads the normalized input only **once**, instead of five times.
   The result is split into left_proj, right_proj, left_gate, right_gate, out_gate.
3. Apply sigmoid to the three gate tensors, broadcast the optional mask and
   obtain the gated tensors ``left`` and ``right``.
4. Fold the hidden dimension into the batch dimension:
       left  → (B·H, N, N)   ,   right → (B·H, N, N)
   Both tensors are kept in half‑precision.
5. Compute the core “tri‑multiplicative” update with a custom Triton kernel:
       out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k] .
   This is exactly a batched GEMM C = A @ Bᵀ for each hidden‑channel slice.
6. Reshape the GEMM output back to (B, N, N, H) and convert to float‑32.
7. Layer‑norm over the hidden dimension, modulate by ``out_gate`` and
   project back to the original dimension with the final linear layer.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton batched GEMM:  C = A @ Bᵀ   (A,B : (batch, N, K),  C : (batch, N, N))
# ---------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A : (batch, M, K)
    stride_bk, stride_bn,          # B : (batch, N, K)   (we read Bᵀ)
    stride_cm, stride_cn,          # C : (batch, M, N)
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the current batch
    a = a_ptr + batch_id * stride_ab
    b = b_ptr + batch_id * stride_bb
    c = c_ptr + batch_id * stride_cb

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M×K)
        a_ptrs = a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_tile = tl.load(a_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0).to(tl.float32)

        # Bᵀ tile (K×N)  – B is stored as (batch, N, K)
        b_ptrs = b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N)

    # write result (FP16)
    c_ptrs = c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict, must contain "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0. unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    bs, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    device = x.device
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Layer‑norm on the last dimension (FP32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(x,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)                     # [bs, N, N, dim]  FP32

    # --------------------------------------------------------------
    # 2. Fuse the five linear layers into a single GEMM
    # --------------------------------------------------------------
    # collect weight matrices (all are (hidden, dim) and bias‑free)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # concatenate → (5*hidden, dim)
    fused_w = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0).to(x.dtype)               # FP16 for speed

    # flatten spatial dims → (B·N·N, dim)
    x_flat = x.to(x.dtype).contiguous().view(bs * N * N, dim)

    # fused linear – reads x only once
    fused_out = F.linear(x_flat, fused_w)                # (B·N·N, 5*hidden)
    fused_out = fused_out.view(bs, N, N, 5, hidden)      # (bs, N, N, 5, hidden)

    # split
    left_proj   = fused_out[..., 0, :]    # (bs, N, N, hidden)
    right_proj  = fused_out[..., 1, :]
    left_gate   = torch.sigmoid(fused_out[..., 2, :])
    right_gate  = torch.sigmoid(fused_out[..., 3, :])
    out_gate    = torch.sigmoid(fused_out[..., 4, :])   # (bs, N, N, hidden)

    # --------------------------------------------------------------
    # 3. Apply mask and gating to obtain the tensors that go into the
    #    heavy O(N³·hidden) multiplication.
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (bs, N, N, 1)
        left  = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # cast to FP16 for the gemm
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out_gate = out_gate.to(torch.float16).contiguous()   # keep for later

    # --------------------------------------------------------------
    # 4. Fold hidden dim into the batch dimension
    # --------------------------------------------------------------
    # (bs, hidden, N, N) → (bs*hidden, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()

    batch_h = bs * hidden
    left_flat  = left.view(batch_h, N, N)
    right_flat = right.view(batch_h, N, N)
    out_gate_flat = out_gate.view(batch_h, N, N)

    # --------------------------------------------------------------
    # 5. Triton batched GEMM  C = A @ Bᵀ   (A,B : (batch_h, N, N))
    # --------------------------------------------------------------
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # block sizes – empirically good for H100 across the tested N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements)
    a_batch, a_m, a_k = left_flat.stride()
    b_batch, b_n, b_k = right_flat.stride()
    c_batch, c_m, c_n = out_flat.stride()

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=a_m,
        stride_ak=a_k,
        stride_bk=b_k,
        stride_bn=b_n,
        stride_cm=c_m,
        stride_cn=c_n,
        stride_ab=a_batch,
        stride_bb=b_batch,
        stride_cb=c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                     # H100 performs best with 8 warps
    )

    # --------------------------------------------------------------
    # 6. Reshape back to (bs, N, N, hidden)  (FP32)
    # --------------------------------------------------------------
    out = out_flat.view(bs, hidden, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()   # (bs, N, N, hidden)
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 7. Post‑processing: LayerNorm over hidden, out‑gate, final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # apply out_gate (broadcasted over the hidden dimension)
    out_gate = out_gate_flat.view(bs, hidden, N, N) \
                       .permute(0, 2, 3, 1).contiguous() \
                       .to(torch.float32)               # (bs, N, N, hidden)
    out = out * out_gate

    # final projection back to original dim
    to_out_w = weights["to_out.weight"]   # shape (dim, hidden)
    out = F.linear(out, to_out_w)         # bias‑free

    return out
