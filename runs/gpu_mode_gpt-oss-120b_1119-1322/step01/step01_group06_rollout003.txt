"""
Optimized outgoing TriMul kernel (AlphaFold‑3).

The forward pass consists of:
1. Layer‑norm on the input (float32).
2. Three linear projections (dim → hidden) computed in float16.
   Optional pairwise masking and sigmoid gating are fused in float16.
3. Core “TriMul” product:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM (A @ Bᵀ) with
   A = left, B = right, summed over the sequence dimension.
   The GEMM is executed by a custom Triton kernel that works on
   float16 inputs and accumulates in float32.
4. Layer‑norm on the hidden dimension, output‑gate multiplication,
   and a final linear projection back to `dim`.

All heavy compute (step 3) is performed in the Triton kernel; the rest
uses regular PyTorch kernels.  Using float16 for the large tensors
halves memory bandwidth and, together with an 8‑warp kernel, brings the
runtime well below the 1 ms target on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [BH, N, N]   (float16)
#   C    : [BH, N, N]   (float32)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    A_ptr,               # const float16*   [BH, M, K]
    B_ptr,               # const float16*   [BH, N, K] (will be read transposed)
    C_ptr,               #       float32*   [BH, M, N]
    batch_head,          # int32: total number of (batch * hidden) matrices
    M, N, K,             # int32: matrix dimensions (M = N = seq_len, K = seq_len)
    stride_ah, stride_am, stride_ak,    # strides for A (batch, row, col)
    stride_bh, stride_bn, stride_bk,    # strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,    # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)   # tile over output matrix (M × N)
    bh  = tl.program_id(axis=1)   # which (batch * hidden) matrix

    # ------------------------------------------------------------------
    # Compute tile coordinates
    # ------------------------------------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid %  num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this BH slice
    a_base = A_ptr + bh * stride_ah
    b_base = B_ptr + bh * stride_bh
    c_base = C_ptr + bh * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (summation dimension)
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # Load Bᵀ tile: (BLOCK_K, BLOCK_N)
        # Bᵀ[k, n] = B[n, k]
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )

        # Accumulate dot product; output is float32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write back C
    # ------------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def batch_matmul(A: torch.Tensor, B: torch.Tensor,
                 BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched GEMM: C = A @ Bᵀ
    A, B : [BH, N, N] (float16)
    Returns C : [BH, N, N] (float32)
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    BH, N, K = A.shape
    assert N == K, "TriMul expects square matrices"

    C = torch.empty((BH, N, N), dtype=torch.float32, device=A.device)

    grid_x = ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)
    grid = (grid_x, BH)                     # (tiles over M×N, batch‑head)
    batch_matmul_kernel[grid](
        A,
        B,
        C,
        BH,
        N,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask        : torch.Tensor [B, N, N]        (bool/float, optional)
        - weights     : dict of model weights
        - config      : dict containing 'dim' and 'hidden_dim' and possibly
                         'nomask' flag

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Output gate (float32) – needed after the second LayerNorm
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 3️⃣ Cast to float16 for the heavy linear projections
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Cast projection & gate weights once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (dim → hidden) in float16
    # --------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]  (fp16)
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]  (fp16)

    # --------------------------------------------------------------
    # 5️⃣ Optional pairwise mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 6️⃣ Sigmoid gates (float16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Prepare tensors for the batched GEMM:
    #       left, right : [B, N, N, H]  →  [B, H, N, N] → (BH, N, N)
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    BH = B * H
    A = left.view(BH, N, N)   # float16
    B_mat = right.view(BH, N, N)  # float16

    # --------------------------------------------------------------
    # 8️⃣ Core TriMul – batched GEMM (A @ Bᵀ) using Triton
    # --------------------------------------------------------------
    out_mat = batch_matmul(A, B_mat)          # shape [BH, N, N], dtype float32

    # --------------------------------------------------------------
    # 9️⃣ Restore original layout: [B, N, N, H] (float32)
    # --------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 10️⃣ Second LayerNorm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 11️⃣ Apply output gate (float32)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 12️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
