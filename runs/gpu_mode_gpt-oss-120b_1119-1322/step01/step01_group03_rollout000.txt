"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (high‑level)
----------------------
1. Layer‑norm on the input tensor *x*  (shape [B,N,N,D]).
2. Linear projections (left/right) and three gating linear layers
   (left_gate, right_gate, out_gate).  All linear layers are bias‑free.
   Everything is cast to FP16 to profit from Tensor‑cores.
3. Optional mask (shape [B,N,N]) is broadcast and applied to the
   projected tensors before gating.
4. Apply sigmoid gates:
        left  = (x @ W_left)  * sigmoid(x @ W_left_gate)  * mask
        right = (x @ W_right) * sigmoid(x @ W_right_gate) * mask
5. Core “tri‑multiplicative” update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply
        C = A @ Bᵀ
   for each hidden channel *h*.  We launch a single Triton kernel that
   processes **B·H** independent GEMMs of shape (N×N) in FP16, accumulating
   in FP32 for numerical stability.
6. Layer‑norm on the hidden result, multiply by *out_gate*,
   and finally project back to the original dimension *D*.

Only the cubic N³ update (step 5) is written in Triton – everything else
leverages highly‑optimized PyTorch kernels.  The Triton kernel uses
block‑sizes (64, 64, 64) that fit well into H100 shared memory and trigger
Tensor‑core execution (FP16×FP16→FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with B·H independent matrices
#           C = A @ Bᵀ          (A,B : [batch, N, N]  in FP16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,   # B: batch, col(k), row(n) strides
    stride_cb, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM (A @ Bᵀ) in FP16 with FP32 accumulation."""
    pid_batch = tl.program_id(2)                # which (batch*hidden) matrix
    pid_m = tl.program_id(0)                    # block row of C
    pid_n = tl.program_id(1)                    # block col of C

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # base pointers for this batch element
    # --------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_ab
    b_base = b_ptr + pid_batch * stride_bb
    c_base = c_ptr + pid_batch * stride_cb

    # --------------------------------------------------------------
    # pointers to the start of the current tile (k = 0)
    # --------------------------------------------------------------
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension (tiling)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_off = k_tile * BLOCK_K

        # load tiles of A and Bᵀ (FP16)
        a = tl.load(
            a_tile_ptr + cur_off * stride_ak,
            mask=mask_m[:, None] & ((offs_k + cur_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_off * stride_bk,
            mask=((offs_k + cur_off) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back FP16 result (cast from FP32)
    # --------------------------------------------------------------
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, D], float32
            - mask         : torch.Tensor or None, shape [B, N, N]
            - weights      : dict of parameters (float32 tensors)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Helper to cast weight tensors on‑the‑fly (avoid extra memory)
    def _to_half(t):
        return t.to(torch.float16)

    # Linear projections (no bias)
    left = F.linear(x_fp16, _to_half(weights["left_proj.weight"]))          # [B,N,N,H] fp16
    right = F.linear(x_fp16, _to_half(weights["right_proj.weight"]))

    # Gating linear layers + sigmoid
    left_gate = F.linear(x_fp16, _to_half(weights["left_gate.weight"])).sigmoid()   # fp16
    right_gate = F.linear(x_fp16, _to_half(weights["right_gate.weight"])).sigmoid()
    out_gate = F.linear(x_fp16, _to_half(weights["out_gate.weight"])).sigmoid()    # fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] -> [B,N,N,1] and cast to fp16
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Apply gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 5) Triton kernel for the N³ update (batched GEMM A @ Bᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then collapse batch*hidden
    left_t = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N] fp16
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_t.view(-1, N, N)   # [B*H, N, N] fp16
    right_flat = right_t.view(-1, N, N)

    total = left_flat.shape[0]  # B * H

    # Allocate output buffer (fp16)
    out_flat = torch.empty_like(left_flat)  # [B*H, N, N] fp16

    # ------------------------------------------------------------------
    # Kernel launch configuration – tuned for H100 Tensor‑cores
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j)
        total,                     # one program per (batch*hidden) matrix
    )

    batched_gemm_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K
        left_flat.stride(0),       # stride_ab (batch stride)
        left_flat.stride(1),       # stride_am (row stride)
        left_flat.stride(2),       # stride_ak (col stride)
        right_flat.stride(0),      # stride_bb
        right_flat.stride(2),      # stride_bk (col stride for B)
        right_flat.stride(1),      # stride_bn (row stride for B)
        out_flat.stride(0),        # stride_cb
        out_flat.stride(1),        # stride_cm
        out_flat.stride(2),        # stride_cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H]   (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7) Hidden‑dim layer‑norm (float32), out_gate multiplication, final proj
    # ------------------------------------------------------------------
    # Cast to float32 for layer‑norm (more stable)
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # out_gate was computed in fp16; cast now to float32
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear projection back to original dimension D (float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
