import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_gating_mask_kernel(a_ptr, b_ptr, mask_ptr, out_ptr,
                             total, H, mask_total,
                             BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total

    # load operands
    a = tl.load(a_ptr + offs, mask=valid, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=valid, other=tl.float16(0.0))

    # compute mask index
    mask_idx = offs // H
    mask_valid = mask_idx < mask_total
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=tl.float16(1.0))

    out = a * b * m
    tl.store(out_ptr + offs, out, mask=valid)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul (Triangular Multiplicative Update).

    Steps:
    1. Layer‑norm on the pair representation.
    2. Linear projections to a hidden dimension.
    3. Optional binary mask and three sigmoid gates (left, right, out).
    4. Fuse mask and gating with a tiny Triton kernel (element‑wise multiply).
    5. Batched matrix multiplication C = A @ Bᵀ using torch.bmm (FP16, tensor cores).
    6. Second Layer‑norm, out‑gate, and final linear projection back to the
       original channel dimension.
    """
    # unpack input
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (fp32) → cast to fp16
    # --------------------------------------------------------------
    left_raw = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right_raw = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------
    # 3. Gating values (sigmoid) – stay in fp16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep fp32 for later

    # --------------------------------------------------------------
    # 4. Fuse mask and gates with a tiny Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → fp16; broadcast over hidden_dim inside kernel
        mask_fp = mask.to(torch.float16)
        B, N, _, _ = left_raw.shape

        total = left_raw.numel()                     # B*N*N*H
        mask_total = mask_fp.numel()                 # B*N*N
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        left = torch.empty_like(left_raw)
        right = torch.empty_like(right_raw)

        fused_gating_mask_kernel[grid](
            left_raw, left_gate, mask_fp, left,
            total, hidden_dim, mask_total,
            BLOCK_SIZE=BLOCK,
        )
        fused_gating_mask_kernel[grid](
            right_raw, right_gate, mask_fp, right,
            total, hidden_dim, mask_total,
            BLOCK_SIZE=BLOCK,
        )
    else:
        left = left_raw * left_gate
        right = right_raw * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ   (FP16)
    # --------------------------------------------------------------
    B = left.shape[0]                 # batch size
    N = left.shape[1]                 # sequence length

    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm uses cuBLAS batched GEMM (tensor cores) → fast
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H] and up‑cast to fp32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 6. Second LayerNorm
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7. Apply out‑gate and final projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```

# Triton‑accelerated outgoing TriMul (Triangular Multiplicative Update).
# ---------------------------------------------------------------
# 1️⃣ Layer‑norm on the pair representation.
# 2️⃣ Linear projections to a hidden dimension (fp16).
# 3️⃣ Three sigmoid gates and an optional binary mask.
# 4️⃣ Fuse mask + gate multiplication with a tiny Triton kernel
#    (element‑wise a * b * mask, broadcasting mask over the hidden dim).
# 5️⃣ Batched GEMM C = A @ Bᵀ using torch.bmm (FP16 tensor‑core GEMM).
# 6️⃣ Second Layer‑norm → out‑gate → final linear back to the original dim.
# ---------------------------------------------------------------
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fused_mask_gate_kernel(
    a_ptr,                 # left/right projection      (fp16)
    b_ptr,                 # corresponding gate tensor  (fp16)
    mask_ptr,              # binary mask                (fp16)
    out_ptr,               # destination                (fp16)
    total_elements,        # B * N * N * H
    hidden_dim,            # H
    mask_elements,         # B * N * N
    BLOCK: tl.constexpr,   # compile‑time block size
):
    """Element‑wise: out = a * b * mask, where mask is broadcast over H."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                # linear indices
    valid = offs < total_elements

    # load a and b (both fp16)
    a = tl.load(a_ptr + offs, mask=valid, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=valid, other=tl.float16(0.0))

    # mask index = linear_index // H  (broadcast mask over the hidden dim)
    mask_idx = offs // hidden_dim
    mask_ok = mask_idx < mask_elements
    m = tl.load(mask_ptr + mask_idx, mask=mask_ok, other=tl.float16(1.0))

    out = a * b * m
    tl.store(out_ptr + offs, out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float, optional)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
                         * "dim" (int)
                         * "hidden_dim" (int)
                         * "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (still fp32) → cast to fp16 for the GEMM
    # ------------------------------------------------------------------
    left_raw  = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden_dim]
    right_raw = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Gating values (sigmoid) – stay in fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))        # fp32, kept for later

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask + gates with a tiny Triton kernel (if a mask is present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → fp16, will be broadcast over hidden_dim inside the kernel
        mask_fp = mask.to(torch.float16)                      # [B,N,N]
        B, N, _, _ = left_raw.shape

        total = left_raw.numel()            # B * N * N * hidden_dim
        mask_total = mask_fp.numel()        # B * N * N
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        # allocate output tensors
        left  = torch.empty_like(left_raw)
        right = torch.empty_like(right_raw)

        # left = left_raw * left_gate * mask
        _fused_mask_gate_kernel[grid](
            left_raw,
            left_gate,
            mask_fp,
            left,
            total,
            hidden_dim,
            mask_total,
            BLOCK=BLOCK,
        )
        # right = right_raw * right_gate * mask
        _fused_mask_gate_kernel[grid](
            right_raw,
            right_gate,
            mask_fp,
            right,
            total,
            hidden_dim,
            mask_total,
            BLOCK=BLOCK,
        )
    else:
        # no mask → simple elementwise multiplication
        left  = left_raw * left_gate
        right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out = left @ rightᵀ   (FP16, tensor‑core)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden_dim, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm maps to cuBLAS “strided‑batched GEMM”, which on H100
    # utilizes tensor cores and is far faster than the reference Triton GEMM.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16

    # reshape back to [B, N, N, hidden_dim] and cast to fp32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
