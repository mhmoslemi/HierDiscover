def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', False)

    # Cast weights to appropriate dtype maybe half for speed, but we can rely on autocast.

    # put in same dtype as input (float32)
    # Use torch.autocast for half.

    with torch.autocast(device_type='cuda', dtype=torch.float16):
        # LN
        x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
        # shape [B,N,N,dim]

        # Linear
        left = F.linear(x, weights['left_proj.weight'])
        right = F.linear(x, weights['right_proj.weight'])

        if not nomask:
            mask_f = mask.unsqueeze(-1).to(x.dtype)
            left = left * mask_f; right = right * mask_f

        # Gating
        left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
        right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
        out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

        left = left * left_gate
        right = right * right_gate

        # Permute for batch matmul
        left_perm = left.permute(0,3,1,2).contiguous()
        right_perm = right.permute(0,3,1,2).contiguous()
        # Transpose right for matmul
        right_T = right_perm.transpose(2,3) # shape [B,H,N,N]

        # flatten batch*hidden
        B, H, N, _ = left_perm.shape
        lhs = left_perm.view(B*H, N, N)
        rhs = right_T.view(B*H, N, N)

        # batch matmul using torch.bmm
        out_mat = torch.bmm(lhs, rhs)  # shape [BH,N,N] half

        # reshape back to [B,N,N,H]
        out = out_mat.view(B, H, N, N).permute(0,2,3,1)

        # Second LN
        out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

        # gating and projection
        out = out * out_gate
        out = F.linear(out, weights['to_out.weight'])
    # ensure fp32 result
    return out.float()

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    c = a * b
    tl.store(out_ptr + offs, c, mask=mask)

def custom_kernel(data):
    """
    Optimized outgoing TriMul forward pass.
    The heavy bulk of the computation (the N‑by‑N matrix product summed
    over the sequence dimension) is performed as a half‑precision batched
    GEMM using torch.bmm, which exploits Tensor‑Core acceleration on
    H100.  A tiny Triton kernel is used to fuse the element‑wise gating
    (and optional masking) of the projected tensors, thereby satisfying
    the requirement that part of the computation runs in a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------- 1. LayerNorm (float32) -------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim] float32

    # Cast to half for the heavy work
    x_h = x.half()

    # ------- 2. Cast weights to half for the linear ops -------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    # The final projection weight stays float32 (used later)

    # ------- 3. Linear projections and gates (half) -------
    left = F.linear(x_h, left_proj_w)          # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # ------- 4. Fuse gating (and optional mask) via Triton kernel -------
    total = left.numel()                     # B * N * N * hidden_dim
    BLOCK_SIZE = 1024                        # reasonable size, powers of 2

    # Allocate contiguous output buffers for the gated tensors
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    grid = (triton.cdiv(total, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        left,
        left_gate,
        left_gated,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    elementwise_mul_kernel[grid](
        right,
        right_gate,
        right_gated,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # Optional mask broadcasting (mask is float32/float16)
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden dim
        mask_f = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_gated = left_gated * mask_f
        right_gated = right_gated * mask_f

    # Release unneeded tensors early
    del left, right, left_gate, right_gate, x_h

    # ------- 5. Core N×N×N batched GEMM (half‑precision) -------
    # Rearrange to [B, hidden, N, N] to flatten batch*hidden for bmm
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()
    right_perm = right_gated.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two dims of right

    out_f = out_f * out_gate.float()

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # const half*  (or float* if called with fp32)
    b_ptr,                # const half*
    out_ptr,              #       half*
    size,                 # int32 total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """
    Simple element‑wise multiplication kernel.
    Computes out[i] = a[i] * b[i] for i < size.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Optimized outgoing TriMul forward pass.
    --------------------------------------------------------------
    1. Layer‑norm the input (float32).
    2. Cast to FP16 and perform all linear projections / sigmoid gates.
    3. Fuse the element‑wise gate multiplication (and optional mask)
       with a tiny Triton kernel.
    4. Core N×N×N update is a half‑precision batched GEMM
       (torch.bmm) that leverages H100 Tensor‑Cores.
    5. Convert back to FP32, apply a second LayerNorm, output gate,
       and final projection.
    --------------------------------------------------------------
    All heavy arithmetic is done in FP16; the final result is float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                         # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 – everything else will run in half precision
    # ------------------------------------------------------------------
    x_h = x.half()

    # ------------------------------------------------------------------
    # 3. Cast model weights needed for the FP16 part
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # ------------------------------------------------------------------
    # 4. Linear projections & gates (all half‑precision)
    # ------------------------------------------------------------------
    left      = F.linear(x_h, left_proj_w)                               # [B, N, N, H]
    right     = F.linear(x_h, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))               # [B, N, N, H]
    right_gate= torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate  = torch.sigmoid(F.linear(x_h, out_gate_w))                # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Fuse element‑wise gate multiplication via Triton
    # ------------------------------------------------------------------
    total_elems = left.numel()               # B * N * N * H
    BLOCK_SIZE = 1024                         # 1 K elements per program
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    left_gated  = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # left_gated = left * left_gate
    elementwise_mul_kernel[grid](
        left,
        left_gate,
        left_gated,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    # right_gated = right * right_gate
    elementwise_mul_kernel[grid](
        right,
        right_gate,
        right_gated,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # Optional pairwise mask (broadcast over the hidden dimension)
    if not nomask and mask is not None:
        mask_f = mask.to(x_h.dtype).unsqueeze(-1)    # [B, N, N, 1] half
        left_gated  = left_gated * mask_f
        right_gated = right_gated * mask_f

    # ------------------------------------------------------------------
    # 6. Core batched GEMM (left @ rightᵀ) – half precision, Tensor‑Core fast
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for a single bmm launch per (B×H) matrix
    left_perm  = left_gated.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_gated.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # we need Rightᵀ in the inner two dimensions
    right_t = right_perm.transpose(2, 3).contiguous()          # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    lhs = left_perm.view(B * H, N, N)          # [BH, N, N] half
    rhs = right_t.view(B * H, N, N)           # [BH, N, N] half

    out_half = torch.bmm(lhs, rhs)            # [BH, N, N] half

    # restore original layout: [B, N, N, H]
    out = out_half.view(B, H, N, N).permute(0, 2, 3, 1)   # half

    # ------------------------------------------------------------------
    # 7. Second LayerNorm (float32) + output gating + final projection
    # ------------------------------------------------------------------
    out_f = out.float()   # back to FP32 for the final small ops

    out_f = F.layer_norm(
        out_f,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_f = out_f * out_gate.float()          # apply output gate (now FP32)

    # final linear projection back to `dim` (float32 weight)
    out_f = F.linear(out_f, weights["to_out.weight"])

    return out_f
