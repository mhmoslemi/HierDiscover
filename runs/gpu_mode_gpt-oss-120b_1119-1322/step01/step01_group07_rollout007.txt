"""
TriMul outgoing update – Triton accelerated implementation.

The kernel computes, for every batch b, position i, j and hidden channel d,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected pair representations after gating
and optional masking.  The computation is a batched matrix multiplication
C = A @ Bᵀ for each (b,d) pair, with A = left[...,d]  (shape N×N)
and B = right[...,d] (shape N×N).  The Triton kernel works directly on the
original layout [B,N,N,H] (no costly reshape/copy) by flattening the
(b,d) dimensions into a single program‑id dimension.  All heavy arithmetic
is performed in fp16 with fp32 accumulation; the rest of the module
(LayerNorms, gating, final projection) stays in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (batched over B * hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # strides (in elements, not bytes)
    batch_stride_a, stride_am, stride_ak, channel_stride_a,
    batch_stride_b, stride_bn, stride_bk, channel_stride_b,
    batch_stride_c, stride_cm, stride_cn, channel_stride_c,
    # problem size
    M, N, K,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each pair (batch, hidden) without any
    intermediate copies.  The input tensors have shape
        A : (B, N, N, H)   -> (batch, M, K)   (M = N, K = N)
        B : (B, N, N, H)   -> (batch, N, K)   (will be read as transposed)
    The output tensor C has the same layout as the inputs:
        (B, N, N, H)
    """
    pid = tl.program_id(0)          # flattened (batch, hidden)
    pid_m = tl.program_id(1)        # tile row
    pid_n = tl.program_id(2)        # tile column

    # ------------------------------------------------------------------
    # Resolve batch and hidden dimension indices from the flattened id
    # ------------------------------------------------------------------
    batch_idx = pid // HIDDEN_DIM
    hidden_idx = pid % HIDDEN_DIM

    # Base pointers for this (batch, hidden) pair
    a_batch = a_ptr + batch_idx * batch_stride_a + hidden_idx * channel_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b + hidden_idx * channel_stride_b
    c_batch = c_ptr + batch_idx * batch_stride_c + hidden_idx * channel_stride_c

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to tiles of A and Bᵀ
    a_tile_ptr = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_off = k * BLOCK_K

        a = tl.load(
            a_tile_ptr + k_off * stride_ak,
            mask=(offs_m[:, None] < M) & ((offs_k[None, :] + k_off) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + k_off * stride_bk,
            mask=((offs_k[:, None] + k_off) < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write result back (store as fp16)
    # ------------------------------------------------------------------
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_store)


def _tri_mul_gemm_outgoing(
    left: torch.Tensor,
    right: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    `left` and `right` have shape [B, N, N, H] and dtype torch.float16.
    Returns a tensor of the same shape and dtype containing the
    multiplicative update.
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    # output tensor (same layout as inputs)
    out = torch.empty_like(left)

    # strides in number of elements (Triton expects element strides)
    a_batch_stride = left.stride(0)
    a_stride_m = left.stride(1)          # i dimension
    a_stride_k = left.stride(2)          # k dimension
    a_channel_stride = left.stride(3)    # hidden dimension ( = 1 for contiguous)

    b_batch_stride = right.stride(0)
    # note: we treat B as (batch, N_j, N_i) → row stride = original j stride,
    # col stride = original i stride
    b_stride_n = right.stride(2)         # original j stride (H)
    b_stride_k = right.stride(1)         # original i stride (N*H)
    b_channel_stride = right.stride(3)   # = 1

    c_batch_stride = out.stride(0)
    c_stride_m = out.stride(1)
    c_stride_n = out.stride(2)
    c_channel_stride = out.stride(3)

    grid = (
        B * H,                                     # flattened (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        batch_stride_a=a_batch_stride,
        stride_am=a_stride_m,
        stride_ak=a_stride_k,
        channel_stride_a=a_channel_stride,
        batch_stride_b=b_batch_stride,
        stride_bn=b_stride_n,
        stride_bk=b_stride_k,
        channel_stride_b=b_channel_stride,
        batch_stride_c=c_batch_stride,
        stride_cm=c_stride_m,
        stride_cn=c_stride_n,
        channel_stride_c=c_channel_stride,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=H,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator (AlphaFold3 style).

    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N] (float32/bool) – may be None
            weights      : dict of model parameters (float32 tensors)
            config       : dict with keys "dim", "hidden_dim",
                           optional "nomask" (bool)

    Returns:
        out : torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear / gating ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on channel dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gating values (sigmoid of linear layers, FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # ------------------------------------------------------------------
    # 6) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core TriMul multiplication – Triton kernel (no copies)
    # ------------------------------------------------------------------
    out_fp16 = _tri_mul_gemm_outgoing(
        left,
        right,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )   # shape [B, N, N, hidden_dim], dtype fp16

    # ------------------------------------------------------------------
    # 8) Convert back to FP32 for the remaining lightweight ops
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 10) Final gating and projection back to the original channel dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
