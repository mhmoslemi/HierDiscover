"""
Simple yet effective optimizer for the convolution‑based objective.

The evaluation function is scale‑invariant:
    score = 2·n·max(a*a) / (sum(a))**2
so we can normalize any candidate to have sum = 1 without changing its
score.  The global optimum for length 2 is known analytically:
    a = (1/3 , 2/3)   →   score = 16/9 ≈ 1.777777…
which already beats the constant construction (score = 2).

The search proceeds as follows

1.  Initialise the current best with the optimal length‑2 vector.
2.  For a range of lengths (3 … 30) perform many cheap hill‑climbing
    moves on random Dirichlet starts:
        – pick two distinct positions i, j,
        – transfer a small fraction of probability from i to j,
        – renormalise to keep the sum = 1,
        – accept the move if the score improves (or with a tiny
          probability to escape flat plateaus).
    Whenever a better sequence is found it replaces the incumbent.
3.  The whole procedure stops after the allotted time (default ≈ 990 s)
    and returns the best sequence discovered.

All arithmetic is performed with NumPy; no calls to the expensive
``evaluate_sequence`` are made during the search (the final answer is
still a valid input for it).  The algorithm respects the security checks
in ``evaluate_sequence`` – every entry is non‑negative, finite and ≤ 1000.

Even if the random hill‑climb does not discover a better vector than the
analytically optimal length‑2 construction, the returned sequence is
guaranteed to be strictly better than the constant baseline, thereby
achieving a score < 2.0.
"""

import math
import time
import numpy as np


def _objective(seq: np.ndarray) -> float:
    """Exact copy of the official objective (scale‑invariant)."""
    n = seq.size
    if n == 0:
        return math.inf
    # protection that the official evaluator performs
    seq = np.maximum(0.0, np.minimum(1000.0, seq.astype(float)))
    s = seq.sum()
    if s < 0.01:
        return math.inf
    conv = np.convolve(seq, seq)          # linear convolution
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(budget_s: float = 990.0, **kwargs) -> list[float]:
    """
    Search for a sequence with a low convolution‑peak score.
    The function returns a plain Python list of non‑negative floats.
    """
    start = time.monotonic()
    # safety: enforce a reasonable time budget
    try:
        limit = float(budget_s)
    except Exception:
        limit = 990.0
    limit = min(limit, 990.0)
    deadline = start + max(1.0, limit)

    # -----------------------------------------------------------------
    # 1) Analytic optimum for length 2  (score = 16/9 ≈ 1.777777…)
    # -----------------------------------------------------------------
    best_seq = np.array([1.0 / 3.0, 2.0 / 3.0], dtype=float)
    best_score = _objective(best_seq)
    print(f"initial length‑2 optimum, score = {best_score:.17g}")

    # -----------------------------------------------------------------
    # 2) Random hill‑climbing on longer lengths
    # -----------------------------------------------------------------
    # lengths to explore – short enough to keep each evaluation cheap
    lengths = list(range(3, 31))

    # parameters for the stochastic local search
    max_iters_per_len = 5000          # total moves tried for each length
    step_fraction = 0.05              # transfer 5 % of the source mass each move
    accept_worse_prob = 0.02          # tiny chance to accept a worsening move

    for n in lengths:
        if time.monotonic() > deadline:
            break

        # a few random restarts per length (here just one for speed)
        a = np.random.dirichlet(np.ones(n)).astype(float)   # sum = 1
        cur_score = _objective(a)

        # keep the best seen for this length
        if cur_score < best_score:
            best_seq = a.copy()
            best_score = cur_score
            print(f"  length={n} initial random score = {best_score:.17g}")

        it = 0
        while it < max_iters_per_len and time.monotonic() < deadline:
            # pick two distinct indices
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * step_fraction
            if delta <= 0.0:
                it += 1
                continue

            # propose move: move `delta` from i to j
            a[i] -= delta
            a[j] += delta

            # renormalise to keep the sum = 1 (numerical drift)
            total = a.sum()
            if total <= 0.0:
                # should never happen, but recover harmlessly
                a[i] += delta
                a[j] -= delta
                it += 1
                continue
            a /= total

            new_score = _objective(a)

            if new_score < cur_score or np.random.rand() < accept_worse_prob:
                # accept (improvement or very unlikely worsening)
                cur_score = new_score
                if new_score < best_score:
                    best_seq = a.copy()
                    best_score = new_score
                    print(f"improved: length={n}, iter={it}, score={best_score:.17g}")
            else:
                # reject: undo the move
                a[j] -= delta
                a[i] += delta
                a /= a.sum()          # renormalise again (exactly restores previous state)

            it += 1

    # -----------------------------------------------------------------
    # 3) Return the best candidate found
    # -----------------------------------------------------------------
    # Ensure the result complies with the evaluator's guards
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # final sanity check
    final_score = _objective(best_seq)
    print(f"final best length={len(best_seq)}, score={final_score:.17g}")

    return [float(x) for x in best_seq]
