def arcsine_weighted_sequence(n):
    # compute x in (0,1)
    i = np.arange(n, dtype=np.float64)
    # shift by 0.5 to avoid zero exactly
    x = (i + 0.5) / n
    # arcsine density proportional to 1/sqrt(x*(1-x))
    w = 1.0 / np.sqrt(x * (1.0 - x))
    # replace any inf (should not happen)
    w[np.isinf(w)] = 0.0
    w = np.nan_to_num(w, nan=0.0)
    # normalize sum to 1
    total = np.sum(w)
    if total == 0:
        w = np.ones(n) / n
    else:
        w = w / total
    return w

- Start from an arcsine-shaped probability distribution (a_i ∝ 1/sqrt(i*(n-1-i))) which makes the self‑convolution almost flat.
- Perform a timed local search that alternates between
    * a targeted mass‑transfer move that directly reduces the entry of the
      convolution that attains the current maximum, and
    * the generic block‑perturbation hill‑climber used in the reference solution.
- Accept a move only if the official `evaluate_sequence` score improves.
- Return the best sequence found before the time budget expires.
"""

import time
import numpy as np

def arcsine_weighted_sequence(n):
    """Return a non–negative sequence of length n whose shape approximates the
    arcsine distribution (density ∝ 1/√(x(1‑x))). The result is normalized to
    have L1 norm 1 and all entries lie well below the evaluator’s 1000‑cap."""
    i = np.arange(n, dtype=np.float64)
    # Normalised coordinate in (0,1) using a half‑step offset to avoid division by zero.
    x = (i + 0.5) / n
    # Avoid division by zero at the boundaries – the half‑step already guarantees
    # a strictly positive argument but we protect against numerical artefacts.
    with np.errstate(divide='ignore', invalid='ignore'):
        w = 1.0 / np.sqrt(x * (1.0 - x))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        w = np.ones(n, dtype=np.float64) / n
    else:
        w = w / total
    return w

def make_block_perturbation(base, rng):
    """Return one O(n) memory non‑negative perturbation of ``base``.
    The implementation mirrors the reference “block‑perturbation” strategy.
    """
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the selected block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # optional symmetrisation – keeps the sequence palindromic half of the time
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)
