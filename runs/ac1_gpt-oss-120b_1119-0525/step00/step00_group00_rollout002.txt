"""
Propose a step‑function sequence (non‑negative heights) that minimizes
    score = 2 * n * max(conv(p,p)) / (sum(p)**2)

The optimizer works in three stages:

1.  Initialise from the previous best sequence (if available) or from a
    simple power‑law shape.
2.  Refine the shape with a smooth‑max (log‑sum‑exp) gradient descent on the
    simplex using the Adam optimiser.  The smoothing parameter β is increased
    gradually (continuation) so that the algorithm first sees a smooth
    objective and finally converges towards the true max.
3.  Exploit the remaining time with cheap random‑logit perturbations
    (softmax of Gaussian vectors) to escape possible local minima.
    The best candidate found is finally returned as a list of floats.

All calculations are performed with FFTs (O(n log n)) so a length of
~5000 can be processed thousands of times within the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(m: int) -> int:
    """Smallest power of two >= max(1,m)."""
    n = max(1, int(m))
    return 1 << (n - 1).bit_length()


def stable_softmax(y: np.ndarray) -> np.ndarray:
    """softmax with numerical stability (y is 1‑D)."""
    y = y - np.max(y)            # shift
    e = np.exp(y, dtype=np.float64)
    s = e.sum()
    if s == 0 or not np.isfinite(s):
        # fallback to uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_self(p: np.ndarray) -> np.ndarray:
    """Full linear convolution p * p (real FFT)."""
    n = p.size
    L = next_pow2(2 * n - 1)
    F = np.fft.rfft(p, n=L)
    c = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    return c


def fast_score(p: np.ndarray) -> float:
    """Fast evaluation of the official score assuming p >= 0."""
    # p may not be perfectly normalised – we keep the exact formula.
    n = p.size
    s = float(p.sum())
    if s <= 0.0:
        return float("inf")
    c = conv_self(p)
    return float(2.0 * n * np.max(c) / (s * s))


def correlation(p_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear correlation of p_rev (len=n) with w (len=2n‑1)."""
    n = p_rev.size
    L = next_pow2(3 * n - 2)          # length needed for linear correlation
    Fr = np.fft.rfft(p_rev, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Fr * Fw, n=L)
    # The centred part [n‑1 : n‑1+n] corresponds to lags 0…n‑1
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Adam‑based smooth‑max optimisation on the simplex
# ----------------------------------------------------------------------
def adam_optimize(p0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine p0 using a continuation on the smoothed max.
    Returns (best_p, best_score).
    """
    # hyper‑parameters
    beta_schedule = [5, 10, 20, 40, 80, 160, 320]   # increasing smoothness
    base_lr = 0.25                                   # initial learning rate
    beta1, beta2 = 0.9, 0.999                        # Adam moments
    eps_adam = 1e-8

    # initialise logits y (softmax(y) = p0)
    eps = 1e-30
    y = np.log(np.maximum(p0, eps))
    y = y - np.mean(y)                # zero‑mean – removes a redundant degree

    m = np.zeros_like(y)              # first moment
    v = np.zeros_like(y)              # second moment
    t_iter = 0

    best_p = p0.copy()
    best_score = fast_score(p0)

    n = p0.size
    # pre‑compute sizes for FFTs (they do not change)
    Lc = next_pow2(2 * n - 1)
    Lcorr = next_pow2(3 * n - 2)

    for beta in beta_schedule:
        # learning rate decays gently with beta
        lr = base_lr / math.sqrt(beta)
        # number of inner iterations (adjustable, but kept modest)
        max_inner = 2500
        for _ in range(max_inner):
            if time.time() > deadline:
                return best_p, best_score
            t_iter += 1

            # ---- p from logits -------------------------------------------------
            p = stable_softmax(y)

            # ---- convolution ---------------------------------------------------
            Fp = np.fft.rfft(p, n=Lc)
            c = np.fft.irfft(Fp * Fp, n=Lc)[: 2 * n - 1]   # full convolution

            # ---- smoothed max (log‑sum‑exp) weights ----------------------------
            max_c = np.max(c)
            w_raw = np.exp(beta * (c - max_c))
            w_sum = w_raw.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(w_raw, 1.0 / w_raw.size)
            else:
                w = w_raw / w_sum        # softmax over the convolution entries

            # ---- gradient w.r.t. p (2 * correlation) -------------------------
            grad_p = 2.0 * correlation(p[::-1], w)

            # ---- pull‑back to log‑it space (mirror descent) -------------------
            dot = np.dot(p, grad_p)
            grad_y = p * (grad_p - dot)

            # ---- Adam update --------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- occasional evaluation -----------------------------------------
            if (t_iter & 31) == 0:      # every ~32 iterations
                sc = fast_score(p)
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    print(f"[Adam] iter={t_iter} β={beta}  score={sc:.9f}")

    return best_p, best_score


# ----------------------------------------------------------------------
# Random‑logit exploration (cheap hill climbing)
# ----------------------------------------------------------------------
def random_logit_search(p_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform many cheap random perturbations of the logits.
    Returns (best_p, best_score) found during the allotted time.
    """
    n = p_start.size
    # initialise from the current best (logits)
    y_best = np.log(np.maximum(p_start, 1e-30))
    y_best = y_best - np.mean(y_best)

    best_p = p_start.copy()
    best_score = fast_score(best_p)

    rng = np.random.default_rng(123456)

    # scale of the random logit perturbations – start moderate, shrink over time
    scale = 1.0

    while time.time() < deadline:
        # gradually shrink the perturbation amplitude
        remaining = deadline - time.time()
        # decay factor so that we start with large steps and finish fine‑grained
        scale = max(0.02, scale * (1.0 - 0.0005))

        y = y_best + rng.normal(scale=scale, size=n)
        p = stable_softmax(y)
        sc = fast_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
            y_best = y.copy()        # keep the good direction
            print(f"[Random] new best score={sc:.9f}")

        # safety break if we are extremely close to the deadline
        if remaining < 0.15:
            break

    return best_p, best_score

def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):

    Return a list of non‑negative heights that (hopefully) achieves a score

    # time handling
    # ------------------------------------------------------------------
    global_budget = float(budget_s) if isinstance(budget_s, (int, float)) else 960.0
    if not np.isfinite(global_budget) or global_budget <= 0:
        global_budget = 960.0
    # keep a small safety margin
    deadline = time.time() + max(30.0, min(950.0, global_budget - 5.0))

    # ------------------------------------------------------------------
    # seeding
    # ------------------------------------------------------------------
    # try to fetch the previously known good sequence
    try:
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
        init_seq = np.maximum(0.0, init_seq)
        if init_seq.sum() <= 0:
            raise ValueError
        init_seq = init_seq / init_seq.sum()
    except Exception:
        # fallback: simple decreasing power‑law (α≈0.5) of length ~4800
        n_fallback = 4806
        i = np.arange(n_fallback, dtype=np.float64)
        init_seq = (i + 1.0) ** (-0.5)
        init_seq = init_seq / init_seq.sum()

    # ------------------------------------------------------------------
    # Stage 1 – smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    print(f"[Stage 1] starting Adam optimisation on length {init_seq.size}")
    p_opt, score_opt = adam_optimize(init_seq, deadline)
    print(f"[Stage 1] best score after Adam = {score_opt:.9f}")

    # ------------------------------------------------------------------
    # Stage 2 – cheap random‑logit hill climbing (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Stage 2] starting random‑logit search")
        p_opt, score_opt = random_logit_search(p_opt, deadline)

    seq_list = [float(x) for x in p_opt]

            print(f"[Final] official evaluator score = {official:.9f}")

        pass
