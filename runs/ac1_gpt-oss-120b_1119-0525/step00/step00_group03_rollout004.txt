def compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.size
    # ensure nonnegative and clip for safety
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    return float(2.0 * n * max_b / (sum_a * sum_a))

"""
A fast heuristic for the self‑convolution min‑max problem.

The algorithm builds an arcsine‑distributed step function (the continuous
distribution whose self‑convolution is perfectly flat) and then locally
refines it.  Refinement repeatedly moves a small amount of mass from the
index that contributes most to the current maximal convolution value to an
index that contributes little.  This flattens the convolution further and
drives the evaluation score towards its theoretical lower bound (~1).

The implementation is fully self‑contained, does not rely on any file I/O,
uses only NumPy and the provided `evaluate_sequence` for validation, and
respects the time budget passed via ``budget_s``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def generate_arcsine(n: int) -> np.ndarray:
    """
    Return a non‑negative sequence of length ``n`` approximating the
    arcsine density on (0, 1).  The sequence is normalised to have an
    L1‑norm of 1, which does not affect the final score because the
    evaluation is homogeneous of degree 0 in the coefficients.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                     # mid‑point of each sub‑interval
    # arcsine density 1/√(x(1‑x))
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, 1000.0)            # respect the hard caps of evaluate_sequence
    a /= np.sum(a)                         # normalise to L1 = 1
    return a


def compute_score(seq: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(b) / (sum(seq)²) where
    b = seq *conv* seq (full discrete convolution).  This mirrors
    ``evaluate_sequence`` but avoids the Python‑level list conversion.
    """
    # Make sure the sequence obeys the same hard bounds as evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = np.sum(seq)
    if sum_a < 0.01:               # same safeguard as evaluate_sequence
        return float('inf')
    n = seq.size
    b = np.convolve(seq, seq)      # full convolution, length 2n‑1
    max_b = np.max(b)
    return float(2.0 * n * max_b / (sum_a * sum_a))


def refine_sequence(seq: np.ndarray, rng: np.random.Generator,
                    deadline: float) -> np.ndarray:
    """
    Local improvement loop that repeatedly
    * finds the index ``k`` where the convolution attains its maximum,
    * identifies the pair of positions that contributes most to this peak,
    * moves a tiny amount of mass from that source to a low‑contributing
      target, and
    * accepts the move if the score strictly improves.

    The loop runs until ``deadline`` (a UNIX timestamp) is reached.
    """
    best_seq = seq.copy()
    best_score = compute_score(best_seq)
    n = best_seq.size
    iter_cnt = 0

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1. locate the current maximal convolution entry and its contributors
        # --------------------------------------------------------------
        b = np.convolve(best_seq, best_seq)
        k_max = int(np.argmax(b))

        # contributions[i] = best_seq[i] * best_seq[k_max - i] when index valid
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros_like(best_seq)
        contrib[mask] = best_seq[i_idx[mask]] * best_seq[j_idx[mask]]

        # --------------------------------------------------------------
        # 2. pick a source index (largest contribution) and a target index
        #    (among the weakest contributors)
        # --------------------------------------------------------------
        src = int(np.argmax(contrib))
        low_thresh = np.percentile(contrib, 10)   # 10‑percentile as “low”
        low_cands = np.where(contrib <= low_thresh)[0]
        if low_cands.size == 0:
            target = int(np.argmin(best_seq))
        else:
            target = int(rng.choice(low_cands))

        if src == target:
            # fall back to a random different index
            target = int(rng.integers(0, n))
            if target == src:
                continue

        # --------------------------------------------------------------
        # 3. propose a tiny transfer of mass from src to target
        # --------------------------------------------------------------
        delta = best_seq[src] * rng.random() * 0.05   # ≤ 5 % of the source weight
        if delta <= 0.0:
            continue
        cand = best_seq.copy()
        cand[src] -= delta
        cand[target] += delta
        cand = np.clip(cand, 0.0, 1000.0)

        # --------------------------------------------------------------
        # 4. accept if the score is strictly better
        # --------------------------------------------------------------
        cand_score = compute_score(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[refine] iter {iter_cnt:6d} → score {best_score:.6f}")

        # occasional random “shake” to avoid stagnation
        if rng.random() < 0.01:
            factor = np.exp(rng.normal(0.0, 0.05))
            shaken = np.clip(best_seq * factor, 0.0, 1000.0)
            shaken_score = compute_score(shaken)
            if shaken_score < best_score:
                best_score = shaken_score
                best_seq = shaken
                print(f"[shake]  iter {iter_cnt:6d} → score {best_score:.6f}")

        iter_cnt += 1

    return best_seq

def propose_candidate(seed: int = 42, budget_s: float = 1600, **kwargs) -> list[float]:

    """

    start = time.monotonic()
