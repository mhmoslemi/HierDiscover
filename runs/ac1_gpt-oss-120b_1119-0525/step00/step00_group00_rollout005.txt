def proj_simplex(v):
    # From Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero maybe
        return np.maximum(v, 0)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent_optimize(p0, max_iter, beta_start=1.0, beta_factor=1.02, lr=0.5, lr_decay=0.99):
    p = p0.copy()
    beta = beta_start

        c = convolve(p, p)   # full
        # compute smooth max weights

def sqrt_series(n):
    n = max(1, int(n))
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        b[k] = b[k-1] * (2.0*k - 1.0) / (2.0*k)
    return b / np.sum(b)

"""
Search for a low‑value non‑negative step sequence.

The algorithm builds on the previous “balance_search” approach but adds a
new fast gradient‑descent refinement that directly minimises a smoothed
max‑convolution.  The method:

1.  Initialise a global best using the known truncated‑sqrt construction,
    a power‑law and an exponential seed (also a Dirichlet random seed).
2.  For each seed run a projected (or multiplicative) gradient descent on
    the smoothed max of the self‑convolution.  The gradient is obtained
    analytically (∇ max ≈ 2 p * w where w are soft‑max weights over the
    convolution coefficients).  After each step the vector is projected
    onto the probability simplex to keep it non‑negative and normalised.
3.  The best candidate is repeatedly polished with longer runs of the
    same gradient optimiser and a final call to the existing
    ``balance_search`` routine for a few extra iterations.
4.  All work respects the global deadline (≈1000 s) and prints progress.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Global timing / bookkeeping
# ------------------------------------------------------------
_DEADLINE = 0.0               # will be set in propose_candidate
_BEST_P = None                # best probability vector (sum = 1)
_BEST_SEQ = None              # best list of floats for the official evaluator
_BEST_SCORE = float("inf")    # best score seen so far
_RNG = np.random.default_rng(12345)

def next_pow2(m):
    """Return the smallest power‑of‑two ≥ m (as int)."""
    n = max(1, int(m))
    return 1 << (n - 1).bit_length()


def proj_simplex(v):
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def finite_positive_array(x):
    """Validate and normalise a vector to a probability distribution."""
    try:
        a = np.asarray(x, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(0.0, a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return None
    a = a / s
    a = np.minimum(1000.0, a)
    s2 = float(np.sum(a))
    if s2 <= 0:
        return None
    return a / s2


def convolve_full(p, q):
    """Full convolution of two 1‑D real arrays (uses FFT for long inputs)."""
    n = p.size + q.size - 1
    if n <= 800:
        return np.convolve(p, q, mode='full')
    L = next_pow2(n)
    Fp = np.fft.rfft(p, n=L)
    Fq = np.fft.rfft(q, n=L)
    c = np.fft.irfft(Fp * Fq, n=L)
    return c[:n]


def fast_score_array(p, lft=0):
    """Fast version of evaluate_sequence: 2*n*max(conv(p,p))."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    if n == 0:
        return float("inf")
    p = np.maximum(0.0, p)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return float("inf")
    # normalise to sum = 1 (the score is scale‑invariant)
    p = p / s
    if n <= 800:
        c = np.convolve(p, p, mode='full')
        mc = float(np.max(c))
    else:
        if lft == 0:
            lft = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, n=lft)
        c = np.fft.irfft(F * F, n=lft)
        c = c[:2 * n - 1]
        mc = float(np.max(c))
    return float(2.0 * n * mc)

def make_power(n, alpha=0.5, delta=1.0):
    """Power‑law seed a_i ∝ (i+δ)^(-α), then normalised."""
    n = max(1, int(n))
    i = np.arange(n, dtype=np.float64)
    p = (i + float(delta)) ** (-float(alpha))
    p = np.maximum(1e-300, p)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return p / s


def make_exponential(n, alpha=1.0):
    """Exponential decay a_i ∝ exp(-α·i/(n‑1))."""
    n = max(1, int(n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    i = np.arange(n, dtype=np.float64) / float(n - 1)
    p = np.exp(-float(alpha) * i)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return p / s


def sqrt_series(n):
    """
    First n coefficients of the series (1‑x)^(-1/2):
        b_0 = 1,   b_{k+1}=b_k·(2k+1)/(2k+2)
    Normalised to sum = 1.
    """
    n = max(1, int(n))
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        b[k] = b[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    s = float(np.sum(b))
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return b / s

def smoothed_max_and_grad(p, beta):
    """
    Return (smooth_max, gradient) of f(p)=max_i (p∗p)_i approximated with
    log‑sum‑exp smoothing parameter beta.
    """
    c = convolve_full(p, p)                     # full convolution, length 2n‑1
    maxc = float(np.max(c))
    # soft‑max weights (stable)
    exp_term = np.exp(beta * (c - maxc))
    w = exp_term / np.sum(exp_term)             # size 2n‑1, sums to 1
    # gradient = 2·(w∗p) truncated to first n entries
    grad_full = convolve_full(w, p)             # length (2n‑1)+n‑1 = 3n‑2
    grad = 2.0 * grad_full[:p.size]             # keep only indices 0..n‑1
    return maxc, grad


def projected_gradient_descent(p0, time_limit,
                               beta_start=5.0, beta_factor=1.02,
                               lr_start=0.5, lr_decay=0.99):
    """
    Simple projected (or multiplicative) gradient descent on the
    smoothed‑max of the self‑convolution.
    """
    p = finite_positive_array(p0)
    if p is None:
        return None, float("inf")
    n = p.size
    end = time.monotonic() + time_limit
    beta = beta_start
    lr = lr_start
    best_local = fast_score_array(p)
    best_vec = p.copy()

    it = 0
    while time.monotonic() < end and time.monotonic() < _DEADLINE:
        it += 1
        _, grad = smoothed_max_and_grad(p, beta)

        # ---- additive update + projection ----
        p_add = proj_simplex(p - lr * grad)
        sc_add = fast_score_array(p_add)

        # ---- multiplicative (exponentiated) update ----
        # protect against overflow
        exp_factor = np.exp(-lr * grad)
        p_mul = p * exp_factor
        if np.sum(p_mul) <= 0:
            p_mul = p.copy()   # fallback
        else:
            p_mul = p_mul / np.sum(p_mul)
        sc_mul = fast_score_array(p_mul)

        # choose the better move
        if sc_mul < sc_add:
            p_new = p_mul
            sc_new = sc_mul
        else:
            p_new = p_add
            sc_new = sc_add

        # keep globally best
        if sc_new < _BEST_SCORE:
            update_best_from_p(p_new)

        if sc_new < best_local:
            best_local = sc_new
            best_vec = p_new.copy()

        p = p_new

        # schedule updates
        beta = min(beta * beta_factor, 200.0)
        lr *= lr_decay
        if lr < 1e-6:
            break

    return best_vec, best_local

        _BEST_SEQ = [float(x) for x in _BEST_P]
    else:
        _BEST_P = finite_positive_array(_BEST_P)
        _BEST_SEQ = [float(x) for x in _BEST_P]

    # one final official evaluation (non‑blocking)
    try:
        official = float(evaluate_sequence(_BEST_SEQ))
        if np.isfinite(official) and official < _BEST_SCORE:
            _BEST_SCORE = official
    except Exception:
        pass

    print(f"final returned score={_BEST_SCORE:.9f}, n={len(_BEST_SEQ)}")
    return _BEST_SEQ
