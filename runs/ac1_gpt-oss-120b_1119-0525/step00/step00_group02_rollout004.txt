    # Determine sequence length

    if s == 0:

        a = a / s

"""
Propose a step‑function sequence that tries to minimise

    2 * n * max(conv(a,a)) / (sum(a)**2)

The objective is equivalent to minimising the peak of the
self‑convolution while keeping the sum of the coefficients fixed.
Uniform coefficients give a value of 2, but the true lower bound is
≈ 2 n/(2 n‑1) → 1 for large n.  The algorithm starts from an
“arcsine” shaped distribution (large near the ends, small in the
middle), then performs a simple projected gradient descent on the
position of the current maximal convolution coefficient.  After each
step the sequence is symmetrised, clipped, and renormalised to keep
the sum constant (scaling does not affect the score).  Occasionally
a random block perturbation is added to escape local minima.
Fast FFT based convolution is used so that many iterations fit into the
time budget.

The function respects the required signature and has no nested
closures or lambdas.
"""

import time
import numpy as np

def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Return full convolution a*a using a power‑of‑two FFT (real input)."""
    n = a.size
    # length of linear convolution = 2*n-1, pick next power of two for FFT speed
    size = 1 << ((2 * n - 1).bit_length())
    # rfft works on real data and is a bit faster than full FFT
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the first 2*n-1 entries (the linear convolution)
    return conv[: 2 * n - 1]


def propose_candidate(seed: int = 42, budget_s: float = 1600, length: int = None, **kwargs):
    """
    Search for a low‑value step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑time budget in seconds (the function stops a few
        seconds before this limit to be safe).
    length : int, optional
        Desired length of the sequence.  If omitted the length of the
        globally supplied ``height_sequence_1`` is used, falling back to
        6000 when unavailable.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum≥0.01).
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base_len = base.size if base.ndim == 1 and base.size > 0 else 0
    except Exception:          # ``height_sequence_1`` may be undefined
        base_len = 0

    if length is None:
        n = max(4_000, base_len)          # at least a few thousand entries
    else:
        n = int(max(2, length))

    # ------------------------------------------------------------------
    # Initial distribution – arcsine shape (large near the edges)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = 0.5 * (a + a[::-1])                # enforce exact symmetry
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / a_sum                       # normalise to sum == 1

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the simple projected gradient descent
    # ------------------------------------------------------------------
    lr = 0.05                               # initial learning rate
    min_lr = 1e-4
    it = 0
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # Full self‑convolution (FFT, O(n log n))
        # ------------------------------------------------------------------
        conv = _fft_convolve_self(best)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]            # not used directly, just for debugging

        # ------------------------------------------------------------------
        # Gradient of the currently maximal convolution coefficient
        #   ∂b_max / ∂a_i = 2 * a_{max_idx - i}   if that index is in range
        # ------------------------------------------------------------------
        grad = np.zeros_like(best)
        lo = max(0, max_idx - n + 1)
        hi = min(max_idx, n - 1)
        if lo <= hi:                       # there is a non‑empty overlapping region
            i_range = np.arange(lo, hi + 1)
            j_range = max_idx - i_range
            grad[i_range] = 2.0 * best[j_range]

        # ------------------------------------------------------------------
        # Gradient step, projection onto the feasible set (non‑negative,
        #   sum fixed, values clipped to [0,1000])
        # ------------------------------------------------------------------
        cand = best - lr * grad
        cand = np.maximum(cand, 0.0)        # non‑negative
        # optional random symmetrisation (helps keep the shape symmetric)
        if rng.random() < 0.5:
            cand = 0.5 * (cand + cand[::-1])

        s = cand.sum()
        if s < 0.01:
            # restart from a uniform distribution if we collapsed to zero
            cand = np.ones_like(cand) / n
            s = cand.sum()

        cand = cand / s                      # renormalise (scale does not matter)
        cand = np.clip(cand, 0.0, 1000.0)    # respect the hard limits of evaluate_sequence

        # ------------------------------------------------------------------
        # Evaluate candidate
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            lr = min(lr * 1.02, 1.0)        # slightly increase step on success
            print(f"[{it:05d}] improvement → {best_score:.6f}")
        else:
            # no improvement – slowly decay the learning rate
            lr = max(lr * 0.99, min_lr)

        # ------------------------------------------------------------------
        # Occasional random block perturbation (helps escape plateaus)
        # ------------------------------------------------------------------
        if rng.random() < 0.02:
            block_w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            block_start = int(rng.integers(0, max(1, n - block_w + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.30)))
            tmp = best.copy()
            tmp[block_start : block_start + block_w] *= factor
            tmp = np.clip(tmp, 0.0, 1000.0)
            tmp_sum = tmp.sum()
            if tmp_sum >= 0.01:
                tmp = tmp / tmp_sum
            tmp = 0.5 * (tmp + tmp[::-1])
            sc = float(evaluate_sequence(tmp.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = tmp
                best_score = sc
                print(f"[{it:05d}] random block improvement → {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final scaling (any positive factor leaves the score unchanged).
    # We simply leave the sum at 1.0 which is safely above the 0.01
    # threshold required by ``evaluate_sequence``.
    # ------------------------------------------------------------------
    result = [float(x) for x in best]
    return result
