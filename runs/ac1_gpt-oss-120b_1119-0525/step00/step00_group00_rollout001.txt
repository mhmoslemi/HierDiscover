def interpolate_shape(p, new_n):
    old_n = p.size
    if old_n == new_n:
        return p.copy()
    xs_old = np.linspace(0, 1, old_n)
    xs_new = np.linspace(0, 1, new_n)
    p_new = np.interp(xs_new, xs_old, p)
    # Ensure non-negative and renormalize
    p_new = np.maximum(0, p_new)
    s = np.sum(p_new)
    if s <= 0:
        p_new = np.ones(new_n) / new_n
    else:
        p_new = p_new / s
    return p_new

def adam_optimize(p0, time_limit):
    start = time.monotonic()
    deadline_local = start + time_limit
    y = np.log(np.maximum(1e-300, p0))
    # optionally scale y to zero-mean to avoid large offsets.
    y = y - np.mean(y)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr0=0.2
    step=0
    best_local_score = fast_score_array(p0)
    best_local_p = p0.copy()
    # For tracking improvement
    last_improve_step = 0
    while time.monotonic() < deadline_local and time.monotonic() < _DEADLINE:
        step+=1
        p = softmax_array(y)
        if p is None:
            break
        # compute convolution
        n = p.shape[0]
        if n <= 800:
            c = np.convolve(p, p)
        else:
            Lc = next_pow2(2*n - 1)
            F = np.fft.rfft(p, n=Lc)
            c = np.fft.irfft(F * F, n=Lc)[:2*n-1]
        max_c = np.max(c)
        # beta schedule
        beta = min(200.0, 0.5 + 0.2 * step) # increase gradually
        # compute weights
        # subtract max for numerical stability
        w_unnorm = np.exp(beta * (c - max_c))
        sum_w = np.sum(w_unnorm)
        if sum_w <= 0 or not np.isfinite(sum_w):
            w = np.ones_like(c) / c.size
        else:
            w = w_unnorm / sum_w
        # gradient wrt p
        if n <= 800:
            corr = np.correlate(p, w, mode='full')
            grad_p = 2.0 * corr[n-1 : n-1+n]
        else:
            # correlation via FFT: conv(p_rev, w)
            Lc2 = next_pow2(3*n - 2)
            F_rev = np.fft.rfft(p[::-1], n=Lc2)
            W = np.fft.rfft(w, n=Lc2)
            conv_rev_w = np.fft.irfft(F_rev * W, n=Lc2)
            # slice
            start_idx = n - 1
            grad_p = 2.0 * conv_rev_w[start_idx : start_idx + n]
        # gradient wrt logits y
        mu = np.dot(p, grad_p)
        grad_y = p * (grad_p - mu)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        lr = lr0 / (np.sqrt(step) + 1)  # decaying LR
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Evaluate new p after update (maybe compute again)

        if p_new is None:

            best_local_p = p_new.copy()
            last_improve_step = step
            update_best_from_p(p_new)
        # Random perturb if stagnation > threshold
        if step - last_improve_step > 200:
            # add small noise

            y = y + perturb

"""
Propose a step‐function sequence with a low evaluation score.

Algorithm
---------
1. Initialise a global best candidate from the previously supplied
   `height_sequence_1` (if any) or from a simple power‑law shape.
2. For several problem sizes (derived from the current best length) create
   an interpolated version of the best shape and improve it with a
   gradient‑based optimiser:
      • Optimisation is performed on the log‑softmax parameters `y`
        (so the probabilities stay on the simplex automatically).
      • The loss is a smooth approximation of `max(conv(p,p))`:
        `L_β(p) = (1/β) log Σ exp(β·c_k)` where `c = p∗p`.
      • Gradients are obtained analytically (`∇_p L = 2·correlation(p,w)`,
        with `w` the softmax over the convolution entries).
      • Adam updates are applied to `y`.  A small random kick is added
        when progress stalls.
   The optimiser runs as long as time permits (up to a few hundred
   iterations per length) and updates the global best solution after
   each improvement.
3. Each refined shape is further polished by the original “balance”
   local search (exponentiated‑gradient iterations) for a short budget.
4. After all lengths are processed a final polishing step is performed
   on the current global best.
5. The best sequence found (as a plain list of floats) is returned.
"""

def next_pow2(m):
    """Smallest power of two >= m (m must be positive integer)."""
    m = int(max(1, m))
    return 1 << ((m - 1).bit_length())


def finite_positive_array(x):
    """Validate, clip and renormalise a raw numeric array to a probability vector."""
    try:
        a = np.asarray(x, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(0.0, a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return None
    a = a / s
    a = np.minimum(1000.0, a)
    s2 = float(np.sum(a))
    if s2 <= 0:
        return None
    return a / s2


def softmax_array(y):
    """Stable softmax returning a probability vector (sum=1)."""
    try:
        y = np.asarray(y, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if y.size == 0 or not np.all(np.isfinite(y)):
        return None
    y = y - float(np.max(y))
    e = np.exp(y - np.max(y))
    s = float(np.sum(e))
    if s == 0 or not np.isfinite(s):
        return None
    p = e / s
    p = np.maximum(0.0, p)
    return p / float(np.sum(p))


def fast_score_array(p, lft=0):
    """Fast evaluation of 2*n*max(conv(p,p)) (sum of p assumed 1)."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    if n == 0:
        return float("inf")
    p = np.maximum(0.0, p)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return float("inf")
    # sum must be 1 for the metric, but we keep the normalisation
    # invariance; the scorer itself does not re‑normalise.
    if n <= 800:
        c = np.convolve(p, p)
        mc = float(np.max(c))
    else:
        if lft == 0:
            lft = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, n=lft)
        c = np.fft.irfft(F * F, n=lft)
        mc = float(np.max(c[:2 * n - 1]))
    return float(2.0 * n * mc / (s * s))


def conv_score_array(p, lft=0):
    """Return full convolution `c` and its maximum for a probability vector."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    p = np.maximum(0.0, p)
    s = float(np.sum(p))
    if s <= 0:
        p = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        p = p / s
    if n <= 800:
        c = np.convolve(p, p)
    else:
        if lft == 0:
            lft = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, n=lft)
        c = np.fft.irfft(F * F, n=lft)
        c = c[:2 * n - 1]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    mc = float(np.max(c)) if c.size else 0.0
    return c, mc

def make_power(n, alpha=0.5, delta=1.0):
    n = max(1, int(n))
    i = np.arange(n, dtype=np.float64)
    p = (i + float(delta)) ** (-float(alpha))
    p = np.maximum(1e-300, p)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return p / s


def make_exponential(n, alpha=1.0):
    n = max(1, int(n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    i = np.arange(n, dtype=np.float64) / float(n - 1)
    p = np.exp(-float(alpha) * i)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return p / s


def truncated_sqrt_sequence(n):
    """First n coefficients of (1-x)^(-1/2), renormalised."""
    n = max(1, int(n))
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            j = k - i
            s += b[i] * b[j]
        bk = (1.0 - s) / 2.0
        if bk < 0.0:
            bk = 0.0
        b[k] = bk
    s = float(np.sum(b))
    if s <= 0 or not np.isfinite(s):
        return None
    return np.maximum(0.0, b) / s


def interpolate_shape(p, new_n):
    """Linear‑interpolate a probability vector `p` to length `new_n`."""
    old_n = p.size
    if old_n == new_n:
        return p.copy()
    xs_old = np.linspace(0.0, 1.0, old_n)
    xs_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(xs_new, xs_old, p)
    new = np.maximum(0.0, new)
    s = np.sum(new)
    if s <= 0 or not np.isfinite(s):
        new = np.ones(new_n, dtype=np.float64) / new_n
    else:
        new = new / s
    return new

def adam_optimize(p0, time_limit=60.0):
    """
    Gradient descent on a smooth approximation of max(conv(p,p)).
    Operates in the log‑softmax space and updates the global best.
    """
    start = time.monotonic()
    deadline = start + max(1.0, float(time_limit))
    y = np.log(np.maximum(1e-300, p0))
    y -= np.mean(y)                       # centre to avoid numerical drift
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.2
    step = 0
    best_local_score = fast_score_array(p0)
    best_local_p = p0.copy()
    last_improve = 0

    while time.monotonic() < deadline and time.monotonic() < _DEADLINE:
        step += 1
        p = softmax_array(y)
        if p is None:
            break
        n = p.size

        # --- convolution (self‑product) ---------------------------------
        if n <= 800:
            c = np.convolve(p, p)
        else:
            Lc = next_pow2(2 * n - 1)
            F = np.fft.rfft(p, n=Lc)
            c = np.fft.irfft(F * F, n=Lc)[:2 * n - 1]

        max_c = float(np.max(c))

        # --- smooth‑max weighting ---------------------------------------
        beta = min(200.0, 0.5 + 0.2 * step)          # slowly increase sharpness
        w_unnorm = np.exp(beta * (c - max_c))
        sw = float(np.sum(w_unnorm))
        if sw <= 0 or not np.isfinite(sw):
            w = np.ones_like(c) / c.size
        else:
            w = w_unnorm / sw

        # --- gradient wrt p (correlation) -------------------------------
        if n <= 800:
            corr = np.correlate(p, w, mode="full")
            grad_p = 2.0 * corr[n - 1 : n - 1 + n]
        else:
            # correlation via FFT: corr(p,w) == conv(p_rev, w)[n-1: n-1+n]
            Lc2 = next_pow2(3 * n - 2)
            F_rev = np.fft.rfft(p[::-1], n=Lc2)
            W = np.fft.rfft(w, n=Lc2)
            conv_rev_w = np.fft.irfft(F_rev * W, n=Lc2)
            start_idx = n - 1
            grad_p = 2.0 * conv_rev_w[start_idx : start_idx + n]

        # --- gradient wrt logits y (Adam works on y) --------------------
        mu = float(np.dot(p, grad_p))
        grad_y = p * (grad_p - mu)

        # --- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        lr = lr0 / (np.sqrt(step) + 1.0)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- evaluate new point -----------------------------------------
        p_new = softmax_array(y)
        if p_new is None:
            continue
        sc = fast_score_array(p_new)
        if np.isfinite(sc) and sc < best_local_score - 1e-12:
            best_local_score = sc
            best_local_p = p_new.copy()
            last_improve = step
            update_best_from_p(p_new)

        # --- occasional random kick if stagnating ----------------------
        if step - last_improve > 200:
            y += 0.01 * _RNG.standard_normal(y.shape)
            last_improve = step  # reset stagnation counter

    return best_local_p, best_local_score

def balance_search(p0, time_limit=60.0):
    """
    Refine a probability vector by minimising max(self‑convolution)
    using the active‑set weighting scheme from the reference implementation.
    """
    p = finite_positive_array(p0)
    if p is None:
        return
    n = p.size
    local_deadline = time.monotonic() + max(1.0, float(time_limit))

    y = np.log(np.maximum(1e-300, p))
    y -= np.mean(y)
    best_p = p.copy()
    best_score = fast_score_array(best_p, 0)
    best_y = y.copy()

    lscore = 0 if n <= 800 else next_pow2(2 * n - 1)
    lv = 0 if n <= 800 else next_pow2(3 * n)

    gamma = 50.0
    stagnation = 0
    it = 0
    grad_steps = (0.01, 0.03, 0.1, 0.3, 1.0, 3.0)
    mult_steps = (0.05, 0.2, 0.7, 2.0)

    while time.monotonic() < local_deadline and time.monotonic() < _DEADLINE:
        it += 1
        c, C = conv_score_array(best_p, lscore)
        if C <= 1e-300 or not np.isfinite(C):
            break

        # --- active set weighting ------------------------------------
        z = c / (C + 1e-300)
        zmax = float(np.max(z)) if z.size else 1.0
        w = np.exp(gamma * (z - zmax))
        sw = float(np.sum(w))
        lam = w / sw if sw > 0 else np.ones_like(c) / c.size

        # --- compute v = Σ_k lam_k p_{k-i} ----------------------------
        if n <= 800:
            v = np.convolve(best_p[::-1], lam, mode="full")[n - 1 : 2 * n - 1]
        else:
            Fp = np.fft.rfft(best_p[::-1], n=lv)
            Fl = np.fft.rfft(lam, n=lv)
            v = np.fft.irfft(Fp * Fl, n=lv)[n - 1 : 2 * n - 1]

        v = np.maximum(1e-300, np.nan_to_num(v))
        vnorm = v / (C + 1e-300)
        mean_v = float(np.dot(best_p, vnorm))
        grad_log = best_p * (vnorm - mean_v)

        trial_score = best_score
        trial_y = None
        trial_p = None

        # --- log‑gradient steps ---------------------------------------
        for step in grad_steps:
            ytry = y - float(step) * grad_log
            ptry = softmax_array(ytry)
            if ptry is None:
                continue
            sc = fast_score_array(ptry, lscore)
            if np.isfinite(sc) and sc < trial_score - 1e-12:
                trial_score = sc
                trial_y = ytry.copy()
                trial_p = ptry

        # --- multiplicative (log‑v) steps -----------------------------
        logv = np.log(vnorm)
        logv -= np.mean(logv)
        for step in mult_steps:
            ytry = y - float(step) * logv
            ptry = softmax_array(ytry)
            if ptry is None:
                continue
            sc = fast_score_array(ptry, lscore)
            if np.isfinite(sc) and sc < trial_score - 1e-12:
                trial_score = sc
                trial_y = ytry.copy()
                trial_p = ptry

        if trial_y is not None:
            y = trial_y
            best_p = trial_p.copy()
            best_y = y.copy()
            best_score = float(trial_score)
            stagnation = 0
            if it % 25 == 0:
                print(f"[balance] n={n} it={it} score={best_score:.9f}")
        else:
            stagnation += 1
            if stagnation % 20 == 0:
                gamma = min(500.0, gamma * 2.0) if gamma < 200.0 else max(5.0, gamma * 0.5)

            if stagnation % 30 == 0:
                xs = np.linspace(0.0, 1.0, n, dtype=np.float64)
                best_kick_y = best_y.copy()
                best_kick_p = best_p.copy()
                best_kick_sc = best_score
                for _ in range(5):
                    amp = 0.04 + 0.08 * (stagnation / 200.0)
                    f1 = 0.5 + 3.0 * _RNG.random()
                    f2 = 1.0 + 5.0 * _RNG.random()
                    ph1 = _RNG.uniform(0.0, 2.0 * math.pi)
                    ph2 = _RNG.uniform(0.0, 2.0 * math.pi)
                    kick = amp * (
                        (0.3 + 0.7 * _RNG.random()) * np.sin(2.0 * math.pi * f1 * xs + ph1)
                        + (0.2 + 0.5 * _RNG.random()) * np.cos(2.0 * math.pi * f2 * xs + ph2)
                        + 0.25 * (xs - 0.5)
                    )
                    ytry = best_y + kick
                    ptry = softmax_array(ytry)
                    if ptry is None:
                        continue
                    sc = fast_score_array(ptry, lscore)
                    if np.isfinite(sc) and sc < best_kick_sc:
                        best_kick_sc = sc
                        best_kick_y = ytry.copy()
                        best_kick_p = ptry
                y = best_kick_y
                best_p = best_kick_p.copy()
                best_y = y.copy()
                best_score = float(best_kick_sc)

    update_best_from_p(best_p)
    return best_p, best_score

def propose_candidate(seed=42, budget_s=1600, **kwargs):
    """
    Search for the step sequence with the smallest evaluation score.
    The function respects a time budget (default ~1000 s) and returns
    a plain list of non‑negative floats.
    """
    global _DEADLINE, _BEST_P, _BEST_SEQ, _BEST_SCORE, _RNG

    # ------------------------------------------------------------------
    # Time handling
    # ------------------------------------------------------------------
    budget_s = float(kwargs.get("budget_s", budget_s))
    if not np.isfinite(budget_s) or budget_s <= 0:
        budget_s = 1000.0
    # leave a safety margin (≈60 s) for final checks
    _DEADLINE = time.monotonic() + max(60.0, min(950.0, budget_s - 60.0))
    _RNG = np.random.default_rng(int(seed) % (2 ** 32))

    # ------------------------------------------------------------------
    # Global best initialisation
    # ------------------------------------------------------------------
    _BEST_P = None
    _BEST_SEQ = None
    _BEST_SCORE = float("inf")
    try:
        prev = finite_positive_array(height_sequence_1)  # may be undefined
        if prev is not None:
            update_best_from_p(prev)
    except Exception:
        pass

    if _BEST_P is None:
        fallback = make_power(1024, 0.5, 1.0)
        update_best_from_p(fallback)

    print(f"[init] best score={_BEST_SCORE:.9f}, n={_BEST_P.size if _BEST_P is not None else 0}")

    # ------------------------------------------------------------------
    # Determine a set of problem sizes to explore
    # ------------------------------------------------------------------
    base_n = _BEST_P.size if _BEST_P is not None else 1024
    factors = [0.7, 0.85, 1.0, 1.15, 1.3]
    lengths = sorted({int(max(256, min(8000, base_n * f))) for f in factors})

    # ------------------------------------------------------------------
    # Main optimisation loop over the chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= _DEADLINE - 30.0:
            break

        # seed from the current global best (interpolated) or pure power law
        if _BEST_P is not None:
            p_start = interpolate_shape(_BEST_P, n)
        else:
            p_start = make_power(n, 0.5, 1.0)

        update_best_from_p(p_start)

        # time budget for this length (roughly split the remaining budget)
        remaining = _DEADLINE - time.monotonic()
        time_alloc = min(150.0, max(20.0, remaining - 40.0))
        if time_alloc < 10.0:
            break

        print(f"[optim] length={n}, time budget={time_alloc:.1f}s")
        # Adam optimiser (smooth‑max)
        p_opt, _ = adam_optimize(p_start, time_limit=time_alloc)

        # short polishing with the balance search
        remaining2 = _DEADLINE - time.monotonic()
        if remaining2 > 20.0:
            balance_search(p_opt, time_limit=min(30.0, remaining2 - 5.0))

    # ------------------------------------------------------------------
    # Final polishing phase (if any time left)
    # ------------------------------------------------------------------
    remaining = _DEADLINE - time.monotonic()
    if remaining > 30.0 and _BEST_P is not None:
        print("[final] polishing the global best")
        balance_search(_BEST_P, time_limit=remaining - 5.0)

    # ------------------------------------------------------------------
    # Ensure a valid return value
    # ------------------------------------------------------------------
    if _BEST_P is None:
        _BEST_P = make_power(1024, 0.5, 1.0)
    else:
        _BEST_P = finite_positive_array(_BEST_P)

    _BEST_SEQ = [float(x) for x in _BEST_P]

    # final official evaluation (non‑blocking)
    try:
        official = float(evaluate_sequence(_BEST_SEQ))
        if np.isfinite(official):
            _BEST_SCORE = official
    except Exception:
        pass

    print(f"[final] returned score={_BEST_SCORE:.9f}, n={len(_BEST_SEQ)}")
    return _BEST_SEQ
