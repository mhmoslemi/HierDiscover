def gradient_step(a, target_sum, step=1e-3, alpha=10.0):
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(3*n))) # for gradient conv length 3n-2
    # compute conv = a * a (auto)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    # compute softmax vector p
    # compute exp(alpha*conv - max) for stability
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()
    # compute gradient: g = 2 * convolution of p with reversed a
    a_rev = a[::-1]
    fp = np.fft.rfft(p, n=L)
    fa_rev = np.fft.rfft(a_rev, n=L)
    grad_full = np.fft.irfft(fp * fa_rev, n=L)[:3*n-2]
    grad = 2 * grad_full[n-1:n-1+n]  # indices n-1 to 2n-2 inclusive = length n
    # update
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    # renorm
    sum_a = a_new.sum()
    if sum_a > 0:
        a_new *= target_sum / sum_a
    return a_new

"""
Iterative linear‑program + mixing optimisation.

The candidate sequence is initialised from the global ``height_sequence_1``
(if it exists) and normalised so that  Σ a_j = sqrt(2·n).
At each iteration we

1. compute the auto‑convolution f∗f and its current maximum M,
2. solve the linear programme
       maximise   Σ b_j
       subject to b_j ≥ 0  and  f∗b ≤ M   (component‑wise)
   (f is a constant parameter for the LP, the only variable is b),
3. renormalise the LP solution to the same ℓ₁‑norm as f,
4. replace f by a convex combination (1‑t)·f + t·g where g is the
   rescaled LP solution.  The step size t starts relatively large and
   decays as 1/√iteration, guaranteeing that the bound never grows.
5. keep the best sequence seen (lowest value of evaluate_sequence).

If the ``cvxpy`` package is unavailable the routine falls back to a simple
random block perturbation (the method used previously).  The loop runs
until the provided time budget expires and finally returns the best
sequence found.
"""

import time
import math
import numpy as np

# ---- helper for fast auto‑convolution (FFT, O(n log n)) -----------------
def _auto_convolve(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using a power‑of‑two FFT length."""
    n = a.shape[0]
    # length needed for full convolution
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


# ---- simple block perturbation (fallback) ---------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative block‑wise multiplicative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def propose_candidate(seed: int = 42, budget_s: float = 1600, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The algorithm mixes an LP based extreme‑point search
    with a diminishing convex‑combination update.
    """
    rng = np.random.default_rng(seed)

    # ---- initialise -------------------------------------------------------
    # use the globally supplied sequence if it looks valid
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise ValueError
    except Exception:
        init = None

    if init is None:
        # fall back to a uniform start of length 4096 (arbitrary but reasonable)
        n = 4096
        init = np.ones(n, dtype=np.float64)
    else:
        n = init.size

    # enforce the explicit bounds required by the evaluator
    f = np.clip(init, 0.0, 1000.0)

    # normalise the ℓ₁‑norm to sqrt(2·n) (the objective is scale‑invariant)
    target_sum = math.sqrt(2.0 * n)
    s = f.sum()
    if s > 0.0:
        f *= target_sum / s
    else:
        f = np.full(n, target_sum / n)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ---- try to set up the linear programme --------------------------------
    use_lp = False
    try:
        import cvxpy as cp

        # parameters that will change each iteration
        param_f = cp.Parameter(n, nonneg=True, value=f)
        param_M = cp.Parameter(nonneg=True)

        # variable representing the new candidate
        b = cp.Variable(n, nonneg=True)

        # f * b is linear in b because f is a constant parameter
        conv_expr = cp.conv(param_f, b)                     # length 2·n‑1
        lp_constraints = [conv_expr[i] <= param_M for i in range(2 * n - 1)]

        lp_obj = cp.Maximize(cp.sum(b))
        lp_prob = cp.Problem(lp_obj, lp_constraints)

        # quick solve to make sure the solver works
        param_M.value = 1.0
        param_f.value = f
        lp_prob.solve(solver="ECOS", warm_start=True, verbose=False, max_iters=200)
        if lp_prob.status in ("optimal", "optimal_inaccurate"):
            use_lp = True
            print("[lp] Linear programme initialised successfully.")
        else:
            print("[lp] Solver failed on dummy problem, will fall back to random search.")
    except Exception as exc:
        print("[lp] cvxpy not available or failed to set up LP:", exc)

    # ---- main optimisation loop -------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    it = 0
    while time.monotonic() < deadline:
        # compute current auto‑convolution and max value
        conv_f = _auto_convolve(f)
        M = float(conv_f.max())

        # evaluate current sequence
        cur_score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = cur_score
            best = f.copy()
            print(f"[{it:04d}] new best: {best_score:.6f} (max={M:.6f})")

        if use_lp:
            # ----- solve the LP ------------------------------------------------
            param_f.value = f
            param_M.value = M
            try:
                lp_prob.solve(solver="ECOS", warm_start=True, verbose=False, max_iters=500)
            except Exception as exc:
                print(f"[{it}] LP solve error:", exc)
                break

            if lp_prob.status not in ("optimal", "optimal_inaccurate"):
                print(f"[{it}] LP not optimal (status: {lp_prob.status}); aborting LP path.")
                use_lp = False
                continue

            g0 = b.value
            if g0 is None:
                print(f"[{it}] LP returned no solution; aborting LP path.")
                use_lp = False
                continue

            # ---- build the mixing candidate ---------------------------------
            g0 = np.maximum(g0, 0.0)                     # defensive clipping
            sum_g0 = g0.sum()
            if sum_g0 <= 0.0:
                print(f"[{it}] LP produced all‑zero vector; aborting LP path.")
                use_lp = False
                continue

            # renormalise to the fixed ℓ₁‑norm
            g = g0 * (target_sum / sum_g0)

            # diminishing convex‑combination step size
            t = 0.30 / math.sqrt(it + 1.0)   # between ~0.3 and small values
            t = min(t, 0.5)                  # never exceed 0.5

            f = (1.0 - t) * f + t * g

            # re‑apply the hard bounds and re‑normalise (scale invariance)
            f = np.clip(f, 0.0, 1000.0)
            total = f.sum()
            if total > 0.0:
                f *= target_sum / total

        else:
            # ----- random block perturbation (fallback) ------------------------
            cand = _make_block_perturbation(f, rng)
            # normalise
            s = cand.sum()
            if s > 0.0:
                cand *= target_sum / s
            cand_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(cand_score) and cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[{it:04d}] random improved: {best_score:.6f}")

            # accept the perturbation only if it improves the true objective
            if cand_score < cur_score:
                f = cand

        it += 1

    # final report
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"[final] best score={final_score:.6f} after {it} iterations (n={n})")
    return [float(x) for x in best]
