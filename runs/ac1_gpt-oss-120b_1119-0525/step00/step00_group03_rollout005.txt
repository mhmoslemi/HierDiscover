def compute_score(a):
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    n = a.shape[0]
    # compute linear conv via FFT
    N = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=N)
    b = np.fft.irfft(A*A, n=N)[:2*n-1]  # real part
    max_b = np.max(b)
    return 2.0 * n * max_b / (S*S)
```

"""
Hill‑climbing search for a low‑evaluation step‑function.

The algorithm starts from the analytic Catalan (square‑root) sequence
a_i = C(2i,i)/4**i, whose self‑convolution is constant in the
infinite limit (ratio ≈ π/2 ≈ 1.571).  A fast FFT based objective
(2·n·max(conv)/sum²) is used for millions of cheap iterations.
Two kinds of perturbations are tried:

* Random block scaling (similar to the reference code).
* Pairwise “gradient‑guided’’ moves: the current maximal convolution
  index kₘₐₓ is located, a random contributing pair (i, kₘₐₓ−i) is
  shrunk a little, and the removed mass is transferred to a random
  index far from the pair.  This directly reduces the dominant term
  of the convolution while keeping the total sum unchanged.

Occasional Gaussian noise and a small chance of symmetrising the
candidate are added to avoid stagnation.  The best candidate seen
throughout the time budget is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_conv_max(a: np.ndarray) -> float:
    """Return max value of the linear convolution a*a using FFT."""
    n = a.shape[0]
    # length for a linear convolution (next power of two for speed)
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    b = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(np.max(b))


def _score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence (exact for the optimizer)."""
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    n = a.shape[0]
    maxb = _fft_conv_max(a)
    return 2.0 * n * maxb / (s * s)

            # use a cheap FFT to locate the max index
            n_len = cand.shape[0]
            N = 1 << ((2 * n_len - 1).bit_length())
            A_fft = np.fft.rfft(cand, n=N)
            conv = np.fft.irfft(A_fft * A_fft, n=N)[: 2 * n_len - 1]
            kmax = int(np.argmax(conv))

            # pick a random contributor of that peak
            i = int(rng.integers(0, n_len))
            j = kmax - i
            if 0 <= j < n_len and i != j:
                # shrink the pair by a modest amount
                eps = rng.uniform(0.02, 0.15)        # 2‑15 % reduction
                reduction = eps * (cand[i] + cand[j])
                cand[i] *= (1.0 - eps)
                cand[j] *= (1.0 - eps)

                # redistribute the removed mass to a random third index
                k = int(rng.integers(0, n_len - 2))
                if k >= i:
                    k += 1
                if k >= j:
                    k += 1
                cand[k] += reduction

        else:
            # --------------------------------------------------------------
            # 2) small Gaussian noise on the whole vector
            # --------------------------------------------------------------
            noise = rng.normal(0.0, 0.03, size=n)
            cand += noise

        # occasional symmetrisation (helps flatten the convolution)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        # enforce bounds required by evaluate_sequence
        cand = np.clip(cand, 0.0, 1000.0)

        # ------------------------------------------------------------------
        # b) evaluate the cheap objective
        # ------------------------------------------------------------------
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        # ------------------------------------------------------------------
        # c) periodic progress report
        # ------------------------------------------------------------------
        if time.monotonic() >= next_report:
            print(f"[{time.monotonic():.2f}] iter={iteration:,} "
                  f"best_score={best_score:.6f}")
            next_report = time.monotonic() + report_interval

    # ----------------------------------------------------------------------
    # 3) final (exact) evaluation – the harness will call evaluate_sequence
    # ----------------------------------------------------------------------
    # Convert to a plain Python list of floats as required.
    return [float(x) for x in best]
