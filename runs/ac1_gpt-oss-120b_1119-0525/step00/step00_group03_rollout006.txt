"""
propose_candidate
=================

Search for a step‑function (non‑negative height sequence) that minimizes
``evaluate_sequence``.  The algorithm follows the iterative improvement
scheme described in the statement:

1.  Start from a reasonable candidate (the supplied ``height_sequence_1``
    or a short geometric decay sequence) and normalise it so that its L¹
    norm equals 1.  For a normalised sequence the evaluation reduces to
    ``2 * n * max(conv(f,f))`` – only the maximal self‑convolution matters.

2.  Given the current sequence ``f`` and its current maximal convolution
    value ``M = max(conv(f,f))`` we solve the linear program

        maximise    sum(g_i)
        subject to  (f * g)_k ≤ M   for all k
                    g_i ≥ 0

    The convolution ``f * g`` is linear in the unknown vector ``g``,
    therefore the problem is an LP with ``n`` variables and ``2n‑1``
    constraints.  CVXPY (ECOS) is used for the optimisation.

3.  The LP returns a vector ``g`` with a larger L¹ norm than ``f`` while
    respecting the same maximal convolution bound ``M``.  We rescale
    ``g`` to have unit L¹ norm (the same normalisation as ``f``) and
    replace the current iterate by a convex combination

        f_new = (1‑t) * f + t * g_scaled          (t ∈ (0,1])

    This keeps the total mass fixed and, by a standard convexity argument,
    never increases the maximal convolution.  With a moderate step size
    (t = 0.5) the value ``M`` drops monotonically.

4.  Steps 2‑3 are repeated until the time budget expires or no further
    improvement is observed.  The best sequence seen during the run is
    returned.

If CVXPY is unavailable the routine falls back to the simple block‑perturb
local search used in the reference implementation.

The whole procedure respects the 1000‑second budget (the caller supplies
``budget_s``) and returns a list of floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper used by the fallback method (identical to the reference code)
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1600, **kwargs):
    """
    Return a sequence of non‑negative heights that (heuristically) minimizes
    ``evaluate_sequence``.
    """
    # --------------------------------------------------------------
    # 1)  Set up the time budget
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # 2)  Initialise a feasible candidate (normalised to L¹ = 1)
    try:
        # Global variable provided by the judging harness
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:        # pragma: no cover
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    n = base.size
    base = np.clip(base, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Produce a small portfolio of starters and pick the best one.
    starters = []

    # (a)  the supplied sequence
    starters.append(base / max(base.sum(), 1e-12))

    # (b)  a few geometric decays – they are known to lower the max‑conv.
    for alpha in (0.8, 1.0, 1.2, 1.5, 2.0):
        if alpha >= n:            # guard against negative ratio
            continue
        r = 1.0 - alpha / n
        if r <= 0.0:
            continue
        geom = r ** np.arange(n, dtype=np.float64)
        # avoid a completely zero vector if r is too small
        if geom.sum() == 0.0:
            continue
        starters.append(geom / geom.sum())

    # pick the starter with the lowest evaluation score
    best_score = float("inf")
    best_seq = None
    for cand in starters:
        val = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(val) and val < best_score:
            best_score = val
            best_seq = cand.copy()
    f = best_seq.copy()
    # --------------------------------------------------------------
    # Helper to compute the current objective (M = max(conv(f,f)))
    def current_max_conv(x):
        return np.convolve(x, x).max()

    # --------------------------------------------------------------
    # 3)  Try the LP‑based refinement if CVXPY is available
    try:
        import cvxpy as cp

        # Parameters that will be updated each iteration
        f_param = cp.Parameter(n, value=f)
        M_param = cp.Parameter(nonneg=True)

        # Decision variable
        g = cp.Variable(n, nonneg=True)

        # Linear convolution (f * g) – CVXPY turns this into a linear
        # expression because ``f_param`` is a constant Parameter.
        conv_fg = cp.conv(f_param, g)          # length 2n‑1

        # One inequality per entry of the convolution
        constraints = [conv_fg <= M_param]

        # Objective: maximise the L1 norm of g
        problem = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # ----- iterative improvement loop ---------------------------------
        iteration = 0
        t = 0.5                                 # convex‑combination step size
        while time.monotonic() < deadline:
            iteration += 1

            # current bound M
            M = float(current_max_conv(f))
            # update the parameters for the LP
            f_param.value = f
            M_param.value = M

            # solve the LP (warm start speeds up successive solves)
            try:
                problem.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            except Exception as e:              # pragma: no cover
                print(f"[iteration {iteration}] LP solve failed: {e}")
                break

            if problem.status not in ("optimal", "optimal_inaccurate"):
                print(f"[iteration {iteration}] LP status = {problem.status}")
                break

            g_sol = g.value
            if g_sol is None:
                break

            sum_g = float(g_sol.sum())
            sum_f = float(f.sum())   # should be 1.0 (normalised)

            # No increase in L1 – we have reached a fixed point
            if sum_g <= sum_f + 1e-12:
                print(f"[iteration {iteration}] No L1 improvement (sum_g={sum_g:.6f})")
                break

            # Rescale g so that its L1 norm equals that of f (keeps total mass fixed)
            scale = sum_f / sum_g
            g_scaled = g_sol * scale

            # Convex combination
            f_new = (1.0 - t) * f + t * g_scaled
            f_new = np.clip(f_new, 0.0, 1000.0)

            # Evaluate the new candidate
            new_score = float(evaluate_sequence(f_new.tolist()))
            if not np.isfinite(new_score):
                # safety – discard and stop the LP‑phase
                print(f"[iteration {iteration}] Invalid score after update.")
                break

            # Compute the new maximal convolution to verify monotonicity
            M_new = float(current_max_conv(f_new))

            if M_new <= M + 1e-15:          # non‑increase accepted
                f = f_new
                if new_score < best_score:
                    best_score = new_score
                    best_seq = f.copy()
                    print(f"[iteration {iteration}] improved score: {best_score:.6f}")
                # optional: shrink step size as we get close to optimum
                t = min(0.9, t * 1.02)
            else:
                # If the bound increased we try a smaller step size; when it
                # becomes tiny we give up the LP‑phase.
                t *= 0.5
                if t < 1e-3:
                    print(f"[iteration {iteration}] step size too small, terminating LP phase.")
                    break
                # do **not** update f – try the next iteration with the reduced t
                continue

        # --------------------------------------------------------------
        # Return the best sequence found by the LP‑based method
        result = best_seq.tolist()
        return [float(x) for x in result]

    except Exception:  # pragma: no cover
        # -----------------------------------------------------------------
        # 4)  Fallback: simple block‑perturb local search (the reference method)
        print("CVXPY not available or failed – falling back to block perturbation.")
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))
        while time.monotonic() < deadline:
            cand = make_block_perturbation(best, rng)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
        return [float(x) for x in best]
