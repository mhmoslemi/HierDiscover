def propose_candidate(seed=12345, budget_s=1000):
    import time, numpy as np, math, sys
    # Set deadline
    deadline = time.time() + max(60.0, min(950.0, budget_s - 60.0))
    rng = np.random.default_rng(seed % (2**32))
    best_seq = None
    best_score = float('inf')
    
    def update(seq):
        nonlocal best_seq, best_score
        # compute fast score and optionally official
        sc = fast_score_array(seq)
        if not np.isfinite(sc):
            return
        if sc < best_score:
            best_score = sc
            best_seq = np.copy(seq)
            print(f"new best: score={sc:.8f}, n={len(seq)}")
    
    # baseline: truncated sqrt at large n
    N = 4806  # use previous best length
    seq = truncated_sqrt_sequence(N)
    if seq is not None:
        update(seq)
    
    # also use power series
    seq = make_power(N, alpha=0.5, delta=1.0)
    update(seq)
    
    # random family search
    eval_budget = 2000
    start_time = time.time()
    for i in range(eval_budget):
        if time.time() >= deadline - 30:
            break
        
        # random choice of family
        choice = rng.integers(0, 5)
        if choice == 0:  # power law
            alpha = rng.uniform(0.2, 1.0)
            delta = rng.uniform(0.5, 5.0)
            p = make_power(N, alpha, delta)
        elif choice == 1:  # exponential
            a = rng.uniform(0.5, 3.0)
            p = make_exponential(N, a)
        elif choice == 2:  # beta
            a = rng.uniform(0.5, 5.0)
            b = rng.uniform(0.5, 5.0)
            p = make_beta(N, a, b)
        elif choice == 3:  # param_to_sequence with sinusoid
            # generate 6-parameter vector
            param = np.empty(6)
            param[0] = rng.uniform(0.2, 1.0)   # alpha
            param[1] = rng.uniform(0.5, 5.0)   # delta
            param[2] = rng.uniform(-2.0, 2.0)  # beta (linear)
            param[3] = rng.uniform(-2.0, 2.0)  # gamma (quadratic)
            param[4] = rng.uniform(0.2, 5.0)   # freq
            param[5] = rng.uniform(-1.0, 1.0)  # amp
            p = param_to_sequence(param, N)
        else:  # random Dirichlet
            random_vec = rng.random(N)
            p = random_vec / np.sum(random_vec)
        
        if p is None:
            continue
        update(p)
    
    # take best candidate(s) for refinement
    if best_seq is not None:
        # perform local refinement using balance_search multiple times
        # Run several restarts from perturbed versions
        for restart in range(5):
            if time.time() >= deadline - 30:
                break
            # perturb best_seq slightly and run balance_search
            y = np.log(best_seq + 1e-300)
            # add Gaussian noise to log domain
            noise = rng.normal(0.0, 0.05, size=y.shape)
            y_pert = y + noise
            p_start = softmax_array(y_pert)
            if p_start is None:
                continue
            # run extended balance search
            # allocate time slice
            remaining = deadline - time.time()
            timeslice = max(30.0, min(120.0, remaining / 4.0))
            balance_search(p_start, time_limit=timeslice)
    
    # Final polishing if time left
    if best_seq is not None and time.time() < deadline - 10:
        # final balance search on best
        remaining = deadline - time.time()
        balance_search(best_seq, time_limit=remaining - 5)
    
    # Return best sequence as list of floats
    if best_seq is None:
        # fallback to power
        best_seq = make_power(N, 0.5, 1.0)
    return [float(x) for x in best_seq]

The algorithm proceeds in several stages:

def next_pow2(m: int) -> int:
    n = max(1, int(m))
    return 1 << (n - 1).bit_length()


def finite_positive_array(x):
    """Normalise a 1‑D array to a probability vector, enforce positivity."""
    try:
        a = np.asarray(x, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(0.0, a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return None
    a = a / s
    a = np.minimum(1000.0, a)
    s2 = float(np.sum(a))
    if s2 <= 0:
        return None
    return a / s2


def softmax_array(y):
    """Stable softmax (used for the log‑parameterisation)."""
    try:
        y = np.asarray(y, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if y.size == 0 or not np.all(np.isfinite(y)):
        return None
    y = y - float(np.mean(y))
    m = float(np.max(y))
    e = np.exp(y - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0:
        return None
    p = e / s
    p = np.maximum(0.0, p)
    s2 = float(np.sum(p))
    if s2 <= 0:
        return None
    return p / s2


def fast_score_array(p, lft=0):
    """Exact evaluation of the objective (fast – uses FFT for large n)."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    if n == 0:
        return float("inf")
    p = np.maximum(0.0, p)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return float("inf")
    if n <= 800:
        c = np.convolve(p, p)
        mc = float(np.max(c)) if c.size else 0.0
    else:
        if lft == 0:
            lft = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, n=lft)
        c = np.fft.irfft(F * F, n=lft)
        c = c[:2 * n - 1]
        mc = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mc) or mc < 0:
        mc = 0.0
    return float(2.0 * n * mc / (s * s))


def conv_score_array(p, lft=0):
    """Return (convolution, max coefficient) for a given p."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    p = np.maximum(0.0, p)
    s = float(np.sum(p))
    if s <= 0:
        p = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        p = p / s
    if n <= 800:
        c = np.convolve(p, p)
        c = c[:2 * n - 1]
    else:
        if lft == 0:
            lft = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, n=lft)
        c = np.fft.irfft(F * F, n=lft)
        c = c[:2 * n - 1]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    mc = float(np.max(c)) if c.size else 0.0
    if mc < 0 or not np.isfinite(mc):
        mc = 0.0
    return c, mc


def update_best_from_p(p):
    """Normalise p, evaluate (fast) and keep the best seen so far."""
    global _BEST_P, _BEST_SEQ, _BEST_SCORE
    p = finite_positive_array(p)
    if p is None:
        return
    sc = fast_score_array(p)
    if np.isfinite(sc) and sc < _BEST_SCORE:
        _BEST_SCORE = sc
        _BEST_P = p.copy()
        _BEST_SEQ = [float(x) for x in p]
        print(f"new best: score={sc:.9f}, n={p.size}")


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def make_power(n, alpha=0.5, delta=1.0):
    n = max(1, int(n))
    i = np.arange(n, dtype=np.float64)
    p = (i + float(delta)) ** (-float(alpha))
    p = np.maximum(1e-300, p)
    return finite_positive_array(p)


def make_exponential(n, alpha=1.0):
    n = max(1, int(n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    i = np.arange(n, dtype=np.float64) / float(n - 1)
    p = np.exp(-float(alpha) * i)
    return finite_positive_array(p)


def truncated_sqrt_sequence(n):
    """
    First n coefficients of (1 - x)^(-1/2) (the binomial series).
    After normalisation their self‑convolution has its first n entries ≈1.
    """
    n = max(1, int(n))
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            j = k - i
            s += b[i] * b[j]
        bk = (1.0 - s) / 2.0
        if bk < 0.0:
            bk = 0.0
        b[k] = bk
    return finite_positive_array(b)


def make_beta(n, a, b):
    """Beta‑distribution shape on {0,…,n‑1} (a,b>0)."""
    if a <= 0 or b <= 0:
        return None
    n = max(1, int(n))
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / float(n)
    p = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    return finite_positive_array(p)


def param_to_sequence(param, n):
    """
    Six‑parameter log‑shape:
      log p_i = -α log(i+δ) + β x + γ x² + amp·sin(2π·freq·x)
    where x = i/(n‑1).  All parameters are real.
    """
    try:
        alpha, delta, beta, gamma, freq, amp = [float(v) for v in param[:6]]
    except Exception:
        return None
    if not all(np.isfinite(v) for v in (alpha, delta, beta, gamma, freq, amp)):
        return None
    if delta <= 1e-12:
        return None
    n = max(1, int(n))
    i = np.arange(n, dtype=np.float64)
    x = i / float(n - 1) if n > 1 else np.zeros(1, dtype=np.float64)
    logp = -alpha * np.log(i + delta) + beta * x + gamma * x * x
    if abs(amp) > 1e-12 and abs(freq) > 1e-12:
        logp = logp + amp * np.sin(2.0 * math.pi * freq * x)
    p = softmax_array(logp)
    return p


# ----------------------------------------------------------------------
# Local optimiser – “balance search’’ (active‑set smoothing)
# ----------------------------------------------------------------------
def balance_search(p0, time_limit=60.0):
    """
    Smoothed‑max gradient descent on log‑probabilities.
    The routine updates the global best sequence whenever it finds improvement.
    """
    p = finite_positive_array(p0)
    if p is None:
        return
    n = p.size
    local_deadline = time.monotonic() + max(1.0, float(time_limit))

    # initialise log‑variables
    y = np.log(np.maximum(1e-300, p))
    y -= np.mean(y)
    best_p = p.copy()
    best_score = fast_score_array(best_p)
    best_y = y.copy()

    # pre‑allocate FFT lengths for large n
    lscore = 0 if n <= 800 else next_pow2(2 * n - 1)
    lv = 0 if n <= 800 else next_pow2(3 * n)

    gamma = 30.0                # sharpness of the soft‑max on c/C
    stagnation = 0
    grad_steps = (0.01, 0.03, 0.1, 0.3, 1.0, 3.0)
    mult_steps = (0.05, 0.2, 0.7, 2.0)

    it = 0
    while time.monotonic() < local_deadline and time.monotonic() < _DEADLINE:
        it += 1
        c, C = conv_score_array(best_p, lscore)
        if C <= 1e-300 or not np.isfinite(C):
            break

        # ---- weights for the (soft) active set -----------------
        z = c / (C + 1e-300)                # normalised convolution
        zmax = float(np.max(z)) if z.size else 1.0
        w = np.exp(gamma * (z - zmax))
        sw = float(np.sum(w))
        if sw <= 0 or not np.isfinite(sw):
            w = np.ones_like(c)
            sw = float(c.size)
        lam = w / sw                        # probability distribution over indices

        # ---- convolution of reversed p with lam (v) -------------
        if n <= 800:
            v = np.convolve(best_p[::-1], lam, mode="full")[n - 1:2 * n - 1]
        else:
            Fp = np.fft.rfft(best_p[::-1], n=lv)
            Fl = np.fft.rfft(lam, n=lv)
            v = np.fft.irfft(Fp * Fl, n=lv)
            v = v[n - 1:2 * n - 1]

        v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
        v = np.maximum(v, 1e-300)
        vnorm = v / (C + 1e-300)           # normalised
        mean_v = float(np.dot(best_p, vnorm))
        grad_log = best_p * (vnorm - mean_v)

        # ---- try a few trial steps (log‑gradient) -------------
        trial_score = best_score
        trial_y = None
        trial_p = None

        for step in grad_steps:
            y_try = best_y - float(step) * grad_log
            p_try = softmax_array(y_try)
            if p_try is None:
                continue
            sc = fast_score_array(p_try, lscore)
            if np.isfinite(sc) and sc < trial_score - 1e-12:
                trial_score, trial_y, trial_p = sc, y_try, p_try

        # ---- multiplicative / exponentiated‑gradient steps -------
        logv = np.log(vnorm)
        logv -= np.mean(logv)
        for step in mult_steps:
            y_try = best_y - float(step) * logv
            p_try = softmax_array(y_try)
            if p_try is None:
                continue
            sc = fast_score_array(p_try, lscore)
            if np.isfinite(sc) and sc < trial_score - 1e-12:
                trial_score, trial_y, trial_p = sc, y_try, p_try

        # ---- accept if we improved -------------------------------
        if trial_y is not None:
            best_y = trial_y
            best_p = trial_p.copy()
            best_score = float(trial_score)
            stagnation = 0
            if it % 25 == 0:
                print(f"balance n={n} it={it} score={best_score:.9f}")
        else:
            stagnation += 1
            # adapt gamma when we are stuck
            if stagnation % 20 == 0:
                if gamma < 500.0:
                    gamma *= 2.0
                else:
                    gamma = max(5.0, gamma * 0.5)

            # occasional random “kick’’ to escape flat regions
            if stagnation % 30 == 0:
                xs = np.linspace(0.0, 1.0, n, dtype=np.float64)
                for _ in range(5):
                    amp = 0.04 + 0.08 * (stagnation / 200.0)
                    f1 = 0.5 + 3.0 * _RNG.random()
                    f2 = 1.0 + 5.0 * _RNG.random()
                    ph1 = _RNG.uniform(0.0, 2.0 * math.pi)
                    ph2 = _RNG.uniform(0.0, 2.0 * math.pi)
                    kick = amp * (
                        (0.3 + 0.7 * _RNG.random()) * np.sin(2.0 * math.pi * f1 * xs + ph1)
                        + (0.2 + 0.5 * _RNG.random()) * np.cos(2.0 * math.pi * f2 * xs + ph2)
                        + 0.25 * (xs - 0.5)
                    )
                    y_try = best_y + kick
                    p_try = softmax_array(y_try)
                    if p_try is None:
                        continue
                    sc = fast_score_array(p_try, lscore)
                    if np.isfinite(sc) and sc < best_score:
                        best_y = y_try.copy()
                        best_p = p_try.copy()
                        best_score = sc

    update_best_from_p(best_p)
    return best_p, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) attains a
    low evaluation score.  The routine runs for at most `budget_s`
    seconds, using a mixture of analytic seeds, random parameter search
    and a dedicated gradient‐based local optimiser.
    """
    global _DEADLINE, _BEST_P, _BEST_SEQ, _BEST_SCORE, _RNG

    # ------------------------------------------------------------------
    # time‑budget handling
    # ------------------------------------------------------------------
    budget_s = float(budget_s) if isinstance(budget_s, (int, float)) else 1000.0
    if not np.isfinite(budget_s) or budget_s <= 0:
        budget_s = 950.0
    # leave a safety margin for the final return
    _DEADLINE = time.time() + max(60.0, min(950.0, budget_s - 60.0))
    _RNG = np.random.default_rng(int(seed) % (2 ** 32))

    # reset globals
    _BEST_P = None
    _BEST_SEQ = None
    _BEST_SCORE = float("inf")

    # ------------------------------------------------------------------
    # 1) Fixed analytic seeds (truncated sqrt is a strong baseline)
    # ------------------------------------------------------------------
    N = 4806                     # primary working length
    for seed_func, name in [
        (truncated_sqrt_sequence, "trunc_sqrt"),
        (lambda n: make_power(n, 0.5, 1.0), "power05"),
        (lambda n: make_power(n, 0.4, 1.0), "power04"),
        (lambda n: make_exponential(n, 0.9), "exp09"),
    ]:
        p = seed_func(N)
        if p is not None:
            print(f"initial seed {name}")
            update_best_from_p(p)

    # ------------------------------------------------------------------
    # 2) Randomised parameter sweep (≈2000 candidates)
    # ------------------------------------------------------------------
    eval_budget = 2000
    for _ in range(eval_budget):
        if time.time() >= _DEADLINE - 30:
            break

        # Choose a family at random
        fam = _RNG.integers(0, 5)
        if fam == 0:           # power‑law
            alpha = _RNG.uniform(0.2, 0.9)
            delta = _RNG.uniform(0.5, 5.0)
            p = make_power(N, alpha, delta)
        elif fam == 1:         # exponential
            a = _RNG.uniform(0.5, 3.0)
            p = make_exponential(N, a)
        elif fam == 2:         # beta‑distribution
            a = _RNG.uniform(0.5, 5.0)
            b = _RNG.uniform(0.5, 5.0)
            p = make_beta(N, a, b)
        elif fam == 3:         # flexible log‑parameter family
            param = np.empty(6)
            param[0] = _RNG.uniform(0.2, 1.0)          # alpha
            param[1] = _RNG.uniform(0.5, 5.0)          # delta
            param[2] = _RNG.uniform(-2.0, 2.0)         # beta (linear term)
            param[3] = _RNG.uniform(-2.0, 2.0)         # gamma (quadratic term)
            param[4] = _RNG.uniform(0.2, 5.0)          # freq
            param[5] = _RNG.uniform(-1.0, 1.0)         # amp
            p = param_to_sequence(param, N)
        else:                  # plain Dirichlet sample
            raw = _RNG.random(N)
            p = raw / np.sum(raw)

        if p is None:
            continue
        update_best_from_p(p)

    # ------------------------------------------------------------------
    # 3) Refine the current best with several balance‑search restarts
    # ------------------------------------------------------------------
    if _BEST_P is not None:
        for restart in range(6):
            if time.time() >= _DEADLINE - 30:
                break
            # small random perturbation in log‑space
            y = np.log(np.maximum(1e-300, _BEST_P))
            y += _RNG.normal(0.0, 0.04, size=y.shape)
            p0 = softmax_array(y)
            if p0 is None:
                continue
            # allocate a time slice for this refinement run
            remaining = _DEADLINE - time.time()
            slice_time = max(20.0, min(120.0, remaining / max(1, 6 - restart)))
            balance_search(p0, time_limit=slice_time)

    # ------------------------------------------------------------------
    # 4) Final polishing on the remaining budget
    # ------------------------------------------------------------------
    if _BEST_P is not None and time.time() < _DEADLINE - 5:
        remaining = _DEADLINE - time.time()
        balance_search(_BEST_P.copy(), time_limit=remaining - 2)

    # ------------------------------------------------------------------
    # 5) Fallback (should never be needed)
    # ------------------------------------------------------------------
    if _BEST_SEQ is None:
        fallback = make_power(N, 0.5, 1.0)
        _BEST_SEQ = [float(x) for x in fallback]

    # final official check (non‑blocking)
    try:
        official = float(evaluate_sequence(_BEST_SEQ))
        if np.isfinite(official):
            _BEST_SCORE = official
    except Exception:
        pass

    print(f"final returned score={_BEST_SCORE:.9f}, n={len(_BEST_SEQ)}")
    return _BEST_SEQ
