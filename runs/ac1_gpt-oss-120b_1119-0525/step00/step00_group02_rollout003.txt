"""
Local optimisation of the L∞‑norm of the auto‑convolution.

The algorithm starts from an arcsine‑like distribution
    a_i ∝ 1 / sqrt((i+0.5)*(N-0.5-i))
which is the discrete analogue of the continuous arcsine law whose
self‑convolution is close to uniform.  From this start we repeatedly
reduce the largest convolution entry by a projected gradient step.
The gradient of the maximum entry `k_max` of `a*a` is simply

    g_i = 2 * a_{k_max-i}   for indices where k_max-i is inside [0,N),

and zero otherwise.  After moving a small amount in the direction
`-g` we clip to the feasible box [0,1000] and renormalise the total
mass (the objective is scale‑invariant).  If a step does not improve the
score we fall back to a cheap random block perturbation (the same
heuristic used in the reference implementation).  The best sequence
found before the time budget expires is returned.

The implementation is fully vectorised, does not use closures or
lambdas, and prints occasional progress reports.
"""

import time
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` (scale‑invariant)."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = float(seq.sum())
    if total < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    n = seq.shape[0]
    return 2.0 * n * max_b / (total * total)


def _gradient_max_convolution(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the convolution entry with index ``k_max`` with respect to ``a``.
    For a non‑negative vector ``a`` the gradient is
        g_i = 2 * a_{k_max-i}   if 0 ≤ i < N and 0 ≤ k_max-i < N,
        g_i = 0                otherwise.
    """
    N = a.shape[0]
    start = max(0, k_max - (N - 1))
    end = min(k_max, N - 1)
    if start > end:
        return np.zeros_like(a)
    idx = np.arange(start, end + 1, dtype=int)
    grad = np.zeros_like(a)
    # ``k_max - idx`` is always a valid index inside the array.
    grad[idx] = 2.0 * a[k_max - idx]
    return grad

                best_score = score
                a = cand
                improved = True
                # keep the step size that succeeded
                eta = trial_eta
                break
            else:
                trial_eta *= 0.5      # reduce step size and try again

        # 4) if gradient step failed, fall back to a cheap random block tweak
        if not improved:
            width = int(rng.integers(max(1, N // 200), max(2, N // 12)))
            start = int(rng.integers(0, max(1, N - width + 1)))
            stop = min(N, start + width)

            cand = a.copy()
            # smaller log‑normal factor than the reference (more cautious)
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand[start:stop] *= factor

            # occasional mirroring – helps to enforce a degree of symmetry
            if rng.random() < 0.30:
                cand = 0.5 * (cand + cand[::-1])

            cand = np.clip(cand, 0.0, 1000.0)
            total = cand.sum()
            if total > 0.0:
                cand = cand * (a.sum() / total)

            score = _score(cand)
            if score < best_score:
                best = cand.copy()
                best_score = score
                a = cand
                improved = True

        # 5) slowly decay the learning rate to stabilise later iterations
        eta *= decay_factor
        it += 1
        if it % report_every == 0:
            print(f"[iter {it:6d}] best score = {best_score:.6f}")

    print(f"[done] iter={it}, best score={best_score:.6f}")
    # Convert to plain Python floats for the caller
    return [float(v) for v in best]
```
