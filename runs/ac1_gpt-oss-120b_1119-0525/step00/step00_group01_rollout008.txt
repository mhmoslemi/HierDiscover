def project_simplex(v):
    # Ensure nonnegative and sum to 1.
    # Implementation: sort descending, compute threshold.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _apply_move(p, conv, i, j, delta):
    # p and conv are numpy arrays
    n = p.shape[0]
    # copy arrays
    p_new = p.copy()
    conv_new = conv.copy()
    # make sure delta <= p[i]
    delta = min(delta, p_new[i])
    if delta <= 0:
        return p_new, conv_new
    p_new[i] -= delta
    p_new[j] += delta
    # update convolution
    conv_new[j:j+n] += 2.0 * delta * p
    conv_new[i:i+n] -= 2.0 * delta * p
    conv_new[2*j] += delta * delta
    conv_new[2*i] += delta * delta
    conv_new[i+j] -= 2.0 * delta * delta
    return p_new, conv_new
```

"""
Propose a sequence that beats the trivial constant construction.

The evaluation function is
    f(a) = 2 * n * max(conv(a,a)) / (sum(a)^2)

The goal is to make the max of the self‑convolution as small as possible.
For a constant vector the max is 1/n and the score equals exactly 2.
A non‑constant distribution can reduce the peak (e.g. [0.5,0.25,0.25] gives
score 1.875).  This search explores short vectors (n≤30) with many random
probability vectors, performs a tiny hill‑climb on each and keeps the best
one found.  The routine is deterministic for a given seed and respects the
1000 s time budget.
"""

import time
import math
import numpy as np

def _objective(seq: np.ndarray) -> float:
    """Exact objective value, assumes non‑negative finite entries."""
    n = seq.size
    if n == 0:
        return math.inf
    total = np.sum(seq)
    if total < 0.01:                     # same safety as the official evaluator
        return math.inf
    conv = np.convolve(seq, seq)
    return 2.0 * n * np.max(conv) / (total * total)


def _hill_climb(p: np.ndarray, rng: np.random.Generator,
                max_iter: int = 200, init_temp: float = 0.01) -> np.ndarray:
    """
    Very small simulated‑annealing hill‑climb that preserves the simplex.
    Moves a random amount of mass from one index to another.
    """
    n = p.size
    best = p.copy()
    best_score = _objective(best)

    cur = p.copy()
    cur_score = best_score
    temp = init_temp

    for _ in range(max_iter):
        # pick a source with positive mass
        src = rng.integers(0, n)
        while cur[src] <= 1e-12:
            src = rng.integers(0, n)

        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)

        # propose moving a random fraction of src to dst
        delta = cur[src] * rng.random() * 0.5          # at most half of src
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_score = _objective(cand)

        # accept if better, otherwise with Metropolis probability
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score

        # cool down a little
        temp *= 0.995

    return best
