def lp_optimize(f, iterations=10, mix=0.5):
    try:
        import cvxpy as cp
    except Exception:
        return f
    n = f.shape[0]
    for it in range(iterations):
        # Build convolution matrix
        conv_mat = np.zeros((2*n-1, n))
        for i in range(n):
            conv_mat[i:i+n, i] = f
        # Variables
        g = cp.Variable(n, nonneg=True)
        t = cp.Variable()
        constraints = [cp.sum(g) == 1, conv_mat @ g <= t]
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            # fallback to SCS or other solver
            prob.solve(solver=cp.SCS, warm_start=True, verbose=False, max_iters=2000)
        if g.value is None:
            break
        g_opt = np.maximum(0.0, np.asarray(g.value, dtype=np.float64).reshape(-1))
        # Mix
        f = (1 - mix) * f + mix * g_opt
        # Normalize
        s = f.sum()
        if s > 0:
            f = f / s
    return f
```

def upscale_sequence(seq, new_n):
    seq = np.asarray(seq, dtype=np.float64)
    old_n = seq.size
    xs_old = np.linspace(0.0, 1.0, old_n)
    xs_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(xs_new, xs_old, seq)
    up = up / np.sum(up)
    return up
```

    f = np.asarray(seq,dtype=float)
    for i in range(iters):
        # build matrix
        n = f.size
        conv_mat = np.zeros((2*n-1, n))
        for j in range(n):
            conv_mat[j:j+n, j] = f
        # variables
        g = cp.Variable(n, nonneg=True)
        t = cp.Variable()
        constraints = [cp.sum(g) == 1, conv_mat @ g <= t]
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True)
        if g.value is None:
            break
        g_opt = np.maximum(0.0, np.asarray(g.value))
        f = (1 - mix) * f + mix * g_opt
        f = f / np.sum(f)
    return f

"""
Propose a step‐function sequence with a low evaluation score.

Algorithm
---------
1.  Start from the classic “truncated‑sqrt’’ sequence (the discrete (1‑x)^‑½
    coefficients).  This shape already gives a score close to the known
    lower bound π/2 ≈ 1.5708.

2.  Refine the shape on a modest length (n≈256) by *exact* linear‑programming
    of the “partner’’ step (the LP described in the original paper:
    for a fixed f minimise  max(f·g) subject to g≥0, Σg=1).  The LP is solved
    with CVXPy/ECOS.  After each LP solution we blend the new partner into the
    current shape (convex combination, renormalise) and repeat a few times.
    This removes the small bias of the gradient‑descent method used before.

3.  Upsample the improved short shape to a larger working length
    (default n=4096) by linear interpolation and renormalise.

4.  Perform a fast “balancing’’ local search on the large shape.
    The routine is identical to the one from the reference implementation
    (exponentiated‑gradient / smooth‑max descent using FFT for speed).  It
    improves the sequence until the deadline is reached.

5.  Return the best sequence found (the official evaluator is called only
    for the final candidate to avoid excessive overhead).

The whole process respects the 1000 s time limit, works with or without
CVXPy, and prints progress messages with intermediate scores.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global state (updated by the optimisation routines)
# ----------------------------------------------------------------------
_DEADLINE = 0.0          # absolute time when we must stop
_BEST_SEQ = None         # best list returned to the caller
_BEST_SCORE = float("inf")
_RNG = np.random.default_rng(12345)


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(m: int) -> int:
    """Smallest power of two >= m (at least 1)."""
    n = max(1, int(m))
    return 1 << (n - 1).bit_length()


def finite_positive_array(x):
    """Convert an arbitrary iterable to a non‑negative, normalised numpy array."""
    try:
        a = np.asarray(x, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(0.0, a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return None
    a = a / s
    a = np.minimum(1000.0, a)
    s2 = float(np.sum(a))
    if s2 <= 0:
        return None
    return a / s2


def softmax_array(y):
    """Numerically stable soft‑max."""
    try:
        y = np.asarray(y, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if y.size == 0 or not np.all(np.isfinite(y)):
        return None
    y = y - float(np.mean(y))
    m = float(np.max(y))
    e = np.exp(y - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0:
        return None
    p = e / s
    p = np.maximum(0.0, p)
    s2 = float(np.sum(p))
    if s2 <= 0:
        return None
    return p / s2


def fast_score_array(p, lft=0):
    """Fast evaluation of 2·n·max(a*a) / (sum a)^2 for a normalised array."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    if n == 0:
        return float("inf")
    p = np.maximum(0.0, p)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return float("inf")
    if n <= 800:
        conv = np.convolve(p, p)
        mc = float(np.max(conv))
    else:
        if lft == 0:
            lft = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, n=lft)
        conv = np.fft.irfft(F * F, n=lft)
        conv = conv[: 2 * n - 1]
        mc = float(np.max(conv))
    if not np.isfinite(mc) or mc < 0:
        mc = 0.0
    return float(2.0 * n * mc / (s * s))


def update_best_from_array(p):
    """Normalise p, evaluate it (officially if possible) and keep the best."""
    global _BEST_SEQ, _BEST_SCORE
    p = finite_positive_array(p)
    if p is None:
        return
    seq = [float(x) for x in p]
    # official evaluator – may be slow, call only for promising candidates
    try:
        sc = float(evaluate_sequence(seq))
    except Exception:
        sc = float("inf")
    # fall back to the fast surrogate if needed
    if not np.isfinite(sc):
        sc = fast_score_array(p)
    if np.isfinite(sc) and sc < _BEST_SCORE:
        _BEST_SCORE = sc
        _BEST_SEQ = seq
        print(f"[+] new best: score={sc:.9f}, n={p.size}")


def make_power(n, alpha=0.5, delta=1.0):
    """Power‑law family a_i ∝ (i+δ)^‑α."""
    n = max(1, int(n))
    i = np.arange(n, dtype=np.float64)
    p = (i + float(delta)) ** (-float(alpha))
    p = np.maximum(1e-300, p)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    return p / s


def truncated_sqrt_sequence(n):
    """
    First n coefficients of (1‑x)^‑½, normalised.
    The un‑scaled convolution is exactly 1 for the first n entries.
    """
    n = max(1, int(n))
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            j = k - i
            s += b[i] * b[j]
        bk = (1.0 - s) / 2.0
        if bk < 0.0:
            bk = 0.0
        b[k] = bk
    s = float(np.sum(b))
    if s <= 0 or not np.isfinite(s):
        return None
    b = np.maximum(0.0, b) / s
    return b


def upscale_sequence(seq, new_n):
    """Linear interpolation from old to new length, re‑normalised."""
    old = np.asarray(seq, dtype=np.float64).reshape(-1)
    old_n = old.size
    if old_n == new_n:
        return old.copy()
    xs_old = np.linspace(0.0, 1.0, old_n)
    xs_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(xs_new, xs_old, old)
    up = np.maximum(0.0, up)
    s = float(np.sum(up))
    if s > 0:
        up /= s
    else:
        up = np.ones(new_n, dtype=np.float64) / new_n
    return up


# ----------------------------------------------------------------------
# Exact linear‑programming step (small n)
# ----------------------------------------------------------------------
def lp_refine_small(f, iters=12, mix=0.3):
    """
    Perform a few LP steps for a short vector f (Σf = 1).
    Each step solves   min_t  s.t.  (f * g)_k ≤ t   , Σg=1,g≥0.
    The solution g is blended into f: f ← (1‑mix)·f + mix·g.
    """
    try:
        import cvxpy as cp
    except Exception:
        # CVXPy not available – bail out gracefully
        return f
    n = f.size
    for it in range(iters):
        if time.monotonic() >= _DEADLINE - 5:
            break
        # Build convolution matrix: conv_mat[k,i] = f[k‑i] when valid, else 0.
        conv_mat = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            conv_mat[i : i + n, i] = f
        # CVX variables
        g = cp.Variable(n, nonneg=True)
        t = cp.Variable()
        constraints = [cp.sum(g) == 1, conv_mat @ g <= t]
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-8, reltol=1e-8)
        except Exception:
            # fallback to SCS if ECOS fails
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=2500, verbose=False)
        if g.value is None:
            break
        g_opt = np.maximum(0.0, np.asarray(g.value, dtype=np.float64).reshape(-1))
        # blend and re‑normalise (both are already normalised)
        f = (1.0 - mix) * f + mix * g_opt
        s = float(np.sum(f))
        if s > 0:
            f = f / s
    return f

# Large‑scale “balancing’’ local search (identical to the reference version)
# ----------------------------------------------------------------------
def conv_score_array(p, lft=0):
    """Return (full convolution, max‑value) for a normalised probability vector."""
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = p.size
    p = np.maximum(0.0, p)
    s = float(np.sum(p))
    if s <= 0:
        p = np.ones(n, dtype=np.float64) / max(1, n)
    else:
        p = p / s
    if n <= 800:
        c = np.convolve(p, p)
        c = c[: 2 * n - 1]
    else:
        if lft == 0:
            lft = next_pow2(2 * n - 1)
        F = np.fft.rfft(p, n=lft)
        c = np.fft.irfft(F * F, n=lft)
        c = c[: 2 * n - 1]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    mc = float(np.max(c)) if c.size else 0.0
    if mc < 0 or not np.isfinite(mc):
        mc = 0.0
    return c, mc


def balance_search(p0, time_limit=60.0):
    """
    Fast FFT‑based smoothing‑max descent on the L∞‑norm of the self‑convolution.
    Operates entirely on the probability simplex.
    """
    p = finite_positive_array(p0)
    if p is None:
        return
    n = p.size
    local_deadline = time.monotonic() + max(1.0, float(time_limit))

    # work in log‑probability space (mirror‑descent on the simplex)
    y = np.log(np.maximum(1e-300, p))
    y -= np.mean(y)
    best_p = p.copy()
    best_y = y.copy()
    best_score = fast_score_array(best_p, 0)

    lscore = 0 if n <= 800 else next_pow2(2 * n - 1)
    lv = 0 if n <= 800 else next_pow2(3 * n)

    gamma = 50.0
    stagnation = 0
    grad_steps = (0.01, 0.03, 0.1, 0.3, 1.0, 3.0)
    mult_steps = (0.05, 0.2, 0.7, 2.0)

    it = 0
    while time.monotonic() < local_deadline and time.monotonic() < _DEADLINE:
        it += 1
        c, C = conv_score_array(best_p, lscore)
        if C <= 1e-300 or not np.isfinite(C):
            break

        # soft‑max across diagonal entries – provides a smooth approximation of the max
        z = c / (C + 1e-300)
        zmax = float(np.max(z)) if z.size else 1.0
        w = np.exp(gamma * (z - zmax))
        sw = float(np.sum(w))
        if sw <= 0 or not np.isfinite(sw):
            w = np.ones_like(c)
            sw = float(c.size)
        lam = w / sw

        # v_i = Σ_k λ_k a_{k‑i}
        if n <= 800:
            v = np.convolve(best_p[::-1], lam, mode="full")[n - 1 : 2 * n - 1]
        else:
            Fp = np.fft.rfft(best_p[::-1], n=lv)
            Fl = np.fft.rfft(lam, n=lv)
            v = np.fft.irfft(Fp * Fl, n=lv)
            v = v[n - 1 : 2 * n - 1]

        v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
        v = np.maximum(v, 1e-300)
        vnorm = v / (C + 1e-300)
        mean_v = float(np.dot(best_p, vnorm))
        grad_log = best_p * (vnorm - mean_v)

        trial_score = best_score
        trial_y = None
        trial_p = None

        # log‑gradient steps
        for step in grad_steps:
            ytry = y - float(step) * grad_log
            ptry = softmax_array(ytry)
            if ptry is None:
                continue
            sc = fast_score_array(ptry, lscore)
            if np.isfinite(sc) and sc < trial_score - 1e-12:
                trial_score = sc
                trial_y = ytry
                trial_p = ptry

        # multiplicative / exponentiated‑gradient steps
        logv = np.log(vnorm)
        logv -= np.mean(logv)
        for step in mult_steps:
            ytry = y - float(step) * logv
            ptry = softmax_array(ytry)
            if ptry is None:
                continue
            sc = fast_score_array(ptry, lscore)
            if np.isfinite(sc) and sc < trial_score - 1e-12:
                trial_score = sc
                trial_y = ytry
                trial_p = ptry

        if trial_y is not None:
            y = trial_y
            best_p = trial_p.copy()
            best_y = y.copy()
            best_score = float(trial_score)
            stagnation = 0
            if it % 25 == 0:
                print(f"[balance] n={n} it={it} score={best_score:.9f}")
        else:
            stagnation += 1
            if stagnation % 20 == 0:
                # sharpen / relax the exponential weighting
                gamma = min(500.0, gamma * 2.0) if gamma < 200.0 else max(5.0, gamma * 0.5)

            if stagnation % 30 == 0:
                # random smooth “kick’’ to escape plateaus
                xs = np.linspace(0.0, 1.0, n, dtype=np.float64)
                best_kick_y = best_y.copy()
                best_kick_sc = best_score
                for _ in range(5):
                    amp = 0.04 + 0.08 * (stagnation / 200.0)
                    f1 = 0.5 + 3.0 * _RNG.random()
                    f2 = 1.0 + 5.0 * _RNG.random()
                    ph1 = _RNG.uniform(0.0, 2.0 * math.pi)
                    ph2 = _RNG.uniform(0.0, 2.0 * math.pi)
                    kick = amp * (
                        (0.3 + 0.7 * _RNG.random()) * np.sin(2.0 * math.pi * f1 * xs + ph1)
                        + (0.2 + 0.5 * _RNG.random()) * np.cos(2.0 * math.pi * f2 * xs + ph2)
                        + 0.25 * (xs - 0.5)
                    )
                    ytry = best_y + kick
                    ptry = softmax_array(ytry)
                    if ptry is None:
                        continue
                    sc = fast_score_array(ptry, lscore)
                    if np.isfinite(sc) and sc < best_kick_sc:
                        best_kick_sc = sc
                        best_kick_y = ytry.copy()
                y = best_kick_y
                best_p = softmax_array(y)
                best_score = float(best_kick_sc)

    update_best_from_array(best_p)
    return best_p, best_score

def propose_candidate(seed=42, budget_s=1600, **kwargs):
    """
    Return a list of non‑negative heights (the step‑function) with a low score.
    The function respects the time budget (default ≈1000 s) and uses a hybrid
    LP + FFT‑based local optimisation.
    """
    global _DEADLINE, _BEST_SEQ, _BEST_SCORE, _RNG

    # ------------------------------------------------------------------
    # time management
    # ------------------------------------------------------------------
    budget_s = float(budget_s) if isinstance(budget_s, (int, float)) else 1600.0
    if not np.isfinite(budget_s) or budget_s <= 0:
        budget_s = 1000.0
    # Keep a small safety margin for the final return
    _DEADLINE = time.monotonic() + max(60.0, min(950.0, budget_s - 60.0))
    _RNG = np.random.default_rng(int(seed) % (2 ** 32))

    _BEST_SEQ = None
    _BEST_SCORE = float("inf")

    print("[*] deadline set to {:.2f} s from now".format(_DEADLINE - time.monotonic()))

    # ------------------------------------------------------------------
    # 1) start from the classic truncated‑sqrt seed (n≈256)
    # ------------------------------------------------------------------
    n_start = 256
    seed_seq = truncated_sqrt_sequence(n_start)
    if seed_seq is None:
        seed_seq = make_power(n_start, 0.5, 1.0)
    update_best_from_array(seed_seq)

    # ------------------------------------------------------------------
    # 2) small‑scale LP refinement (if CVXPy is available)
    # ------------------------------------------------------------------
    if time.monotonic() < _DEADLINE - 30:
        refined_small = lp_refine_small(seed_seq, iters=12, mix=0.3)
        update_best_from_array(refined_small)
    else:
        refined_small = seed_seq

    # ------------------------------------------------------------------
    # 3) upscale to a larger working length
    # ------------------------------------------------------------------
    large_n = 4096
    upscaled = upscale_sequence(refined_small, large_n)
    update_best_from_array(upscaled)

    # ------------------------------------------------------------------
    # 4) final polishing with the FFT‑based balancing search
    # ------------------------------------------------------------------
    remaining = _DEADLINE - time.monotonic()
    if remaining > 30:
        print("[*] starting balance search (n={}) – {:.1f}s left".format(large_n, remaining))
        balance_search(upscaled, time_limit=remaining - 10)

    # ------------------------------------------------------------------
    # 5) final fallback – ensure something is returned
    # ------------------------------------------------------------------
    if _BEST_SEQ is None:
        _BEST_SEQ = [1.0]

    # final official evaluation (does not affect the returned score)
    try:
        official = float(evaluate_sequence(_BEST_SEQ))
        if np.isfinite(official):
            _BEST_SCORE = official
    except Exception:
        pass

    print("[*] final returned score={:.9f}, n={}".format(_BEST_SCORE, len(_BEST_SEQ)))
    return _BEST_SEQ
