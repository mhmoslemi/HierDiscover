def proj_simplex(v):
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def proj_simplex(v):
    # v: 1d numpy array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def search_length(n, time_budget):
    # random initial weights
    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    # Simulated annealing parameters
    start = time.time()
    T0 = 0.1
    Tmin = 1e-5
    decay = 0.9995 # each iteration warm
    T = T0
    while time.time() < time_budget:
        # propose
        i = np.random.randint(0, n)
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # determine step amount
        eps = np.random.exponential(scale=0.1)  # random move size fraction
        eps = min(eps, a[i])  # can't move more than available
        # propose new array copy
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # compute score
        score_new = compute_score(a_new)
        # accept criteria
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            # maybe print progress
        if score_new < compute_score(a) or np.random.rand() < np.exp(-(score_new - compute_score(a))/T):
            a = a_new
        # update T
        T = max(Tmin, T * decay)
    return best_a, best_score
```

def refine(a, n, max_iters, initial_lr):
    a = a.copy()
    for it in range(max_iters):
        b = np.convolve(a, a)
        M = b.max()
        # compute grad
        idx = np.where(b >= M - 1e-12)[0]  # indices achieving max
        grad = np.zeros_like(a)
        for k in idx:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            grad[i_range] += 2 * a[j_range]
        # step
        lr = initial_lr / np.sqrt(it + 1)
        # exponential update
        a = a * np.exp(-lr * grad)
        # renorm
        total = a.sum()
        if total == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / total
        # optionally stop early if M small
    return a, 2 * n * M

"""
Propose a step‑function (a list of non‑negative heights) that minimizes the
evaluation function

    2 * n * max(a * a) / (sum(a)**2)

where ``a*a`` is the full discrete convolution of the sequence with itself.
The optimum is not the constant sequence – by choosing non‑uniform heights the
maximum convolution entry can be reduced below the uniform value 1/n.
The algorithm performs a fast stochastic search:

* Small problem sizes (n = 2 … 40) are explored because the factor *n* in the
  objective penalises large lengths.
* For each length many random starts (Dirichlet samples) are improved by a
  simulated‑annealing style random walk that moves a little mass from one
  index to another.
* The current best solution is always kept.  A simple multiplicative gradient
  step (sub‑gradient of the max‑convolution) is applied after the annealing
  phase to fine‑tune the best candidate.
* The whole search respects the time budget (default 1000 s) and returns a
  valid sequence as soon as time runs out.

The code is pure NumPy / Python and uses no external files or network IO.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score_of(a: np.ndarray) -> float:
    """
    Compute the objective value for a probability vector ``a``.
    The sum of ``a`` is assumed to be 1 (scale‑invariant), so the score is
    simply ``2 * n * max(convolution)``.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b))


def _subgradient_max_convolution(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(a*a)`` with respect to ``a``.
    For each index ``k`` where the convolution attains its maximum,
    the contribution to the gradient at position ``i`` is ``2 * a[k-i]``.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    max_idx = np.where(conv >= max_val - 1e-14)[0]

    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        grad[i_range] += 2.0 * a[j_range]
    return grad


def _multiplicative_update(a: np.ndarray, grad: np.ndarray, lr: float) -> np.ndarray:
    """
    Perform a multiplicative (mirror‑descent) update that stays on the simplex.
    ``lr`` is a learning‑rate like factor.
    """
    # avoid overflow/underflow: clip gradient to a reasonable range
    step = np.exp(-lr * grad)
    a_new = a * step
    total = a_new.sum()
    if total <= 0.0:
        # degenerate case – fall back to uniform
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


def _anneal_one_restart(a: np.ndarray, max_iter: int, init_temp: float) -> tuple:
    """
    Simulated‑annealing improvement of a single candidate vector.
    Returns the best vector seen during this run and its score.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _score_of(cur)
    best_vec = cur.copy()
    best_score = cur_score

    T = init_temp
    decay = 0.9995  # geometric cooling

    for it in range(max_iter):
        # ----- propose a small mass transfer -----
        i = random.randrange(n)
        if cur[i] <= 1e-12:
            continue                       # nothing to move
        # pick a distinct destination index
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        # step size: a decreasing fraction of the source amount
        frac = 0.10 * (0.99 ** it)
        delta = min(frac * cur[i], cur[i])

        new_vec = cur.copy()
        new_vec[i] -= delta
        new_vec[j] += delta

        # ----- evaluate -----
        new_score = _score_of(new_vec)

        # ----- acceptance test -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-9))
            if random.random() < prob:
                accept = True

        if accept:
            cur = new_vec
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()

        # ----- temperature schedule -----
        T *= decay

        # occasional tiny jitter to avoid stagnation
        if (it + 1) % 200 == 0:
            jitter = np.random.normal(scale=0.001, size=n)
            cur = np.clip(cur + jitter, 0.0, None)
            s = cur.sum()
            if s > 0:
                cur /= s
            else:
                cur = np.full(n, 1.0 / n)
            cur_score = _score_of(cur)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()

    return best_vec, best_score


def _gradient_refine(a: np.ndarray, steps: int = 600, init_lr: float = 0.5) -> np.ndarray:
    """
    Fine‑tune a vector using a simple multiplicative sub‑gradient step.
    """
    n = a.shape[0]
    vec = a.copy()
    lr = init_lr
    for s in range(steps):
        grad = _subgradient_max_convolution(vec)
        vec = _multiplicative_update(vec, grad, lr)
        lr *= 0.99               # slowly shrink step
        # early stop if the max‑convolution is already very small
        if 2.0 * n * np.max(np.convolve(vec, vec)) < 1.3:
            break
    return vec

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes the
    provided ``evaluate_sequence`` function.
    The routine respects the given time budget (seconds) and returns the best
    sequence found (as a Python list of floats).  The implementation combines
    random restarts, simulated annealing and a short gradient‑based refinement.
    """
    start = time.monotonic()
    # a small safety margin so that the script can finish cleanly
    deadline = start + max(1.0, float(budget_s) - 0.5)

    random.seed(seed)
    np.random.seed(seed)

    # global best
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1️⃣  Known optimum for length 2 (analytical)
    # ------------------------------------------------------------------
    a2 = np.array([2.0 / 3.0, 1.0 / 3.0])
    score2 = _score_of(a2)
    best_seq = a2.tolist()
    best_score = score2
    print(f"initial length=2 analytical optimum score={best_score:.12f}")

    # ------------------------------------------------------------------
    # 2️⃣  Stochastic search over a range of lengths
    # ------------------------------------------------------------------
    lengths = [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20, 25, 30, 40]
    random.shuffle(lengths)

    # parameters that adapt to the remaining time budget
    for n in lengths:
        if time.monotonic() > deadline:
            break

        # how many independent restarts we will attempt for this length
        # (more restarts for small n, fewer for larger n)
        if n <= 8:
            restarts = 30
        elif n <= 20:
            restarts = 20
        else:
            restarts = 12

        for _ in range(restarts):
            if time.monotonic() > deadline:
                break

            # random start on the simplex (Dirichlet)
            a = np.random.rand(n)
            a /= a.sum()

            # a few deterministic starts (uniform, slightly perturbed)
            if _ == 0:
                a = np.full(n, 1.0 / n)

            # run a short annealing phase
            max_iter = 2500
            init_temp = 0.04
            cand_vec, cand_score = _anneal_one_restart(a, max_iter, init_temp)

            # keep globally best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_vec.tolist()
                print(f"new global best length={n} score={best_score:.12f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final refinement of the best candidate with gradient steps
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_best = np.asarray(best_seq, dtype=np.float64)
        a_refined = _gradient_refine(a_best, steps=800, init_lr=0.6)
        refined_score = _score_of(a_refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refined.tolist()
            print(f"refined best score={best_score:.12f}")

    # ------------------------------------------------------------------
    # 4️⃣  Verify with the authoritative evaluator (optional safety)
    # ------------------------------------------------------------------
    try:
        eval_score = evaluate_sequence(best_seq)   # type: ignore
        print(f"authoritative evaluator score={eval_score:.12f}")
    except Exception as e:
        print("warning: evaluation failed:", e)

    # guarantee a non‑zero sum (the evaluator rejects sums < 0.01)
    if best_seq is None or sum(best_seq) < 0.01:
        # fallback: safe constant sequence
        best_seq = [1.0] * 2
        print("fallback to trivial safe sequence")

    return [float(x) for x in best_seq]
