def propose_candidate(seed=42, budget_s=1600, **kwargs):
    """Search for a step height sequence minimizing evaluate_sequence.
    Uses simulated annealing with multiple mutation types: block scaling,
    pairwise mass transfer, symmetric averaging, and random jitter.
    """
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # initialize sequence
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        init = np.ones(256, dtype=np.float64)

    # ensure valid shape
    init = np.clip(init, 0.0, 1000.0)
    if init.ndim != 1 or init.size == 0:
        init = np.ones(256, dtype=np.float64)

    # optional random restart: begin with uniform with same length
    n = init.size
    # maybe randomly perturb initial if too high score
    current = init.copy()
    current_score = float(evaluate_sequence(current.tolist()))
    best = current.copy()
    best_score = current_score

    # Temperature schedule
    T0 = 0.2 * max(0.01, best_score)  # initial temperature roughly 20% of current score
    # We'll decay temperature linearly to near zero at deadline
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        # compute remaining fraction
        elapsed = time.monotonic() - start_time
        fraction = elapsed / (deadline - start_time)
        # temperature linearly decreasing
        T = T0 * max(0.0, 1.0 - fraction)

        # choose mutation type
        r = rng.random()
        if r < 0.4:
            candidate = block_perturb(current, rng)
        elif r < 0.65:
            candidate = pairwise_swap(current, rng)
        elif r < 0.85:
            candidate = symmetric_average(current, rng)
        else:
            candidate = random_noise(current, rng)

        # evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(score):
            # reject invalid
            continue
        delta = score - current_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < math.exp(-delta / T)
        if accept:
            current = candidate
            current_score = score
            if score < best_score:
                best = candidate.copy()
                best_score = score
                print(f"New best score {best_score:.6f} at time {elapsed:.1f}s")
        # occasional restart to best if stuck (like every 500 iterations)

    # Ensure sum not too small (though should be fine)

    return [float(x) for x in final]

def block_perturb(seq, rng):
    candidate = seq.copy()
    n = candidate.size
    # choose block width
    min_width = max(1, n // 250)  # smaller width for fine-tuning
    max_width = max(2, n // 8)
    width = int(rng.integers(min_width, max_width+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # apply scaling factor drawn from lognormal
    factor = np.exp(rng.normal(0.0, 0.07))  # smaller sigma for fine adjustments
    candidate[start:stop] *= factor
    # maybe reflect with some probability
    if rng.random() < 0.3:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip after operation to enforce bounds
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

pairwise_swap: select two distinct indices; transfer mass delta.

```
def pairwise_swap(seq, rng, max_fraction=0.1):
    candidate = seq.copy()
    n = candidate.size
    i, j = rng.choice(n, size=2, replace=False)
    # Transfer some mass from larger to smaller
    if candidate[i] > candidate[j]:
        src, dst = i, j
    else:
        src, dst = j, i
    # compute transfer amount as fraction of source
    amount = candidate[src] * rng.uniform(0.0, max_fraction)
    candidate[src] -= amount
    candidate[dst] += amount
    # clip
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate
```

def symmetric_average(seq, rng, prob=0.5):

"""
propose_candidate

Search for a non‑negative step‑function (a list of heights) that minimizes the
evaluation function `evaluate_sequence`.  The algorithm is a simple simulated
annealing / hill‑climbing hybrid:

*   Start from the best known construction (`height_sequence_1`) or from a
    uniform vector if that is unavailable.
*   Repeatedly propose a mutated candidate using one of several inexpensive
    move types:
    - block scaling (the “perturbation” used in the reference code)
    - pairwise mass transfer between two positions
    - averaging with the reversed sequence (enforces symmetry)
    - small random jitter on a subset of entries
*   Accept the candidate if it improves the score or, with a probability
    depending on a linearly decreasing temperature, even if it is worse
    (standard simulated‑annealing acceptance).  This allows the search to
    escape shallow local minima.
*   Keep the best sequence ever seen and report intermediate improvements.
*   Stop after the allotted time budget and return the best sequence found.

The implementation uses only ``numpy``, ``time`` and ``math`` and respects the
global limits on the number of CPU cores and on the total runtime.
"""
import time
import math
import numpy as np

# Helper mutation functions (all operate in‑place on a copy of the sequence)
# ----------------------------------------------------------------------

def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor and optionally
    symmetrize the whole sequence."""
    cand = seq.copy()
    n = cand.size
    # block width from a few elements up to a moderate fraction of n
    min_w = max(1, n // 300)
    max_w = max(2, n // 8)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.08)))   # modest perturbation
    cand[start:start + width] *= factor
    # occasional mirroring (helps symmetry)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def pairwise_transfer(seq: np.ndarray, rng: np.random.Generator,
                     max_fraction: float = 0.07) -> np.ndarray:
    """Move a fraction of mass from the larger of two random indices to the
    smaller one."""
    cand = seq.copy()
    n = cand.size
    i, j = rng.choice(n, size=2, replace=False)
    if cand[i] > cand[j]:
        src, dst = i, j
    else:
        src, dst = j, i
    # transfer up to `max_fraction` of the source amount
    amount = cand[src] * rng.uniform(0.0, max_fraction)
    cand[src] -= amount
    cand[dst] += amount
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def symmetric_average(seq: np.ndarray, rng: np.random.Generator,
                      prob: float = 0.45) -> np.ndarray:
    """Replace the sequence by the average of itself and its reversal."""
    cand = seq.copy()
    if rng.random() < prob:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def jitter(seq: np.ndarray, rng: np.random.Generator,
           sigma: float = 0.025) -> np.ndarray:
    """Add small Gaussian noise to a random 1‑2 % of the entries."""
    cand = seq.copy()
    n = cand.size
    num = max(1, int(n * 0.015))                # perturb ~1.5 % of indices
    idx = rng.choice(n, size=num, replace=False)
    noise = rng.normal(0.0, sigma, size=num)
    cand[idx] += noise
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    """

    # leave a small safety margin so the platform can finish cleanly

    init = np.clip(init, 0.0, 1000.0)
    if init.ndim != 1 or init.size == 0:

    # current state (the “annealer” walks)
    current = init.copy()
    current_score = float(evaluate_sequence(current.tolist()))

    # best ever seen
    best = current.copy()
    best_score = current_score

    print(f"[INIT] length={current.size}, score={best_score:.6f}")

    # temperature schedule – start at ~20 % of the current score and decay
    T0 = 0.2 * max(0.01, best_score)
    start_time = time.monotonic()
    iteration = 0

    # ------------------------------------------------------------------
    # main loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        # linearly decreasing temperature
        elapsed = time.monotonic() - start_time
