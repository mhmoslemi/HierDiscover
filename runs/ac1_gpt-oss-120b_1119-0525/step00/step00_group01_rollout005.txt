def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step function whose self‑convolution is as flat as possible.
    The algorithm performs projected sub‑gradient descent on the maximum convolution entry.
    It starts from a random vector (or the previous construction if available) and iteratively
    reduces the dominant convolution term.  The final sequence is returned.
    """
    import time, math, numpy as np, random
    start = time.monotonic()
    deadline = start + max(1.0, float(budget_s) - 10.0)  # keep margin

    # Determine target length
    try:
        target_len = int(len(globals().get("height_sequence_1", [])))
        if not (1 <= target_len <= 8000):
            target_len = 6962
    except Exception:
        target_len = 6962

    # We may use a slightly smaller length for speed.
    n = min(target_len, 5000)   # cap
    print(f"Optimization length = {n}")

    # Use previous construction if present
    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            arr = np.asarray(parent, dtype=np.float64)
            if arr.ndim == 1 and 1 <= arr.size <= 8000:
                # resize or truncate/extend to n
                if arr.size != n:
                    # simple resize by interpolation or truncation
                    arr = np.interp(np.linspace(0, arr.size-1, n), np.arange(arr.size), arr)
                a = np.maximum(arr, 0.0)
                if a.sum() < 0.01:
                    a = np.ones(n) / n
                else:
                    a = a / a.sum()
                print("Initialized from previous construction")
        except Exception:
            a = None

    if 'a' not in locals() or a is None:
        rng = np.random.RandomState(seed)
        a = rng.rand(n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        print("Initialized random start")

    best_seq = a.copy()
    best_score = float('inf')
    # compute initial score
    conv = np.convolve(a, a)
    max_b = conv.max()
    best_score = 2 * n * max_b / (a.sum() ** 2)
    print(f"initial score = {best_score:.6f}")

    # Main loop
    iteration = 0
    no_improve = 0
    while time.monotonic() < deadline:
        iteration += 1
        # compute convolution and max index
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        max_b = float(conv[max_idx])
        # compute current objective
        obj = 2.0 * n * max_b / (a.sum() ** 2)

        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iteration}: improved score = {best_score:.8f} (k={max_idx})")
        else:
            no_improve += 1

        # Build gradient of the dominant term
        # shift left (i+k) contribution
        left = np.concatenate((a[max_idx:], np.zeros(max_idx, dtype=a.dtype)))
        # shift right (i-k) contribution
        right = np.concatenate((np.zeros(max_idx, dtype=a.dtype), a[:n - max_idx]))
        grad = left + right

        # simple projected sub‑gradient step with back‑tracking line search
        lr = 0.2  # initial step size
        for _ in range(10):
            a_new = a - lr * grad
            # enforce non‑negativity
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-12:
                # cannot renormalise – shrink step
                lr *= 0.5
                continue
            a_new = a_new / s
            # evaluate new max_b
            conv_new = np.convolve(a_new, a_new)
            max_b_new = conv_new.max()
            if max_b_new < max_b - 1e-14:
                a = a_new
                break
            else:
                lr *= 0.5
        else:
            # if line search failed, add a small random perturbation
            rng = np.random.RandomState()
            a = a + 1e-4 * rng.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # occasional reset if no progress for many steps
        if no_improve >= 500 and iteration % 1000 == 0:
            rng = np.random.RandomState(seed + iteration)
            a = rng.rand(n)
            a = a / a.sum()
            no_improve = 0
            print(f"Iter {iteration}: random restart")

        # break early if already close to theoretical lower bound
        if best_score <= 1.01 * (2.0 * n / (2 * n - 1)):
            print("Reached near‑optimal bound, stop.")
            break

    # Return best sequence found (convert to list of floats)
    final_seq = best_seq.tolist()
    # verify with evaluate_sequence to avoid unexpected issues
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final evaluated score = {final_score:.10f}")
    except Exception:
        pass

    return final_seq
