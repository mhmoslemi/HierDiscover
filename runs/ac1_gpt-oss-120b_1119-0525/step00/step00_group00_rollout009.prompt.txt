def truncated_sqrt_sequence(n):
    """
    First n coefficients of (1 - x)^(-1/2), normalized.
    Their self-convolution has its first n coefficients exactly 1 (unscaled).
    """
    n = max(1, int(n))
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            j = k - i
            s += b[i] * b[j]
        bk = (1.0 - s) / 2.0
        if bk < 0.0:
            bk = 0.0
        b[k] = bk
    s = float(np.sum(b))
    if s <= 0 or not np.isfinite(s):
        return None
    b = np.maximum(0.0, b) / s
    return b

            best_p = p
            best_param = param.copy()
