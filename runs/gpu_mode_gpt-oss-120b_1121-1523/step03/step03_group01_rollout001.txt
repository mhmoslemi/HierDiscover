def custom_kernel(data):
   input_tensor, mask, weights, config = data
   B, N, _, dim = input_tensor.shape # dim == config["dim"], verify.

   # LN on input:
   eps = 1e-5
   # using torch.nn.functional.layer_norm
   # careful: layer_norm expects normalized_shape param, which is dim
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

   # Compute projections and gates
   left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"]) # shape B,N,N, hidden_dim
   right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
   left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

   # Apply mask if present
   if not config.get("nomask", False):
       # mask shape (B,N,N) -> unsqueeze(-1)
       mask_ = mask.unsqueeze(-1).to(input_tensor.dtype)
       left_proj = left_proj * mask_
       right_proj = right_proj * mask_
   # else mask omitted (skip)

   # Apply gates:
   left = left_proj * left_gate
   right = right_proj * right_gate

   # Compute core TriMul via Triton
   # allocate out tensor
   hidden_dim = config["hidden_dim"]
   out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

   # Determine strides
   left_strides = left.stride()
   right_strides = right.stride()
   out_strides = out.stride()

   block_m = 32
   block_n = 32
   block_k = 32
   block_d = 8  # hidden per block

   # Compute grid
   grid = (triton.cdiv(N, block_m), triton.cdiv(N, block_n), B * triton.cdiv(hidden_dim, block_d))

   trimul_outgoing_kernel[grid](
       left, right, out,
       B, N, hidden_dim,
       left_strides[0], left_strides[1], left_strides[2], left_strides[3],
       right_strides[0], right_strides[1], right_strides[2], right_strides[3],
       out_strides[0], out_strides[1], out_strides[2], out_strides[3],
       BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k, BLOCK_D=block_d,
       SEQ_LEN=N
   )
   # Note: Use .contiguous() maybe.

   # Apply to_out_norm
   out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

   # Apply out_gate gating
   out = out * out_gate

   # Final linear
   out = torch.nn.functional.linear(out, weights["to_out.weight"])

   return out

"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The operation works on a 4‑D pairwise tensor X ∈ ℝ^{B×N×N×C}.

1.  Layer‑norm on the last dimension (C).
2.  Two linear projections → left/right ∈ ℝ^{B×N×N×H}
    (H = hidden_dim) and three gating vectors (sigmoid) of the same shape.
3.  Apply mask (if present) and per‑feature gates.
4.  Core “TriMul”:
       out_{b,i,j,d} = Σ_{k=0}^{N‑1} left_{b,i,k,d} * right_{b,j,k,d}
    i.e. for each hidden channel d we perform a matrix multiplication
    left_d @ right_dᵀ.  This 3‑D N×N×H contraction is realised in a
    custom Triton kernel that tiles the i‑, j‑, k‑ and d‑axes.
5.  Layer‑norm on the hidden dimension, multiplicative out‑gate,
    and a final linear projection back to C.

Only the heavy N³ contraction is accelerated with Triton; all other
operations are performed with standard PyTorch (they are memory‑bound
and inexpensive compared with the N³ work).  The kernel works for any
batch size B∈{1,2}, sequence length N up to 1024 and hidden dimension
H up to 384, handling arbitrary padding via masks.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,           # const float*  (B,N,N,H)   – left projection
    right_ptr,          # const float*  (B,N,N,H)   – right projection
    out_ptr,            # float*        (B,N,N,H)   – result
    B, N, H,            # int scalar sizes
    # strides for left
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,
    # strides for right
    stride_b_r, stride_i_r, stride_k_r, stride_d_r,
    # strides for out
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
    SEQ_LEN: tl.constexpr,
):
    ############################################################
    # block identifiers
    pid_i = tl.program_id(0)          # tile along i   (first N axis)
    pid_j = tl.program_id(1)          # tile along j   (second N axis)
    pid_z = tl.program_id(2)          # combined (batch, d‑chunk)

    # ----- decode batch index and hidden‑channel chunk -------------
    num_d_chunks = tl.cdiv(H, BLOCK_D)
    batch_idx   = pid_z // num_d_chunks
    d_chunk_idx = pid_z % num_d_chunks
    d_offset    = d_chunk_idx * BLOCK_D

    # ----- starting offsets for i, j and d -------------------------
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # ----- build 1‑D index vectors --------------------------------
    i_range = i_offset + tl.arange(0, BLOCK_M)
    j_range = j_offset + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)          # will be added inside loop
    d_range = d_offset + tl.arange(0, BLOCK_D)

    # ----- out‑of‑bounds masks ------------------------------------
    i_mask = i_range < N
    j_mask = j_range < N
    d_mask = d_range < H

    # accumulator: (BLOCK_M, BLOCK_N, BLOCK_D) in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over the contracting dimension k
    # -----------------------------------------------------------------
    for start_k in range(0, SEQ_LEN, BLOCK_K):
        k_offset = start_k + k_range                     # (BLOCK_K,)
        k_mask   = k_offset < SEQ_LEN

        # combined masks for the three dimensions
        mask_left  = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
        mask_right = j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]

        # pointer arithmetic (all strides are in elements, not bytes)
        left_ptrs = (left_ptr
                     + batch_idx * stride_b_l
                     + (i_range[:, None, None] * stride_i_l)
                     + (k_offset[None, :, None] * stride_k_l)
                     + (d_range[None, None, :] * stride_d_l))

        right_ptrs = (right_ptr
                      + batch_idx * stride_b_r
                      + (j_range[:, None, None] * stride_i_r)   # stride_i_r = stride for the “j” axis
                      + (k_offset[None, :, None] * stride_k_r)
                      + (d_range[None, None, :] * stride_d_r))

        left_tile  = tl.load(left_ptrs,  mask=mask_left,  other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # ----- per‑channel dot product (BLOCK_D independent dot calls) -----
        for d in range(BLOCK_D):
            # shapes: (BLOCK_M, BLOCK_K) and (BLOCK_N, BLOCK_K)
            left_slice  = left_tile[:, :, d]
            right_slice = right_tile[:, :, d]
            acc[:, :, d] += tl.dot(left_slice, right_slice)   # sum over k

    # -----------------------------------------------------------------
    # write back
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr
                + batch_idx * stride_b_o
                + (i_range[:, None, None] * stride_i_o)
                + (j_range[None, :, None] * stride_j_o)
                + (d_range[None, None, :] * stride_d_o))

    store_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the forward pass of the “outgoing” TriMul module.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N]  (ignored if config["nomask"] is True)
            - weights      : dict of all linear / layer‑norm parameters
            - config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2) linear projections + gating
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", False):
        # mask is 0/1 float; reshape for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) core TriMul (N³) – Triton
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty((B, N, N, H), dtype=torch.float32, device=input_tensor.device)

    # strides (in elements) for the three tensors
    s_b_l, s_i_l, s_k_l, s_d_l = left.stride()
    s_b_r, s_i_r, s_k_r, s_d_r = right.stride()
    s_b_o, s_i_o, s_j_o, s_d_o = out.stride()

    # tile sizes – tuned for H100, keep them powers of two
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden‑dim slice per block (H up to 384 ⇒ ≤48 blocks)

    # launch grid
    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * triton.cdiv(H, BLOCK_D),      # (batch, d‑chunk)
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_b_l, s_i_l, s_k_l, s_d_l,
        s_b_r, s_i_r, s_k_r, s_d_r,
        s_b_o, s_i_o, s_j_o, s_d_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        SEQ_LEN=N,
    )

    # ------------------------------------------------------------------
    # 4) post‑processing: hidden‑dim layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])   # (B,N,N,C)

    return out
