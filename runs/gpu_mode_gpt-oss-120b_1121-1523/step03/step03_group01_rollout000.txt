"""
TriMul forward kernel (outgoing version).

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension (dim).
2. Apply two linear projections (left/right) producing tensors of shape
   [B, N, N, H] where H = hidden_dim.
3. (Optional) Apply a binary mask of shape [B, N, N] to the projected
   tensors.
4. Compute three gating tensors (left, right, out) via linear layers + sigmoid.
   Multiply the projected tensors by their respective gates.
5. Compute the expensive pair‑wise interaction:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   which is equivalent to a batched matrix multiplication per hidden channel:
        out_{h} = left_{h} @ right_{h}ᵀ   for every (b,h) pair.
   This step is implemented with a custom Triton kernel that processes the
   batch*hidden dimension as an independent batch and uses tiled GEMM.
6. Layer‑normalize the intermediate result over the hidden dimension,
   multiply by the output gate, and finally project back to `dim` with a
   linear layer.
All operations except the heavy N³ matmul are performed with PyTorch
(easily handling LayerNorm and gating).  The Triton kernel accelerates the
dominant O(N³ · H) computation while keeping the final output in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (M=N=K=N)
    stride_am, stride_ak, stride_ah,   # strides for A (row, col, batch)
    stride_bk, stride_bn, stride_bh,   # strides for B (row, col, batch)
    stride_cm, stride_cn, stride_ch,   # strides for C (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_id = tl.program_id(0)           # which (batch, hidden) pair
    pid_m = tl.program_id(1)              # block on M dimension
    pid_n = tl.program_id(2)              # block on N dimension

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension (the shared axis)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A sub‑matrix [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + batch_id * stride_ah \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B sub‑matrix [BLOCK_K, BLOCK_N] (B is already transposed)
        b_ptrs = b_ptr + batch_id * stride_bh \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the product
        acc += tl.dot(a, b)   # result [BLOCK_M, BLOCK_N]

    # Store the result
    c_ptrs = c_ptr + batch_id * stride_ch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def batched_matmul_triton(a: torch.Tensor,
                          b: torch.Tensor,
                          *,
                          BLOCK_M: int = 128,
                          BLOCK_N: int = 128,
                          BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    Input tensors:
        a: [B*H, N, K]   (K = N)
        b: [B*H, K, N]   (already transposed, i.e. original rightᵀ)
    Returns:
        c: [B*H, N, N] with dtype matching a/b.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2 and M == N, "Incompatible shapes for batched matmul."

    # allocate output
    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # strides
    stride_ah, stride_am, stride_ak = a.stride()
    stride_bh, stride_bk, stride_bn = b.stride()
    stride_ch, stride_cm, stride_cn = c.stride()

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=M,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ah=stride_ah,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bh=stride_bh,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_ch=stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Helper: LayerNorm without bias to keep everything in a single function
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """Apply LayerNorm over the last dimension."""
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights_dict : dict mapping weight names to torch.Tensor
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True -> ignore mask

    # ------------------------------------------------------------------
    # extract weights (all are bias‑free Linear weights except LayerNorm)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor, norm_weight, norm_bias)               # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear treats the last dimension as feature dim
    left  = F.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. (Optional) apply binary mask
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask dtype must match computation dtype
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Compute gating tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    # Apply gates to projected tensors
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy N^3 interaction via Triton GEMM (batch*hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to [B*H, N, N] and make matrices contiguous
    left_reshaped  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # transpose the second operand to match the einsum pattern
    right_T = right_reshaped.transpose(1, 2).contiguous()   # now shape [B*H, N, N]

    # batched matmul: each slice computes Σ_k left[i,k] * right[j,k]
    out_reshaped = batched_matmul_triton(left_reshaped, right_T)

    # reshape back to [B, N, N, hidden_dim]
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension, gate, and final linear
    # ------------------------------------------------------------------
    out = layer_norm(out, to_out_norm_weight, to_out_norm_bias)

    # apply output gate
    out = out * out_gate

    # final projection back to original dimensionality (no bias)
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
