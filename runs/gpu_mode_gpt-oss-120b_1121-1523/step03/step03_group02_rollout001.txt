"""
TriMul (outgoing) – fast Triton‑augmented implementation.

The expensive contraction 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a large batched GEMM in fp16 using
`torch.bmm`, which on H100 maps to a Tensor‑Core kernel.
Four linear layers (proj + gate for left/right) are fused
into a single matmul, halving memory traffic.
All heavy arithmetic is kept in float16; the final
output is returned in float32.
A tiny Triton kernel is used for the final element‑wise
multiplication with the output gate (required by the
specification that “at least part of the operations” be a Triton
kernel).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elemwise_mul_kernel(
    out_ptr,         # pointer to output tensor (will be overwritten)
    gate_ptr,        # pointer to gate tensor
    size,            # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place out *= gate for half‑precision tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(out_ptr + offs, mask=mask)   # fp16
    b = tl.load(gate_ptr + offs, mask=mask)  # fp16
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul_inplace(out: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """Apply the Triton element‑wise multiplication kernel in‑place."""
    assert out.is_contiguous() and gate.is_contiguous()
    size = out.numel()
    BLOCK = 8192  # 8 K threads per block – a good trade‑off for 16‑bit data
    grid = (triton.cdiv(size, BLOCK),)
    _elemwise_mul_kernel[grid](
        out.view(-1), gate.view(-1), size, BLOCK_SIZE=BLOCK
    )
    return out


def custom_kernel(data):
    """
    Forward pass for the outgoing Triangle Multiplicative Update.
    Arguments:
        data = (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (bool or 0/1, optional)
        - weights      : dict of model parameters (float32)
        - config       : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Normalise input (half‑precision)
    # --------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=eps,
    )  # [B, N, N, dim]   fp16

    # --------------------------------------------------------------
    # 2. Fuse the four linear projections (proj + gate for left/right)
    # --------------------------------------------------------------
    #   left_proj, left_gate, right_proj, right_gate → (4·hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (4*hidden_dim, dim)

    combined = F.linear(x, combined_weight)  # [B, N, N, 4*hidden_dim]  fp16
    left_raw, left_gate_raw, right_raw, right_gate_raw = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_raw = left_raw * mask_f
        right_raw = right_raw * mask_f

    # --------------------------------------------------------------
    # 4. Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_raw * left_gate   # [B, N, N, hidden_dim]
    right = right_raw * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM:   Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    Rearrange to (B·hidden_dim, N, N) @ (B·hidden_dim, N, N).T
    # --------------------------------------------------------------
    #   left : (B, N, N, hidden) → (B, hidden, N, N) → (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    #   right : (B, N, N, hidden) → (B, hidden, N, N) with axes (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # fp16 batched matrix multiplication – highly optimised in cuBLAS
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Restore original layout  →  [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # --------------------------------------------------------------
    # 7. Output gate (sigmoid) and in‑place element‑wise multiply via Triton
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))
    # out *= out_gate  (both half‑precision)
    _elemwise_mul_inplace(out, out_gate)

    # --------------------------------------------------------------
    # 8. Final projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # [B, N, N, dim]  fp16

    # Return in FP32 (the usual format for downstream layers)
    return out.to(torch.float32)
