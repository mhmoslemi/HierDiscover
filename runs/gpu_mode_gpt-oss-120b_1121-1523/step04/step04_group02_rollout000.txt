"""
TriMul (outgoing) – fast Triton implementation
================================================

The heavy part of the outgoing Triangle Multiplicative Update is the
contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix‑multiply:
    left  : (B·D, N, N)   – rows = i, cols = k
    right : (B·D, N, N)   – rows = k, cols = j
    out   : (B·D, N, N)   – rows = i, cols = j

We flatten the (batch, head) dimensions (B·D) and launch a Triton GEMM
kernel that works on FP16 inputs while accumulating in FP32.  The rest
(LayerNorm, linear projections, sigmoid gates) is performed with
high‑level PyTorch ops – they are cheap compared to the O(N³) contraction.

Key ideas
---------
* **FP16 loads** – the two input tensors are cast to half‑precision before
  the kernel; this cuts memory traffic by 2× on H100.
* **FP32 accumulation** – numerical stability is retained by accumulating in
  float32.
* **Large tiles** – BLOCK_M/N = 128 and BLOCK_K = 64 give good occupancy for
  N up to 1024 on an H100.
* **Batch flattening** – a 3‑D launch grid (M‑blocks, N‑blocks, batch) lets
  Triton handle the B·D independent GEMMs without any extra synchronisation.
* **Optional mask** – if a mask is supplied we apply it before the kernel
  (the extra element‑wise multiply costs negligible time).

The kernel returns a tensor of shape `[B, N, N, hidden_dim]` which is then
passed through the output LayerNorm, out‑gate and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) with FP16 inputs, FP32 accumulator
# -------------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,                 # *const half*, shape (batch, M, K)
    b_ptr,                 # *const half*, shape (batch, K, N)
    c_ptr,                 # *float*,       shape (batch, M, N)
    BATCH, M, N, K,        # problem dimensions
    stride_ah, stride_am, stride_ak,   # strides of A (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # strides of B (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with FP32 accumulator.

    Computes for each batch p:
        C[p, i, j] = Σ_k A[p, i, k] * B[p, k, j]
    """
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col
    pid_b = tl.program_id(2)      # batch index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the current batch
    a_batch = a_ptr + pid_b * stride_ah
    b_batch = b_ptr + pid_b * stride_bh

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k                     # absolute K indices

        # Boundary masks
        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        # Load tiles of A and B (half‑precision)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
            dtype=tl.float16,
        )
        b = tl.load(
            b_batch
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
            dtype=tl.float16,
        )

        # Cast to FP32 before dot to keep accumulation in FP32
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)
        acc += tl.dot(a_f32, b_f32)   # (BLOCK_M, BLOCK_N) FP32

    # Write the result
    c = c_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c, acc, mask=mask_c)


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Performs the ``... i k d, ... j k d -> ... i j d`` contraction using the
    Triton GEMM kernel above.  Input tensors are expected to be
    ``[B, N, N, D]`` (D = hidden_dim) and the result has the same shape.
    """
    B, N, _, D = left.shape

    # --------------------------------------------------------------
    # 1) Rearrange to (batch, M, K) and (batch, K, N) where
    #    batch = B * D, M = N, K = N, N = N
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, D, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, D, k, j)

    batch = B * D
    left_flat = left_perm.view(batch, N, N)            # (batch, i, k)
    right_flat = right_perm.view(batch, N, N)          # (batch, k, j)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the kernel (memory bandwidth reduction)
    # --------------------------------------------------------------
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output in FP32 (required by subsequent LayerNorm)
    out_fp32 = torch.empty((batch, N, N), dtype=torch.float32, device=left.device)

    # --------------------------------------------------------------
    # 3) Kernel launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks in M dimension
        triton.cdiv(N, BLOCK_N),   # blocks in N dimension
        batch,                     # batch dimension (B * D)
    )

    _batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        batch,          # BATCH
        N, N, N,        # M, N, K   (all equal to sequence length)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 4) Reshape back to original layout [B, N, N, D]
    # --------------------------------------------------------------
    out = out_fp32.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    dim = config["dim"]                # input feature dimension (c)
    hidden_dim = config["hidden_dim"]  # projection dimension (c_z)
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # -----------------------------------------------------------------
    if not nomask:
        # mask is broadcasted over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) projected from the same normalized input
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Expensive contraction using the high‑performance Triton kernel
    # -----------------------------------------------------------------
    out = _batched_matmul_fp16(left, right)   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 6) Output LayerNorm + out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7) Final linear projection back to ``dim``
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
