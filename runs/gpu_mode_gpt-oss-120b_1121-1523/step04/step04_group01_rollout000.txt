"""
TriMul (outgoing) – fast Triton implementation
================================================

This module implements the forward pass of the “TriMul” block used in AlphaFold‑3
(and derived models).  The heavy O(N³·C) part

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is carried out by a custom Triton kernel that performs a batched matrix
multiplication for every hidden channel `d`.  All other operations
(LayerNorm, Linear projections and gates) are expressed with regular
torch.nn.functional calls – this keeps the code simple while still giving a
large speed‑up for the cubic kernel.

The kernel works on tensors of shape [B, N, N, H] (H = hidden_dim) and
uses a 3‑D grid:
    axis‑0 : batch * hidden dim
    axis‑1 : i‑tile (M)
    axis‑2 : j‑tile (N)

Each program computes a (BLOCK_M × BLOCK_N) tile of the output matrix for a
single hidden channel and accumulates over the k‑dimension in blocks of
size BLOCK_K.  The implementation uses float32 accumulation (the inputs are
float32 as well) and stores the result back to a tensor of the same shape.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel for the cubic contraction
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # const float*   [B, N, N, H]
    right_ptr,                    # const float*   [B, N, N, H]
    out_ptr,                      # float*         [B, N, N, H]
    B, N, H,                      # int scalars
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,        # tile size for i dimension
    BLOCK_N: tl.constexpr,        # tile size for j dimension
    BLOCK_K: tl.constexpr):       # tile size for k (reduction) dimension

    pid_bh = tl.program_id(0)               # combined batch & hidden index
    pid_m  = tl.program_id(1)               # i‑tile index
    pid_n  = tl.program_id(2)               # j‑tile index

    # -----------------------------------------------------------------
    # Decode batch and hidden‑channel from pid_bh
    # -----------------------------------------------------------------
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # -----------------------------------------------------------------
    # Offsets for the current tile
    # -----------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Base pointers for the chosen batch & hidden channel
    # -----------------------------------------------------------------
    left_base  = left_ptr  + batch_idx * stride_lb + hidden_idx * stride_lh
    right_base = right_ptr + batch_idx * stride_rb + hidden_idx * stride_rh

    # -----------------------------------------------------------------
    # Accumulator for the output tile
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over k in BLOCK_K steps
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # Load left[i, k]  -> (BLOCK_M, BLOCK_K)
        left_ptrs = (left_base
                     + offs_i[:, None] * stride_li
                     + k_base[None, :] * stride_lk)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # Load right[j, k] -> (BLOCK_N, BLOCK_K)
        right_ptrs = (right_base
                      + offs_j[:, None] * stride_ri
                      + k_base[None, :] * stride_rk)
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)

        # Convert to float32 (input may be fp16) and accumulate
        acc += tl.dot(left_tile.to(tl.float32), right_tile.to(tl.float32))

    # -----------------------------------------------------------------
    # Write the tile back to the output tensor
    # -----------------------------------------------------------------
    out_base = out_ptr + batch_idx * stride_ob + hidden_idx * stride_oh
    out_ptrs = out_base \
               + offs_i[:, None] * stride_oi \
               + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_tiled(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches ``tri_mul_kernel``.
    left / right : [B, N, N, H]  (float32)
    returns out   : [B, N, N, H]  (float32)
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape

    # Output buffer
    out = torch.empty_like(left)

    # Block sizes – tuned for H100 (shared‑memory limited ~64 KiB)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*hidden, ceil(N/BM), ceil(N/BN))
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Strides (in elements, not bytes)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4)          # 4 warps gives good occupancy on H100
    return out

    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input layer‑norm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_w,
                     bias=norm_b)

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden_dim (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3) Optional masking (mask shape [B,N,N])
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is float and broadcastable
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) – also linear without bias
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Cubic contraction – the O(N³) kernel
    # -----------------------------------------------------------------
    out = tri_mul_tiled(left.contiguous(), right.contiguous())
    # out shape: [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 6) Output layer‑norm (over hidden_dim) + out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate

    # -----------------------------------------------------------------
    # 7) Final linear projection back to the model dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B,N,N,dim]

    return out
```
