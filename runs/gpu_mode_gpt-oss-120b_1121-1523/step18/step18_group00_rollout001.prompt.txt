import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel – highly‑optimized “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32 for numerical stability).
2.  Cast to FP16 for the bulk of the work.
3.  Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
    into a single matmul:   (B·N·N, 4·H) = (B·N·N, C) @ (4·H, C).T
4.  Split the result, optionally apply the pair‑wise mask with a tiny Triton
    kernel (broadcast over the hidden dimension).  The mask is only multiplied
    into the *projected* tensors, exactly as in the reference implementation.
5.  Apply sigmoid gating to the left/right projected tensors.
6.  Core O(N³·H) update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    realised with a single `torch.einsum` which avoids the costly
    `permute‑contiguous‑view‑bmm` pattern and therefore removes large memory
    copies.
7.  Layer‑norm the hidden tensor, compute the output‑gate, multiply and project
    back to the original channel dimension.
8.  Return a float‑32 tensor.

The kernel in step 4 is the only Triton part (required by the task); all
compute‑heavy work stays in PyTorch but is now performed on half‑precision
tensors and with far fewer memory movements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: broadcast a mask over the hidden dimension and
#                apply it in‑place to two tensors (left & right)
# -------------------------------------------------------------
@triton.jit
def _mask_mul_two_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """left[b,i,j,h] *= mask[b,i,j];  right[b,i,j,h] *= mask[b,i,j]"""
    b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load a (BLOCK_I, BLOCK_J) tile of the mask and broadcast to H
    # -----------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i[:, None] * stride_mask_i
        + j[None, :] * stride_mask_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                 # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # -----------------------------------------------------------------
    # Loop over the hidden dimension
    # -----------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # left
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # right
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # apply mask
        left = left * mask_tile
        right = right * mask_tile
        # store back
        tl.store(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_two_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """In‑place broadcast mask multiplication for left and right tensors."""
    B, N, _, H = left.shape
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_mul_two_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # tensors are mutated in‑place


# -------------------------------------------------------------
# Outgoing TriMul – forward‑only implementation
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor torch.Tensor [B, N, N, C] (float32)
            - mask_tensor   torch.Tensor [B, N, N]   (bool/float, may be omitted)
            - weights dict of model parameters (float32)
            - config dict with keys:
                * dim          (int)   – input channel C
                * hidden_dim   (int)   – hidden dimension H
                * nomask       (bool)  – True ⇒ no mask
                * layernorm_eps (float, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # C
    hidden = config["hidden_dim"]       # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # Extract weights (all are float32 in the checkpoint)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]       # (H, C)
    right_proj_weight = weights["right_proj.weight"]     # (H, C)

    left_gate_weight = weights["left_gate.weight"]       # (H, C)
    right_gate_weight = weights["right_gate.weight"]     # (H, C)
    out_gate_weight = weights["out_gate.weight"]         # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]   # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]       # (H,)
    to_out_weight = weights["to_out.weight"]             # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Half‑precision for the heavy path
    # -----------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, C]  fp16
    B, N, _, C = x_h.shape

    # -----------------------------------------------------------------
    # 3️⃣ Fused projection + gating (single matmul → 4·H outputs)
    # -----------------------------------------------------------------
    # Build a fused weight matrix of shape (4*H, C)
    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    # Flatten the input to (B·N·N, C) and apply the fused linear
    x_flat = x_h.view(-1, C)                     # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)  # (B·N·N, 4*H)

    # Split and reshape back to (B, N, N, H)
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – in‑place multiplication via a tiny Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # mask is float‑like (bool or 0/1) – cast to half for the kernel
        mask_h = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_two_triton(left_proj, right_proj, mask_h)

    # -----------------------------------------------------------------
    # 5️⃣ Sigmoid gates
    # -----------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update – a single einsum avoids costly permutes/copies
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Output LayerNorm + out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final projection back to the original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B, N, N, C) fp16

    # -----------------------------------------------------------------
    # Return float32 to stay compatible with the reference module
    # -----------------------------------------------------------------
    return out.float()
