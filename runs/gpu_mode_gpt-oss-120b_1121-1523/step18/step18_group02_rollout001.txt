def mask_mul_kernel(x_ptr, mask_ptr, B, N, H, stride_x_b, stride_x_i, stride_x_j, stride_x_h, stride_m_b, stride_m_i, stride_m_j, BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    b = tl.program_id(0)
    i = tl.program_id(1) * BLOCK_I + tl.arange(0, BLOCK_I)
    j = tl.program_id(2) * BLOCK_J + tl.arange(0, BLOCK_J)
    i_ok = i < N
    j_ok = j < N

    # Load mask tile (B,N,N)
    mask_val = tl.load(mask_ptr + b*stride_m_b + i[:,None]*stride_m_i + j[None,:]*stride_m_j,
                       mask=i_ok[:,None] & j_ok[None,:], other=0.0)  # shape (BLOCK_I, BLOCK_J)
    # Broadcast mask over h dimension inside loop.

    # Iterate over H in BLOCK_H steps
    h = 0
    while h < H:
        hi = h + tl.arange(0, BLOCK_H)
        hi_ok = hi < H
        # Compute x pointer
        x = tl.load(x_ptr + b*stride_x_b + i[:,None,None]*stride_x_i + j[None,:,None]*stride_x_j + hi[None,None,:]*stride_x_h,

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)
    device = input_tensor.device
    # Step1: LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Step2 cast to fp16
    x_h = x.to(torch.float16)
    B,N,_,C = x_h.shape
    # Step3: projections and gates
    # Concatenate weights
    proj_weight = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"]], dim=0).to(dtype=torch.float16, device=device)
    gate_weight = torch.cat([weights["left_gate.weight"], weights["right_gate.weight"]], dim=0).to(dtype=torch.float16, device=device)
    # Linear
    x_flat = x_h.view(-1, C)   # (B*N*N, C)
    proj = F.linear(x_flat, proj_weight)   # (B*N*N, 2*hidden)
    gate = F.linear(x_flat, gate_weight)   # (B*N*N, 2*hidden)
    proj = proj.view(B,N,N,2*hidden)
    gate = gate.view(B,N,N,2*hidden)
    left_proj, right_proj = torch.split(proj, hidden, dim=-1)
    left_gate, right_gate = torch.split(gate, hidden, dim=-1)
    # step 4: mask
    if not nomask:
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)
    # step5: gates
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)
    # step6: core multiplication
    # reshape for batched matmul:
    left_flat = left.permute(0,3,1,2).contiguous().view(B*hidden, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*hidden, N, N)
    # compute out = left @ right.T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, hidden, N, N).permute(0,2,3,1).contiguous()
    # step7: output layernorm
    out = F.layer_norm(out,
                       (hidden,),
                       weight=weights["to_out_norm.weight"].to(dtype=out.dtype, device=device),
                       bias=weights["to_out_norm.bias"].to(dtype=out.dtype, device=device),
                       eps=eps)
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(dtype=torch.float16, device=device)))
    out = out * out_gate
    # step8: final linear
    out = F.linear(out, weights["to_out.weight"].to(dtype=torch.float16, device=device))
    # step9: cast to float32
    return out.float()
```

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3 style).

The forward pass consists of:
1️⃣ LayerNorm on the input (FP32 for numerical stability).
2️⃣ Two fused linear projections (left/right) and two fused linear gates, all in
   FP16.
3️⃣ Optional in‑place mask multiplication (broadcast over the hidden dimension)
   performed by a tiny Triton kernel – this satisfies the “use at least one
   Triton kernel” requirement.
4️⃣ Sigmoid gating.
5️⃣ Core O(N³·H) update: for each head (hidden dimension) we need
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix multiplication left @ rightᵀ for every
   hidden head.  We achieve this with a single torch.bmm call (uses cuBLAS
   Tensor‑cores on H100) after reshaping to (B·H, N, N).
6️⃣ Output LayerNorm + out‑gate sigmoid.
7️⃣ Final linear projection back to the original channel dimension.
All heavy arithmetic runs in FP16 with FP32 accumulation (cuBLAS does this
automatically).  The only Triton component is the mask‑multiply kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1️⃣ In‑place mask multiplication kernel (broadcast over hidden dim H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    x_ptr, mask_ptr,
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Perform: x[b,i,j,h] *= mask[b,i,j]   (mask broadcast over h)
    """
    b = tl.program_id(0)                     # batch
    pid_i = tl.program_id(1)                 # tile i
    pid_j = tl.program_id(2)                 # tile j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load a mask tile (BLOCK_I x BLOCK_J) – mask is FP16 after casting
    # -----------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                                 # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]  # (BLOCK_I, BLOCK_J, 1) – broadcast over H

    # -----------------------------------------------------------------
    # Loop over the hidden dimension H in chunks of BLOCK_H
    # -----------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # Load a tile of x: (BLOCK_I, BLOCK_J, BLOCK_H)
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # Apply mask
        x = x * mask_tile

        # Store back
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor):
    """In‑place mask multiplication using the kernel above."""
    B, N, _, H = x.shape
    s_x = x.stride()
    s_m = mask.stride()
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32
    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_mul_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# 2️⃣ Main Triton‑enabled TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3) using:
      – Triton for the optional mask multiplication,
      – Torch bmm (cuBLAS/Tensor‑cores) for the O(N³·H) core update.

    Returns a float‑32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                     # C
    hidden = config["hidden_dim"]           # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)               # (B, N, N, dim)
    B, N, _, C = x_h.shape

    # ------------------------------------------------------------------
    # 3️⃣ Fused projections & gates (two linear calls each)
    # ------------------------------------------------------------------
    # Concatenate left/right projection weights → (2*hidden, dim)
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)

    # Concatenate left/right gate weights → (2*hidden, dim)
    gate_weight = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)

    # Flatten the spatial dimensions for a single GEMM
    x_flat = x_h.view(-1, C)                       # (B*N*N, dim)

    proj = F.linear(x_flat, proj_weight)           # (B*N*N, 2*hidden)
    gate = F.linear(x_flat, gate_weight)           # (B*N*N, 2*hidden)

    proj = proj.view(B, N, N, 2 * hidden)
    gate = gate.view(B, N, N, 2 * hidden)

    left_proj, right_proj = torch.split(proj, hidden, dim=-1)   # (B,N,N,H)
    left_gate, right_gate = torch.split(gate, hidden, dim=-1)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional in‑place mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update via batched matmul (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) – each head becomes an independent batch item
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Batched GEMM: left @ rightᵀ  (FP16 matmul, FP32 accumulation)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"]
        .to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"]
        .to(dtype=out.dtype, device=device),
        eps=eps,
    )

    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(dtype=torch.float16, device=device))
    )  # (B, N, N, hidden)  FP16

    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype=torch.float16, device=device),
    )  # (B, N, N, dim)  FP16

    # ------------------------------------------------------------------
    # 9️⃣ Return FP32 tensor (matches reference implementation)
    # ------------------------------------------------------------------
    return out.float()
