```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
custom_kernel: highly‑optimized outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm on the input (float32 for numeric stability).
2. Cast to fp16 and apply a *single* fused linear projection that produces
   the left/right projections and their gates (4·hidden_dim outputs).
3. Optional mask is broadcast‑multiplied (plain torch op – cheap compared
   with the O(N³) work).
4. Apply sigmoid gates.
5. Core O(N³·H) update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication with batch = B·H.
   Implemented with a custom Triton kernel that fuses the whole
   batch dimension, uses fp16 arithmetic and fp32 accumulation.
6. Output LayerNorm + out‑gate (both in fp16).
7. Final linear projection back to the original channel dimension.
8. Return float32 tensor (matches the reference PyTorch module).

The only Triton‑implemented part is the batched GEMM (step 5); everything
else uses PyTorch’s highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (computes C = A @ Bᵀ for each batch)
#   A: (BATCH, M, K)   fp16
#   B: (BATCH, K, N)   fp16  (will be accessed as transposed, i.e. Bᵀ)
#   C: (BATCH, M, N)   fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, M, N, K,                    # logical sizes
    stride_ah, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, k‑dim, n‑dim)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)                 # which (B*H) batch
    pid_m = tl.program_id(1)                    # tile over M (i dimension)
    pid_n = tl.program_id(2)                    # tile over N (j dimension)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers to the start of the tiles (will be offset in the K‑loop)
    # --------------------------------------------------------------
    a_tile_ptr = (
        a_ptr
        + batch_id * stride_ah
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_tile_ptr = (
        b_ptr
        + batch_id * stride_bh
        + offs_k[:, None] * stride_bk          # k dimension (row of Bᵀ)
        + offs_n[None, :] * stride_bn          # n dimension (col of Bᵀ)
    )

    # --------------------------------------------------------------
    # Accumulator (fp32 for precision)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # compute mask for the current K tile (handles tail)
        cur_k = k_tile * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_tile_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write result (cast back to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + batch_id * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.

    A: (BATCH, M, K) fp16 (contiguous)
    B: (BATCH, K, N) fp16 (contiguous)
    Returns C: (BATCH, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in element units
    s_a = A.stride()
    s_b = B.stride()
    s_c = C.stride()

    # block sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        s_a[0],
        s_a[1],
        s_a[2],
        s_b[0],
        s_b[2],      # B stride for k‑dim (col stride) → transposed access
        s_b[1],      # B stride for n‑dim (row stride) → transposed access
        s_c[0],
        s_c[1],
        s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implemented with a fused Triton batched GEMM.

    Args:
        data: tuple (input, mask, weights, config)
            input   torch.Tensor [B, N, N, C]          (float32)
            mask    torch.Tensor [B, N, N]            (bool/float, optional)
            weights dict of model parameters
            config  dict with keys:
                * dim          (int)   – input channel C
                * hidden_dim   (int)   – hidden dimension H
                * nomask       (bool)  – True ⇒ ignore mask
                * layernorm_eps (float, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -------------------------- unpack ------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # C
    hidden = config["hidden_dim"]           # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------- LayerNorm ---------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  float32

    # ---------------------- cast to fp16 ----------------------------
    x_h = x.half()          # fp16
    B, N, _, C = x_h.shape

    # ------------------ fused projection + gates -------------------
    # Build (4*H, C) weight; all sub‑weights are (H, C)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4H, C)

    x_flat = x_h.view(-1, C)                     # (B·N·N, C)
    combined = F.linear(x_flat, combined_weight)  # (B·N·N, 4H)

    left_proj, right_proj, left_gate, right_gate = torch.split(
        combined, hidden, dim=-1
    )  # each (B·N·N, H)

    # reshape back to 4‑D layout
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate = left_gate.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)

    # -------------------------- mask --------------------------------
    if not nomask:
        # broadcast mask over the hidden dimension
        mask_fp = mask_tensor.to(dtype=x_h.dtype, device=device).unsqueeze(-1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # -------------------------- gating ------------------------------
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # ------------------- prepare batched GEMM -----------------------
    # shape needed for GEMM: (B, H, N, N) → view as (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N) # (B*H, N, N)

    # Triton batched matmul (computes left @ rightᵀ)
    out_mat = _batched_matmul_triton(left_mat, right_mat)  # (B*H, N, N) fp16

    # reshape back to (B, N, N, H)
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp16

    # -------------------- output LayerNorm + out_gate ---------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))  # (B,N,N,H)
    out = out * out_gate

    # -------------------------- final linear -----------------------
    out = F.linear(out, weights["to_out.weight"].half())   # (B, N, N, C) fp16

    # -------------------------- return -----------------------------
    return out.float()
```
