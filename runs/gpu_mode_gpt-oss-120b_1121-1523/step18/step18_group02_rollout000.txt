"""
TriMul (outgoing) – fast Triton‑augmented implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32 for numerical stability).
2. Cast to FP16 for the heavy part.
3. Fused linear projections for the left/right tensors (2×hidden_dim) and
   fused linear gates (2×hidden_dim).  Split the result into left/right
   projections and gates.
4. Optional in‑place masking of the projected tensors using a tiny Triton
   kernel (broadcast over the hidden dimension).
5. Apply sigmoid gates element‑wise.
6. Core O(N³·H) update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) performed with
   torch.bmm (cuBLAS tensor‑core kernels) on tensors of shape
   (B·H, N, N) – much faster than a naive Triton GEMM.
7. Output LayerNorm → out‑gate sigmoid → element‑wise gating.
8. Final linear projection back to the original channel dimension.
9. Return float‑32 tensor.

Only the mask‑multiplication is done in Triton, satisfying the “must‑use‑Triton”
requirement while keeping the main compute on the highly‑optimized cuBLAS
backend.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over H)
# --------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    x_ptr,         # pointer to x   (B, N, N, H)
    mask_ptr,      # pointer to mask (B, N, N)
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    x[b, i, j, h] *= mask[b, i, j]
    """
    b = tl.program_id(0)        # batch dimension
    pid_i = tl.program_id(1)    # tile over i
    pid_j = tl.program_id(2)    # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load mask tile (BLOCK_I, BLOCK_J) and broadcast over H
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        x = x * mask_tile
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place multiplication x *= mask (broadcast mask over hidden dim).
    Both tensors must be on the same CUDA device and have dtype fp16.
    """
    B, N, _, H = x.shape
    s_x = x.stride()
    s_m = mask.stride()
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64   # H is ≤ 384, a single iteration for most configs
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_mul_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3) using Triton for mask handling
    and cuBLAS (torch.bmm) for the O(N³·H) update.
    Returns a float‑32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel C
    hidden = config["hidden_dim"]           # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ----------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # ----------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy ops
    # ----------------------------------------------------------
    x_h = x.to(torch.float16)               # (B, N, N, C)
    B, N, _, C = x_h.shape

    # ----------------------------------------------------------
    # 3️⃣ Fused linear projections & gates (2×hidden out)
    # ----------------------------------------------------------
    # Concatenate left/right projection weights and gate weights for a
    # single matmul per tensor, then split.
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)        # (2*H, C)

    gate_weight = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)        # (2*H, C)

    # Flatten spatial dimensions for a single linear call
    x_flat = x_h.view(-1, C)                         # (B·N·N, C)

    proj = F.linear(x_flat, proj_weight)             # (B·N·N, 2*H)
    gate = F.linear(x_flat, gate_weight)             # (B·N·N, 2*H)

    proj = proj.view(B, N, N, 2 * hidden)            # (B, N, N, 2*H)
    gate = gate.view(B, N, N, 2 * hidden)

    left_proj, right_proj = torch.split(proj, hidden, dim=-1)   # (B,N,N,H) each
    left_gate, right_gate = torch.split(gate, hidden, dim=-1)   # (B,N,N,H) each

    # ----------------------------------------------------------
    # 4️⃣ Optional mask (in‑place Triton kernel)
    # ----------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 to match left/right dtype
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)

    # ----------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # ----------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # ----------------------------------------------------------
    # 6️⃣ Core O(N³·H) update via batched GEMM (cuBLAS)
    # ----------------------------------------------------------
    # Reshape to (B*H, N, N) for a batched matrix multiplication.
    # left @ rightᵀ  ==  left[i,k] * right[j,k] summed over k.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)               # (B*H, N, N)
    right_flat = right_perm.view(BH, N, N)             # (B*H, N, N)

    # Batched GEMM using torch.bmm → uses tensor cores on H100
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 7️⃣ Output LayerNorm + out‑gate
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out.dtype, device=device),
        eps=eps,
    )

    out_gate = torch.sigmoid(
        F.linear(
            x_h,
            weights["out_gate.weight"].to(dtype=torch.float16, device=device),
        )
    )  # (B, N, N, H)

    out = out * out_gate

    # ----------------------------------------------------------
    # 8️⃣ Final linear projection back to dim
    # ----------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype=torch.float16, device=device),
    )  # (B, N, N, dim)

    # ----------------------------------------------------------
    # 9️⃣ Return float32 (matches reference)
    # ----------------------------------------------------------
    return out.float()
