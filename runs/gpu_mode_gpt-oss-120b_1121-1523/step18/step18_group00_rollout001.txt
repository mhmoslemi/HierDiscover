"""
custom_kernel – high‑performance outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32 → float16 for speed).
2.  Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
    into a single matmul of shape (B·N·N, 4·H) to obtain all projected
    tensors in one pass.
3.  (Optional) apply the pair‑wise mask in‑place to the projected left/right
    tensors with a tiny Triton kernel that broadcasts the mask over the hidden
    dimension.
4.  Sigmoid gates → `left = left_proj * sigmoid(left_gate)`,
    `right = right_proj * sigmoid(right_gate)`.
5.  Core O(N³·H) update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    realized as a batched GEMM:
        out_h = left_h @ right_hᵀ   for every head `h`.
    This is performed with a single `torch.bmm` on the reshaped tensor
    (B·H, N, N), which is many independent half‑precision matrix
    multiplications and fully exploits the H100 tensor cores.
6.  Hidden‑dim layer‑norm, output gate (sigmoid) and final linear projection
    back to the original channel dimension `dim`.
7.  Return a float‑32 tensor (compatible with the reference module).

Only the mask‑multiplication step is written in Triton – the rest stays in
PyTorch but runs completely in half‑precision, giving a large speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: broadcast mask over the hidden dimension and
#                multiply it in‑place into two tensors (left & right)
# -------------------------------------------------------------
@triton.jit
def _mask_mul_two_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    left[b,i,j,h] *= mask[b,i,j];  right[b,i,j,h] *= mask[b,i,j]
    Assumes tensors are contiguous in memory (as they will be after
    the preceding half‑precision ops) and that mask holds 0/1 values.
    """
    b = tl.program_id(0)                # batch dimension
    pid_i = tl.program_id(1)            # tile over i
    pid_j = tl.program_id(2)            # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # ---------------------------------------------------------
    # Load a (BLOCK_I, BLOCK_J) tile of the mask and broadcast
    # ---------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i[:, None] * stride_mask_i
        + j[None, :] * stride_mask_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # ---------------------------------------------------------
    # Loop over the hidden dimension
    # ---------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # left
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # right
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # apply mask
        left = left * mask_tile
        right = right * mask_tile
        # store back
        tl.store(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_two_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place broadcast‑multiplication of a (B,N,N) mask onto the
    hidden dimension of `left` and `right` tensors (B,N,N,H).
    """
    B, N, _, H = left.shape
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_mul_two_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # in‑place modification; no return


# -------------------------------------------------------------
# Outgoing TriMul – forward‑only (no autograd needed)
# -------------------------------------------------------------

    """

    # Unpack inputs
    # -----------------------------------------------------------------

    device = input_tensor.device

    # Load weights (all stored as float32)
    # -----------------------------------------------------------------

    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numeric stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C]   float32

    # -----------------------------------------------------------------
    # 2️⃣ Switch to fp16 for the heavy path
    # -----------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, C]   fp16
    B, N, _, C = x_h.shape

    # -----------------------------------------------------------------
    # 3️⃣ Fuse left/right projection + gating linear layers
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)        # (4*H, C)

    x_flat = x_h.view(-1, C)                        # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)      # (B·N·N, 4*H)

    # Split and reshape back to (B, N, N, H)
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – in‑place via tiny Triton kernel
    # -----------------------------------------------------------------
    if (not nomask) and mask_tensor is not None:
        # Cast mask to half‑precision (0/1 values) – broadcast later
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_two_triton(left_proj, right_proj, mask_h)

    # -----------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -----------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    left = left_proj * left_gate          # (B,N,N,H) fp16
    right = right_proj * right_gate       # (B,N,N,H) fp16

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update – batched GEMM over the hidden dimension
    #    out_h = left_h @ right_hᵀ  for each head h.
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    B_h, H_, N_, _ = left_t.shape  # B_h == B, H_ == hidden
    left_mat = left_t.view(B * hidden, N, N)          # (B*H, N, N)
    right_mat = right_t.view(B * hidden, N, N)        # (B*H, N, N)

    # Batched matrix multiply: left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)    # (B,N,N,H) fp16

    # -----------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final projection back to input dim C
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Return float32 for compatibility with reference implementation
    # -----------------------------------------------------------------
    return out.float()
```
