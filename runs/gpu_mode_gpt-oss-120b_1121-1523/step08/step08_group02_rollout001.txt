"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1. **Layer‑norm** over the input channel dimension (FP32) – performed with
   ``torch.nn.functional.layer_norm`` because the reduction axis is tiny
   (C ∈ {128,384}) and the overhead of a custom reduction would outweigh any
   gain.

2. **Linear projections** (no bias) → ``left_proj`` and ``right_proj``.
   The result is cast to FP16 to exploit Tensor‑Core GEMM later.

3. **Sigmoid gates** for the two sides and the output gate.
   The side‑gates are also cast to FP16; the output gate stays FP32.

4. **Fuse batch & hidden dimensions**: reshape  
   ``[B, N, N, H] → [B·H, N, N]`` (contiguous, FP16).  This yields a single
   huge batched matrix‑multiply problem.

5. **Element‑wise gating + optional mask** – a single Triton kernel
   ``_fuse_proj_gate_mask_kernel`` loads tiles of the left/right
   projections, multiplies them by the corresponding gate and (if present)
   by the pairwise mask (the mask is broadcast across the hidden dimension
   without copying).  The kernel writes the gated tensors ``left_gated``,
   ``right_gated`` in FP16.

6. **Batched GEMM** – ``torch.bmm`` performs the heavy
   ``N × N`` matrix multiplies for all ``B·H`` matrices in one launch,
   using cuBLAS/Tensor‑Core (FP16 × FP16 → FP32 accumulation).

7. **Reshape back** to ``[B, N, N, H]`` (FP32) and apply the hidden‑dim
   LayerNorm, the output‑gate, and the final linear projection back to the
   original channel dimension.

Only the element‑wise gating / mask step is written in Triton; all heavy
linear algebra re‑uses the highly‑optimised cuBLAS kernels, giving sub‑ms
latency on an H100 for the required configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of projection, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    # input tensors
    proj_a_ptr,   gate_a_ptr,   # left projection & its gate  [B·H, N, N]  (fp16)
    proj_b_ptr,   gate_b_ptr,   # right projection & its gate [B·H, N, N]  (fp16)
    mask_ptr,                     # mask [B, N, N] (fp16) – broadcasted across H
    # output tensors
    out_a_ptr, out_b_ptr,         # gated left / gated right   [B·H, N, N]
    # matrix sizes
    M, N,                         # both equal to seq_len
    # strides for the four inputs
    stride_pa_batch, stride_pa_m, stride_pa_n,
    stride_ga_batch, stride_ga_m, stride_ga_n,
    stride_pb_batch, stride_pb_m, stride_pb_n,
    stride_gb_batch, stride_gb_m, stride_gb_n,
    # strides for the outputs (identical layout to the inputs)
    stride_oa_batch, stride_oa_m, stride_oa_n,
    stride_ob_batch, stride_ob_m, stride_ob_n,
    # mask strides (original mask layout)
    mask_batch_stride, mask_stride_m, mask_stride_n,
    # compile‑time constants
    H: tl.constexpr,                # hidden dimension (used only for mask mapping)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_bh = tl.program_id(0)          # index over (batch * hidden)
    pid_m  = tl.program_id(1)          # row tile
    pid_n  = tl.program_id(2)          # col tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Load tiles of the four FP16 tensors
    # ------------------------------------------------------------------
    a = tl.load(
        proj_a_ptr + pid_bh * stride_pa_batch
                      + offs_m[:, None] * stride_pa_m
                      + offs_n[None, :] * stride_pa_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    ag = tl.load(
        gate_a_ptr + pid_bh * stride_ga_batch
                       + offs_m[:, None] * stride_ga_m
                       + offs_n[None, :] * stride_ga_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    b = tl.load(
        proj_b_ptr + pid_bh * stride_pb_batch
                      + offs_m[:, None] * stride_pb_m
                      + offs_n[None, :] * stride_pb_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    bg = tl.load(
        gate_b_ptr + pid_bh * stride_gb_batch
                       + offs_m[:, None] * stride_gb_m
                       + offs_n[None, :] * stride_gb_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )

    left  = a * ag
    right = b * bg

    # ------------------------------------------------------------------
    # Optional mask – broadcasted across the hidden dimension
    # ------------------------------------------------------------------
    if HAS_MASK:
        # map the flattened batch*hidden index back to the original batch
        batch_idx = pid_bh // H
        mask_base = mask_ptr + batch_idx * mask_batch_stride
        m = tl.load(
            mask_base
            + offs_m[:, None] * mask_stride_m
            + offs_n[None, :] * mask_stride_n,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Store gated results
    # ------------------------------------------------------------------
    tl.store(
        out_a_ptr + pid_bh * stride_oa_batch
                        + offs_m[:, None] * stride_oa_m
                        + offs_n[None, :] * stride_oa_n,
        left,
        mask=mask_m[:, None] & mask_n[None, :],
    )
    tl.store(
        out_b_ptr + pid_bh * stride_ob_batch
                        + offs_m[:, None] * stride_ob_m
                        + offs_n[None, :] * stride_ob_n,
        right,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, N, N, C]  (float32, on CUDA)
        - mask_tensor  : torch.Tensor[bs, N, N]    (bool/float, optional)
        - weights_dict : dict of model parameters (see reference)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor[bs, N, N, C] (float32)
        The processed tensor.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype                # expected torch.float32

    dim        = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]          # H
    eps        = 1e-5
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the channel dimension (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  (FP32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) + sigmoid gates
    # ------------------------------------------------------------------
    # Projections to hidden dim H – keep FP16 for the GEMM later
    left_proj  = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # side‑gates (sigmoid) – also FP16
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    # output‑gate stays FP32 (applied after the hidden‑dim LayerNorm)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))       # [B,N,N,H] FP32

    # ------------------------------------------------------------------
    # 3) Merge batch and hidden dimensions
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_batch = B * hidden_dim                               # B·H

    # reshape to [B·H, N, N] – keep data contiguous for the GEMM
    left_proj  = left_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    left_gate  = left_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)

    # ------------------------------------------------------------------
    # 4) Optional mask – broadcast across hidden dim (no copy)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> FP16
        mask_fp16 = mask_tensor.to(torch.float16)
    else:
        # dummy tensor – never read because HAS_MASK==0
        mask_fp16 = left_proj  # any tensor of compatible shape

    # ------------------------------------------------------------------
    # 5) Fuse projection, gate and mask (Triton)
    # ------------------------------------------------------------------
    left_gated  = torch.empty_like(left_proj, dtype=torch.float16, device=device)
    right_gated = torch.empty_like(right_proj, dtype=torch.float16, device=device)

    # Choose block sizes (all seq_lens in the test suite are multiples of 64)
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128

    grid = (
        total_batch,                 # one program per (batch * hidden) matrix
        triton.cdiv(N, BLOCK_M),    # rows
        triton.cdiv(N, BLOCK_N),    # cols
    )
    HAS_MASK = 0 if nomask else 1

    _fuse_proj_gate_mask_kernel[grid](
        # pointers
        proj_a_ptr=left_proj,
        gate_a_ptr=left_gate,
        proj_b_ptr=right_proj,
        gate_b_ptr=right_gate,
        mask_ptr=mask_fp16,
        out_a_ptr=left_gated,
        out_b_ptr=right_gated,
        # sizes
        M=N,
        N=N,
        # strides for left projection / left gate
        stride_pa_batch=left_proj.stride(0), stride_pa_m=left_proj.stride(1), stride_pa_n=left_proj.stride(2),
        stride_ga_batch=left_gate.stride(0), stride_ga_m=left_gate.stride(1), stride_ga_n=left_gate.stride(2),
        # strides for right projection / right gate
        stride_pb_batch=right_proj.stride(0), stride_pb_m=right_proj.stride(1), stride_pb_n=right_proj.stride(2),
        stride_gb_batch=right_gate.stride(0), stride_gb_m=right_gate.stride(1), stride_gb_n=right_gate.stride(2),
        # output strides (identical layout)
        stride_oa_batch=left_gated.stride(0), stride_oa_m=left_gated.stride(1), stride_oa_n=left_gated.stride(2),
        stride_ob_batch=right_gated.stride(0), stride_ob_m=right_gated.stride(1), stride_ob_n=right_gated.stride(2),
        # mask strides (original mask layout)
        mask_batch_stride=mask_fp16.stride(0),
        mask_stride_m=mask_fp16.stride(1),
        mask_stride_n=mask_fp16.stride(2),
        # compile‑time constants
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 6) Batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # right matrix must be transposed for the inner‑product over k
    right_gated_t = right_gated.transpose(1, 2)          # [B·H, N, N]
    out_fp16 = torch.bmm(left_gated, right_gated_t)     # [B·H, N, N]  (FP16, acc in FP32)

    # ------------------------------------------------------------------
    # 7) Restore shape & final layers
    # ------------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)   # [B, H, N, N]
        .permute(0, 2, 3, 1)                 # [B, N, N, H]
        .contiguous()
        .to(dtype)                           # back to FP32
    )

    # hidden‑dim LayerNorm + output gate
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # final linear projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
```
