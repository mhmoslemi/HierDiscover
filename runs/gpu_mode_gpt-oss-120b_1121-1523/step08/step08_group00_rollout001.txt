def tri_mul_outgo_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)  # 0 <= bh < B*H
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # masks
    i_mask = i_start + tl.arange(0, BLOCK_M) < N
    j_mask = j_start + tl.arange(0, BLOCK_N) < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute base pointers
    left_base = left_ptr + b*stride_l_b + h*stride_l_h
    right_base = right_ptr + b*stride_r_b + h*stride_r_h
    out_base = out_ptr + b*stride_o_b + h*stride_o_h

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # Load left tile (M,K)
        left = tl.load(
            left_base
            + (i_start + tl.arange(0, BLOCK_M))[:, None] * stride_l_i
            + (k_start + tl.arange(0, BLOCK_K))[None, :] * stride_l_k,
            mask=i_mask[:, None] & (k_start + tl.arange(0, BLOCK_K) < N)[None,:],
            other=0.0,
        )
        # Load right tile (K,N) (note the order for dot)
        right = tl.load(
            right_base
            + (k_start + tl.arange(0, BLOCK_K))[:, None] * stride_r_k
            + (j_start + tl.arange(0, BLOCK_N))[None, :] * stride_r_j,
            mask=(k_start + tl.arange(0, BLOCK_K) < N)[:, None] & j_mask[None, :],
            other=0.0,
        )
        # Compute dot product
        acc += tl.dot(left, right)  # left MxK, right KxN

        # Increment pointers? Not needed as we compute base each iteration.

    # Write output
    out_ptrs = out_base + (i_start + tl.arange(0, BLOCK_M))[:, None] * stride_o_i + (j_start + tl.arange(0, BLOCK_N))[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=i_mask[:, None] & j_mask[None, :])

def tri_mul_outgo_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute combined batch
    bh = tl.program_id(0)
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_mask = i_range < N
    j_mask = j_range < N

    # pointers base for this batch/h
    left_batch = left_ptr + b * stride_l_b + h * stride_l_h
    right_batch = right_ptr + b * stride_r_b + h * stride_r_h
    out_batch = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_valid = (k_base + k_range) < N

        # load left tile M x K
        left = tl.load(
            left_batch
            + (i_range[:, None] * stride_l_i)
            + ((k_base + k_range)[None, :] * stride_l_k),
            mask=i_mask[:, None] & k_valid[None, :],
            other=0.0,
        )
        # load right tile K x N (note swapped axes)
        right = tl.load(
            right_batch

            + (j_range[None, :] * stride_r_j),
            mask=k_valid[:, None] & j_mask[None, :],
            other=0.0,

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. Input LayerNorm FP32
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # 2. Fused projection & gating (FP16)
    x_fp16 = x.half()
    combined_weight = torch.cat([left_proj_weight, left_gate_weight,
                                 right_proj_weight, right_gate_weight], dim=0).to(dtype=torch.float16)
    combined = F.linear(x_fp16, combined_weight)  # shape [B,N,N,4*H]
    left_proj, left_gate, right_proj, right_gate = torch.split(combined, hidden_dim, dim=-1)

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    left = (left_proj * left_gate).contiguous()
    right = (right_proj * right_gate).contiguous()

    # 3. optional mask
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # 4. Core O(N^3) via Triton GEMM
    B, N, _, H = left.shape

    # strides in elements
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_outgo_kernel[grid](
        left,
        right,
        out,
        B, H, N,

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
--------------------------------------------------

Algorithm
~~~~~~~~~
1. **Layer‑Norm** on the 4‑D input (FP32).
2. **Fused projection + gating**:
   a single half‑precision linear layer produces
   left‑proj, left‑gate, right‑proj and right‑gate.
   The gates are passed through sigmoid and applied.
3. **Optional mask** – an in‑place Triton kernel multiplies left and right
   by the scalar mask (broadcast over the hidden dimension).
4. **Core O(N³) update** – for each hidden slice *h* we compute
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` as a
   batched matrix multiplication `left @ rightᵀ`.  This is realised
   by a custom Triton GEMM kernel that:
   * treats each (batch, hidden) pair as an independent program id,
   * tiles the i‑, j‑ and k‑dimensions (64×64×64),
   * uses FP16 loads and a FP32 accumulator (Tensor‑Core friendly),
   * writes the result directly as FP32.
5. **Output LayerNorm → output‑gate → final linear** (all FP32).

Only the mask‑apply and the heavy O(N³) contraction are written in
Triton; everything else leverages highly‑optimised PyTorch kernels.
The implementation respects the required signatures and works for all
test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F

import triton.language as tl

# Triton kernel: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,                     # fp16 *
    right_ptr,                    # fp16 *
    mask_ptr,                     # fp16 *
    B, N, H,                      # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Multiply `left` and `right` (shape [B,N,N,H]) by a scalar mask
    `mask` (shape [B,N,N]) broadcast over the hidden dimension.
    One program processes a BLOCK_I × BLOCK_J tile of the (i,j) plane.
    """
    b = tl.program_id(0)
    ti = tl.program_id(1)
    tj = tl.program_id(2)

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # scalar mask (broadcast over H)
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]      # (BLOCK_I, BLOCK_J, 1)

    # load left / right tiles
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )

    # apply mask
    left = left * mask
    right = right * mask

    # store back
    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )

# Triton kernel: batched GEMM for the O(N³) TriMul update
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # fp16 *, fp16 *, fp32 *
    B, H, N,                                    # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (b, h) pair:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left  : [B, N, N, H]  (i, k)
    right : [B, N, N, H]  (j, k) – note that the k‑dimension is shared,
                                 hence we load right as (K, N) to obtain a
                                 dot product of (M, K) × (K, N).
    out   : [B, N, N, H]  (FP32 accumulator)
    """
    bh = tl.program_id(0)                         # 0 ≤ bh < B*H
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_mask = i < N
    j_mask = j < N

    # base pointers for this (b, h) slice
    left_batch  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_batch = right_ptr + b * stride_r_b + h * stride_r_h
    out_batch   = out_ptr   + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_base = kk * BLOCK_K
        k_valid = (k_base + k) < N

        # left tile: (M, K)
        left = tl.load(
            left_batch
            + (i[:, None] * stride_l_i)
            + ((k_base + k)[None, :] * stride_l_k),
            mask=i_mask[:, None] & k_valid[None, :],
            other=0.0,
        )
        # right tile: (K, N)  – we load it transposed w.r.t. its original layout
        right = tl.load(
            right_batch
            + ((k_base + k)[:, None] * stride_r_k)
            + (j[None, :] * stride_r_j),
            mask=k_valid[:, None] & j_mask[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(left, right)   # fp16 × fp16 → fp32 accumulation

    # write result
    out_ptrs = out_batch + (i[:, None] * stride_o_i) + (j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model weights (all stored as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # (H, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                                   # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2) Fused projection + gating (FP16)
    #    One linear produces left_proj, left_gate, right_proj, right_gate.
    # ------------------------------------------------------------------
    x_fp16 = x.half()
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16)
    combined = F.linear(x_fp16, combined_weight)        # [B, N, N, 4*H] (fp16)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    left = (left_proj * left_gate).contiguous()   # [B, N, N, H] fp16
    right = (right_proj * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 3) Optional mask (in‑place via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 4) Core O(N³) update – batched GEMM using a custom Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty(B, N, N, H, dtype=torch.float32, device=left.device)

    # strides expressed in elements (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                   # program id 0  (batch × hidden)
        (N + BLOCK_M - 1) // BLOCK_M,            # program id 1  (i‑tiles)
        (N + BLOCK_N - 1) // BLOCK_N,            # program id 2  (j‑tiles)
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, H, N,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, output‑gate and final projection (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                                   # FP32

    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # FP32, shape [B,N,N,H]
    out = out * out_gate

    out = F.linear(out, to_out_weight)                  # back to dim, FP32
    return out
```
This implementation keeps the heavy O(N³) computation on the GPU via a specialised Triton GEMM, uses an in‑place mask kernel, and performs all other steps with PyTorch’s highly‑optimised primitives. It satisfies the required API and works for all configurations listed in the problem statement.
