def triton_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)   # i tile
    pid2 = tl.program_id(2)   # j tile

    b = pid0 // H
    h = pid0 % H

    i_offset = pid1 * BLOCK_M
    j_offset = pid2 * BLOCK_N

    # range indices
    i = i_offset + tl.arange(0, BLOCK_M)
    j = j_offset + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_ok = i < N
    j_ok = j < N
    k_ok = k < N

    # Initialize accumulator

    for kk in range(0, tl.cdiv(N, BLOCK_K)):

@triton.jit
def outgoing_triton_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr, # if nomask, we pass a dummy pointer
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_j,
    stride_r_b, stride_r_h, stride_r_i, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    stride_m_b, stride_m_i, stride_m_j,  # mask strides (if used)
    USE_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # combined batch*head
    pid1 = tl.program_id(1)  # i tile
    pid2 = tl.program_id(2)  # j tile

    # compute b, h
    b = pid0 // H
    h = pid0 % H

    i_base = pid1 * BLOCK_M
    j_base = pid2 * BLOCK_N

    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # bounds
    i_mask = i_range < N
    j_mask = j_range < N
    k_mask = k_range < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute offsets base for given b and h
    left_base = left_ptr + b*stride_l_b + h*stride_l_h
    right_base = right_ptr + b*stride_r_b + h*stride_r_h
    out_base = out_ptr + b*stride_o_b + h*stride_o_h

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k_idx = k_base + k_range
        cur_k_mask = k_idx < N

        # Load left tile A[i,k]
        a = tl.load(
            left_base
            + i_range[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_j,
            mask=i_mask[:, None] & cur_k_mask[None, :],
            other=0.0,
        )

        # Load right tile B_T[k,j] => right[b,h,k,j]
        b_tile = tl.load(
            right_base
            + k_idx[:, None] * stride_r_i
            + j_range[None, :] * stride_r_j,
            mask=cur_k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )

        if USE_MASK:
            # Load mask tiles for left (i,k) and right (j,k) 
            # mask shape [B,N,N] ; left uses mask[i,k], right uses mask[j,k]
            mask_i_k = tl.load(
                mask_ptr + b*stride_m_b
                + i_range[:, None] * stride_m_i
                + k_idx[None, :] * stride_m_j,
                mask=i_mask[:, None] & cur_k_mask[None, :],
                other=1.0,
            )
            mask_j_k = tl.load(
                mask_ptr + b*stride_m_b
                + j_range[:, None] * stride_m_i
                + k_idx[None, :] * stride_m_j,
                mask=j_mask[:, None] & cur_k_mask[None, :],
                other=1.0,
            )
            a = a * mask_i_k
            b_tile = b_tile * mask_j_k

        # Accumulate
        acc += tl.dot(a, b_tile)

        # advance offsets: compute next k tile automatically using loops; although using pointer arithmetic we compute each tile based on indices.
    # Store output accumulate
    tl.store(
        out_base
        + i_range[:, None] * stride_o_i
        + j_range[None, :] * stride_o_j,
        acc,
        mask=i_mask[:, None] & j_mask[None, :],
    )

"""
custom_kernel – highly‑optimized “outgoing” TriMul (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32).
2️⃣ Fuse the four linear projections (left‑proj, left‑gate,
   right‑proj, right‑gate) into a single half‑precision GEMM.
   Split → left_proj, left_gate, right_proj, right_gate.
   Apply sigmoid‑gate → `left`, `right`  (fp16).
3️⃣ Core O(N³) update:   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   Implemented as a batched half‑precision GEMM with a custom Triton
   kernel that accumulates in fp32.  Optional mask is applied inside the
   kernel (broadcast over the hidden dimension).
4️⃣ Output LayerNorm + sigmoid‑gate (`out_gate`) + final projection
   back to the original channel dimension (float32).

All heavy arithmetic (step 2 & 3) runs on Tensor‑Core fp16 → fp32,
minimising memory traffic and kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) update, optional mask.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,      # fp16 *
    right_ptr,     # fp16 *
    out_ptr,       # fp32 *
    mask_ptr,      # fp16 *   (may be dummy)
    B, H, N,       # runtime sizes
    # strides for left (B,H,N,N)   (elements, not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_j,
    # strides for right (B,H,N,N)
    stride_r_b, stride_r_h, stride_r_i, stride_r_j,
    # strides for output (B,H,N,N)
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    # strides for mask (B,N,N)
    stride_m_b, stride_m_i, stride_m_j,
    # compile‑time constants
    USE_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ   (batched over B and H)."""
    pid0 = tl.program_id(0)          # combined batch‑head dimension
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    # ------------------------------------------------------------------
    # Decode batch and head indices
    # ------------------------------------------------------------------
    b = pid0 // H
    h = pid0 % H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    i_base = pid1 * BLOCK_M
    j_base = pid2 * BLOCK_N
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_ok = i < N
    j_ok = j < N
    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Base pointers for the current (b,h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h
    mask_base  = mask_ptr  + b * stride_m_b          # used only if USE_MASK

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        k_idx   = k_start + k
        k_ok = k_idx < N

        # left[i, k]  (shape BLOCK_M × BLOCK_K)
        a = tl.load(
            left_base
            + i[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_j,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        )

        # rightᵀ[k, j] == right[k, j] (shape BLOCK_K × BLOCK_N)
        b_tile = tl.load(
            right_base
            + k_idx[:, None] * stride_r_i
            + j[None, :] * stride_r_j,
            mask=k_ok[:, None] & j_ok[None, :],
            other=0.0,
        )

        if USE_MASK:
            # mask for left  : mask[i, k]
            mask_i_k = tl.load(
                mask_base
                + i[:, None] * stride_m_i
                + k_idx[None, :] * stride_m_j,
                mask=i_ok[:, None] & k_ok[None, :],
                other=1.0,
            )
            # mask for right : mask[j, k]
            mask_j_k = tl.load(
                mask_base
                + j[:, None] * stride_m_i
                + k_idx[None, :] * stride_m_j,
                mask=j_ok[:, None] & k_ok[None, :],
                other=1.0,
            )
            a = a * mask_i_k
            b_tile = b_tile * mask_j_k

        # Accumulate (fp16 × fp16 → fp32 via Tensor‑Core)
        acc += tl.dot(a, b_tile)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    tl.store(
        out_base
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j,
        acc,
        mask=i_ok[:, None] & j_ok[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor | None,
                         hidden_dim: int) -> torch.Tensor:
    """
    Batched half‑precision GEMM `left @ rightᵀ` with optional mask.
    left/right: [B, N, N, H] (fp16)
    mask:      [B, N, N]   (fp16) or ``None`` for no mask
    Returns:   out  [B, N, N, H] (fp32)
    """
    device = left.device
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    # Permute to (B, H, N, N) and make contiguous for simple stride logic
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out = torch.empty_like(left_perm, dtype=torch.float32)

    # strides in *elements* (not bytes)
    s_l = left_perm.stride()
    s_r = right_perm.stride()
    s_o = out.stride()

    if mask is None:
        mask_fp16 = torch.empty((1,), dtype=torch.float16, device=device)
        stride_m = (0, 0, 0)
        use_mask = 0
    else:
        mask_fp16 = mask.to(dtype=torch.float16)
        stride_m = mask_fp16.stride()
        use_mask = 1

    # ------------------------------------------------------------------
    # Tuning parameters – these work well on H100 for the given ranges.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_warps = 8

    grid = (
        B * H,                                  # combined batch‑head
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out,
        mask_fp16,
        B, H, N,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        stride_m[0], stride_m[1], stride_m[2],
        USE_MASK=use_mask,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )
    # Back‑to (B, N, N, H)
    return out.permute(0, 2, 3, 1).contiguous()


# ----------------------------------------------------------------------
# Entry point – forward‑only “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating (half‑precision)
    # ------------------------------------------------------------------
    x_h = x.half()  # fp16
    # concatenate the four projection matrices (order matters!)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
        ],
        dim=0,
    ).to(dtype=torch.float16)  # (4*H, C)

    # One GEMM → [B, N, N, 4*H]
    combined = F.linear(x_h, combined_weight)  # fp16

    # Split back into the four branches
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )

    # Apply sigmoid‑gates
    left = left_proj * left_gate.sigmoid()   # (fp16)
    right = right_proj * right_gate.sigmoid()  # (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Core O(N³) update – Triton batched GEMM
    # ------------------------------------------------------------------
    mask = None if nomask else mask_tensor
    out = _triton_batched_gemm(left, right, mask, hidden_dim)  # fp32, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Output LayerNorm + output‑gate + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # fp32

    # output gate comes from the *normed* input `x`
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # (B,N,N,H)
    out = out * out_gate

    # final linear back to original channel dimension
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim) fp32
    return out
