```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def proj_gate_mask_kernel(
    x_ptr, left_ptr, right_ptr, mask_ptr,
    B, N, C, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    wlp_ptr, wlg_ptr, wrp_ptr, wrg_ptr,
    stride_w_c, stride_w_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
    NOMASK: tl.constexpr
):
    b = tl.program_id(0)
    ti = tl.program_id(1)
    tj = tl.program_id(2)

    i0 = ti * BLOCK_I
    j0 = tj * BLOCK_J

    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # loop over hidden dimension blocks
    for h_off in range(0, H, BLOCK_H):
        h = h_off + tl.arange(0, BLOCK_H)
        h_mask = h < H

        # accumulators in fp32
        acc_lp = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
        acc_rp = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # reduction over C
        for c_off in range(0, C, BLOCK_C):
            c = c_off + tl.arange(0, BLOCK_C)
            c_mask = c < C

            # load x block
            x = tl.load(
                x_ptr
                + b * stride_x_b
                + i[:, None, None] * stride_x_i
                + j[None, :, None] * stride_x_j
                + c[None, None, :] * stride_x_c,
                mask=i_mask[:, None, None] & j_mask[None, :, None] & c_mask[None, None, :],
                other=0.0,
            )  # (BLOCK_I, BLOCK_J, BLOCK_C)

            # load weight blocks
            w_lp = tl.load(
                wlp_ptr
                + c[:, None] * stride_w_c
                + h[None, :] * stride_w_h,
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            )  # (BLOCK_C, BLOCK_H)

            w_lg = tl.load(
                wlg_ptr
                + c[:, None] * stride_w_c
                + h[None, :] * stride_w_h,
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            )
            w_rp = tl.load(
                wrp_ptr
                + c[:, None] * stride_w_c
                + h[None, :] * stride_w_h,
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            )
            w_rg = tl.load(
                wrg_ptr
                + c[:, None] * stride_w_c
                + h[None, :] * stride_w_h,
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            )

            # compute partial contributions
            # x: (I,J,C) ; w: (C,H) -> prod: (I,J,C,H)
            x_exp = x[:, :, :, None]          # (I,J,C,1)
            w_lp_exp = w_lp[None, None, :, :] # (1,1,C,H)
            acc_lp += tl.sum(x_exp * w_lp_exp, axis=2)

            w_lg_exp = w_lg[None, None, :, :]
            acc_lg += tl.sum(x_exp * w_lg_exp, axis=2)

            w_rp_exp = w_rp[None, None, :, :]
            acc_rp += tl.sum(x_exp * w_rp_exp, axis=2)

            w_rg_exp = w_rg[None, None, :, :]
            acc_rg += tl.sum(x_exp * w_rg_exp, axis=2)

        # Apply sigmoid gates
        gate_l = tl.sigmoid(acc_lg)
        gate_r = tl.sigmoid(acc_rg)

        left_val = acc_lp * gate_l
        right_val = acc_rp * gate_r

        # Optional mask
        if not NOMASK:
            # mask shape (B,N,N) broadcast over H
            mask = tl.load(
                mask_ptr
                + b * stride_m_b
                + i[:, None] * stride_m_i
                + j[None, :] * stride_m_j,
                mask=i_mask[:, None] & j_mask[None, :],
                other=1.0,
            )  # (BLOCK_I, BLOCK_J)
            mask = mask[:, :, None]  # (BLOCK_I, BLOCK_J, 1)
            left_val = left_val * mask
            right_val = right_val * mask

        # Cast to fp16 for storage
        left_h = left_val.to(tl.float16)
        right_h = right_val.to(tl.float16)

        # store
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],

"""
Fast outgoing TriMul (AlphaFold3) – Triton‑fused projection, gating & mask.

Algorithm (float32 → fp16 → fp32)
--------------------------------------------------------------
1. LayerNorm the input (FP32) → x.
2. One fused Triton kernel computes, for every pair (b,i,j):
   • left_proj = x[b,i,j,:] @ W_left_proj
   • left_gate = sigmoid(x[b,i,j,:] @ W_left_gate)
   • right_proj = x[b,i,j,:] @ W_right_proj
   • right_gate = sigmoid(x[b,i,j,:] @ W_right_gate)
   • left  = left_proj * left_gate   (optionally * mask)
   • right = right_proj * right_gate (optionally * mask)
   All tensors are half‑precision; only H values are written per element.
3. Batched half‑precision GEMM:
   left_mat  = left.permute(0,3,1,2)   # (B,H,N,N)
   right_mat = right.permute(0,3,1,2)  # (B,H,N,N)
   out_mat   = torch.matmul(left_mat,
                            right_mat.transpose(-2,-1))         # fp16 → fp32 acc.
4. LayerNorm + output‑gate + final projection (FP32) → result.
The kernel fuses four linear layers, four sigmoid gates and the optional
masking into a single launch, halving memory traffic compared with the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused projection + gating + (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    # tensor pointers
    x_ptr,               # fp16 *  (B,N,N,C)
    left_ptr,            # fp16 *  (B,N,N,H) – output
    right_ptr,           # fp16 *  (B,N,N,H) – output
    mask_ptr,            # fp16 *  (B,N,N)   – optional
    # runtime sizes
    B, N, C, H,
    # strides (in elements, not bytes)
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    # weight pointers (fp16)
    wlp_ptr, wlg_ptr, wrp_ptr, wrg_ptr,
    stride_w_c, stride_w_h,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
    NOMASK: tl.constexpr
):
    """Compute left/right projections, their sigmoid gates and apply mask."""
    b = tl.program_id(0)          # batch dimension
    ti = tl.program_id(1)         # i‑tile
    tj = tl.program_id(2)         # j‑tile

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles (BLOCK_H) --------------------------------
    # ------------------------------------------------------------------
    for h_off in range(0, H, BLOCK_H):
        h = h_off + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # accumulators in fp32
        acc_lp = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
        acc_rp = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Reduce over C in BLOCK_C chunks
        # ------------------------------------------------------------------
        for c_off in range(0, C, BLOCK_C):
            c = c_off + tl.arange(0, BLOCK_C)
            c_ok = c < C

            # ----- load x[b,i,j,c] -------------------------------------------------
            x = tl.load(
                x_ptr
                + b * stride_x_b
                + i[:, None, None] * stride_x_i
                + j[None, :, None] * stride_x_j
                + c[None, None, :] * stride_x_c,
                mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
                other=0.0,
            )                       # (BLOCK_I, BLOCK_J, BLOCK_C)

            # ----- load weight tiles -------------------------------------------------
            # shape (BLOCK_C, BLOCK_H)
            w_lp = tl.load(
                wlp_ptr + c[:, None] * stride_w_c + h[None, :] * stride_w_h,
                mask=c_ok[:, None] & h_ok[None, :],
                other=0.0,
            )
            w_lg = tl.load(
                wlg_ptr + c[:, None] * stride_w_c + h[None, :] * stride_w_h,
                mask=c_ok[:, None] & h_ok[None, :],
                other=0.0,
            )
            w_rp = tl.load(
                wrp_ptr + c[:, None] * stride_w_c + h[None, :] * stride_w_h,
                mask=c_ok[:, None] & h_ok[None, :],
                other=0.0,
            )
            w_rg = tl.load(
                wrg_ptr + c[:, None] * stride_w_c + h[None, :] * stride_w_h,
                mask=c_ok[:, None] & h_ok[None, :],
                other=0.0,
            )

            # ----- accumulate -------------------------------------------------------
            # expand dims for broadcasting: (I,J,C,1) * (1,1,C,H) → (I,J,C,H)
            x_exp = x[:, :, :, None]               # (BI,BJ,BC,1)
            # left projection
            acc_lp += tl.sum(x_exp * w_lp[None, None, :, :], axis=2)
            # left gate
            acc_lg += tl.sum(x_exp * w_lg[None, None, :, :], axis=2)
            # right projection
            acc_rp += tl.sum(x_exp * w_rp[None, None, :, :], axis=2)
            # right gate
            acc_rg += tl.sum(x_exp * w_rg[None, None, :, :], axis=2)

        # ------------------------------------------------------------------
        # Apply sigmoid gates
        # ------------------------------------------------------------------
        gate_l = tl.sigmoid(acc_lg)      # (BI,BJ,BH)
        gate_r = tl.sigmoid(acc_rg)

        left_val  = acc_lp * gate_l
        right_val = acc_rp * gate_r

        # ------------------------------------------------------------------
        # Optional mask (broadcast over H)
        # ------------------------------------------------------------------
        if tl.constexpr(NOMASK):
            # no mask – nothing to do
            pass
        else:
            mask = tl.load(
                mask_ptr
                + b * stride_m_b
                + i[:, None] * stride_m_i
                + j[None, :] * stride_m_j,
                mask=i_ok[:, None] & j_ok[None, :],
                other=1.0,
            )                               # (BI,BJ)
            mask = mask[:, :, None]         # (BI,BJ,1)
            left_val  = left_val  * mask
            right_val = right_val * mask

        # ------------------------------------------------------------------
        # Store results (cast back to fp16)
        # ------------------------------------------------------------------
        left_h  = left_val.to(tl.float16)
        right_h = right_val.to(tl.float16)

        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper to launch the fused projection kernel
# ----------------------------------------------------------------------
def _proj_gate_mask_triton(x: torch.Tensor,
                           left: torch.Tensor,
                           right: torch.Tensor,
                           mask: torch.Tensor | None,
                           weight_dict: dict,
                           nomask: bool):
    """In‑place Triton launch that fills `left` and `right`."""
    device = x.device
    B, N, _, C = x.shape
    H = left.shape[-1]

    # strides (elements)
    sx = x.stride()
    sl = left.stride()
    sr = right.stride()
    if not nomask:
        sm = mask.stride()
    else:
        # dummy strides – will never be used because NOMASK=True
        sm = (0, 0, 0)

    # weight pointers (half‑precision)
    wlp = weight_dict["left_proj.weight"]
    wlg = weight_dict["left_gate.weight"]
    wrp = weight_dict["right_proj.weight"]
    wrg = weight_dict["right_gate.weight"]

    # weight strides (C,H)
    sw = wlp.stride()                     # same for all four matrices
    stride_w_c, stride_w_h = sw[0], sw[1]

    BLOCK_I, BLOCK_J, BLOCK_H, BLOCK_C = 8, 8, 32, 64

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    proj_gate_mask_kernel[grid](
        x,
        left,
        right,
        mask if not nomask else left,   # dummy pointer safe when NOMASK=True
        B, N, C, H,
        sx[0], sx[1], sx[2], sx[3],
        sl[0], sl[1], sl[2], sl[3],
        sr[0], sr[1], sr[2], sr[3],
        sm[0], sm[1], sm[2],
        wlp, wlg, wrp, wrg,
        stride_w_c, stride_w_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_C=BLOCK_C,
        NOMASK=nomask,
        num_warps=4,
    )
    # left & right are now populated (in‑place)


# ----------------------------------------------------------------------
# Entry‑point: forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (FP32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (all FP32 in the checkpoint)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]    # (C, H)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]    # (C, H)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]     # (dim, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B,N,N,dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Half‑precision projection + gating (+ optional mask) via Triton
    # ------------------------------------------------------------------
    x_h = x.half()                           # FP16
    B, N, _, C = x_h.shape

    # allocate output tensors (FP16)
    left  = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # cast weights to FP16 once
    w_dict = {
        "left_proj.weight": left_proj_weight.half(),
        "left_gate.weight": left_gate_weight.half(),
        "right_proj.weight": right_proj_weight.half(),
        "right_gate.weight": right_gate_weight.half(),
    }

    mask_h = mask_tensor.to(dtype=torch.float16) if not nomask else None
    _proj_gate_mask_triton(x_h, left, right, mask_h, w_dict, nomask)

    # ------------------------------------------------------------------
    # 3️⃣ Core O(N³) update – batched half‑precision GEMM
    # ------------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()

    # fp16 matmul, Tensor‑Core accumulation → fp16 result (faster & less BW)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))   # (B,H,N,N) fp16
    out_mat = out_mat.to(torch.float32)        # cast to FP32 for the following LN

    # back to (B,N,N,H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Output LayerNorm, gating and final projection (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )   # FP32

    # output gate comes from the *normed* input `x`
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # (B,N,N,hidden_dim) FP32
    out = out * out_gate

    out = F.linear(out, to_out_weight)       # (B,N,N,dim) FP32
    return out
