import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1. Layer‑norm over the channel dimension (torch, FP32).
2. Linear projections → left/right tensors (FP32 → FP16).
   Compute three sigmoid gates (FP32 → FP16).
3. Fuse the hidden dimension into the batch dimension:
        [B, N, N, H] → [B·H, N, N]   (contiguous, FP16)
4. One Triton kernel multiplies the two projections with their
   sigmoid gates and (optionally) the pairwise mask, producing two
   gated tensors `left_gated` and `right_gated` (still FP16).
5. Batched matrix‑multiply (B·H independent GEMMs) using
   `torch.bmm`, which maps to cuBLAS/Tensor‑Core GEMM on H100.
6. Convert the result back to [B, N, N, H], apply the hidden‑dim
   LayerNorm, the output‑gate, and a final linear projection back
   to the original channel dimension (FP32).

Only the element‑wise gating (step 4) is written in Triton; the heavy
matrix multiplication re‑uses the highly‑optimized cuBLAS implementation,
giving sub‑millisecond runtimes on the provided configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask for BOTH sides
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    # input tensors
    a_ptr,  ag_ptr,            # left projection  & left gate   [B·H, N, N] (fp16)
    b_ptr,  bg_ptr,            # right projection & right gate   [B·H, N, N] (fp16)
    mask_ptr,                   # mask (broadcasted)            [B·H, N, N] (fp16) – ignored if HAS_MASK==0
    # output tensors
    a_out_ptr, b_out_ptr,      # gated left / gated right       [B·H, N, N] (fp16)
    # sizes
    M, N,                       # matrix dimensions (M=N=seq_len)
    # strides for the four input tensors
    stride_a_batch, stride_a_m, stride_a_n,
    stride_ag_batch, stride_ag_m, stride_ag_n,
    stride_b_batch, stride_b_m, stride_b_n,
    stride_bg_batch, stride_bg_m, stride_bg_n,
    # strides for the two outputs (identical layout to inputs)
    stride_aout_batch, stride_aout_m, stride_aout_n,
    stride_bout_batch, stride_bout_m, stride_bout_n,
    # mask strides (broadcasted across hidden dim)
    mask_batch_stride, mask_stride_m, mask_stride_n,
    # compile‑time constants
    H: tl.constexpr,                # hidden dimension (for mask mapping)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)            # batch*hidden index
    pid_m = tl.program_id(1)            # row block
    pid_n = tl.program_id(2)            # col block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --- base pointers for the current [N,N] matrix --------------------
    a_base   = a_ptr    + pid_b * stride_a_batch
    ag_base  = ag_ptr   + pid_b * stride_ag_batch
    b_base   = b_ptr    + pid_b * stride_b_batch
    bg_base  = bg_ptr   + pid_b * stride_bg_batch

    # --- load tiles ----------------------------------------------------
    a  = tl.load(a_base  + offs_m[:, None] * stride_a_m + offs_n[None, :] * stride_a_n,
                 mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    ag = tl.load(ag_base + offs_m[:, None] * stride_ag_m + offs_n[None, :] * stride_ag_n,
                 mask=mask_m[:, None] & mask_n[None, :], other=0.0)

    b  = tl.load(b_base  + offs_m[:, None] * stride_b_m + offs_n[None, :] * stride_b_n,
                 mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    bg = tl.load(bg_base + offs_m[:, None] * stride_bg_m + offs_n[None, :] * stride_bg_n,
                 mask=mask_m[:, None] & mask_n[None, :], other=0.0)

    left  = a  * ag
    right = b  * bg

    # --- optional mask (broadcasted across the hidden dimension) -----
    if HAS_MASK:
        # which original batch does this (B·H) index belong to?
        mask_batch = pid_b // H
        mask_base = mask_ptr + mask_batch * mask_batch_stride
        m = tl.load(mask_base + offs_m[:, None] * mask_stride_m + offs_n[None, :] * mask_stride_n,
                    mask=mask_m[:, None] & mask_n[None, :], other=0.0)
        left  = left  * m
        right = right * m

    # --- store results -------------------------------------------------
    tl.store(a_out_ptr + pid_b * stride_aout_batch
                     + offs_m[:, None] * stride_aout_m + offs_n[None, :] * stride_aout_n,
             left,
             mask=mask_m[:, None] & mask_n[None, :])

    tl.store(b_out_ptr + pid_b * stride_bout_batch
                     + offs_m[:, None] * stride_bout_m + offs_n[None, :] * stride_bout_n,
             right,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, C]   (float32, on CUDA)
            mask_tensor  : [B, N, N]      (bool/float, optional)
            weights_dict : model parameters (see reference module)
            config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        out : [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack ------------------------------------------------------------
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype       # expected torch.float32

    dim        = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]          # H
    eps        = 1e-5
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  (FP32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) + sigmoid gates
    # ------------------------------------------------------------------
    # proj → FP16 (to feed the GEMM later)
    left_proj  = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))       # keep FP32

    # ------------------------------------------------------------------
    # 3) Merge batch and hidden dimensions → [B·H, N, N]
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_batch = B * hidden_dim                              # B·H

    # reorder to [B, H, N, N] and make contiguous before view()
    left_proj  = left_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    left_gate  = left_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcasted across the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)                     # [B, N, N]
        # expand across the hidden dimension without copying
        mask_exp = mask_fp16.unsqueeze(1).expand(-1, hidden_dim, -1, -1)  # [B, H, N, N]
        mask_view = mask_exp.reshape(total_batch, N, N)                 # [B·H, N, N]
    else:
        # dummy tensor – will never be read because HAS_MASK==0
        mask_view = left_proj

    # ------------------------------------------------------------------
    # 5) Fuse projection, gate and mask (Triton) → gated tensors
    # ------------------------------------------------------------------
    left_gated  = torch.empty_like(left_proj, dtype=torch.float16, device=device)
    right_gated = torch.empty_like(right_proj, dtype=torch.float16, device=device)

    # block sizes – all test seq_len are multiples of 64, fallback to 32 for tiny N
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128

    grid = (
        total_batch,                 # one program per (batch*hidden) matrix
        triton.cdiv(N, BLOCK_M),    # row tiles
        triton.cdiv(N, BLOCK_N),    # col tiles
    )

    HAS_MASK = 0 if nomask else 1

    _fuse_proj_gate_mask_kernel[grid](
        # pointers
        a_ptr=left_proj,
        ag_ptr=left_gate,
        b_ptr=right_proj,
        bg_ptr=right_gate,
        mask_ptr=mask_view,
        a_out_ptr=left_gated,
        b_out_ptr=right_gated,
        # sizes
        M=N,
        N=N,
        # strides for left projection / left gate
        stride_a_batch=left_proj.stride(0), stride_a_m=left_proj.stride(1), stride_a_n=left_proj.stride(2),
        stride_ag_batch=left_gate.stride(0), stride_ag_m=left_gate.stride(1), stride_ag_n=left_gate.stride(2),
        # strides for right projection / right gate
        stride_b_batch=right_proj.stride(0), stride_b_m=right_proj.stride(1), stride_b_n=right_proj.stride(2),
        stride_bg_batch=right_gate.stride(0), stride_bg_m=right_gate.stride(1), stride_bg_n=right_gate.stride(2),
        # strides for outputs (identical layout)
        stride_aout_batch=left_gated.stride(0), stride_aout_m=left_gated.stride(1), stride_aout_n=left_gated.stride(2),
        stride_bout_batch=right_gated.stride(0), stride_bout_m=right_gated.stride(1), stride_bout_n=right_gated.stride(2),
        # mask strides (same layout as the projection tensors)
        mask_batch_stride=mask_view.stride(0),
        mask_stride_m=mask_view.stride(1),
        mask_stride_n=mask_view.stride(2),
        # compile‑time constants
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 6) Batched matrix multiplication (cuBLAS / Tensor‑Core)
    # ------------------------------------------------------------------
    # right tensor must be transposed for the GEMM
    right_gated_t = right_gated.transpose(1, 2)          # [B·H, N, N]
    out_fp16 = torch.bmm(left_gated, right_gated_t)     # [B·H, N, N]  (FP16, acc in FP32)

    # ------------------------------------------------------------------
    # 7) Reshape back to [B, N, N, H] (FP32) and final layers
    # ------------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)                        # [B, N, N, H]
        .contiguous()
        .to(dtype)                                  # FP32
    )

    # hidden‑dim LayerNorm + out‑gate
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # final linear projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
