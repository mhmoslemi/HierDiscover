"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (forward only)

Algorithm
---------
1. Layer‑norm over the input channel dimension (FP32).
2. Linear projections (dim → hidden_dim) and three sigmoid gates.
   Projections are cast to FP16, gates stay FP16 except the final out_gate
   (kept in FP32).
3. Collapse the batch and hidden dimensions → [B·H, N, N] (contiguous FP16).
4. Triton kernel fuses:
      * element‑wise multiplication of projection × gate,
      * optional broadcasting mask,
   producing the gated tensors `left_gated` and `right_gated`.
5. Batched matrix‑multiply (cuBLAS/Tensor‑Core) on the gated tensors:
      out_fp16 = left_gated @ right_gatedᵀ   (per (B·H) matrix)
6. Reshape back to [B, N, N, H] (FP32), apply hidden‑dim LayerNorm,
   multiply by the out‑gate, and a final linear projection to the original
   channel dimension.
Only the element‑wise fusion (step 4) is written in Triton; the heavy GEMM
uses the highly‑optimised cuBLAS implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and (optional) mask for BOTH sides
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    # input tensors
    a_ptr, ag_ptr,            # left projection & left gate   [B·H, N, N] (fp16)
    b_ptr, bg_ptr,            # right projection & right gate [B·H, N, N] (fp16)
    mask_ptr,                 # mask (broadcasted)            [B·H, N, N] (fp16) – ignored if HAS_MASK==0
    # output tensors
    a_out_ptr, b_out_ptr,    # gated left / gated right       [B·H, N, N] (fp16)
    # matrix sizes
    M, N,                     # N == M == seq_len
    # strides for the four input tensors
    stride_a_batch, stride_a_m, stride_a_n,
    stride_ag_batch, stride_ag_m, stride_ag_n,
    stride_b_batch, stride_b_m, stride_b_n,
    stride_bg_batch, stride_bg_m, stride_bg_n,
    # strides for the two outputs (identical layout to inputs)
    stride_aout_batch, stride_aout_m, stride_aout_n,
    stride_bout_batch, stride_bout_m, stride_bout_n,
    # mask strides (broadcasted across hidden dim)
    mask_batch_stride, mask_stride_m, mask_stride_n,
    # compile‑time constants
    H: tl.constexpr,                # hidden dimension (for mask mapping)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)            # batch*hidden index
    pid_m = tl.program_id(1)            # row block
    pid_n = tl.program_id(2)            # col block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Load tiles of the four tensors
    # ------------------------------------------------------------------
    a = tl.load(
        a_ptr + pid_b * stride_a_batch
               + offs_m[:, None] * stride_a_m
               + offs_n[None, :] * stride_a_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    ag = tl.load(
        ag_ptr + pid_b * stride_ag_batch
                + offs_m[:, None] * stride_ag_m
                + offs_n[None, :] * stride_ag_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    b = tl.load(
        b_ptr + pid_b * stride_b_batch
               + offs_m[:, None] * stride_b_m
               + offs_n[None, :] * stride_b_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    bg = tl.load(
        bg_ptr + pid_b * stride_bg_batch
                + offs_m[:, None] * stride_bg_m
                + offs_n[None, :] * stride_bg_n,
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left  = a * ag
    right = b * bg

    # ------------------------------------------------------------------
    # Optional mask (broadcasted across the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # map the (B·H) index back to the original batch
        mask_batch = pid_b // H
        mask_base = mask_ptr + mask_batch * mask_batch_stride
        m = tl.load(
            mask_base
            + offs_m[:, None] * mask_stride_m
            + offs_n[None, :] * mask_stride_n,
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Store gated results
    # ------------------------------------------------------------------
    tl.store(
        a_out_ptr + pid_b * stride_aout_batch
                  + offs_m[:, None] * stride_aout_m
                  + offs_n[None, :] * stride_aout_n,
        left,
        mask=mask_m[:, None] & mask_n[None, :],
    )
    tl.store(
        b_out_ptr + pid_b * stride_bout_batch
                  + offs_m[:, None] * stride_bout_m
                  + offs_n[None, :] * stride_bout_n,
        right,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, C]   (float32, on CUDA)
            mask_tensor  : [B, N, N]      (bool/float, optional)
            weights_dict : model parameters (same keys as in the reference module)
            config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        out : [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype                     # expected torch.float32

    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Layer‑norm over the channel dimension (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C] (FP32)

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) + sigmoid gates
    # --------------------------------------------------------------
    # All linear ops are carried out in FP32 then cast to FP16 for the
    # subsequent batched GEMM – this is the fastest path on H100.
    left_proj = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))      # FP32

    # --------------------------------------------------------------
    # 3) Collapse batch & hidden dimensions → [B·H, N, N]
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_batch = B * hidden_dim                     # B·H

    # Shape -> [B, H, N, N] then view as [B·H, N, N]
    left_proj  = left_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    left_gate  = left_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)

    # --------------------------------------------------------------
    # 4) Optional mask (broadcasted across hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N]  →  [B·H, N, N] (broadcast)
        mask_fp16 = mask_tensor.to(torch.float16)
        mask_exp = mask_fp16.unsqueeze(1).expand(-1, hidden_dim, -1, -1)   # [B, H, N, N]
        mask_view = mask_exp.reshape(total_batch, N, N)
    else:
        # dummy placeholder – never read when HAS_MASK==0
        mask_view = left_proj

    # --------------------------------------------------------------
    # 5) Fuse projection, gate and mask (Triton)
    # --------------------------------------------------------------
    left_gated  = torch.empty_like(left_proj, dtype=torch.float16, device=device)
    right_gated = torch.empty_like(right_proj, dtype=torch.float16, device=device)

    # Choose block sizes (all test seq_len are multiples of 64)
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128

    grid = (
        total_batch,                # batch*hidden index
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
    )

    HAS_MASK = 0 if nomask else 1

    _fuse_proj_gate_mask_kernel[grid](
        # pointers
        a_ptr=left_proj,
        ag_ptr=left_gate,
        b_ptr=right_proj,
        bg_ptr=right_gate,
        mask_ptr=mask_view,
        a_out_ptr=left_gated,
        b_out_ptr=right_gated,
        # sizes
        M=N,
        N=N,
        # strides for left projection / left gate
        stride_a_batch=left_proj.stride(0), stride_a_m=left_proj.stride(1), stride_a_n=left_proj.stride(2),
        stride_ag_batch=left_gate.stride(0), stride_ag_m=left_gate.stride(1), stride_ag_n=left_gate.stride(2),
        # strides for right projection / right gate
        stride_b_batch=right_proj.stride(0), stride_b_m=right_proj.stride(1), stride_b_n=right_proj.stride(2),
        stride_bg_batch=right_gate.stride(0), stride_bg_m=right_gate.stride(1), stride_bg_n=right_gate.stride(2),
        # strides for outputs (identical layout)
        stride_aout_batch=left_gated.stride(0), stride_aout_m=left_gated.stride(1), stride_aout_n=left_gated.stride(2),
        stride_bout_batch=right_gated.stride(0), stride_bout_m=right_gated.stride(1), stride_bout_n=right_gated.stride(2),
        # mask strides
        mask_batch_stride=mask_view.stride(0),
        mask_stride_m=mask_view.stride(1),
        mask_stride_n=mask_view.stride(2),
        # compile‑time constants
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # --------------------------------------------------------------
    # 6) Batched matrix multiplication (cuBLAS/Tensor‑Core)
    # --------------------------------------------------------------
    # right_gatedᵀ is needed because the einsum does "left[i,k] * right[j,k]"
    right_gated_t = right_gated.transpose(1, 2)       # [B·H, N, N]
    out_fp16 = torch.bmm(left_gated, right_gated_t)  # [B·H, N, N], acc in FP32

    # --------------------------------------------------------------
    # 7) Reshape back & final post‑processing
    # --------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)   # [B, H, N, N]
        .permute(0, 2, 3, 1)                # [B, N, N, H]
        .contiguous()
        .to(dtype)                          # back to FP32
    )

    # hidden‑dim LayerNorm + out‑gate
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # final projection back to original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
