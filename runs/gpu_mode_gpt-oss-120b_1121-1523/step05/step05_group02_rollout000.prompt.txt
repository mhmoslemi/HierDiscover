```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
TriMul – “outgoing” version (AlphaFold‑3) implemented in Triton.

Algorithm
---------
1. Layer‑norm over the input channel dimension.
2. Fuse the linear projection and its gate for the left and right sides:
        left = (x @ W_left) * sigmoid(x @ W_left_gate)
        right = (x @ W_right) * sigmoid(x @ W_right_gate)
   (weights are concatenated so that each side needs only ONE matmul.)
3. Optional mask is applied (broadcast over the hidden dimension).
4. Core O(N³) operation:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM L @ Rᵀ for every hidden channel.  The heavy
   computation is performed by a Triton kernel that streams the A and Bᵀ
   tiles through shared memory and uses Tensor‑Core friendly fp16 × fp16 → fp32.
5. Layer‑norm over the hidden dimension, multiplication by the output‑gate,
   and a final linear projection back to the original channel size.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for all (batch, hidden) matrices.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,                    # A strides (row‑major within a matrix)
    stride_bk, stride_bn,                    # Bᵀ strides
    stride_cm, stride_cn,                    # C strides (row‑major)
    stride_abatch, stride_ah,                # batch‑ and hidden‑strides for A
    stride_bbatch, stride_bh,                # batch‑ and hidden‑strides for Bᵀ
    stride_cbatch,                           # stride between consecutive output matrices
    H,                                       # number of hidden channels (runtime scalar)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # 0‑D grid layout :
    #   axis 0 : (batch * hidden) matrix index
    #   axis 1 : tile row   (M / BLOCK_M)
    #   axis 2 : tile col   (N / BLOCK_N)
    # ------------------------------------------------------------------
    pid = tl.program_id(0)            # linear index over (batch*hidden)
    pid_m = tl.program_id(1)          # tile‑row index
    pid_n = tl.program_id(2)          # tile‑col index

    # Recover batch and hidden indices from the flattened pid
    pid_batch = pid // H
    pid_hidden = pid - pid_batch * H   # same as pid % H but without % operator

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for the current (batch, hidden) matrix
    a_base = a_ptr + pid_batch * stride_abatch + pid_hidden * stride_ah
    b_base = b_ptr + pid_batch * stride_bbatch + pid_hidden * stride_bh

    # Accumulator (FP32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)               # number of K‑tiles
    for k in range(num_k):
        k_off = k * BLOCK_K

        # Pointers to the current K‑tile of A and Bᵀ
        a_tile_ptr = a_base + offs_m[:, None] * stride_am + (k_off + offs_k)[None, :] * stride_ak
        b_tile_ptr = b_base + (k_off + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # OOB guards
        mask_a = (offs_m[:, None] < M) & ((k_off + offs_k)[None, :] < K)
        mask_b = ((k_off + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_tile_ptr, mask=mask_a, other=0.0)
        b = tl.load(b_tile_ptr, mask=mask_b, other=0.0)

        # fp16 × fp16 → fp32 accumulation (Tensor‑Core friendly)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write the result back – C is laid out as (total, M, N) contiguous
    c_base = c_ptr + pid * stride_cbatch
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc, mask=mask_c)


def triton_batched_matmul(A: torch.Tensor, B_T: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM (A @ B_T) where A and B_T have shape (B, H, N, N) –
    each hidden channel of each batch is a separate matrix.
    Returns C of shape (B*H, N, N) in torch.float32.
    """
    assert A.is_cuda and B_T.is_cuda
    B, H, M, K = A.shape               # M == K == N
    _, _, K2, N = B_T.shape
    assert K == K2, "Inner dimensions must match"

    total = B * H                       # total number of independent matrices

    # Output tensor (float32)
    C = torch.empty((total, M, N), dtype=torch.float32, device=A.device)

    # Heuristic block sizes – trade‑off between occupancy and shared‑memory usage
    if M <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (total,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Strides (in elements, not bytes)
    stride_am, stride_ak = A.stride(2), A.stride(3)
    stride_bk, stride_bn = B_T.stride(2), B_T.stride(3)
    stride_cm, stride_cn = C.stride(1), C.stride(2)

    stride_abatch, stride_ah = A.stride(0), A.stride(1)
    stride_bbatch, stride_bh = B_T.stride(0), B_T.stride(1)
    stride_cbatch = C.stride(0)

    batched_gemm_kernel[grid](
        A, B_T, C,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_abatch, stride_ah,
        stride_bbatch, stride_bh,
        stride_cbatch,
        H,                                 # runtime scalar hidden‑channel count
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the “outgoing” TriMul operator.
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns a torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Fuse projection + gate for the left side
    #    (weight concatenation saves one matmul per side)
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)   # (2*H, dim)
    left_comb = F.linear(x, left_comb_weight)                                 # [B, N, N, 2*H]
    left = left_comb[..., :hidden_dim]                                        # projection
    left_gate = left_comb[..., hidden_dim:].sigmoid()                         # gate
    left = (left * left_gate).to(torch.float16)                               # FP16 for the GEMM

    # ------------------------------------------------------------------
    # 3) Fuse projection + gate for the right side
    # ------------------------------------------------------------------
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)
    right_comb = F.linear(x, right_comb_weight)
    right = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Optional mask (applied after gating – mathematically equivalent)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5) Core TriMul – batched GEMM L @ Rᵀ using Triton
    # ------------------------------------------------------------------
    # Rearrange to (batch, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2)        # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)      # [B, H, N, N]
    right_T = right_perm.transpose(-2, -1)    # transpose last two dims → Rᵀ

    # Batched GEMM → (B*H, N, N) float32
    out_mat = triton_batched_matmul(left_perm, right_T)   # shape (B*H, N, N)

    # Restore shape [B, N, N, H]
    B, N, _, _ = input_tensor.shape
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_gate = F.linear(x, out_gate_weight).sigmoid()     # [B, N, N, H]
    out = out * out_gate
    out = F.linear(out, to_out_weight)                    # project back to `dim`
    return out
```
