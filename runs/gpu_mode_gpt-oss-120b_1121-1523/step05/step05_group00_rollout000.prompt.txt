import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel - Triton‑accelerated “outgoing” TriMul (Triangle Multiplicative Update)

Algorithm
---------
1️⃣ Layer‑norm the input tensor (shape [B,N,N,C]).
2️⃣ Linear projections → left/right (shape [B,N,N,H]) and three sigmoid gates.
3️⃣ Apply the optional pairwise mask and the left/right gates.
4️⃣ Core O(N³·H) step:
   for each hidden index h (and batch b) compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is exactly a batched matrix multiplication
        out_h = left_h @ right_hᵀ .
   A custom Triton GEMM kernel (block‑wise, fp16 arithmetic, fp32 accumulation)
   is used to compute the whole batch × hidden dimension in one launch.
5️⃣ Layer‑norm on the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to the original channel size C.
6️⃣ Return the result (float32, shape [B,N,N,C]).

The Triton kernel handles arbitrary batch·hidden sizes (≤ 2·384) and
sequence lengths that are multiples of 64 (the block sizes chosen below).
All other operations stay in PyTorch; only the expensive N³ reduction is
offloaded to Triton.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ  (fp16 inputs → fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def _trmul_gemm_kernel(
    a_ptr,                     # left  [B·H, N, N]  (fp16)
    b_ptr,                     # right [B·H, N, N]  (fp16)
    c_ptr,                     # out   [B·H, N, N]  (fp16)
    M, N, K,                   # all equal to seq_len
    stride_am, stride_ak,      # row/col strides of A
    stride_bn, stride_bk,      # row/col strides of B (original layout)
    stride_cm, stride_cn,      # row/col strides of C
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block index along M (i dimension)
    pid_n = tl.program_id(1)          # block index along N (j dimension)
    pid_b = tl.program_id(2)          # batch*hidden index

    # Pointers to the current batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # Offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 (to avoid overflow)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----------- Load A block (M × K) ----------
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----------- Load Bᵀ block (K × N) ----------
        # Bᵀ[k, n] == B[n, k] in the original layout
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ----------- Dot product ----------
        acc += tl.dot(a, b)

    # ----------- Store result ----------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel for
    the cubic reduction.  The function expects

        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    where
        input_tensor : [B, N, N, C]  (float32)
        mask_tensor  : [B, N, N]    (bool/float, may be ignored)
        weights_dict : contains all linear / LN parameters
        config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑norm on the channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → left / right  (shape [B,N,N,H])
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (multiply before gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask may be bool/int – cast to same dtype as x and broadcast
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates from the *original* normalized input
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy O(N³·H) reduction via Triton
    #    Convert to [B·H, N, N] fp16 for the kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # Bring hidden dimension into the leading axis for easy batching
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    total_batch = B * hidden_dim

    left_fp16 = left.view(total_batch, N, N).to(torch.float16)
    right_fp16 = right.view(total_batch, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # Choose a reasonable block size (all test seq_len are multiples of 64)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),        # number of blocks along M
        triton.cdiv(N, BLOCK_N),        # number of blocks along N
        total_batch,                    # batch*hidden dimension
    )

    _trmul_gemm_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_fp16,
        M=N,
        N=N,
        K=N,
        stride_am=left_fp16.stride(1),   # row stride within a matrix
        stride_ak=left_fp16.stride(2),   # col stride (reduction)
        stride_bn=right_fp16.stride(1),  # row stride of B (original)
        stride_bk=right_fp16.stride(2),  # col stride of B (original)
        stride_cm=out_fp16.stride(1),
        stride_cn=out_fp16.stride(2),
        stride_ab=left_fp16.stride(0),   # batch stride
        stride_bb=right_fp16.stride(0),
        stride_cb=out_fp16.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] (float32 for the remaining ops)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)                     # fp16 → fp32

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
