"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) accelerated with Triton

Algorithm
---------
1️⃣ Layer‑norm the input tensor x ∈ [B,N,N,C] (float32).
2️⃣ Cast x to fp16 and perform the four linear projections (left‑proj / left‑gate
   and right‑proj / right‑gate) in **two** fused matmuls:
       [W_left;W_left_gate]  →  [left , left_gate]
       [W_right;W_right_gate] → [right, right_gate]
   (weights are cast to fp16 as well).  Sigmoid is applied to the gate tensors.
3️⃣ Apply the optional pairwise mask and element‑wise gating:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
4️⃣ Core O(N³·H) update: for each hidden index h we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix multiplication left @ rightᵀ.
   A custom Triton kernel (_trmul_gemm_kernel) computes all
   B·H matrices in a single launch.  The kernel works on fp16 data,
   uses Tensor‑Core‑accelerated `tl.dot(..., out_dtype=tl.float32)` and
   handles arbitrary (non‑multiple‑of‑64) sequence lengths via masking.
5️⃣ Convert the result back to fp32, apply a hidden‑dim LayerNorm,
   multiply by the out‑gate (computed from the original normalized x),
   and finally a linear projection back to C channels.
6️⃣ Return the fp32 tensor of shape [B,N,N,C].

All heavy tensor work (the N³ reduction) runs on the GPU in Triton;
the remaining ops stay in PyTorch but are heavily fused and run in fp16
where possible, yielding a large speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ
#   A : [B·H, N, N]  (fp16)
#   B : [B·H, N, N]  (fp16)
#   C : [B·H, N, N]  (fp16)  ←  result
# ----------------------------------------------------------------------
@triton.jit
def _trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_am, stride_ak,               # A row / col strides
    stride_bn, stride_bk,               # B row / col strides (original layout)
    stride_cm, stride_cn,               # C row / col strides
    stride_ab, stride_bb, stride_cb,    # batch stride for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch*hidden index

    # Pointers to the current batch slice
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K) – fp16 → keep fp16 for TensorCore
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0))

        # Load Bᵀ tile (K × N) – note the swap of strides (k,n)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0))

        # Accumulate with Tensor‑Core‑enabled fp16 dot → fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result as fp16
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor[B, N, N, C] (float32)
        mask_tensor  : torch.Tensor[B, N, N]  (bool/float, optional)
        weights_dict : dict with all linear / LN parameters (float32)
        config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype_fp32 = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  float32

    # --------------------------------------------------------------
    # 2) Cast to fp16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x_norm.half()                     # [B, N, N, C]  fp16

    # --------------------------------------------------------------
    # 3) Fused left projection + left gate (shape: 2*H)
    # --------------------------------------------------------------
    w_left_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)   # [(2*H), C]

    left_cat = F.linear(x_fp16, w_left_cat)    # [B, N, N, 2*H]  fp16
    left = left_cat[..., :hidden_dim]          # proj
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])  # gate (fp16)

    # --------------------------------------------------------------
    # 4) Fused right projection + right gate (shape: 2*H)
    # --------------------------------------------------------------
    w_right_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)

    right_cat = F.linear(x_fp16, w_right_cat)  # [B, N, N, 2*H]  fp16
    right = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # --------------------------------------------------------------
    # 5) Apply gates (and optional mask) – all still fp16
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if use_mask:
        # mask shape [B, N, N] → broadcast over hidden dim
        mask = mask_tensor.to(dtype=torch.float16, device=device).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 6) Prepare tensors for the Triton GEMM kernel
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # Move hidden dimension to the leading axis: [B, H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    total_batch = B * hidden_dim                # B·H
    left_fp16 = left.view(total_batch, N, N)    # [B·H, N, N]  fp16
    right_fp16 = right.view(total_batch, N, N)  # [B·H, N, N]  fp16

    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7) Triton launch – choose block sizes based on seq_len
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = BLOCK_K = 128

    grid = (
        triton.cdiv(N, BLOCK_M),        # blocks along i
        triton.cdiv(N, BLOCK_N),        # blocks along j
        total_batch,                    # batch*hidden
    )

    _trmul_gemm_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_fp16,
        M=N,
        N=N,
        K=N,
        stride_am=left_fp16.stride(1),          # row stride of A
        stride_ak=left_fp16.stride(2),          # col stride of A
        stride_bn=right_fp16.stride(1),         # row stride of B (original)
        stride_bk=right_fp16.stride(2),         # col stride of B (original)
        stride_cm=out_fp16.stride(1),
        stride_cn=out_fp16.stride(2),
        stride_ab=left_fp16.stride(0),   # batch stride A
        stride_bb=right_fp16.stride(0),  # batch stride B
        stride_cb=out_fp16.stride(0),    # batch stride C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, H] and convert to fp32 for the final steps
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype_fp32)                     # fp32 for LN & final linear

    # ------------------------------------------------------------------
    # 9) Hidden‑dim LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 10) Out‑gate (computed from the original normalized x, float32)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,H]
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to C (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,C]

    return out
