@triton.jit
def _fused_proj_gate_mul_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    N_ELEM: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
    MASK_ENABLED: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    idx = offset + tl.arange(0, BLOCK_SIZE)
    mask = idx < N_ELEM
    a = tl.load(proj_ptr + idx, mask=mask, other=0.0)
    b = tl.load(gate_ptr + idx, mask=mask, other=0.0)
    if MASK_ENABLED:
        m = tl.load(mask_ptr + idx, mask=mask, other=1.0)
        res = a * b * m
    else:
        res = a * b
    tl.store(out_ptr + idx, res, mask=mask)

Highly‑optimised outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input (FP16).
2. Linear projections → left/right (FP16).
3. Linear gates → left/right/out (FP16, sigmoid).
4. Fuse (proj * gate * optional mask) in‑place with a tiny Triton kernel

5. Heavy reduction: a batched GEMM over (B·H) matrices of size N×N.

   Tensor‑Core FP16×FP16→FP32 and is far faster than a naïve Triton loop.
6. Layer‑norm over the hidden dimension, apply the output gate and a final

7. Cast the result to FP32 (the model output type).

def _fused_proj_gate_mask_kernel(
    proj_ptr,            # *fp16   – projection tensor (input & output)
    gate_ptr,            # *fp16   – sigmoid gate
    mask_ptr,            # *fp16   – optional mask (0/1), may be nullptr

    HAS_MASK: tl.constexpr,        # compile‑time flag (0/1)
):

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]  (float32)
        - mask         : torch.Tensor   [B, N, N]      (bool or float)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys
            * "dim"         – input feature dimension
            * "hidden_dim" – internal hidden dimension (c_z)
            * "nomask"      – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Model weights (cast to the compute dtype – fp16)
    # --------------------------------------------------------------
    dtype = torch.float16
    # Layer‑norm weights
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)

    # Linear projections
    left_proj_w  = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    # Gates
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    # Output side
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    to_out_w      = weights["to_out.weight"].to(torch.float32)   # final proj stays FP32

    # --------------------------------------------------------------
    # 1) Input layer‑norm (FP16)
    # --------------------------------------------------------------
    # Cast input to fp16 for the whole pipeline
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    ).to(dtype)                      # [B, N, N, dim]  fp16

    # --------------------------------------------------------------
    # 2) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # [B, N, N, H]
    right_proj = F.linear(x, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3) Gates (sigmoid, FP16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H]

    # --------------------------------------------------------------
    # 4) Fuse (proj * gate * optional mask) in‑place via Triton
    # --------------------------------------------------------------
    total_elems = left_proj.numel()
    BLOCK_SIZE = 16384                     # good trade‑off for large tensors
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    if nomask:
        # No mask – simple proj*gate
        _fused_proj_gate_mask_kernel[grid](
            left_proj,
            left_gate,
            left_proj,               # dummy pointer – never read when HAS_MASK=0
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=0,
        )
        _fused_proj_gate_mask_kernel[grid](
            right_proj,
            right_gate,
            right_proj,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=0,
        )
    else:
        # Prepare mask in FP16 and broadcast over hidden dim
        mask_fp16 = mask.to(dtype).unsqueeze(-1)                # [B,N,N,1]
        # Ensure contiguous layout for the kernel
        mask_fp16 = mask_fp16.contiguous()

        _fused_proj_gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=1,
        )
        _fused_proj_gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=1,
        )

    # --------------------------------------------------------------
    # 5) Heavy reduction: batched GEMM (B·H matrices of size N×N)
    # --------------------------------------------------------------
    B, N, _, H = left_proj.shape          # left_proj == right_proj == [B,N,N,H]

    # Rearrange to (B*H, N, N) for torch.bmm
    left_mat  = left_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B*H, N, N]
    # right needs a transpose on the last two dimensions (k ↔ j)
    right_mat = right_proj.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # [B*H, N, N]

    # Batched GEMM – FP16 inputs, FP32 accumulation (Tensor‑Core on H100)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N]  dtype=fp16 → fp32 internally

    # Reshape back to [B, N, N, H] and keep FP16 for the next layer‑norm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # --------------------------------------------------------------
    # 6) Output layer‑norm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                           # still FP16

    # Apply the output gate (still FP16)
    out = out * out_gate

    # Final linear projection back to `dim` (output must be FP32)
    out = F.linear(out, to_out_w)   # FP32 result

    return out
