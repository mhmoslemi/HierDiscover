@triton.jit
def _mul_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, N_ELEM, H, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_raw = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask value: broadcast across H
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    # sigmoid
    sig = 1.0 / (1.0 + tl.exp(-gate_raw))
    out = proj * sig * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

"""
Highly‑optimised forward pass for the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input (FP32).
2. Cast to FP16 and perform **one** large GEMM that computes all five
   linear projections/gates (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) in a single `torch.nn.functional.linear`.
3. Split the result into the five tensors.
4. Fuse “projection × sigmoid(gate) × optional mask” with a tiny Triton
   element‑wise kernel – this removes two separate element‑wise passes.
5. Heavy O(N³) contraction: for each hidden channel we need a matrix
   product `left[..., i,k] @ right[..., k,j]`.  This is performed as a
   **single strided‑batched GEMM** (`torch.bmm`) on tensors of shape
   `[B*H, N, N]`.
6. Layer‑norm over the hidden dimension, apply the output gate and the
   final linear projection back to `dim`.
7. Return the FP32 result.

All heavy arithmetic is done in FP16 (Tensor‑Core friendly) while the
few normalisation / gating steps stay in FP32 for stability.  The Triton
kernel is mandatory (step 4) and is written to broadcast the optional
mask without materialising an expanded mask tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _mul_gate_mask_kernel(
    proj_ptr,          # *fp16  – projection tensor (contiguous)
    gate_ptr,          # *fp16  – raw gate tensor (pre‑sigmoid)
    mask_ptr,          # *fp16  – mask tensor of shape [B*N*N] (0/1)
    out_ptr,           # *fp16  – destination
    N_ELEM: tl.constexpr,   # total number of elements (=B*N*N*H)
    H: tl.constexpr,        # hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    m = offs < N_ELEM

    proj = tl.load(proj_ptr + offs, mask=m, other=0.0)
    gate_raw = tl.load(gate_ptr + offs, mask=m, other=0.0)

    # broadcast mask: each hidden element shares the same mask entry
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=1.0)

    # sigmoid in fp16
    sig = 1.0 / (1.0 + tl.exp(-gate_raw))

    out = proj * sig * mask_val
    tl.store(out_ptr + offs, out, mask=m)


@triton.jit
def _mul_gate_kernel(
    proj_ptr,
    gate_ptr,
    out_ptr,
    N_ELEM: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    m = offs < N_ELEM

    proj = tl.load(proj_ptr + offs, mask=m, other=0.0)
    gate_raw = tl.load(gate_ptr + offs, mask=m, other=0.0)

    sig = 1.0 / (1.0 + tl.exp(-gate_raw))
    out = proj * sig
    tl.store(out_ptr + offs, out, mask=m)


# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        - mask         : torch.Tensor   [B, N, N]        (bool/float)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – hidden dimension used internally
            * "nomask"     – if True, the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Grab weight tensors (FP32) and cast to the precision we need
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]    # (dim, hidden)

    # --------------------------------------------------------------
    # 1) Input layer‑norm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_weight,
                     bias=norm_bias, eps=1e-5)          # [B, N, N, dim] FP32

    # --------------------------------------------------------------
    # 2) One‑shot linear that yields all five intermediate tensors
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # concatenate the five weight matrices (out_features stacked)
    concat_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w], dim=0
    ).to(torch.float16)                                   # (5*hidden, dim)

    # single GEMM: [B*N*N, dim] @ [dim, 5*hidden]^T -> [B*N*N, 5*hidden]
    all_inter = F.linear(x_fp16, concat_w)                # [B, N, N, 5*hidden] FP16

    # --------------------------------------------------------------
    # 3) Split into the five components
    # --------------------------------------------------------------
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        all_inter, hidden_dim, dim=-1
    )    # each: [B, N, N, hidden] FP16

    # --------------------------------------------------------------
    # 4) Fuse proj × sigmoid(gate) × optional mask with Triton
    # --------------------------------------------------------------
    total_elems = left_proj_raw.numel()          # B * N * N * hidden
    BLOCK_SIZE = 1024
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    left  = torch.empty_like(left_proj_raw)      # [B, N, N, hidden] FP16
    right = torch.empty_like(right_proj_raw)

    if not nomask:
        # mask → FP16, flattened (B*N*N,)
        mask_fp16 = mask.to(torch.float16).contiguous().flatten()
        # left = left_proj * sigmoid(left_gate) * mask
        _mul_gate_mask_kernel[grid](
            left_proj_raw,
            left_gate_raw,
            mask_fp16,
            left,
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right = right_proj * sigmoid(right_gate) * mask
        _mul_gate_mask_kernel[grid](
            right_proj_raw,
            right_gate_raw,
            mask_fp16,
            right,
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        _mul_gate_kernel[grid](
            left_proj_raw,
            left_gate_raw,
            left,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        _mul_gate_kernel[grid](
            right_proj_raw,
            right_gate_raw,
            right,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
        )

    # --------------------------------------------------------------
    # 5) Heavy O(N³) reduction: batched GEMM over (B * hidden) matrices
    # --------------------------------------------------------------
    # reshape to [B, hidden, N, N] (left) and [B, hidden, N, N] (rightᵀ)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k as rows)

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    left_mat  = left_perm.view(batch_h, N, N)   # [B*H, N, N]  (FP16)
    right_mat = right_perm.view(batch_h, N, N)  # [B*H, N, N]  (FP16)

    # Strided‑batched GEMM – FP16 inputs, FP32 accumulation (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)    # [B*H, N, N]  (FP32)

    # restore original layout: [B, N, N, hidden]
    out = (
        out_mat.view(B, H, N, N)               # [B, H, N, N]
        .permute(0, 2, 3, 1)                   # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # --------------------------------------------------------------
    # 6) Output norm, output‑gate and final projection
    # --------------------------------------------------------------
    # output gate (sigmoid, FP32)
    out_gate = torch.sigmoid(out_gate_raw.to(torch.float32))

    # hidden‑dim layer‑norm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # apply output gate
    out = out * out_gate

    # final linear back to `dim` (FP32)
    out = F.linear(out, to_out_w)   # [B, N, N, dim] FP32

    return out
