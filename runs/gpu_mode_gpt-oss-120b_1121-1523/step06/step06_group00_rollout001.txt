"""
TriMul (outgoing) – highly‑optimized Triton‑aware forward pass
================================================================

Algorithm (all heavy arithmetic in fp16, final output fp32)
---------------------------------------------------------
1. Layer‑norm on the input tensor (float32).
2. Fused projection + gate for the left and right branches:
      * Concatenate proj‑ and gate‑weights → a single linear layer.
      * Split the 2·H output, apply sigmoid to the gate part and
        multiply with the proj part → tensor `[B, N, N, H]` (fp16).
3. Optional mask:
      * A tiled Triton kernel processes a `BLOCK_I × BLOCK_J` tile of
        the `(i,j)` plane at once and multiplies the whole hidden
        dimension (H≤128) by `mask[b,i,j]`.  This reduces the launch
        count from `B·N·N` to `B·⌈N/BLOCK_I⌉·⌈N/BLOCK_J⌉` (≈1 k blocks
        for N=1024) while keeping the work per block fully
        register‑resident.
4. Core O(N³) update – for each hidden channel we need
      `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is exactly a batch of `B·H` independent matrix multiplies
   (`N×N` × `N×Nᵀ`).  We reshape to `(B*H, N, N)` and call
   `torch.bmm`, which uses cuBLAS/H100 Tensor‑Core GEMM with
   fp16 inputs / fp32 accumulation.
5. Output layer‑norm, output‑gate (sigmoid) and final linear back to
   the original channel dimension (`dim`).  The final tensor is
   float32.

All kernels are written for Triton 3.3.1 and tested on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled mask application (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    left_ptr,            # fp16 *
    right_ptr,           # fp16 *
    mask_ptr,            # fp16 *
    B, N, H,             # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Multiply left and right tensors by mask[b,i,j] (broadcast over H).

    One program instance processes a `BLOCK_I × BLOCK_J` tile of the
    `(i,j)` plane and the whole hidden dimension (H≤BLOCK_H) in registers.
    """
    b  = tl.program_id(0)
    ti = tl.program_id(1)
    tj = tl.program_id(2)

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    # ------------------------------------------------------------------
    # Index vectors for the tile
    # ------------------------------------------------------------------
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # ------------------------------------------------------------------
    # Load mask (scalar per (i,j)) and broadcast over H
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]                     # (BLOCK_I, BLOCK_J, 1)

    # ------------------------------------------------------------------
    # Load a tile of left & right, apply mask, store back
    # ------------------------------------------------------------------
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )

    left = left * mask
    right = right * mask

    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """In‑place half‑precision mask multiplication via a tiled Triton kernel."""
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    # ensure mask is half‑precision
    mask_h = mask.to(left.dtype)

    # strides (in elements)
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_h.stride()

    # tile sizes – chosen to give ~1 k blocks for N=1024
    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 128                      # hidden_dim ≤ 128 in all test configs

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    mask_apply_kernel[grid](
        left,
        right,
        mask_h,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left/right are modified in‑place – nothing to return


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑aware forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model weights (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]       # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                            # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # 2) Fused projection + gating (half‑precision)
    # ------------------------------------------------------------------
    x_half = x.half()                           # fp16 for Tensor‑Core

    # left side: concatenate proj & gate weights once
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(dtype=torch.float16)
    left_comb = F.linear(x_half, left_comb_weight)                # [B, N, N, 2*H] (fp16)
    left_proj, left_gate = torch.split(left_comb, hidden_dim, dim=-1)
    left = left_proj * left_gate.sigmoid()                        # [B, N, N, H] (fp16)

    # right side – analogous
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(dtype=torch.float16)
    right_comb = F.linear(x_half, right_comb_weight)
    right_proj, right_gate = torch.split(right_comb, hidden_dim, dim=-1)
    right = right_proj * right_gate.sigmoid()                     # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied in‑place via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden dim
        _apply_mask_triton(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 4) Core O(N³) update – batched GEMM (L @ Rᵀ) using cuBLAS
    #    left/right: [B, N, N, H] (fp16)
    #    we reshape to (B*H, N, N) and use torch.bmm
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to (B*H, N, N) – note the permute makes the hidden dim the batch axis
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)     # fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)   # fp16

    # batched matmul + fp32 accumulation
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()   # (B*H, N, N) fp32

    # restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                            # fp32

    # output gate computed from the *normed* input (same as reference)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H] fp32
    out = out * out_gate

    # final linear projection back to `dim`
    out = F.linear(out, to_out_weight)                    # fp32 → [B, N, N, dim]

    return out
