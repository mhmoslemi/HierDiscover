@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # combined batch*head
    pid1 = tl.program_id(1)  # i tile
    pid2 = tl.program_id(2)  # j tile

    # compute batch and head
    b = pid0 // H
    h = pid0 % H

    # Base offsets
    i_base = pid1 * BLOCK_I
    j_base = pid2 * BLOCK_J

    # create range indices
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    # mask for i and j within N
    i_mask = i < N
    j_mask = j < N

    # initialize accumulator as float32 zeros
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension tiles
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_start * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # compute mask combined
        mask = i_mask[:, None] & k_mask[None, :]
        # load left tile (i,k)
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k,
            mask=mask,
            other=0.0,
        )

        # right tile (k,j)
        mask_r = k_mask[:, None] & j_mask[None, :]
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + k[:, None] * stride_r_k
            + j[None, :] * stride_r_j,
            mask=mask_r,
            other=0.0,
        )

        # cast to float32 for accumulation
        left_f = left.to(tl.float32)
        right_f = right.to(tl.float32)

        # dot product
        acc += tl.dot(left_f, right_f)  # (BLOCK_I, BLOCK_J)

    # store output
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

"""
Optimized Triton implementation of the *outgoing* Triangle‑Multiplicative‑Update
(TriMul) as used in AlphaFold‑3.  

The kernel fuses the O(N³) reduction `out[i,j] = Σₖ left[i,k] * right[j,k]`
over the hidden dimension `h` with a custom batched GEMM written in Triton.
A second Triton kernel optionally multiplies `left`/`right` by a binary mask
(broadcast over `h`).  All heavy linear algebra is performed in FP16 while
LayerNorms and the final projection stay in FP32 for numerical stability.

The implementation follows the reference PyTorch module but removes the
gradient path and keeps the forward pass fully GPU‑resident.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel 1: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    mask_ptr,               # fp16 *
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Multiply left/right [B,N,N,H] by mask [B,N,N] (broadcast over H)."""
    b = tl.program_id(0)          # batch
    i_tile = tl.program_id(1)     # i‑tile
    j_tile = tl.program_id(2)     # j‑tile

    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # ---------- load mask (scalar per i,j) ----------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]          # (BLOCK_I, BLOCK_J, 1)

    # ---------- left ----------
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    # ---------- right ----------
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )

    # ---------- apply mask ----------
    left = left * mask
    right = right * mask

    # ---------- store ----------
    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """In‑place multiplication of ``left`` and ``right`` by ``mask`` using Triton."""
    B, N, _, H = left.shape
    mask_h = mask.to(dtype=left.dtype)

    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_h.stride()

    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 64                      # H ≤ 128 → 64 gives good occupancy

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_apply_kernel[grid](
        left,
        right,
        mask_h,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left and right are modified in‑place


# ----------------------------------------------------------------------
# Triton kernel 2: core O(N³) reduction (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Grid:
        program_id(0) → combined batch*head index  (p = b*H + h)
        program_id(1) → i‑tile
        program_id(2) → j‑tile
    """
    pid0 = tl.program_id(0)          # combined batch*head
    pid1 = tl.program_id(1)          # i tile
    pid2 = tl.program_id(2)          # j tile

    # decode batch and head
    b = pid0 // H
    h = pid0 % H

    i_base = pid1 * BLOCK_I
    j_base = pid2 * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # ----- left[i,k] -----
        mask_l = i_mask[:, None] & k_mask[None, :]
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k,
            mask=mask_l,
            other=0.0,
        )

        # ----- right[k,j] -----
        mask_r = k_mask[:, None] & j_mask[None, :]
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + k[:, None] * stride_r_k
            + j[None, :] * stride_r_j,
            mask=mask_r,
            other=0.0,
        )

        # ---- accumulate (cast to FP32) ----
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # store result (cast back to FP16)
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
    )
    store_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)


def _tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched Triton implementation of the outgoing TriMul reduction.
    left/right are FP16 tensors of shape [B,N,N,H].
    Returns out (FP16) of shape [B,N,N,H].
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16)

    # strides (in element units)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # grid: (batch*head, ceil(N/BLOCK_I), ceil(N/BLOCK_J))
    grid = (
        B * H,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2) load model weights (all float32 on device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # (H, dim)
    right_proj_weight = weights["right_proj.weight"]    # (H, dim)
    left_gate_weight = weights["left_gate.weight"]      # (H, dim)
    right_gate_weight = weights["right_gate.weight"]    # (H, dim)
    out_gate_weight = weights["out_gate.weight"]        # (H, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (dim, H)

    # ------------------------------------------------------------------
    # 3) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 4) Fused projection + gating (FP16)
    # ------------------------------------------------------------------
    x_fp16 = x_norm.half()   # cast once

    # combine the four projection/gate weight matrices → (4*H, dim)
    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)

    # linear produces (B, N, N, 4*H) in FP16
    combined = F.linear(x_fp16, combined_weight)  # bias=None

    # split into four tensors of shape [B,N,N,H] (fp16)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )

    # apply sigmoid gates
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 5) Optional mask (in‑place via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        assert mask_tensor is not None, "mask must be provided when nomask=False"
        _apply_mask_triton(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 6) Core O(N³) update – Triton batched GEMM
    # ------------------------------------------------------------------
    out_fp16 = _tri_mul_outgoing_triton(left, right)   # [B,N,N,H] fp16

    # cast to FP32 for the following LayerNorm
    out = out_fp16.float()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )   # FP32

    # Out‑gate depends on the *normed* input (same as the reference)
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))   # [B,N,N,H] FP32
    out = out * out_gate

    # Final linear projection back to ``dim``
    out = F.linear(out, to_out_weight)   # [B,N,N,dim] FP32

    return out
```
