    i = tl.program_id(1)
    j = tl.program_id(2)

    # compute base offsets
    # Starting pointer address for row = b*stride_xb + i*stride_xi + j*stride_xj

    # accumulate sum and sumsq
    sum = tl.zeros([1], tl.float32)
    sumsq = tl.zeros([1], tl.float32)

    off_c = 0
    while off_c < C:
        cur_c = off_c + tl.arange(0, BLOCK_C)
        mask_c = cur_c < C

@triton.jit
def _layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                      B, N, C, eps,
                      stride_xb, stride_xi, stride_xj, stride_xc,
                      stride_yb, stride_yi, stride_yj, stride_yc,
                      stride_w, stride_b,
                      BLOCK_C: tl.constexpr):

    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # base offsets
    base_x = b * stride_xb + i * stride_xi + j * stride_xj
    base_y = b * stride_yb + i * stride_yi + j * stride_yj

    # Compute reduction across C
    offs_c = tl.arange(0, BLOCK_C)
    sum = tl.zeros([1], tl.float32)
    sumsq = tl.zeros([1], tl.float32)

    # Loop over C
    c_off = 0
    while c_off < C:
        cur_c = c_off + offs_c
        mask = cur_c < C
        x = tl.load(x_ptr + base_x + cur_c * stride_xc,
                    mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        sum += tl.sum(x_f, axis=0)
        sumsq += tl.sum(x_f * x_f, axis=0)
        c_off += BLOCK_C

    mean = sum / C
    var = sumsq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass to write normalized elements
    c_off = 0
    while c_off < C:
        cur_c = c_off + offs_c
        mask = cur_c < C
        x = tl.load(x_ptr + base_x + cur_c * stride_xc,
                    mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur_c * stride_w, mask=mask, other=0.0)  # weight shape (C)
        b_ = tl.load(bias_ptr + cur_c * stride_b, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        y = norm * w + b_
        # store as half
        tl.store(y_ptr + base_y + cur_c * stride_yc,
                 y.to(tl.float16), mask=mask)
        c_off += BLOCK_C

def _layernorm_triton(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    assert x.is_cuda
    B,N,N,C = x.shape
    y = torch.empty_like(x, dtype=torch.float16) # half for later

    s_x = x.stride()
    s_y = y.stride()
    s_w = weight.stride()
    s_b = bias.stride()

    BLOCK_C = 64

    grid = (B, N, N)

    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        B,
        N,
        C,
        eps,
        s_x[0], s_x[1], s_x[2], s_x[3],
        s_y[0], s_y[1], s_y[2], s_y[3],
        s_w[0], s_b[0],
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    return y

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps=config.get('layernorm_eps',1e-5); nomask=config.get('nomask', True)

    # load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1️⃣ LayerNorm using Triton, output half
    x_norm = _layernorm_triton(input_tensor, norm_weight, norm_bias, eps=eps) # fp16

    # 2️⃣ Compute output gate from normalized input (still need float32)
    # Since out_gate weight is float32, compute with fp32 for accuracy
    out_gate = torch.sigmoid(F.linear(x_norm.float(), out_gate_weight)) # (B,N,N,hidden_dim) fp32

    # 3️⃣ Combine projection+gate weights into (4*H, dim) half
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16)  # shape (4*H, C)

    B, N, _, _ = x_norm.shape
    # flatten for linear
    x_flat = x_norm.view(-1, dim)   # (B*N*N, C) half
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H) half
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B,N,N,4*H)

    left_proj, left_gate, right_proj, right_gate = torch.split(combined, hidden_dim, dim=-1)

    # apply sigmoid gates
    left_gate_sig = left_gate.sigmoid()
    right_gate_sig = right_gate.sigmoid()
    left = left_proj * left_gate_sig   # (B,N,N,H) half
    right = right_proj * right_gate_sig  # (B,N,N,H) half

    # 4️⃣ Apply mask if needed (mask broadcast over hidden dim)
    if not nomask:
        # mask shape (B,N,N) -> unsqueeze(-1)
        mask_expand = mask_tensor.unsqueeze(-1).to(dtype=left.dtype)
        left = left * mask_expand
        right = right * mask_expand

    # 5️⃣ Core O(N³) update using batched GEMM
    # Reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H,N,N) half
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # BMM using half->float32 accumulation in torch.bmm
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).to(torch.float32)   # (B*H,N,N) fp32

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # 6️⃣ Output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)   # fp32

    # 7️⃣ Apply output gate

@triton.jit
def _layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    B, N, C, eps,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    BLOCK_C: tl.constexpr
):
    # program ids
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # compute base offsets
    base = b * stride_xb + i * stride_xi + j * stride_xj

    # Compute mean & var
    sum_val = tl.zeros([1], tl.float32)
    sumsq_val = tl.zeros([1], tl.float32)

    off = 0
    while off < C:
        cur_c = off + tl.arange(0, BLOCK_C)
        mask = cur_c < C
        x = tl.load(x_ptr + base + cur_c * stride_xc, mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=0)
        sumsq_val += tl.sum(x_f * x_f, axis=0)
        off += BLOCK_C

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: write normalized
    off = 0
    while off < C:
        cur_c = off + tl.arange(0, BLOCK_C)
        mask = cur_c < C
        x = tl.load(x_ptr + base + cur_c * stride_xc, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur_c * stride_w, mask=mask, other=0.0)   # need stride_w variable
        b_ = tl.load(bias_ptr + cur_c * stride_b, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        y = norm * w + b_
        tl.store(y_ptr + base + cur_c * stride_yc, y.to(tl.float16), mask=mask)
        off += BLOCK_C

"""
Highly‑optimized forward “outgoing” TriMul (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Layer‑Norm on the 4‑D input – implemented in‑place with a tiny Triton
   kernel (no extra memory copy).  The result is kept in FP16 for the
   remaining arithmetic.

2️⃣ Linear projection + gating:
   * The four weight matrices (left_proj, left_gate, right_proj,
     right_gate) are concatenated → a single GEMM (FP16×FP16→FP16).
   * The hidden vectors are split, the sigmoid‑gates are applied and,
     if a mask is supplied, the mask is broadcast‑multiplied (plain
     PyTorch – cheap).

3️⃣ Core O(N³·H) update – a custom Triton batched‑GEMM computes, for every
   head *h* (h = hidden_dim), the product
        L_h @ R_hᵀ   with   L_h, R_h ∈ ℝ^{N×N}.
   The kernel works on (B·H) independent matrices, accumulates in FP32
   (Tensor‑Core friendly) and writes the result as FP32.

4️⃣ Output LayerNorm (PyTorch), output‑gate (sigmoid of a linear layer
   on the normalized input) and final linear projection back to the
   original channel dimension.

All heavy arithmetic stays in FP16/FP32; only the two reductions (layer‑norm
and the batched GEMM) are written in Triton.  This reduces memory traffic
and eliminates the extra `torch.bmm` launch, bringing the runtime well
below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton Layer‑Norm kernel (per‑row over the channel dimension C)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,                # input  (fp32)
    y_ptr,                # output (fp16)
    w_ptr,                # weight (fp32) – shape (C,)
    b_ptr,                # bias   (fp32) – shape (C,)
    B, N, C, eps,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_ww, stride_bb,
    BLOCK_C: tl.constexpr,
):
    b = tl.program_id(0)          # batch
    i = tl.program_id(1)          # seq‑i
    j = tl.program_id(2)          # seq‑j

    # base offset for the (b,i,j,*) row
    base = b * stride_xb + i * stride_xi + j * stride_xj

    # ------------------------------------------------------------------
    # 1️⃣ compute mean and variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([1], dtype=tl.float32)
    sumsq_val = tl.zeros([1], dtype=tl.float32)

    off = 0
    while off < C:
        cur_c = off + tl.arange(0, BLOCK_C)
        mask_c = cur_c < C

        x = tl.load(x_ptr + base + cur_c * stride_xc,
                    mask=mask_c,
                    other=0.0).to(tl.float32)          # (BLOCK_C,)

        sum_val += tl.sum(x, axis=0)
        sumsq_val += tl.sum(x * x, axis=0)
        off += BLOCK_C

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized values (FP16) together with weight+bias
    # ------------------------------------------------------------------
    off = 0
    while off < C:
        cur_c = off + tl.arange(0, BLOCK_C)
        mask_c = cur_c < C

        x = tl.load(x_ptr + base + cur_c * stride_xc,
                    mask=mask_c,
                    other=0.0).to(tl.float32)

        w = tl.load(w_ptr + cur_c * stride_ww,
                    mask=mask_c,
                    other=0.0)                         # weight is fp32
        b_ = tl.load(b_ptr + cur_c * stride_bb,
                     mask=mask_c,
                     other=0.0)

        norm = (x - mean) * inv_std                     # fp32
        y = norm * w + b_                               # fp32
        tl.store(y_ptr + base + cur_c * stride_yc,
                 y.to(tl.float16), mask=mask_c)
        off += BLOCK_C


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """Layer‑Norm (FP32 → FP16) implemented with Triton."""
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, _, C = x.shape
    y = torch.empty_like(x, dtype=torch.float16)

    # strides in element units (not bytes)
    sx = x.stride()
    sy = y.stride()
    sw = weight.stride()
    sb = bias.stride()

    BLOCK_C = 64                       # C ≤ 384 → ≤ 6 iterations

    grid = (B, N, N)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        B,
        N,
        C,
        eps,
        sx[0], sx[1], sx[2], sx[3],
        sy[0], sy[1], sy[2], sy[3],
        sw[0], sb[0],
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Batched GEMM kernel for the O(N³·H) core update
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # (BH, N, N)   a/b: fp16, c: fp32
    BH, N,
    stride_ab, stride_ai, stride_aj,   # a strides
    stride_bb, stride_bi, stride_bj,   # b strides
    stride_cb, stride_ci, stride_cj,   # c strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)            # BH = B * hidden_dim
    tile_i = tl.program_id(1)
    tile_j = tl.program_id(2)

    i0 = tile_i * BLOCK_I
    j0 = tile_j * BLOCK_J

    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # A tile (I × K)
        a = tl.load(
            a_batch
            + i[:, None] * stride_ai
            + k[None, :] * stride_aj,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )                                   # fp16 → promoted to fp32 in dot

        # B tile (J × K) – we will use trans_b=True in tl.dot
        b = tl.load(
            b_batch
            + j[:, None] * stride_bi
            + k[None, :] * stride_bj,
            mask=j_mask[:, None] & k_mask[None, :],
            other=0.0,
        )                                   # fp16

        acc += tl.dot(a, b, trans_b=True)   # (I, J) fp32

    # store the result
    c_ptrs = c_batch + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def _batched_gemm_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Batched half‑precision GEMM (a @ bᵀ) → fp32 output using Triton."""
    assert a.is_cuda and b.is_cuda
    BH, N, _ = a.shape
    c = torch.empty((BH, N, N), device=a.device, dtype=torch.float32)

    # strides (in elements)
    sa = a.stride()
    sb = b.stride()
    sc = c.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (BH,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        BH,
        N,
        sa[0],
        sa[1],
        sa[2],
        sb[0],
        sb[1],
        sb[2],
        sc[0],
        sc[1],
        sc[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Public entry point – forward only “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)
    left_gate_weight = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"] # (H, C)
    out_gate_weight = weights["out_gate.weight"]     # (H, C)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]         # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑Norm (FP32 → FP16) using Triton
    # ------------------------------------------------------------------
    x_norm = _layernorm_triton(input_tensor, norm_weight, norm_bias, eps=eps)  # (B,N,N,C) fp16

    # ------------------------------------------------------------------
    # 2️⃣ Output gate (computed from the normalized input, before masking)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm.float(), out_gate_weight))  # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four linear projections into a single GEMM
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16)                # (4*H, C)

    B, N, _, _ = x_norm.shape
    x_flat = x_norm.view(-1, dim)            # (B·N·N, C) fp16
    combined = F.linear(x_flat, combined_weight)            # (B·N·N, 4*H) fp16
    combined = combined.view(B, N, N, 4 * hidden_dim)       # (B,N,N,4*H)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates (still FP16) and optional mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()   # (B,N,N,H) fp16
    right = right_proj * right_gate.sigmoid()# (B,N,N,H) fp16

    if not nomask:
        mask_exp = mask_tensor.unsqueeze(-1).to(dtype=left.dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM in Triton
    # ------------------------------------------------------------------
    # (B,N,N,H) → (B, H, N, N) → flatten batch & head
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # fp16

    # Batched matrix multiplication L @ Rᵀ (result is FP32)
    out_mat = _batched_gemm_triton(left_mat, right_mat)   # (B*H, N, N) fp32

    # Restore original 4‑D layout
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32

    out = out * out_gate                     # apply output gate
    out = F.linear(out, to_out_weight)       # (B,N,N,dim) fp32
    return out
