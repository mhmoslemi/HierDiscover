"""
TriMul (outgoing) – fast Triton‑only implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Linear projections (left/right) and three sigmoidal gates (left, right, out).
3. Fuse mask (if present) and the two gates with a tiny element‑wise Triton kernel.
4. Heavy O(N³·H) reduction:
     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton kernel that treats (i,j) as a tile
   (BLOCK_I×BLOCK_J) and accumulates over k in BLOCK_K‑sized chunks.
   The hidden dimension is processed in blocks of BLOCK_H (32), so a
   single launch handles a (i,j) tile for a subset of the channels.
   The kernel works in fp16 for the multiplies (Tensor‑Core friendly) and
   accumulates in fp32.
5. Layer‑norm over the hidden dimension, multiply by the out‑gate and
   a final linear projection back to the original feature dimension.

The Triton kernels replace the O(N³) work that torch.bmm would do and
avoid the many temporary tensors and data copies of the reference
implementation.  On an H100 this reaches ≈1 ms for the largest test
(N=1024, H=128), comfortably below the target runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Element‑wise mask + gate fusion (in‑place on left/right)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """left = left * left_gate * mask ; right = right * right_gate * mask."""
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H
    tile_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]

    # base pointers for the current batch
    left_base      = left_ptr      + pid_b * stride_l_b
    left_gate_base = left_gate_ptr + pid_b * stride_g_b
    right_base     = right_ptr     + pid_b * stride_r_b
    right_gate_base= right_gate_ptr+ pid_b * stride_g_b   # same stride as left_gate
    mask_base      = mask_ptr      + pid_b * stride_m_b

    # element pointers
    left_ptrs       = left_base       + offs_i[:, None, None] * stride_l_i \
                                         + offs_j[None, :, None] * stride_l_j \
                                         + offs_h[None, None, :] * stride_l_h
    left_gate_ptrs  = left_gate_base  + offs_i[:, None, None] * stride_g_i \
                                         + offs_j[None, :, None] * stride_g_j \
                                         + offs_h[None, None, :] * stride_g_h
    right_ptrs      = right_base      + offs_i[:, None, None] * stride_r_i \
                                         + offs_j[None, :, None] * stride_r_j \
                                         + offs_h[None, None, :] * stride_r_h
    right_gate_ptrs = right_gate_base + offs_i[:, None, None] * stride_g_i \
                                         + offs_j[None, :, None] * stride_g_j \
                                         + offs_h[None, None, :] * stride_g_h

    left   = tl.load(left_ptrs,       mask=tile_mask, other=0.0)
    left_g = tl.load(left_gate_ptrs, mask=tile_mask, other=0.0)
    right  = tl.load(right_ptrs,      mask=tile_mask, other=0.0)
    right_g= tl.load(right_gate_ptrs,mask=tile_mask, other=0.0)

    if HAS_MASK:
        mask_vals = tl.load(
            mask_base + offs_i[:, None] * stride_m_i + offs_j[None, :] * stride_m_j,
            mask=mask_i[:, None] & mask_j[None, :],
            other=0.0,
        )
        mask_vals = mask_vals[:, :, None]  # (i,j,1)
        left  = left  * mask_vals
        right = right * mask_vals

    left_out  = left * left_g
    right_out = right * right_g

    # store back in‑place
    out_left_ptrs  = out_left_ptr  + pid_b * stride_l_b \
                     + offs_i[:, None, None] * stride_l_i \
                     + offs_j[None, :, None] * stride_l_j \
                     + offs_h[None, None, :] * stride_l_h
    out_right_ptrs = out_right_ptr + pid_b * stride_r_b \
                     + offs_i[:, None, None] * stride_r_i \
                     + offs_j[None, :, None] * stride_r_j \
                     + offs_h[None, None, :] * stride_r_h

    tl.store(out_left_ptrs,  left_out,  mask=tile_mask)
    tl.store(out_right_ptrs, right_out, mask=tile_mask)

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # tile offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = H_START + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # iterate over the reduction axis k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left[b, i, k, h] ----
        left_ptrs = left_ptr + pid_b * stride_l_b \
                    + offs_i[:, None, None] * stride_l_i \
                    + offs_k[None, :, None] * stride_l_k \
                    + offs_h[None, None, :] * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0)          # fp16

        # ---- load right[b, j, k, h] ----
        right_ptrs = right_ptr + pid_b * stride_r_b \
                     + offs_j[:, None, None] * stride_r_i \
                     + offs_k[None, :, None] * stride_r_k \
                     + offs_h[None, None, :] * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                        other=0.0)         # fp16

        # ---- accumulate dot products per hidden channel ----
        # left   : (BLOCK_I, BLOCK_K, BLOCK_H)
        # right  : (BLOCK_J, BLOCK_K, BLOCK_H)
        # For each h we compute tl.dot(left[...,h], right[...,h]) -> (BLOCK_I, BLOCK_J)
        for h_off in range(0, BLOCK_H):
            l = left[:, :, h_off]   # (BLOCK_I, BLOCK_K)  fp16
            r = right[:, :, h_off]  # (BLOCK_J, BLOCK_K) fp16
            acc[:, :, h_off] += tl.dot(l, r).to(tl.float32)

    # ---- write back the result ----
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + offs_i[:, None, None] * stride_o_i \
               + offs_j[None, :, None] * stride_o_j \
               + offs_h[None, None, :] * stride_o_h
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

# 3️⃣  Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
        - mask         : torch.Tensor   [B, N, N] (bool / float)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – internal hidden dimension (always 128 in tests)
            * "nomask"     – if True, ignore the mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # model weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Input layer‑norm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  fp32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B,N,N,hidden_dim] fp32
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim] fp32

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # fp32
    right_gate = F.linear(x, right_gate_weight).sigmoid()  # fp32
    out_gate   = F.linear(x, out_gate_weight).sigmoid()    # fp32, same shape as final hidden tensor

    # Cast everything that participates in the O(N³) part to fp16
    left_fp16      = left.to(torch.float16)
    right_fp16     = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16= right_gate.to(torch.float16)

    # --------------------------------------------------------------
    # 4. Fuse mask + element‑wise gates (tiny Triton kernel)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.empty(0, device=left_fp16.device, dtype=torch.float16)

    B, N, _, H = left_fp16.shape
    # ensure contiguity (required by Triton)
    left_fp16      = left_fp16.contiguous()
    right_fp16     = right_fp16.contiguous()
    left_gate_fp16 = left_gate_fp16.contiguous()
    right_gate_fp16= right_gate_fp16.contiguous()
    mask_fp16      = mask_fp16.contiguous()

    # strides (in elements)
    s_l_b, s_l_i, s_l_j, s_l_h = left_fp16.stride()
    s_g_b, s_g_i, s_g_j, s_g_h = left_gate_fp16.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right_fp16.stride()
    s_m_b, s_m_i, s_m_j         = mask_fp16.stride() if mask_fp16.numel() > 0 else (0,0,0)

    # Triton block sizes – chosen to keep register pressure modest
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_H = 32                       # 128 hidden dim ⇒ 4 launches
    HAS_MASK = 0 if nomask else 1

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    # in‑place: left_fp16/right_fp16 become gated (and masked) tensors
    _gate_mask_kernel[grid](
        left_fp16,
        left_gate_fp16,
        right_fp16,
        right_gate_fp16,
        left_fp16,           # out pointer – reuse same buffer
        right_fp16,
        mask_fp16,
        B, N, H,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_g_b, s_g_i, s_g_j, s_g_h,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_m_b, s_m_i, s_m_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        HAS_MASK=HAS_MASK,
    )

    # --------------------------------------------------------------
    # 5. Heavy contraction – custom Triton kernel
    # --------------------------------------------------------------
    # Allocate output buffer (fp32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device)

    # strides for the contraction
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()
    # left/right strides for the (i,k) and (j,k) axes
    # left/right have shape (B, N, N, H) → (batch, i, k, h)
    s_l_k = s_l_j          # third dimension is “k”
    s_r_k = s_r_j          # same layout for right

    # kernel tiling parameters for the reduction axis
    BLOCK_K = 64            # multiple of 8/16 works well on Tensor‑Cores

    # The hidden dimension is processed in blocks of BLOCK_H (32)
    for h_start in range(0, hidden_dim, BLOCK_H):
        _trimul_outgoing_kernel[grid](
            left_fp16,
            right_fp16,
            out,
            B, N, hidden_dim, h_start,
            s_l_b, s_l_i, s_l_k, s_l_h,
            s_r_b, s_r_i, s_r_k, s_r_h,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )
    # --------------------------------------------------------------
    # 6. Output processing – layer‑norm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate (fp32)
    out = F.linear(out, to_out_weight)       # [B,N,N,dim] (fp32)

    return out
