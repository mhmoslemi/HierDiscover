import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast Triton implementation
================================================

The core of the module is the N³ reduction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched GEMM for every hidden channel `h`.  A custom
Triton kernel computes this reduction in FP16 with FP32 accumulation,
treating the pair (batch, hidden) as a single virtual batch dimension.
No permute‑copy of the tensors is required – the kernel works directly
on the original `[B, N, N, H]` layout by supplying the appropriate
strides.  The surrounding operations (LayerNorm, linear projections,
gates, final LayerNorm and projection) are performed in PyTorch.

Result: the heavy O(B·H·N³) work stays on the GPU, memory traffic is
minimised and the kernel runs well below the target ~1 ms on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # *fp16  left  [batch,H,N,N] (viewed as [batch,N,N,H])
    b_ptr,                     # *fp16  right [batch,H,N,N] (viewed as [batch,N,N,H])
    c_ptr,                     # *fp16  out   [batch,H,N,N]
    M, N, K,                   # problem size (int32) – all equal to seq_len
    stride_a_batch, stride_am, stride_ak, stride_ah,   # left strides
    stride_b_batch, stride_bk, stride_bn, stride_bh,   # right strides (row/col swapped)
    stride_c_batch, stride_cm, stride_cn, stride_ch,   # out   strides
    hidden_dim,                # total hidden channels (int)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B for many (batch, hidden) pairs.

    A has shape (M, K)   – rows are `i`, cols are `k`
    B has shape (K, N)   – rows are `k`, cols are `j`
    C has shape (M, N)

    The kernel is launched with a 3‑D grid:
        (batch_h, ceil(M/BLOCK_M), ceil(N/BLOCK_N))
    where   batch_h = B * hidden_dim.
    """
    pid = tl.program_id(0)               # combined (batch, hidden) index
    batch_idx = pid // hidden_dim
    h_idx = pid % hidden_dim

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + h_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_b_batch + h_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_c_batch + h_idx * stride_ch

    # block indices for the output matrix C
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (better precision for the huge sums)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension (the summed dimension)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tiles of A and B
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles, zero‑pad out‑of‑bounds elements
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16×FP16 → FP32 accumulation via tensor‑core dot
        acc += tl.dot(a, b)

    # write the result tile back to C (store as FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
        - mask         : torch.Tensor   [B, N, N]   (bool / float)
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – hidden dimension used in internal projections
            * "nomask"     – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (float32 → fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]  (float32)
    right = F.linear(x, right_proj_weight)

    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid).  Gates are kept in fp16 except the final out‑gate.
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid().to(left.dtype)
    right_gate = F.linear(x, right_gate_weight).sigmoid().to(right.dtype)
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # stays float32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy N³ reduction – batched GEMM with Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape               # left shape = [B, N, N, H]
    out = torch.empty_like(left, dtype=torch.float16)   # same shape

    #--- strides for the original layout [B, N, N, H] (contiguous) ----------
    # left
    stride_a_batch, stride_am, stride_ak, stride_ah = left.stride()
    # right – we need rows = k (dim2) and cols = j (dim1), so we swap the two inner strides
    stride_b_batch, stride_bi, stride_bj, stride_bh = right.stride()
    stride_bk = stride_bj          # row stride (k)
    stride_bn = stride_bi          # column stride (j)

    # out
    stride_c_batch, stride_cm, stride_cn, stride_ch = out.stride()

    # block sizes – tuned for H100 tensor‑cores (FP16×FP16 → FP32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    batch_h = B * hidden_dim
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        N,                # M
        N,                # N
        N,                # K
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_ah,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_c_batch,
        stride_cm,
        stride_cn,
        stride_ch,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Cast to FP32, final LayerNorm, output gate and projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, N, N, hidden_dim]

    out = out * out_gate                       # gating after norm (float32)
    out = F.linear(out, to_out_weight)        # back to `dim`

    return out
