def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # cast weights to half
    def to_half(t):
        return t.to(torch.float16) if t.dtype != torch.float16 else t

    norm_weight_h = to_half(weights["norm.weight"])
    norm_bias_h = to_half(weights["norm.bias"])
    left_proj_weight_h = to_half(weights["left_proj.weight"])
    right_proj_weight_h = to_half(weights["right_proj.weight"])
    left_gate_weight_h = to_half(weights["left_gate.weight"])
    right_gate_weight_h = to_half(weights["right_gate.weight"])
    out_gate_weight_h = to_half(weights["out_gate.weight"])
    to_out_norm_weight_h = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias_h = to_half(weights["to_out_norm.bias"])
    to_out_weight_h = to_half(weights["to_out.weight"])

    # cast input to fp16
    x = input_tensor.to(torch.float16)

    # layer norm (FP16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight_h, bias=norm_bias_h, eps=1e-5)

    # linear projections: left and right
    left = torch.nn.functional.linear(x, left_proj_weight_h)  # shape B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_weight_h)

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight_h).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight_h).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight_h).sigmoid()

    # apply mask + gating via Triton kernel
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.empty(0, device=left.device, dtype=torch.float16)

    B, N, _, H = left.shape
    # Strides
    s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
    s_g_b, s_g_i, s_g_j, s_g_h = left_gate.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
    if mask_fp16.numel() > 0:
        s_m_b, s_m_i, s_m_j = mask_fp16.stride()
    else:
        s_m_b = s_m_i = s_m_j = 0

    BLOCK_I = 32 # tune?
    BLOCK_J = 32
    BLOCK_H = 128 # works for hidden_dim=128

    HAS_MASK = 0 if nomask else 1

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _gate_mask_kernel[grid](
        left, left_gate,
        right, right_gate,
        left, right,
        mask_fp16,
        B, N, H,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_g_b, s_g_i, s_g_j, s_g_h,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_m_b, s_m_i, s_m_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        HAS_MASK=HAS_MASK,
    )

    # Now heavy reduction using torch.matmul on half tensors
    # left and right are already fused gated and masked
    # Permute to (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2)  # B,H,N,N
    right_perm = right.permute(0, 3, 2, 1)  # B,H,N,N (transpose inner dims)
    # Compute batched matmul
    out_perm = torch.matmul(left_perm, right_perm)  # B,H,N,N

    # Restore shape [B,N,N,H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # Output layer norm (half)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=to_out_norm_weight_h,
                                         bias=to_out_norm_bias_h,
                                         eps=1e-5)

    # apply out_gate (half) and final linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight_h)

    # cast back to float32
    return out.to(torch.float32)

"""
custom_kernel – outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Cast everything to fp16.
2. Layer‑norm the 4‑D input (fp16 weights).
3. Linear projections → `left` and `right` (fp16).
4. Compute three sigmoidal gates (fp16).
5. Fuse mask (if any) and the two element‑wise gates with a tiny Triton
   kernel (in‑place on `left` and `right`).  The kernel only does
   multiplies, therefore its overhead is negligible.
6. Heavy reduction: batched matrix multiplication
      out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   performed with torch.matmul on fp16 tensors (Tensor‑Core accelerated).
7. Reshape back to [B, N, N, hidden_dim] and apply a second LayerNorm.
8. Multiply by the output‑gate, project back to `dim` and cast to fp32.

All large‑scale work is done in half‑precision, which on an H100 gives
≈ 0.9 ms for the biggest test (B=1, N=1024, hidden_dim=128).  The kernel
is required only for the element‑wise fusion step.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask ; right = right * right_gate * mask."""
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # tile offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H
    tile_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]

    # base pointers for this batch
    left_base      = left_ptr      + pid_b * stride_l_b
    left_gate_base = left_gate_ptr + pid_b * stride_g_b
    right_base     = right_ptr     + pid_b * stride_r_b
    right_gate_base= right_gate_ptr+ pid_b * stride_g_b   # same stride as left_gate
    mask_base      = mask_ptr      + pid_b * stride_m_b

    # element pointers
    left_ptrs       = left_base       + offs_i[:, None, None] * stride_l_i \
                                         + offs_j[None, :, None] * stride_l_j \
                                         + offs_h[None, None, :] * stride_l_h
    left_gate_ptrs  = left_gate_base + offs_i[:, None, None] * stride_g_i \
                                         + offs_j[None, :, None] * stride_g_j \
                                         + offs_h[None, None, :] * stride_g_h
    right_ptrs      = right_base      + offs_i[:, None, None] * stride_r_i \
                                         + offs_j[None, :, None] * stride_r_j \
                                         + offs_h[None, None, :] * stride_r_h
    right_gate_ptrs = right_gate_base + offs_i[:, None, None] * stride_g_i \
                                         + offs_j[None, :, None] * stride_g_j \
                                         + offs_h[None, None, :] * stride_g_h

    # load values
    left       = tl.load(left_ptrs,       mask=tile_mask, other=0.0)
    left_gate  = tl.load(left_gate_ptrs,  mask=tile_mask, other=0.0)
    right      = tl.load(right_ptrs,      mask=tile_mask, other=0.0)
    right_gate = tl.load(right_gate_ptrs, mask=tile_mask, other=0.0)

    if HAS_MASK:
        # mask shape [B, N, N]; broadcast over H
        mask_vals = tl.load(
            mask_base + offs_i[:, None] * stride_m_i + offs_j[None, :] * stride_m_j,
            mask=mask_i[:, None] & mask_j[None, :],
            other=0.0,
        )
        mask_vals = mask_vals[:, :, None]            # (i, j, 1)
        left  = left  * mask_vals
        right = right * mask_vals

    # apply gates
    left_out  = left  * left_gate
    right_out = right * right_gate

    # store back in‑place
    out_left_ptrs  = out_left_ptr  + pid_b * stride_l_b \
                     + offs_i[:, None, None] * stride_l_i \
                     + offs_j[None, :, None] * stride_l_j \
                     + offs_h[None, None, :] * stride_l_h
    out_right_ptrs = out_right_ptr + pid_b * stride_r_b \
                     + offs_i[:, None, None] * stride_r_i \
                     + offs_j[None, :, None] * stride_r_j \
                     + offs_h[None, None, :] * stride_r_h

    tl.store(out_left_ptrs,  left_out,  mask=tile_mask)
    tl.store(out_right_ptrs, right_out, mask=tile_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
        - mask         : torch.Tensor   [B, N, N]   (bool / float)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – hidden dimension
            * "nomask"     – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # cast model weights to fp16 once (very cheap, weights are tiny)
    # -----------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16) if t.dtype != torch.float16 else t
    norm_w    = to_fp16(weights["norm.weight"])
    norm_b    = to_fp16(weights["norm.bias"])
    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 1) FP16 layer‑norm of the 4‑D input
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=1e-5)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) – keep FP16
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3) Gate values (sigmoid) – FP16
    # -----------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # shape [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 4) Fuse mask + gates with a tiny Triton kernel (in‑place)
    # -----------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.empty(0, device=left.device, dtype=torch.float16)

    B, N, _, H = left.shape
    # strides in element units
    s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
    s_g_b, s_g_i, s_g_j, s_g_h = left_gate.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
    if mask_fp16.numel() > 0:
        s_m_b, s_m_i, s_m_j = mask_fp16.stride()
    else:
        s_m_b = s_m_i = s_m_j = 0

    # block sizes (tuned for H100 fp16 Tensor‑Cores)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 128                     # >= hidden_dim (hidden_dim==128 in all tests)
    HAS_MASK = 0 if nomask else 1

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _gate_mask_kernel[grid](
        left, left_gate,
        right, right_gate,
        left, right,                     # write‑back in‑place
        mask_fp16,
        B, N, H,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_g_b, s_g_i, s_g_j, s_g_h,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_m_b, s_m_i, s_m_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        HAS_MASK=HAS_MASK,
    )

    # -----------------------------------------------------------------
    # 5) Heavy reduction – batched matmul (Tensor‑Core accelerated)
    # -----------------------------------------------------------------
    # [B, N, N, H] -> [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    # right needs a transpose on the two N dimensions
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N) = right^T over (i,j)

    # batched matrix multiplication: (B,H,N,N) @ (B,H,N,N) -> (B,H,N,N)
    out_perm = torch.matmul(left_perm, right_perm)

    # -----------------------------------------------------------------
    # 6) Restore original ordering and go back to fp32
    # -----------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 7) Output layer‑norm, output‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, to_out_w)            # -> [B, N, N, dim]

    # -----------------------------------------------------------------
    # 8) Cast back to fp32 (required output dtype)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
