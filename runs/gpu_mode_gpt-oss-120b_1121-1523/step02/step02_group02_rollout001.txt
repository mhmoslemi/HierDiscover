"""
TriMul (outgoing) – fast Triton‑accelerated implementation
=========================================================

The heavy N³ reduction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is implemented as a batched GEMM (one GEMM per (batch, hidden) pair).
We first bring the hidden dimension to the second axis and make the
summation dimension K contiguous in memory (a cheap permute+contiguous).
This yields co‑alesced loads for both operands and lets the kernel hit
the full Tensor‑Core bandwidth of an H100.

All linear layers and LayerNorm are performed in FP16 so the whole
pipeline stays in half‑precision except the final cast to FP32.
The kernel keeps a FP32 accumulator for numerical stability.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A:[M,K], B:[K,N]) → C:[M,N]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # *fp16  left  [batch_h, M, K]
    b_ptr,                     # *fp16  right [batch_h, K, N]
    c_ptr,                     # *fp16  out   [batch_h, M, N]
    M, N, K,                   # problem size (int32)
    stride_a_batch, stride_am, stride_ak, stride_ah,   # A strides
    stride_b_batch, stride_bk, stride_bn, stride_bh,   # B strides
    stride_c_batch, stride_cm, stride_cn, stride_ch,   # C strides
    hidden_dim,                # total hidden channels (int)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes many independent GEMMs (one per (batch, hidden) pair) in FP16
    with FP32 accumulation.  A and B are already laid out so that the K
    dimension is contiguous (stride of 1) – this is achieved by a
    permute‑contiguous before kernel launch.
    """
    pid = tl.program_id(0)               # combined (batch, hidden) index
    batch_idx = pid // hidden_dim
    h_idx     = pid % hidden_dim

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + h_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_b_batch + h_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_c_batch + h_idx * stride_ch

    # tile indices inside the output matrix C
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (high precision for the large sums)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tiles of A and B
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot: FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # write the result tile back (store as FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)

        return t.to(torch.float16)

    norm_weight = _to_half(weights["norm.weight"])
    norm_bias   = _to_half(weights["norm.bias"])

    left_proj_weight = _to_half(weights["left_proj.weight"])
    right_proj_weight = _to_half(weights["right_proj.weight"])
    left_gate_weight  = _to_half(weights["left_gate.weight"])
    right_gate_weight = _to_half(weights["right_gate.weight"])
    out_gate_weight   = _to_half(weights["out_gate.weight"])

    to_out_norm_weight = _to_half(weights["to_out_norm.weight"])
    to_out_norm_bias   = _to_half(weights["to_out_norm.bias"])
    to_out_weight      = _to_half(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float16)
    # --------------------------------------------------------------
    # layer_norm works on the last dimension; we keep it in FP16 to
    # stay in the Tensor‑Core domain.
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    ).to(torch.float16)                     # [B, N, N, dim] (fp16)

    # --------------------------------------------------------------
    # 2) Fused projections + gates (all in FP16)
    #    left_proj + left_gate  → single matmul, same for right
    # --------------------------------------------------------------
    # combine left projection & gate weights
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)   # (2*H, dim)
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2*H, dim)

    # one matmul for left branch
    left_comb = F.linear(x, left_comb_weight)        # [B, N, N, 2*H] (fp16)
    left = left_comb[..., :hidden_dim]               # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()  # fp16

    # one matmul for right branch
    right_comb = F.linear(x, right_comb_weight)      # [B, N, N, 2*H] (fp16)
    right = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4) Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Heavy N³ reduction – batched GEMM via Triton
    #    We rewrite tensors to make the K dimension contiguous:
    #       left  → [B, H, N, N]   (i, k)
    #       right → [B, H, N, N]   (k, j)  (the inner two dims are swapped)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                            # left shape = [B, N, N, H]

    # bring hidden dimension to the second axis and make K contiguous
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]  (i,k)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]  (k,j)

    # allocate output in the same permuted layout
    out_t = torch.empty_like(left_t)                  # [B, H, N, N] (fp16)

    # strides for left_t
    stride_a_batch, stride_ah, stride_am, stride_ak = left_t.stride()
    # strides for right_t
    stride_b_batch, stride_bh, stride_bk, stride_bn = right_t.stride()
    # strides for out_t
    stride_c_batch, stride_ch, stride_cm, stride_cn = out_t.stride()

    # block sizes tuned for H100 Tensor‑Cores (FP16×FP16 → FP32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    batch_h = B * hidden_dim
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        a_ptr=left_t,
        b_ptr=right_t,
        c_ptr=out_t,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ah=stride_ah,
        stride_b_batch=stride_b_batch,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bh=stride_bh,
        stride_c_batch=stride_c_batch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_ch=stride_ch,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 performs best with 8 warps per block
        num_stages=3,         # 3‑stage pipeline for pre‑fetching
    )

    # permute back to the original layout [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 6) Final LayerNorm, output gate and projection (stay in FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, N, N, H] (fp16)

    # out‑gate (computed from the original x, still in FP16)
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # [B, N, N, H] (fp16)

    out = out * out_gate                               # gating after norm
    out = F.linear(out, to_out_weight)                 # back to `dim` (fp16)

    # --------------------------------------------------------------
    # 7) Cast result to float32 as required by the interface
    # --------------------------------------------------------------
    return out.to(torch.float32)
