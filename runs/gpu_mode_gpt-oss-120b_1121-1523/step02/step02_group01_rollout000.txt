def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # linear:
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # mask
    if mask is not None:

        left = left * mask_expand
        right = right * mask_expand
    # gating
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()

    out = triton_matmul_batched(left, right)  # shape [B,N,N,H] result

    out = out * out_gate
    # final linear out -> dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]
    return out

@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bj, stride_bk, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # get batch and hidden dims
    total_batch_hidden = B * H
    b = pid // H
    h = pid % H

    # offsets for the output tile
    offs_m = tl.arange(0, BLOCK_M) + tl.program_id(axis=1) * BLOCK_M
    offs_n = tl.arange(0, BLOCK_N) + tl.program_id(axis=2) * BLOCK_N

    # mask out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to A and B
    a_ptrs = a_ptr + b * stride_am + h * stride_ah + offs_m[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    # For B we load tile B_T: we want B_T rows = k, cols = j. So we need B entries (b, j, k, h) but use k as row, j as column.
    # offs_k for K loop.
    # We'll compute pointer inside loop.
    # Use granular pointer offset updates inside K loop.

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_offset
        # mask k
        mask_k = offs_k < N

        # Load A tile [M, K]
        a_load = tl.load(a_ptrs + (offs_k[None, :] * stride_ak), mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # Load B tile [N, K] from B (original shape (b, j, k, h)). We want B_T means B_T[k, j] = B[b, j, k, h]
        # So offsets: row index k (k_offset) -> stride_bk; column index j (offs_n) -> stride_bj
        b_ptrs = b_ptr + b * stride_bm + h * stride_bh + offs_n[None, :] * stride_bj + offs_k[:, None] * stride_bk
        # But shape expects (K, N) for dot (K vs N?). Actually we want to compute dot between A[M,K] and B_T[K,N] => so we need B_T shape K,N.
        # So we need to load B_T as shape [K,N] where each row corresponds to offs_k, each column to offs_n.
        b_load = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # dims K,N

        # compute dot
        acc += tl.dot(a_load, b_load)

        # pointer increments optional: we can recompute each iteration.

    # Cast to output dtype, store
    c_ptrs = c_ptr + b * stride_cm + h * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel definitions
@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bj, stride_bk, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    Using a tiled GEMM where right is accessed transposed.
    A: left, shape (B,N,N,H) with axes (i,k)
    B: right, shape (B,N,N,H) with axes (j,k) -> we treat transposed to (k,j)
    C: out, shape (B,N,N,H) with axes (i,j)
    """
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch and hidden index
    total_batch_hidden = B * H
    b = pid // H
    h = pid % H

    # Base pointers for this (b,h) pair
    a_base = a_ptr + b * stride_am + h * stride_ah
    b_base = b_ptr + b * stride_bm + h * stride_bh
    c_base = c_ptr + b * stride_cm + h * stride_ch

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a = tl.load(
            a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B tile (K x N) -> note we read B transposed
        b = tl.load(
            b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        acc += tl.dot(a, b)

    # Write back result
    c = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c, acc, mask=mask_i[:, None] & mask_j[None, :])

def tritmul_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using Triton: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
    left/right: (B, N, N, H) possibly float16.
    Returns out in float32.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Allocate output in float32
    out = torch.empty((B, N, N, H), device=left.device, dtype=torch.float32)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Extract strides
    stride_am, stride_ai, stride_ak, stride_ah = left.stride()
    stride_bm, stride_bj, stride_bk, stride_bh = right.stride()
    stride_cm, stride_ci, stride_cj, stride_ch = out.stride()

    trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_am,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_bh,
        stride_cm,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Outgoing TriMul module forward pass.
    Performs LayerNorm, linear projections, mask & gating, a fused
    batch GEMM via Triton (the core triangular multiplication),
    followed by output LayerNorm, gating and linear projection.
    """
    input_tensor, mask, weights, config = data
    # config contains dim (input channel), hidden_dim (intermediate channel)
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm over input channels -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections (no bias) -----
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]    # (hidden_dim, dim)
    left_gate_weight = weights["left_gate.weight"]      # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]        # (hidden_dim, dim)

    # Project input to hidden dimension
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Apply mask if provided (mask shape [B,N,N]) -----
    if mask is not None:
        # mask may be bool; cast to same dtype
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Compute gates (sigmoid) -----
    left_gate = F.linear(x, left_gate_weight).sigmoid()   # [B,N,N,hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Convert to half precision for GEMM (optional) -----
    # Keeping the original dtype for correctness; H100 performs well on fp16.
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # ----- Core TriMul via Triton (batched GEMM) -----
    # Result shape: [B,N,N,hidden_dim] (float32)
    out = tritmul_batched_gemm(left_half, right_half)

    # ----- Output LayerNorm over hidden_dim -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ----- Apply output gate -----
    out = out * out_gate

    # ----- Final linear projection back to dim -----
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out

"""
Outgoing TriMul (Triangular Multiplicative Update) forward pass.

The implementation follows the reference PyTorch module:
1. Layer‐norm over the input channel dimension.
2. Two linear projections (left/right) and three gate projections (left,
   right, out) without bias.
3. Optional binary mask applied to the projected tensors.
4. Element‑wise gating (sigmoid) applied to the left/right tensors.
5. Core operation:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   – implemented as a fused batched GEMM in Triton (left @ rightᵀ per
   hidden slice).
6. Layer‑norm over the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original channel size.

All heavy work (the N³ contraction) is performed in a custom Triton kernel,
while the surrounding linear / norm / sigmoid work is delegated to
torch.nn.functional for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: fused batched GEMM for the triangular multiplication.
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bj, stride_bk, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    where `left` (a_ptr) and `right` (b_ptr) have shape (B, N, N, H)
    and the contraction is over the third axis (k).  The kernel treats
    the operation as a batched matrix multiplication:
        left[b, :, :, h]      (N × N)    ×   right[b, :, :, h]ᵀ (N × N)
    giving (N × N).  Each (b, h) pair is assigned to a separate
    program_id(0); program_id(1) and program_id(2) tile the M (=i) and
    N (=j) dimensions respectively.
    """
    pid = tl.program_id(0)          # combined batch * hidden index
    pid_i = tl.program_id(1)        # tile over i dimension
    pid_j = tl.program_id(2)        # tile over j dimension

    # Decode batch `b` and hidden index `h` from the global pid
    b = pid // H
    h = pid % H

    # Base pointers for this (b, h) slice
    a_base = a_ptr + b * stride_am + h * stride_ah
    b_base = b_ptr + b * stride_bm + h * stride_bh
    c_base = c_ptr + b * stride_cm + h * stride_ch

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M × K) – shape (i, k)
        a = tl.load(
            a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile transposed (K × N) – we read B originally as (j, k)
        # By swapping the strides we obtain Bᵀ(k, j).
        b = tl.load(
            b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

    # Write the result tile back to C
    c = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c, acc, mask=mask_i[:, None] & mask_j[:, None])


# --------------------------------------------------------------------------- #
# Python helper that launches the Triton kernel.
# --------------------------------------------------------------------------- #
def _trimul_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM wrapper for the TriMul contraction.
    `left` and `right` must have shape (B, N, N, H) and be on the same
    CUDA device.  They are expected to be in FP16 (or FP32); the kernel
    accumulates in FP32 and returns a FP32 result.
    """
    assert left.shape == right.shape, "left and right must share shape"
    B, N, _, H = left.shape

    # Output in FP32 (required by later LayerNorm)
    out = torch.empty((B, N, N, H), device=left.device, dtype=torch.float32)

    # Tile sizes – tuned for H100 (64 is a good trade‑off)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # one program per (b, h)
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over j
    )

    # Strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak, stride_ah = left.stride()
    stride_bm, stride_bj, stride_bk, stride_bh = right.stride()
    stride_cm, stride_ci, stride_cj, stride_ch = out.stride()

    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_am,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_bh,
        stride_cm,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------------- #
# Entry point required by the test harness.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask:        torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights:     dict of model parameters
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1️⃣ LayerNorm over the input channel dimension (dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"] # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------- #
    # 3️⃣ Apply optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------- #
    if mask is not None:
        # Ensure mask is float and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4️⃣ Compute gates (sigmoid) from the *original* input `x`
    # ------------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()   # [B, N, N, hidden_dim]
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------- #
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 6️⃣ Cast to FP16 for the heavy N³ contraction (memory‑bandwidth friendly)
    # ------------------------------------------------------------------- #
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------- #
    # 7️⃣ Core TriMul contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------- #
    # out shape: [B, N, N, hidden_dim]  (FP32)
    out = _trimul_batched_gemm(left_half, right_half)

    # ------------------------------------------------------------------- #
    # 8️⃣ Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 9️⃣ Multiply by the output gate (element‑wise)
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 🔟 Final linear projection back to the original channel size
    # ------------------------------------------------------------------- #
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)         # [B, N, N, dim]

    return out
