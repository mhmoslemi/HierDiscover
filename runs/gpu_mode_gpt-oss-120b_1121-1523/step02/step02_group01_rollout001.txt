"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Compute four linear projections of `x` without bias:
   * left_proj   : (B,N,N, H)   = x @ W_left   (H = hidden_dim)
   * right_proj  : (B,N,N, H)   = x @ W_right
   * left_gate   : (B,N,N, H)   = sigmoid(x @ W_left_gate)
   * right_gate  : (B,N,N, H)   = sigmoid(x @ W_right_gate)
   * out_gate    : (B,N,N, H)   = sigmoid(x @ W_out_gate)   (used later)
3. Fuse mask application, gating and broadcasting with a tiny Triton kernel:
   `left  = left_proj  * left_gate  * mask`
   `right = right_proj * right_gate * mask`
   The kernel iterates over the hidden dimension in blocks.
4. Perform the core triangular multiplication:
   * Permute tensors to batch‑wise matrix‑multiply form:
        L = left .permute(0,3,1,2)   # (B,H,N,N)   i‑k
        R = right.permute(0,3,2,1)   # (B,H,N,N)   k‑j
   * out_hidden = torch.matmul(L, R)  # (B,H,N,N) → Σ_k left[i,k]·right[j,k]
   * out = out_hidden.permute(0,2,3,1)   # (B,N,N,H)
5. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and finally project back to the original feature dimension with a linear
   weight `W_to_out`.
6. Return the result (float32).

The heavy N³ operation is performed by PyTorch’s highly‑optimized
batched GEMM (torch.matmul), while the element‑wise mask / gating step is
fused in Triton to satisfy the “part of the ops must run in Triton” rule.
All tensors are kept on the CUDA device throughout the computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    # Input tensors
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # Output tensors
    out_left_ptr, out_right_ptr,
    # Shape / size
    B, N, H,
    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_outl_b, stride_outl_i, stride_outl_j, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_j, stride_outr_h,
    BLOCK_H: tl.constexpr,
):
    """
    Fuse mask multiplication and gating:
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    One program (block) handles a single (b,i,j) position and loops over the
    hidden dimension H in chunks of BLOCK_H.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from the linear id
    b = pid // (N * N)
    resid = pid % (N * N)
    i = resid // N
    j = resid % N

    # base offsets for each tensor (without the hidden index)
    off_l  = b * stride_l_b  + i * stride_l_i  + j * stride_l_j
    off_r  = b * stride_r_b  + i * stride_r_i  + j * stride_r_j
    off_lg = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    off_rg = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    off_mask = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    off_outl = b * stride_outl_b + i * stride_outl_i + j * stride_outl_j
    off_outr = b * stride_outr_b + i * stride_outr_i + j * stride_outr_j

    # scalar mask for this (b,i,j)
    mask_val = tl.load(mask_ptr + off_mask)

    # loop over the hidden dimension
    for offs_h in range(0, H, BLOCK_H):
        cur_off = offs_h * stride_l_h
        mask = offs_h + tl.arange(0, BLOCK_H) < H

        left      = tl.load(left_ptr      + off_l  + cur_off, mask=mask, other=0.0)
        right     = tl.load(right_ptr     + off_r  + cur_off, mask=mask, other=0.0)
        left_gate = tl.load(left_gate_ptr + off_lg + cur_off, mask=mask, other=0.0)
        right_gate= tl.load(right_gate_ptr+ off_rg + cur_off, mask=mask, other=0.0)

        out_l = left * left_gate * mask_val
        out_r = right * right_gate * mask_val

        tl.store(out_left_ptr  + off_outl + cur_off, out_l, mask=mask)
        tl.store(out_right_ptr + off_outr + cur_off, out_r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate values
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right_proj = F.linear(x, weights["right_proj.weight"])        # (B,N,N,hidden)

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # (B,N,N,hidden)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3. Optional mask handling
    # ------------------------------------------------------------------
    if nomask:
        mask = torch.ones_like(mask_tensor, dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(dtype=x.dtype, device=x.device)

    # make tensors contiguous for deterministic stride extraction
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    B, N, _, _ = left_proj.shape  # (B,N,N,hidden)

    # allocate output buffers for the fused mask+gate step
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # ------------------------------------------------------------------
    # 4. Triton kernel: mask * gate fusion
    # ------------------------------------------------------------------
    # strides in elements (not bytes)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left_proj.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right_proj.stride()
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h = left_gate.stride()
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()
    stride_outl_b, stride_outl_i, stride_outl_j, stride_outl_h = left.stride()
    stride_outr_b, stride_outr_i, stride_outr_j, stride_outr_h = right.stride()

    total_positions = B * N * N
    # launch one program (block) per (b,i,j)
    grid = (total_positions,)

    _mask_gate_kernel[grid](
        # pointers
        left_proj.data_ptr(),
        right_proj.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask.data_ptr(),
        left.data_ptr(),
        right.data_ptr(),
        # shape
        B, N, hidden_dim,
        # strides
        stride_l_b, stride_l_i, stride_l_j, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_h,
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        stride_outl_b, stride_outl_i, stride_outl_j, stride_outl_h,
        stride_outr_b, stride_outr_i, stride_outr_j, stride_outr_h,
        BLOCK_H=32,  # hidden_dim ≤ 384, 32 is a good compile‑time block
    )

    # ------------------------------------------------------------------
    # 5. Core triangular multiplication (einsum) via batched GEMM
    # ------------------------------------------------------------------
    # left: (B,N,N,H) -> (B,H,N,N)   (i,k)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    # right: (B,N,N,H) -> (B,H,N,N)   (k,j)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)

    # batched matrix multiply over the N dimension
    out_hidden = torch.matmul(left_perm, right_perm)  # (B, H, N, N)

    # back to (B,N,N,H)
    out = out_hidden.permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Second LayerNorm, out_gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate  # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
