"""
Fast outgoing TriMul (AlphaFold‑3) – forward only (no autograd)

Algorithm
---------
1️⃣ Layer‑norm the input (float32) and cast to fp16.
2️⃣ Fuse the four linear layers (left/right projections and their gates) into a
   single fp16 GEMM: (B·N·N, C) × (4·H, C)ᵀ → (B·N·N, 4·H).
3️⃣ Split the fused result back to left_proj, right_proj, left_gate, right_gate
   tensors of shape (B,N,N,H) in fp16.
4️⃣ Optional mask – broadcast‑multiply the (B,N,N) mask onto the hidden
   dimension of *both* left_proj and right_proj using a tiny Triton kernel.
5️⃣ Apply sigmoid gates and compute the core contraction
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  via a batched GEMM:
   for each hidden slice h we compute left_h @ right_hᵀ.
6️⃣ Hidden‑dim LayerNorm (fp16), multiply by an output‑gate (sigmoid),
   and a final linear projection back to the original channel dimension.
7️⃣ Return the result as float32 (compatible with the PyTorch reference).

Only the mask‑multiplication step is implemented as a custom Triton kernel;
the rest leverages highly‑optimized PyTorch kernels (fp16 GEMM, batched‑mm,
layer‑norm).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: broadcast mask over the hidden dimension and
#                multiply it in‑place into two tensors (left & right)
# -------------------------------------------------------------
@triton.jit
def _mask_mul_two_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    left[b,i,j,h] *= mask[b,i,j];  right[b,i,j,h] *= mask[b,i,j]
    """
    b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load mask tile (BLOCK_I, BLOCK_J)
    mask_tile = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i[:, None] * stride_mask_i
        + j[None, :] * stride_mask_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # Load left tile
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # Load right tile
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # Apply mask
        left = left * mask_tile
        right = right * mask_tile

        # Store back
        tl.store(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_two_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place broadcast‑multiply a (B,N,N) mask onto the hidden dimension of
    `left` and `right` tensors (both shape B,N,N,H) using Triton.
    """
    B, N, _, H = left.shape
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_mul_two_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # No return – operation is in-place


# -------------------------------------------------------------
# Outgoing TriMul – forward‑only (no autograd needed)
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys:
                * dim (int)           – input dimension C
                * hidden_dim (int)    – hidden dimension H
                * nomask (bool)       – if True, ignore mask
                * layernorm_eps (float, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                    # C
    hidden = config["hidden_dim"]          # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # Load weights (all stored as float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy path
    # -----------------------------------------------------------------
    x_h = x.half()                         # [B, N, N, C]  fp16
    B, N, _, C = x_h.shape

    # -----------------------------------------------------------------
    # 3️⃣ Fuse left/right projection + gating linear layers
    #    (4 * H) outputs in a single GEMM)
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_weight,
         right_proj_weight,
         left_gate_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    x_flat = x_h.view(-1, C)                  # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)  # (B·N·N, 4*H)

    # Split back to the four tensors and reshape to (B, N, N, H)
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – apply via Triton (in‑place)
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 (0.0 or 1.0) and broadcast‑multiply
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_two_triton(left_proj, right_proj, mask_h)

    # -----------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -----------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    left = left_proj * left_gate       # (B,N,N,H) fp16
    right = right_proj * right_gate    # (B,N,N,H) fp16

    # -----------------------------------------------------------------
    # 6️⃣ Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched matrix multiplication per hidden slice.
    # -----------------------------------------------------------------
    # Rearrange for batched GEMM: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Merge batch & hidden dimensions for a single bmm call
    left_mat = left_perm.view(B * hidden, N, N)            # (B*H, N, N)
    right_mat = right_perm.view(B * hidden, N, N)          # (B*H, N, N)

    # Batched matmul: left_mat @ right_matᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    # output gate: sigmoid( linear(x_h) )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final projection back to input dimension C
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())  # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Return float32 for compatibility with reference implementation
    # -----------------------------------------------------------------
    return out.float()
