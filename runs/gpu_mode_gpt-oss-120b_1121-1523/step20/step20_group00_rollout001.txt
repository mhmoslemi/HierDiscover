@triton.jit
def _mask_mul_input_kernel(x_ptr, mask_ptr, out_ptr,
                           B, N, C,
                           stride_xb, stride_xi, stride_xj, stride_xc,
                           stride_maskb, stride_maski, stride_maskj,
                           BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    i = i_start + tl.arange(0, BLOCK_I)
    j = j_start + tl.arange(0, BLOCK_J)
    i_valid = i < N
    j_valid = j < N
    # Load mask tile
    mask = tl.load(mask_ptr + b*stride_maskb + i[:,None]*stride_maski + j[None,:]*stride_maskj,
                   mask=i_valid[:,None] & j_valid[None,:],
                   other=0.0)
    mask = mask[..., None]  # broadcast C
    # Loop over C
    c = 0
    while c < C:
        c_range = tl.arange(0, BLOCK_C)
        c_idx = c + c_range
        c_valid = c_idx < C
        # Load x tile
        x = tl.load(x_ptr + b*stride_xb + i[:,None,None]*stride_xi + j[None,:,None]*stride_xj + c_idx[None,None,:]*stride_xc,
                    mask=i_valid[:,None,None] & j_valid[None,:,None] & c_valid[None,None,:],
                    other=0.0)
        # Multiply
        out = x * mask
        tl.store(out_ptr + b*stride_xb + i[:,None,None]*stride_xi + j[None,:,None]*stride_xj + c_idx[None,None,:]*stride_xc,
                 out,
                 mask=i_valid[:,None,None] & j_valid[None,:,None] & c_valid[None,None,:])
        c += BLOCK_C
```

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
* All heavy arithmetic is performed in fp16 (Tensor‑Core friendly).
* The optional pairwise mask is applied by a tiny Triton kernel that
  multiplies the half‑precision tensor in‑place (broadcast over the channel dim).
* Linear projections for the “left” and “right” branches are fused into a
  single half‑precision GEMM (2·H outputs) – same for the two gating layers.
* The O(N³·H) core update is expressed with a single `torch.einsum`,
  which avoids the expensive `permute(...).contiguous()` copy required by the
  naïve `bmm` formulation.
* LayerNorms and the final projection are kept in fp16; only the final
  result is cast back to fp32 for API compatibility.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: broadcast mask over the channel dimension and
#                multiply the input tensor (half‑precision) in‑place.
# --------------------------------------------------------------
@triton.jit
def _mask_mul_input_kernel(
    x_ptr,                # half * [B,N,N,C]
    mask_ptr,             # half * [B,N,N]
    out_ptr,              # half * [B,N,N,C]
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_msb, stride_msi, stride_msj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """out[b,i,j,c] = x[b,i,j,c] * mask[b,i,j]  (mask broadcast over C)"""
    b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)      # tile i
    pid_j = tl.program_id(2)      # tile j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    i_ok = i < N
    j_ok = j < N

    # Load mask tile (BLOCK_I, BLOCK_J)
    mask = tl.load(
        mask_ptr + b * stride_msb + i[:, None] * stride_msi + j[None, :] * stride_msj,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                      # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1) – broadcast over C

    # Process the channel dimension in chunks
    c_start = 0
    while c_start < C:
        c = c_start + tl.arange(0, BLOCK_C)
        c_ok = c < C
        # Load x tile (BLOCK_I, BLOCK_J, BLOCK_C)
        x = tl.load(
            x_ptr
            + b * stride_xb
            + i[:, None, None] * stride_xi
            + j[None, :, None] * stride_xj
            + c[None, None, :] * stride_xc,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
            other=0.0,
        )
        out = x * mask
        tl.store(
            out_ptr
            + b * stride_xb
            + i[:, None, None] * stride_xi
            + j[None, :, None] * stride_xj
            + c[None, None, :] * stride_xc,
            out,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
        )
        c_start += BLOCK_C


def _apply_mask_to_input(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply a pair‑wise mask to a half‑precision tensor `x`
    (shape [B,N,N,C]) using a Triton kernel.
    Returns a new tensor (still fp16).
    """
    B, N, _, C = x.shape
    out = torch.empty_like(x)

    # launch configuration
    BLOCK_I, BLOCK_J, BLOCK_C = 32, 32, 32
    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_mul_input_kernel[grid](
        x,
        mask,
        out,
        B,
        N,
        C,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    return out

# Outgoing TriMul – forward only
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys
            * dim            – input channel dimension C
            * hidden_dim     – hidden dimension H
            * nomask         – True → mask omitted
            * layernorm_eps  – optional epsilon for LayerNorm

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # C
    hidden = config["hidden_dim"]       # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------
    # 1️⃣ Cast to fp16 early & LayerNorm (half‑precision)
    # -------------------------------------------------
    x_h = input_tensor.half()   # [B,N,N,C] fp16
    x = F.layer_norm(
        x_h,
        (dim,),
        weight=weights["norm.weight"].half(),
        bias=weights["norm.bias"].half(),
        eps=eps,
    )  # still fp16

    # -------------------------------------------------
    # 2️⃣ Apply optional mask (broadcast over C) using Triton
    # -------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        x = _apply_mask_to_input(x, mask_h)   # masked fp16 tensor
    # else x already masked (no mask)

    # -------------------------------------------------
    # 3️⃣ Fused linear projection for left/right branches
    # -------------------------------------------------
    # concatenate projection weights → (2*H, C)
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).half()                     # (2*H, C)

    # flatten for a single GEMM
    x_flat = x.view(-1, dim)                # (B*N*N, C)
    proj_out = F.linear(x_flat, proj_weight)   # (B*N*N, 2*H)

    left_proj, right_proj = torch.split(proj_out, hidden, dim=1)
    left_proj = left_proj.view(B, N, N, hidden)   # (B,N,N,H)
    right_proj = right_proj.view(B, N, N, hidden)

    # -------------------------------------------------
    # 4️⃣ Gating linear layers (un‑masked input)
    # -------------------------------------------------
    gate_weight = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).half()                     # (2*H, C)

    gate_out = F.linear(x_flat, gate_weight)   # (B*N*N, 2*H)
    left_gate_pre, right_gate_pre = torch.split(gate_out, hidden, dim=1)
    left_gate = left_gate_pre.view(B, N, N, hidden).sigmoid()
    right_gate = right_gate_pre.view(B, N, N, hidden).sigmoid()

    # -------------------------------------------------
    # 5️⃣ Apply gates (mask already baked into projections)
    # -------------------------------------------------
    left = left_proj * left_gate               # (B,N,N,H)
    right = right_proj * right_gate            # (B,N,N,H)

    # -------------------------------------------------
    # 6️⃣ Core O(N³·H) update – fused einsum
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # -------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B,N,N,H)

    # -------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm (fp16)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )

    # -------------------------------------------------
    # 8️⃣ Output gate (computed on the *un‑masked* normalized x)
    # -------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"].half())
    ).view(B, N, N, hidden)      # (B,N,N,H)

    # -------------------------------------------------
    # 9️⃣ Apply gate and final projection back to original dim
    # -------------------------------------------------
    out = out * out_gate                     # (B,N,N,H)
    out = F.linear(out.view(-1, hidden), weights["to_out.weight"].half())
    out = out.view(B, N, N, dim)             # (B,N,N,C)

    # -------------------------------------------------
    # 10️⃣ Cast back to fp32 for API compatibility
    # -------------------------------------------------
    return out.float()
