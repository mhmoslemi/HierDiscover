@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    # we can also add num_stages etc
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # offsets for blocked shape
    a_ptrs = A_ptr + batch_id * stride_A_batch + offs_m[:, None] * stride_A_m + (tl.arange(0, BLOCK_K))[None, :] * stride_A_k
    b_ptrs = B_ptr + batch_id * stride_B_batch + (tl.arange(0, BLOCK_K))[:, None] * stride_B_k + offs_n[None, :] * stride_B_n

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute actual k offsets
        k_offset = k * BLOCK_K
        # NOTE: k_start + arange may exceed K, need mask.
        k_valid = k_offset + tl.arange(0, BLOCK_K) < K

        # load A tile
        a = tl.load(
            A_ptr + batch_id * stride_A_batch
                + offs_m[:, None] * stride_A_m
                + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_A_k,
            mask=(offs_m[:, None] < M) & k_valid[None, :],
            other=0.0,
        )
        b = tl.load(
            B_ptr + batch_id * stride_B_batch
                + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_B_k
                + offs_n[None, :] * stride_B_n,
            mask=k_valid[:, None] & (offs_n[None, :] < N),
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)
    # store result
    c = acc.to(tl.float16)

    # store to C (with mask)
    tl.store(
        C_ptr + batch_id * stride_C_batch + offs_m[:, None] * stride_C_m + offs_n[None, :] * stride_C_n,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets for M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = A_ptr + batch * stride_a_batch
    b_ptrs = B_ptr + batch * stride_b_batch
    c_ptrs = C_ptr + batch * stride_c_batch

    # compute mask for M,N
    m_mask = offs_m < M
    n_mask = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute offsets for K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for k
        k_mask = offs_k < K

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptrs
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=m_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # load B tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptrs
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=k_mask[:, None] & n_mask[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)
    # convert acc to fp16
    c = acc.to(tl.float16)
    # store result
    tl.store(
        c_ptrs
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=m_mask[:, None] & n_mask[None, :],
    )

"""
Outgoing TriMul (AlphaFold‑3) – forward only, highly‑fused Triton implementation.

Algorithm
---------
1. Layer‑norm the input (float32) → fp16.
2. Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into one matrix multiply (B·N·N, 4·H).
3. Apply the optional pair‑wise mask *in‑place* with a tiny Triton kernel.
4. Compute sigmoid gates and multiply them into the projected tensors.
5. Core O(N³·H) update: a custom Triton kernel computes
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   (no extra transpose,
   no costly permute+contiguous copies).  The kernel works directly on the
   original [B, N, N, H] layout.
6. Hidden‑dim layer‑norm, output gate (sigmoid) and final projection back to
   the input dimension.
7. Return a float‑32 tensor of shape [B, N, N, C].

All heavy arithmetic runs in fp16, accumulation in fp32.
"""

# Triton kernel: broadcast mask over the hidden dimension and multiply

@triton.jit
def _mask_mul_two_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    left[b,i,j,h] *= mask[b,i,j];  right[b,i,j,h] *= mask[b,i,j]
    """
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    mask_tile = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i[:, None] * stride_mask_i
        + j[None, :] * stride_mask_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left = left * mask_tile
        right = right * mask_tile
        tl.store(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H

# ----------------------------------------------------------------------
# Triton kernel: batched O(N³·H) contraction (the core TriMul)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    The tensors have shape:
        left  : (B, N, N, H)   – stride (b,i,k,h)
        right : (B, N, N, H)   – same layout
        out   : (B, N, N, H)   – same layout
    """
    pid_bh = tl.program_id(0)        # combined batch*hidden index
    pid_i = tl.program_id(1)         # tile over i (output rows)
    pid_j = tl.program_id(2)         # tile over j (output columns)

    # decode batch and hidden ids
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    # pointers to the slice (b,h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_idx   = k_start + tl.arange(0, BLOCK_K)
        k_ok    = k_idx < N

        # left tile: (i,k)
        a = tl.load(
            left_base
            + i[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        )
        # right tile transposed: (k,j)  (original right[j,k])
        b_tile = tl.load(
            right_base
            + k_idx[:, None] * stride_r_k   # k dimension becomes “row”
            + j[None, :] * stride_r_i,      # j dimension becomes “column”
            mask=k_ok[:, None] & j_ok[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b_tile)

    # store result (fp16)
    tl.store(
        out_base
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j,
        acc.to(tl.float16),
        mask=i_ok[:, None] & j_ok[None, :],
    )

        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C] float32

    # -----------------------------------------------------------------
    # 2️⃣ Switch to fp16 for compute‑heavy path
    # -----------------------------------------------------------------
    x_h = x.half()
    B, N, _, C = x_h.shape

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the four linear layers (proj + gates) into one matmul
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_weight,
         right_proj_weight,
         left_gate_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)          # (4·H, C)

    x_flat = x_h.view(-1, C)                           # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)        # (B·N·N, 4·H) fp16
    fused_out = fused_out.view(B, N, N, 4 * hidden)    # (B,N,N,4·H)

    # Slice the fused result (views, no extra copy)
    left_proj      = fused_out[..., 0            : hidden]          # (B,N,N,H)
    right_proj     = fused_out[..., hidden       : 2 * hidden]      # (B,N,N,H)
    left_gate_pre  = fused_out[..., 2 * hidden   : 3 * hidden]      # (B,N,N,H)
    right_gate_pre = fused_out[..., 3 * hidden   : 4 * hidden]      # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – in‑place via Triton kernel
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_two_triton(left_proj, right_proj, mask_h)

    # -----------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -----------------------------------------------------------------
    left_gate  = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    left = left_proj * left_gate          # (B,N,N,H) fp16
    right = right_proj * right_gate       # (B,N,N,H) fp16

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update – custom Triton batched GEMM
    # -----------------------------------------------------------------
    # Output tensor (B,N,N,H) fp16, allocated once
    out = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)

    # Strides for the three tensors (batch, i, j, h)
    s_l = left.stride()   # (b,i,k,h)
    s_r = right.stride()
    s_o = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * hidden,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        hidden,
        N,
        s_l[0], s_l[1], s_l[2], s_l[3],   # left strides
        s_r[0], s_r[1], s_r[2], s_r[3],   # right strides
        s_o[0], s_o[1], s_o[2], s_o[3],   # out strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # -----------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final projection back to input dim C
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Return float32 for compatibility with the reference implementation
    # -----------------------------------------------------------------
    return out.float()
