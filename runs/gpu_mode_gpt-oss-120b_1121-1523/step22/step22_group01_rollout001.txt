python code:

```
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    bh = tl.program_id(0)  # combined batch and head
    pid_i = tl.program_id(1)  # tile across i
    pid_j = tl.program_id(2)  # tile across j

    # Compute batch index b and head index h
    b = bh // H
    h = bh % H

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i < N
    j_valid = j < N

    # pointers base for this (b,h)
    left_ptr += b * stride_left_b + h * stride_left_h
    right_ptr += b * stride_right_b + h * stride_right_h
    out_ptr += b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute range of k for this tile
        k_start = k * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        k_valid = k_idx < N

        # load left tile (M x K)
        # offsets: i * stride_left_i + k_idx * stride_left_k
        a = tl.load(
            left_ptr + i[:, None] * stride_left_i + k_idx[None, :] * stride_left_k,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        # load right tile (K x N) i.e., B_T
        # offsets: k_idx[:, None] * stride_right_k + j[None, :] * stride_right_j
        b_tile = tl.load(
            right_ptr + k_idx[:, None] * stride_right_k + j[None, :] * stride_right_j,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        # matrix multiplication accumulate
        acc += tl.dot(a, b_tile)  # both fp16 -> fp32 accumulation

    # store result to out
    # offsets: i * stride_out_i + j * stride_out_j
    out = acc.to(tl.float32)
    tl.store(
        out_ptr + i[:, None] * stride_out_i + j[None, :] * stride_out_j,
        out,
        mask=i_valid[:, None] & j_valid[None, :],
    )
```

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh = tl.program_id(0)   # batch * head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and head
    b = bh // H
    h = bh % H

    # compute tile start coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for within range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers base for this (b,h)
    a_ptr = left_ptr + b * stride_left_b + h * stride_left_h
    b_ptr = right_ptr + b * stride_right_b + h * stride_right_h
    c_ptr = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (BLOCK_K, BLOCK_N) - need transposed
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,

    # store result

        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )

Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm

1. Layer‑Norm over the channel dimension (float32).

   all heads for a batch element) – this removes the many tiny cuBLAS
   launches of the naïve torch.bmm implementation.
6. Output LayerNorm (float32) + out‑gate (sigmoid of a linear).
7. Final linear projection back to “dim” (fp16) and cast to fp32.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask * sigmoid(gate) fused for left / right tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_j, stride_rg_k, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)    # batch
    pid_i = tl.program_id(1)  # i tile
    pid_j = tl.program_id(2)  # j tile

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # mask tile (float16), broadcast over H
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # -------- LEFT ----------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_k
            + h[None, None, :] * stride_l_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_k
            + h[None, None, :] * stride_lg_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        left_res = left * mask_tile * left_gate_sig
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_k
            + h[None, None, :] * stride_l_h,
            left_res,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        # -------- RIGHT ----------
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_j
            + j[None, :, None] * stride_r_k
            + h[None, None, :] * stride_r_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_j
            + j[None, :, None] * stride_rg_k
            + h[None, None, :] * stride_rg_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
        right_res = right * mask_tile * right_gate_sig
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_j
            + j[0, :, None] * stride_r_k
            + h[None, None, :] * stride_r_h,
            right_res,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_gate_triton(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
):
    """In‑place fusion of mask, sigmoid‑gate and multiplication."""
    B, N, _, H = left_proj.shape
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the O(N³·H) TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B,
    N,
    H,
    stride_l_b,
    stride_l_i,
    stride_l_k,
    stride_l_h,
    stride_r_b,
    stride_r_j,
    stride_r_k,
    stride_r_h,
    stride_o_b,
    stride_o_i,
    stride_o_j,
    stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    with fp16 inputs and fp32 accumulation.
    """
    # (0) batch/head program id
    bh = tl.program_id(0)                # combined batch * head
    pid_i = tl.program_id(1)             # tile over i
    pid_j = tl.program_id(2)             # tile over j

    # decode batch and head indices
    b = bh // H
    h = bh % H

    # (1) tile coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    # base pointers for this (b, h)
    a_ptr = left_ptr + b * stride_l_b + h * stride_l_h
    b_ptr = right_ptr + b * stride_r_b + h * stride_r_h
    c_ptr = out_ptr + b * stride_o_b + h * stride_o_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # (2) reduction over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        k_ok = k < N

        # left tile  (i, k) -> shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        )
        # right tile (k, j) – note the transpose on the second dimension
        b = tl.load(
            b_ptr
            + k[:, None] * stride_r_k
            + j[None, :] * stride_r_j,
            mask=k_ok[:, None] & j_ok[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # (3) write result
    tl.store(
        c_ptr
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j,
        acc,
        mask=i_ok[:, None] & j_ok[None, :],
    )


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched contraction.
    left / right : [B, N, N, H]  (fp16)
    returns out   : [B, N, N, H]  (fp32)
    """
    B, N, _, H = left.shape
    # output in fp32
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # Tunable tile sizes – 64 works well for N≥256 and is still fine for smaller N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3 flavour).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float, optional)
        - weights      : dict of model weights
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``layernorm_eps`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                     # C
    hidden = config["hidden_dim"]           # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)               # [B, N, N, dim] fp16
    B, N, _, C = x_h.shape
    M = B * N * N                           # flattened spatial dimension

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gate linear (4 * hidden) via one GEMM
    # ------------------------------------------------------------------
    # concatenate projection and gate weights (each (hidden, dim))
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)  # (4*H, dim)

    # flatten spatial dims for a single matmul
    x_flat = x_h.view(M, C)                 # (M, dim)
    proj_gate = F.linear(x_flat, proj_gate_w)  # (M, 4*H)

    # split into the four tensors
    left_proj = proj_gate[:, 0:hidden]
    right_proj = proj_gate[:, hidden:2 * hidden]
    left_gate = proj_gate[:, 2 * hidden:3 * hidden]
    right_gate = proj_gate[:, 3 * hidden:4 * hidden]

    # reshape back to (B, N, N, H)
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate = left_gate.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (independent linear)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden)  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask + sigmoid gates (Triton fusion when mask present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float, cast to fp16
        mask_fp = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_gate_triton(left_proj, right_proj, left_gate, right_gate, mask_fp)
        left = left_proj
        right = right_proj
    else:
        # No mask – just apply sigmoid gates
        left = left_proj * torch.sigmoid(left_gate)
        right = right_proj * torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction (batched matmul) via Triton
    # ------------------------------------------------------------------
    # out_hidden: (B, N, N, hidden) float32
    out_hidden = _tri_mul_triton(left, right)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (float32) + out‑gate
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out_hidden.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out_hidden.dtype, device=device),
        eps=eps,
    )
    # apply out‑gate (cast gate to float32)
    out_hidden = out_hidden * out_gate.to(out_hidden.dtype)

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dim (fp16)
    # ------------------------------------------------------------------
    # Cast to fp16 for the final linear (Tensor Cores)
    out_hidden_fp16 = out_hidden.to(torch.float16)

    to_out_w = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out_fp16 = F.linear(out_hidden_fp16, to_out_w)   # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to fp32 (matching reference implementation)
    # ------------------------------------------------------------------
    return out_fp16.float()
