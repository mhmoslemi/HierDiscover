    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # decode batch and hidden
    b = pid // H
    h = pid % H
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    i_valid = i < N
    j_valid = j < N
    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    k = 0

        k_range = k + tl.arange(0, cur_k)

"""
Fast outgoing TriMul (AlphaFold‑3) – forward only
------------------------------------------------
1️⃣ Layer‑norm the input (FP32) → cast to fp16.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
   single GEMM (FP16).
3️⃣ Core O(N³·H) contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] *
                  sigmoid(left_gate[b,i,k,h]) *
                  sigmoid(right_gate[b,j,k,h]) *
                  mask[b,i,k] * mask[b,j,k]
   Implemented as a custom Triton kernel that:
   – loads tiles of left/right and their gate tensors,
   – applies sigmoid‑gate and mask on‑the‑fly,
   – accumulates the matrix‑multiply in FP32,
   – writes the result (FP16) for each hidden slice (batch×hidden).
   This fuses masking, gating and the O(N³) update, eliminating extra
   memory copies and kernel launches.
4️⃣ Hidden‑dim LayerNorm + output gate (sigmoid) and final linear projection
   back to the original channel dimension (all in FP16, accumulation in FP32).
5️⃣ Cast the final result to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: fused mask, sigmoid‑gate and O(N³) update
# -------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # input tensors
    left_ptr,               # (B,N,N,H) – left projection (FP16)
    right_ptr,              # (B,N,N,H) – right projection (FP16)
    left_gate_ptr,          # (B,N,N,H) – left‑gate pre‑sigmoid (FP16)
    right_gate_ptr,         # (B,N,N,H) – right‑gate pre‑sigmoid (FP16)
    mask_ptr,               # (B,N,N) – mask (FP16, 0/1)
    # output tensor
    out_ptr,                # (B,N,N,H) – result (FP16)

    # dimensions
    B, N, H,

    # strides for the four tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes one tile of the outgoing TriMul contraction for a specific
    (batch, hidden) pair. All tensors are half‑precision; accumulation is
    performed in FP32.
    """
    # ----------------------------
    # Decode fused batch‑hidden index
    # ----------------------------
    pid = tl.program_id(0)                # fused (b, h)
    pid_i = tl.program_id(1)              # tile over i
    pid_j = tl.program_id(2)              # tile over j

    b = pid // H
    h = pid % H

    # ----------------------------
    # Tile coordinates
    # ----------------------------
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # ----------------------------
    # Accumulator (FP32)
    # ----------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----------------------------
    # Loop over the reduction axis k
    # ----------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_offset = kk * BLOCK_K
        k = k_offset + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # ---- mask left (i,k) ----
        mask_left = tl.load(
            mask_ptr + b * stride_m_b
                      + i[:, None] * stride_m_i
                      + k[None, :] * stride_m_j,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        # ---- left projection ----
        left = tl.load(
            left_ptr + b * stride_l_b
                       + i[:, None] * stride_l_i
                       + k[None, :] * stride_l_k
                       + h * stride_l_h,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        # ---- left‑gate pre‑sigmoid ----
        left_gate_pre = tl.load(
            left_gate_ptr + b * stride_lg_b
                           + i[:, None] * stride_lg_i
                           + k[None, :] * stride_lg_k
                           + h * stride_lg_h,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        # ---- sigmoid for left gate (FP32) ----
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_pre.to(tl.float32)))
        # ---- cast left to FP32 and apply gate & mask ----
        left_f32 = left.to(tl.float32) * left_gate * mask_left.to(tl.float32)

        # ---- mask right (j,k) – note the transposed layout ----
        mask_right = tl.load(
            mask_ptr + b * stride_m_b
                      + k[:, None] * stride_m_j   # rows = k (second dim of mask)
                      + j[None, :] * stride_m_i,  # cols = j (first dim of mask)
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        # ---- right projection (k,j) ----
        right = tl.load(
            right_ptr + b * stride_r_b
                        + k[:, None] * stride_r_k   # rows = k
                        + j[None, :] * stride_r_i   # cols = j
                        + h * stride_r_h,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        # ---- right‑gate pre‑sigmoid (k,j) ----
        right_gate_pre = tl.load(
            right_gate_ptr + b * stride_rg_b
                            + k[:, None] * stride_rg_k
                            + j[None, :] * stride_rg_i
                            + h * stride_rg_h,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        # ---- sigmoid for right gate (FP32) ----
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_pre.to(tl.float32)))
        # ---- cast right to FP32 and apply gate & mask ----
        right_f32 = right.to(tl.float32) * right_gate * mask_right.to(tl.float32)

        # ---- multiply‑accumulate (matrix‑multiply) ----
        acc += tl.dot(left_f32, right_f32)   # (BLOCK_I, BLOCK_J)

    # ----------------------------
    # Store the result (FP16)
    # ----------------------------
    out_ptrs = out_ptr + b * stride_o_b \
                       + i[:, None] * stride_o_i \
                       + j[None, :] * stride_o_j \
                       + h * stride_o_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])


# -------------------------------------------------------------
# Helper to launch the fused contraction kernel
# -------------------------------------------------------------
def _fused_trimul(left_proj, right_proj,
                  left_gate_pre, right_gate_pre,
                  mask, hidden):
    """
    Executes the Triton kernel that fuses mask, sigmoid‑gate and the
    O(N³·H) update.

    Arguments are all FP16 tensors.
    Returns a tensor ``out`` of shape (B, N, N, hidden) in FP16.
    """
    B, N, _, H = left_proj.shape
    device = left_proj.device

    # strides (B,N,N,H)
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate_pre.stride()
    s_rg = right_gate_pre.stride()
    s_m = mask.stride()

    # output tensor
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    s_o = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _tri_mul_kernel[grid](
        left_proj,
        right_proj,
        left_gate_pre,
        right_gate_pre,
        mask,
        out,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# -------------------------------------------------------------
# Outgoing TriMul – forward‑only entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (AlphaFold‑3) with a fused Triton kernel for the
    heavy O(N³·H) update.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys:
                * dim            – input channel C (int)
                * hidden_dim     – hidden dimension H (int)
                * nomask         – bool, True ⇒ ignore mask
                * layernorm_eps  – optional epsilon for LayerNorm

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # C
    hidden = config["hidden_dim"]           # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    assert C == dim, f"Channel mismatch: got {C}, expected {dim}"

    # -----------------------------------------------------------------
    # Model weights (all FP32 on the same device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] float32

    # -----------------------------------------------------------------
    # 2️⃣ Switch to fp16 for the heavy path
    # -----------------------------------------------------------------
    x_h = x.half()  # [B, N, N, C] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Fuse four linear maps (left/right + gates) into one GEMM
    #    Output order: left_proj, right_proj, left_gate, right_gate
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_weight,
         right_proj_weight,
         left_gate_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)          # (4*H, C)

    x_flat = x_h.view(-1, dim)                         # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)        # (B·N·N, 4*H)

    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Prepare mask (FP16). If no mask or nomask, use all‑ones mask.
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction with fused mask & gates (Triton)
    # -----------------------------------------------------------------
    out_hidden = _fused_trimul(
        left_proj, right_proj,
        left_gate_pre, right_gate_pre,
        mask_fp16, hidden,
    )  # (B, N, N, H) fp16

    # -----------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H) fp16
    out_hidden = out_hidden * out_gate

    # -----------------------------------------------------------------
    # 7️⃣ Final projection back to the original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out_hidden, to_out_weight.half())  # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Return float32 tensor compatible with the reference implementation
    # -----------------------------------------------------------------
    return out.float()
