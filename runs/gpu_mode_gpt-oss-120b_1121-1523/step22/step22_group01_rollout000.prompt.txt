import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
tri_mul_triton.py
-----------------
Highly‑optimized Triton‑accelerated implementation of the *outgoing* TriMul
operator (AlphaFold‑3 style).

Algorithm
~~~~~~~~~
1. **LayerNorm** on the input (FP32, bias + weight).
2. Cast to **FP16** for all heavy arithmetic.
3. **Fused projection+gate linear layers** – a single GEMM produces
   `left_proj, right_proj, left_gate, right_gate` (4× hidden_dim).
4. **Optional mask + sigmoid gating** – a custom Triton kernel fuses
   mask multiplication and the sigmoid gate, writing the final
   `left` and `right` tensors in‑place.
5. **Core O(N³·H) update** – batched matrix multiplication
   `left @ rightᵀ` for each head (hidden dimension) using `torch.bmm`.
6. **Output LayerNorm** on the result.
7. **Element‑wise out‑gate** (sigmoid of a linear on the normalized input).
8. **Final linear projection** back to `dim`.
9. Return FP32 output.

The only Triton component is the mask‑gate fusion (step 4).  All
matrix multiplications are performed with Tensor‑Core‑enabled FP16
(GEMM) and FP32 accumulation, which is optimal on H100.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication, sigmoid gating and in‑place write
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_gb_b, stride_gb_i, stride_gb_j, stride_gb_h,
    stride_gt_b, stride_gt_i, stride_gt_j, stride_gt_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For each (b,i,j) tile:
        left[b,i,j,:]  = left_proj[b,i,j,:]  * mask[b,i,j] * sigmoid(left_gate[b,i,j,:])
        right[b,i,j,:] = right_proj[b,i,j,:] * mask[b,i,j] * sigmoid(right_gate[b,i,j,:])
    """
    b = tl.program_id(0)                     # batch dimension
    pid_i = tl.program_id(1)                 # tile over i
    pid_j = tl.program_id(2)                 # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load a mask tile (BLOCK_I x BLOCK_J) – mask is FP16
    # -----------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]  # broadcast over H

    # -----------------------------------------------------------------
    # Loop over hidden dimension in blocks of BLOCK_H
    # -----------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # ----- LEFT side -------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + b * stride_gb_b
            + i[:, None, None] * stride_gb_i
            + j[None, :, None] * stride_gb_j
            + h[None, None, :] * stride_gb_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # sigmoid
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        # apply mask and gate
        left_res = left * mask_tile * left_gate_sig
        # store result back into left_proj (in‑place)
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_res,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        # ----- RIGHT side ------------------------------------------------
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr
            + b * stride_gt_b
            + i[:, None, None] * stride_gt_i
            + j[None, :, None] * stride_gt_j
            + h[None, None, :] * stride_gt_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
        right_res = right * mask_tile * right_gate_sig
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_res,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        h_start += BLOCK_H


def _apply_mask_gate_triton(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
):
    """In‑place fusion of mask‑multiply and sigmoid‑gate for left/right tensors."""
    B, N, _, H = left_proj.shape
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8  # small enough to fit in shared memory

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Triton‑enabled TriMul forward (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (FP32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float, optional)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``layernorm_eps`` and ``nomask``.

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (FP32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # C
    hidden = config["hidden_dim"]       # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for numerical safety)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the rest of the pipeline works in half precision
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)           # (B, N, N, dim)
    B, N, _, C = x_h.shape
    M = B * N * N                        # flattened batch‑spatial size

    # ------------------------------------------------------------------
    # 3️⃣ One‑shot linear for left/right projections **and** all gates
    #    (weight shape: (4*hidden, dim))
    # ------------------------------------------------------------------
    # Concatenate weights in the order: left_proj, right_proj,
    # left_gate, right_gate
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, dim)

    # Flatten spatial dimensions for a single GEMM
    x_flat = x_h.view(M, C)                     # (M, dim)

    # One GEMM produces all four  tensors at once
    proj_gate = F.linear(x_flat, proj_gate_weight)   # (M, 4*H)

    # Split into the four components
    left_proj = proj_gate[:, 0 : hidden]
    right_proj = proj_gate[:, hidden : 2 * hidden]
    left_gate = proj_gate[:, 2 * hidden : 3 * hidden]
    right_gate = proj_gate[:, 3 * hidden : 4 * hidden]

    # Reshape back to (B, N, N, H)
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate = left_gate.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate linear (independent from the core matmul)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask + sigmoid gating fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is half‑precision and on the same device
        mask_fp = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_gate_triton(left_proj, right_proj, left_gate, right_gate, mask_fp)
        # After the kernel, left_proj/right_proj already contain:
        #   left_proj = left_proj * mask * sigmoid(left_gate)
        #   right_proj = right_proj * mask * sigmoid(right_gate)
        left = left_proj
        right = right_proj
    else:
        # No mask – just apply sigmoid gates
        left = left_proj * torch.sigmoid(left_gate)
        right = right_proj * torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update: batched matmul left @ rightᵀ for each head
    # ------------------------------------------------------------------
    # Re‑order to (B, H, N, N) → (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Batched GEMM (FP16 × FP16 → FP32 accumulation)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32 internal reduction) + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out.dtype, device=device),
        eps=eps,
    )
    # Apply out‑gate (computed earlier)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out = F.linear(out, to_out_weight)  # (B, N, N, dim)  FP16

    # ------------------------------------------------------------------
    # 9️⃣ Return FP32 tensor (matches reference implementation)
    # ------------------------------------------------------------------
    return out.float()
