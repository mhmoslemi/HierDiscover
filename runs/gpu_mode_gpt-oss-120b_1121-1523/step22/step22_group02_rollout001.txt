"""
Fast outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (FP32) → x.
2️⃣ Cast x to FP16 and fuse the four linear branches (left‑proj/right‑proj and the
   two gates) into one GEMM, producing tensors of shape [B,N,N,H] in FP16.
3️⃣ Apply sigmoid to the gate tensors (still FP16).  If a mask is present,
   multiply both projected tensors by the mask (broadcast over H) – this is a
   simple element‑wise torch kernel.
4️⃣ Core O(N³·H) update: for each hidden dimension perform a matrix‑multiply
   left[b,i,k,h] @ right[b,k,j,h] across the sequence dimension k.
   Implemented as a batched GEMM in Triton (batch = B·H) with FP16×FP16 → FP32
   accumulation to exploit H100 Tensor‑Cores.
5️⃣ Layer‑Norm over the hidden dimension, multiply by the output‑gate, project
   back to the original channel dimension and return FP32.
All heavy arithmetic stays in FP16 while the custom Triton kernel handles the
massive O(N³·H) work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Batched matmul kernel for the O(N³·H) update
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, N,
    stride_a_b, stride_a_i, stride_a_k,
    stride_b_b, stride_b_k, stride_b_j,
    stride_c_b, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j]  (FP16×FP16 → FP32)"""
    b = tl.program_id(0)          # batch index (B*H)
    pid_i = tl.program_id(1)      # block row
    pid_j = tl.program_id(2)      # block col

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_ok = k < N

        # Load A[i,k] tile
        a = tl.load(
            A_ptr
            + b * stride_a_b
            + i[:, None] * stride_a_i
            + k[None, :] * stride_a_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        )
        # Load B[k,j] tile
        b_tile = tl.load(
            B_ptr
            + b * stride_b_b
            + k[:, None] * stride_b_k
            + j[None, :] * stride_b_j,
            mask=k_ok[:, None] & j_ok[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b_tile)   # FP32 accumulation

    # Write result back (cast to FP16 for storage)
    tl.store(
        C_ptr
        + b * stride_c_b
        + i[:, None] * stride_c_i
        + j[None, :] * stride_c_j,
        acc.to(tl.float16),
        mask=i_ok[:, None] & j_ok[None, :],
    )


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """Batched GEMM for left @ right.  A/B/C have shape (BATCH, N, N) FP16."""
    assert A.is_cuda and B.is_cuda
    BATCH, N, _ = A.shape
    C = torch.empty_like(A)

    # Tunable block sizes – 64⁴ tiles give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BATCH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    s_a = A.stride()
    s_b = B.stride()
    s_c = C.stride()

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        BATCH,
        N,
        s_a[0],
        s_a[1],
        s_a[2],
        s_b[0],
        s_b[1],
        s_b[2],
        s_c[0],
        s_c[1],
        s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------
# 3️⃣ Entry‑point custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Model weights (already on GPU)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear branches (FP16)
    # ------------------------------------------------------------------
    x_h = x.half()          # [B, N, N, C]  fp16
    B, N, _, _ = x_h.shape

    # Combine weights for a single GEMM: (4*H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=x_h.device)

    # Linear over flattened [B*N*N, C] → [B*N*N, 4*H]
    x_flat = x_h.view(-1, dim)               # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to same dtype and broadcast over hidden dim
        mask_h = mask_tensor.to(dtype=left.dtype)
        left = left * mask_h[..., None]
        right = right * mask_h[..., None]

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update via batched Triton GEMM
    #    left  : (B, N, N, H) → (B*H, N, N) where dim1 = i, dim2 = k
    #    right : (B, N, N, H) → (B*H, N, N) where dim1 = k, dim2 = j
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    out_mat = _batched_matmul_triton(left_mat, right_mat)   # (B*H, N, N) fp16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm, output‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H) fp16
    out = out * out_gate

    out = F.linear(out, to_out_weight.half())   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # Return FP32 (matches reference)
    # ------------------------------------------------------------------
    return out.float()
