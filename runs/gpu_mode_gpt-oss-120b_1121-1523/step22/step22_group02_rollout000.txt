import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          BATCH, N,
                          stride_a_batch, stride_a_i, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_j,
                          stride_c_batch, stride_c_i, stride_c_j,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        a = tl.load(A_ptr + b * stride_a_batch + i[:, None] * stride_a_i + k[None, :] * stride_a_k,
                    mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b_tile = tl.load(B_ptr + b * stride_b_batch + k[:, None] * stride_b_k + j[None, :] * stride_b_j,
                         mask=k_mask[:, None] & j_mask[None, :], other=0.0)
        acc += tl.dot(a, b_tile)
    # store
    tl.store(C_ptr + b * stride_c_batch + i[:, None] * stride_c_i + j[None, :] * stride_c_j,
             acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    ...
    # Input layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # FP16 conversion
    x_h = x.half()
    B,N,_,_ = x_h.shape

    # Combine weights
    combined_weight = torch.cat([left_proj_weight, left_gate_weight,
                                 right_proj_weight, right_gate_weight], dim=0).to(dtype=torch.float16)

    # Linear
    x_flat = x_h.view(-1, dim)
    combined = F.linear(x_flat, combined_weight)   # [B*N*N, 4*hidden_dim]
    combined = combined.view(B, N, N, 4*hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(combined, hidden_dim, dim=-1)

    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    if not nomask:
        # mask shape [B, N, N] -> broadcast to hidden dim
        mask_f = mask_tensor.to(dtype=left.dtype)
        left = left * mask_f[..., None]
        right = right * mask_f[..., None]

    # Reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # For right: need shape (B, N, N, H) where second axis is j dimension; we need (B, H, N, N) with inner order for dot as (k,j)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = _batched_matmul_triton(left_mat, right_mat)   # (B*H,N,N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight.half(),
                       bias=to_out_norm_bias.half(), eps=eps)

    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)

    out = out * out_gate

    out = F.linear(out, to_out_weight.half())

    return out.float()

"""
Fast Triton TriMul (outgoing) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (FP32).
2️⃣ Cast to FP16 and fuse the four linear branches (proj+gate) into a single GEMM.
3️⃣ Apply sigmoid gates (still FP16) and optionally mask the tensors (broadcast over H).
4️⃣ Core O(N³·H) update: a batched matmul kernel that computes,
      out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   for every hidden channel h (implemented as B·H independent N×N @ N×N matmuls in
   FP16 with FP32 accumulation, using larger 128×128 tiles and double‑buffered K‑loop).
5️⃣ Output Layer‑Norm, output‑gate, final projection back to the original channel dimension.
All heavy arithmetic stays in FP16 to exploit H100 Tensor‑Cores.  The final result
is returned as FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Batched matmul kernel for O(N³·H) update
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j]   (FP16 × FP16 → FP32 accumulation)
    Shapes for each tensor: (BATCH, N, N) in row‑major order.
    """
    b = tl.program_id(0)               # batch (B*H)
    pid_i = tl.program_id(1)           # tile row
    pid_j = tl.program_id(2)           # tile col

    # tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # --------------------------------------------------------------
    # Double‑buffered K‑loop (prefetch next tile while computing current)
    # --------------------------------------------------------------
    for k_tile in range(num_k):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_ok = k < N

        # A tile (i, k) – contiguously stored in memory
        a = tl.load(
            A_ptr + b * stride_a_batch
                     + i[:, None] * stride_a_i
                     + k[None, :] * stride_a_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        )
        # B tile (k, j)
        b_tile = tl.load(
            B_ptr + b * stride_b_batch
                     + k[:, None] * stride_b_k
                     + j[None, :] * stride_b_j,
            mask=k_ok[:, None] & j_ok[None, :],
            other=0.0,
        )
        # Dot product over the K dimension (Tensor‑Core friendly)
        acc += tl.dot(a, b_tile)

    # Store the result (down‑cast to FP16)
    tl.store(
        C_ptr + b * stride_c_batch
               + i[:, None] * stride_c_i
               + j[None, :] * stride_c_j,
        acc.to(tl.float16),
        mask=i_ok[:, None] & j_ok[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched (B*H, N, N) matmul: left @ right (both FP16).

    Args:
        left, right: tensors of shape (B*H, N, N), dtype=torch.float16, contiguous.
    Returns:
        Tensor of same shape and dtype = torch.float16.
    """
    assert left.is_cuda and right.is_cuda
    BATCH, N, _ = left.shape
    out = torch.empty_like(left)

    # Tile sizes – chosen to fully occupy Tensor‑Cores on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        BATCH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    s_a = left.stride()
    s_b = right.stride()
    s_c = out.stride()

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        BATCH,
        N,
        s_a[0],
        s_a[1],
        s_a[2],
        s_b[0],
        s_b[1],
        s_b[2],
        s_c[0],
        s_c[1],
        s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,      # pipelined pre‑fetching of K‑tiles
    )
    return out


# ----------------------------------------------------------------------
# 3️⃣ Entry‑point custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack & config
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Model weights (already on GPU)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]            # (C,)
    norm_bias = weights["norm.bias"]                # (C,)

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four linear branches (FP16)
    # ------------------------------------------------------------------
    x_h = x.half()                                 # FP16 view
    B, N, _, _ = x_h.shape

    # Concatenate the four weight matrices → one GEMM
    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16)                      # (4*H, C)

    # Linear over flattened [B*N*N, C] → [B*N*N, 4*H]
    x_flat = x_h.view(-1, dim)                     # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)   # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    # Split back into four tensors
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each: [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Sigmoid gates (still FP16) and optional mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    if not nomask:
        # Broadcast mask over the hidden dimension (FP16)
        mask_fp16 = mask_tensor.to(dtype=left.dtype)
        left = left * mask_fp16[..., None]   # shape (B,N,N,1) → (B,N,N,H)
        right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – batched matmul via Triton
    #    left:  (B, N, N, H) → (B*H, N, N)  where inner axes are (i, k)
    #    right: (B, N, N, H) → (B*H, N, N)  where inner axes are (k, j)
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = _batched_matmul_triton(left_mat, right_mat)   # (B*H, N, N), FP16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, output‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )   # FP16

    # Output gate is computed on the original (FP32) normed tensor for higher precision
    out_gate_fp32 = torch.sigmoid(F.linear(x, out_gate_weight))      # (B, N, N, H), FP32
    out_gate = out_gate_fp32.to(dtype=out.dtype)                     # cast to FP16
    out = out * out_gate

    out = F.linear(out, to_out_weight.half())   # (B, N, N, C), FP16

    # ------------------------------------------------------------------
    # Return FP32 (matches reference)
    # ------------------------------------------------------------------
    return out.float()
