import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Fast outgoing TriMul (AlphaFold‑3) – forward only

Algorithm
---------
1️⃣ Layer‑norm the input (FP32) → FP16.
2️⃣ Fuse the four linear layers (left/right + their gates) into a single
   GEMM that outputs 4×hidden (`left_proj, right_proj, left_gate, right_gate`).
3️⃣ If a mask is provided, a single Triton kernel fuses *masking*,
   *sigmoid‑gate* and element‑wise multiplication:
   left = left_proj * sigmoid(left_gate) * mask
   right = right_proj * sigmoid(right_gate) * mask
   (In the “nomask’’ case we only apply the sigmoid‑gate with PyTorch.)
4️⃣ Core O(N³·H) update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched FP16 matrix‑multiply
   (torch.matmul on tensors of shape (B, H, N, N)).
5️⃣ Hidden‑dim LayerNorm, output‑gate (sigmoid) and final linear projection
   back to the original channel dimension.
6️⃣ Cast the result back to FP32.

Only the mask‑gate step is written in Triton (required by the problem).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid‑gate and in‑place multiplication
# -------------------------------------------------------------
@triton.jit
def _mask_gate_mul_kernel(
    # tensor pointers
    left_ptr, right_ptr,                # (B,N,N,H) – will be overwritten with gated values
    left_gate_ptr, right_gate_ptr,      # (B,N,N,H) – pre‑gate values
    mask_ptr,                           # (B,N,N) – 0/1 mask (half)

    # dimensions
    B, N, H,
    # strides
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    In‑place:
        left[b,i,j,h]  = left_proj[b,i,j,h] *
                         sigmoid(left_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
        right[b,i,j,h] = right_proj[b,i,j,h] *
                         sigmoid(right_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
    All tensors are assumed to be half‑precision.
    """

    b = tl.program_id(0)                 # batch dimension
    pid_i = tl.program_id(1)             # tile over i
    pid_j = tl.program_id(2)             # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # -----------------------------------------------------------------
    # Load the (i,j) mask tile once and broadcast over H
    # -----------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )                     # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]    # (BLOCK_I, BLOCK_J, 1) – broadcast over H

    # -----------------------------------------------------------------
    # Loop over the hidden dimension in chunks of BLOCK_H
    # -----------------------------------------------------------------
    h_offset = 0
    while h_offset < H:
        h = h_offset + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # Load left_proj, right_proj, left_gate_pre, right_gate_pre
        left   = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right  = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        left_g = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right_g = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Sigmoid (computed in FP32 for stability, then cast back)
        # -----------------------------------------------------------------
        left_gate  = 1.0 / (1.0 + tl.exp(-left_g.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_g.to(tl.float32)))
        left_gate  = left_gate.to(tl.float16)
        right_gate = right_gate.to(tl.float16)

        # Apply mask and gate
        left_out  = left * left_gate * mask_tile
        right_out = right * right_gate * mask_tile

        # Store back in‑place
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )

        h_offset += BLOCK_H

                                left_gate_pre, right_gate_pre,

    In‑place fusion of mask, sigmoid‑gate and multiplication.
    All tensors are FP16 and have shape (B, N, N, H) except mask (B, N, N).
    """
    B, N, _, H = left_proj.shape

    # strides for the four tensors
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate_pre.stride()
    s_rg = right_gate_pre.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

        left_proj,
        right_proj,
        left_gate_pre,
        right_gate_pre,

        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

# -------------------------------------------------------------
# Outgoing TriMul – forward‑only entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys
                * dim            – input channel C (int)
                * hidden_dim     – hidden dimension H (int)
                * nomask         – bool, True ⇒ ignore mask
                * layernorm_eps  – optional epsilon for LayerNorm

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                   # C
    hidden = config["hidden_dim"]         # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    assert C == dim, f"Channel dimension mismatch (got {C}, expected {dim})"

    # -----------------------------------------------------------------
    # Model weights (all FP32 on the same device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – keeps numerical stability
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Switch to fp16 for the heavy path
    # -----------------------------------------------------------------
    x_h = x.half()                         # [B, N, N, C] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Fuse left/right projections + their gates into a single GEMM
    #    Output order: left_proj, right_proj, left_gate_pre, right_gate_pre
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)      # (4*H, C)

    x_flat = x_h.view(-1, dim)                     # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)    # (B·N·N, 4*H)

    # Split and reshape back to (B, N, N, H)
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – fused mask + sigmoid‑gate (Triton)
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Convert mask to half (0.0 / 1.0)
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_and_gate_triton(
            left_proj, right_proj,
            left_gate_pre, right_gate_pre,
            mask_h,
        )
        # left_proj/right_proj now hold the gated & masked tensors
    else:
        # No mask: only apply sigmoid‑gate (pure PyTorch – cheap)
        left_gate = left_gate_pre.sigmoid()
        right_gate = right_gate_pre.sigmoid()
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # At this point:
    #   left_proj == left  , right_proj == right   (both FP16, shape B,N,N,H)

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction (batched GEMM)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) – make the tensors contiguous for cuBLAS
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Batched matrix multiplication: (B,H,N,N) @ (B,H,N,N)ᵀ → (B,H,N,N)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Bring back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1)   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    # Output gate (sigmoid) computed from the *original* normalized input
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7️⃣ Final projection back to the original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Return float32 tensor (compatible with reference implementation)
    # -----------------------------------------------------------------
    return out.float()
