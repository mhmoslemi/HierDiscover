import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Fast outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (FP32) → x.
2️⃣ Cast x to FP16 and fuse the four linear branches
   (left‑proj, left‑gate, right‑proj, right‑gate) into one GEMM.
3️⃣ Apply sigmoid to the two gate tensors (still FP16).
4️⃣ If a mask is present, an in‑place Triton kernel multiplies both
   projected tensors by the mask (broadcast over the hidden dimension).
5️⃣ Core O(N³·H) update is a batched matrix‑multiply:
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   realized as a custom Triton kernel that performs a batch of
   (B·H) 𝑁×𝑁 @ 𝑁×𝑁 matmuls in FP16 (accumulate in FP32).
6️⃣ Layer‑Norm over the hidden dimension, multiply by the output‑gate,
   project back to the original channel dimension and return FP32.
All heavy arithmetic stays in FP16 to exploit H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ In‑place mask kernel (broadcast over hidden dim H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_lr_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)
    ti = tl.program_id(1)
    tj = tl.program_id(2)

    i = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    j = tj * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load 2‑D mask tile
    mask = tl.load(
        mask_ptr + b * stride_m_b
                  + i[:, None] * stride_m_i
                  + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    num_h = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h):
        cur_h = hi * BLOCK_H + tl.arange(0, BLOCK_H)
        h_ok = cur_h < H

        # left
        left = tl.load(
            left_ptr + b * stride_l_b
                     + i[:, None, None] * stride_l_i
                     + j[None, :, None] * stride_l_j
                     + cur_h[None, None, :] * stride_l_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left = left * mask
        tl.store(
            left_ptr + b * stride_l_b
                     + i[:, None, None] * stride_l_i
                     + j[None, :, None] * stride_l_j
                     + cur_h[None, None, :] * stride_l_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        # right
        right = tl.load(
            right_ptr + b * stride_r_b
                      + i[:, None, None] * stride_r_i
                      + j[None, :, None] * stride_r_j
                      + cur_h[None, None, :] * stride_r_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right = right * mask
        tl.store(
            right_ptr + b * stride_r_b
                      + i[:, None, None] * stride_r_i
                      + j[None, :, None] * stride_r_j
                      + cur_h[None, None, :] * stride_r_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )


def _apply_mask_lr_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place mask multiplication for (left, right) tensors (mask broadcast over H)."""
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    mask = mask.to(dtype=left.dtype)

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    _mask_lr_kernel[grid](
        left, right, mask,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # tensors are modified in‑place


# ----------------------------------------------------------------------
# 2️⃣ Batched matmul kernel for the O(N³·H) update
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, N,
    stride_a_b, stride_a_i, stride_a_k,
    stride_b_b, stride_b_k, stride_b_j,
    stride_c_b, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j]   (FP16 × FP16 → FP32 accumulation)
    Shapes: A/B/C – (BATCH, N, N)
    """
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_ok = k < N

        # A tile (i, k)
        a = tl.load(
            A_ptr
            + b * stride_a_b
            + i[:, None] * stride_a_i
            + k[None, :] * stride_a_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        )
        # B tile (k, j)
        b_tile = tl.load(
            B_ptr
            + b * stride_b_b
            + k[:, None] * stride_b_k
            + j[None, :] * stride_b_j,
            mask=k_ok[:, None] & j_ok[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b_tile)   # acc is FP32, dot result is cast automatically

    tl.store(
        C_ptr
        + b * stride_c_b
        + i[:, None] * stride_c_i
        + j[None, :] * stride_c_j,
        acc.to(tl.float16),
        mask=i_ok[:, None] & j_ok[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched (B*H, N, N) matmul: left @ right  (both FP16)."""
    assert left.is_cuda and right.is_cuda
    BATCH, N, _ = left.shape
    out = torch.empty_like(left)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BATCH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    s_a = left.stride()
    s_b = right.stride()
    s_c = out.stride()

    _batched_matmul_kernel[grid](
        left, right, out,
        BATCH, N,
        s_a[0], s_a[1], s_a[2],
        s_b[0], s_b[1], s_b[2],
        s_c[0], s_c[1], s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# 3️⃣ Entry‑point custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (already on GPU)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                         # [B, N, N, C]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four linear branches (FP16)
    # ------------------------------------------------------------------
    x_h = x.half()           # [B, N, N, C]  fp16
    B, N, _, _ = x_h.shape

    # Concatenate weight matrices for a single GEMM
    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16)          # (4*H, C)

    # Linear over flattened [B*N*N, C] → [B*N*N, 4*H]
    x_flat = x_h.view(-1, dim)        # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each: [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_lr_triton(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM via Triton
    #    left  : (B, N, N, H) → (B*H, N, N)  where axis0 = i, axis1 = k
    #    right : (B, N, N, H) → (B*H, N, N)  where axis0 = k, axis1 = j
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = _batched_matmul_triton(left_mat, right_mat)   # (B*H, N, N) fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm, output‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )                         # fp16

    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H) fp16
    out = out * out_gate

    out = F.linear(out, to_out_weight.half())  # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # Return FP32 (matches reference)
    # ------------------------------------------------------------------
    return out.float()
