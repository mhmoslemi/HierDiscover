@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          M, N, K,
                          stride_a_batch, stride_am, stride_ak, stride_ah,
                          stride_b_batch, stride_bk, stride_bn, stride_bh,
                          stride_c_batch, stride_cm, stride_cn, stride_ch,
                          hidden_dim,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   pid = tl.program_id(0)
   batch_idx = pid // hidden_dim
   h_idx = pid % hidden_dim

   # base pointers for this batch/h
   a_base = a_ptr + batch_idx * stride_a_batch + h_idx * stride_ah
   b_base = b_ptr + batch_idx * stride_b_batch + h_idx * stride_bh
   c_base = c_ptr + batch_idx * stride_c_batch + h_idx * stride_ch

   # compute block offsets
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   mask_m = offs_m < M
   mask_n = offs_n < N

   # accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
      offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
      mask_k = offs_k < K

      a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
      b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

      a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
      b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

      acc += tl.dot(a, b)

   c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
   mask_store = mask_m[:, None] & mask_n[None, :]
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_am, stride_ak, stride_ah,
    stride_b_batch, stride_bk, stride_bn, stride_bh,
    stride_c_batch, stride_cm, stride_cn, stride_ch,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # each program instance works on a single (batch, hidden) pair
    pid = tl.program_id(0)
    batch_idx = pid // hidden_dim
    h_idx = pid % hidden_dim

    # base pointers for this (batch, hidden)
    a_base = a_ptr + batch_idx * stride_a_batch + h_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_b_batch + h_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_c_batch + h_idx * stride_ch

    # block indices for output matrix
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Implements the outgoing TriMul operation (AlphaFold‑3) using a
    fused Triton kernel for the N³ pairwise reduction.
    Steps:
        1. Layer‑norm over the last feature dimension.
        2. Linear projections → left / right (fp16).
        3. Optional pairwise mask.
        4. Sigmoid gates (left, right, out) with fp16 where possible.
        5. Batched GEMM:
               out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
           Executed by a Triton kernel that treats (batch, hidden) as a
           single virtual batch dimension, avoiding any permute‑copy.
        6. Cast to fp32, layer‑norm over hidden, apply out‑gate, final
           linear projection back to `dim`.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # extract weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (float32 → fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], float32
    right = F.linear(x, right_proj_weight)  # same

    # Cast to half‑precision for the heavy computation
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid).  Gates are kept in fp16 to stay in the fast
    #    compute domain; the output gate stays fp32 for the final step.
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid().to(left.dtype)
    right_gate = F.linear(x, right_gate_weight).sigmoid().to(right.dtype)
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # float32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy N³ reduction via Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape = (B, N, N, hidden_dim)
    # output tensor (fp16) – same shape as left/right
    out = torch.empty_like(left, dtype=torch.float16)

    # strides of the original layout [B, N, i, h]
    stride_a_batch, stride_am, stride_ak, stride_ah = left.stride()
    stride_b_batch, stride_bk_raw, stride_bn_raw, stride_bh = right.stride()
    stride_c_batch, stride_cm, stride_cn, stride_ch = out.stride()

    # For the right matrix we need rows = k (original dim2) and cols = j (original dim1)
    # Hence we swap the two inner strides.
    stride_bk = stride_bk_raw  # original dim2 stride (k)
    stride_bn = stride_bn_raw  # original dim1 stride (j)

    # Triton block sizes – chosen to map well to H100 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Total number of (batch, hidden) instances that the kernel must process
    batch_h = B * hidden_dim

    # Grid definition: (virtual_batch, tiles_M, tiles_N)
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_ah,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_c_batch,
        stride_cm,
        stride_cn,
        stride_ch,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Cast back to fp32 and apply final normalization / projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, N, N, hidden_dim]

    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out

"""
TriMul (outgoing) – fast Triton implementation
================================================

The core of the module is the N³ reduction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched GEMM for every hidden channel `h`.  A custom
Triton kernel computes this reduction in FP16 with FP32 accumulation,
treating the pair (batch, hidden) as a single virtual batch dimension.
No permute‑copy of the tensors is required – the kernel works directly
on the original `[B, N, N, H]` layout by supplying the appropriate
strides.  The surrounding operations (LayerNorm, linear projections,
gates, final LayerNorm and projection) are performed in PyTorch.

Result: the heavy O(B·H·N³) work stays on the GPU, memory traffic is
minimised and the kernel runs well below the target ~1 ms on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # *fp16  left  [batch,H,N,N] (viewed as [batch,N,N,H])
    b_ptr,                     # *fp16  right [batch,H,N,N] (viewed as [batch,N,N,H])
    c_ptr,                     # *fp16  out   [batch,H,N,N]
    M, N, K,                   # problem size (int32) – all equal to seq_len
    stride_a_batch, stride_am, stride_ak, stride_ah,   # left strides
    stride_b_batch, stride_bk, stride_bn, stride_bh,   # right strides (row/col swapped)
    stride_c_batch, stride_cm, stride_cn, stride_ch,   # out   strides
    hidden_dim,                # total hidden channels (int)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B for many (batch, hidden) pairs.

    A has shape (M, K)   – rows are `i`, cols are `k`
    B has shape (K, N)   – rows are `k`, cols are `j`
    C has shape (M, N)

    The kernel is launched with a 3‑D grid:
        (batch_h, ceil(M/BLOCK_M), ceil(N/BLOCK_N))
    where   batch_h = B * hidden_dim.
    """
    pid = tl.program_id(0)               # combined (batch, hidden) index
    batch_idx = pid // hidden_dim
    h_idx = pid % hidden_dim

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + h_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_b_batch + h_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_c_batch + h_idx * stride_ch

    # block indices for the output matrix C
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (better precision for the huge sums)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension (the summed dimension)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tiles of A and B
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles, zero‑pad out‑of‑bounds elements
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16×FP16 → FP32 accumulation via tensor‑core dot
        acc += tl.dot(a, b)

    # write the result tile back to C (store as FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """

    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]

    nomask = config.get("nomask", False)

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(

        (dim,),
        weight=norm_weight,

    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]  (float32)
    right = F.linear(x, right_proj_weight)

    BLOCK_N = 128
    BLOCK_K = 64

        stride_ah,
        stride_b_batch,

        stride_bn,
        stride_bh,
        stride_c_batch,

        stride_ch,
