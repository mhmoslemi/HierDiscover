@triton.jit
def _apply_mask_lr_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)
    ti = tl.program_id(1)
    tj = tl.program_id(2)

    i = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    j = tj * BLOCK_J + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_mask = i < N
    j_mask = j < N
    h_mask = h < H

    # mask tile (i,j)
    mask = tl.load(mask_ptr + b*stride_m_b + i[:,None]*stride_m_i + j[None,:]*stride_m_j,
                   mask=i_mask[:,None] & j_mask[None,:], other=0.0)  # (BLOCK_I,BLOCK_J)
    mask = mask[:, :, None]  # broadcast over H

    # load left tile
    left = tl.load(left_ptr + b*stride_l_b + i[:,None,None]*stride_l_i + j[None,:,None]*stride_l_j + h[None,None,:]*stride_l_h,
                   mask=i_mask[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:],
                   other=0.0)
    left = left * mask
    tl.store(left_ptr + b*stride_l_b + i[:,None,None]*stride_l_i + j[None,:,None]*stride_l_j + h[None,None,:]*stride_l_h,
             left,
             mask=i_mask[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:])

    # load right tile
    right = tl.load(right_ptr + b*stride_r_b + i[:,None,None]*stride_r_i + j[None,:,None]*stride_r_j + h[None,None,:]*stride_r_h,
                   mask=i_mask[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:],
                   other=0.0)
    right = right * mask
    tl.store(right_ptr + b*stride_r_b + i[:,None,None]*stride_r_i + j[None,:,None]*stride_r_j + h[None,None,:]*stride_r_h,
              right,
              mask=i_mask[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:])
```

    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,

"""
Fast outgoing TriMul forward pass (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (float32).
2️⃣ Cast to FP16 and fuse the four linear branches
   (left‑proj, left‑gate, right‑proj, right‑gate) into one GEMM.
3️⃣ Apply sigmoid gates, then (optional) mask the two projected tensors.
   The mask‑multiplication is performed by a tiny Triton kernel that
   updates *both* tensors in‑place, satisfying the “must‑use‑kernel” rule.
4️⃣ Core O(N³·H) update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h].
   This is realized as a batched matrix‑multiply:
   (B·H, N, N) @ (B·H, N, N) → (B·H, N, N) which is far faster than
   a raw ein­summ.
5️⃣ Layer‑Norm over the hidden dimension, multiply by the output gate
   (computed from the normalized input) and a final FP16 linear projection
   back to the original channel dimension.  The result is returned as FP32.

All arithmetic (except the two LayerNorms) stays in FP16 so the kernel
uses Tensor‑Core friendly ops on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – in‑place mask multiplication for the two projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_lr_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    """x_left[b,i,j,h] *= mask[b,i,j]; x_right[b,i,j,h] *= mask[b,i,j]"""
    b  = tl.program_id(0)          # batch
    ti = tl.program_id(1)          # tile i
    tj = tl.program_id(2)          # tile j

    i = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    j = tj * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # ------------------------------------------------------------------
    # Load the 2‑D mask tile (broadcast later over H)
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr + b * stride_m_b
                  + i[:, None] * stride_m_i
                  + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                               # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]         # (BLOCK_I, BLOCK_J, 1)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension (H) in BLOCK_H‑wide chunks
    # ------------------------------------------------------------------
    num_h = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h):
        cur_h = hi * BLOCK_H + tl.arange(0, BLOCK_H)
        h_ok = cur_h < H

        # ----- left ----------------------------------------------------
        left = tl.load(
            left_ptr + b * stride_l_b
                     + i[:, None, None] * stride_l_i
                     + j[None, :, None] * stride_l_j
                     + cur_h[None, None, :] * stride_l_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left = left * mask
        tl.store(
            left_ptr + b * stride_l_b
                     + i[:, None, None] * stride_l_i
                     + j[None, :, None] * stride_l_j
                     + cur_h[None, None, :] * stride_l_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        # ----- right ---------------------------------------------------
        right = tl.load(
            right_ptr + b * stride_r_b
                      + i[:, None, None] * stride_r_i
                      + j[None, :, None] * stride_r_j
                      + cur_h[None, None, :] * stride_r_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right = right * mask
        tl.store(
            right_ptr + b * stride_r_b
                      + i[:, None, None] * stride_r_i
                      + j[None, :, None] * stride_r_j
                      + cur_h[None, None, :] * stride_r_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )


def _apply_mask_lr_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place mask multiplication for (left,right) tensors (mask broadcast over H)."""
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    mask = mask.to(dtype=left.dtype)            # same dtype (fp16)

    # Tile sizes – tuned for H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64                 # ≤ hidden_dim; will loop if H > BLOCK_H

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    _mask_lr_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # tensors are updated in‑place


# ----------------------------------------------------------------------
# Triton‑enabled forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Weights (assumed already on the correct device)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32, kept for numeric stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                       # [B, N, N, C]   fp32

    # --------------------------------------------------------------
    # 2️⃣ FP16 cast + fused projection (produces left/right & gates)
    # --------------------------------------------------------------
    x_h = x.half()          # [B, N, N, C]   fp16

    # Concatenate the four linear weight matrices once
    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16)                     # (4*H, C)

    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                    # (B*N*N, C)

    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – fused Triton kernel (broadcast over H)
    # --------------------------------------------------------------
    if not nomask:
        _apply_mask_lr_triton(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update  – batched GEMM instead of einsum
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for a single bmm call
    left_perm = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right needs to be transposed on the last two axes (j,k) → (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_perm, right_perm)                     # (B*H, N, N) fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)    # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, output‑gate and final projection (stay fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )   # fp16

    # output gate – computed from the *normalized* input (same x_h)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))   # (B,N,N,H) fp16
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, to_out_weight.half())                         # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # Return FP32 tensor (matches reference implementation)
    # --------------------------------------------------------------
    return out.float()
