```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
Highly‑optimized Triton‑aware forward pass for the *outgoing* TriMul operator.

Algorithm (FP16/BF16 everywhere except the tiny LayerNorms that stay in FP32):
1. Layer‑Norm the input (FP32) – small cost, numerically stable.
2. Cast to half‑precision and perform ONE fused linear projection that yields
   left_proj, left_gate, right_proj, right_gate in a single matmul.
3. In‑place sigmoid on the gate tensors and fuse them into the projections:
   left = left_proj * sigmoid(left_gate) ,  right = right_proj * sigmoid(right_gate)
4. (Optional) apply the pairwise mask in‑place with a tiny Triton kernel.
   The mask is broadcast over the hidden dimension.
5. Core O(N³) update:
   – view tensors as (B, H, N, N) (no copy) and call batched half‑precision GEMM.
   – result is permuted back to (B, N, N, H) and kept in half‑precision.
6. Output LayerNorm + output‑gate (sigmoid) + final linear projection back to `dim`
   are all performed in half‑precision; only the final cast back to FP32 is done
   at the very end.
Only the mask‑application is written in Triton; everything else uses
high‑throughput PyTorch kernels (fp16 GEMM, fused linear, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    mask_ptr,               # fp16 *
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Multiply left/right [B,N,N,H] by mask [B,N,N] (broadcast over H)."""
    b = tl.program_id(0)          # batch index
    i_tile = tl.program_id(1)     # i‑tile
    j_tile = tl.program_id(2)     # j‑tile

    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # ----- load mask (scalar per i,j) -----
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # ----- load left tile -----
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
        other=0.0,
    )
    # ----- load right tile -----
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
        other=0.0,
    )

    # ----- apply mask (broadcast over H) -----
    left = left * mask
    right = right * mask

    # ----- store back -----
    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
    )


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place elementwise multiplication of `left`/`right` by `mask`
    using a small Triton kernel. All tensors are expected to be
    half‑precision and on the same CUDA device.
    """
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    mask_fp16 = mask.to(dtype=left.dtype)

    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_fp16.stride()

    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 128                     # hidden_dim ≤ 384 → max 384, we cap at 128 per block

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_apply_kernel[grid](
        left,
        right,
        mask_fp16,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left & right are modified in‑place


def custom_kernel(data):
    """
    Triton‑enhanced forward pass for the *outgoing* TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Load model weights (all stored as float32 on GPU)
    # --------------------------------------------------------------
    # Input LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Fused projection weights – concatenate for a single matmul
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]      # (hidden_dim, dim)

    # Output side
    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]       # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 3) Input LayerNorm (still in FP32 for numerical stability)
    # --------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 4) Cast to FP16 and perform a single fused linear projection
    # --------------------------------------------------------------
    x_fp16 = x_norm_fp32.half()   # cast once

    # Stack the four weight matrices → (4*H, dim)
    combined_weight_fp16 = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=x_fp16.device)

    # Linear: (B,N,N,dim) @ (dim,4*H)^T → (B,N,N,4*H)
    combined = F.linear(x_fp16, combined_weight_fp16)   # (B,N,N,4*H)

    # Split without copy
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each -> (B,N,N,H) fp16

    # In‑place sigmoid gates (fp16)
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()

    # Fuse gates into projections (still fp16)
    left = left_proj.mul_(left_gate)      # left now holds gated projection
    right = right_proj.mul_(right_gate)   # right holds gated projection

    # --------------------------------------------------------------
    # 5) Optional mask (in‑place via Triton)
    # --------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 6) Core O(N³) update – batched GEMM (FP16)
    # --------------------------------------------------------------
    # left  : (B,N,N,H) → (B,H,N,N)
    # right : (B,N,N,H) → (B,H,N,N) with last two axes transposed
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)        # (B, H, N, N)

    # Batched matmul over the (B·H) dimension
    out_perm = torch.matmul(left_perm, right_perm)  # (B, H, N, N) fp16

    # Back to (B,N,N,H) and keep fp16 for the remaining ops
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm + output‑gate + final projection (all FP16)
    # --------------------------------------------------------------
    # Output LayerNorm (cast weights to fp16)
    to_out_norm_weight_fp16 = to_out_norm_weight.half()
    to_out_norm_bias_fp16   = to_out_norm_bias.half()

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
        eps=eps,
    )   # still fp16

    # Output gate derived from the *normed* input (use fp16 copy)
    x_norm_fp16 = x_norm_fp32.half()
    out_gate_weight_fp16 = out_gate_weight.half()
    out_gate = torch.sigmoid(F.linear(x_norm_fp16, out_gate_weight_fp16))   # (B,N,N,H) fp16
    out = out * out_gate

    # Final linear back to original dim (FP16 weight)
    to_out_weight_fp16 = to_out_weight.half()
    out = F.linear(out, to_out_weight_fp16)   # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 8) Cast back to FP32 (required by downstream code)
    # --------------------------------------------------------------
    return out.float()
```
