"""
TriMul “outgoing” kernel – fused element‑wise gating/mask + batched GEMM.

Algorithm
---------
1. Layer‑norm over the channel dimension (float32, torch).
2. Linear projections (fp16) to hidden_dim and their sigmoid gates (fp16).
3. Optional pairwise mask (fp16) – broadcast across the hidden dimension.
4. A single Triton kernel computes, for every (batch, hidden) pair,
   the matrix product   C = (P_left ⊙ G_left ⊙ M) @ (P_right ⊙ G_right ⊙ M)^T .
   The kernel fuses the element‑wise multiplications with the reduction,
   eliminating the two intermediate “fused” tensors.
5. The result (fp16) is reshaped to [B, N, N, H] and cast to fp32.
6. Hidden‑dim layer‑norm, out‑gate multiplication and final linear
   projection back to the original channel dimension are performed with
   torch (these steps are tiny compared to the O(N³·H) work).

The heavy O(N³·H) work is therefore carried out by a single highly‑
optimized Triton GEMM‑like kernel, cutting kernel launch overhead
and memory traffic dramatically.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused projection‑gate‑mask -> batched GEMM
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fused_gemm_kernel(
    # Tensor pointers
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    # Shapes / parameters
    N, H,
    # Strides (batch, i, k for each tensor)
    stride_lp_batch, stride_lp_i, stride_lp_k,
    stride_lg_batch, stride_lg_i, stride_lg_k,
    stride_rp_batch, stride_rp_i, stride_rp_k,
    stride_rg_batch, stride_rg_i, stride_rg_k,
    mask_batch_stride, mask_i_stride, mask_k_stride,
    stride_out_batch, stride_out_i, stride_out_j,
    # Compile‑time constants
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """ Compute
        out[b,h,i,j] = Σ_k  (left_proj*left_gate*mask)[b,h,i,k] *
                      (right_proj*right_gate*mask)[b,h,j,k]
        with fp16 arithmetic, accumulation in fp32.
    """
    pid_m = tl.program_id(0)   # output row block
    pid_n = tl.program_id(1)   # output col block
    pid_b = tl.program_id(2)   # batch*hidden index  (0 .. B*H-1)

    # Offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k_iter in range(0, num_k):
        k_off = k_iter * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        valid_k = offs_k < N

        # --------------------------------------------------------------
        # Load left projection + gate (+ mask) : shape (M, K)
        # --------------------------------------------------------------
        lp_ptr = (left_proj_ptr + pid_b * stride_lp_batch
                  + offs_i[:, None] * stride_lp_i
                  + offs_k[None, :] * stride_lp_k)
        lg_ptr = (left_gate_ptr + pid_b * stride_lg_batch
                  + offs_i[:, None] * stride_lg_i
                  + offs_k[None, :] * stride_lg_k)

        lp = tl.load(lp_ptr,
                     mask=mask_i[:, None] & valid_k[None, :],
                     other=0.0)
        lg = tl.load(lg_ptr,
                     mask=mask_i[:, None] & valid_k[None, :],
                     other=0.0)
        left = lp * lg

        if HAS_MASK:
            batch_id = pid_b // H
            mask_base = mask_ptr + batch_id * mask_batch_stride
            m_left = tl.load(mask_base + offs_i[:, None] * mask_i_stride
                                            + offs_k[None, :] * mask_k_stride,
                             mask=mask_i[:, None] & valid_k[None, :],
                             other=1.0)
            left = left * m_left

        # --------------------------------------------------------------
        # Load right projection + gate (+ mask) : shape (K, N)
        # --------------------------------------------------------------
        rp_ptr = (right_proj_ptr + pid_b * stride_rp_batch
                  + offs_k[:, None] * stride_rp_k
                  + offs_j[None, :] * stride_rp_i)
        rg_ptr = (right_gate_ptr + pid_b * stride_rg_batch
                  + offs_k[:, None] * stride_rg_k
                  + offs_j[None, :] * stride_rg_i)

        rp = tl.load(rp_ptr,
                     mask=valid_k[:, None] & mask_j[None, :],
                     other=0.0)
        rg = tl.load(rg_ptr,
                     mask=valid_k[:, None] & mask_j[None, :],
                     other=0.0)
        right = rp * rg

        if HAS_MASK:
            batch_id = pid_b // H
            mask_base = mask_ptr + batch_id * mask_batch_stride
            m_right = tl.load(mask_base + offs_k[:, None] * mask_k_stride
                                             + offs_j[None, :] * mask_i_stride,
                              mask=valid_k[:, None] & mask_j[None, :],
                              other=1.0)
            right = right * m_right

        # --------------------------------------------------------------
        # Accumulate dot product
        # --------------------------------------------------------------
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store output tile (fp16, accumulation already in fp32)
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr + pid_b * stride_out_batch
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, C]   (float32)
            mask_tensor  : [B, N, N]      (bool / float, optional)
            weights_dict : model parameters (float32)
            config_dict  : {"dim": C,
                            "hidden_dim": H,
                            "nomask": bool, ...}

    Returns:
        out : [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)
    HAS_MASK = 0 if nomask else 1

    batch_size, seq_len, _, _ = input_tensor.shape
    B, N = batch_size, seq_len
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                       # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2. Projections & gates (fp16)
    # ------------------------------------------------------------------
    # Cast once to fp16 to let torch use tensor‑core kernels
    x_fp16 = x.to(torch.float16)

    # Weights in fp16 – tiny copies, negligible cost
    w_left_proj  = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate  = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)

    left_proj  = F.linear(x_fp16, w_left_proj)               # [B,N,N,H] fp16
    right_proj = F.linear(x_fp16, w_right_proj)

    left_gate  = torch.sigmoid(F.linear(x_fp16, w_left_gate))   # fp16 sigmoid
    right_gate = torch.sigmoid(F.linear(x_fp16, w_right_gate))

    # out‑gate stays in fp32 (used after hidden‑dim LN)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H] fp32

    # ------------------------------------------------------------------
    # 3. Rearrange tensors to [B*H, N, N] (contiguous)
    # ------------------------------------------------------------------
    total_batch = B * H                         # B·H

    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    left_proj = left_proj.view(total_batch, N, N)   # [B·H, N, N]
    right_proj = right_proj.view(total_batch, N, N)
    left_gate = left_gate.view(total_batch, N, N)
    right_gate = right_gate.view(total_batch, N, N)

    # ------------------------------------------------------------------
    # 4. Optional mask (fp16) – broadcast across hidden dim
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()  # [B,N,N]
    else:
        # dummy – will never be accessed because HAS_MASK==0
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 5. Fused GEMM kernel (batched matrix multiply with gating & mask)
    # ------------------------------------------------------------------
    out_fp16 = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # Tile sizes – empirically good for H100 fp16 GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        total_batch,                      # batch*hidden
    )

    # Strides (in element units)
    lp_batch_s, lp_i_s, lp_k_s = left_proj.stride()
    lg_batch_s, lg_i_s, lg_k_s = left_gate.stride()
    rp_batch_s, rp_i_s, rp_k_s = right_proj.stride()
    rg_batch_s, rg_i_s, rg_k_s = right_gate.stride()
    out_batch_s, out_i_s, out_j_s = out_fp16.stride()
    mask_batch_s, mask_i_s, mask_k_s = mask_fp16.stride() if not nomask else (0, 0, 0)

    _tri_mul_fused_gemm_kernel[grid](
        left_proj_ptr=left_proj,
        left_gate_ptr=left_gate,
        right_proj_ptr=right_proj,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp16,
        out_ptr=out_fp16,
        N=N,
        H=H,
        stride_lp_batch=lp_batch_s,
        stride_lp_i=lp_i_s,
        stride_lp_k=lp_k_s,
        stride_lg_batch=lg_batch_s,
        stride_lg_i=lg_i_s,
        stride_lg_k=lg_k_s,
        stride_rp_batch=rp_batch_s,
        stride_rp_i=rp_i_s,
        stride_rp_k=rp_k_s,
        stride_rg_batch=rg_batch_s,
        stride_rg_i=rg_i_s,
        stride_rg_k=rg_k_s,
        mask_batch_stride=mask_batch_s,
        mask_i_stride=mask_i_s,
        mask_k_stride=mask_k_s,
        stride_out_batch=out_batch_s,
        stride_out_i=out_i_s,
        stride_out_j=out_j_s,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to [B, N, N, H] and cast to fp32
    # ------------------------------------------------------------------
    out = (
        out_fp16
        .view(B, hidden_dim, N, N)                     # [B, H, N, N]
        .permute(0, 2, 3, 1)                            # [B, N, N, H]
        .contiguous()
        .to(dtype)                                      # fp16 → fp32
    )

    # ------------------------------------------------------------------
    # 7. Hidden‑dim layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                               # out‑gate (fp32)
    out = F.linear(out, weights["to_out.weight"])      # back to dim C

    return out
