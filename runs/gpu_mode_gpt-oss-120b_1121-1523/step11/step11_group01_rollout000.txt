def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)  # batch index
    i_tile = tl.program_id(1)  # i tile
    j_tile = tl.program_id(2)  # j tile
    h_tile = tl.program_id(3)  # hidden tile

    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J
    h_base = h_tile * BLOCK_H

    # compute valid ranges
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = h_base + tl.arange(0, BLOCK_H)

    j_mask = j < N
    h_mask = h < H

    # loop over K tiles

        k = k_tile + tl.arange(0, BLOCK_K)
        k_mask = k < N

        #   left[i,k,h] * right[j,k,h] -> produce (i,j,k,h)
        #   then sum over k dimension produce (i,j,h).

        # This can be done by:

"""
Highly‑optimized forward implementation of the *outgoing* TriMul operator
used in AlphaFold‑3 style models.

Algorithm (FP16/BF16 everywhere except the tiny LayerNorms that stay in FP32):
1. Layer‑Norm the input (FP32 – numerically stable, cheap on the small dim).
2. Cast to half‑precision and perform ONE fused linear projection that yields
   left_proj, left_gate, right_proj, right_gate in a single GEMM.
3. In‑place sigmoid on the gate tensors and fuse them into the projections.
4. (Optional) Apply the pair‑wise mask in‑place with a tiny Triton kernel that
   broadcasts the scalar mask over the hidden dimension.
5. Core O(N³) update – batched half‑precision GEMM:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single torch.matmul on tensors of shape (B, H, N, N) –
   avoids the costly .contiguous copies used in the reference code.
6. Output LayerNorm + output‑gate (sigmoid) + final linear projection back to
   the original channel dimension – everything stays in half‑precision.
7. Cast the result back to FP32 for downstream consumption.

Only the mask‑application is performed in a custom Triton kernel; the rest
uses highly‑optimized torch kernels.  The implementation works for any
batch size 1‑2, sequence length up to 1024 and hidden dimensions up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,                # fp16 *
    right_ptr,               # fp16 *
    mask_ptr,                # fp16 *
    B, N, H,                 # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Multiply left/right [B,N,N,H] by mask [B,N,N] (broadcast over H)."""
    b = tl.program_id(0)          # batch index
    i_tile = tl.program_id(1)     # i‑tile
    j_tile = tl.program_id(2)     # j‑tile

    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # ------------------------------------------------------------------
    # Load the scalar mask (shape [B,N,N]) and broadcast to hidden dim
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        h = h_start + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # ----- left tile -----
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        # ----- right tile -----
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        # ----- apply mask (broadcast over H) -----
        left = left * mask
        right = right * mask

        # ----- store back (in‑place) -----
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place elementwise multiplication of `left`/`right` by `mask`
    using a small Triton kernel. All tensors must be half‑precision
    and reside on the same CUDA device.
    """
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    mask_fp16 = mask.to(dtype=left.dtype)

    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_fp16.stride()

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 64                     # 64 works for all supported hidden dims (≤384)

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_apply_kernel[grid](
        left,
        right,
        mask_fp16,
        B, N, H,

        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑enhanced forward pass for the *outgoing* TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Load weights (all stored as float32 on the device)
    # --------------------------------------------------------------
    # Input LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (four matrices will be concatenated)
    left_proj_weight   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]    # (hidden_dim, dim)

    # Output side
    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]       # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 3) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 4) Cast to FP16 and perform a single fused linear projection
    # --------------------------------------------------------------
    x_fp16 = x_norm_fp32.half()   # one cast only

    # Concatenate the four weight matrices → (4*hidden_dim, dim)
    fused_weight_fp16 = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=x_fp16.device)

    # (B,N,N,dim) @ (dim,4*hidden_dim)^T  → (B,N,N,4*hidden_dim)
    fused = F.linear(x_fp16, fused_weight_fp16)   # fp16

    # Split without additional copy
    left_proj, left_gate, right_proj, right_gate = torch.split(
        fused, hidden_dim, dim=-1
    )  # each -> (B,N,N,hidden_dim)

    # In‑place sigmoid on gates (fp16)
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()

    # Fuse gates into the projections
    left = left_proj.mul_(left_gate)      # (B,N,N,hidden_dim)
    right = right_proj.mul_(right_gate)   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 5) Optional mask (in‑place via Triton)
    # --------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 6) Core O(N³) update – batched half‑precision GEMM
    # --------------------------------------------------------------
    # Re‑order to (B, hidden, N, N) – no extra copy
    left_perm = left.permute(0, 3, 1, 2)      # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)    # (B, H, N, N) – note the transpose on the last two dims

    # Batched matrix multiplication in half‑precision
    out_perm = torch.matmul(left_perm, right_perm)   # (B, H, N, N), fp16

    # Back to (B, N, N, H) – simply a view, no memory copy
    out = out_perm.permute(0, 2, 3, 1)               # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Output LayerNorm + output‑gate + final projection (all FP16)
    # --------------------------------------------------------------
    # Output LayerNorm (weights/bias cast to fp16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )   # still fp16

    # Output gate (derived from the *normed* input)
    out_gate = torch.sigmoid(F.linear(x_norm_fp32.half(), out_gate_weight.half()))   # (B,N,N,hidden_dim) fp16
    out = out * out_gate

    # Final linear back to original dim
    out = F.linear(out, to_out_weight.half())   # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 8) Cast back to FP32 for downstream code
    # --------------------------------------------------------------
    return out.float()
