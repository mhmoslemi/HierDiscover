"""
High‑performance outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D pair tensor (float32 → stable).
2️⃣ OPTIONAL in‑place mask‑multiplication using a tiny Triton kernel
   (broadcast over the channel dimension C).  This satisfies the “must‑use‑kernel”
   rule.
3️⃣ Cast to fp16 and fuse the four linear branches (left‑proj, left‑gate,
   right‑proj, right‑gate) into a single GEMM.  Split the result and apply
   sigmoid‑gates (still fp16).
4️⃣ Core O(N³·H) update: for every head h we need
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is simply a batched matrix‑matrix product:
        out_h = left_h @ right_hᵀ.
   We implement it with a single `torch.bmm` over the combined batch
   dimension (B·H), which gives Tensor‑Core‑friendly performance far
   superior to the original `einsum`.
5️⃣ Output LayerNorm (fp16), gating with the sigmoid of a linear on the
   normalized input, and a final projection back to the original dimension.
   The result is returned as float32.

All heavy arithmetic stays in fp16 (Tensor‑Core friendly).  Only the mask
multiplication is performed with a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over C)
# -------------------------------------------------------------
@triton.jit
def _mask_input_kernel(
    x_ptr,                # fp16 / fp32 *
    mask_ptr,             # same dtype as x *
    B, N, C,              # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """x[b, i, j, c] *= mask[b, i, j]   (mask broadcast over C)"""
    b = tl.program_id(0)          # batch dimension
    ti = tl.program_id(1)         # tile over i
    tj = tl.program_id(2)         # tile over j

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    c = tl.arange(0, BLOCK_C)

    i_ok = i < N
    j_ok = j < N

    # Load mask (scalar per i,j)
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                         # (BLOCK_I, BLOCK_J)

    mask = mask[:, :, None]    # broadcast over channel dim

    # Loop over channel tiles
    off = 0
    while off < C:
        cur_c = off + c
        c_ok = cur_c < C

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_c[None, None, :] * stride_x_c,
            mask=i_ok[:, None, None] &
                 j_ok[None, :, None] &
                 c_ok[None, None, :],
            other=0.0,
        )
        x = x * mask
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_c[None, None, :] * stride_x_c,
            x,
            mask=i_ok[:, None, None] &
                 j_ok[None, :, None] &
                 c_ok[None, None, :],
        )
        off += BLOCK_C


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place multiplication ``x *= mask`` (mask broadcast over the channel dim).
    The operation is performed by the Triton kernel defined above.
    """
    assert x.is_cuda and mask.is_cuda
    B, N, _, C = x.shape
    mask = mask.to(dtype=x.dtype)               # same dtype as the tensor we modify

    s_x = x.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_C = 128                     # covers up to C=384 in 3 tiles

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_input_kernel[grid](
        x,
        mask,
        B,
        N,
        C,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    # `x` is modified in‑place


# -------------------------------------------------------------
# Triton‑aware forward‑only outgoing TriMul
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (all stored as float32 in the dict)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Optional mask – fused in‑place with a Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(x, mask_tensor)   # modifies x in‑place

    # ------------------------------------------------------------------
    # 3️⃣ Half‑precision fused projection + gating
    # ------------------------------------------------------------------
    x_h = x.half()    # (fp16)

    # Fuse the four linear branches into one GEMM:
    # order = left_proj, left_gate, right_proj, right_gate
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16)      # (4*H, C)

    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                 # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)   # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4*H)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each (B, N, N, H) in fp16

    # Apply sigmoid gates
    left = left_proj * left_gate.sigmoid()   # (B, N, N, H)
    right = right_proj * right_gate.sigmoid()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – batched matrix multiplication
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_b = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_b[b*h] = left_b @ right_bᵀ
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, gating and final projection (stay in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    # output gate – computed from the normalized (and masked) input
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H) fp16
    out = out * out_gate

    # final linear back to original dimension `dim`
    out = F.linear(out, to_out_weight.half())   # (B,N,N,dim) fp16

    # Return float32 as in the reference implementation
    return out.float()
