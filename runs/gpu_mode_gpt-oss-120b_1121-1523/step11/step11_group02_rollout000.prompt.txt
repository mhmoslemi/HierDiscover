```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm over the channel dimension C.
2. Linear projections → left/right tensors of shape [B,N,N,H] (fp16).
   Independent sigmoid gates → left_gate/right_gate (fp16) and out_gate (fp32).
3. Fuse projection, gate and optional pairwise mask into two
   tensors `left_fused` and `right_fused` with a small Triton kernel.
   After fusion we have the exact operands required for the core
   reduction:
       out_h = left_fused_h @ right_fused_h.T   (batched GEMM)
   The heavy matrix multiplication is delegated to `torch.bmm`,
   which on an H100 uses the highly‑optimized CUTLASS kernels and
   tensor‑cores.
4. Reshape the result back to [B,N,N,H] (float32) and apply the
   hidden‑dim LayerNorm, the out‑gate and the final linear projection
   back to the original channel dimension C.
5. The only Triton code is the element‑wise fusion kernel; all
   compute‑intensive work is performed by the library GEMM.

The kernel loads a single element from the projection, its gate and
writes the fused result (and optionally the mask) – vastly reducing
memory traffic compared with the naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection + gate (+ optional mask) into one tensor
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    proj_ptr,                # [B·H,N,N] projection
    gate_ptr,                # [B·H,N,N] sigmoid gate
    out_ptr,                 # [B·H,N,N] fused result (fp16)
    mask_ptr,                # [B,N,N] pairwise mask (fp16) – broadcasted across H
    # Sizes
    M, N,                    # both equal to seq_len
    # Strides
    stride_proj_batch, stride_proj_m, stride_proj_n,
    stride_gate_batch, stride_gate_m, stride_gate_n,
    stride_out_batch,  stride_out_m,  stride_out_n,
    mask_batch_stride, mask_stride_m, mask_stride_n,
    # Compile‑time constants
    H: tl.constexpr,        # hidden dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    out = proj ⊙ gate ⊙ (mask if present)
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch*hidden index

    # ---- Offsets inside the tile -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- Pointers to the current matrix -----------------------------------------
    proj = proj_ptr + pid_b * stride_proj_batch
    gate = gate_ptr + pid_b * stride_gate_batch
    out  = out_ptr  + pid_b * stride_out_batch

    # ---- Load projection and gate -------------------------------------------------
    a = tl.load(
        proj + (offs_m[:, None] * stride_proj_m + offs_n[None, :] * stride_proj_n),
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    g = tl.load(
        gate + (offs_m[:, None] * stride_gate_m + offs_n[None, :] * stride_gate_n),
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    out_val = a * g                     # element‑wise product

    # ---- Optional mask -----------------------------------------------------------
    if HAS_MASK:
        # Which original batch does this (batch*hidden) belong to?
        batch_id = pid_b // H
        mask = mask_ptr + batch_id * mask_batch_stride
        m = tl.load(
            mask + (offs_m[:, None] * mask_stride_m + offs_n[None, :] * mask_stride_n),
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        out_val = out_val * m

    # ---- Store result -------------------------------------------------------------
    tl.store(
        out + (offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n),
        out_val,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, C]   (float32)
            mask_tensor  : [B, N, N]      (bool/float, optional)
            weights_dict : model parameters
            config_dict  : {"dim": C,
                            "hidden_dim": H,
                            "nomask": bool, ...}

    Returns:
        out : [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Layer‑norm over channel dimension (C)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, H] (fp16)
    #    and sigmoid gates (fp16 for hidden, fp32 for out‑gate)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # fp32

    # --------------------------------------------------------------
    # 3) Rearrange to [B·H, N, N] (contiguous) – convenient for
    #    the batched GEMM and the fusion kernel.
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_batch = B * hidden_dim                     # B·H

    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    left_proj = left_proj.view(total_batch, N, N)    # [B·H, N, N]
    right_proj = right_proj.view(total_batch, N, N)
    left_gate = left_gate.view(total_batch, N, N)
    right_gate = right_gate.view(total_batch, N, N)

    # --------------------------------------------------------------
    # 4) Fuse projection, gate and (optional) mask into two tensors.
    #    The mask is **not** expanded – the kernel computes the
    #    correct batch index on‑the‑fly.
    # --------------------------------------------------------------
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # Choose tile size for the element‑wise kernel
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        total_batch,                      # batch*hidden
    )
    HAS_MASK = 0 if nomask else 1

    # Strides for the tensors (all in element units)
    proj_batch_stride = left_proj.stride(0)
    proj_m_stride = left_proj.stride(1)
    proj_n_stride = left_proj.stride(2)

    gate_batch_stride = left_gate.stride(0)
    gate_m_stride = left_gate.stride(1)
    gate_n_stride = left_gate.stride(2)

    out_batch_stride = left_fused.stride(0)
    out_m_stride = left_fused.stride(1)
    out_n_stride = left_fused.stride(2)

    if not nomask:
        mask = mask_tensor.to(torch.float16)                # [B, N, N]
        mask_batch_stride = mask.stride(0)
        mask_m_stride = mask.stride(1)
        mask_n_stride = mask.stride(2)
    else:
        # dummy values – will never be read because HAS_MASK==0
        mask = torch.empty(0, device=device, dtype=torch.float16)
        mask_batch_stride = mask_m_stride = mask_n_stride = 0

    # ---- Left side fusion -------------------------------------------------
    _fuse_proj_gate_mask_kernel[grid](
        proj_ptr=left_proj,
        gate_ptr=left_gate,
        out_ptr=left_fused,
        mask_ptr=mask,
        M=N,
        N=N,
        stride_proj_batch=proj_batch_stride,
        stride_proj_m=proj_m_stride,
        stride_proj_n=proj_n_stride,
        stride_gate_batch=gate_batch_stride,
        stride_gate_m=gate_m_stride,
        stride_gate_n=gate_n_stride,
        stride_out_batch=out_batch_stride,
        stride_out_m=out_m_stride,
        stride_out_n=out_n_stride,
        mask_batch_stride=mask_batch_stride,
        mask_stride_m=mask_m_stride,
        mask_stride_n=mask_n_stride,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # ---- Right side fusion ------------------------------------------------
    _fuse_proj_gate_mask_kernel[grid](
        proj_ptr=right_proj,
        gate_ptr=right_gate,
        out_ptr=right_fused,
        mask_ptr=mask,
        M=N,
        N=N,
        stride_proj_batch=proj_batch_stride,
        stride_proj_m=proj_m_stride,
        stride_proj_n=proj_n_stride,
        stride_gate_batch=gate_batch_stride,
        stride_gate_m=gate_m_stride,
        stride_gate_n=gate_n_stride,
        stride_out_batch=out_batch_stride,
        stride_out_m=out_m_stride,
        stride_out_n=out_n_stride,
        mask_batch_stride=mask_batch_stride,
        mask_stride_m=mask_m_stride,
        mask_stride_n=mask_n_stride,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # --------------------------------------------------------------
    # 5) Core batched GEMM: (B·H, N, N) @ (B·H, N, N)^T
    #    Torch’s batched GEMM (CUTLASS) is far faster than a generic Triton GEMM.
    # --------------------------------------------------------------
    out_fp16 = torch.bmm(left_fused, right_fused.transpose(1, 2))

    # --------------------------------------------------------------
    # 6) Rest of the pipeline – hidden‑dim LayerNorm, out‑gate and final projection.
    # --------------------------------------------------------------
    out = (
        out_fp16
        .view(B, hidden_dim, N, N)                 # [B, H, N, N]
        .permute(0, 2, 3, 1)                        # [B, N, N, H]
        .contiguous()
        .to(dtype)                                  # fp16 → fp32
    )

    # Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply out‑gate (still fp32)
    out = out * out_gate

    # Final linear projection back to C
    out = F.linear(out, weights["to_out.weight"])

    return out
```
