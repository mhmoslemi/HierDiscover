"""
TriMul “outgoing” (AlphaFold‑3) – fast forward implementation.

Algorithm
---------
1. Layer‑norm over the channel dimension C.
2. Two fused linear passes:
   * One linear produces both left‑ and right‑projections (2·H outputs).
   * A second linear produces the two gates plus the out‑gate (3·H outputs).
3. Split the results, apply sigmoid to the gates and fuse
   projection·gate·mask with a tiny Triton kernel (one kernel per side).
   This eliminates four separate linear passes and an extra element‑wise
   multiplication.
4. Batched GEMM (torch.bmm) on the fused tensors – each hidden slice is a
   separate matrix multiplication, handled by CUTLASS/Tensor‑cores.
5. Hidden‑dim layer‑norm, out‑gate multiplication and final linear
   projection back to the original feature dimension.
All heavy arithmetic stays in FP16; only the small LayerNorms and the
final projection use FP32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fuse_proj_gate_mask_kernel(
    proj_ptr,                # [B·H,N,N] projection (fp16)
    gate_ptr,                # [B·H,N,N] sigmoid gate (fp16)
    out_ptr,                 # [B·H,N,N] fused result (fp16)
    mask_ptr,                # [B,N,N] pairwise mask (fp16) – broadcasted across H
    # sizes
    M, N,                    # both equal to seq_len
    # strides (in element units)
    stride_proj_batch, stride_proj_m, stride_proj_n,
    stride_gate_batch, stride_gate_m, stride_gate_n,
    stride_out_batch,  stride_out_m,  stride_out_n,
    mask_batch_stride, mask_stride_m, mask_stride_n,
    # compile‑time constants
    H: tl.constexpr,        # hidden dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """ out = proj ⊙ gate ⊙ (mask if present) """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this tile
    proj = proj_ptr + pid_b * stride_proj_batch
    gate = gate_ptr + pid_b * stride_gate_batch
    out  = out_ptr  + pid_b * stride_out_batch

    # projection & gate
    a = tl.load(
        proj + (offs_m[:, None] * stride_proj_m + offs_n[None, :] * stride_proj_n),
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    g = tl.load(
        gate + (offs_m[:, None] * stride_gate_m + offs_n[None, :] * stride_gate_n),
        mask=mask_m[:, None] & mask_n[None, :],
        other=0.0,
    )
    out_val = a * g                     # element‑wise product

    # optional mask
    if HAS_MASK:
        batch_id = pid_b // H
        mask = mask_ptr + batch_id * mask_batch_stride
        m = tl.load(
            mask + (offs_m[:, None] * mask_stride_m + offs_n[None, :] * mask_stride_n),
            mask=mask_m[:, None] & mask_n[None, :],
            other=0.0,
        )
        out_val = out_val * m

    tl.store(
        out + (offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n),
        out_val,
        mask=mask_m[:, None] & mask_n[None, :],
    )

        stride_proj_batch=proj_batch_stride,
        stride_proj_m=proj_m_stride,
        stride_proj_n=proj_n_stride,
        stride_gate_batch=gate_batch_stride,
        stride_gate_m=gate_m_stride,
        stride_gate_n=gate_n_stride,
        stride_out_batch=out_batch_stride,
        stride_out_m=out_m_stride,
        stride_out_n=out_n_stride,
        mask_batch_stride=mask_batch_stride,
        mask_stride_m=mask_m_stride,
        mask_stride_n=mask_n_stride,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # right side
    _fuse_proj_gate_mask_kernel[grid](
        proj_ptr=right_proj_r,
        gate_ptr=right_gate_r,
        out_ptr=right_fused,
        mask_ptr=mask,
        M=N,
        N=N,
        stride_proj_batch=proj_batch_stride,
        stride_proj_m=proj_m_stride,
        stride_proj_n=proj_n_stride,
        stride_gate_batch=gate_batch_stride,
        stride_gate_m=gate_m_stride,
        stride_gate_n=gate_n_stride,
        stride_out_batch=out_batch_stride,
        stride_out_m=out_m_stride,
        stride_out_n=out_n_stride,
        mask_batch_stride=mask_batch_stride,
        mask_stride_m=mask_m_stride,
        mask_stride_n=mask_n_stride,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 4) Core batched GEMM: (B·H, N, N) @ (B·H, N, N)^T  (FP16)
    # ------------------------------------------------------------------
    out_fp16 = torch.bmm(left_fused, right_fused.transpose(1, 2))   # (B·H, N, N)

    # ------------------------------------------------------------------
    # 5) Hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = (
        out_fp16
        .view(B, hidden_dim, N, N)            # (B, H, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, H)
        .contiguous()
        .to(torch.float32)                  # fp16 → fp32
    )

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # apply out‑gate (FP32)

    # final linear back to original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
