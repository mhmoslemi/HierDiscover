import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Fast “outgoing” TriMul forward pass (AlphaFold‑3 style) – Triton mask kernel
and a fully‑fusion‑friendly PyTorch implementation that stays in fp16 as long
as possible.

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D tensor (float32 → keep numerics stable).
2️⃣ If a mask is provided, fuse it *in‑place* with a tiny Triton kernel
   (broadcast over the channel dimension).
3️⃣ Cast the normalized tensor to fp16 and apply one fused linear layer that
   produces the four branches (left‑proj, left‑gate, right‑proj, right‑gate)
   in a single GEMM.  Split the result and apply sigmoid‑gates.
4️⃣ Core O(N³·H) update:   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   → performed with a single `torch.einsum` (avoid the costly
   permute‑+‑contiguous‑+‑view used for a batched bmm).
5️⃣ Layer‑Norm over the hidden dimension, multiply by the output‑gate
   (computed from the normalized input) and a final fp16 linear projection
   back to the original `dim`.  The result is returned as float32.

All heavy arithmetic stays in fp16 (Tensor‑Core friendly) and only the
few LayerNorm calls up‑cast to fp32 internally.  The mask‑multiplication is
the only part that runs on Triton, satisfying the “must‑use‑kernel” rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over C)
# -------------------------------------------------------------
@triton.jit
def _mask_input_kernel(
    x_ptr,                 # fp16 / fp32 *
    mask_ptr,              # same dtype as x *
    B, N, C,               # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr
):
    """x[b, i, j, c] *= mask[b, i, j]   (mask broadcast over C)"""
    b = tl.program_id(0)          # batch dimension
    ti = tl.program_id(1)         # tile i
    tj = tl.program_id(2)         # tile j

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    c = tl.arange(0, BLOCK_C)

    i_ok = i < N
    j_ok = j < N

    # ---- load mask (scalar per i,j) ---------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)

    mask = mask[:, :, None]   # broadcast over C

    # ---- loop over the channel dimension -----------------------------
    off = 0
    while off < C:
        cur_c = off + c
        c_ok = cur_c < C

        # load a tile of x
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_c[None, None, :] * stride_x_c,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
            other=0.0,
        )
        # apply mask
        x = x * mask
        # store back
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_c[None, None, :] * stride_x_c,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
        )
        off += BLOCK_C


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place multiplication ``x *= mask`` (mask broadcast over the channel dim)."""
    assert x.is_cuda and mask.is_cuda
    B, N, _, C = x.shape
    mask_cast = mask.to(dtype=x.dtype)    # same dtype as the tensor we modify

    # strides expressed in element counts (not bytes)
    s_x = x.stride()
    s_m = mask_cast.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    # C can be up to 384 → three 128‑wide tiles are enough
    BLOCK_C = 128

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_input_kernel[grid](
        x,
        mask_cast,
        B,
        N,
        C,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    # ``x`` is modified in‑place


# -------------------------------------------------------------
# Triton‑aware forward‑only outgoing TriMul
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (all stored as float32 in the dict)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Optional mask – fused in‑place with a Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(x, mask_tensor)   # x is modified in‑place

    # ------------------------------------------------------------------
    # 3️⃣ Half‑precision fused projection + gating
    # ------------------------------------------------------------------
    x_h = x.half()                                    # (fp16)

    # concatenate the four linear weights once (order: left_proj, left_gate,
    # right_proj, right_gate) → shape (4*H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16)

    B, N, _, _ = x_h.shape
    # flatten to (B·N·N, C) and apply the fused linear projection
    x_flat = x_h.view(-1, dim)                         # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)      # (B*N*N, 4*H)

    # reshape back to 4‑branch tensor
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4*H)

    # split into the four branches
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: (B, N, N, H)  fp16

    # apply sigmoid gates directly in fp16
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – single fused einsum (no extra permutes)
    # ------------------------------------------------------------------
    # out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, gating and final projection (stay in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    # output gate computed from the *normalized* input (still in fp16)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H) fp16
    out = out * out_gate

    # final linear back to original dimension `dim`
    out = F.linear(out, to_out_weight.half())                       # (B,N,N,dim) fp16

    # return float32 (as in the reference implementation)
    return out.float()
