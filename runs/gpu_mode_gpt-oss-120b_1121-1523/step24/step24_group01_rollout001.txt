@triton.jit
def _apply_out_gate_kernel(
    out_ptr, # (B,N,N,H)
    gate_ptr, # (B,N,N,H) (pre-sigmoid)
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # load out and gate
    out = tl.load(out_ptr + b*stride_out_b + i[:,None,None]*stride_out_i + j[None,:,None]*stride_out_j + h[None,None,:]*stride_out_h,
                  mask=i_valid[:,None,None] & j_valid[None,:,None] & h_valid[None,None,:],
                  other=0.0)
    gate = tl.load(gate_ptr + b*stride_gate_b + i[:,None,None]*stride_gate_i + j[None,:,None]*stride_gate_j + h[None,None,:]*stride_gate_h,
                  mask=i_valid[:,None,None] & j_valid[None,:,None] & h_valid[None,None,:],
                  other=0.0)

    # sigmoid on gate (FP32 for stability)
    gate_fp32 = gate.to(tl.float32)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate_fp32))
    gate_sig = gate_sig.to(tl.float16)

    out = out * gate_sig

    tl.store(out_ptr + b*stride_out_b + i[:,None,None]*stride_out_i + j[None,:,None]*stride_out_j + h[None,None,:]*stride_out_h,
             out,
             mask=i_valid[:,None,None] & j_valid[None,:,None] & h_valid[None,None,:])

"""
Fast outgoing TriMul (AlphaFold‑3) – forward only.
Algorithm
---------
1️⃣ Layer‑norm the input (FP32→FP16).
2️⃣ A single fused linear projects the input to four tensors
   (left, right, left‑gate, right‑gate) in FP16.
3️⃣ One Triton kernel fuses the sigmoid‑gate, optional mask and
   element‑wise multiplication, overwriting left/right in‑place.
   The same kernel is used for the “no‑mask’’ case (mask = 1).
4️⃣ Core contraction Σₖ left[b,i,k,h]·right[b,j,k,h] is a batched
   GEMM (torch.bmm) on the view (B·H, N, N).
5️⃣ Hidden‑dim LayerNorm (FP16), out‑gate (linear + sigmoid) is
   fused by a tiny Triton kernel, then the final linear projection.
6️⃣ Cast back to FP32.
The implementation keeps all heavy matrix‑multiplies in cuBLAS (the
fast path) and only the cheap element‑wise gating steps in Triton,
giving a sub‑millisecond forward pass on H100 for the test configs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel : mask + sigmoid‑gate (in‑place on left/right)
# -------------------------------------------------------------
@triton.jit
def _mask_gate_mul_kernel(
    # pointers
    left_ptr, right_ptr,               # (B,N,N,H) – overwritten
    left_gate_ptr, right_gate_ptr,     # (B,N,N,H) – pre‑gate
    mask_ptr,                          # (B,N,N) – optional (half), ignored if HAS_MASK==0
    # problem size
    B, N, H,
    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    # tile sizes (compile‑time)
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr, HAS_MASK: tl.constexpr,
):
    """
    In‑place:
        left  = left  * sigmoid(left_gate)  * mask
        right = right * sigmoid(right_gate) * mask
    All tensors are half‑precision.
    """
    b = tl.program_id(0)               # batch
    pid_i = tl.program_id(1)           # tile over i
    pid_j = tl.program_id(2)           # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # ------- mask (broadcast over H) ----------
    if HAS_MASK:
        mask_tile = tl.load(
            mask_ptr
            + b * stride_m_b
            + i[:, None] * stride_m_i
            + j[None, :] * stride_m_j,
            mask=i_valid[:, None] & j_valid[None, :],
            other=0.0,
        )                     # (BLOCK_I, BLOCK_J)
        mask_tile = mask_tile[:, :, None]    # (BLOCK_I, BLOCK_J, 1)
    else:
        mask_tile = tl.full((BLOCK_I, BLOCK_J, 1), 1.0, dtype=tl.float16)

    # ------- loop over hidden dimension ----------
    h_offset = 0
    while h_offset < H:
        h = h_offset + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # ----- load tensors -----
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        left_g = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right_g = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )

        # ----- sigmoid (FP32 for stability) -----
        left_gate = 1.0 / (1.0 + tl.exp(-left_g.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_g.to(tl.float32)))
        left_gate = left_gate.to(tl.float16)
        right_gate = right_gate.to(tl.float16)

        # ----- apply mask and gate -----
        left_out = left * left_gate * mask_tile
        right_out = right * right_gate * mask_tile

        # ----- store back in‑place -----
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )

        h_offset += BLOCK_H


def _apply_mask_and_gate_triton(left_proj, right_proj,
                                left_gate_pre, right_gate_pre,
                                mask, has_mask):
    """
    Fuse sigmoid‑gate, optional mask and in‑place multiplication.
    All tensors are FP16.
    """
    B, N, _, H = left_proj.shape
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate_pre.stride()
    s_rg = right_gate_pre.stride()
    s_m = mask.stride() if mask is not None else (0, 0, 0)

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_mul_kernel[grid](
        left_proj,
        right_proj,
        left_gate_pre,
        right_gate_pre,
        mask if has_mask else left_proj,   # dummy ptr – never read if HAS_MASK==0
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        HAS_MASK=int(has_mask),
        num_warps=4,
    )
    # left_proj / right_proj now contain gated (and masked) values


# -------------------------------------------------------------
# Triton kernel : apply sigmoid‑gate to hidden output (no mask)
# -------------------------------------------------------------
@triton.jit
def _apply_out_gate_kernel(
    out_ptr, gate_ptr,
    B, N, H,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    in‑place:
        out = out * sigmoid(gate)
    All tensors FP16.
    """
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # load tensors
    out = tl.load(
        out_ptr
        + b * stride_o_b
        + i[:, None, None] * stride_o_i
        + j[None, :, None] * stride_o_j
        + h[None, None, :] * stride_o_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    gate = tl.load(
        gate_ptr
        + b * stride_g_b
        + i[:, None, None] * stride_g_i
        + j[None, :, None] * stride_g_j
        + h[None, None, :] * stride_g_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    # sigmoid (FP32)
    gate_f32 = gate.to(tl.float32)
    gate_sig = 1.0 / (1.0 + tl.exp(-gate_f32))
    gate_sig = gate_sig.to(tl.float16)

    out = out * gate_sig

    tl.store(
        out_ptr
        + b * stride_o_b
        + i[:, None, None] * stride_o_i
        + j[None, :, None] * stride_o_j
        + h[None, None, :] * stride_o_h,
        out,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )


def _apply_out_gate_triton(out_tensor, gate_tensor):
    """Fuse sigmoid+multiply for the out‑gate."""
    B, N, _, H = out_tensor.shape

    s_g = gate_tensor.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _apply_out_gate_kernel[grid](
        out_tensor,
        gate_tensor,
        B, N, H,
        s_o[0], s_o[1], s_o[2], s_o[3],
        s_g[0], s_g[1], s_g[2], s_g[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # out_tensor is now multiplied by sigmoid(gate)


# -------------------------------------------------------------
# Outgoing TriMul – forward‑only entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data : (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys
                * dim            – input channel C
                * hidden_dim     – hidden dimension H
                * nomask         – bool (True ⇒ ignore mask)
                * layernorm_eps  – optional eps for LayerNorm (float)

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # ---------------------------------------------------------
    # unpack
    # ---------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    assert C == dim, f"dim mismatch: got {C}, expected {dim}"

    # ---------------------------------------------------------
    # model weights (all FP32 on device)
    # ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]          # (H, C)
    right_proj_weight = weights["right_proj.weight"]        # (H, C)

    left_gate_weight = weights["left_gate.weight"]          # (H, C)
    right_gate_weight = weights["right_gate.weight"]        # (H, C)
    out_gate_weight = weights["out_gate.weight"]            # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]      # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]          # (H,)
    to_out_weight = weights["to_out.weight"]                # (C, H)

    # ---------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32 → FP16 for the heavy path)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C] float32

    x_h = x.half()                     # FP16

    # ---------------------------------------------------------
    # 2️⃣  Fuse four linear projections (left, right, left‑gate, right‑gate)
    # ---------------------------------------------------------
    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)    # (4*H, C)

    x_flat = x_h.view(-1, dim)                    # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)   # (B·N·N, 4·H)

    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # ---------------------------------------------------------
    # 3️⃣  Mask + sigmoid‑gate (single Triton kernel)
    # ---------------------------------------------------------
    if mask_tensor is None:
        # no mask supplied – treat as all‑ones
        mask_h = torch.empty(1, dtype=torch.float16, device=device)  # dummy
        has_mask = False
    else:
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        has_mask = not nomask

    _apply_mask_and_gate_triton(
        left_proj,
        right_proj,
        left_gate_pre,
        right_gate_pre,
        mask_h,
        has_mask,
    )
    # gate‑pre tensors are no longer needed
    del left_gate_pre, right_gate_pre

    # ---------------------------------------------------------
    # 4️⃣  Core contraction – batched GEMM
    # ---------------------------------------------------------
    # reshape to (B*H, N, N)
    left_bh = left_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_bh = right_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # (B*H, N, N) @ (B*H, N, N).T  →  (B*H, N, N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_bh.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # 5️⃣  Hidden‑dim LayerNorm (FP16)
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )

    # ---------------------------------------------------------
    # 6️⃣  Output gate (linear + sigmoid) – fused Triton kernel
    # ---------------------------------------------------------
    out_gate_pre = F.linear(x_h, out_gate_weight.half())   # (B,N,N,H) fp16
    _apply_out_gate_triton(out, out_gate_pre)            # in‑place multiply
    del out_gate_pre  # free

    # ---------------------------------------------------------
    # 7️⃣  Final projection back to C
    # ---------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,C) fp16

    # ---------------------------------------------------------
    # Return FP32 (matching reference)
    # ---------------------------------------------------------
    return out.float()
