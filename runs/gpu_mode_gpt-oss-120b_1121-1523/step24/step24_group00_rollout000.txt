@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    m_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # load A
        a = tl.load(A_ptr + batch_id * stride_a_batch + m_offset[:, None] * stride_a_m + k_offset[None, :] * stride_a_k,
                    mask=(m_offset[:, None] < M) & (k_offset[None, :] < K),
                    other=0.0)
        # load B
        b = tl.load(B_ptr + batch_id * stride_b_batch + k_offset[:, None] * stride_b_k + n_offset[None, :] * stride_b_n,
                    mask=(k_offset[:, None] < K) & (n_offset[None, :] < N),
                    other=0.0)
        # compute dot
        acc += tl.dot(a, b)
        # advance
    # Convert to fp16 and store
    c = acc.to(tl.float16)
    tl.store(C_ptr + batch_id * stride_c_batch + m_offset[:, None] * stride_c_m + n_offset[None, :] * stride_c_n,
             c,
             mask=(m_offset[:, None] < M) & (n_offset[None, :] < N))

                         stride_b_batch, stride_b_k, stride_b_n,
                         stride_c_batch, stride_c_m, stride_c_n,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # batch id
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid indices

"""
Fast outgoing TriMul (AlphaFold‑3) – forward only

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32) → cast to FP16.
2️⃣ Fuse the four linear maps (left/right projections + their gates) into a single
   GEMM that produces a tensor of shape (B, N, N, 4·H).  Split it back into
   left_proj, right_proj, left_gate_pre, right_gate_pre.
3️⃣ If a mask is present, launch a tiny Triton kernel that fuses:
       * sigmoid on the gate pre‑activations,
       * multiplication by the mask,
       * element‑wise multiplication with the projections.
   The kernel works in‑place on the projected tensors.
4️⃣ Core O(N³·H) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a **single** batched GEMM (B·H independent N×N matmuls)
   using a custom Triton kernel – far cheaper than launching H separate cuBLAS
   kernels.
5️⃣ Hidden‑dim LayerNorm + output gate (sigmoid) and final linear projection
   back to the original channel dimension.
6️⃣ Cast the result back to FP32 for compatibility with the reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + sigmoid‑gate fusion (in‑place on the projections)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_mul_kernel(
    # tensors (all fp16)
    left_ptr, right_ptr,                 # (B,N,N,H) – will be overwritten
    left_gate_ptr, right_gate_ptr,       # (B,N,N,H) – pre‑gate values
    mask_ptr,                            # (B,N,N) – fp16 mask (0/1)

    # static shapes
    B, N, H,
    # strides
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    In‑place update:
        left[b,i,j,h]  = left_proj[b,i,j,h] *
                         sigmoid(left_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
        right[b,i,j,h] = right_proj[b,i,j,h] *
                         sigmoid(right_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
    All tensors are fp16.  The kernel tiles over (i,j,h).
    """
    b = tl.program_id(0)           # batch dimension
    pid_i = tl.program_id(1)       # tile over i
    pid_j = tl.program_id(2)       # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # Load mask tile and broadcast over H
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=tl.float16(0.0),
    )                                   # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    h_offset = 0
    while h_offset < H:
        h = h_offset + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # Load projection and gate tiles
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )
        left_g = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )
        right_g = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )

        # Sigmoid (computed in FP32 for stability)
        left_gate = 1.0 / (1.0 + tl.exp(-left_g.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_g.to(tl.float32)))
        left_gate = left_gate.to(tl.float16)
        right_gate = right_gate.to(tl.float16)

        # Apply mask and gate
        left_out = left * left_gate * mask_tile
        right_out = right * right_gate * mask_tile

        # Store the gated & masked values back in‑place
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        h_offset += BLOCK_H


def _apply_mask_and_gate_triton(left_proj, right_proj,
                                left_gate_pre, right_gate_pre,
                                mask):
    """
    Fuse mask, sigmoid‑gate and element‑wise multiplication using Triton.
    All tensors must be fp16.
    The kernel overwrites ``left_proj`` and ``right_proj`` with their gated &
    masked values.
    """
    B, N, _, H = left_proj.shape
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_mul_kernel[grid](
        left_proj, right_proj,
        left_gate_pre, right_gate_pre,
        mask,
        B, N, H,
        *left_proj.stride(),
        *right_proj.stride(),
        *left_gate_pre.stride(),
        *right_gate_pre.stride(),
        *mask.stride(),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left_proj / right_proj now contain gated & masked values


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where A,B are (batch, N, N) fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for each batch id:
        C = A @ Bᵀ          (both in fp16, accumulation in fp32)
    Shapes:
        A : (batch, M, K)   – row‑major
        B : (batch, K, N)   – already transposed w.r.t. the original right‑proj
        C : (batch, M, N)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            A_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )
        b = tl.load(
            B_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )
        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_triton(A, B):
    """
    Wrapper around the above kernel.
    A : (batch, M, K) fp16, contiguous
    B : (batch, K, N) fp16, contiguous (already transposed form of the right proj)
    Returns C : (batch, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty_like(A)

    # strides for the flattened (batch, M, K) tensors
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        A, B, C,
        batch, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys
                * dim            – input channel C (int)
                * hidden_dim     – hidden dimension H (int)
                * nomask         – bool, True ⇒ ignore mask
                * layernorm_eps  – optional epsilon for LayerNorm

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    assert C == dim, f"Channel mismatch: got {C}, expected {dim}"

    # -----------------------------------------------------------------
    # Model weights (FP32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) → fp16 for the heavy path
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] float32
    x_h = x.half()  # fp16

    # -----------------------------------------------------------------
    # 2️⃣ Fuse left/right proj + gates into a single GEMM (4*H)
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_weight,
         right_proj_weight,
         left_gate_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)           # (4·H, C)

    x_flat = x_h.view(-1, dim)                         # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)        # (B·N·N, 4·H)

    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask – fuse mask + sigmoid‑gate via Triton
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_and_gate_triton(
            left_proj, right_proj,
            left_gate_pre, right_gate_pre,
            mask_h,
        )
        # left_proj / right_proj now contain gated & masked values
    else:
        # No mask: only apply sigmoid‑gate (pure PyTorch)
        left_gate = left_gate_pre.sigmoid()
        right_gate = right_gate_pre.sigmoid()
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # free gate pre‑activations early
    del left_gate_pre, right_gate_pre

    # -----------------------------------------------------------------
    # 4️⃣ Core contraction: batched GEMM (left @ rightᵀ) across hidden dim
    # -----------------------------------------------------------------
    # Re‑order dimensions so that every hidden channel becomes a separate batch.
    # After the permutations the memory layout is contiguous (row‑major) for each
    # (b, h) matrix – ideal for the custom kernel.
    left_contig = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_contig = right_proj.permute(0, 3, 2, 1).contiguous() # (B, H, N, N), already transposed

    A = left_contig.view(-1, N, N)   # (B·H, N, N)
    B = right_contig.view(-1, N, N)  # (B·H, N, N)   (k as rows, j as cols)

    C = _batched_gemm_triton(A, B)   # (B·H, N, N)  fp16

    # Reshape back to (B, N, N, H)
    out = C.view(B, hidden, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final projection back to original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Cast back to FP32 (the reference implementation works in FP32)
    # -----------------------------------------------------------------
    return out.float()
