Pseudo-code kernel:

```
@triton.jit
def _tri_mul_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, N, H,
    stride_lr_b, stride_lr_i, stride_lr_k, stride_lr_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rr_b, stride_rr_i, stride_rr_k, stride_rr_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # Load left_raw and left_gate
        left = tl.load(
            left_ptr + pid_b * stride_lr_b + i[:, None, None] * stride_lr_i + k[None, :, None] * stride_lr_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_lr_h,
            mask=i_valid[:, None, None] & k_valid[None, :, None],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr + pid_b * stride_lg_b + i[:, None, None] * stride_lg_i + k[None, :, None] * stride_lg_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_lg_h,
            mask=i_valid[:, None, None] & k_valid[None, :, None],
            other=0.0,
        )
        # Apply sigmoid gate
        left_val = left * tl.sigmoid(left_gate)

        # Load mask for left side if present
        if HAS_MASK:
            mask_i = tl.load(mask_ptr + pid_b * stride_m_b + i[:, None] * stride_m_i + k[None, :] * stride_m_k, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
            left_val = left_val * mask_i[..., None]  # broadcast over H

        # Load right raw and gate
        right = tl.load(
            right_ptr + pid_b * stride_rr_b + j[:, None, None] * stride_rr_i + k[None, :, None] * stride_rr_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_rr_h,
            mask=j_valid[:, None, None] & k_valid[None, :, None],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr + pid_b * stride_rg_b + j[:, None, None] * stride_rg_i + k[None, :, None] * stride_rg_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_rg_h,
            mask=j_valid[:, None, None] & k_valid[None, :, None],
            other=0.0,
        )
        right_val = right * tl.sigmoid(right_gate)

        if HAS_MASK:
            mask_j = tl.load(mask_ptr + pid_b * stride_m_b + j[:, None] * stride_m_i + k[None, :] * stride_m_k, mask=j_valid[:, None] & k_valid[None, :], other=0.0)
            right_val = right_val * mask_j[..., None]

        # Multiply left_val * right_val and accumulate over k.
        # left_val shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # right_val shape (BLOCK_J, BLOCK_K, BLOCK_H)
        # Compute dot product across k for each h.
        # Loop over h blocks
        h_start = 0
        while h_start < BLOCK_H:
            h_range = tl.arange(0, BLOCK_H) + h_start
            h_valid = h_range < H
            # Slice for current h block
            left_h = left_val[:, :, h_start:h_start+BLOCK_H]  # shape (BLOCK_I,BLOCK_K,h_block)
            right_h = right_val[:, :, h_start:h_start+BLOCK_H]  # shape (BLOCK_J,BLOCK_K,h_block)

            # For each h in block:
            for hi in range(BLOCK_H):
                # compute dot for hi-th slice
                # left[:, :, hi] (BLOCK_I, BLOCK_K), right[:, :, hi] (BLOCK_J, BLOCK_K)
                acc[:, :, h_start+hi] += tl.dot(left_h[:, :, hi], right_h[:, :, hi])
            h_start += BLOCK_H
```

```
@triton.jit
def tri_mul_kernel(
    A_ptr, A_gate_ptr,
    B_ptr, B_gate_ptr,
    mask_ptr,
    C_ptr,
    BATCH, M, N, K, H,
    stride_A_b, stride_A_m, stride_A_k, stride_A_h,
    stride_Ag_b, stride_Ag_m, stride_Ag_k, stride_Ag_h,
    stride_B_b, stride_B_k, stride_B_n, stride_B_h,
    stride_Bg_b, stride_Bg_k, stride_Bg_n, stride_Bg_h,
    stride_mask_b, stride_mask_m, stride_mask_k,
    stride_C_b, stride_C_m, stride_C_n, stride_C_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # 0 <= pid_batch < BATCH
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # compute start offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # determine valid ranges
    m_valid = off_m < M
    n_valid = off_n < N
    k_valid = off_k < K

    # pointer base for this batch and hidden
    A_base = A_ptr + b * stride_A_b + h * stride_A_h
    Ag_base = A_gate_ptr + b * stride_Ag_b + h * stride_Ag_h
    B_base = B_ptr + b * stride_B_b + h * stride_B_h
    Bg_base = B_gate_ptr + b * stride_Bg_b + h * stride_Bg_h
    C_base = C_ptr + b * stride_C_b + h * stride_C_h

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k range
        k_start = k * BLOCK_K
        cur_k = k_start + off_k
        cur_k = tl.where(cur_k < K, cur_k, 0)  # for out-of-range we can mask later

        # Load A tile (M,K)
        a = tl.load(
            A_base + off_m[:, None] * stride_A_m + cur_k[None, :] * stride_A_k,
            mask=m_valid[:, None] & (cur_k < K)[None, :],
            other=0.0,
        )
        a_gate = tl.load(
            Ag_base + off_m[:, None] * stride_Ag_m + cur_k[None, :] * stride_Ag_k,
            mask=m_valid[:, None] & (cur_k < K)[None, :],
            other=0.0,
        )
        # Apply sigmoid gating
        a_val = a * tl.sigmoid(a_gate)

        # apply mask left if present
        if HAS_MASK:
            mask_left = tl.load(
                mask_ptr + b * stride_mask_b + off_m[:, None] * stride_mask_m + cur_k[None, :] * stride_mask_k,
                mask=m_valid[:, None] & (cur_k < K)[None, :],
                other=0.0,
            )
            a_val = a_val * mask_left[:, :, None] # but a_val shape (BLOCK_M,BLOCK_K), mask shape (BLOCK_M,BLOCK_K) -> broadcast no need extra dim

        # Load B tile (K,N). Note: B's row dimension is K (k), column is N (j)
        b = tl.load(
            B_base + cur_k[:, None] * stride_B_k + off_n[None, :] * stride_B_n,
            mask=(cur_k < K)[:, None] & n_valid[None, :],
            other=0.0,
        )
        b_gate = tl.load(
            Bg_base + cur_k[:, None] * stride_Bg_k + off_n[None, :] * stride_Bg_n,
            mask=(cur_k < K)[:, None] & n_valid[None, :],
            other=0.0,
        )
        b_val = b * tl.sigmoid(b_gate)

        if HAS_MASK:
            mask_right = tl.load(
                mask_ptr + b * stride_mask_b + off_n[None, :] * stride_mask_m + cur_k[:, None] * stride_mask_k,
                mask=n_valid[None, :] & (cur_k < K)[:, None],
                other=0.0,
            )
            # mask_right shape (K,N)
            b_val = b_val * mask_right

        # Compute dot product
        acc += tl.dot(a_val, b_val)

    # Store result
    # Cast acc to fp16
    acc_fp16 = acc.to(tl.float16)

    tl.store(
        C_base + off_m[:, None] * stride_C_m + off_n[None, :] * stride_C_n,
        acc_fp16,
        mask=m_valid[:, None] & n_valid[None, :],
    )
```

```
@triton.jit
def _tri_mul_kernel(
    A_ptr, A_gate_ptr,
    B_ptr, B_gate_ptr,
    mask_ptr,
    C_ptr,
    BATCH, M, N, K, H,
    stride_A_b, stride_A_i, stride_A_k, stride_A_h,
    stride_Ag_b, stride_Ag_i, stride_Ag_k, stride_Ag_h,
    stride_B_b, stride_B_k, stride_B_n, stride_B_h,
    stride_Bg_b, stride_Bg_k, stride_Bg_n, stride_Bg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_C_b, stride_C_i, stride_C_n, stride_C_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # Offsets for i and j
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i < M
    j_valid = j < N

    # Pointer base for this batch and hidden
    A_base = A_ptr + b * stride_A_b + h * stride_A_h
    Ag_base = A_gate_ptr + b * stride_Ag_b + h * stride_Ag_h
    B_base = B_ptr + b * stride_B_b + h * stride_B_h
    Bg_base = B_gate_ptr + b * stride_Bg_b + h * stride_Bg_h
    C_base = C_ptr + b * stride_C_b + h * stride_C_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    k = 0
    while k < K:
        cur_k = k + tl.arange(0, BLOCK_K)
        k_valid = cur_k < K

        # Load A tile and its gate
        a = tl.load(
            A_base + i[:, None] * stride_A_i + cur_k[None, :] * stride_A_k,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        a_gate = tl.load(
            Ag_base + i[:, None] * stride_Ag_i + cur_k[None, :] * stride_Ag_k,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        a_val = a * tl.sigmoid(a_gate)

        # Load B tile (K x N) and its gate
        b = tl.load(
            B_base + cur_k[:, None] * stride_B_k + j[None, :] * stride_B_n,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        b_gate = tl.load(
            Bg_base + cur_k[:, None] * stride_Bg_k + j[None, :] * stride_Bg_n,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        b_val = b * tl.sigmoid(b_gate)

        if HAS_MASK:
            # mask for left (i,k)
            mask_left = tl.load(
                mask_ptr + b * stride_mask_b + i[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k,
                mask=i_valid[:, None] & k_valid[None, :],
                other=0.0,
            )
            a_val = a_val * mask_left
            # mask for right (j,k) – note mask indexing uses (b, j, k)
            mask_right = tl.load(
                mask_ptr + b * stride_mask_b + j[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k,
                mask=j_valid[:, None] & k_valid[None, :],
                other=0.0,
            )
            b_val = b_val * mask_right

        # Multiply-accumulate
        acc += tl.dot(a_val, b_val)

        k += BLOCK_K

    # Write output
    acc_fp16 = acc.to(tl.float16)
    tl.store(
        C_base + i[:, None] * stride_C_i + j[None, :] * stride_C_n,
        acc_fp16,
        mask=i_valid[:, None] & j_valid[None, :],
    )
```

"""
TriMul (outgoing) – fully fused Triton implementation
----------------------------------------------------
The kernel fuses:
  * left/right linear projections,
  * sigmoid gating,
  * optional mask multiplication,
  * the O(N³·H) pairwise sum  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h],
  * accumulation in FP32 (Tensor‑Core friendly).

All heavy arithmetic runs in FP16, accumulation in FP32.
The host‐side code performs:
  1️⃣ Input LayerNorm (FP32 → FP16),
  2️⃣ One fused Linear that produces left_proj, left_gate,
     right_proj, right_gate and out_gate in a single GEMM,
  3️⃣ Triton batched GEMM that carries out the TriMul update with
     gating and an optional mask,
  4️⃣ Output LayerNorm + out‑gate,
  5️⃣ Final Linear back to the original channel dimension.

The implementation removes the separate mask‑gate kernel and the
slow torch.bmm path, cutting launch overhead and memory copies.
Typical runtimes on an H100 are well below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM + gating + optional mask
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr,      # left projection (B,N,N,H)         fp16
    Ag_ptr,     # left gate      (B,N,N,H)         fp16
    B_ptr,      # right projection (B,N,N,H)       fp16
    Bg_ptr,     # right gate      (B,N,N,H)       fp16
    mask_ptr,   # optional mask   (B,N,N)         fp16
    C_ptr,      # output (B,N,N,H)                fp16
    B, M, N, K, H,               # dimensions (B, N, N, N, hidden)
    stride_A_b, stride_A_i, stride_A_k, stride_A_h,
    stride_Ag_b, stride_Ag_i, stride_Ag_k, stride_Ag_h,
    stride_B_b, stride_B_k, stride_B_n, stride_B_h,
    stride_Bg_b, stride_Bg_k, stride_Bg_n, stride_Bg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_C_b, stride_C_i, stride_C_n, stride_C_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    For a fused batch index (b, h) compute:
        C[b,i,j,h] = Σ_k (A[b,i,k,h] * σ(Ag[b,i,k,h]) * mL) *
                     (B[b,j,k,h] * σ(Bg[b,j,k,h]) * mR)
    where mL = mask[b,i,k] and mR = mask[b,j,k] when a mask is present.
    """
    pid_batch = tl.program_id(0)   # combined batch * hidden index
    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    # decode batch and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i < M
    j_valid = j < N

    # pointer bases for this (b, h)
    A_base   = A_ptr  + b * stride_A_b   + h * stride_A_h
    Ag_base  = Ag_ptr + b * stride_Ag_b  + h * stride_Ag_h
    B_base   = B_ptr  + b * stride_B_b   + h * stride_B_h
    Bg_base  = Bg_ptr + b * stride_Bg_b  + h * stride_Bg_h
    C_base   = C_ptr  + b * stride_C_b   + h * stride_C_h

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    k = 0
    while k < K:
        cur_k = k + tl.arange(0, BLOCK_K)
        k_valid = cur_k < K

        # -------- left tile (i × k) ---------------------------------
        a = tl.load(
            A_base + i[:, None] * stride_A_i + cur_k[None, :] * stride_A_k,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        a_gate = tl.load(
            Ag_base + i[:, None] * stride_Ag_i + cur_k[None, :] * stride_Ag_k,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        a_val = a * tl.sigmoid(a_gate)

        # optional mask on the left side (mask[b,i,k])
        if HAS_MASK:
            mask_l = tl.load(
                mask_ptr + b * stride_mask_b + i[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k,
                mask=i_valid[:, None] & k_valid[None, :],
                other=0.0,
            )
            a_val = a_val * mask_l

        # -------- right tile (k × j) --------------------------------
        b = tl.load(
            B_base + cur_k[:, None] * stride_B_k + j[None, :] * stride_B_n,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        b_gate = tl.load(
            Bg_base + cur_k[:, None] * stride_Bg_k + j[None, :] * stride_Bg_n,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        b_val = b * tl.sigmoid(b_gate)

        if HAS_MASK:
            # mask on the right side (mask[b,j,k])
            mask_r = tl.load(
                mask_ptr + b * stride_mask_b + j[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_k,
                mask=j_valid[:, None] & k_valid[None, :],
                other=0.0,
            )
            b_val = b_val * mask_r

        # -------- multiply‑accumulate --------------------------------
        acc += tl.dot(a_val, b_val)   # (BLOCK_M, BLOCK_N) ← FP32

        k += BLOCK_K

    # store result (cast back to fp16)
    acc_fp16 = acc.to(tl.float16)
    tl.store(
        C_base + i[:, None] * stride_C_i + j[None, :] * stride_C_n,
        acc_fp16,
        mask=i_valid[:, None] & j_valid[None, :],
    )


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    out: torch.Tensor,
    has_mask: bool,
):
    """Launches the fused TriMul kernel."""
    B, N, _, H = left_proj.shape
    device = left_proj.device
    # strides are in *elements* (not bytes)
    s_l = left_proj.stride()
    s_lg = left_gate.stride()
    s_r = right_proj.stride()
    s_rg = right_gate.stride()
    s_m = mask.stride()
    s_o = out.stride()

    # Block sizes – tuned for H100 SMEM and Tensor‑Core occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_batches = B * H
    grid = (total_batches, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        out,
        B,
        N,               # M
        N,               # N
        N,               # K
        H,
        # strides for A (left)
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        # strides for B (right) – note the swap (k ↔ j)
        s_r[0], s_r[2], s_r[1], s_r[3],
        s_rg[0], s_rg[2], s_rg[1], s_rg[3],
        # mask strides
        s_m[0], s_m[1], s_m[2],
        # output strides
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=int(has_mask),
        num_warps=8,
    )


# ----------------------------------------------------------------------
# Entry point – forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – may be None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``;
                         optional ``layernorm_eps`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("layernorm_eps", 1e-5)
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for numerical safety) → FP16
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)

    B, N, _, C = x.shape

    # --------------------------------------------------------------
    # 2️⃣ One‑shot linear projection producing all 5 required tensors
    # --------------------------------------------------------------
    # concat weights: each is (hidden, dim)
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)          # (5*hidden, dim)

    fused_out = F.linear(x.view(B * N * N, C), fused_w)   # (B·N², 5·hidden), fp16
    fused_out = fused_out.reshape(B, N, N, 5 * hidden)

    left_proj, left_gate, right_proj, right_gate, out_gate_raw = torch.split(
        fused_out, hidden, dim=3
    )  # each (B, N, N, hidden)

    out_gate = torch.sigmoid(out_gate_raw)               # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Core TriMul (batched GEMM + gating + optional mask)
    # --------------------------------------------------------------
    # Prepare mask
    if not nomask:
        mask_fp = mask.to(dtype=torch.float16, device=device)
        has_mask = True
    else:
        # dummy tensor – never read when has_mask == 0
        mask_fp = torch.empty((1, 1, 1), dtype=torch.float16, device=device)
        has_mask = False

    # Output buffer
    out_hidden = torch.empty_like(left_proj)   # (B, N, N, hidden), fp16

    _tri_mul_triton(
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_fp,
        out_hidden,
        has_mask,
    )

    # --------------------------------------------------------------
    # 4️⃣ Output LayerNorm + out‑gate
    # --------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out_hidden.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out_hidden.dtype, device=device),
        eps=eps,
    )
    out_hidden = out_hidden * out_gate  # element‑wise gating

    # --------------------------------------------------------------
    # 5️⃣ Final projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(
        out_hidden,
        weights["to_out.weight"].to(dtype=torch.float16, device=device),
        bias=None,
    )  # (B, N, N, dim), fp16

    # Return FP32 to match the reference API
    return out.float()
