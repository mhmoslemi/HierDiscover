    B, N, H,

"""
Fast outgoing TriMul (AlphaFold‑3) – forward only

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input in FP32 → cast to fp16 for the heavy path.
2️⃣ Fuse the four linear maps (proj + gates) into a single GEMM that yields
   tensors (left_proj, right_proj, left_gate_pre, right_gate_pre) of shape
   (B, N, N, H) in fp16.
3️⃣ If a mask is present, launch a tiny Triton kernel that fuses:
       * sigmoid on the gate pre‑activations,
       * multiplication by the mask,
       * element‑wise multiplication with the projections.
   When there is no mask we apply only the sigmoid‑gates in PyTorch.
4️⃣ Core contraction: compute for every hidden slice h
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (B·H independent N×N matmuls).  We implement
   it with a custom Triton kernel that works on the original strided layout
   and thus avoids the costly contiguity copies required by torch.matmul.
5️⃣ Hidden‑dim LayerNorm + output gate (sigmoid) and final linear projection
   back to the original channel dimension.
6️⃣ Cast the result back to FP32 (as required by the reference model).

All heavy arithmetic is performed in fp16 (Tensor‑Core friendly);
the only FP32 work is the first LayerNorm and the final cast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel 1: fuse mask, sigmoid‑gate and multiplication
# --------------------------------------------------------------
@triton.jit
def _mask_gate_mul_kernel(
    left_ptr, right_ptr,                # (B,N,N,H) – will be overwritten
    left_gate_ptr, right_gate_ptr,      # (B,N,N,H) – pre‑gate values
    mask_ptr,                           # (B,N,N)   – fp16 mask (0/1)

    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    In‑place update:
        left[b,i,j,h]  = left_proj[b,i,j,h] *
                         sigmoid(left_gate_pre[b,i,j,h]) * mask[b,i,j]
        right[b,i,j,h] = right_proj[b,i,j,h] *
                         sigmoid(right_gate_pre[b,i,j,h]) * mask[b,i,j]
    All tensors are fp16.  The kernel tiles over (i,j,h).

    pid_i = tl.program_id(1)       # tile over i
    pid_j = tl.program_id(2)       # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # load mask and broadcast over H
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b
                 + i[:, None] * stride_m_i
                 + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=tl.float16(0.0),
    )[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    h_off = 0
    while h_off < H:
        h = h_off + tl.arange(0, BLOCK_H)
        h_valid = h < H
        # mask for loading A/B/G tensors
        mask_ijkh = i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]

        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=mask_ijkh,
            other=tl.float16(0.0),
        )
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=mask_ijkh,
            other=tl.float16(0.0),
        )
        left_g = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=mask_ijkh,
            other=tl.float16(0.0),
        )
        right_g = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=mask_ijkh,
            other=tl.float16(0.0),
        )
        # sigmoid in fp32 for stability
        left_gate = 1.0 / (1.0 + tl.exp(-left_g.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_g.to(tl.float32)))
        left_gate = left_gate.to(tl.float16)
        right_gate = right_gate.to(tl.float16)

        left_out = left * left_gate * mask_tile
        right_out = right * right_gate * mask_tile

        # store back in‑place
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_out,
            mask=mask_ijkh,
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_out,
            mask=mask_ijkh,
        )
        h_off += BLOCK_H


def _apply_mask_and_gate_triton(left_proj, right_proj,
                                left_gate_pre, right_gate_pre,
                                mask):
    """Fuse mask, sigmoid‑gate and element‑wise multiplication using Triton."""
    B, N, _, H = left_proj.shape
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate_pre.stride()
    s_rg = right_gate_pre.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_mul_kernel[grid](
        left_proj,
        right_proj,
        left_gate_pre,
        right_gate_pre,
        mask,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left_proj / right_proj now contain gated & masked values


# --------------------------------------------------------------
# Triton kernel 2: batched GEMM for the O(N³·H) contraction
# --------------------------------------------------------------
@triton.jit
def _trimul_outgoing_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    for all batches b and hidden channels h.
    The kernel treats (b,h) as a combined batch dimension to launch a
    strided batched GEMM without any data copies.
    """
    pid_batch = tl.program_id(0)          # 0 .. B*H-1  (combined batch & hidden)
    pid_i = tl.program_id(1)              # tile along i
    pid_j = tl.program_id(2)              # tile along j

    # ------------------------------------------------------------------
    # Decode batch and hidden index
    # ------------------------------------------------------------------
    b = pid_batch // H
    h = pid_batch % H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i0 < N
    j_mask = j0 < N

    # ------------------------------------------------------------------
    # Base pointers for this (b,h) slice
    # ------------------------------------------------------------------
    a_base = left_ptr + b * stride_l_b + h * stride_l_h
    b_base = right_ptr + b * stride_r_b + h * stride_r_h
    c_base = out_ptr + b * stride_o_b + h * stride_o_h

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k0 < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + i0[:, None] * stride_l_i
            + k0[None, :] * stride_l_k,
            mask=i_mask[:, None] & k_mask[None, :],
            other=tl.float16(0.0),
        ).to(tl.float32)

        # B tile (transposed on‑the‑fly): shape (BLOCK_K, BLOCK_N)
        #   we want B_T[k, j] = right[b, j, k, h]
        b_tile = tl.load(
            b_base
            + k0[:, None] * stride_r_k
            + j0[None, :] * stride_r_j,
            mask=k_mask[:, None] & j_mask[None, :],
            other=tl.float16(0.0),
        ).to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(a, b_tile)

    # ------------------------------------------------------------------
    # Write result (cast back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base \
        + i0[:, None] * stride_o_i \
        + j0[None, :] * stride_o_j
    tl.store(c_ptrs, c, mask=i_mask[:, None] & j_mask[None, :])


def _trimul_matmul_triton(left, right, out):
    """Launch the batched GEMM kernel for the contraction."""
    B, N, _, H = left.shape
    # strides for the tensors
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # tile sizes – tuned for H100 Tensor‑Core performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                               # combined batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
    )
    _trimul_outgoing_matmul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_j, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # out now contains the contracted result


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys
                * dim            – input channel C (int)
                * hidden_dim     – hidden dimension H (int)
                * nomask         – bool, True ⇒ ignore mask
                * layernorm_eps  – optional epsilon for LayerNorm

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # C
    hidden = config["hidden_dim"]      # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    assert C == dim, f"Channel mismatch: got {C}, expected {dim}"

    # ----------------------------------------------------------
    # Model weights (FP32, already on device)
    # ----------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ----------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) → fp16 for the heavy path
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                 # [B, N, N, C] fp32
    x_h = x.half()                    # cast to fp16

    # ----------------------------------------------------------
    # 2️⃣ Fuse left/right proj + gates into a single GEMM (output 4*H)
    # ----------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_weight, right_proj_weight,
         left_gate_weight, right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    x_flat = x_h.view(-1, dim)                 # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)  # (B·N·N, 4*H)

    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # ----------------------------------------------------------
    # 3️⃣ Optional mask – fuse mask + sigmoid‑gate via Triton
    # ----------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_and_gate_triton(
            left_proj, right_proj,
            left_gate_pre, right_gate_pre,
            mask_h,
        )
    else:
        # No mask: only apply sigmoid‑gate (pure PyTorch)
        left_gate = left_gate_pre.sigmoid()
        right_gate = right_gate_pre.sigmoid()
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # Free the pre‑gate tensors early
    del left_gate_pre, right_gate_pre

    # ----------------------------------------------------------
    # 4️⃣ Core contraction – custom Triton GEMM
    # ----------------------------------------------------------
    out = torch.empty_like(left_proj)   # (B,N,N,H) fp16
    _trimul_matmul_triton(left_proj, right_proj, out)

    # ----------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm + output gate
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # ----------------------------------------------------------
    # 6️⃣ Final projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,C) fp16

    # ----------------------------------------------------------
    # Cast back to FP32 for compatibility with the reference implementation
    # ----------------------------------------------------------
    return out.float()
