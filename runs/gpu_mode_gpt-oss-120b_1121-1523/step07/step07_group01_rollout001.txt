@triton.jit
def trilu_mul_gemm_gate_kernel(
    left_ptr, right_ptr, gate_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    batch_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and hidden
    b = batch_h // H
    h = batch_h % H

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # compute base offset
    b_off = b * stride_b
    h_off = h

    # pointers
    a_ptrs = left_ptr + b_off + h_off + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k
    b_ptrs = right_ptr + b_off + h_off + offs_n[:, None] * stride_i + offs_k[None, :] * stride_k
    # load tiles with mask
    a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
    b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles for this k block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        a_ptrs += BLOCK_K * stride_k
        b_ptrs += BLOCK_K * stride_k

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _trimul_gemm_gate_kernel(
    left_ptr, right_ptr, gate_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_hidden = tl.program_id(0)  # ranges 0..B*H-1
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and hidden
    b = batch_hidden // H
    h = batch_hidden % H

    # offsets for block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # dynamic mask for dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base offsets
    b_off = b * stride_b
    h_off = h  # stride_h = 1

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to A and B tiles
        a_ptr = left_ptr + b_off + h_off + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        b_ptr = right_ptr + b_off + h_off + offs_j[:, None] * stride_i + offs_k[None, :] * stride_k

        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Multiply by out_gate
    gate_ptr_tile = gate_ptr + b_off + h_off + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    gate = tl.load(gate_ptr_tile, mask=mask_i[:, None] & mask_j[None, :], other=1.0)  # shape (BLOCK_M,BLOCK_N)
    acc = acc * gate

    # store result as fp16
    out = acc.to(tl.float16)
    out_ptr_tile = out_ptr + b_off + h_off + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptr_tile, out, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on dim
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # cast x to half for compute
    x_h = x.half()

    # cast weights to half
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"].half()

    # projections
    left = F.linear(x_h, left_proj_w)  # (B,N,N,hidden_dim)
    right = F.linear(x_h, right_proj_w)
    # gates
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # apply mask
    if not nomask:
        mask = mask_tensor.to(x_h.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # apply gates (in-place)
    left = left * left_gate
    right = right * right_gate

    # main tri-mul + out_gate
    B, N, _, _ = left.shape
    # output tensor
    out = torch.empty_like(left, dtype=torch.float16, device=device)  # shape (B,N,N,hidden_dim)
    
    # strides
    stride_b, stride_i, stride_k, stride_h = left.stride()  # (B,N,N,H) layout
    # Should be (N*N*H, N*H, H, 1)
    # assert stride_h == 1 and stride_k == hidden_dim and stride_i == N*hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    total_batch = B * hidden_dim
    grid = (total_batch, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    _trimul_gemm_gate_kernel[grid](
        left, right, out_gate, out,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Cast to float32
    out_fp32 = out.float()
    # to_out_norm (layernorm on hidden_dim)
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)
    # multiply with out_gate (float32)
    out_norm = out_norm * out_gate.float()
    # final linear projection (to original dim)
    # Use fp16 weight and fp16 input for speed
    out_final = F.linear(out_norm.half(), to_out_w).float()
    return out_final

"""
TriMul (outgoing) – fully fused Triton implementation
----------------------------------------------------
* Layer‑norm on the input channels (C) – torch (fast, small work).
* All linear projections, sigmoid gates and optional masking are done in
  fp16.  The tensors `left`, `right` and `out_gate` stay in the original
  layout ``[B, N, N, H]`` (no costly ``permute`` / copy).
* The O(N³·H) reduction is a batched GEMM that is written entirely in
  Triton.  It computes for every hidden‑dimension ``h``:

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] * out_gate[b,i,j,h]

  The hidden dimension is fused into the batch axis (total_batch = B*H)
  and the kernel operates directly on the original strides, thus avoiding
  the ``permute(...).contiguous()`` that was the main overhead of the
  reference implementation.
* The element‑wise ``out_gate`` multiplication is fused into the GEMM,
  removing an extra Triton kernel launch.
* The remaining cheap post‑processing (layer‑norm on the hidden dim and
  a final linear projection back to ``dim``) is kept in torch; they are
  memory‑bound and negligible compared with the cubic work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM + out_gate multiplication.
# Input/output layout: [B, N, N, H] (contiguous, stride order
# (B,N,N,H) → (stride_b, stride_i, stride_k, 1)).
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_gate_kernel(
    left_ptr,        # fp16   [B,N,N,H]
    right_ptr,       # fp16   [B,N,N,H]
    gate_ptr,        # fp16   [B,N,N,H]
    out_ptr,         # fp16   [B,N,N,H]  (output)
    B, N, H,         # runtime scalars
    stride_b, stride_i, stride_k,   # strides for the three tensors
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 0‑dim = batch * hidden, 1‑dim = i‑tile, 2‑dim = j‑tile
    batch_h = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # ----- decode batch and hidden‑dim ---------------------------------
    b = batch_h // H          # batch index
    h = batch_h %  H          # hidden index (0 ≤ h < H)

    # ----- tile offsets -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offset for the current (b, h)
    batch_offset = b * stride_b          # move to the right batch
    hidden_offset = h                    # stride_h == 1 → just add h

    # accumulator in fp32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over the reduction dimension k -------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_off  = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A = left[b, i, k, h]   shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr + batch_offset + hidden_offset \
                + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B = right[b, j, k, h]  shape (BLOCK_N, BLOCK_K)
        b_ptr = right_ptr + batch_offset + hidden_offset \
                + offs_j[:, None] * stride_i + offs_k[None, :] * stride_k
        b = tl.load(b_ptr,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        # dot‑product over k
        acc += tl.dot(a, b)

    # ----- apply out_gate (still in fp16) ------------------------------
    gate_tile_ptr = gate_ptr + batch_offset + hidden_offset \
                    + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    gate = tl.load(gate_tile_ptr,
                   mask=mask_i[:, None] & mask_j[None, :],
                   other=1.0)                # scalar 1 → no effect if missing
    acc = acc * gate.to(tl.float32)          # fp32 * fp32

    # ----- write result (fp16) -----------------------------------------
    out_tile_ptr = out_ptr + batch_offset + hidden_offset \
                   + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_tile_ptr, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point – “outgoing” TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, C]      (float32)
        * mask_tensor  : torch.Tensor [B, N, N]       (bool/float) – may be ignored
        * weights_dict : dict of model weights (see docstring above)
        * config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    eps = 1e-5

    dim        = cfg["dim"]          # C
    hidden_dim = cfg["hidden_dim"]   # H
    nomask     = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Layer‑norm on the input channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 and perform all linear / gate ops in half‑precision
    # ------------------------------------------------------------------
    x_h = x.half()

    # ----- cast all linear weights to fp16 once -----------------------
    left_proj_w   = w["left_proj.weight"].half()
    right_proj_w  = w["right_proj.weight"].half()
    left_gate_w   = w["left_gate.weight"].half()
    right_gate_w  = w["right_gate.weight"].half()
    out_gate_w    = w["out_gate.weight"].half()
    to_out_w      = w["to_out.weight"].half()          # (C, H)  →  fp16

    # ----- projections -------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B,N,N,H] fp16
    right = F.linear(x_h, right_proj_w)  # [B,N,N,H] fp16

    # ----- sigmoid gates (still fp16) ---------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # ----- optional mask (broadcast over hidden dim) ------------------
    if not nomask:
        m = mask.to(x_h.dtype).unsqueeze(-1)       # [B,N,N,1]
        left = left * m
        right = right * m

    # ----- apply gates in‑place ----------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Core O(N³·H) reduction – fused Triton GEMM + out_gate
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, N, N, H
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides for the contiguous layout [B,N,N,H]
    stride_b, stride_i, stride_k, stride_h = left.stride()   # (N*N*H, N*H, H, 1)
    # sanity check (only for debugging; can be removed for speed)
    # assert stride_h == 1 and stride_k == hidden_dim and stride_i == N * hidden_dim

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batch = B * hidden_dim
    grid = (total_batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _trimul_gemm_gate_kernel[grid](
        left, right, out_gate, out,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4) Post‑processing (layer‑norm on hidden dim, out_gate already fused)
    # ------------------------------------------------------------------
    out_f32 = out.float()                         # cast back to fp32 for LN
    out_norm = F.layer_norm(
        out_f32,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    # (the out_gate multiplication was already fused inside the GEMM)

    # ------------------------------------------------------------------
    # 5) Final linear projection back to original dim (fp16 for speed)
    # ------------------------------------------------------------------
    # cast to fp16, apply final linear, cast back to fp32
    final = F.linear(out_norm.half(), to_out_w).float()   # (B,N,N,dim)

    return final
