"""
TriMul (outgoing) – heavily‑fused fp16 implementation
----------------------------------------------------
* Layer‑norm on the input (C‑dim) is done in fp32 (high‑accuracy).
* The five linear layers (proj, gate‑proj, out‑gate) are fused into a
  single half‑precision GEMM:  Y = X @ Wcatᵀ   where Wcat = [Wₗ, Wₗg, Wᵣ, Wᵣg, Wₒg].
* Mask, sigmoid‑gates and the element‑wise left/right gating are applied
  in fp16 without extra memory copies.
* The O(N³ · H) “tri‑mul” is performed with a single batched half‑precision
  GEMM (torch.bmm).  Input tensors are kept in fp16 throughout to avoid
  costly cast‑backs.
* A tiny Triton kernel fuses the final element‑wise multiplication with the
  out‑gate (still half‑precision).  All remaining ops (layer‑norm on the
  hidden dimension, final linear projection) are executed with the highly
  optimized PyTorch kernels on fp16 data.
* The whole pipeline reduces the number of large memory passes from five
  (separate linear layers) to two (the fused projection and the batched GEMM),
  which brings the end‑to‑end latency for the largest test case (B=1, N=1024,
  H=128, C=384) well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = out * gate   (both half‑precision, same shape)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_kernel(
    out_ptr,          # fp16*  – flattened output tensor
    gate_ptr,         # fp16*  – flattened gate tensor (same shape)
    total_elem: tl.constexpr,   # total number of elements (runtime scalar)
    BLOCK_SIZE: tl.constexpr,   # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elem

    out_val  = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor[B, N, N, C]      (float32)
        * mask_tensor  : torch.Tensor[B, N, N]        (bool/float) – may be ignored
        * weights_dict : dict with the following keys (float32 tensors)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight", "to_out.bias"
        * config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}
    Returns
    -------
    out : torch.Tensor[B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    eps = cfg.get("eps", 1e-5)

    B, N, _, C = x.shape
    H = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 2) Input layer‑norm (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  (fp32)

    # --------------------------------------------------------------
    # 3) Cast to fp16 and fuse the five linear maps into one GEMM
    # --------------------------------------------------------------
    x_fp16 = x.half()                     # [B,N,N,C] fp16
    # concatenate weights: [left_proj, left_gate, right_proj, right_gate, out_gate]
    w_cat_fp16 = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).half()                               # (5*H, C)

    # fused projection -> shape (B,N,N,5*H)
    fused = F.linear(x_fp16, w_cat_fp16)   # bias=None
    # split back
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, H, dim=-1
    )  # each: [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 4) Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B,N,N] → [B,N,N,1] half
        mask_fp16 = mask.to(x_fp16.dtype).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 5) Apply sigmoid gates and element‑wise left/right gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # fp16
    right_gate = torch.sigmoid(right_gate) # fp16

    left = left_proj * left_gate           # [B,N,N,H] fp16
    right = right_proj * right_gate        # [B,N,N,H] fp16
    out_gate = torch.sigmoid(out_gate)     # fp16   (used later)

    # --------------------------------------------------------------
    # 6) Core O(N³·H) reduction – batched half‑precision GEMM
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) with the hidden dimension as batch dim
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # reshape back to (B, N, N, H)  (still fp16)
    out = (
        out_b.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 7) Hidden‑dim layer‑norm (fp16)  +  out‑gate multiplication
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )  # [B,N,N,H] fp16

    # Triton fused multiplication out = out * out_gate
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_elems = out_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    _apply_gate_kernel[grid](
        out_ptr=out_flat,
        gate_ptr=gate_flat,
        total_elem=total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # --------------------------------------------------------------
    # 8) Final linear projection back to original channel dim C
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].half(),
        bias=weights["to_out.bias"].half(),
    )  # [B,N,N,C] fp16

    # Cast back to float32 for external consumption
    return out.float()
