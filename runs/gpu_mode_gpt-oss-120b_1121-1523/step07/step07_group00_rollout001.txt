"""
TriMul (outgoing) – fast forward‑only implementation

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to FP16 and perform ONE fused linear projection that simultaneously
   computes left‑proj, left‑gate, right‑proj and right‑gate (weights are
   concatenated).  Split the result and apply the sigmoid gate.
3. Optional mask – applied in‑place with a tiny Triton kernel that multiplies
   the whole hidden dimension by mask[b,i,j].
4. Core O(N³) update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Realised as a single `torch.einsum` (equivalent to a batched GEMM).  The
   contraction runs in FP16 with Tensor‑Core acceleration; the result is
   cast back to FP32 for the following LayerNorm.
5. Output LayerNorm, output‑gate and final projection back to the original
   channel dimension (float32).

All heavy arithmetic is FP16; only the two LayerNorms and the final Linear
operate in FP32 to preserve numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,                     # fp16 *
    right_ptr,                    # fp16 *
    mask_ptr,                     # fp16 *
    B, N, H,                      # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Multiply `left` and `right` (shape [B,N,N,H]) by a scalar mask
    `mask` (shape [B,N,N]) broadcast over the hidden dimension.
    One program instance processes a `BLOCK_I × BLOCK_J` tile of the
    (i,j) plane and the whole hidden dimension (H ≤ BLOCK_H) in registers.
    """
    b = tl.program_id(0)
    ti = tl.program_id(1)
    tj = tl.program_id(2)

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # Load mask (scalar per (i,j)) and broadcast over H
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # Load left tile
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    # Load right tile
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )

    # Apply mask
    left = left * mask
    right = right * mask

    # Store back
    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """In‑place mask multiplication with a tiled Triton kernel."""
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    mask_h = mask.to(left.dtype)                     # fp16 mask

    # strides expressed in number of elements (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_h.stride()

    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 128                     # hidden_dim ≤ 128 in all tests

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_apply_kernel[grid](
        left,
        right,
        mask_h,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left/right are modified in‑place


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑aware forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Load model weights (all stored as float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # (H, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm  (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                                   # [B, N, N, dim] (fp32)

    # --------------------------------------------------------------
    # 2) Fused projection + gate (FP16)
    #    One linear call produces: left_proj, left_gate, right_proj, right_gate
    # --------------------------------------------------------------
    x_fp16 = x.half()

    # concatenate all four weight matrices -> shape (4*H, dim)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16)

    # Linear (B,N,N,4*H)
    combined = F.linear(x_fp16, combined_weight)        # no bias
    # split
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )
    # apply sigmoid gates
    left = left_proj * left_gate.sigmoid()               # [B,N,N,H] fp16
    right = right_proj * right_gate.sigmoid()            # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 3) Optional mask (in‑place via Triton)
    # --------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 4) Core O(N³) update – batched GEMM using einsum
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # einsum maps cleanly onto a batched GEMM (FP16 inputs, FP32 accumulation)
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)
    out = out.float()                                   # cast to FP32 for LN

    # --------------------------------------------------------------
    # 5) Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                                   # FP32

    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # FP32, shape [B,N,N,H]
    out = out * out_gate

    out = F.linear(out, to_out_weight)                  # back to dim (FP32)

    return out
