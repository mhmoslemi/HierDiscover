"""
custom_kernel – fast “outgoing” TriMul forward pass (AlphaFold3)

Algorithm
---------
1️⃣ Layer‑norm the input (float32) → `x`.
2️⃣ Fuse the four linear projections (left‑proj, left‑gate, right‑proj,
   right‑gate) into **one** half‑precision GEMM:
   * weight matrix `W_combined` has shape (4·H, C).
   * `F.linear(x_half, W_combined)` yields `[B, N, N, 4·H]`.
   * Split → `left_proj`, `left_gate`, `right_proj`, `right_gate`.
   * Apply sigmoid‑gate and multiply → `left`, `right` (fp16).

3️⃣ Optional mask: a single Triton kernel multiplies both `left` and `right`
   by `mask[b,i,j]` (broadcast over the hidden dimension).  This halves the
   number of launches compared with two separate element‑wise kernels.

4️⃣ Core O(N³) update – batched half‑precision matrix multiplication:
   * `left_mat  = left.permute(0, 3, 1, 2)`   # (B, H, N, N)
   * `right_mat = right.permute(0, 3, 1, 2)`  # (B, H, N, N)
   * `out_mat   = torch.matmul(left_mat, right_mat.transpose(-2, -1))`
     (Tensor‑Core fp16 → fp32 accumulation).
   * reshape back to `[B, N, N, H]`.

5️⃣ Output LayerNorm + output‑gate (sigmoid) + final projection
   (all float32) → result `[B, N, N, C]`.

Only the mask‑application is done with Triton; all heavy arithmetic uses
high‑throughput cuBLAS/Tensor‑Core GEMM and fused PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication for both left & right.
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    mask_ptr,               # fp16 *
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Multiply `left` and `right` by `mask` (broadcast over H)."""
    b  = tl.program_id(0)          # batch
    ti = tl.program_id(1)          # i‑tile
    tj = tl.program_id(2)          # j‑tile

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_ok = i < N
    j_ok = j < N
    h_ok = h < H

    # Load mask[i, j] (scalar) and broadcast over H
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                     # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]    # (BLOCK_I, BLOCK_J, 1)

    # Load left & right tiles
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        other=0.0,
    )

    left = left * mask
    right = right * mask

    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
    )


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor) -> None:
    """Launch the Triton kernel that multiplies `left` & `right` by `mask` (in‑place)."""
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    mask_h = mask.to(left.dtype)                 # fp16 mask

    # strides in elements (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_h.stride()

    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 128                               # hidden_dim ≤ 128 in all configs

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_apply_kernel[grid](
        left,
        right,
        mask_h,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left & right are modified in‑place


# ----------------------------------------------------------------------
# Entry‑point: forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑aware forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (all float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]     # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]         # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                       # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating (half‑precision)
    # ------------------------------------------------------------------
    x_half = x.half()                                   # fp16

    # Combine the four weight matrices into one big linear layer
    # order: left_proj, left_gate, right_proj, right_gate
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight,
         right_proj_weight, right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16)  # (4*H, C)

    # One half‑precision GEMM → [B, N, N, 4*H]
    combined = F.linear(x_half, combined_weight)

    # Split back to the four branches
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )

    # Apply sigmoid‑gates
    left = left_proj * left_gate.sigmoid()    # [B, N, N, H] (fp16)
    right = right_proj * right_gate.sigmoid() # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (in‑place Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³) update via batched half‑precision GEMM
    #    left, right : (B, N, N, H) → (B, H, N, N)
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2)      # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2)    # (B, H, N, N)

    # Batched matmul: left @ rightᵀ   → (B, H, N, N) in fp32
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1)).to(torch.float32)

    # Back to (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                             # fp32

    # output‑gate computed from the *normed* input `x`
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # (B,N,N,H) fp32
    out = out * out_gate

    # final linear back to original channel dimension `dim`
    out = F.linear(out, to_out_weight)                       # (B,N,N,dim) fp32

    return out
