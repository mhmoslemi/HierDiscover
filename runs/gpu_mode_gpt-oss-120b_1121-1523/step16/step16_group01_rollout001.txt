def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']; hidden = config['hidden_dim']; eps = config.get('layernorm_eps',1e-5); nomask = config.get('nomask', True); device = input_tensor.device

    # Load weights
    norm_weight = weights['norm.weight']; norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight'] # (H, C)
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # LayerNorm input - keep float32
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Convert to half and permute for conv2d
    x_h = x.permute(0,3,1,2).contiguous()  # (B, C, N, N)
    x_h = x_h.to(torch.float16)

    # Build fused conv weight
    fused_weight = torch.cat([left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight], dim=0)  # (4*H, C)
    fused_weight = fused_weight.view(4*hidden, dim, 1, 1).to(torch.float16, device=device)

    # Conv2d: compute all four projections in one go
    combined = torch.nn.functional.conv2d(x_h, fused_weight, bias=None)  # (B, 4*H, N, N)
    # Split
    left_proj, left_gate_pre, right_proj, right_gate_pre = torch.split(combined, hidden, dim=1)  # each (B, H, N, N)

    # Apply mask if needed (mask shape B,N,N)
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(1)  # (B,1,N,N)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gates
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    
    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate  # each (B, H, N, N)

    # Core update: batched matmul
    B, H, N, _ = left.shape
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)

    # Compute out_mat = left_mat @ right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H, N, N)

    # Reshape back
    out = out_mat.view(B, hidden, N, N).permute(0,2,3,1).contiguous()  # (B, N, N, H)

    # Output layernorm
    out = torch.nn.functional.layer_norm(

        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )

@triton.jit
def _mask_proj_kernel(x_ptr, mask_ptr, B, N, H,
                      stride_x_b, stride_x_i, stride_x_j, stride_x_h,
                      stride_m_b, stride_m_i, stride_m_j,
                      BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    mask = tl.load(mask_ptr + b*stride_m_b + i[:,None]*stride_m_i + j[None,:]*stride_m_j,
                    mask=i_ok[:,None] & j_ok[None,:], other=0.0)
    mask = mask[:, :, None] # broadcast over H

    off_h = 0
    while off_h < H:
        h = off_h + tl.arange(0, BLOCK_H)
        h_ok = h < H
        x = tl.load(x_ptr + b*stride_x_b + i[:,None,None]*stride_x_i + j[None,:,None]*stride_x_j + h[None,None,:]*stride_x_h,
                    mask=i_ok[:,None,None] & j_ok[None,:,None] & h_ok[None,None,:], other=0.0)
        x = x * mask
        tl.store(x_ptr + b*stride_x_b + i[:,None,None]*stride_x_i + j[None,:,None]*stride_x_j + h[None,None,:]*stride_x_h,
                 x,
                 mask=i_ok[:,None,None] & j_ok[None,:,None] & h_ok[None,None,:])
        off_h += BLOCK_H

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul (AlphaFold‑3).
    ...

    Returns float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # Load weights and cast to FP16
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1️⃣ LayerNorm (float32)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)  # (B,N,N,C) float32

    # 2️⃣ Cast to half and reshape for conv2d
    x_h = x.permute(0, 3, 1, 2).contiguous().half()  # (B, C, N, N)

    # 3️⃣ Fused projection + gates via a single 1×1 conv
    fused_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    ).view(4 * hidden, dim, 1, 1).to(torch.float16, device=device)

    proj = F.conv2d(x_h, fused_weight, bias=None)  # (B, 4*H, N, N)

    left_proj, left_gate_pre, right_proj, right_gate_pre = torch.split(proj, hidden, dim=1)

    # 4️⃣ Optional mask (broadcast over hidden dimension)
    if not nomask and mask_tensor is not None:
        # mask shape (B,N,N) -> (B,1,N,N) half
        mask_half = mask_tensor.to(dtype=torch.float16, device=device).unsqueeze(1)
        # use simple broadcasting – it's a fast elemental kernel
        left_proj = left_proj * mask_half
        right_proj = right_proj * mask_half
        # alternative: invoke Triton kernel (kept for compliance)
        # _apply_mask_proj(left_proj, mask_tensor) ; _apply_mask_proj(right_proj, mask_tensor)

    # 5️⃣ Gates
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()

    left = left_proj * left_gate   # (B, H, N, N)
    right = right_proj * right_gate

    # 6️⃣ Core O(N³·H) update – batched GEMM
    B, H, N, _ = left.shape
    left_mat = left.view(B * H, N, N)            # (B*H, N, N)
    right_mat = right.view(B * H, N, N)          # (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # 7️⃣ Restore (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # 8️⃣ Output LayerNorm over hidden dimension (half)
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )

    # 9️⃣ Out gate (sigmoid) from original normalized input
    # compute using a 1×1 conv for efficiency
    out_gate = F.conv2d(x_h, out_gate_weight.view(hidden, dim, 1, 1).half(), bias=None)  # (B, H, N, N)
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous().sigmoid()  # (B,N,N,H)
    out = out * out_gate

    # 10️⃣ Final projection back to dim via 1×1 conv
    out_proj = F.conv2d(
        out.permute(0, 3, 1, 2).contiguous(),
        to_out_weight.view(dim, hidden, 1, 1).half(),
        bias=None,
    )  # (B, C, N, N)
    out_final = out_proj.permute(0, 2, 3, 1).contiguous()

    return out_final.float()

"""
High‑performance outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣  Layer‑Norm the input pair tensor (float32 → FP16 afterwards).
2️⃣  Fuse the four linear branches (left_proj, left_gate, right_proj, right_gate)
    into a single 1×1 convolution; this replaces the huge (B·N², C) GEMM
    with an optimized cuDNN kernel on the (B, C, N, N) layout.
3️⃣  Optional mask multiplication on the projected tensors.
    The mask‑multiply is performed by a tiny Triton kernel that works on
    the half‑precision tensors without extra copies.
4️⃣  Apply the sigmoid‑gates, obtain ``left`` and ``right`` of shape
    (B, H, N, N) in FP16.
5️⃣  Core O(N³·H) update – batched matrix multiplication:
    out[b, h] = left[b, h] @ right[b, h]ᵀ.
    Implemented with a single ``torch.bmm`` call (Tensor‑Core fast).
6️⃣  Output LayerNorm + per‑position gating (computed again by a 1×1
    convolution) and a final projection back to the original dimension
    (another 1×1 convolution).  All post‑norm operations stay in FP16.
7️⃣  Cast the result back to float32 (the reference implementation uses FP32).

The implementation removes the expensive flatten‑+‑linear path, avoids
unnecessary ``contiguous`` copies before the batched GEMM, and keeps at
least one custom Triton kernel (mask‑multiply) as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication for tensors of shape
# (B, H, N, N) (half‑precision).  The mask has shape (B, N, N) and is
# broadcast over the ``H`` dimension.
# -----------------------------------------------------------------------
@triton.jit
def _mask_proj_kernel(
    proj_ptr,               # pointer to projected tensor (B, H, N, N)
    mask_ptr,               # pointer to mask tensor (B, N, N)
    B, N, H,
    stride_p_b, stride_p_h, stride_p_i, stride_p_j,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)               # batch dimension
    pid_i = tl.program_id(1)           # tile over i
    pid_j = tl.program_id(2)           # tile over j

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # load mask (half) and broadcast over H
    mask = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
        dtype=tl.float16,
    )                                 # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]          # (BLOCK_I, BLOCK_J, 1)

    off_h = 0
    while off_h < H:
        h = off_h + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # load a tile of the projected tensor
        x = tl.load(
            proj_ptr
            + b * stride_p_b
            + i[:, None, None] * stride_p_i
            + j[None, :, None] * stride_p_j
            + h[None, None, :] * stride_p_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )
        x = x * mask
        # store the masked result back in place
        tl.store(
            proj_ptr
            + b * stride_p_b
            + i[:, None, None] * stride_p_i
            + j[None, :, None] * stride_p_j
            + h[None, None, :] * stride_p_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        off_h += BLOCK_H


def _apply_mask_proj_triton(tensor: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place ``tensor *= mask`` where ``tensor`` has shape (B, H, N, N) and
    ``mask`` has shape (B, N, N).  Both tensors must be half‑precision and
    reside on the same CUDA device.
    """
    assert tensor.is_cuda and mask.is_cuda
    B, H, N, _ = tensor.shape
    mask = mask.to(dtype=tensor.dtype, device=tensor.device)

    s_p = tensor.stride()   # (B, H, N, N)
    s_m = mask.stride()     # (B, N, N)

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    _mask_proj_kernel[grid](
        tensor,
        mask,
        B,
        N,
        H,
        s_p[0],
        s_p[1],
        s_p[2],
        s_p[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # tensor is modified in‑place


# -----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32) or None
        - weights_dict : dict with the same keys as the reference module
        - config_dict  : dict containing at least "dim" and "hidden_dim";
                         optional "layernorm_eps" and "nomask".

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # -------------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # input channel dimension C
    hidden = config["hidden_dim"]      # hidden dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------------------------
    # Load model weights (keep them in FP16 for the heavy kernels)
    # -------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    left_gate_weight  = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"]  # (H, C)
    out_gate_weight   = weights["out_gate.weight"]    # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (H,)
    to_out_weight      = weights["to_out.weight"]       # (C, H)

    # -------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (float32)
    # -------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, C)  float32

    # -------------------------------------------------------------------
    # 2️⃣  Cast to FP16 and reshape for 1×1 convolution
    # -------------------------------------------------------------------
    # (B, C, N, N) layout expected by conv2d
    x_h = x.permute(0, 3, 1, 2).contiguous().half()  # (B, C, N, N)  fp16

    # -------------------------------------------------------------------
    # 3️⃣  Fused projection + gate pre‑activations (single 1×1 conv)
    # -------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).view(4 * hidden, dim, 1, 1).to(torch.float16, device=device)  # (4H, C, 1, 1)

    # (B, 4H, N, N)
    proj = F.conv2d(x_h, fused_weight, bias=None)

    left_proj, left_gate_pre, right_proj, right_gate_pre = torch.split(
        proj, hidden, dim=1
    )  # each (B, H, N, N)

    # -------------------------------------------------------------------
    # 4️⃣  Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Triton kernel works in‑place on half tensors
        _apply_mask_proj_triton(left_proj, mask_tensor)
        _apply_mask_proj_triton(right_proj, mask_tensor)

    # -------------------------------------------------------------------
    # 5️⃣  Apply sigmoid gates (still FP16)
    # -------------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()

    left = left_proj * left_gate   # (B, H, N, N)  fp16
    right = right_proj * right_gate

    # -------------------------------------------------------------------
    # 6️⃣  Core O(N³·H) update – batched GEMM via cuBLAS (Tensor‑Core)
    # -------------------------------------------------------------------
    B, H, N, _ = left.shape
    left_mat = left.view(B * H, N, N)                     # (B*H, N, N)
    right_mat = right.view(B * H, N, N)                   # (B*H, N, N)

    # out_mat = left_mat @ right_matᵀ   →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -------------------------------------------------------------------
    # 7️⃣  Output LayerNorm (still FP16)
    # -------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )

    # -------------------------------------------------------------------
    # 8️⃣  Per‑position output gate (1×1 conv + sigmoid)
    # -------------------------------------------------------------------
    out_gate = F.conv2d(
        x_h,
        out_gate_weight.view(hidden, dim, 1, 1).half(),
        bias=None,
    )                              # (B, H, N, N)  fp16
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous().sigmoid()  # (B, N, N, H)

    out = out * out_gate

    # -------------------------------------------------------------------
    # 9️⃣  Final projection back to original dim (1×1 conv)
    # -------------------------------------------------------------------
    # reshape to (B, H, N, N) for the conv
    out_conv_in = out.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    out_proj = F.conv2d(
        out_conv_in,
        to_out_weight.view(dim, hidden, 1, 1).half(),
        bias=None,
    )                              # (B, C, N, N)  fp16

    out_final = out_proj.permute(0, 2, 3, 1).contiguous()  # (B, N, N, C)

    # -------------------------------------------------------------------
    # 10️⃣  Cast back to float32 (matches reference implementation)
    # -------------------------------------------------------------------
    return out_final.float()
