"""
custom_kernel: highly‑optimized forward‑only “outgoing” TriMul (AlphaFold‑3)
--------------------------------------------------------------------------

The implementation follows the reference PyTorch model but fuses several
operations to minimise memory traffic:

1. LayerNorm on the input (float32 for numeric stability).
2. One fused linear projection that simultaneously computes
   left/right projections **and** their gates (4·hidden features).
3. Optional mask multiplication – executed by a tiny Triton kernel
   that broadcasts the mask over the hidden dimension.
4. Element‑wise sigmoid gating.
5. Core O(N³·H) update realised as a batched GEMM
   ``out = left @ rightᵀ`` (torch.bmm).  The required ``permute`` /
   ``contiguous`` steps are cheap compared with the GEMM.
6. Output LayerNorm + out‑gate.
7. Final linear projection back to the original channel dimension.

All heavy arithmetic runs in FP16; only the two LayerNorms use FP32
statistics.  The routine returns a float32 tensor to match the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over H)
# -------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    x_ptr, mask_ptr,
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """x[b,i,j,h] *= mask[b,i,j] (mask broadcast over the hidden dim)."""
    b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    # tile coordinates
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load a (BLOCK_I, BLOCK_J) mask tile and broadcast it over H
    # -----------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                                 # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # -----------------------------------------------------------------
    # Loop over the hidden dimension
    # -----------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        x = x * mask_tile
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor):
    """In‑place mask multiplication with the Triton kernel above."""
    B, N, _, H = x.shape
    s_x = x.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_mul_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # x is mutated in‑place


# -------------------------------------------------------------
# Outgoing TriMul – forward‑only implementation
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: tuple (input, mask, weights, config)
            - input   torch.Tensor [B, N, N, C] (float32)
            - mask    torch.Tensor [B, N, N]   (bool/float)
            - weights dict of model parameters (float32)
            - config  dict with keys:
                * dim          (int)   – input channel C
                * hidden_dim   (int)   – hidden dimension H
                * nomask       (bool)  – True ⇒ no mask
                * layernorm_eps (float, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # C
    hidden = config["hidden_dim"]           # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # Model weights (float32 → cast to half when used)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the computation
    # -----------------------------------------------------------------
    x_h = x.half()          # [B, N, N, C]  fp16
    B, N, _, C = x_h.shape

    # -----------------------------------------------------------------
    # 3️⃣ Fused projection + gating (single matmul → 4·H outputs)
    # -----------------------------------------------------------------
    # Build a single weight matrix of shape (4*H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    # Flatten input to (B·N·N, C) and apply the fused linear
    x_flat = x_h.view(-1, C)                     # (B·N·N, C)
    combined = F.linear(x_flat, combined_weight)  # (B·N·N, 4*H)

    # Split back into the four branches
    left_proj, right_proj, left_gate, right_gate = torch.split(
        combined, hidden, dim=-1
    )  # each (B·N·N, H)

    # Reshape to the original 4‑D layout
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate = left_gate.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – in‑place multiplication (broadcast over H)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)

    # -----------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -----------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update – batched GEMM (left @ rightᵀ)
    #    left/right : (B, N, N, H)
    #    reshape to (B*H, N, N) → torch.bmm
    # -----------------------------------------------------------------
    # (B, H, N, N) contiguous layout required for bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Output LayerNorm (still in fp16) and out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to C channels
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B, N, N, C) fp16

    # -----------------------------------------------------------------
    # Return float32 to stay compatible with the reference module
    # -----------------------------------------------------------------
    return out.float()
