def batched_matmul_tiled(left, right):
    # left, right: (B*H, N, N) float16
    B_H, N, K = left.shape
    assert K == N
    out = torch.empty((B_H, N, N), dtype=torch.float16, device=left.device)
    # define strides: left stride (batch, i, k)
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    # right stride: (batch, j, k)
    stride_right_batch = right.stride(0)
    stride_right_j = right.stride(1) # actually N dimension (like column)
    stride_right_k = right.stride(2)
    # c strides similar
    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)

    BLOCK_M = 32 (or 64)
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (B_H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

        num_warps=4, # or dynamic
        # maybe enable N=0?
    )
    return out

"""
high‑performance outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣  Layer‑Norm the input pair representation (FP32 → FP16).
2️⃣  Linear projections + linear gates are fused into two GEMMs.
3️⃣  Optional mask multiplication (broadcast over the hidden channel) – done by a
    tiny Triton kernel (mandatory “use a kernel” rule).
4️⃣  Core O(N³·H) update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    This is a batched matrix multiplication (B·H independent GEMMs) implemented
    in pure Triton to avoid the extra CUDA kernel launch overhead of torch.bmm.
5️⃣  Output LayerNorm, per‑position gating and final linear projection back to
    the original channel dimension (all in FP16).  The result is returned as
    float32 to match the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# 1️⃣  Triton kernel: in‑place mask * tensor (broadcast over hidden dim)
# -----------------------------------------------------------------------
@triton.jit
def _mask_channel_kernel(
    x_ptr,                # pointer to tensor [B,N,N,H]
    mask_ptr,             # pointer to mask   [B,N,N]
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    b = tl.program_id(0)               # batch index
    pid_i = tl.program_id(1)           # tile over i
    pid_j = tl.program_id(2)           # tile over j

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # load mask (scalar) and broadcast over hidden dimension
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    h_off = 0
    while h_off < H:
        h = h_off + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # load tensor slice, apply mask, store back
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        x = x * mask
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_off += BLOCK_H


def _apply_mask_torch(x: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place x *= mask (mask broadcast over the hidden dimension) using Triton."""
    B, N, _, H = x.shape
    s_x = x.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _mask_channel_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# -----------------------------------------------------------------------
# 2️⃣  Triton kernel: batched GEMM (C = A @ Bᵀ)  – core O(N³·H) update
# -----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,            # [BATCH, M, K]   (A)
    b_ptr,            # [BATCH, N, K]   (B) – we will treat as K×N tile
    c_ptr,            # [BATCH, M, N]   (C)
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)          # which (batch * hidden) we are processing
    pid_m = tl.program_id(1)          # tile over rows (i)
    pid_n = tl.program_id(2)          # tile over cols (j)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    m_mask = offs_m < M
    n_mask = offs_n < N

    # Pointers to the start of the tiles for A and B
    a_ptrs = (
        a_ptr
        + batch * stride_a_batch
        + (offs_m[:, None] * stride_a_m)
        + (tl.arange(0, BLOCK_K)[None, :] * stride_a_k)
    )
    b_ptrs = (
        b_ptr
        + batch * stride_b_batch
        + (tl.arange(0, BLOCK_K)[:, None] * stride_b_k)   # K dimension as rows
        + (offs_n[None, :] * stride_b_n)                  # N dimension as cols
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # mask for the K‑tile (handle tail)
        k_mask = (tl.arange(0, BLOCK_K) < K - cur_k)

        a = tl.load(
            a_ptrs + cur_k * stride_a_k,
            mask=m_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs + cur_k * stride_b_k,
            mask=k_mask[:, None] & n_mask[None, :],
            other=0.0,
        )
        # A: (BLOCK_M, BLOCK_K), B: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = (
        c_ptr
        + batch * stride_c_batch
        + (offs_m[:, None] * stride_c_m)
        + (offs_n[None, :] * stride_c_n)
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=m_mask[:, None] & n_mask[None, :],
    )


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (BATCH, N, N) in FP16.
    Returns tensor of the same shape and dtype.
    """
    BATCH, M, K = a.shape
    _, N, _ = b.shape
    assert K == N, "inner dimensions must match"

    c = torch.empty((BATCH, M, N), dtype=torch.float16, device=a.device)

    # strides in element units
    s_a = a.stride()
    s_b = b.stride()
    s_c = c.stride()

    # Tuning parameters – reasonable defaults for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BATCH,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        BATCH,
        M,
        N,
        K,
        s_a[0],
        s_a[1],
        s_a[2],
        s_b[0],
        s_b[2],
        s_b[1],
        s_c[0],
        s_c[1],
        s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c

    Forward‑only outgoing TriMul.

    hidden = config["hidden_dim"]     # H

    nomask = config.get("nomask", True)

    device = input_tensor.device
    # -------------------------------------------------
    # Load weights (they are already on the correct device)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    left_gate_weight  = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"]  # (H, C)
    out_gate_weight   = weights["out_gate.weight"]    # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (H,)
    to_out_weight      = weights["to_out.weight"]       # (C, H)

    # -------------------------------------------------
    # 1️⃣ LayerNorm (float32)
    # -------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  float32

    # -------------------------------------------------
    # 2️⃣ Linear projections (FP16)
    # -------------------------------------------------
    x_h = x_norm.half()                     # (B, N, N, C)

    left_proj = F.linear(x_h, left_proj_weight.half())   # (B,N,N,H)
    right_proj = F.linear(x_h, right_proj_weight.half()) # (B,N,N,H)

    # -------------------------------------------------
    # 3️⃣ Optional mask – broadcast over hidden dim
    # -------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask is [B, N, N] – apply to both left_proj and right_proj
        _apply_mask_torch(left_proj, mask_tensor)
        _apply_mask_torch(right_proj, mask_tensor)

    # -------------------------------------------------
    # 4️⃣ Gates (FP16) – applied after mask
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight.half()))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.half())) # (B,N,N,H)

    left = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # -------------------------------------------------
    B, N, _, H = left.shape

    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, i, k)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, j, k)

    # out_b has shape (B*H, N, N) where the inner product is taken over k
    out_b = _batched_matmul(left_b, right_b)  # (B*H, N, N) FP16

    # reshape back to (B, N, N, H)
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    out = F.layer_norm(

        bias=to_out_norm_bias.half(),

    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)

    out = F.linear(out, to_out_weight.half())  # (B,N,N,C) FP16

    # -------------------------------------------------
    # Return float32 tensor to match reference implementation
    # -------------------------------------------------
    return out.float()
