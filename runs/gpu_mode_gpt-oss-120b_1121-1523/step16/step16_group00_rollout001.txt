def _tri_mul_matmul_kernel(left, right, out, BLOCK_I=64, BLOCK_J=64, BLOCK_K=64):
    B, N, _, H = left.shape # (B,N,N,H) shape
    left_perm = left.permute(0,3,1,2).contiguous() # shape (B,H,N,N)
    right_perm = right.permute(0,3,1,2).contiguous()
    B_H = B*H
    left_flat = left_perm.view(B_H, N, N)
    right_flat = right_perm.view(B_H, N, N)
    # output shape (B_H,N,N)
    out_flat = torch.empty_like(left_flat)  # same dtype half

    # compute strides
    s_l = left_flat.stride()   # (N*N, N, 1)
    s_r = right_flat.stride()
    s_o = out_flat.stride()

    grid = (B_H, (N+BLOCK_I-1)//BLOCK_I, (N+BLOCK_J-1)//BLOCK_J)
    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B_H,
        N,
        s_l[0], s_l[1], s_l[2],
        s_r[0], s_r[1], s_r[2],
        s_o[0], s_o[1], s_o[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape out back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    left_ptr, right_ptr, out_ptr,
    batch_head, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,   # we treat rows i here but will load transposed
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    bh = tl.program_id(0) # batch*head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute indices
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # pointers offset base for this bh
    left_base = left_ptr + bh * stride_l_batch
    right_base = right_ptr + bh * stride_r_batch
    out_base = out_ptr + bh * stride_o_batch

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.
    ...
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # Unpack weights, cast to half
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    device = input_tensor.device

    # LayerNorm over last dim
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    x_h = x.to(torch.float16)   # (B,N,N,C) half

    B, N, _, C = x_h.shape

    # Linear projections: fused using concatenated weights
    proj_weight = torch.cat([left_proj_weight, right_proj_weight], dim=0).to(device=device, dtype=torch.float16)  # (2*H, C)
    gate_weight = torch.cat([left_gate_weight, right_gate_weight], dim=0).to(device=device, dtype=torch.float16)  # (2*H, C)

    x_flat = x_h.view(-1, C)  # (B*N*N, C)

    proj = F.linear(x_flat, proj_weight)   # (B*N*N, 2*H)
    gate = F.linear(x_flat, gate_weight)   # (B*N*N, 2*H)

    proj = proj.view(B, N, N, 2*hidden)   # (B,N,N,2*H)
    gate = gate.view(B, N, N, 2*hidden)

    left_proj, right_proj = torch.split(proj, hidden, dim=-1)
    left_gate, right_gate = torch.split(gate, hidden, dim=-1)

    # Optional mask multiply
    if not nomask:
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)

    # Apply sigmoid gates (in half)
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # Core update via Triton GEMM
    out = _tri_mul_gemm(left, right, hidden)   # custom function to call kernel and reshape

    # Output layernorm
    out = F.layer_norm(
        out,
        (hidden,),

        bias=to_out_norm_bias.to(dtype=out.dtype, device=device),

def _tri_mul_gemm(left, right, hidden):
    """left,right shape (B,N,N,H) half; returns out shape (B,N,N,H) half"""
    B, N, _, H = left.shape
    # permute to (B,H,N,N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    batch_head = B * H
    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    out_flat = torch.empty_like(left_flat)
    # Get strides
    s_l = left_flat.stride()
    s_r = right_flat.stride()
    s_o = out_flat.stride()
    # Launch kernel
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64
    grid = (batch_head, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_head,
        N,
        s_l[0], s_l[1], s_l[2],
        s_r[0], s_r[1], s_r[2],
        s_o[0], s_o[1], s_o[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # reshape to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3 style).

The forward pass consists of:
1️⃣ LayerNorm on the input (float32 → float16 for the heavy part).
2️⃣ Two fused linear projections (proj) and two fused linear gates (gate) – all
   performed in FP16.
3️⃣ Optional in‑place masking of the projected tensors (tiny Triton kernel).
4️⃣ Element‑wise sigmoid‑gate application.
5️⃣ Core O(N³·H) update: for each (b, head) we compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton GEMM kernel (batched matrix multiplication).
6️⃣ Output LayerNorm, out‑gate sigmoid, and final linear projection back to
   the original channel dimension.
All heavy arithmetic runs in FP16 with FP32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1️⃣  In‑place mask multiplication (broadcast over hidden dimension H)
# ----------------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    x_ptr, mask_ptr,
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    x[b,i,j,h] *= mask[b,i,j]  (mask broadcast over h)
    """
    b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load mask tile (BLOCK_I x BLOCK_J) and broadcast over H
    # -----------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                               # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # -----------------------------------------------------------------
    # Loop over the hidden dimension H
    # -----------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        x = x * mask_tile
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor):
    """In‑place mask multiplication using the kernel above."""
    B, N, _, H = x.shape
    s_x = x.stride()
    s_m = mask.stride()
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64
    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_mul_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# 2️⃣  Batched GEMM kernel for the O(N³·H) TriMul update
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_head, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each (batch_head) index:
        out[i,j] = Σ_k left[i,k] * right[j,k]
    where left and right have shape (N, N) and right is accessed transposed.
    Accumulation is in FP32, final result stored as FP16.
    """
    bh = tl.program_id(0)          # batch * head index
    pid_i = tl.program_id(1)       # tile over rows (i)
    pid_j = tl.program_id(2)       # tile over cols (j)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # base pointers for this batch_head
    left_base  = left_ptr  + bh * stride_l_batch
    right_base = right_ptr + bh * stride_r_batch
    out_base   = out_ptr   + bh * stride_o_batch

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # left tile: (BLOCK_I, BLOCK_K)
        left = tl.load(
            left_base
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # right tile transposed: (BLOCK_K, BLOCK_J)
        right = tl.load(
            right_base
            + k[:, None] * stride_r_k   # k plays the role of row in the transposed view
            + j[None, :] * stride_r_i,  # j is the column
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )
        # dot over k → (BLOCK_I, BLOCK_J)
        acc += tl.cast(tl.dot(left, right), tl.float32)

    # Store result
    out = acc.to(tl.float16)
    tl.store(
        out_base
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j,
        out,
        mask=i_mask[:, None] & j_mask[None, :],
    )


def _tri_mul_gemm(
    left: torch.Tensor,
    right: torch.Tensor,
    hidden_dim: int,
) -> torch.Tensor:
    """
    Batched GEMM for the TriMul core.
    left / right: (B, N, N, H)  FP16
    returns out: (B, N, N, H)    FP16
    """
    B, N, _, H = left.shape
    # (B, H, N, N)  →  (B*H, N, N) contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_head = B * H
    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides for the contiguous (B*H, N, N) tensors
    s_l = left_flat.stride()   # (N*N, N, 1)
    s_r = right_flat.stride()
    s_o = out_flat.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        batch_head,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_head,
        N,
        s_l[0],
        s_l[1],
        s_l[2],
        s_r[0],
        s_r[1],
        s_r[2],
        s_o[0],
        s_o[1],
        s_o[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# 3️⃣  Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3) using Triton for
    mask‑multiplication and the O(N³·H) matrix update.
    Returns a float‑32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                         # input channel C
    hidden = config["hidden_dim"]               # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (float32 → keep for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣  Cast to FP16 for all subsequent heavy ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)          # (B, N, N, C)
    B, N, _, C = x_h.shape

    # ------------------------------------------------------------------
    # 3️⃣  Fused projections and gates (two linear calls each)
    # ------------------------------------------------------------------
    proj_weight = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"]], dim=0).to(
        dtype=torch.float16, device=device
    )  # (2*H, C)

    gate_weight = torch.cat([weights["left_gate.weight"], weights["right_gate.weight"]], dim=0).to(
        dtype=torch.float16, device=device
    )  # (2*H, C)

    x_flat = x_h.view(-1, C)                     # (B·N·N, C)

    proj = F.linear(x_flat, proj_weight)          # (B·N·N, 2*H)
    gate = F.linear(x_flat, gate_weight)          # (B·N·N, 2*H)

    proj = proj.view(B, N, N, 2 * hidden)          # (B, N, N, 2*H)
    gate = gate.view(B, N, N, 2 * hidden)

    left_proj, right_proj = torch.split(proj, hidden, dim=-1)   # each (B,N,N,H)
    left_gate, right_gate = torch.split(gate, hidden, dim=-1)   # each (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣  Optional mask (in‑place, broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)

    # ------------------------------------------------------------------
    # 5️⃣  Apply sigmoid gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 6️⃣  Core O(N³·H) update via Triton GEMM
    # ------------------------------------------------------------------
    out = _tri_mul_gemm(left, right, hidden)   # (B, N, N, H)  FP16

    # ------------------------------------------------------------------
    # 7️⃣  Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out.dtype, device=device),
        eps=eps,
    )

    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(dtype=torch.float16, device=device))
    )  # (B, N, N, H)  FP16

    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣  Final projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(dtype=torch.float16, device=device))

    # ------------------------------------------------------------------
    # 9️⃣  Return float32 (matches reference implementation)
    # ------------------------------------------------------------------
    return out.float()
