"""
High‑performance outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the pair tensor (float32 → stable).
2️⃣ Fuse the four linear branches (left_proj, left_gate, right_proj, right_gate)
   into a single fp16 GEMM over the flattened (B·N·N, C) matrix.
3️⃣ (Optional) Apply the pair mask in‑place with a tiny Triton kernel
   (mask is broadcast over the hidden dimension H).
4️⃣ Apply sigmoid gates (still fp16) and element‑wise multiply the gated
   projections.
5️⃣ Core O(N³·H) update: for each hidden channel h we need
        out_h = left_h @ right_hᵀ .
   This is realised as a *batched* fp16 GEMM by collapsing the batch and
   hidden dimensions (B·H, N, N) – far faster than a generic einsum.
6️⃣ Output LayerNorm, out‑gate, final linear projection back to C.
7️⃣ Cast the result to float32 (as in the reference implementation).

All heavy work is done in fp16 (tensor‑core friendly).  The Triton kernel
is only required for the mask‑multiplication step, satisfying the problem
constraint while keeping overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication on a 4‑D tensor
# -------------------------------------------------------------------------
@triton.jit
def _mask_4d_kernel(
    x_ptr,                # fp16/fp32 *
    mask_ptr,             # same dtype as x *
    B, N, H,              # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """x[b,i,j,h] *= mask[b,i,j]   (mask broadcast over hidden dimension H)"""
    b = tl.program_id(0)            # batch index
    pid_i = tl.program_id(1)        # tile over i
    pid_j = tl.program_id(2)        # tile over j

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load mask (scalar per (i,j))
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                     # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension H
    off_h = 0
    while off_h < H:
        h = off_h + tl.arange(0, BLOCK_H)
        h_ok = h < H

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        x = x * mask
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        off_h += BLOCK_H


def _apply_mask_triton_4d(tensor: torch.Tensor, mask: torch.Tensor) -> None:

    """

    B, N, _, H = tensor.shape
    mask = mask.to(dtype=tensor.dtype)

    s_x = tensor.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

        s_m[2],

        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,

    )

# -------------------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Model weights (stored as float32; convert to fp16 when used)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]          # (C,)
    norm_bias = weights["norm.bias"]              # (C,)

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm the input (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four linear branches into one fp16 GEMM
    # ------------------------------------------------------------------
    x_h = x.half()                               # fp16
    B, N, _, _ = x_h.shape

    # Concatenate weights: [left_proj; left_gate; right_proj; right_gate] → (4H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    # Linear over flattened pair dimension
    x_flat = x_h.view(-1, dim)                    # (B·N·N, C)
    combined = F.linear(x_flat, combined_weight)  # (B·N·N, 4·H)
    combined = combined.view(B, N, N, 4 * hidden_dim)   # (B, N, N, 4·H)

    # Split into the four tensors (views, no copy)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over hidden dim) – Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # Mask is float (0/1) – convert once to fp16
        mask_half = mask_tensor.to(dtype=torch.float16)
        _apply_mask_triton_4d(left_proj, mask_half)
        _apply_mask_triton_4d(right_proj, mask_half)

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate_sig = left_gate.sigmoid()      # (B, N, N, H)
    right_gate_sig = right_gate.sigmoid()    # (B, N, N, H)

    # Apply gates
    left = left_proj * left_gate_sig          # (B, N, N, H)
    right = right_proj * right_gate_sig       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM across hidden dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Equivalent to: for each h, out_h = left_h @ right_hᵀ
    # ------------------------------------------------------------------
    # Bring hidden dim to batch dimension for a single torch.bmm call
    # left: (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matmul (fp16 uses Tensor Cores)
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))   # (B·H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final projection (stay fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # (B, N, N, H) fp16

    # Out‑gate computed from the *unmasked* normalized input x_h
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B, N, N, H) fp16
    out = out * out_gate                                            # (B, N, N, H)

    # Final linear back to original channel dimension C
    out = F.linear(out, to_out_weight.half())                       # (B, N, N, C) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (as in the reference implementation)
    # ------------------------------------------------------------------
    return out.float()
```
