"""
Fast outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣  LayerNorm the 4‑D input in FP32.
2️⃣  Cast to FP16 and fuse the five linear branches (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a single GEMM.
3️⃣  Apply sigmoid gates.
4️⃣  Permute the gated tensors to bring the hidden dimension to the second
    axis (shape [B, H, N, N]) – this makes the k‑dimension contiguous.
5️⃣  Core update (O(N³·H)) is executed by a custom Triton kernel that
    computes, for every (b, h, i, j),

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] *
                       mask[b,i,k] * mask[b,j,k]          (mask optional)

    The kernel is a classic tiled GEMM with FP16 inputs, FP32 accumulation
    and optional mask multiplication fused into the load.
6️⃣  Permute the result back to [B, N, N, H] and apply a hidden‑dim LayerNorm,
    the output gate and a final FP16 linear projection.
7️⃣  Return the result in FP32.

Only the cubic “TriMul” update (and the optional mask‑multiply) is
implemented in Triton; all other ops use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM with optional mask multiplication.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,                # FP16 [B, H, N, N]
    right_ptr,               # FP16 [B, H, N, N]
    out_ptr,                 # FP16 [B, H, N, N]
    mask_ptr,                # FP16 [B, N, N] (broadcast over H)
    B, N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    stride_m_b, stride_m_i, stride_m_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    MASKED: tl.constexpr               # 1 → mask is applied, 0 → mask omitted
):
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] (masked)."""
    batch_h = tl.program_id(0)       # combines batch and hidden dim
    i_tile  = tl.program_id(1)       # tile over i
    j_tile  = tl.program_id(2)       # tile over j

    b = batch_h // H                 # original batch index
    h = batch_h %  H                 # hidden index

    # Tile coordinates
    i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    i_ok = i < N
    j_ok = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_ok = k < N

        # ----- left tile (i,k) -------------------------------------------------
        left = tl.load(
            left_ptr + b * stride_l_b
                      + h * stride_l_h
                      + i[:, None] * stride_l_i
                      + k[None, :] * stride_l_k,
            mask = i_ok[:, None] & k_ok[None, :],
            other = 0.0,
        )
        # ----- right tile (j,k) ------------------------------------------------
        right = tl.load(
            right_ptr + b * stride_r_b
                       + h * stride_r_h
                       + j[:, None] * stride_r_i
                       + k[None, :] * stride_r_k,
            mask = j_ok[:, None] & k_ok[None, :],
            other = 0.0,
        )

        if MASKED:
            # mask for left (b,i,k)
            mask_l = tl.load(
                mask_ptr + b * stride_m_b
                         + i[:, None] * stride_m_i
                         + k[None, :] * stride_m_k,
                mask = i_ok[:, None] & k_ok[None, :],
                other = 1.0,
            )
            left = left * mask_l

            # mask for right (b,j,k)
            mask_r = tl.load(
                mask_ptr + b * stride_m_b
                         + j[:, None] * stride_m_i
                         + k[None, :] * stride_m_k,
                mask = j_ok[:, None] & k_ok[None, :],
                other = 1.0,
            )
            right = right * mask_r

        # dot product (FP16 → FP32 accumulation)
        acc += tl.dot(left, right)

    # Store FP16 result
    tl.store(
        out_ptr + b * stride_o_b
                 + h * stride_o_h
                 + i[:, None] * stride_o_i
                 + j[None, :] * stride_o_j,
        acc.to(tl.float16),
        mask = i_ok[:, None] & j_ok[None, :],
    )
    # no return


def _batched_trimul(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    left, right : [B, N, N, H] (FP16) – already gated.
    mask        : [B, N, N] (FP16) or None.
    Returns out : [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    device = left.device

    # Permute to (B, H, N, N) – makes the k‑dimension contiguous.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (b, h, j, k)

    out = torch.empty_like(left_perm)

    # Mask handling
    if mask is None:
        mask_tensor = torch.empty((1, 1, 1), dtype=torch.float16, device=device)  # dummy
        masked = 0
    else:
        mask_tensor = mask.to(dtype=torch.float16, device=device).contiguous()
        masked = 1

    # Block sizes – tuned for H100, hidden_dim ≤ 384
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    s_l = left_perm.stride()
    s_r = right_perm.stride()
    s_o = out.stride()
    s_m = mask_tensor.stride()

    _trimul_kernel[grid](
        left_perm,
        right_perm,
        out,
        mask_tensor,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        MASKED=masked,
        num_warps=8,
    )
    # Convert back to the original layout [B, N, N, H]
    return out.permute(0, 2, 3, 1).contiguous()


# ----------------------------------------------------------------------
# Forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C] (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 + fuse the five linear branches
    # ------------------------------------------------------------------
    x_half = x.half()                                 # fp16
    B, N, _, _ = x_half.shape
    x_flat = x_half.view(-1, dim)                      # (B·N·N, C)

    # concatenate the five weight matrices (no bias)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(dtype=torch.float16)                         # (5·H, C)

    # single GEMM → (B·N·N, 5·H)
    combined = F.linear(x_flat, combined_weight)      # fp16
    combined = combined.view(B, N, N, 5 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()
    out_gate = out_gate.sigmoid()                     # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – custom Triton GEMM (mask fused if needed)
    # ------------------------------------------------------------------
    mask = None if nomask else mask_tensor
    out = _batched_trimul(left, right, mask, hidden_dim)   # fp16, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm + output‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].half(),   # (dim, H)
        bias=None,
    )   # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Return FP32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
