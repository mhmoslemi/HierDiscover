"""
High‑performance forward‑only “outgoing” TriMul (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input in FP32 (numerical stability).
2️⃣  Cast to FP16 and fuse the five linear branches (proj‑, gate‑weights) into a
    single GEMM (`torch.nn.functional.linear`).  The result is split back into
    *left_proj*, *left_gate*, *right_proj*, *right_gate* and *out_gate*.
3️⃣  Apply the sigmoid gates.
4️⃣  If a mask is present, an in‑place Triton kernel multiplies both *left* and
    *right* by the mask (broadcast over the hidden dimension).
5️⃣  Core update (the O(N³·H) step):
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
    is realised as a batched GEMM on tensors reshaped to (B·H, N, N).
6️⃣  Layer‑Norm over the hidden dimension, scaling by *out_gate*.
7️⃣  Final FP16 linear projection back to the original channel dimension.
8️⃣  Return the result in FP32.

Only the mask‑multiplication is performed with a custom Triton kernel, satisfying
the “must‑use‑kernel” rule while keeping the heavy O(N³) work in the highly‑
optimised cuBLAS batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – in‑place mask multiplication for left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_lr_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Multiply left[b,i,j,h] and right[b,i,j,h] by mask[b,i,j] (in‑place).

    Arguments
    ---------
    left_ptr, right_ptr : pointers to FP16 tensors [B,N,N,H]
    mask_ptr            : pointer to FP16 mask   [B,N,N]
    B, N, H             : runtime sizes
    stride_…            : strides (in elements) of the tensors
    BLOCK_…             : compile‑time tile sizes
    """
    b = tl.program_id(0)                    # batch dimension
    ti = tl.program_id(1)                   # i‑tile
    tj = tl.program_id(2)                   # j‑tile

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    i = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    j = tj * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # --------------------------------------------------------------
    # Load mask tile (shape BLOCK_I x BLOCK_J) and broadcast over H
    # --------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                                 # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]           # (BLOCK_I, BLOCK_J, 1)

    # --------------------------------------------------------------
    # Loop over hidden dimension in chunks of BLOCK_H
    # --------------------------------------------------------------
    num_h = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h):
        cur_h = hi * BLOCK_H + tl.arange(0, BLOCK_H)
        h_ok = cur_h < H

        # ----- left -------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + cur_h[None, None, :] * stride_l_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left = left * mask
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + cur_h[None, None, :] * stride_l_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        # ----- right ------------------------------------------------
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + cur_h[None, None, :] * stride_r_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right = right * mask
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + cur_h[None, None, :] * stride_r_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # --------------------------------------------------------------
    dim = config["dim"]                     # input channel dim C
    hidden_dim = config["hidden_dim"]       # projection dim H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  (fp32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 and fuse the five linear branches
    # --------------------------------------------------------------
    x_h = x.half()                                    # fp16
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                         # (B·N·N, C)

    # Concatenate the five weight matrices (no bias).  Shape: (5·H, C)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],    # (H, C)
            weights["left_gate.weight"],    # (H, C)
            weights["right_proj.weight"],   # (H, C)
            weights["right_gate.weight"],   # (H, C)
            weights["out_gate.weight"],     # (H, C)
        ],
        dim=0,
    ).to(dtype=torch.float16)

    # Single GEMM → (B·N·N, 5·H)
    combined = F.linear(x_flat, combined_weight)      # fp16
    combined = combined.view(B, N, N, 5 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()
    out_gate = out_gate.sigmoid()                     # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – in‑place Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        if mask_tensor is None:
            raise ValueError("mask_tensor must be provided when nomask=False")
        _apply_mask_lr_twin(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # left: (B, N, N, H) -> (B, H, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right: permute to (B, H, N, N) with k‑j ordering (transpose last two dims)
    right_b = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N) → (B·H, N, N)
    out_mat = torch.bmm(left_b, right_b)               # fp16

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final projection back to the original channel dimension
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].half(),   # (dim, H)
        bias=None,
    )   # (B, N, N, dim) fp16

    # --------------------------------------------------------------
    # 8️⃣ Return FP32 (API contract)
    # --------------------------------------------------------------
    return out.float()
```
