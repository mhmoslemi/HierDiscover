"""
High‑performance outgoing TriMul (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D pair tensor (float32 → stable).
2️⃣ Fuse the four linear branches (left‑proj, left‑gate, right‑proj, right‑gate)
   into a single half‑precision GEMM.
3️⃣ If a mask is present, multiply *only* the projected tensors (left‑proj &
   right‑proj) by the mask (broadcast over the hidden dimension) using a tiny
   Triton kernel – this respects the original “gate‑then‑mask” order.
4️⃣ Apply sigmoid gates and obtain the gated left/right tensors (half‑precision).
5️⃣ Core O(N³·H) update: treat each hidden slice as an independent matrix
   multiplication and perform all of them with a single batched half‑precision
   cuBLAS gemm (torch.bmm).
6️⃣ Output LayerNorm (half‑precision), element‑wise out‑gate and final
   projection back to the original channel dimension.
7️⃣ Return the result in float32 (as in the reference implementation).

Only the mask‑multiplication step is implemented in Triton; the rest of the
forward pass uses fused PyTorch kernels (half‑precision linear layers,
batched matrix multiplication, etc.) which are already highly optimized on
H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication on a 4‑D tensor
# -------------------------------------------------------------------------
@triton.jit
def _mask_4d_kernel(
    x_ptr,               # fp16/fp32 *
    mask_ptr,            # same dtype as x *
    B, N, H,             # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """x[b,i,j,h] *= mask[b,i,j]  (mask broadcast over hidden dimension H)"""
    b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load mask (scalar per (i,j)) and broadcast over H
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    off_h = 0
    while off_h < H:
        h = off_h + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # Load a tile of x
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # Apply mask (broadcast)
        x = x * mask
        # Store back
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        off_h += BLOCK_H

    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear branches (half‑precision)
    # ------------------------------------------------------------------
    x_h = x.half()   # (B, N, N, C) in fp16

    # Concatenate the four weight matrices: [left_proj, left_gate, right_proj, right_gate]
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)  # (4*H, C)

    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                     # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4*H)

    # Split the fused result
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask on the *projected* tensors (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; cast to half for the kernel
        mask_half = mask_tensor.to(dtype=torch.float16)
        _apply_mask_triton_4d(left_proj, mask_half)
        _apply_mask_triton_4d(right_proj, mask_half)

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and obtain gated tensors
    # ------------------------------------------------------------------
    left_gate_sig = left_gate.sigmoid()
    right_gate_sig = right_gate.sigmoid()
    left = left_proj * left_gate_sig   # (B, N, N, H) fp16
    right = right_proj * right_gate_sig

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    # Reshape for batched GEMM: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating, final projection (stay in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    # out_gate is computed from the *unmasked* normalized input (per the reference)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B, N, N, H) fp16
    out = out * out_gate

    out = F.linear(out, to_out_weight.half())   # (B, N, N, C) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (as in original implementation)
    # ------------------------------------------------------------------
    return out.float()
```
