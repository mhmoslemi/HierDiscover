"""
Highly‑optimized forward implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  

Algorithm
---------
1. Layer‑Norm the input (FP32 for numeric stability).
2. Cast to FP16 and perform a *single* fused linear projection that yields
   left‑proj, left‑gate, right‑proj and right‑gate (4× hidden_dim features).
3. Apply sigmoid to the gate tensors and fuse them into the projections.
4. (Optional) Apply the pairwise mask in‑place with a tiny Triton kernel.  The
   mask has shape [B,N,N] and is broadcast over the hidden dimension H.
5. Core O(N³) update:  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched half‑precision matrix multiplication using a
   single ``torch.matmul`` on views of shape (B, H, N, N) → avoids costly
   ``permute(...).contiguous()`` copies.
6. Output LayerNorm → output‑gate (sigmoid) → final linear projection back to
   the original channel dimension (all in FP16).
7. Cast the result to FP32 for downstream consumption.

Only the mask‑application step is written in Triton, satisfying the “use
Triton” requirement while keeping the heavy N³ work on cuBLAS which is
optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over H)
# --------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,                # fp16 *
    right_ptr,               # fp16 *
    mask_ptr,                # fp16 *
    B, N, H,                 # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Multiply left/right [B,N,N,H] by mask [B,N,N] (broadcast over H)."""
    b = tl.program_id(0)          # batch index
    i_tile = tl.program_id(1)     # i‑tile
    j_tile = tl.program_id(2)     # j‑tile

    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # Load mask (scalar) and broadcast over H
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )               # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension H in blocks
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # Load left tile
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        # Load right tile
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )

        # Apply broadcast mask
        left = left * mask
        right = right * mask

        # Store back in‑place
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """In‑place multiplication of ``left`` and ``right`` by a broadcast mask."""
    B, N, _, H = left.shape
    mask_fp16 = mask.to(dtype=left.dtype, device=left.device)

    # Strides (in elements, not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_fp16.stride()

    # Tile sizes – tuned for H100
    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 64

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_apply_kernel[grid](
        left,
        right,
        mask_fp16,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left and right are modified in‑place


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑enhanced forward pass for the *outgoing* TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Load weights (FP32 on device) and cast to FP16 where possible
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]       # (dim, hidden_dim)

    # Concatenate the four projection matrices into one large GEMM weight.
    fused_weight_fp16 = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=input_tensor.device)

    out_gate_weight_fp16    = out_gate_weight.to(torch.float16, input_tensor.device)
    to_out_norm_weight_fp16 = to_out_norm_weight.to(torch.float16, input_tensor.device)
    to_out_norm_bias_fp16   = to_out_norm_bias.to(torch.float16, input_tensor.device)
    to_out_weight_fp16      = to_out_weight.to(torch.float16, input_tensor.device)

    # ------------------------------------------------------------------
    # 3) Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 4) Fused linear projection + gating (FP16)
    # ------------------------------------------------------------------
    x_fp16 = x_norm_fp32.half()   # [B,N,N,dim] fp16

    # One big linear produces: left_proj | left_gate | right_proj | right_gate
    fused = F.linear(x_fp16, fused_weight_fp16)    # [B,N,N,4*hidden_dim] fp16
    left_proj, left_gate, right_proj, right_gate = torch.split(
        fused, hidden_dim, dim=-1
    )  # each [B,N,N,hidden_dim]

    # In‑place sigmoid on gates
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()

    # Fuse gates into the projections (in‑place)
    left = left_proj.mul_(left_gate)     # [B,N,N,hidden_dim]
    right = right_proj.mul_(right_gate)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5) Optional mask (in‑place via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 6) Core O(N³) update – batched half‑precision GEMM
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Move hidden dimension to the batch axis (viewable without copy)
    left_view  = left.permute(0, 3, 1, 2)               # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2)              # (B, H, N, N)

    # Right‑hand side needs to be transposed on the last two axes
    right_t = right_view.transpose(-2, -1)              # (B, H, N, N) with (k,j)

    # Batched matmul across B and H simultaneously
    out_view = torch.matmul(left_view, right_t)         # (B, H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_view.permute(0, 2, 3, 1)                  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm → output gate → final linear (all FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
        eps=eps,
    )  # fp16

    # output gate derived from the *normed* input (half‑precision)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight_fp16))  # (B,N,N,hidden_dim)
    out = out * out_gate

    # final projection back to original dim
    out = F.linear(out, to_out_weight_fp16)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 8) Cast back to FP32
    # ------------------------------------------------------------------
    return out.float()
