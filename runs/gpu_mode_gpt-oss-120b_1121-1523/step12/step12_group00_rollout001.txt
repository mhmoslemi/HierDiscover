def custom_kernel(data):
    """Forward‑only outgoing TriMul (AlphaFold‑3 style) with Triton mask kernel.

    The function follows the PyTorch reference but fuses the five linear
    projections (left_proj, left_gate, right_proj, right_gate, out_gate)
    into a single GEMM, applies the optional mask with a custom Triton kernel,
    performs a batched matrix multiply via ``torch.bmm`` and finishes with
    a second LayerNorm, output‑gate scaling and a final linear projection.
    The result is returned as ``float32``.
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input layer‑norm (float32 for stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B,N,N,C]  fp32

    # --------------------------------------------------------------
    # 2. Cast to fp16 and compute the five linear branches in one GEMM
    # --------------------------------------------------------------
    x_h = x.half()
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                     # (B*N*N, C)

    # pack the five weight matrices (no bias)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16)                     # (5*H, C)

    combined = F.linear(x_flat, combined_weight)   # (B*N*N, 5*H)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 3. Apply sigmoid gates
    # --------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()
    out_gate = out_gate.sigmoid()                 # (B,N,N,H)

    # --------------------------------------------------------------
    # 4. Optional mask – in‑place Triton kernel (broadcast over H)
    # --------------------------------------------------------------
    if not nomask:
        apply_mask_lr_twin(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 5. Core O(N³·H) update – batched GEMM
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_b, right_b)                      # (B*H, N, N) fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm + output‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].half())

    # --------------------------------------------------------------
    # 8. Cast to fp32 for the API contract
    # --------------------------------------------------------------
    return out.float()

        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # tensors are updated in‑place


# ----------------------------------------------------------------------
# Forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] in float32.
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]                     # input channel dim C
    hidden_dim = config["hidden_dim"]       # projection dim H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                            # [B, N, N, C]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ FP16 cast + fused projection (five linear branches)
    # ------------------------------------------------------------------
    x_h = x.half()                               # fp16
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                    # (B·N·N, C)

    # concatenate the five weight matrices (no bias)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16)                     # (5·H, C)

    # single GEMM → (B·N·N, 5·H)
    combined = F.linear(x_flat, combined_weight)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()
    out_gate = out_gate.sigmoid()                # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – in‑place Triton kernel (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_lr_twin(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM
    # ------------------------------------------------------------------
    # (B, H, N, N) → (B·H, N, N)
    left_b  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_b, right_b)                 # (B·H, N, N) fp16

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + output‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].half())   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Return FP32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
