"""
High‑performance outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D pair tensor (float32 → stable).
2️⃣ OPTIONAL mask multiplication (broadcast over the channel dimension C) performed
   by a tiny Triton kernel (required “use a kernel” rule).
3️⃣ Cast to fp16 and fuse the four linear branches (left‑proj, left‑gate,
   right‑proj, right‑gate) into a single GEMM.  Split the result and apply
   sigmoid‑gates (still fp16).
4️⃣ Core O(N³·H) update – batched matrix multiplication:
        out[b,h,i,j] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a single torch.matmul on tensors of shape (B,H,N,N) which
   lets cuBLAS/torch‑stdlib use Tensor‑Cores without any extra copies.
5️⃣ Output LayerNorm (fp16), gating with sigmoid of a linear on the normalized
   (and masked) input, final linear back to the original dimension.
   The result is returned as float32.

The kernel in step 2 is the only Triton kernel; all heavy arithmetic stays in
FP16 to exploit Tensor‑Cores on H100.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over C)
# -------------------------------------------------------------
@triton.jit
def _mask_input_kernel(
    x_ptr,                # fp16 / fp32 *
    mask_ptr,             # same dtype as x *
    B, N, C,              # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """x[b,i,j,c] *= mask[b,i,j]  (mask broadcast over channel C)"""
    b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load mask (scalar per (i,j)), cast to the same dtype as x
    # -----------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                                     # (BLOCK_I, BLOCK_J)
    # expand mask for broadcasting over C
    mask = mask[:, :, None]                # (BLOCK_I, BLOCK_J, 1)

    # -----------------------------------------------------------------
    # Loop over channel tiles
    # -----------------------------------------------------------------
    off_c = 0
    while off_c < C:
        c = off_c + tl.arange(0, BLOCK_C)
        c_ok = c < C

        # Load a tile of x
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + c[None, None, :] * stride_x_c,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
            other=0.0,
        )
        # Apply mask
        x = x * mask
        # Store back
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + c[None, None, :] * stride_x_c,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
        )
        off_c += BLOCK_C


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place multiplication ``x *= mask`` (mask broadcast over channel dim).
    The operation is performed by the Triton kernel defined above.
    """
    assert x.is_cuda and mask.is_cuda, "Both tensors must reside on CUDA"
    B, N, _, C = x.shape
    mask = mask.to(dtype=x.dtype)

    # strides
    s_x = x.stride()
    s_m = mask.stride()

    # Tile sizes – chosen to cover C up to 384 without too many iterations
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_C = 64

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_input_kernel[grid](
        x,
        mask,
        B,
        N,
        C,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    # x is modified in‑place


# -------------------------------------------------------------
# Triton‑aware forward‑only outgoing TriMul
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (stored as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Optional mask – apply before projection (same semantics)
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(x, mask_tensor)   # modifies x in‑place

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 and fuse the four linear branches
    # ------------------------------------------------------------------
    x_h = x.half()    # (fp16)

    # Build a single weight matrix of shape (4*H, C) for the fused GEMM
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)  # (4*H, C)

    B, N, _, _ = x_h.shape
    x_flat = x_h.reshape(-1, dim)                      # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)       # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4*H)

    # Split back into the four branches
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each (B, N, N, H)

    # Apply sigmoid gates
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()
    # left/right now fp16, shape (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – batched matrix multiplication via torch.matmul
    # ------------------------------------------------------------------
    # Bring head dimension next to batch to use torch.matmul's batch semantics
    left_perm = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)   (transpose k↔j)

    # Batched matmul: out_perm[b, h, i, j] = Σₖ left_perm[b, h, i, k] * right_perm[b, h, k, j]
    out_perm = torch.matmul(left_perm, right_perm)   # (B, H, N, N)

    # Restore original layout
    out = out_perm.permute(0, 2, 3, 1)   # (B, N, N, H)   fp16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, gating and final projection (stay in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    # Output gate computed from the (masked) normalized input `x_h`
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B, N, N, H) fp16
    out = out * out_gate

    # Final linear back to original dimension `dim`
    out = F.linear(out, to_out_weight.half())   # (B, N, N, dim) fp16

    # Return float32 as in the reference implementation
    return out.float()
