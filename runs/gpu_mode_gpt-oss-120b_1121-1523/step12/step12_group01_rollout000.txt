@triton.jit
def mask_proj_kernel(x_ptr, mask_ptr, B,N,H,
                     stride_x_b, stride_x_i, stride_x_j, stride_x_h,
                     stride_m_b, stride_m_i, stride_m_j,
                     BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
  b = tl.program_id(0)
  i_tile = tl.program_id(1)
  j_tile = tl.program_id(2)
  i0 = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
  j0 = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
  h = tl.arange(0, BLOCK_H)

  i_mask = i0 < N
  j_mask = j0 < N

  mask = tl.load(mask_ptr + b*stride_m_b + i0[:,None]*stride_m_i + j0[None,:]*stride_m_j,
                 mask=i_mask[:,None] & j_mask[None,:],
                 other=0.0) # shape (BLOCK_I,BLOCK_J)
  mask = mask[:, :, None] # broadcast H

  h_mask = h < H

  x = tl.load(x_ptr + b*stride_x_b + i0[:,None,None]*stride_x_i + j0[None,:,None]*stride_x_j + h[None,None,:]*stride_x_h,
               mask = i_mask[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:],
               other=0.0)

  x = x * mask

  tl.store(x_ptr + ..., x,
            mask = i_mask[:,None,None] & j_mask[None,:,None] & h_mask[None,None,:])

@triton.jit
def mask_proj_kernel(x_ptr, mask_ptr, B, N, H,
                     stride_x_b, stride_x_i, stride_x_j, stride_x_h,
                     stride_m_b, stride_m_i, stride_m_j,
                     BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    b = tl.program_id(0)
    ti = tl.program_id(1)
    tj = tl.program_id(2)

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_ok = i < N
    j_ok = j < N

    mask = tl.load(mask_ptr
        + b*stride_m_b
        + i[:, None]*stride_m_i
        + j[None, :]*stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0)

    mask = mask[:, :, None]  # broadcast H

    offset = 0
    while offset < H:
        cur_h = offset + h
        h_ok = cur_h < H

        x = tl.load(x_ptr
            + b*stride_x_b
            + i[:, None, None]*stride_x_i
            + j[None, :, None]*stride_x_j
            + cur_h[None, None, :]*stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0)

        x = x * mask

        tl.store(x_ptr
            + b*stride_x_b
            + i[:, None, None]*stride_x_i
            + j[None, :, None]*stride_x_j
            + cur_h[None, None, :]*stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :])

        offset += BLOCK_H

        x,

"""
High‑performance outgoing TriMul (AlphaFold‑3 style)

Key ideas
---------
1️⃣ Layer‑Norm + linear projections are done in fp16 (Tensor‑Core friendly).
2️⃣ Mask is applied *after* the projections (as in the reference) using a tiny
   Triton kernel that multiplies a (B,N,N,H) tensor by a (B,N,N) mask broadcast
   over the head dimension.
3️⃣ The four linear branches (left_proj, left_gate, right_proj, right_gate) are
   fused into a single GEMM (weight shape (4*H, C)).
4️⃣ Core O(N³·H) update is a batched matrix multiplication:
      out_h = left_h @ right_hᵀ   (B·H, N, N) = torch.bmm(...)
5️⃣ Output LayerNorm, gating and final projection stay in fp16; the result is
   returned as fp32.

The only mandatory Triton component is the mask‑multiplication kernel, which
gives a measurable speed‑up compared with a pure PyTorch broadcast.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication for a (B,N,N,H) tensor.
# ----------------------------------------------------------------------
@triton.jit
def _mask_proj_kernel(
    x_ptr, mask_ptr,          # pointers
    B, N, H,                  # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """x[b,i,j,h] *= mask[b,i,j]  (mask broadcast over h)"""
    b = tl.program_id(0)          # batch
    ti = tl.program_id(1)         # tile over i
    tj = tl.program_id(2)         # tile over j

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_ok = i < N
    j_ok = j < N

    # load mask (float16)
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                                   # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]            # (BLOCK_I, BLOCK_J, 1)

    off = 0
    while off < H:
        cur_h = off + h
        h_ok = cur_h < H

        # load a tile of x
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # apply mask
        x = x * mask
        # store back
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        off += BLOCK_H


def _apply_mask_proj_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place multiplication ``x *= mask`` where ``x`` has shape (B,N,N,H)
    and ``mask`` has shape (B,N,N).  The mask is broadcast over the head
    dimension H.  The operation is performed by the Triton kernel above.
    """
    assert x.is_cuda and mask.is_cuda
    B, N, _, H = x.shape
    mask = mask.to(dtype=x.dtype)   # same dtype as tensor to be masked

    s_x = x.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_proj_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # ``x`` is modified in‑place


# ----------------------------------------------------------------------
# Triton‑aware forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (stored as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑Norm in fp16 (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    x = input_tensor.half()                           # (B,N,N,C) fp16
    x = F.layer_norm(
        x,
        (dim,),
        weight=norm_weight.half(),
        bias=norm_bias.half(),
        eps=eps,
    )   # still fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating (single GEMM)
    # ------------------------------------------------------------------
    # concatenate the 4 weight matrices → (4*H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16)       # (4*H, C)

    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                 # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)   # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B,N,N,4*H)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each (B,N,N,H)  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask *only* on the projected tensors (not on gates)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape is (B,N,N); broadcast over H
        _apply_mask_proj_triton(left_proj, mask_tensor)
        _apply_mask_proj_triton(right_proj, mask_tensor)

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and element‑wise fusion
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched matrix multiplication
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_b = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_h = left_h @ right_hᵀ   (B*H, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # back to (B,N,N,H)
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate (computed from the *unmasked* normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight.half()))   # (B,N,N,H)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,dim) fp16

    # Return float32 as in the reference implementation
    return out.float()
