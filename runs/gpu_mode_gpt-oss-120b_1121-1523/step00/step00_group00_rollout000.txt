@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ak,
                        stride_bk, stride_bn,
                        stride_cm, stride_cn,
                        batch_stride_a, batch_stride_b, batch_stride_c,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                        ACC_TYPE: tl.constexpr):
   # program ids
   pid_batch = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # Compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)

   # Mask out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # Pointers offsets: each batch add offset
   a_ptrs = A_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
   b_ptrs = B_ptr + pid_batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=ACC_TYPE)

   # Loop over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None,:] < K), other=0.0)
       b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None,:], other=0.0)
       acc += tl.dot(a, b)  # a shape (BLOCK_M, BLOCK_K), b shape (BLOCK_K, BLOCK_N)
       # Advance pointers for next block
       a_ptrs += BLOCK_K * stride_ak
       b_ptrs += BLOCK_K * stride_bk

   # Write result
   c_ptrs = C_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
   tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

    # A shape (batch, M, K), B shape (batch, K, N)
    # Guarantee contiguous
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert Kb == K

    # Determine block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid
    num_blocks_M = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_N = (N + BLOCK_N - 1) // BLOCK_N

    grid = (batch, num_blocks_M, num_blocks_N)

    # Compute strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()

def triton_matmul(A, B):
    # Expect A shape (batch, M, K) dtype float16, B shape (batch, K, N) dtype float16
    # Output float32.

    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # Create output
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid dimensions
    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Strides
    a_batch_stride, a_m_stride, a_k_stride = A.stride()
    b_batch_stride, b_k_stride, b_n_stride = B.stride()
    c_batch_stride, c_m_stride, c_n_stride = C.stride()

    batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        a_m_stride, a_k_stride,
        b_k_stride, b_n_stride,
        c_m_stride, c_n_stride,
        a_batch_stride, b_batch_stride, c_batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

@triton.jit
def batched_gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_abatch, stride_bbatch, stride_cbatch,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_abatch
    b_batch_ptr = b_ptr + pid_batch * stride_bbatch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + (k_offset + offs_k)[None, :] * stride_ak
        b_ptrs = b_batch_ptr + (k_offset + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Masks
        mask_a = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)
        acc += tl.dot(a, b)

    # Write back results
    c_ptrs = c_ptr + pid_batch * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    # get config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = config.get("layernorm_eps", 1e-5)

    # load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Cast to half for kernel compute
    left = left.to(torch.float16) * left_gate.to(torch.float16)
    right = right.to(torch.float16) * right_gate.to(torch.float16)

    # Permute to [B, hidden_dim, N, N] and flatten batch*hidden_dim
    B, N, _, _ = left.shape  # note _ = hidden_dim placeholder; final shape after cast? Actually shapes: left shape [B,N,N,hidden_dim]; after permute we get shape [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right's last two dims for matmul
    right_T = right_perm.transpose(-2, -1).contiguous()

    # Flatten batch*hidden_dim
    A = left_perm.view(B * hidden_dim, N, N)
    Bmat = right_T.view(B * hidden_dim, N, N)

    # Core TriMul via Triton
    out_mat = triton_matmul(A, Bmat)  # shape (B*hidden_dim, N, N), dtype float32

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear to original dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
Triton implementation of AlphaFold‑3 “outgoing” TriMul.

The forward pass consists of:
  1. LayerNorm over the input channel dimension.
  2. Linear projections (left/right) and three gating projections.
  3. Optional mask application.
  4. Element‑wise gating of the projected tensors.
  5. Core O(N³) operation:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]      (i.e. L @ Rᵀ per channel)
     This is realised as a batched GEMM (B·H matrices of shape N×N) in a
     Triton kernel for maximum throughput.
  6. LayerNorm over the hidden dimension, multiplication by the output gate,
     and a final linear projection back to the original channel size.

Only the heavy matrix‑multiply step is executed on the GPU via Triton; all
other algebraic operations are performed with standard PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where each batch corresponds to one
# hidden channel of the TriMul operation.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (M = N = seq_len, K = seq_len)
    stride_am, stride_ak,         # strides for A (batch, M, K)
    stride_bk, stride_bn,         # strides for B (batch, K, N) – B already transposed
    stride_cm, stride_cn,         # strides for C (batch, M, N)
    stride_abatch, stride_bbatch, stride_cbatch,  # strides for the batch dimension
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # which (batch * hidden) matrix
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for the current batch
    a_batch = a_ptr + pid_batch * stride_abatch
    b_batch = b_ptr + pid_batch * stride_bbatch

    # Accumulator in FP32 (regardless of input precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for k in range(num_k):
        k_offset = k * BLOCK_K
        # Pointers to the current K‑tile
        a_tile_ptr = a_batch + offs_m[:, None] * stride_am + (k_offset + offs_k)[None, :] * stride_ak
        b_tile_ptr = b_batch + (k_offset + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_tile_ptr, mask=mask_a, other=0.0)
        b = tl.load(b_tile_ptr, mask=mask_b, other=0.0)

        # FP16 inputs → FP32 accumulation
        acc += tl.dot(a, b)

    # Write back the result
    c_batch = c_ptr + pid_batch * stride_cbatch
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, acc, mask=mask_c)


def triton_batched_matmul(A: torch.Tensor, B_T: torch.Tensor) -> torch.Tensor:
    """
    Computes C = A @ B_T for a batch of matrices using the Triton kernel above.
    A : (B, M, K)   – torch.float16
    B_T: (B, K, N) – torch.float16 (already transposed, i.e. original Bᵀ)
    Returns C : (B, M, N) – torch.float32
    """
    assert A.is_contiguous() and B_T.is_contiguous()
    B, M, K = A.shape
    _, K2, N = B_T.shape
    assert K == K2

    # Output tensor (FP32 for numerical stability)
    C = torch.empty((B, M, N), dtype=torch.float32, device=A.device)

    # Block sizes – tuned for H100 (shared memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch, num_tiles_M, num_tiles_N)
    grid = (B,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Strides (in elements, not bytes)
    stride_am, stride_ak = A.stride()[1], A.stride()[2]
    stride_bk, stride_bn = B_T.stride()[1], B_T.stride()[2]
    stride_cm, stride_cn = C.stride()[1], C.stride()[2]

    stride_abatch, stride_bbatch, stride_cbatch = A.stride()[0], B_T.stride()[0], C.stride()[0]

    batched_gemm_kernel[grid](
        A, B_T, C,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_abatch, stride_bbatch, stride_cbatch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the “outgoing” TriMul operator.
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns a torch.Tensor of shape [bs, seq_len, seq_len, dim].
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (left/right) – no bias
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask is float (0 / 1); broadcast across hidden_dim
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # later used after out_norm

    # Apply gates (cast to FP16 for the kernel)
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Core TriMul (batched GEMM L @ Rᵀ) using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim

    # Rearrange to (batch, hidden_dim, N, N) and flatten batch*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Transpose last two dims of right for Rᵀ
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    # Merge batch and channel dimensions for the kernel
    A = left_perm.view(B * hidden_dim, N, N)      # (B*H, N, N), FP16
    B_mat = right_T.view(B * hidden_dim, N, N)   # (B*H, N, N), FP16

    # Triton batched GEMM → (B*H, N, N) FP32
    out_mat = triton_batched_matmul(A, B_mat)

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # final projection back to `dim`

    return out
