"""
custom_kernel
==============

Implements the **outgoing** TriMul operation (AlphaFold‑3) using a mixed
PyTorch‑Triton pipeline.

Algorithm
---------
1. Layer‑norm the input tensor (`norm`).
2. Linear projections → `left`, `right`.
3. (Optional) apply the pairwise mask.
4. Compute three sigmoid gates from the normalized input.
5. Apply the left/right gates to the projected tensors.
6. **Heavy part:**  Batched matrix multiplication
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a GEMM for every hidden channel `h` and batch `b`.  The
   multiplication is performed in a Triton kernel in FP16 with FP32
   accumulation.
7. Reshape the result back to [B, N, N, H] (H = hidden_dim) and cast to
   fp32.
8. Layer‑norm over the hidden dimension (`to_out_norm`), apply the
   output gate and a final linear projection back to `dim`.

The kernel works for any batch size (1‑2), sequence length up to 1024
and hidden dimension 128 (as used in the tests).  It runs on H100 GPUs
with triton‑3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,            # *mut fp16
    b_ptr,            # *mut fp16
    c_ptr,            # *mut fp16
    M, N, K,          # problem size (int32)
    stride_a_batch, stride_am, stride_ak,   # a: [batch, M, K]
    stride_b_batch, stride_bk, stride_bn,   # b: [batch, K, N]
    stride_c_batch, stride_cm, stride_cn,   # c: [batch, M, N]
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A: [batch,M,K] , B: [batch,K,N])"""

    pid_batch = tl.program_id(0)          # which batch * hidden channel
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    # Compute offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of A and B
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + (offs_m[:, None] * stride_am)
            + (offs_k[None, :] * stride_ak)
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + (offs_k[:, None] * stride_bk)
            + (offs_n[None, :] * stride_bn)
        )

        # Load tiles, zero‑pad out‑of‑bounds elements
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate (FP16 * FP16 → FP32)
        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn)
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
        - mask         : torch.Tensor   [B, N, N]   (bool or float)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – hidden dimension used in all internal projections
            * "nomask"     – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # extract weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool – cast to same dtype as projections
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy reduction – batched GEMM with Triton
    # ------------------------------------------------------------------
    # Cast to FP16 before the kernel for speed
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Rearrange tensors so that each hidden channel becomes a separate batch
    # left:  [B, N, N, H] -> [B, H, N, N]  (row‑major i,k)
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    # right: [B, N, N, H] -> [B, H, K, N] where K==N (k as row, j as col)
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    # Collapse the (B, H) dimensions → a single batch dimension for the kernel
    left_mat = left_perm.view(batch_h, N, N)   # [batch_h, M=N, K=N]
    right_mat = right_perm.view(batch_h, N, N) # [batch_h, K=N, N]

    # Allocate output tensor (FP16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides (in elements)
    stride_a_batch, stride_am, stride_ak = left_mat.stride()
    stride_b_batch, stride_bk, stride_bn = right_mat.stride()
    stride_c_batch, stride_cm, stride_cn = out_mat.stride()

    # Triton block sizes – tuned for H100 FP16 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch_h, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape kernel output back to [B, N, N, H] and convert to FP32
    # ------------------------------------------------------------------
    out = (
        out_mat.view(B, H, N, N)            # [B, H, N, N]
        .permute(0, 2, 3, 1)                # [B, N, N, H]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7) Output layer‑norm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, N, N, hidden_dim]

    out = out * out_gate                       # gating after norm
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return out
