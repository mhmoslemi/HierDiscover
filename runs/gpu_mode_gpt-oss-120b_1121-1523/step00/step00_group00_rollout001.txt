@triton.jit
def layernorm_kernel(
    input_ptr, output_ptr,
    M, D,
    stride_i_m, stride_i_d,
    stride_o_m, stride_o_d,
    epsilon: tl.constexpr,
    weight_ptr, bias_ptr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    # each program covers a block of M in rows and D in columns.
    # Determine row and column offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)

    mask_m = offs_m < M
    mask_d = offs_d < D

    # load input (M,D)
    input = tl.load(input_ptr + offs_m[:, None] * stride_i_m + offs_d[None, :] * stride_i_d,
                    mask=mask_m[:, None] & mask_d[None, :], other=0.0)

    # compute mean over D dimension
    mean = tl.mean(input, axis=1)  # note tl.mean not available; we can do sum then divide
    sum_d = tl.sum(input, axis=1)
    mean = sum_d / D

    # compute variance: average of (x-mean)^2
    diff = input - mean[:, None]
    var = tl.sum(diff * diff, axis=1) / D

    # normalize
    inv_std = tl.rsqrt(var + epsilon)
    normalized = diff * inv_std[:, None]

    if weight_ptr != 0:
        weight = tl.load(weight_ptr + offs_d * stride_w) # weight stride unknown, assume contiguous
    else:
        weight = 1.0

    if bias_ptr != 0:
        bias = tl.load(bias_ptr + offs_d * stride_b)
    else:
        bias = 0.0

    out = normalized * weight[None, :] + bias[None, :]
    tl.store(output_ptr + offs_m[:, None] * stride_o_m + offs_d[None, :] * stride_o_d,
             out, mask=mask_m[:, None] & mask_d[None, :])

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # Get program IDs.
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets for i and j
    i_idx = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_idx = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for bounds
    mask_i = i_idx < N
    mask_j = j_idx < N

    # Loop over hidden dimension in blocks
    for h_start in range(0, H, BLOCK_H):
        h_idx = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_idx < H

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # Loop over k dimension
        for k_start in range(0, N, BLOCK_K):
            k_idx = k_start + tl.arange(0, BLOCK_K)
            mask_k = k_idx < N

            # Compute pointers for left and right
            # left_ptr shape (B,N,N,H)
            left_ptrs = left_ptr + pid_b * stride_l_b \
                                      + i_idx[:, None, None] * stride_l_i \
                                      + k_idx[None, :, None] * stride_l_k \
                                      + h_idx[None, None, :] * stride_l_h

            right_ptrs = right_ptr + pid_b * stride_r_b \
                                        + j_idx[:, None, None] * stride_r_j \
                                        + k_idx[None, :, None] * stride_r_k \
                                        + h_idx[None, None, :] * stride_r_h

            # load with mask
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                           other=0.0)
            right = tl.load(right_ptrs,
                            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                            other=0.0)

            # For each h subblock compute dot and accumulate
            # We'll loop over h dimension within block
            for hh in range(0, BLOCK_H):
                # Only compute if within bound
                if_mask = mask_h[hh]
                # Continue? Can't use continue; but we can compute anyway; off-border values are zero due to mask.

                              + i_idx[:, None, None] * stride_o_i \

                              + h_idx[None, None, :] * stride_o_h

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract dims
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; seqlen = config.get("seqlen", input_tensor.shape[1]); bs = input_tensor.shape[0]
    # get weight tensors
    # note they are torch tensors.

    # Normalization for input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute layer norm over last dim
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # left_proj and right_proj
    left_w = weights["left_proj.weight"]   # shape [hidden_dim, dim]
    right_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_w)    # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_w)

    # mask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)  # [B,N,N,1] broadcast
    else:
        mask_unsq = torch.ones_like(left[..., :1])  # no mask case

    # left_gate
    left_gate_w = weights["left_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate_w = weights["right_gate.weight"]
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))

    # apply mask and gates
    left = left * mask_unsq * left_gate
    right = right * mask_unsq * right_gate

    # Ensure contiguous and cast to float16 for speed maybe
    # We'll use float16 for kernel and later cast back to float32
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # Allocate out buffer
    out = torch.empty((bs, seqlen, seqlen, hidden_dim), dtype=torch.float16, device=left.device)

    # Compute strides
    # left stride: [B, N, N, H]
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Call kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8  # maybe 8

    grid = (bs, (seqlen + BLOCK_M - 1)//BLOCK_M, (seqlen + BLOCK_N - 1)//BLOCK_N)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        bs, seqlen, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # out now is float16; cast back to float32
    out = out.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight,
                                         bias=to_out_norm_bias)

    # out_gate
    out_gate_w = weights["out_gate.weight"]
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))
    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_w)  # final shape [B,N,N,dim]

    return output

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bound mask
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over hidden dim
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(num_h_blocks):
        offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # Loop over k dimension
        num_k_blocks = tl.cdiv(N, BLOCK_K)
        for kb in range(num_k_blocks):
            offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Compute pointers
            # left: B, i, k, h

                                 + offs_i[:, None, None] * stride_l_i \
                                 + offs_k[None, :, None] * stride_l_k \
                                 + offs_h[None, None, :] * stride_l_h

            # right: B, j, k, h

                                   + offs_j[:, None, None] * stride_r_j \

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of blocks along hidden dim
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(num_h_blocks):
        offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for the current hidden block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # loop over the reduction dimension k
        num_k_blocks = tl.cdiv(N, BLOCK_K)
        for kb in range(num_k_blocks):
            offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # pointers
            left_ptrs = left_ptr + pid_b * stride_l_b \
                                   + offs_i[:, None, None] * stride_l_i \
                                   + offs_k[None, :, None] * stride_l_k \
                                   + offs_h[None, None, :] * stride_l_h

            right_ptrs = right_ptr + pid_b * stride_r_b \
                                     + offs_j[:, None, None] * stride_r_j \
                                     + offs_k[None, :, None] * stride_r_k \
                                     + offs_h[None, None, :] * stride_r_h

            # load tiles (float16 -> cast to fp32 for accumulation)
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],

                            other=0.0,
                            dtype=tl.float16).to(tl.float32)

            for hh in range(0, BLOCK_H):

def custom_kernel(data):
    """TriMul (outgoing) forward pass.

    This implementation follows the PyTorch reference but fuses the
    N‑dimensional contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
    into a custom Triton kernel. All other linear / LayerNorm / gating
    operations are performed in PyTorch for simplicity.
    """
    import torch
    import triton

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # batch and seq_len
    B, N, _, _ = input_tensor.shape

    # ----- LayerNorm on input -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ----- Linear projections -----
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ----- Gating tensors -----
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ----- Apply mask and gates -----
    if mask is None:
        mask = torch.ones_like(left[..., :1])
    mask = mask.unsqueeze(-1)  # [B,N,N,1]

    left = left * mask * left_gate
    right = right * mask * right_gate

    # ----- Cast to half for the Triton kernel (memory‑friendly) -----
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)

    # ----- Prepare stride arguments for the kernel -----
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ----- Kernel launch configuration -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # 128 hidden -> 16 blocks of 8

    grid = (B,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # back to fp32
    out = out.to(torch.float32)

    # ----- LayerNorm after contraction -----
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----- Gating and final linear -------
    out = out * out_gate
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_w)   # shape (B,N,N,dim)

    return output

"""
TriMul (outgoing) forward pass.

The implementation mirrors the reference PyTorch module but fuses the costly
N‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

into a custom Triton kernel. All other operations (LayerNorm, linear
projections, sigmoid gates) are performed with PyTorch for simplicity.

The kernel works on blocks of (i, j) and a block of the hidden dimension.
Inside each block we loop over the reduction dimension `k`, load tiles of
`left` and `right` (stored as FP16 to reduce memory traffic) and accumulate
the dot‑product for each hidden channel using `tl.dot`.  Accumulation is
done in FP32 and the final result is stored back as FP16 before the
following LayerNorm and output projection.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                # float16 *
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       using block‑wise tiling.
    """
    pid_b = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # --------------------------------------------------------------
    # Offsets for the (i,j) tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Loop over hidden‑dimension blocks
    # --------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(num_h_blocks):
        offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for this hidden block (M,N,Hb) in FP32
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # ----------------------------------------------------------
        # Reduction over k
        # ----------------------------------------------------------
        num_k_blocks = tl.cdiv(N, BLOCK_K)
        for kb in range(num_k_blocks):
            offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # pointers to the current tiles
            left_ptrs = (
                left_ptr
                + pid_b * stride_l_b
                + offs_i[:, None, None] * stride_l_i
                + offs_k[None, :, None] * stride_l_k
                + offs_h[None, None, :] * stride_l_h
            )
            right_ptrs = (
                right_ptr
                + pid_b * stride_r_b
                + offs_j[:, None, None] * stride_r_j
                + offs_k[None, :, None] * stride_r_k
                + offs_h[None, None, :] * stride_r_h
            )

            # load tiles (FP16) and up‑cast to FP32 for accumulation
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None]
                & mask_k[None, :, None]
                & mask_h[None, None, :],
                other=0.0,
                dtype=tl.float16,
            ).to(tl.float32)

            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None]
                & mask_k[None, :, None]
                & mask_h[None, None, :],
                other=0.0,
                dtype=tl.float16,
            ).to(tl.float32)

            # ------------------------------------------------------
            # Compute dot product for each hidden channel in the block
            # ------------------------------------------------------
            # Unroll over the hidden‑channel dimension (compile‑time constant)
            for hh in range(0, BLOCK_H):
                # left[:, :, hh] : (M, K)   right[:, :, hh] : (N, K)
                l = left[:, :, hh]
                r = right[:, :, hh]
                # (M,K) @ (N,K)^T -> (M,N)
                acc[:, :, hh] += tl.dot(l, r, trans_b=True)

        # ----------------------------------------------------------
        # Write back the computed block
        # ----------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_o_b
            + offs_i[:, None, None] * stride_o_i
            + offs_j[None, :, None] * stride_o_j
            + offs_h[None, None, :] * stride_o_h
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None]
            & mask_j[None, :, None]
            & mask_h[None, None, :],
        )


def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool). May be
                    all‑ones when ``nomask`` is True.
            - weights: dict with the following keys (all tensors are on CUDA):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight"
                * "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict containing at least "dim" and "hidden_dim".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gating tensors (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 4) Apply mask and gates
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]

    left = left * mask * left_gate
    right = right * mask * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the triton kernel (memory‑friendly)
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # allocate output buffer for the contraction
    out = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float16, device=left.device
    )

    # ------------------------------------------------------------------
    # 6) Triton kernel launch (block‑wise matmul over the reduction dim)
    # ------------------------------------------------------------------
    # strides in the contiguous layout [B,N,N,hidden]
    stride_l = left.stride()   # (B,N,N,hidden)
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # 128 → 16 blocks, 384 → 48 blocks

    grid = (
        B,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # back to FP32 for the remaining FP32 ops
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
