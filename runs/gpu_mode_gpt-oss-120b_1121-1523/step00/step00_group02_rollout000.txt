@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_il, stride_kl, stride_dl,
    stride_ir, stride_kr, stride_dr,
    stride_ol, stride_oj, stride_dl_out,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    d = pid_d

    # if start out of range then exit
    if i_start >= N or j_start >= N or d >= hidden_dim:
        return

    # masks for i and j dims
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)
    mask_i = i_range < N
    mask_j = j_range < N

    # Out pointer base for this d
    out_ptr_base = out_ptr + d * stride_dl_out

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over K
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left block (I,K)
        left_ptrs = left_ptr + i_range[:, None] * stride_il + k_range[None, :] * stride_kl + d * stride_dl
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed (K,J)
        # right shape: (j,k) in memory; we want (k,j)
        right_ptrs = right_ptr + k_range[:, None] * stride_kr + j_range[None, :] * stride_ir + d * stride_dr
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute GEMM: left (I,K) @ right (K,J)
        acc += tl.dot(left, right)   # yields (I,J)
    # Store result to out
    out_ptrs = out_ptr_base + i_range[:, None] * stride_ol + j_range[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16) if tl.dtype == tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_i, stride_k, stride_d,
    # same for right (maybe separate but we can reuse)
    out_stride_i, out_stride_j, out_stride_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i = pid_i * BLOCK_I
    j = pid_j * BLOCK_J
    d = pid_d

    if i >= N or j >= N or d >= hidden_dim:
        return

    # index ranges
    i_range = i + tl.arange(0, BLOCK_I)
    j_range = j + tl.arange(0, BLOCK_J)
    k_range_base = tl.arange(0, BLOCK_K)

    mask_i = i_range < N
    mask_j = j_range < N

    # out base pointer offset for d
    out_base = out_ptr + d * out_stride_d

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K
        k_range = k + k_range_base
        mask_k = k_range < N

        # Load left block (i,k)
        left_ptrs = left_ptr + d * stride_d + i_range[:, None] * stride_i + k_range[None, :] * stride_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        left = left.to(tl.float32)

        # Load right block (k,j)
        right_ptrs = right_ptr + d * stride_d + k_range[:, None] * stride_k + j_range[None, :] * stride_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)
        right = right.to(tl.float32)

        # dot product: left[I,K] @ right[K,J] => shape (I,J)
        acc += tl.dot(left, right)  # using default K reduction

    out_ptrs = out_base + i_range[:, None] * out_stride_i + j_range[None, :] * out_stride_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # LayerNorm: using weight bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # We'll compute using torch.nn.functional.layer_norm
    # normalized_shape = dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    # left_proj and right_proj
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present (mask shape B,N,N)
    if mask is not None:
        # broadcast mask to last dimension
        mask = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert left/right to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # Allocate output (hidden_dim)
    out_hidden = torch.empty((batch_size, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=left.device)

    # Run kernel per batch
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    num_i = (seq_len + BLOCK_I - 1) // BLOCK_I
    num_j = (seq_len + BLOCK_J - 1) // BLOCK_J

    for b in range(batch_size):
        left_ptr = left_fp16[b].data_ptr()
        right_ptr = right_fp16[b].data_ptr()
        out_ptr = out_hidden[b].data_ptr()
        # strides
        # left stride: (i, k, d)
        left_stride_i = left_fp16.stride(1)  # stride for i axis
        left_stride_k = left_fp16.stride(2)  # stride for k axis
        left_stride_d = left_fp16.stride(3)  # stride for d axis
        # right stride same as left
        right_stride_i = right_fp16.stride(1)
        right_stride_k = right_fp16.stride(2)
        right_stride_d = right_fp16.stride(3)
        out_stride_i = out_hidden.stride(1)
        out_stride_j = out_hidden.stride(2)
        out_stride_d = out_hidden.stride(3)

        grid = (num_i, num_j, hidden_dim)
        tri_mul_kernel[grid](
            left_ptr, right_ptr, out_ptr,
            seq_len, hidden_dim,
            left_stride_i, left_stride_k, left_stride_d,
            # (right strides not needed because we reuse left strides but can pass anyway)
            out_stride_i, out_stride_j, out_stride_d,
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
            num_warps=4,
        )
    # out_hidden now holds the summed product across k for each hidden_dim. Equivalent to result of einsum.

    # Continue with to_out_norm, out_gate, to_out linear

    # LayerNorm on out_hidden
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    output = torch.nn.functional.linear(out_ln, to_out_weight)  # shape B,N,N,dim

    return output

"""
TriMul forward (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Linear projections for left/right edge representations and for three gates.
3. Apply the optional pair‑wise mask.
4. Gating: left = left * sigmoid(left_gate), right = right * sigmoid(right_gate).
5. Triangle multiplication (core O(N³) work):
   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d].
   This is a batched sum‑of‑outer‑products and can be written as
   out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T.
   A custom Triton kernel tiles the i‑, j‑ and k‑axes (32×32×32) and
   accumulates the result in FP32 while loading the inputs in FP16
   (bandwidth‑friendly).  The kernel is launched with a 3‑D grid
   (i‑blocks, j‑blocks, hidden‑dim) and is executed once per batch.
6. Layer‑norm the hidden‑dim result, multiply by the output gate and
   apply the final linear projection back to the original channel‑size.

Only the heavy triangle multiplication is performed on the GPU via Triton;
all other operations (norms, linears, sigmoids) use native PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,       # const float16 *
    right_ptr,      # const float16 *
    out_ptr,        #       float32 *
    N,              # int32  (sequence length)
    hidden_dim,     # int32  (hidden dimension)

    # strides for left/right tensors (same layout)
    stride_i,       # int32  stride of the first N dimension  (i / j axis)
    stride_k,       # int32  stride of the second N dimension (k axis)
    stride_d,       # int32  stride of the channel axis (usually 1)

    # strides for output tensor
    out_stride_i,   # int32
    out_stride_j,   # int32
    out_stride_d,   # int32

    BLOCK_I: tl.constexpr,   # tile size in i dimension
    BLOCK_J: tl.constexpr,   # tile size in j dimension
    BLOCK_K: tl.constexpr,   # tile size in k dimension
):
    # ---------------------------------------------------------------
    # Program IDs define the current tile (i‑block, j‑block, channel)
    # ---------------------------------------------------------------
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_d = tl.program_id(2)          # channel index (0 … hidden_dim‑1)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    d = pid_d

    # Out‑of‑bounds guard (early exit)
    if (i_start >= N) or (j_start >= N) or (d >= hidden_dim):
        return

    # ---------------------------------------------------------------
    # Index ranges for the current tile
    # ---------------------------------------------------------------
    i_range = i_start + tl.arange(0, BLOCK_I)      # [i_start, i_start+BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)      # [j_start, j_start+BLOCK_J)
    k_base  = tl.arange(0, BLOCK_K)               # used inside the k‑loop

    mask_i = i_range < N
    mask_j = j_range < N

    # ----------------------------------------------------------------
    # Base pointer for the selected channel (d)
    # ----------------------------------------------------------------
    left_base  = left_ptr  + d * stride_d
    right_base = right_ptr + d * stride_d
    out_base   = out_ptr   + d * out_stride_d

    # ----------------------------------------------------------------
    # Accumulator (FP32) for the tile
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----------------------------------------------------------------
    # Loop over k in tiles
    # ----------------------------------------------------------------
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for tile_k in range(0, num_tiles_k):
        k_start = tile_k * BLOCK_K
        k_range = k_start + k_base
        mask_k = k_range < N

        # -----------------------------
        # Load left[i, k, d]  (shape: BLOCK_I × BLOCK_K)
        # -----------------------------
        left_ptrs = left_base \
                    + i_range[:, None] * stride_i \
                    + k_range[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)
        left = left.to(tl.float32)

        # -----------------------------
        # Load right[j, k, d] transposed to (k, j) for GEMM
        #   right[k, j, d] == right[j, k, d] in memory
        # -----------------------------
        right_ptrs = right_base \
                     + k_range[:, None] * stride_k \
                     + j_range[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)
        right = right.to(tl.float32)

        # -----------------------------
        # Accumulate the matrix product: (I×K) @ (K×J) → (I×J)
        # -----------------------------
        acc += tl.dot(left, right)   # reduces over K

    # ---------------------------------------------------------------
    # Store the computed tile back to the output tensor (FP32)
    # ---------------------------------------------------------------
    out_ptrs = out_base \
               + i_range[:, None] * out_stride_i \
               + j_range[None, :] * out_stride_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim]
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input channel dimension (c_z)
    hidden_dim = config["hidden_dim"] # hidden dimension (c)

    # -----------------------------------------------------------------
    # Helper to fetch weight/bias tensors (already on the correct device)
    # -----------------------------------------------------------------
    def get(name):
        return weights[name]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32)
    # -----------------------------------------------------------------
    norm_weight = get("norm.weight")
    norm_bias   = get("norm.bias")
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w  = get("left_proj.weight")      # (hidden_dim, dim)
    right_proj_w = get("right_proj.weight")     # (hidden_dim, dim)

    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        # ensure float mask (0/1) and broadcast
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear transforms of the *pre‑norm* input)
    # -----------------------------------------------------------------
    left_gate_w  = get("left_gate.weight")   # (hidden_dim, dim)
    right_gate_w = get("right_gate.weight")  # (hidden_dim, dim)
    out_gate_w   = get("out_gate.weight")    # (hidden_dim, dim)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Triangle multiplication – fused Triton kernel
    # -----------------------------------------------------------------
    # Cast to FP16 for bandwidth reduction (kernel accumulates in FP32)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float32,
                             device=left_fp16.device)

    # Tile sizes – chosen to match typical H100 warp geometry
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # Strides (in elements) for the contiguous tensors
    s_i = left_fp16.stride(1)   # stride of the first N axis (i / j)
    s_k = left_fp16.stride(2)   # stride of the second N axis (k)
    s_d = left_fp16.stride(3)   # stride of the channel axis (usually 1)

    out_s_i = out_hidden.stride(1)
    out_s_j = out_hidden.stride(2)
    out_s_d = out_hidden.stride(3)

    # Launch the kernel per batch (B ≤ 2, cheap Python loop)
    for b in range(B):
        # Pointers to the start of the batch slice
        left_ptr  = left_fp16[b].data_ptr()
        right_ptr = right_fp16[b].data_ptr()
        out_ptr   = out_hidden[b].data_ptr()

        grid = (
            (N + BLOCK_I - 1) // BLOCK_I,    # number of i‑tiles
            (N + BLOCK_J - 1) // BLOCK_J,    # number of j‑tiles
            hidden_dim,                      # one program per channel
        )
        tri_mul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            s_i, s_k, s_d,
            out_s_i, out_s_j, out_s_d,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            num_warps=8,                     # tuned for H100
        )

    # -----------------------------------------------------------------
    # 6️⃣ Post‑multiplication LayerNorm, output gate, final linear
    # -----------------------------------------------------------------
    to_out_norm_w = get("to_out_norm.weight")
    to_out_norm_b = get("to_out_norm.bias")
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # [B, N, N, hidden_dim]

    # Apply output gate (shape already matches)
    out_hidden = out_hidden * out_gate

    # Final projection back to input dimension (no bias)
    to_out_w = get("to_out.weight")   # (dim, hidden_dim)
    output = F.linear(out_hidden, to_out_w)  # [B, N, N, dim]

    return output
