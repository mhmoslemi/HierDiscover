"""
TriMul (outgoing) – fast forward implementation
------------------------------------------------
* LayerNorm on the 4‑D pair tensor (float32 → fp16)
* Optional mask applied in‑place by a tiny Triton kernel (broadcast over C)
* Fused linear projection of the 4 branches (left/right proj & gates) plus the
  output‑gate into a single GEMM (half precision)
* Element‑wise sigmoid gates (in‑place)
* Core O(N³·H) update *left·rightᵀ* performed by a custom “flash‑attention‑style”
  Triton kernel that computes the full batched matrix product Q·Kᵀ in fp16
  (accumulating in fp32) – this replaces the huge torch.matmul call.
* Output LayerNorm, gating, and final linear back to the original dim.
* The final tensor is returned as float32 (identical to the reference module).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Mask kernel (broadcast mask over the channel dimension)
# ----------------------------------------------------------------------
@triton.jit
def _mask_input_kernel(
    x_ptr, mask_ptr,
    B, N, C,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """x[b, i, j, c] *= mask[b, i, j]   (mask is broadcast over C)."""
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # load a tile of mask (scalar per (i,j)) and broadcast over C
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                     # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # loop over C tiles
    off_c = 0
    while off_c < C:
        c = off_c + tl.arange(0, BLOCK_C)
        c_ok = c < C

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + c[None, None, :] * stride_x_c,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
            other=0.0,
        )
        x = x * mask
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + c[None, None, :] * stride_x_c,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
        )
        off_c += BLOCK_C


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place x *= mask (mask broadcast over the channel dimension)."""
    assert x.is_cuda and mask.is_cuda
    B, N, _, C = x.shape
    mask = mask.to(dtype=x.dtype)

    s_x = x.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_C = 64

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_input_kernel[grid](
        x,
        mask,
        B,
        N,
        C,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    # x is modified in‑place


# ----------------------------------------------------------------------
# 2. Core O(N³·H) kernel – fused left·rightᵀ (flash‑attention style)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    B, H, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]"""
    pid = tl.program_id(0)           # combined batch‑head axis
    pid_i = tl.program_id(1)         # tile over i
    pid_j = tl.program_id(2)         # tile over j

    b = pid // H                     # batch index
    h = pid % H                      # head index

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over the reduction dimension k
    # ------------------------------------------------------------------
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = start_k * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        k_ok = k < N

        # left block  (i, k, h)
        left = tl.load(
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + i[:, None] * stride_li
            + k[None, :] * stride_lk,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        ).to(tl.float32)

        # right block (j, k, h)
        right = tl.load(
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + j[:, None] * stride_rj
            + k[None, :] * stride_rk,
            mask=j_ok[:, None] & k_ok[None, :],
            other=0.0,
        ).to(tl.float32)

        # dot‑product over k (with Kᵀ)
        acc += tl.dot(left, right, trans_b=True)

    # write back (cast to fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + i[:, None] * stride_oi
        + j[None, :] * stride_oj,
        out,
        mask=i_ok[:, None] & j_ok[None, :],
    )


def _batched_qk_t_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    left/right: (B, N, N, H)  – fp16
    returns out of same shape (B, N, N, H) (fp16)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B,N,N,H)

    # strides for the four‑dim tensors
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # batch‑head axis
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑tiles
    )
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        B,
        H,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# 3. Entry point – the forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (AlphaFold‑3 style).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weights (float32 in the checkpoint)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]       # (H, C)
    right_proj_weight = weights["right_proj.weight"]     # (H, C)

    left_gate_weight = weights["left_gate.weight"]       # (H, C)
    right_gate_weight = weights["right_gate.weight"]     # (H, C)
    out_gate_weight = weights["out_gate.weight"]         # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]   # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]       # (H,)
    to_out_weight = weights["to_out.weight"]             # (C, H)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑Norm on the pair tensor (float32 → fp16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                     # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Optional mask (in‑place via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(x, mask_tensor)   # modifies x in‑place

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 and fuse the four linear branches + out‑gate
    # ------------------------------------------------------------------
    x_h = x.half()     # (B, N, N, C)

    # build a single weight matrix of shape (5*H, C)
    fused_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (5*H, C)

    B, N, _, _ = x_h.shape
    x_flat = x_h.reshape(-1, dim)                # (B*N*N, C)

    fused = F.linear(x_flat, fused_weight)       # (B*N*N, 5*H)
    fused = fused.view(B, N, N, 5 * hidden_dim)  # (B, N, N, 5*H)

    # split into the five branches
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )   # each (B, N, N, H)

    # in‑place sigmoid to avoid extra allocations
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

    # gated projections
    left = left_proj * left_gate                # (B, N, N, H), fp16
    right = right_proj * right_gate             # (B, N, N, H), fp16

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – custom Triton kernel (Q·Kᵀ)
    # ------------------------------------------------------------------
    out = _batched_qk_t_matmul(left, right)      # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, gate, final projection (stay in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )   # fp16

    out = out * out_gate                         # gating

    out = F.linear(out, to_out_weight.half())   # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # Return in float32 (identical to the reference implementation)
    # ------------------------------------------------------------------
    return out.float()
