@triton.jit
def _make_left_mat_kernel(
    combined_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_comb_b, stride_comb_i, stride_comb_k, stride_comb_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr, NOMASK: tl.constexpr
):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)

    i_ok = i < N
    k_ok = k < N

    # Load mask
    if NOMASK:
        mask_tile = tl.full((BLOCK_I, BLOCK_K), 1.0, dtype=tl.float16)
    else:
        mask_tile = tl.load(
            mask_ptr + b * stride_mask_b
                       + i[:, None] * stride_mask_i
                       + k[None, :] * stride_mask_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
            dtype=tl.float16,
        )
    mask_tile = mask_tile[:, :, None]  # (I,K,1)

    # base offset for output batch*H dimension
    batch_offset = b * H * stride_out_b

    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h_tiles):
        h = hi * BLOCK_H + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # Load left_proj
        left_proj = tl.load(
            combined_ptr + b * stride_comb_b
                         + i[:, None, None] * stride_comb_i
                         + k[None, :, None] * stride_comb_k
                         + h[None, None, :] * stride_comb_d,
            mask=i_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load left_gate (offset H)
        left_gate = tl.load(
            combined_ptr + b * stride_comb_b
                         + i[:, None, None] * stride_comb_i
                         + k[None, :, None] * stride_comb_k
                         + (h + H)[None, None, :] * stride_comb_d,
            mask=i_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Sigmoid
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))

        left_val = left_proj * left_gate * mask_tile

        # Store to out_ptr at position (b*H + h, i, k)
        tl.store(
            out_ptr + batch_offset
                    + h[:, None, None] * stride_out_b
                    + i[:, None, None] * stride_out_i
                    + k[None, :, None] * stride_out_k,
            left_val,
            mask=i_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
        )
```

"""
Fast outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the input tensor (float32 → fp32 stable).
2️⃣ Cast to fp16 and compute the four linear branches (proj + gate) in a single
   GEMM (weight of shape (4·H, C)).  The result is a tensor
   `combined` of shape [B, N, N, 4·H].
3️⃣ A fused Triton kernel reads `combined` together with the optional mask,
   applies the sigmoid gates, multiplies by the mask and writes the two
   projected tensors directly into the layout required for the batched GEMM:
      * `left_mat`  – (B·H, N, N)  with rows = i, cols = k
      * `right_mat` – (B·H, N, N)  with rows = k, cols = j
   This eliminates the costly permute‑+‑contiguous copies of the reference.
4️⃣ Batched matrix‑multiply (FP16 Tensor‑Core)   out_mat = left_mat @ right_mat
   yields (B·H, N, N).
5️⃣ Reshape back to [B, N, N, H] and apply the output LayerNorm, output‑gate
   (computed from the normalized input) and the final projection.
6️⃣ Return FP32 tensor.

All heavy arithmetic stays in fp16, only the first LayerNorm runs in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – produce left_mat (i,k) and right_mat (k,j) from combined
# ----------------------------------------------------------------------
@triton.jit
def _make_left_mat_kernel(
    combined_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_comb_b, stride_comb_i, stride_comb_k, stride_comb_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    NOMASK: tl.constexpr
):
    """Write left_proj * sigmoid(left_gate) (optionally masked) into left_mat."""
    b  = tl.program_id(0)          # batch
    ti = tl.program_id(1)          # tile i
    tk = tl.program_id(2)          # tile k

    i = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    k = tk * BLOCK_K + tl.arange(0, BLOCK_K)

    i_ok = i < N
    k_ok = k < N

    # --------------------------------------------------------------
    # Load mask if required (shape BLOCK_I x BLOCK_K)
    # --------------------------------------------------------------
    if NOMASK:
        mask_tile = tl.full((BLOCK_I, BLOCK_K), 1.0, dtype=tl.float16)
    else:
        mask_tile = tl.load(
            mask_ptr + b * stride_mask_b
                     + i[:, None] * stride_mask_i
                     + k[None, :] * stride_mask_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
            dtype=tl.float16,
        )
    # Broadcast over hidden dimension
    mask_tile = mask_tile[:, :, None]   # (I, K, 1)

    # ------------------------------------------------------------------
    # Offsets for the output batch dimension (B*H)
    # ------------------------------------------------------------------
    batch_offset = b * H * stride_out_b   # start of this batch in left_mat

    # ------------------------------------------------------------------
    # Loop over hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    num_h = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h):
        h = hi * BLOCK_H + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # left projection slice (offset 0:H)
        left_proj = tl.load(
            combined_ptr + b * stride_comb_b
                         + i[:, None, None] * stride_comb_i
                         + k[None, :, None] * stride_comb_k
                         + h[None, None, :] * stride_comb_d,
            mask=i_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # left gate slice (offset H:2H)
        left_gate = tl.load(
            combined_ptr + b * stride_comb_b
                         + i[:, None, None] * stride_comb_i
                         + k[None, :, None] * stride_comb_k
                         + (h + H)[None, None, :] * stride_comb_d,
            mask=i_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # sigmoid gate
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))

        # fused value
        val = left_proj * left_gate * mask_tile   # (I, K, H)

        # Store to left_mat   layout: (batch*H, i, k)
        ptr = out_ptr + batch_offset \
                     + h[:, None, None] * stride_out_b \
                     + i[:, None, None] * stride_out_i \
                     + k[None, :, None] * stride_out_k
        tl.store(
            ptr,
            val,
            mask=i_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
        )


@triton.jit
def _make_right_mat_kernel(
    combined_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_comb_b, stride_comb_i, stride_comb_k, stride_comb_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k,
    BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    NOMASK: tl.constexpr
):
    """Write right_proj * sigmoid(right_gate) (optionally masked) into right_mat."""
    b  = tl.program_id(0)          # batch
    tj = tl.program_id(1)          # tile j (output column)
    tk = tl.program_id(2)          # tile k (shared dimension)

    j = tj * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tk * BLOCK_K + tl.arange(0, BLOCK_K)

    j_ok = j < N
    k_ok = k < N

    if NOMASK:
        mask_tile = tl.full((BLOCK_J, BLOCK_K), 1.0, dtype=tl.float16)
    else:
        mask_tile = tl.load(
            mask_ptr + b * stride_mask_b
                     + j[:, None] * stride_mask_i
                     + k[None, :] * stride_mask_k,
            mask=j_ok[:, None] & k_ok[None, :],
            other=0.0,
            dtype=tl.float16,
        )
    mask_tile = mask_tile[:, :, None]   # (J, K, 1)

    batch_offset = b * H * stride_out_b

    num_h = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h):
        h = hi * BLOCK_H + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # right projection slice (offset 2H:3H)
        right_proj = tl.load(
            combined_ptr + b * stride_comb_b
                         + j[:, None, None] * stride_comb_i   # i‑dim of combined = j here
                         + k[None, :, None] * stride_comb_k
                         + (h + 2 * H)[None, None, :] * stride_comb_d,
            mask=j_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # right gate slice (offset 3H:4H)
        right_gate = tl.load(
            combined_ptr + b * stride_comb_b
                         + j[:, None, None] * stride_comb_i
                         + k[None, :, None] * stride_comb_k
                         + (h + 3 * H)[None, None, :] * stride_comb_d,
            mask=j_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

        val = right_proj * right_gate * mask_tile   # (J, K, H)

        # Store to right_mat   layout: (batch*H, k, j)
        ptr = out_ptr + batch_offset \
                     + h[:, None, None] * stride_out_b \
                     + k[:, None, None] * stride_out_i \
                     + j[None, :, None] * stride_out_k
        tl.store(
            ptr,
            val,
            mask=j_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :],
        )


def _project_to_mat_triton(combined: torch.Tensor, mask: torch.Tensor,
                           left_mat: torch.Tensor, right_mat: torch.Tensor,
                           nomask: bool):
    """Fused kernel that fills left_mat and right_mat from `combined`."""
    B, N, _, fourH = combined.shape
    H = fourH // 4
    device = combined.device

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 64

    # ----------------------------------------------
    # left_mat
    # ----------------------------------------------
    grid_left = (B,
                 (N + BLOCK_I - 1) // BLOCK_I,
                 (N + BLOCK_K - 1) // BLOCK_K)

    s_comb = combined.stride()
    s_mask = mask.stride()
    s_l = left_mat.stride()

    _make_left_mat_kernel[grid_left](
        combined,
        mask,
        left_mat,
        B,
        N,
        H,
        s_comb[0], s_comb[1], s_comb[2], s_comb[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_l[0], s_l[1], s_l[2],
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        NOMASK=int(nomask),
        num_warps=4,
    )

    # ----------------------------------------------
    # right_mat
    # ----------------------------------------------
    grid_right = (B,
                  (N + BLOCK_J - 1) // BLOCK_J,
                  (N + BLOCK_K - 1) // BLOCK_K)

    s_r = right_mat.stride()

    _make_right_mat_kernel[grid_right](
        combined,
        mask,
        right_mat,
        B,
        N,
        H,
        s_comb[0], s_comb[1], s_comb[2], s_comb[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_r[0], s_r[1], s_r[2],
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        NOMASK=int(nomask),
        num_warps=4,
    )

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  fp32

    # --------------------------------------------------------------
    # 2️⃣ FP16 cast + fused projection (four linear branches in one GEMM)
    # --------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, C]  fp16
    B, N, _, _ = x_h.shape

    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16)          # (4*H, C)

    # Linear on flattened tensor
    x_flat = x_h.view(-1, dim)                     # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4H)

    # --------------------------------------------------------------
    # 3️⃣ Allocate contiguous buffers for the two projected tensors
    # --------------------------------------------------------------
    left_mat = torch.empty(
        (B * hidden_dim, N, N), dtype=torch.float16, device=x.device
    )
    right_mat = torch.empty_like(left_mat)

    # --------------------------------------------------------------
    # 4️⃣ Fuse gating, optional mask and layout transformation
    # --------------------------------------------------------------
    mask_fp16 = mask_tensor.to(dtype=torch.float16)
    _project_to_mat_triton(combined, mask_fp16, left_mat, right_mat, nomask)

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM (L @ Rᵀ) in fp16
    # --------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)                # (B*H, N, N) fp16

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm, output‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    # output gate computed from the normalized **input**
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H) fp16
    out = out * out_gate

    # final linear back to original dimension
    out = F.linear(out, to_out_weight.half())                      # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # Return FP32 tensor (matches reference implementation)
    # --------------------------------------------------------------
    return out.float()
