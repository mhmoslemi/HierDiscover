"""
Fast forward‑only outgoing TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣ Layer‑norm the input (FP32) and cast to FP16.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
   single half‑precision GEMM (shape B·N·N × 4·H).
3️⃣ If a mask is supplied, apply it in‑place to the projected left/right tensors
   with a tiny Triton kernel (broadcast over the hidden dimension).
4️⃣ Apply sigmoid gates to the projected tensors.
5️⃣ Core O(N³·H) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched FP16 matrix‑multiply (torch.bmm) after reshaping
   to (B·H, N, N).  This maps to a cuBLAS GEMM on H100.
6️⃣ Hidden‑dim LayerNorm, output‑gate (sigmoid) and final linear projection back
   to the original channel dimension.
7️⃣ Cast the result back to FP32.

The only Triton‑implemented part is the mask‑multiplication kernel; everything
else uses highly‑optimized PyTorch FP16 kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication for two tensors (left & right)
# ------------------------------------------------------------------------
@triton.jit
def _mask_mul_two_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    left[b,i,j,h] *= mask[b,i,j];   right[b,i,j,h] *= mask[b,i,j]
    All tensors are assumed contiguous.
    """
    b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)      # tile over i dimension
    pid_j = tl.program_id(2)      # tile over j dimension

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # Load mask tile (BLOCK_I, BLOCK_J)
    mask_tile = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i[:, None] * stride_mask_i
        + j[None, :] * stride_mask_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)

    # Broadcast the mask over the hidden dimension
    mask_tile = mask_tile[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension in tiles
    h_offset = 0
    while h_offset < H:
        h = h_offset + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # Load left and right tiles
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )

        # Apply mask (broadcasted)
        left = left * mask_tile
        right = right * mask_tile

        # Store back
        tl.store(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            left,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            right,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )

        h_offset += BLOCK_H


def _apply_mask_two_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place mask broadcast: left *= mask, right *= mask.
    left/right: (B,N,N,H) half‑precision, contiguous.
    mask:       (B,N,N) half‑precision (0/1).
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    _mask_mul_two_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # Mask applied in‑place; nothing to return


# ------------------------------------------------------------------------
# Outgoing TriMul – forward‑only entry point
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys
                * dim            – input channel C (int)
                * hidden_dim     – hidden dimension H (int)
                * nomask         – bool, True ⇒ ignore mask
                * layernorm_eps  – optional epsilon for LayerNorm

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = config.get("layernorm_eps", 1e-5)

    assert C == dim, f"Expected input dim {dim}, got {C}"
    device = input_tensor.device

    # -----------------------------------------------------------------
    # Load weights (all stored as float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – keeps numerical stability
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy work
    # -----------------------------------------------------------------
    x_h = x.half()  # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # 3️⃣ Fuse four linear projections into a single GEMM:
    #    left_proj, right_proj, left_gate, right_gate
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)  # (4*H, C)

    # Flatten leading dims for a batched GEMM
    x_flat = x_h.view(-1, C)                         # (B·N·N, C)

    fused_out = F.linear(x_flat, fused_weight)      # (B·N·N, 4*H)

    # Split and reshape back to (B,N,N,H)
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden).contiguous()
    right_proj = right_proj.view(B, N, N, hidden).contiguous()
    left_gate_pre = left_gate_pre.view(B, N, N, hidden).contiguous()
    right_gate_pre = right_gate_pre.view(B, N, N, hidden).contiguous()

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – in‑place via Triton
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Ensure mask is half‑precision (0/1) and contiguous
        mask_h = mask_tensor.to(dtype=torch.float16, device=device).contiguous()
        _apply_mask_two_triton(left_proj, right_proj, mask_h)

    # -----------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (use float32 for stability then cast)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre.float()).half()
    right_gate = torch.sigmoid(right_gate_pre.float()).half()

    left = left_proj * left_gate   # (B,N,N,H) fp16
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Core contraction: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM over (B*H, N, N)
    # -----------------------------------------------------------------
    # Reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Batched matmul (left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    # Output gate: computed from the normalized input x (float32) → half
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight.float())).half()  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())  # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Return FP32 tensor
    # -----------------------------------------------------------------
    return out.float()
