def batch_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)  # batch * hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for rows and columns
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to start of block
    a_ptrs = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = B_ptr + batch_id * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a block of A
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K), other=0.0)
        # load a block of B (note B is not transposed; we need B_T)

        acc += tl.dot(a, b)
        # advance pointers

        b_ptrs += BLOCK_K * stride_b_k
    # write back
    c_ptrs = C_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # Setup dims
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    device = input_tensor.device

    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)   # float32

    x_h = x.to(torch.float16)   # cast to half

    # flatten x for linear ops
    x_flat = x_h.view(-1, dim)

    # fused projection weight (left+right)
    proj_weight = torch.cat([weights['left_proj.weight'], weights['right_proj.weight']], dim=0).to(dtype=torch.float16, device=device)
    proj = F.linear(x_flat, proj_weight)   # (B*N*N, 2*hidden)
    left_proj, right_proj = torch.split(proj, hidden, dim=-1)
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)

    # fused gate weight (left_gate+right_gate+out_gate)
    gate_weight = torch.cat([weights['left_gate.weight'], weights['right_gate.weight'], weights['out_gate.weight']], dim=0).to(dtype=torch.float16, device=device)
    gates = F.linear(x_flat, gate_weight)  # (B*N*N, 3*hidden)
    left_gate, right_gate, out_gate = torch.split(gates, hidden, dim=-1)
    left_gate = left_gate.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)
    out_gate = out_gate.view(B, N, N, hidden)

    # apply mask if exists
    if not config.get("nomask", True):
        mask = mask_tensor.unsqueeze(-1).to(dtype=torch.float16, device=device)  # (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # gating
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # core batched GEMM using Triton
    # permute to (B, H, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    batch_hidden = B * hidden
    left_mat = left_perm.view(batch_hidden, N, N)
    right_mat = right_perm.view(batch_hidden, N, N)
    out_mat = torch.empty_like(left_mat)  # dtype=fp16

    # Kernel call
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (batch_hidden, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    batch_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_hidden, N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        # maybe num_stages=3
    )
    # reshape back
    out = out_mat.view(B, hidden, N, N).permute(0,2,3,1).contiguous()   # shape (B,N,N,hidden)

    # output layernorm + gating
    out_norm = F.layer_norm(out, (hidden,), weight=weights['to_out_norm.weight'].to(torch.float16, device=device), bias=weights['to_out_norm.bias'].to(torch.float16, device=device), eps=eps)
    out_gate_sig = torch.sigmoid(out_gate)   # shape (B,N,N,hidden)
    out = out_norm * out_gate_sig

    # final linear
    out = F.linear(out, weights['to_out.weight'].to(torch.float16, device=device))   # shape (B,N,N,dim)
    return out.float()

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3 style).

The forward pass consists of:
  1️⃣ LayerNorm on the input (FP32 for stability).
  2️⃣ Two fused linear projections (left/right) and a single fused
      linear for the three gates (left_gate, right_gate, out_gate) – all
      in FP16.
  3️⃣ Optional mask multiplication (broadcast over the hidden dim);
      performed with a simple Torch element‑wise op (the heavy work is
      still in Triton, satisfying the “use Triton” requirement).
  4️⃣ Sigmoid gating.
  5️⃣ Core O(N³·H) update: a batched matrix multiplication for every
      hidden head.  This is implemented as a custom Triton kernel that
      processes the whole (batch × hidden) dimension in one launch,
      avoiding the overhead of many tiny torch‑bmm calls.
  6️⃣ Output LayerNorm + out‑gate multiplication.
  7️⃣ Final linear projection back to the original channel dimension.
All arithmetic after the initial LayerNorm runs in FP16 with FP32
accumulation; the result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1️⃣ Batched GEMM kernel (left @ rightᵀ for every hidden head)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,                     # left  tensor   [B*H, M, K]   (M=N, K=N)
    B_ptr,                     # right tensor   [B*H, K, N]   (stored as normal, we read transposed)
    C_ptr,                     # output tensor  [B*H, M, N]
    BATCH, M, N, K,            # logical sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.

    A : [batch, M, K]   (row‑major)
    B : [batch, K, N]   (row‑major, we access it as Bᵀ)
    C : [batch, M, N]
    """
    batch_id = tl.program_id(0)          # batch × hidden
    pid_m = tl.program_id(1)             # tile over rows  (M)
    pid_n = tl.program_id(2)             # tile over cols  (N)

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch element
    a_base = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m
    b_base = B_ptr + batch_id * stride_b_batch + offs_n[None, :] * stride_b_m

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over the K dimension
    # ------------------------------------------------------------------
    iters = tl.cdiv(K, BLOCK_K)
    for k in range(iters):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)          # (BLOCK_K,)

        mask_k = k_range < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + k_range[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ tile (we read B as transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base + k_range[:, None] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # FP32 accumulation

    # ------------------------------------------------------------------
    # write back the result (cast to FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# 2️⃣ Triton‑enabled TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                     # input channel dimension C
    hidden = config["hidden_dim"]           # hidden dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy arithmetic
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)               # (B, N, N, C)
    B, N, _, C = x_h.shape

    # --------------------------------------------------------------
    # 3️⃣ Fused projections (left/right) and fused gates
    #    (left_gate, right_gate, out_gate) – all in one matmul each
    # --------------------------------------------------------------
    # flatten the spatial dimensions: (B*N*N, C)
    x_flat = x_h.view(-1, C)

    # ---- left/right projection (2*hidden) ---------------------------------
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)               # (2*H, C)

    proj = F.linear(x_flat, proj_weight)                    # (B*N*N, 2*H)
    proj = proj.view(B, N, N, 2 * hidden)
    left_proj, right_proj = torch.split(proj, hidden, dim=-1)  # (B,N,N,H)

    # ---- gate projections (3*hidden) --------------------------------------
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)               # (3*H, C)

    gates = F.linear(x_flat, gate_weight)                   # (B*N*N, 3*H)
    gates = gates.view(B, N, N, 3 * hidden)
    left_gate, right_gate, out_gate = torch.split(gates, hidden, dim=-1)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask multiplication (broadcast over hidden)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left = left_proj * torch.sigmoid(left_gate)   # (B,N,N,H)
    right = right_proj * torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update – batched GEMM via Triton
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch×head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_hidden = B * hidden
    left_mat = left_perm.view(batch_hidden, N, N)       # (B*H, N, N)
    right_mat = right_perm.view(batch_hidden, N, N)     # (B*H, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_m, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_hidden,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_m,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"]
        .to(dtype=torch.float16, device=device),
        bias=weights["to_out_norm.bias"]
        .to(dtype=torch.float16, device=device),
        eps=eps,
    )
    out_gate = torch.sigmoid(out_gate)       # (B,N,N,H)
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype=torch.float16, device=device),
    )  # (B, N, N, C)  FP16

    # --------------------------------------------------------------
    # 9️⃣ Return FP32 tensor (matches reference)
    # --------------------------------------------------------------
    return out.float()
