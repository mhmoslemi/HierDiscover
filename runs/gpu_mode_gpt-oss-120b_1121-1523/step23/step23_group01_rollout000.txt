"""
Fast outgoing TriMul – forward‑only (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (FP32 → FP16).
2️⃣  Fuse the 4 linear projections (left, right, left‑gate, right‑gate) into one
    GEMM (shape (4*H, C)) and reshape to (B, N, N, H) tensors.
3️⃣  If a mask is present, a Triton kernel fuses:
        * loading of left/right projections,
        * sigmoid of pre‑gate tensors,
        * element‑wise mask,
        * multiplication → gated & masked tensors.
   Otherwise only a cheap PyTorch sigmoid‑gate is applied.
4️⃣  Core contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply:
        left[b,:, :,h] @ right[b,:, :,h].T
   Implemented via ``torch.bmm`` on a view (B*H, N, N).
5️⃣  Hidden‑dim LayerNorm (FP16), element‑wise output‑gate, and final linear
    projection back to the original channel dimension.
6️⃣  Cast the result back to FP32 (matching the reference implementation).

The most expensive operation (step 4) is handled by CuTensor‑accelerated
batched GEMM, while the mask‑gate fusion is done in Triton as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: mask + sigmoid‑gate + element‑wise multiplication
# -------------------------------------------------------------
@triton.jit
def _mask_gate_mul_kernel(
    # -------------- pointers --------------
    left_ptr, right_ptr,               # (B,N,N,H) – will be overwritten
    left_gate_ptr, right_gate_ptr,     # (B,N,N,H) – pre‑gate values
    mask_ptr,                          # (B,N,N) – 0/1 mask (half)

    # -------------- problem size ----------
    B, N, H,
    # -------------- strides ---------------
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,

    # -------------- tile sizes ------------
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    In‑place:
        left[b,i,j,h]  = left_proj[b,i,j,h] *
                         sigmoid(left_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
        right[b,i,j,h] = right_proj[b,i,j,h] *
                         sigmoid(right_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
    All tensors are half‑precision.
    """
    b = tl.program_id(0)               # batch
    pid_i = tl.program_id(1)           # tile over i
    pid_j = tl.program_id(2)           # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # load mask tile and broadcast over H
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )                     # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]    # (BLOCK_I, BLOCK_J, 1)

    h_offset = 0
    while h_offset < H:
        h = h_offset + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # Load tensors
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        left_g = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right_g = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability, back to fp16
        left_gate = 1.0 / (1.0 + tl.exp(-left_g.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_g.to(tl.float32)))
        left_gate = left_gate.to(tl.float16)
        right_gate = right_gate.to(tl.float16)

        # apply mask & gate
        left_out = left * left_gate * mask_tile
        right_out = right * right_gate * mask_tile

        # store in‑place
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )

        h_offset += BLOCK_H


def _apply_mask_and_gate_triton(left_proj, right_proj,
                                left_gate_pre, right_gate_pre,
                                mask):

    In‑place Triton fusion of mask, sigmoid‑gate and multiplication.
    Expected dtype: torch.float16.
    Shapes:

    B, N, _, H = left_proj.shape
    # strides (in elements, not bytes)
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate_pre.stride()
    s_rg = right_gate_pre.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32   # fits H up to 384 with a few loop iterations

        left_proj,
        right_proj,
        left_gate_pre,
        right_gate_pre,
        mask,

        s_l[0], s_l[1], s_l[2], s_l[3],

        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    """
    Forward‑only outgoing TriMul.

    Args:
        data: (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of parameters (float32 tensors)
            - config        : dict with keys

                * hidden_dim     – hidden dimension H (int)
                * nomask         – bool (True ⇒ ignore mask)

        torch.Tensor [B, N, N, C] (float32) – the TriMul result.
    """

    # Unpack and sanity‑check
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # C
    hidden = config["hidden_dim"]     # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    assert C == dim, f"dim mismatch: got {C}, expected {dim}"
    torch.backends.cuda.matmul.allow_tf32 = True   # speed‑up on H100

    # -----------------------------------------------------------------
    # Load model weights (all FP32 tensors on same device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) – keep numeric stability
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy path
    # -----------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, C] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Fuse four linear projections (left, right, left_gate, right_gate)
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)    # (4*H, C)

    # Linear over the last dimension
    x_flat = x_h.view(-1, dim)                   # (B*N*N, C)
    fused_out = F.linear(x_flat, fused_weight)  # (B*N*N, 4*H)

    # Split
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Apply mask + sigmoid‑gate (Triton fusion if mask present)
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Convert mask to half (0/1) – broadcasting works fine
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_and_gate_triton(
            left_proj, right_proj,
            left_gate_pre, right_gate_pre,
            mask_h,
        )
        # left_proj/right_proj now hold gated & masked values
    else:
        # No mask scenario – cheap PyTorch sigmoid‑gate
        left_gate = left_gate_pre.sigmoid()
        right_gate = right_gate_pre.sigmoid()
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # Release gate tensors early
    del left_gate_pre, right_gate_pre

    # -----------------------------------------------------------------
    # 5️⃣ Core contraction O(N³·H) via batched GEMM
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    = left[b,:, :,h] @ right[b,:, :,h].T
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_bh = left_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N)      # (B*H, N, N)
    right_bh = right_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N)    # (B*H, N, N)

    # batched matrix multiply with transpose on the second operand
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))                  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_bh.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm (FP16), output‑gate sigmoid, and final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    # output gate depends on original normalized x (still in half)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))      # (B, N, N, H)
    out = out * out_gate

    # final projection back to dim C
    out = F.linear(out, to_out_weight.half())    # (B, N, N, C) fp16

    # -----------------------------------------------------------------
    # Return FP32 tensor (compatible with reference)
    # -----------------------------------------------------------------
    return out.float()
```
