Now implementing the kernel:

```
@triton.jit
def batched_gemm_kernel(left_ptr, right_ptr, out_ptr,
                        BATCH, M, N, K,
                        stride_l_batch, stride_l_m, stride_l_k,
                        stride_r_batch, stride_r_k, stride_r_n,
                        stride_o_batch, stride_o_m, stride_o_n,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
   pid_batch = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)
   # offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # mask
   m_valid = offs_m < M
   n_valid = offs_n < N
   # accumulate
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
   # iterate over K
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
       k_valid = k_offset < K
       # Load A (left)
       a_ptr = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_m + k_offset[None, :] * stride_l_k
       a = tl.load(a_ptr,
                   mask=m_valid[:, None] & k_valid[None, :],
                   other=0.0)
       # Load B (right)
       b_ptr = right_ptr + pid_batch * stride_r_batch + k_offset[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
       b = tl.load(b_ptr,
                   mask=k_valid[:, None] & n_valid[None, :],
                   other=0.0)
       # multiply
       acc += tl.dot(a, b, out_dtype=tl.float32)
   # write output
   out_ptrs = out_ptr + pid_batch * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
   # cast to FP16 for store
   out = acc.to(tl.float16)
   tl.store(out_ptrs, out,
            mask=m_valid[:, None] & n_valid[None, :])
```

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # combined batch*hidden index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden from pid
    b = pid // H
    h = pid % H

    # compute i, j offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_ok = k_offset < N

        # load left block
        a_ptr = left_ptr + b*stride_l_b + h*stride_l_h + i[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=i_ok[:, None] & k_ok[None,:], other=0.0)

        # load right block (transposed)
        b_ptr = right_ptr + b*stride_r_b + h*stride_r_h + k_offset[:, None] * stride_r_k + j[None, :] * stride_r_j
        b = tl.load(b_ptr, mask=k_ok[:, None] & j_ok[None, :], other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store
    out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_ok[:, None] & j_ok[None, :])
```

"""
Fast outgoing TriMul (AlphaFold‑3 style) kernel.

The implementation follows the reference TriMul forward pass:
1️⃣ LayerNorm on the input (FP32).
2️⃣ FP16 fused linear projection that simultaneously computes
   left_proj, right_proj, left_gate, right_gate and out_gate.
3️⃣ In‑place sigmoid on the three gate tensors.
4️⃣ Optional mask multiplication (broadcast over the hidden dimension) using a tiny Triton kernel.
5️⃣ Core O(N³·H) update  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
   implemented as a custom batched GEMM in Triton – the batch dimension is (batch*hidden).
6️⃣ LayerNorm on the hidden dimension, gated by out_gate.
7️⃣ Final FP16 linear projection back to the original channel dimension.
8️⃣ Return FP32 (API contract).

The heavy O(N³·H) step is the only operation performed with Triton; all other
operations use PyTorch which already runs at near‑peak performance on an H100.
The Triton GEMM fuses the batch‑wise matrix multiplications without extra
copies, dramatically cutting launch overhead compared to `torch.matmul`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_lr_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    """In‑place multiply left/right tensors by a binary mask."""
    b  = tl.program_id(0)          # batch
    ti = tl.program_id(1)          # i‑tile
    tj = tl.program_id(2)          # j‑tile

    i = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    j = tj * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # load mask tile (broadcast later over H)
    mask = tl.load(
        mask_ptr + b * stride_m_b
                  + i[:, None] * stride_m_i
                  + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                                 # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]          # (BLOCK_I, BLOCK_J, 1)

    num_h = tl.cdiv(H, BLOCK_H)
    for hi in range(num_h):
        cur_h = hi * BLOCK_H + tl.arange(0, BLOCK_H)
        h_ok = cur_h < H

        # ----- left ----------------------------------------------------
        left = tl.load(
            left_ptr + b * stride_l_b
                     + i[:, None, None] * stride_l_i
                     + j[None, :, None] * stride_l_j
                     + cur_h[None, None, :] * stride_l_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left = left * mask
        tl.store(
            left_ptr + b * stride_l_b
                     + i[:, None, None] * stride_l_i
                     + j[None, :, None] * stride_l_j
                     + cur_h[None, None, :] * stride_l_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )

        # ----- right ---------------------------------------------------
        right = tl.load(
            right_ptr + b * stride_r_b
                      + i[:, None, None] * stride_r_i
                      + j[None, :, None] * stride_r_j
                      + cur_h[None, None, :] * stride_r_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right = right * mask
        tl.store(
            right_ptr + b * stride_r_b
                      + i[:, None, None] * stride_r_i
                      + j[None, :, None] * stride_r_j
                      + cur_h[None, None, :] * stride_r_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )


def _apply_mask_lr_twin(left: torch.Tensor,
                        right: torch.Tensor,
                        mask: torch.Tensor) -> None:
    """Launch the mask kernel (in‑place)."""
    B, N, _, H = left.shape
    # ensure mask has the same dtype as left/right (fp16)
    mask = mask.to(dtype=left.dtype)

    BLOCK_I = 128
    BLOCK_J = 128
    BLOCK_H = 64

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    _mask_lr_kernel[grid](
        left,
        right,
        mask,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # tensors are updated in‑place


# ----------------------------------------------------------------------
# Triton kernel – batched matrix multiplication for the TriMul core
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] for all b,i,j,h.
    The batch dimension is flattened as (b * H + h) → program_id(0).
    """
    pid = tl.program_id(0)   # flattened batch*head
    pid_i = tl.program_id(1) # i‑tile
    pid_j = tl.program_id(2) # j‑tile

    # recover original batch and hidden indices
    b = pid // H
    h = pid % H

    # tile offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    # accumulator in fp32 (tensor‑core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_ok = k < N

        # ---------------- left block (i,k) ---------------------------
        a_ptr = left_ptr + b * stride_l_b + h * stride_l_h \
                + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        a = tl.load(a_ptr,
                    mask=i_ok[:, None] & k_ok[None, :],
                    other=0.0)

        # ---------------- right block (k,j) --------------------------
        # note: right is accessed as right[b, j, k, h]
        b_ptr = right_ptr + b * stride_r_b + h * stride_r_h \
                + k[:, None] * stride_r_k + j[None, :] * stride_r_j
        b = tl.load(b_ptr,
                    mask=k_ok[:, None] & j_ok[None, :],
                    other=0.0)

        # fused dot (fp16 * fp16 → fp32 accumulation)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store result (cast back to fp16)
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h \
               + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=i_ok[:, None] & j_ok[None, :])


def _batched_tri_mul(left: torch.Tensor,
                     right: torch.Tensor,
                     hidden_dim: int) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.  left/right are [B,N,N,H] fp16.
    Returns out of the same shape.
    """
    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty_like(left)  # same shape, fp16

    # strides for the three tensors
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # tile sizes – empirically good for H100 up to hidden=384
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] in float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # [B, N, N, C] (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ FP16 fused projection ( five linear branches )
    # ------------------------------------------------------------------
    x_h = x.half()                    # fp16
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)        # (B·N·N, C)

    # concatenate the five weight matrices (bias= False)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16)        # (5·H, C)

    combined = F.linear(x_flat, combined_weight)   # (B·N·N, 5·H)
    combined = combined.view(B, N, N, 5 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 3️⃣ In‑place sigmoid on gates
    # ------------------------------------------------------------------
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates & optional mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate

    if not nomask:
        # mask: [B, N, N]  broadcast over H inside the kernel
        _apply_mask_lr_twin(left, right, mask_tensor)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – fused batched GEMM via Triton
    # ------------------------------------------------------------------
    # left/right are already in layout [B, N, N, H]; the kernel expects that.
    out = _batched_tri_mul(left, right, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + output‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].half())   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Return FP32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
