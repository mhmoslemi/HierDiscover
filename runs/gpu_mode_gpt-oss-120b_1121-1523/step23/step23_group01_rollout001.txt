2️⃣ Fuse the four linear maps (left/right projections + their gates) into a
   single GEMM that produces a tensor of shape (B,N,N,4*H).  Split it back
   into left_proj, right_proj, left_gate_pre, right_gate_pre.
3️⃣ If a mask is present, launch a tiny Triton kernel that fuses:
       * sigmoid on the gate pre‑activations,
       * multiplication by the mask,
       * element‑wise multiplication with the projections.
   The kernel works in‑place on the projected tensors.
   When there is no mask we simply apply the sigmoid gates in PyTorch.
4️⃣ Core O(N³·H) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (L * Rᵀ) performed with torch.matmul,
   which is executed by cuBLAS/Tensor‑Core kernels.
5️⃣ Hidden‑dim LayerNorm + output gate (sigmoid) and final linear projection
   back to the original channel dimension.
6️⃣ Cast the result back to FP32 for compatibility with the reference model.

The only mandatory Triton component is the mask‑gate‑fusion kernel; the rest
uses highly‑optimized PyTorch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid‑gate and element‑wise multiplication
# -----------------------------------------------------------------
@triton.jit
def _mask_gate_mul_kernel(
    # tensor pointers (all fp16)
    left_ptr, right_ptr,                # (B,N,N,H) – will be overwritten
    left_gate_ptr, right_gate_ptr,      # (B,N,N,H) – pre‑gate values
    mask_ptr,                           # (B,N,N)   – fp16 mask (0/1)

    # shapes
    B, N, H,
    # strides
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,

    # tile sizes (compile‑time)
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    In‑place update:
        left[b,i,j,h]  = left_proj[b,i,j,h]  *
                         sigmoid(left_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
        right[b,i,j,h] = right_proj[b,i,j,h] *
                         sigmoid(right_gate_pre[b,i,j,h]) *
                         mask[b,i,j]
    All tensors are fp16.  The kernel tiles over (i,j,h).
    """
    b = tl.program_id(0)           # batch dimension
    pid_i = tl.program_id(1)       # tile index over i
    pid_j = tl.program_id(2)       # tile index over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N

    # Load mask tile (BLOCK_I, BLOCK_J) and broadcast to H
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b
                       + i[:, None] * stride_m_i
                       + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=tl.float16(0.0),
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]        # (BLOCK_I, BLOCK_J, 1)

    h_offset = 0
    while h_offset < H:
        h = h_offset + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # Load projection and gate tensors (BLOCK_I, BLOCK_J, BLOCK_H)
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )
        left_g = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )
        right_g = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )

        # Sigmoid in fp32 for stability, then back to fp16
        left_gate = 1.0 / (1.0 + tl.exp(-left_g.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_g.to(tl.float32)))
        left_gate = left_gate.to(tl.float16)
        right_gate = right_gate.to(tl.float16)

        # Apply mask and gate
        left_out = left * left_gate * mask_tile
        right_out = right * right_gate * mask_tile

        # Store in‑place
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right_out,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        )
        h_offset += BLOCK_H


def _apply_mask_and_gate_triton(left_proj, right_proj,
                                left_gate_pre, right_gate_pre,
                                mask):
    """
    Fuse mask, sigmoid‑gate and element‑wise multiplication using Triton.
    All tensors must be fp16 and have shapes:
        left_proj / right_proj          : (B, N, N, H)
        left_gate_pre / right_gate_pre  : (B, N, N, H)
        mask                            : (B, N, N)
    The kernel updates left_proj and right_proj in‑place.
    """
    B, N, _, H = left_proj.shape
    s_l = left_proj.stride()
    s_r = right_proj.stride()
    s_lg = left_gate_pre.stride()
    s_rg = right_gate_pre.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    _mask_gate_mul_kernel[grid](
        left_proj,
        right_proj,
        left_gate_pre,
        right_gate_pre,
        mask,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left_proj / right_proj now hold gated & masked values


def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys
                * dim            – input channel C (int)
                * hidden_dim     – hidden dimension H (int)
                * nomask         – bool, True ⇒ ignore mask
                * layernorm_eps  – optional epsilon for LayerNorm

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # C
    hidden = config["hidden_dim"]      # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    assert C == dim, f"Channel mismatch: got {C}, expected {dim}"

    # -----------------------------------------------------------------
    # Model weights (FP32, already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) → fp16 for the heavy path
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] float32
    x_h = x.half()  # fp16

    # -----------------------------------------------------------------
    # 2️⃣ Fuse left/right proj + gates into one GEMM (output 4*H)
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_weight, right_proj_weight,
         left_gate_weight, right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    x_flat = x_h.view(-1, dim)                 # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)  # (B·N·N, 4*H)

    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask – fuse mask + sigmoid‑gate via Triton
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # Convert mask to fp16 (0.0 / 1.0)
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_and_gate_triton(
            left_proj, right_proj,
            left_gate_pre, right_gate_pre,
            mask_h,
        )
        # left_proj/right_proj now contain gated & masked values
    else:
        # No mask: only apply sigmoid‑gate (pure PyTorch)
        left_gate = left_gate_pre.sigmoid()
        right_gate = right_gate_pre.sigmoid()
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # Release intermediate gate tensors
    del left_gate_pre, right_gate_pre

    # -----------------------------------------------------------------
    # 4️⃣ Core contraction: batched matmul (L * Rᵀ) over the sequence dim
    # -----------------------------------------------------------------
    # Reshape to (B*H, N, N) for BLAS
    left_mat = left_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H,N,N)
    right_mat = right_proj.permute(0, 3, 1, 2).reshape(B * hidden, N, N) # (B*H,N,N)

    # Multiply left * rightᵀ  → (B*H, N, N)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm + output gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 6️⃣ Final projection back to original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B,N,N,C) fp16

    # -----------------------------------------------------------------
    # Return float32 tensor (compatible with reference code)
    # -----------------------------------------------------------------
    return out.float()
