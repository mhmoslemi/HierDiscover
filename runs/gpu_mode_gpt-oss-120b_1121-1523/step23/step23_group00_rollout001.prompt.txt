import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Highly‑optimized forward‑only implementation of the “outgoing” TriMul operator
(AlphaFold‑3 style).  The algorithm follows the reference PyTorch version but
fuses the mask‑gate step in a custom Triton kernel and keeps all heavy
arithmetic in FP16 (Tensor‑Core friendly).  The core O(N³·H) update is performed
with a single batched GEMM (torch.bmm) on the GPU.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication + sigmoid gating
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute, for each (b,i,j) tile:
        left  = left_proj  * mask * sigmoid(left_gate)
        right = right_proj * mask * sigmoid(right_gate)
    The result overwrites left_proj / right_proj (in‑place).
    """
    b = tl.program_id(0)                 # batch dimension
    pid_i = tl.program_id(1)             # i‑tile
    pid_j = tl.program_id(2)             # j‑tile

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # ------------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – broadcast over H
    # ------------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=i_mask[:, None] & j_mask[None, :],
        other=0.0,
    )[:, :, None]       # shape (BLOCK_I, BLOCK_J, 1)

    # ------------------------------------------------------------------
    # Loop over hidden dimension
    # ------------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_mask = h < H

        # ----- LEFT ----------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        left = left * mask_tile * tl.sigmoid(left_gate)
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
        )

        # ----- RIGHT ---------------------------------------------------
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        right = right * mask_tile * tl.sigmoid(right_gate)
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
        )

        h_start += BLOCK_H


def _apply_mask_gate_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
) -> None:
    """
    In‑place fusion of ``mask * sigmoid(gate)`` with the left/right projections.
    All tensors must be on the same CUDA device and have dtype ``torch.float16``.
    """
    B, N, _, H = left.shape
    # strides for the four tensors (all 4‑D)
    s_l = left.stride()
    s_r = right.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()
    s_m = mask.stride()

    # Choose tile sizes – H is typically a multiple of 8, we use a block of 16
    BLOCK_I = 32
    BLOCK_J = 32
    # Make sure BLOCK_H divides H nicely, otherwise the loop will handle the tail
    BLOCK_H = 16 if H >= 16 else H

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )


# ----------------------------------------------------------------------
# Triton‑enabled TriMul forward (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3 style).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (FP32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float, optional)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``layernorm_eps`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (FP32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # input channel dimension C
    hidden = config["hidden_dim"]       # hidden dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B, N, N, C)  FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the bulk of the computation runs in half precision
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    B, N, _, C = x_h.shape
    M = B * N * N   # flattened batch‑spatial dimension

    # --------------------------------------------------------------
    # 3️⃣ One‑shot linear for left/right projections **and** the four gates
    #    (weight shape: (4*hidden, dim))
    # --------------------------------------------------------------
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)       # (4*H, C)

    # (M, 4*H) result of the concatenated GEMM
    proj_gate = F.linear(x_h.view(M, C), proj_gate_weight)

    # Split the concatenated result
    left_proj   = proj_gate[:, 0            : hidden].view(B, N, N, hidden)
    right_proj  = proj_gate[:, hidden       : 2 * hidden].view(B, N, N, hidden)
    left_gate   = proj_gate[:, 2 * hidden   : 3 * hidden].view(B, N, N, hidden)
    right_gate  = proj_gate[:, 3 * hidden   : 4 * hidden].view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 4️⃣ Out‑gate (independent linear) – also in FP16
    # --------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate = torch.sigmoid(F.linear(x_h.view(M, C), out_gate_weight)).view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 5️⃣ Mask + gate fusion (Triton).  If no mask we fall back to a
    #    simple element‑wise implementation which is equally fast.
    # --------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 and ensure it is contiguous
        mask_fp = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_gate_triton(left_proj, right_proj, left_gate, right_gate, mask_fp)
        left = left_proj
        right = right_proj
    else:
        left = left_proj * torch.sigmoid(left_gate)
        right = right_proj * torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update: batched matrix multiply left @ rightᵀ
    #    (B, H, N, N) → (B*H, N, N) for a single torch.bmm call.
    # --------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # FP16 × FP16 → FP32 accumulation (default cuBLAS behaviour)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32) + element‑wise out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out.dtype, device=device),
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension
    # --------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out = F.linear(out, to_out_weight)   # (B, N, N, C)  FP16

    # --------------------------------------------------------------
    # 9️⃣ Return FP32 tensor (matches reference implementation)
    # --------------------------------------------------------------
    return out.float()
