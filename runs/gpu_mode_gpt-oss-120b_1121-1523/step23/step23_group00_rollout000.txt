"""
Optimized outgoing TriMul implementation (AlphaFold‑3 style).

Algorithm
---------
1. Input LayerNorm (FP32 for numeric safety) → cast to FP16.
2. Single fused GEMM computes **left_proj**, **left_gate**, **right_proj**,
   **right_gate**, **out_gate** in one pass (weights concatenated).
3. If a mask is present, a Triton kernel fuses the mask, sigmoid gating and
   writes the gated **left** and **right** tensors back in‑place.
   Otherwise simple element‑wise multiplication with the sigmoid of the
   gate tensors is used.
4. Core O(N³·H) update is performed with a batched GEMM:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   The batch dimension is flattened as B·H to let cuBLAS handle all matrices
   in a single launch.
5. Output LayerNorm → element‑wise multiplication with the **out_gate**
   → final linear projection back to the original channel dimension.
All heavy arithmetic runs in FP16 (Tensor‑Core friendly) and the only
non‑trivial kernel is the mask‑gate fusion.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask * sigmoid(gate) fused for left and right tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For each (b,i,j) tile compute in‑place:
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
    """
    b = tl.program_id(0)                     # batch axis
    pid_i = tl.program_id(1)                 # i‑tile
    pid_j = tl.program_id(2)                 # j‑tile

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N
    ij_mask = i_valid[:, None] & j_valid[None, :]

    # Load mask tile and broadcast over H
    mask_tile = tl.load(
        mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
        mask=ij_mask,
        other=0.0,
    )[:, :, None]  # shape (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_valid = h < H
        h_mask = i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]

        # ----- LEFT ----------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=h_mask,
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=h_mask,
            other=0.0,
        )
        left = left * mask_tile * tl.sigmoid(left_gate)
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left,
            mask=h_mask,
        )

        # ----- RIGHT ---------------------------------------------------
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=h_mask,
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=h_mask,
            other=0.0,
        )
        right = right * mask_tile * tl.sigmoid(right_gate)
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right,
            mask=h_mask,
        )

        h_start += BLOCK_H


def _apply_mask_gate_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
) -> None:
    """In‑place fuse mask, sigmoid‑gate and multiplication for left/right."""
    B, N, _, H = left.shape
    # Strides are in elements (not bytes) – Torch gives that directly.
    s_l = left.stride()
    s_r = right.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()
    s_m = mask.stride()

    # Tile sizes – tuned for H100 SMEM.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 16 if H >= 16 else H

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – may be None
        - weights      : dict containing all model parameters (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # C
    hidden = config["hidden_dim"]       # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float16)  # (B, N, N, C) in fp16

    B, N, _, C = x.shape
    M = B * N * N  # flattened batch‑spatial dimension

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left_proj, left_gate, right_proj, right_gate, out_gate
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each of shape [hidden, dim])
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)  # shape (5*H, C)

    # Single GEMM → (M, 5*H)
    fused_out = F.linear(x.view(M, C), fused_weight)  # fp16

    # Split and reshape
    split = torch.split(fused_out, hidden, dim=1)  # 5 tensors of shape (M, H)
    left_raw, left_gate_raw, right_raw, right_gate_raw, out_gate_raw = split
    # Reshape to (B, N, N, H)
    left_raw = left_raw.view(B, N, N, hidden)
    left_gate_raw = left_gate_raw.view(B, N, N, hidden)
    right_raw = right_raw.view(B, N, N, hidden)
    right_gate_raw = right_gate_raw.view(B, N, N, hidden)
    out_gate = torch.sigmoid(out_gate_raw.view(B, N, N, hidden))

    # ------------------------------------------------------------------
    # 3️⃣ Mask & gate fusion (or plain element‑wise when no mask)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_gate_triton(left_raw, right_raw, left_gate_raw, right_gate_raw, mask_fp)
        left = left_raw  # now holds gated left
        right = right_raw
    else:
        left = left_raw * torch.sigmoid(left_gate_raw)
        right = right_raw * torch.sigmoid(right_gate_raw)

    # Free gate tensors early
    del left_gate_raw, right_gate_raw, left_raw, right_raw

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update via batched GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N) for cuBLAS batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # (B*H, N, N) × (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore (B, N, N, H)
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out.dtype, device=device),
        eps=eps,
    )
    out = out * out_gate  # element‑wise gating

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out = F.linear(out, to_out_weight)  # (B, N, N, C) in fp16

    # ------------------------------------------------------------------
    # 7️⃣ Return FP32 tensor (matches reference API)
    # ------------------------------------------------------------------
    return out.float()
```
