"""
Highly‑optimized forward‑only implementation of the “outgoing” TriMul
operator (AlphaFold‑3 style).  The implementation fuses
* mask‑application into the input projection,
* sigmoid‑gating of the left/right projections via a tiny Triton kernel,
* and keeps the bulk of the arithmetic in FP16 (Tensor‑Core friendly).
All heavy linear layers are done with a single GEMM per logical group,
so the number of memory passes is minimal.
"""
import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place sigmoid‑gate (mask already applied earlier)
# ----------------------------------------------------------------------
@triton.jit
def _gate_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] projected tensors (in‑place)
    left_gate_ptr, right_gate_ptr,      # [B,N,N,H] gate tensors (sigmoid applied here)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Elementwise:
        left  = left  * sigmoid(left_gate)
        right = right * sigmoid(right_gate)
    one program processes a (BLOCK_I × BLOCK_J) tile of the (i,j) plane
    for a single batch element, looping over the hidden dimension H.
    """
    b = tl.program_id(0)               # batch axis
    pid_i = tl.program_id(1)           # i‑tile axis
    pid_j = tl.program_id(2)           # j‑tile axis

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_valid = i < N
    j_valid = j < N
    ij_mask  = i_valid[:, None] & j_valid[None, :]

    # ------------------------------------------------------------------
    # Loop over H dimension
    # ------------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_valid = h < H

        # ---- LEFT ----------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=ij_mask[:, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + b * stride_lg_b
            + i[:, None, None] * stride_lg_i
            + j[None, :, None] * stride_lg_j
            + h[None, None, :] * stride_lg_h,
            mask=ij_mask[:, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        left = left * tl.sigmoid(left_gate)
        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left,
            mask=ij_mask[:, :, None] & h_valid[None, None, :],
        )

        # ---- RIGHT ---------------------------------------------------
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=ij_mask[:, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr
            + b * stride_rg_b
            + i[:, None, None] * stride_rg_i
            + j[None, :, None] * stride_rg_j
            + h[None, None, :] * stride_rg_h,
            mask=ij_mask[:, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        right = right * tl.sigmoid(right_gate)
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right,
            mask=ij_mask[:, :, None] & h_valid[None, None, :],
        )

        h_start += BLOCK_H


def _apply_gate_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
) -> None:
    """In‑place sigmoid gating of left/right projections (mask already baked in)."""
    B, N, _, H = left.shape
    # strides (all tensors are contiguous in the last dimension)
    s_l = left.stride()
    s_r = right.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    # Choose a block size that divides H nicely; fall back to tail handling.
    BLOCK_H = 32 if H >= 32 else H

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )


# ----------------------------------------------------------------------
# Triton‑enabled TriMul forward (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only “outgoing” TriMul.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (FP32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float)
        - weights      : dict of model parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``layernorm_eps`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (FP32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # input channel dimension C
    hidden = config["hidden_dim"]       # hidden dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 for speed; keeps numerics acceptable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B, N, N, C) – still FP32
    x = x.to(torch.float16)            # switch to half precision

    B, N, _, C = x.shape
    M = B * N * N                       # flattened batch‑spatial dimension

    # --------------------------------------------------------------
    # 2️⃣ Apply mask *before* the left/right projections (mask is broadcast
    #    over the channel dimension).  This avoids a separate mask‑gate
    #    kernel for the projections.
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask_tensor.to(dtype=torch.float16, device=device).unsqueeze(-1)
        x_masked = x * mask_fp
    else:
        x_masked = x

    # --------------------------------------------------------------
    # 3️⃣ Linear projections
    #    - Left/Right projections use the *masked* input.
    #    - Gates (left, right, out) use the original (unmasked) input.
    # --------------------------------------------------------------
    # concatenate left & right projection weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0).to(
        dtype=torch.float16, device=device
    )                         # (2*H, C)

    # concatenate gate weights (left_gate, right_gate, out_gate)
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0).to(
        dtype=torch.float16, device=device
    )                         # (3*H, C)

    # Linear on the masked input -> left & right projections
    proj = F.linear(x_masked.view(M, C), proj_weight)           # (M, 2H)
    left_proj = proj[:, :hidden].view(B, N, N, hidden)
    right_proj = proj[:, hidden: 2 * hidden].view(B, N, N, hidden)

    # Linear on the *unmasked* input -> three gate tensors
    gates = F.linear(x.view(M, C), gate_weight)                # (M, 3H)
    left_gate = gates[:, :hidden].view(B, N, N, hidden)
    right_gate = gates[:, hidden: 2 * hidden].view(B, N, N, hidden)
    out_gate = torch.sigmoid(gates[:, 2 * hidden : 3 * hidden].view(B, N, N, hidden))

    # --------------------------------------------------------------
    # 4️⃣ Fuse sigmoid‑gate for left/right projections via Triton
    # --------------------------------------------------------------
    _apply_gate_triton(left_proj, right_proj, left_gate, right_gate)

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update: batched matmul L @ Rᵀ for each hidden dim
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) → one large batched GEMM
    left_flat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # FP16 × FP16 → FP32 accumulation (Tensor‑Core)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore shape (B, N, N, H)
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (still in FP16, good enough) + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=out.dtype, device=device),
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final projection back to original dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out = F.linear(out, to_out_w)   # (B, N, N, dim) in FP16

    # --------------------------------------------------------------
    # 8️⃣ Return FP32 tensor (matches reference implementation)
    # --------------------------------------------------------------
    return out.float()
