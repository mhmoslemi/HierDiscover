@triton.jit
def _batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                         M, N, K,
                         stride_a_batch, stride_a_m, stride_a_k,
                         stride_b_batch, stride_b_m, stride_b_k,
                         stride_c_batch, stride_c_m, stride_c_n,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for valid indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (BLOCK_M, BLOCK_K)
        a = tl.load(
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile transposed: B^T has shape (K, N)
        # Original B layout: (batch, row=M (i?), col=K (j?) ) but we need B^T. So we swap indices:
        # We load element B[pid_batch, offs_n (col), offs_k (row)] (i.e., B[batch, col?, row?]) using strides stride_b_m for row and stride_b_k for col
        b = tl.load(
            B_ptr
            + pid_batch * stride_b_batch
            + offs_n[None, :] * stride_b_k   # B col stride (original col) corresponds to N dimension? Actually B original shape: (batch, N, N). Here we want B_T[n,k]=B[k,n] essentially.
            + offs_k[:, None] * stride_b_m,
            mask=mask_n[None, :] & mask_k[:, None],
            other=0.0,
        )
        # B loads shape (BLOCK_K, BLOCK_N) after transposition

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store result
    C = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        C,
        mask=mask_m[:, None] & mask_n[None, :],
    )

@triton.jit
def _mask_4d_kernel(
    x_ptr,                # fp16/fp32 *
    mask_ptr,             # same dtype as x *
    B, N, H,              # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,

    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    j0 = pid_j * BLOCK_J

    j_ok = j < N

            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        off_h += BLOCK_H


def _apply_mask_triton_4d(tensor: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place ``tensor *= mask`` where mask is broadcast over the hidden dim."""
    assert tensor.is_cuda and mask.is_cuda, "Both tensors must be on CUDA"
    B, N, _, H = tensor.shape
    mask = mask.to(dtype=tensor.dtype)

    s_x = tensor.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_4d_kernel[grid](
        tensor,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ   (fp16 * fp16 → fp32)
# -------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)     # batch = B * H
    pid_m = tl.program_id(1)        # tile over output rows (i)
    pid_n = tl.program_id(2)        # tile over output cols (j)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    m_ok = offs_m < M
    n_ok = offs_n < N

    # accumulator (float32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        k_ok = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=m_ok[:, None] & k_ok[None, :],
            other=0.0,
        )

        # Bᵀ tile: we read B in transposed order.
        # Original B layout: (batch, N, K) with strides (stride_b_m, stride_b_k)
        # We need element B_T[row = k, col = n] = B[batch, n, k]
        b = tl.load(
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_m      # k‑dimension uses original row stride
            + offs_n[None, :] * stride_b_k,     # n‑dimension uses original col stride
            mask=k_ok[:, None] & n_ok[None, :],
            other=0.0,
        )

        # accumulate dot product over the K‑tile
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=m_ok[:, None] & n_ok[None, :],
    )


def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM ``C = left @ rightᵀ`` for tensors of shape
    (B*H, N, N).  All tensors must be contiguous and half‑precision.
    """
    assert left.is_cuda and right.is_cuda
    batch, M, K = left.shape
    _, N, K2 = right.shape
    assert K == K2, "inner dimensions must match"
    # output tensor
    out = torch.empty((batch, M, N), dtype=left.dtype, device=left.device)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()   # (batch,N,K)
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # tile sizes – tuned for fp16 on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left,
        right,
        out,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# -------------------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Model weights (float32 in the checkpoint)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]          # (H, C)
    right_proj_weight = weights["right_proj.weight"]        # (H, C)
    left_gate_weight = weights["left_gate.weight"]          # (H, C)
    right_gate_weight = weights["right_gate.weight"]        # (H, C)
    out_gate_weight = weights["out_gate.weight"]            # (H, C)
    to_out_norm_weight = weights["to_out_norm.weight"]      # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]          # (H,)
    to_out_weight = weights["to_out.weight"]                # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑Norm the input (float32 → stable)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                                          # [B, N, N, C] – float32

    # -----------------------------------------------------------------
    # 2️⃣ Fuse proj + gate into a single half‑precision GEMM
    # -----------------------------------------------------------------
    x_h = x.half()                                            # fp16 view
    B, N, _, _ = x_h.shape

    # concatenate the four weight matrices (proj/gate) → shape (4·H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=x.device)               # (4·H, C)

    # (B·N·N, C) @ (4·H, C)ᵀ → (B·N·N, 4·H)
    x_flat = x_h.view(-1, dim)                                # (B·N·N, C)
    combined = F.linear(x_flat, combined_weight)              # (B·N·N, 4·H)
    combined = combined.view(B, N, N, 4 * hidden_dim)         # (B, N, N, 4·H)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )                                                         # each (B, N, N, H)

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(dtype=torch.float16)
        _apply_mask_triton_4d(left_proj, mask_h)
        _apply_mask_triton_4d(right_proj, mask_h)

    # -----------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and combine
    # -----------------------------------------------------------------
    left_gate_sig = left_gate.sigmoid()
    right_gate_sig = right_gate.sigmoid()
    left = left_proj * left_gate_sig          # (B, N, N, H)
    right = right_proj * right_gate_sig       # (B, N, N, H)

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM per hidden channel
    # -----------------------------------------------------------------
    # reshape to (B·H, N, N) → each hidden channel becomes a batch element
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)      # (B·H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)    # (B·H, N, N)

    # batched GEMM: left @ rightᵀ   (using a custom Triton kernel)
    out_mat = _batched_gemm_triton(left_mat, right_mat)  # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate and final projection (stay in fp16)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )                                                         # (B, N, N, H)

    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B, N, N, H)
    out = out * out_gate                                            # (B, N, N, H)

    out = F.linear(out, to_out_weight.half())                       # (B, N, N, C) – fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 (as in the reference implementation)
    # -----------------------------------------------------------------
    return out.float()
```
