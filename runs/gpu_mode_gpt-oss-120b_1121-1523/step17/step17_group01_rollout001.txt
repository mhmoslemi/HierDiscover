"""
High‑performance outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the input pair tensor (float32 → stable).
2️⃣ Fuse the four linear branches (left_proj, left_gate, right_proj, right_gate)
   into a single FP16 GEMM that produces a tensor of shape (B, N, N, 4·H).
3️⃣ One Triton kernel per side computes
      out = proj * sigmoid(gate)            (and optional mask)
   This replaces two separate mask‑mul, sigmoid and element‑wise
   multiplication kernels, saving memory traffic.
4️⃣ Core O(N³·H) update: for each hidden channel h,
      out_h = left_h @ right_hᵀ .
   Realised as a batched FP16 GEMM by reshaping to (B·H, N, N).
5️⃣ Output LayerNorm, out‑gate (sigmoid) and final linear projection.
   All heavy work stays in FP16 (Tensor‑Core friendly); the result
   is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# Triton kernel: gated projection + optional mask
# -------------------------------------------------------------------------
@triton.jit
def _gated_proj_kernel(
    proj_ptr,        # FP16 *  – left/right projection (B,N,N,H)
    gate_ptr,        # FP16 *  – left/right gate (B,N,N,H)
    out_ptr,         # FP16 *  – output tensor (B,N,N,H)
    mask_ptr,        # FP16 *  – mask tensor (B,N,N) or dummy
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_out_b,  stride_out_i,  stride_out_j,  stride_out_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    MASKED: tl.constexpr,
):
    """Compute `out = proj * sigmoid(gate)` and optionally apply `* mask`.

    proj/gate/out have shape (B,N,N,H).  mask has shape (B,N,N) and is
    broadcast over H.  All tensors are stored in FP16.
    """
    b = tl.program_id(0)                   # batch index
    pid_i = tl.program_id(1)               # tile over i
    pid_j = tl.program_id(2)               # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load mask once per (i,j) tile if required
    # -----------------------------------------------------------------
    if MASKED:
        mask = tl.load(
            mask_ptr
            + b * stride_mask_b
            + i[:, None] * stride_mask_i
            + j[None, :] * stride_mask_j,
            mask=i_ok[:, None] & j_ok[None, :],
            other=1.0,                     # default = 1 (no masking)
        )                                 # (BLOCK_I, BLOCK_J)
        mask = mask[:, :, None]           # (BLOCK_I, BLOCK_J, 1)
    else:
        mask = None                        # let the compiler prune the block

    # -----------------------------------------------------------------
    # Loop over hidden dimension H in blocks
    # -----------------------------------------------------------------
    off_h = 0
    while off_h < H:
        cur_h = off_h + tl.arange(0, BLOCK_H)
        h_ok = cur_h < H

        # load projection slice
        proj = tl.load(
            proj_ptr
            + b * stride_proj_b
            + i[:, None, None] * stride_proj_i
            + j[None, :, None] * stride_proj_j
            + cur_h[None, None, :] * stride_proj_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # load gate slice
        gate = tl.load(
            gate_ptr
            + b * stride_gate_b
            + i[:, None, None] * stride_gate_i
            + j[None, :, None] * stride_gate_j
            + cur_h[None, None, :] * stride_gate_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )

        # apply mask to the projection (if any)
        if MASKED:
            proj = proj * mask

        # sigmoid gate (stable implementation)
        gate_sig = 1.0 / (1.0 + tl.exp(-gate))

        # gated output
        out = proj * gate_sig

        # store result
        tl.store(
            out_ptr
            + b * stride_out_b
            + i[:, None, None] * stride_out_i
            + j[None, :, None] * stride_out_j
            + cur_h[None, None, :] * stride_out_h,
            out,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        off_h += BLOCK_H


# -------------------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                   # C
    hidden_dim = config["hidden_dim"]     # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # Model weights (float32 → cast to fp16 when used)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]          # (H, C)
    right_proj_weight = weights["right_proj.weight"]        # (H, C)

    left_gate_weight = weights["left_gate.weight"]          # (H, C)
    right_gate_weight = weights["right_gate.weight"]        # (H, C)
    out_gate_weight = weights["out_gate.weight"]            # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]      # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]          # (H,)
    to_out_weight = weights["to_out.weight"]                # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑Norm over channel dimension C (float32 for numeric stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] – float32

    # -----------------------------------------------------------------
    # 2️⃣ Fuse four linear branches into a single FP16 GEMM
    # -----------------------------------------------------------------
    x_h = x.half()                         # FP16 tensor
    B, N, _, _ = x_h.shape

    # Build combined weight: (4·H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)

    # Flatten pair dimension and apply linear
    x_flat = x_h.view(-1, dim)             # (B·N·N, C)
    combined = F.linear(x_flat, combined_weight)               # (B·N·N, 4·H)
    combined = combined.view(B, N, N, 4 * hidden_dim)          # (B, N, N, 4·H)

    # -----------------------------------------------------------------
    # 3️⃣ Split combined tensor into per‑side projection & gate tensors
    # -----------------------------------------------------------------
    proj_left = combined[..., :hidden_dim]                     # (B,N,N,H)
    gate_left = combined[..., hidden_dim:2 * hidden_dim]
    proj_right = combined[..., 2 * hidden_dim:3 * hidden_dim]
    gate_right = combined[..., 3 * hidden_dim:4 * hidden_dim]

    # Allocate output containers for the gated & (optionally) masked tensors
    left = torch.empty_like(proj_left)
    right = torch.empty_like(proj_right)

    # -----------------------------------------------------------------
    # 4️⃣ Apply mask (if any) + sigmoid gating in a fused Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        mask_half = mask_tensor.to(dtype=torch.float16)
        mask_dummy = mask_half                         # real mask
        MASKED = 1
    else:
        # Dummy mask – never read because MASKED==0
        mask_dummy = torch.empty(0, dtype=torch.float16, device=device)
        MASKED = 0

    # block sizes – tuned for H100 Tensor‑cores
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    # left side
    _gated_proj_kernel[grid](
        proj_left,
        gate_left,
        left,
        mask_dummy,
        B,
        N,
        hidden_dim,
        *proj_left.stride(),
        *gate_left.stride(),
        *left.stride(),
        *mask_half.stride() if not nomask else (0, 0, 0),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        MASKED=MASKED,
        num_warps=4,
    )
    # right side
    _gated_proj_kernel[grid](
        proj_right,
        gate_right,
        right,
        mask_dummy,
        B,
        N,
        hidden_dim,
        *proj_right.stride(),
        *gate_right.stride(),
        *right.stride(),
        *mask_half.stride() if not nomask else (0, 0, 0),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        MASKED=MASKED,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM across hidden dimension
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # reshape to (B·H, N, N) for a single torch.bmm call
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))   # (B·H, N, N)

    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final projection (all fp16)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # (B, N, N, H) fp16

    # out‑gate from the *unmasked* normalized input x_h
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))   # (B, N, N, H) fp16
    out = out * out_gate                                              # (B, N, N, H)

    # final linear back to original channel dimension C
    out = F.linear(out, to_out_weight.half())                         # (B, N, N, C) fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (as in the reference implementation)
    # -----------------------------------------------------------------
    return out.float()
