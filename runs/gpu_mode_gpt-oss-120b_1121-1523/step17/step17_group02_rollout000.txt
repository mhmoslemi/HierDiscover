@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)   # batch
    h = tl.program_id(1)   # head
    pid_i = tl.program_id(2)
    pid_j = tl.program_id(3)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = offs_i < N
    j_mask = offs_j < N

    # initialization
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in FP32 for precision

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # load right tile: shape (BLOCK_N, BLOCK_K)
        right = tl.load(
            right_ptr + b * stride_right_b + h * stride_right_h + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k,
            mask=j_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # right is used transposed: we need right[j,k] to multiply left[i,k].
        # It's shape (BLOCK_N, BLOCK_K). The dot product across k:
        acc += tl.dot(left, right, trans_b=True)  # note: tl.dot expects (M,K)*(K,N).
        # But we loaded right as (BLOCK_N, BLOCK_K), we need (K,N) shape. Using trans_b=True to treat right as transposed
        # Even better: use tl.dot(left, right) with transposed right.
        # We'll use tl.dot directly:
        # acc += tl.dot(left, right, trans_b=True)

    # store output (cast to fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j,
        out,
        mask=i_mask[:, None] & j_mask[None, :],
    )

def batched_matmul_triton_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols

    a_ptrs = a_ptr + batch_id * stride_abatch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bbatch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a = tl.load(
            a_ptr + batch_id * stride_abatch + offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak,
            mask=(offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptr + batch_id * stride_bbatch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # multiply-accumulate: a: (M,K), b: (K,N)

    c_ptrs = c_ptr + batch_id * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

        c_ptrs,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),

@triton.jit
def batched_gemm_triton_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a = tl.load(
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # shape (BM, BK)

        # Load B tile (transposed)
        b = tl.load(
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # shape (BK, BN)

        # FP32 accumulate
        acc += tl.dot(a, b)

    # write out
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

"""
High‑performance *outgoing* TriMul (AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the pairwise input (float32 → stable).
2️⃣ Fuse the four linear branches (left_proj, left_gate, right_proj, right_gate)
   into one 4× larger GEMM (float32) and split the result.
3️⃣ (Optional) Apply the pair mask by simple broadcasting (no Triton needed).
4️⃣ Apply sigmoid gates (float32) and multiply.
5️⃣ Cast the gated tensors to fp16 and compute the O(N³·H) update with a
   custom batched‑GEMM Triton kernel (one kernel launch per head‑batch).
   The kernel simply does:
        out[b,i,j] = Σₖ left[b,i,k] * right[b,k,j]   (fp16 × fp16 → fp32 acc)
6️⃣ Output LayerNorm, out‑gate, final linear projection (all fp16).
7️⃣ Cast the result back to float32 (as in the reference implementation).

The custom Triton kernel implements the heavy N³ inner loop and therefore
satisfies the “must contain a Triton kernel” requirement while the surrounding
operations stay in highly‑optimised PyTorch/Tensor‑Core code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the core TriMul contraction
# -------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,  # left  (batch, M, K)   – fp16
    b_ptr,  # right (batch, K, N)   – fp16 (already transposed)
    c_ptr,  # out   (batch, M, N)   – fp16
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Perform a batched GEMM: C[b] = A[b] @ B[b]  (fp16 → fp32 acc)."""
    batch_id = tl.program_id(0)      # batch × head dimension
    pid_m = tl.program_id(1)         # tile row
    pid_n = tl.program_id(2)         # tile col

    # -------------------------------------------------------------
    # Block offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # Accumulator (fp32 for numeric stability)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (already stored as K‑major): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)

    # -------------------------------------------------------------
    # Write back (cast to fp16)
    # -------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # input channel C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # Load model weights (stored as float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C] – float32

    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the four linear branches (float32)
    # -----------------------------------------------------------------
    # Concatenate the four weight matrices (H, C) → (4·H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    )  # (4*H, C), float32

    x_flat = x.view(-1, dim)                   # (B·N·N, C)
    combined = F.linear(x_flat, combined_weight)  # (B·N·N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4H)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each (B, N, N, H), still float32

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask_tensor.to(dtype=left_proj.dtype)  # float32
        left_proj = left_proj * mask_f[..., None]
        right_proj = right_proj * mask_f[..., None]

    # -----------------------------------------------------------------
    # 4️⃣ Sigmoid gates & gated projections (float32)
    # -----------------------------------------------------------------
    left_gate_sig = left_gate.sigmoid()
    right_gate_sig = right_gate.sigmoid()

    left = left_proj * left_gate_sig       # (B, N, N, H) float32
    right = right_proj * right_gate_sig    # (B, N, N, H) float32

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM in fp16 via Triton
    # -----------------------------------------------------------------
    # Cast to fp16 for tensor‑core execution
    left_h = left.half()
    right_h = right.half()

    # Re‑order to (B, H, N, N) – the head dimension becomes part of the batch
    left_h = left_h.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_h = right_h.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N)  (k, j) ordering

    # Collapse batch × head into one dimension for the kernel
    batch_h = B * hidden_dim
    left_h = left_h.view(batch_h, N, N)   # (batch_h, M=N, K=N)
    right_h = right_h.view(batch_h, N, N) # (batch_h, K=N, N=N)
    out_h = torch.empty_like(left_h)      # (batch_h, N, N), fp16

    # Kernel launch configuration
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        N, N, N,
        left_h.stride(0), left_h.stride(1), left_h.stride(2),
        right_h.stride(0), right_h.stride(1), right_h.stride(2),
        out_h.stride(0), out_h.stride(1), out_h.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (B, N, N, H)
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final projection (fp16)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # (B, N, N, H) fp16

    # out‑gate computed from the *unmasked* normalized input
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B, N, N, H) float32
    out_gate = out_gate.half()

    out = out * out_gate                                      # (B, N, N, H) fp16

    # Final linear projection back to original channel dimension
    out = F.linear(out, to_out_weight.half())                 # (B, N, N, C) fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32 (as in the reference implementation)
    # -----------------------------------------------------------------
    return out.float()
```
