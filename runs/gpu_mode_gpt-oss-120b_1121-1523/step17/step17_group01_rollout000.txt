@triton.jit
def _gate_kernel(
    combined_ptr, mask_ptr, left_ptr, right_ptr,
    B, N, H,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    stride_m_b, stride_m_i, stride_m_j,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    # compute block indices
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # compute mask tile
    mask = tl.load(mask_ptr + b * stride_m_b + i[:, None] * stride_m_i + j[None, :] * stride_m_j,
                   mask=i_ok[:, None] & j_ok[None, :],
                   other=1.0)  # fill with 1 if out of range
    mask = mask[:, :, None]  # broadcast to hidden dim

    # compute base offset for combined without hidden offset
    base_comb = b * stride_c_b + i[:, None] * stride_c_i + j[None, :] * stride_c_j

    # compute base offsets for left/right outputs
    base_left = b * stride_l_b + i[:, None] * stride_l_i + j[None, :] * stride_l_j
    base_right = b * stride_r_b + i[:, None] * stride_r_i + j[None, :] * stride_r_j

    off_h = 0
    while off_h < H:
        h = off_h + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # compute hidden offset for combined segments
        # left_proj segment offset = 0
        # left_gate segment offset = H
        # right_proj segment offset = 2*H
        # right_gate segment offset = 3*H

        # indices for combined: base_comb + (segment_offset + h) * stride_c_h
        # Since stride_c_h is stride for the last dimension, usually 1.
        # Simplify by adding segment_offset to h then mul stride.
        # Keep as vector.

        # left_proj
        left_proj = tl.load(combined_ptr + base_comb + (h) * stride_c_h,
                            mask=i_ok[:, None] & j_ok[None, :] & h_ok[None, None],
                            other=0.0)
        # left_gate
        left_gate = tl.load(combined_ptr + base_comb + (h + H) * stride_c_h,
                            mask=i_ok[:, None] & j_ok[None, :] & h_ok[None, None],
                            other=0.0)
        # right_proj
        right_proj = tl.load(combined_ptr + base_comb + (h + 2*H) * stride_c_h,
                             mask=i_ok[:, None] & j_ok[None, :] & h_ok[None, None],
                             other=0.0)
        # right_gate
        right_gate = tl.load(combined_ptr + base_comb + (h + 3*H) * stride_c_h,
                             mask=i_ok[:, None] & j_ok[None, :] & h_ok[None, None],
                             other=0.0)

        # sigmoid
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

        # apply mask and gating
        left_out = left_proj * mask * left_gate_sig
        right_out = right_proj * mask * right_gate_sig

        # store to left and right tensors
        tl.store(left_ptr + base_left + h * stride_l_h,
                 left_out,
                 mask=i_ok[:, None] & j_ok[None, :] & h_ok[None, None])
        tl.store(right_ptr + base_right + h * stride_r_h,
                 right_out,
                 mask=i_ok[:, None] & j_ok[None, :] & h_ok[None, None])

        off_h += BLOCK_H

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)
    device = input_tensor.device
    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # half
    x_h = x.half()
    B,N,N2, C = x_h.shape
    assert N==N2
    # Combined linear
    combined_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"], weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(dtype=torch.float16, device=device)
    x_flat = x_h.view(-1, dim)
    combined_flat = F.linear(x_flat, combined_weight)  # shape (B*N*N, 4*H)
    combined = combined_flat.view(B, N, N, 4*hidden_dim)
    # Allocate left/right output buffers
    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    # Mask
    if nomask:
        mask_h = torch.ones((B,N,N), dtype=torch.float16, device=device)
    else:
        mask_h = mask_tensor.to(dtype=torch.float16)
    # Get strides
    s_comb = combined.stride()
    s_mask = mask_h.stride()
    s_left = left.stride()
    s_right = right.stride()
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _gate_kernel[grid](
        combined,
        mask_h,
        left,
        right,
        B, N, hidden_dim,
        s_comb[0], s_comb[1], s_comb[2], s_comb[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # BMM
    left_perm = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_perm = right.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    out_perm = torch.bmm(left_perm, right_perm.transpose(1,2))   # (B*H,N,N)
    out = out_perm.view(B, hidden_dim, N, N).permute(0,2,3,1)   # (B,N,N,H)

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].half(),
                       bias=weights["to_out_norm.bias"].half(), eps=eps)

    # out_gate linear from x_h
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))  # shape (B,N,N,H)
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"].half())
    return out.float()

"""
High‑performance outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Layer‑Norm the input (float32 → stable).
2️⃣ Fuse the four linear branches (left_proj, left_gate, right_proj, right_gate)
   into a single fp16 GEMM over the flattened (B·N·N, C) matrix.
3️⃣ One Triton kernel: split the fused result, apply the optional pair mask,
   compute sigmoid gates and fuse the gated projections → left, right tensors.
   (All work stays in fp16, mask is broadcast over the hidden dimension.)
4️⃣ Batched fp16 GEMM (B·H, N, N) implements the core O(N³·H) update:
   out_h = left_h @ right_hᵀ.
5️⃣ Layer‑Norm over the hidden dimension, out‑gate, and final linear
   projection back to the original channel dimension C.
6️⃣ Cast the result to float32 (as in the reference implementation).

Only the mask‑gate fusion is done with a custom Triton kernel; all heavy
matrix‑multiplications rely on cuBLAS/Tensor‑Core‑accelerated GEMMs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# Triton kernel: split fused linear output, apply mask, sigmoid gates and
# write the gated left/right tensors (broadcast mask over hidden dim H).
# -------------------------------------------------------------------------
@triton.jit
def _gate_kernel(
    comb_ptr,          # fp16*   – fused linear output (B,N,N,4H)
    mask_ptr,          # fp16*   – mask (B,N,N), broadcast over H
    left_ptr,          # fp16*   – output left tensor (B,N,N,H)
    right_ptr,         # fp16*   – output right tensor (B,N,N,H)
    B, N, H,           # runtime sizes
    # strides of the tensors
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    stride_m_b, stride_m_i, stride_m_j,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # --------------------------------------------------------------
    # 1‑D / 2‑D tile indices
    # --------------------------------------------------------------
    b = tl.program_id(0)                # batch
    pid_i = tl.program_id(1)            # i‑tile
    pid_j = tl.program_id(2)            # j‑tile

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # --------------------------------------------------------------
    # Load mask (scalar per (i,j)), broadcast over H
    # --------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=1.0,                     # if out‑of‑range → no masking
    )                                   # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]            # (BLOCK_I, BLOCK_J, 1)

    # --------------------------------------------------------------
    # Base offsets for the (b,i,j) position (hidden dimension handled later)
    # --------------------------------------------------------------
    idx_i = i[:, None, None]           # (BLOCK_I,1,1)
    idx_j = j[None, :, None]           # (1,BLOCK_J,1)

    base_comb = b * stride_c_b + idx_i * stride_c_i + idx_j * stride_c_j
    base_left = b * stride_l_b + idx_i * stride_l_i + idx_j * stride_l_j
    base_right = b * stride_r_b + idx_i * stride_r_i + idx_j * stride_r_j

    # --------------------------------------------------------------
    # Loop over the hidden dimension in chunks
    # --------------------------------------------------------------
    off_h = 0
    while off_h < H:
        h = off_h + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # ------------------------------
        # Load four slices from the fused tensor
        #   left_proj, left_gate, right_proj, right_gate
        # ------------------------------
        left_proj = tl.load(
            comb_ptr
            + base_comb
            + (h) * stride_c_h,
            mask=i_ok[:, None, None]
                 & j_ok[None, :, None]
                 & h_ok[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            comb_ptr
            + base_comb
            + (h + H) * stride_c_h,
            mask=i_ok[:, None, None]
                 & j_ok[None, :, None]
                 & h_ok[None, None, :],
            other=0.0,
        )
        right_proj = tl.load(
            comb_ptr
            + base_comb
            + (h + 2 * H) * stride_c_h,
            mask=i_ok[:, None, None]
                 & j_ok[None, :, None]
                 & h_ok[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            comb_ptr
            + base_comb
            + (h + 3 * H) * stride_c_h,
            mask=i_ok[:, None, None]
                 & j_ok[None, :, None]
                 & h_ok[None, None, :],
            other=0.0,
        )

        # ------------------------------
        # Sigmoid (1 / (1+exp(-x))) – fp16 friendly
        # ------------------------------
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

        # ------------------------------
        # Apply mask and gates
        # ------------------------------
        left_out = left_proj * mask * left_gate_sig
        right_out = right_proj * mask * right_gate_sig

        # ------------------------------
        # Store gated tensors
        # ------------------------------
        tl.store(
            left_ptr
            + base_left
            + h * stride_l_h,
            left_out,
            mask=i_ok[:, None, None]
                 & j_ok[None, :, None]
                 & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + base_right
            + h * stride_r_h,
            right_out,
            mask=i_ok[:, None, None]
                 & j_ok[None, :, None]
                 & h_ok[None, None, :],
        )

        off_h += BLOCK_H


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the full pair representation (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear branches (fp16 GEMM)
    # ------------------------------------------------------------------
    x_h = x.half()                     # fp16
    B, N, _, _ = x_h.shape

    # combined weight = [left_proj; left_gate; right_proj; right_gate] → (4H, C)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    # linear over the flattened pair dimension
    x_flat = x_h.view(-1, dim)                  # (B·N·N, C)
    combined_flat = F.linear(x_flat, combined_weight)  # (B·N·N, 4·H)
    combined = combined_flat.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4·H)

    # ------------------------------------------------------------------
    # 3️⃣ Triton kernel: split, mask, sigmoid‐gate and write left/right
    # ------------------------------------------------------------------
    # allocate output buffers (fp16)
    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # mask: convert to fp16; if no mask → all‑ones tensor
    if nomask:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask_tensor.to(dtype=torch.float16)

    # strides (int) needed by the Triton kernel
    s_comb = combined.stride()
    s_mask = mask_fp16.stride()
    s_left = left.stride()
    s_right = right.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    _gate_kernel[grid](
        combined,
        mask_fp16,
        left,
        right,
        B,
        N,
        hidden_dim,
        s_comb[0],
        s_comb[1],
        s_comb[2],
        s_comb[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – batched GEMM across hidden dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for a single bmm call
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication (fp16 – TensorCores)
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))   # (B·H, N, N)

    # back to (B, N, N, H)
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, out‑gate, final projection (stay in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )  # (B, N, N, H) fp16

    # out‑gate computed from the *unmasked* normalized input (fp16)
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))  # (B,N,N,H) fp16
    out = out * out_gate

    # final projection back to C
    out = F.linear(out, weights["to_out.weight"].half())                       # (B,N,N,C) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Cast to float32 (as in the reference implementation)
    # ------------------------------------------------------------------
    return out.float()
```
