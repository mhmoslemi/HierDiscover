def _apply_mask_kernel(
    left_ptr, right_ptr, mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_j, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_j, stride_r_h,
    stride_m_batch, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):

    pid_i = tl.program_id(1) # i tile
    pid_j = tl.program_id(2) # j tile

    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_j = offs_j < N

                     + offs_h[None, None, :] * stride_l_h,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right = tl.load(...)
        if HAS_MASK:

                mask_ptr + pid_b * stride_m_batch
                         + offs_i[:, None] * stride_m_i
                         + offs_j[None, :] * stride_m_j,
                mask=mask_i[:, None] & mask_j[None, :],

    in_ptr, out_ptr, mask_ptr,
    B, N, H,

    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    mask_stride_batch, mask_stride_i, mask_stride_j,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

            in_ptr + pid_b * stride_in_batch

                     + offs_j[None, :, None] * stride_in_j

"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm (high‑level)
----------------------
1️⃣ Layer‑norm the input on the channel dimension (FP32).
2️⃣ Cast to FP16 and perform **one** large GEMM that computes
   *left_proj*, *left_gate*, *right_proj*, *right_gate* and *out_gate*
   in a single matmul by concatenating the five weight matrices.
3️⃣ Apply sigmoid to the three gate tensors and multiply the two
   projection tensors by their corresponding gates → *left_gated*,
   *right_gated*.
4️⃣ Optional pair‑wise mask is applied with a tiny Triton kernel
   (identity when mask is absent, otherwise element‑wise scaling).
5️⃣ Reshape to [B·H, N, N] and run a batched FP16 × FP16 → FP32 GEMM
   (Tensor‑Core) to obtain the hidden‑dim output.
6️⃣ Restore the original shape, Layer‑norm over the hidden dimension,
   multiply by *out_gate* and finish with the output linear projection.
   The final tensor is FP32 of shape [B, N, N, C].

The heavy linear algebra (steps 2 and 5) use cuBLAS/Tensor‑Core.
Only the mask‑application (step 4) is written in Triton, satisfying the
requirement that “part of the operations” be a custom kernel while
keeping the overall runtime well below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: apply (optional) pair‑wise mask to the two gated tensors
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    left_in_ptr, right_in_ptr, mask_ptr,          # inputs
    left_out_ptr, right_out_ptr,                  # outputs
    B, N, H,                                      # dimensions
    # strides for the left tensor [B, N, N, H]
    stride_li_b, stride_li_i, stride_li_j, stride_li_h,
    # strides for the right tensor [B, N, N, H]
    stride_ri_b, stride_ri_i, stride_ri_j, stride_ri_h,
    # strides for the output tensors (same layout as inputs)
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    # strides for the mask tensor [B, N, N]
    stride_m_b, stride_m_i, stride_m_j,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)            # batch index
    pid_i = tl.program_id(1)            # tile over first sequence axis
    pid_j = tl.program_id(2)            # tile over second sequence axis

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # iterate over the hidden dimension inside the block
    for off_h in range(0, H, BLOCK_H):
        offs_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ---------- load left and right ----------
        left = tl.load(
            left_in_ptr
            + pid_b * stride_li_b
            + offs_i[:, None, None] * stride_li_i
            + offs_j[None, :, None] * stride_li_j
            + offs_h[None, None, :] * stride_li_h,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_in_ptr
            + pid_b * stride_ri_b
            + offs_i[:, None, None] * stride_ri_i
            + offs_j[None, :, None] * stride_ri_j
            + offs_h[None, None, :] * stride_ri_h,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ---------- optional mask ----------
        if HAS_MASK:
            m = tl.load(
                mask_ptr
                + pid_b * stride_m_b
                + offs_i[:, None] * stride_m_i
                + offs_j[None, :] * stride_m_j,
                mask=mask_i[:, None] & mask_j[None, :],
                other=1.0,
            )
            # broadcast mask over the hidden dimension
            left = left * m[:, :, None]
            right = right * m[:, :, None]

        # ---------- store ----------
        tl.store(
            left_out_ptr
            + pid_b * stride_lo_b
            + offs_i[:, None, None] * stride_lo_i
            + offs_j[None, :, None] * stride_lo_j
            + offs_h[None, None, :] * stride_lo_h,
            left,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )
        tl.store(
            right_out_ptr
            + pid_b * stride_ro_b
            + offs_i[:, None, None] * stride_ro_i
            + offs_j[None, :, None] * stride_ro_j
            + offs_h[None, None, :] * stride_ro_h,
            right,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, N, N, C]  (float32, CUDA)
        - mask_tensor  : torch.Tensor[bs, N, N]    (bool/float, same device)
        - weights_dict : model parameters (see reference implementation)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor[bs, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    dim = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]   # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over channel dimension (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B, N, N, C]  FP32

    # ------------------------------------------------------------------
    # 2) One fused linear for all projections + gates + out_gate
    # ------------------------------------------------------------------
    # Cast once to FP16 (Tensor‑Core friendly)
    x_fp16 = x.half()               # [B, N, N, C]  FP16

    # Assemble the big weight matrix  (5·H × C) and move to the same device
    w_cat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().to(device)             # FP16

    # Linear:  (B·N·N, C) × (5·H, C)^T  → (B, N, N, 5·H)
    proj_all = F.linear(x_fp16, w_cat)   # FP16

    # ------------------------------------------------------------------
    # 3) Split, apply sigmoid gates and compute gated projections
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj_raw   = proj_all[..., 0:H]                     # [B,N,N,H]
    left_gate_raw   = proj_all[..., H:2*H]
    right_proj_raw  = proj_all[..., 2*H:3*H]
    right_gate_raw  = proj_all[..., 3*H:4*H]
    out_gate_raw    = proj_all[..., 4*H:5*H]

    left_gate  = torch.sigmoid(left_gate_raw)   # FP16
    right_gate = torch.sigmoid(right_gate_raw) # FP16
    out_gate   = torch.sigmoid(out_gate_raw)   # FP16

    left_gated = left_proj_raw * left_gate      # FP16
    right_gated = right_proj_raw * right_gate   # FP16

    # ------------------------------------------------------------------
    # 4) Optional mask – Triton kernel (identity when nomask)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(dtype=torch.float16, device=device)   # [B,N,N]
    else:
        # dummy tensor – never read because HAS_MASK == 0
        mask_fp16 = left_gated

    left_masked = torch.empty_like(left_gated)
    right_masked = torch.empty_like(right_gated)

    B, N, _, _ = left_gated.shape

    # Tile sizes – power‑of‑2 for best occupancy
    BLOCK_I = 128 if N >= 128 else 64
    BLOCK_J = BLOCK_I
    BLOCK_H = 32 if hidden_dim >= 32 else hidden_dim

    grid = (B,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J))

    HAS_MASK = 0 if nomask else 1

    _apply_mask_kernel[grid](
        # inputs
        left_gated, right_gated, mask_fp16,
        # outputs
        left_masked, right_masked,
        # dimensions
        B, N, hidden_dim,
        # strides for left input
        left_gated.stride(0), left_gated.stride(1), left_gated.stride(2), left_gated.stride(3),
        # strides for right input
        right_gated.stride(0), right_gated.stride(1), right_gated.stride(2), right_gated.stride(3),
        # strides for left output (identical layout)
        left_masked.stride(0), left_masked.stride(1), left_masked.stride(2), left_masked.stride(3),
        # strides for right output
        right_masked.stride(0), right_masked.stride(1), right_masked.stride(2), right_masked.stride(3),
        # mask strides
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        # compile‑time constants
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 5) Batched GEMM: left * rightᵀ   (Tensor‑Core)
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N]   (contiguous layout required by torch.bmm)
    left = left_masked.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right = right_masked.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # rightᵀ for the inner product over the last axis
    out_fp16 = torch.bmm(left, right.transpose(1, 2))       # FP16 (accumulated in FP32)

    # restore [B, N, N, H] and convert to FP32 for the following LayerNorm
    out = (
        out_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, H]
        .contiguous()
        .float()
    )

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm, output‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()                     # broadcasted over dim

    out = F.linear(out, weights["to_out.weight"])    # back to original dim

    return out
