"""
Highly‑optimized Triton‑aware forward for the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑Norm the input (FP32).
2. Cast to FP16 and perform a *single* fused linear projection that computes
   left_proj, left_gate, right_proj and right_gate in one kernel.
3. In‑place sigmoid on the gate tensors and fuse the gate into the projections:
   left = left_proj * sigmoid(left_gate) ,  right = right_proj * sigmoid(right_gate)
4. (Optional) apply the pairwise mask in‑place with a tiny Triton kernel.
5. Core O(N³) update:
   – view the tensors as (B, H, N, N) (no copy, just a permute) and call
     torch.matmul which internally runs a batched GEMM over the (B·H) matrices.
   – the result is permuted back to (B, N, N, H) and cast to FP32.
6. Output LayerNorm + output‑gate (sigmoid) + final linear projection back to `dim`.

Only the mask‑application is written in Triton; everything else uses
high‑throughput PyTorch kernels (FP16 GEMM, fused linear, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    mask_ptr,               # fp16 *
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Multiply left/right [B,N,N,H] by mask [B,N,N] (broadcast over H)."""
    b = tl.program_id(0)          # batch index
    i_tile = tl.program_id(1)     # i‑tile
    j_tile = tl.program_id(2)     # j‑tile

    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # ----- load mask (scalar per i,j) -----
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # ----- load left tile -----
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
        other=0.0,
    )
    # ----- load right tile -----
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
        other=0.0,
    )

    # ----- apply mask (broadcast over H) -----
    left = left * mask
    right = right * mask

    # ----- store back -----
    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
    )

# Main entry point

def custom_kernel(data):
    """
    Triton‑enhanced forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Load model weights (all stored as float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 3) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 4) Fused projection + gate (FP16)
    # --------------------------------------------------------------
    x_fp16 = x_norm.half()   # cast once

    # concatenate the four weight matrices → (4*H, dim)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=x_fp16.device)

    # Linear produces (B,N,N,4*H) – bias is None
    combined = F.linear(x_fp16, combined_weight)   # (B,N,N,4*H)

    # split into four (B,N,N,H) tensors (views, no copy)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each = fp16

    # in‑place sigmoid for the gate tensors
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()

    # fuse gate into projection (in‑place, re‑using proj buffers)
    left = left_proj.mul_(left_gate)    # left now holds the gated projection
    right = right_proj.mul_(right_gate)  # right holds gated projection

    # --------------------------------------------------------------
    # 5) Optional mask (in‑place via Triton)
    # --------------------------------------------------------------
    if not nomask:
        _apply_mask_triton(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 6) Core O(N³) update – batched GEMM (FP16) without extra copies
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape via permute (no data movement) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)  (k,j) order already matches matmul

    # batched matrix multiplication over the (B·H) batch dimension
    out_perm = torch.matmul(left_perm, right_perm)  # (B, H, N, N) FP16

    # permute back to (B, N, N, H) and cast to FP32 for the following LayerNorm
    out = out_perm.permute(0, 2, 3, 1).contiguous().float()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )   # FP32

    # output gate derived from the *normed* input x_norm (float32)
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))   # (B,N,N,H) FP32
    out = out * out_gate

    # final linear projection back to original dim (float32)
    out = F.linear(out, to_out_weight)   # (B,N,N,dim) FP32

    return out
```
