"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Layer‑norm over the input channel dimension C (FP32) – done with
   ``torch.nn.functional.layer_norm`` (C ≤ 384, reduction is cheap).

2️⃣ Linear projections to the hidden dimension H (no bias) → FP16.
   Gates (sigmoid of a linear projection) are also computed in FP32 and
   cast to FP16.

3️⃣ Fuse the element‑wise multiplication *projection × gate × mask* for
   both the left and the right side in a single Triton kernel.
   The kernel also stores the right side **transposed** so that the later
   batched matrix‑multiply does not need an extra ``transpose`` (no copy).

4️⃣ Batched GEMM: ``torch.bmm(left_gated, right_gated_T)``.
   This uses Tensor‑Core FP16×FP16→FP32 which is the fastest implementation
   on H100.

5️⃣ Reshape back to ``[B, N, N, H]`` (FP32), apply the hidden‑dim LayerNorm,
   multiply by the output gate and finally project back to the original
   channel dimension C.

Only step 3 is written in Triton – the heaviest computation (the N³ GEMM)
leverages cuBLAS/Tensor‑Core, while Triton removes the extra transpose
and fuses mask handling.  The kernel works for any square sequence length
(N) up to 1024 and automatically handles the presence/absence of a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right projections, gates and (optional) mask.
# The right side is written transposed (output index (j,i) for input (i,j)).
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    # input tensors (all B*H × N × N, FP16)
    proj_a_ptr, gate_a_ptr,           # left projection & its gate
    proj_b_ptr, gate_b_ptr,           # right projection & its gate
    mask_ptr,                         # mask [B, N, N] (FP16) – broadcasted across H
    # output tensors
    out_a_ptr, out_b_ptr,             # left gated, right gated (transposed)
    # matrix dimensions
    M, N,
    # strides for left projection / gate
    stride_pa_batch, stride_pa_m, stride_pa_n,
    stride_ga_batch, stride_ga_m, stride_ga_n,
    # strides for right projection / gate
    stride_pb_batch, stride_pb_m, stride_pb_n,
    stride_gb_batch, stride_gb_m, stride_gb_n,
    # output strides (identical layout to inputs)
    stride_oa_batch, stride_oa_m, stride_oa_n,
    stride_ob_batch, stride_ob_m, stride_ob_n,
    # mask strides (original mask layout)
    mask_batch_stride, mask_stride_m, mask_stride_n,
    # compile‑time constants
    H: tl.constexpr,                 # hidden dimension (used for batch→mask mapping)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_bh = tl.program_id(0)   # index over (batch * hidden)
    pid_m  = tl.program_id(1)   # tile row
    pid_n  = tl.program_id(2)   # tile column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds check
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_2d = mask_m[:, None] & mask_n[None, :]

    # ------------------------------------------------------------------
    # Load a tile from each of the four FP16 tensors
    # ------------------------------------------------------------------
    a = tl.load(
        proj_a_ptr + pid_bh * stride_pa_batch
                        + offs_m[:, None] * stride_pa_m
                        + offs_n[None, :] * stride_pa_n,
        mask=mask_2d,
        other=0.0,
    )
    ag = tl.load(
        gate_a_ptr + pid_bh * stride_ga_batch
                        + offs_m[:, None] * stride_ga_m
                        + offs_n[None, :] * stride_ga_n,
        mask=mask_2d,
        other=0.0,
    )
    b = tl.load(
        proj_b_ptr + pid_bh * stride_pb_batch
                        + offs_m[:, None] * stride_pb_m
                        + offs_n[None, :] * stride_pb_n,
        mask=mask_2d,
        other=0.0,
    )
    bg = tl.load(
        gate_b_ptr + pid_bh * stride_gb_batch
                        + offs_m[:, None] * stride_gb_m
                        + offs_n[None, :] * stride_gb_n,
        mask=mask_2d,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # Optional mask (broadcasted across the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        batch_idx = pid_bh // H                     # original batch index
        mask_base = mask_ptr + batch_idx * mask_batch_stride
        m = tl.load(
            mask_base
            + offs_m[:, None] * mask_stride_m
            + offs_n[None, :] * mask_stride_n,
            mask=mask_2d,
            other=0.0,
        )
        a  = a  * m
        b  = b  * m
        ag = ag * 1.0          # gates already loaded, keep as‑is
        bg = bg * 1.0

    # ------------------------------------------------------------------
    # Apply gates (sigmoid already applied in PyTorch before the call)
    # ------------------------------------------------------------------
    left  = a * ag                     # [BLOCK_M, BLOCK_N]
    right = b * bg                     # [BLOCK_M, BLOCK_N]

    # ------------------------------------------------------------------
    # Store left (same orientation)
    # ------------------------------------------------------------------
    tl.store(
        out_a_ptr + pid_bh * stride_oa_batch
                        + offs_m[:, None] * stride_oa_m
                        + offs_n[None, :] * stride_oa_n,
        left,
        mask=mask_2d,
    )

    # ------------------------------------------------------------------
    # Store right **transposed** → rows become offs_n, cols become offs_m
    # ------------------------------------------------------------------
    tl.store(
        out_b_ptr + pid_bh * stride_ob_batch
                        + offs_n[:, None] * stride_ob_m   # row index = original col
                        + offs_m[None, :] * stride_ob_n,  # col index = original row
        right,
        mask=mask_2d,
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, N, N, C]  (float32, on CUDA)
        - mask_tensor  : torch.Tensor[bs, N, N]    (bool/float, optional)
        - weights_dict : dict of model parameters
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor[bs, N, N, C] (float32)
        The processed tensor.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype                     # expect torch.float32

    dim        = config["dim"]          # C
    hidden_dim = config["hidden_dim"]   # H
    eps        = 1e-5
    nomask     = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Layer‑norm over the channel dimension (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  (FP32)

    # --------------------------------------------------------------
    # 2) Linear projections + gates
    # --------------------------------------------------------------
    # Projections to hidden dimension H (no bias).  Cast to FP16 for
    # subsequent GEMM.
    left_proj  = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # Side‑gates (sigmoid).  Compute in FP32 then cast to FP16.
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    # Output gate stays FP32 (applied after hidden‑dim LayerNorm).
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))          # [B, N, N, H] (FP32)

    # --------------------------------------------------------------
    # 3) Merge batch and hidden dimensions → shape (B·H, N, N)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_batch = B * hidden_dim    # B·H

    # Rearrange to [B·H, N, N] (contiguous layout, FP16)
    left_proj = left_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_proj = right_proj.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)

    # --------------------------------------------------------------
    # 4) Prepare mask (FP16) – broadcast across the hidden dimension
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)                     # [B, N, N]
    else:
        # dummy tensor – never dereferenced because HAS_MASK = 0
        mask_fp16 = left_proj

    # --------------------------------------------------------------
    # 5) Triton kernel – fuse projection, gate and (optional) mask.
    #    The kernel also writes the right side transposed.
    # --------------------------------------------------------------
    left_gated  = torch.empty_like(left_proj, dtype=torch.float16, device=device)
    right_gated_t = torch.empty_like(right_proj, dtype=torch.float16, device=device)

    # Block size selection (all test seq_len are multiples of 64)
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128

    grid = (
        total_batch,                # program per (batch * hidden)
        triton.cdiv(N, BLOCK_M),    # rows
        triton.cdiv(N, BLOCK_N),    # cols
    )
    HAS_MASK = 0 if nomask else 1

    _fuse_proj_gate_mask_kernel[grid](
        # input pointers
        proj_a_ptr=left_proj,
        gate_a_ptr=left_gate,
        proj_b_ptr=right_proj,
        gate_b_ptr=right_gate,
        mask_ptr=mask_fp16,
        # output pointers
        out_a_ptr=left_gated,
        out_b_ptr=right_gated_t,
        # matrix sizes
        M=N,
        N=N,
        # strides for left projection / left gate
        stride_pa_batch=left_proj.stride(0), stride_pa_m=left_proj.stride(1), stride_pa_n=left_proj.stride(2),
        stride_ga_batch=left_gate.stride(0), stride_ga_m=left_gate.stride(1), stride_ga_n=left_gate.stride(2),
        # strides for right projection / right gate
        stride_pb_batch=right_proj.stride(0), stride_pb_m=right_proj.stride(1), stride_pb_n=right_proj.stride(2),
        stride_gb_batch=right_gate.stride(0), stride_gb_m=right_gate.stride(1), stride_gb_n=right_gate.stride(2),
        # output strides (identical layout)
        stride_oa_batch=left_gated.stride(0), stride_oa_m=left_gated.stride(1), stride_oa_n=left_gated.stride(2),
        stride_ob_batch=right_gated_t.stride(0), stride_ob_m=right_gated_t.stride(1), stride_ob_n=right_gated_t.stride(2),
        # mask strides (original mask layout)
        mask_batch_stride=mask_fp16.stride(0),
        mask_stride_m=mask_fp16.stride(1),
        mask_stride_n=mask_fp16.stride(2),
        # compile‑time constants
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HAS_MASK=HAS_MASK,
    )

    # --------------------------------------------------------------
    # 6) Batched GEMM (Tensor‑Core) – left_gated @ right_gated_t
    # --------------------------------------------------------------
    # torch.bmm performs FP16×FP16 with FP32 accumulation.
    out_fp16 = torch.bmm(left_gated, right_gated_t)   # [B·H, N, N]

    # --------------------------------------------------------------
    # 7) Restore original shape and finish with hidden‑dim LayerNorm,
    #    output‑gate and final projection.
    # --------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)   # [B, H, N, N]
        .permute(0, 2, 3, 1)                # [B, N, N, H]
        .contiguous()
        .to(dtype)                          # back to FP32
    )

    # Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply output gate (still FP32)
    out = out * out_gate

    # Final linear projection back to channel dimension C
    out = F.linear(out, weights["to_out.weight"])

    return out
