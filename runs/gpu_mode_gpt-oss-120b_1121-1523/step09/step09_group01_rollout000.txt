      # we can reuse same kernel for left/right; but we can implement a simple triton kernel that multiplies x in-place with mask broadcast over dim.
      # We'll create a __apply_mask_input kernel (similar but only one tensor).
  else:
      # no mask

  # Cast to half
  x_half = x.half()

  # Combined weight concatenation as before (do once maybe precomputed)
  combined_weight = torch.cat([...], dim=0).to(dtype=torch.float16)
  # Flatten x for linear
  B, N, _, dim = x.shape
  x_flat = x_half.view(-1, dim)  # (B*N*N, dim)
  combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden_dim)
  combined_out = combined_out.view(B, N, N, 4*hidden_dim)

  # Split
  left_proj, left_gate, right_proj, right_gate = torch.split(combined_out, hidden_dim, dim=-1)

  # Apply gates
  left = left_proj * left_gate.sigmoid()
  right = right_proj * right_gate.sigmoid()

  # Core batched matmul:
  # reshape to (B*hidden_dim, N, N)
  left_mat = left.permute(0, 3, 1, 2).contiguous().reshape(B*hidden_dim, N, N)
  right_mat = right.permute(0, 3, 1, 2).contiguous().reshape(B*hidden_dim, N, N)
  out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
  out_mat = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

  # to_out_norm etc.
  out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

  out_gate = torch.sigmoid(F.linear(x, out_gate_weight)) # x is input after layernorm

  out = out * out_gate
  out = torch.nn.functional.linear(out, to_out_weight)

  return out

"""
Fast “outgoing” TriMul forward pass (AlphaFold‑3 style).

Algorithm (all heavy arithmetic stays in FP16 to exploit H100 Tensor‑Cores)

1️⃣ Layer‑Norm the 4‑D input (FP32).
2️⃣ If a mask is present, fuse it *into* the normalized input with a tiny
   Triton kernel (mask is broadcast over the channel dimension).
3️⃣ Flatten the tensor to (B·N·N, C) and apply one fused linear projection
   (weights of the four branches concatenated).  The result is reshaped
   back to (B,N,N,4·H) and split into left‑proj, left‑gate, right‑proj,
   right‑gate.
4️⃣ Apply sigmoid‑gates (still FP16) → tensors `left` and `right`.
5️⃣ Core O(N³·H) update:
   * reshape to (B·H, N, N) and use a single batched GEMM
     (`torch.bmm`) which runs on Tensor‑Cores (FP16 × FP16 → FP32).
6️⃣ Layer‑Norm over the hidden dimension, multiply by the *output‑gate*
   (computed from the normed input) and a final FP32 linear projection
   back to `dim`.

Only the mask‑application is done with Triton; all other work uses
high‑throughput PyTorch kernels.  The implementation follows the
reference forward pass exactly while dramatically reducing memory traffic
and launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication on a 4‑D tensor
#   x[b,i,j,c] *= mask[b,i,j]      (mask broadcast over channel dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_input_kernel(
    x_ptr,                 # fp32 / fp16 *
    mask_ptr,              # same dtype as x *
    B, N, C,               # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr
):
    b = tl.program_id(0)          # batch
    ti = tl.program_id(1)         # i‑tile
    tj = tl.program_id(2)         # j‑tile

    i_base = ti * BLOCK_I
    j_base = tj * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    c = tl.arange(0, BLOCK_C)

    i_ok = i < N
    j_ok = j < N

    # ----- load mask (scalar per i,j) -----
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                     # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # ----- loop over channel dimension -----
    off = 0
    while off < C:
        cur_c = off + c
        c_ok = cur_c < C

        # load a tile of x
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_c[None, None, :] * stride_x_c,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
            other=0.0,
        )
        # apply mask
        x = x * mask
        # store back
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + cur_c[None, None, :] * stride_x_c,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
        )
        off += BLOCK_C


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place element‑wise multiplication `x *= mask` (mask broadcast over channel)."""
    assert x.is_cuda and mask.is_cuda
    B, N, _, C = x.shape
    # ensure mask has same dtype as x
    mask_cast = mask.to(dtype=x.dtype)

    # strides in elements (not bytes)
    s_x = x.stride()
    s_m = mask_cast.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_C = 128                     # C ≤ 384 → up to 3 iterations

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_input_kernel[grid](
        x,
        mask_cast,
        B,
        N,
        C,
        s_x[0], s_x[1], s_x[2], s_x[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    # x is modified in‑place


# ----------------------------------------------------------------------
# Triton‑aware forward‑only outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights (all float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)
    left_gate_weight = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"] # (H, C)
    out_gate_weight = weights["out_gate.weight"]     # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]         # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                       # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Optional mask – fuse directly into the normalized tensor
    # ------------------------------------------------------------------
    if not nomask:
        # in‑place mask multiplication using Triton
        _apply_mask_triton(x, mask_tensor)

    # ------------------------------------------------------------------
    # 3️⃣ Half‑precision fused projection + gating
    # ------------------------------------------------------------------
    x_h = x.half()                         # fp16

    # Concatenate the four projection weights (order: left_proj, left_gate,
    # right_proj, right_gate) → shape (4*H, C)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16)

    B, N, _, _ = x_h.shape
    # flatten to (B·N·N, C) for a single GEMM
    x_flat = x_h.view(-1, dim)                               # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)            # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)       # (B,N,N,4*H)

    # split into the four branches
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each (B,N,N,H)   fp16

    # sigmoid gates (still fp16) and apply
    left = left_proj * left_gate.sigmoid()   # (B,N,N,H)  fp16
    right = right_proj * right_gate.sigmoid()  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³) update – batched half‑precision GEMM
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for a single bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).to(torch.float32)

    # reshape back to (B, N, N, H)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B,N,N,H)  fp32

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32

    # output gate (computed from the normed *input* x)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,H) fp32
    out = out * out_gate

    # final linear back to original dimension `dim`
    out = F.linear(out, to_out_weight)                      # (B,N,N,dim) fp32

    return out
```
