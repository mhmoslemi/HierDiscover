"""
tri_mul_outgoing.py
-------------------

High‑performance *outgoing* TriMul implementation (AlphaFold‑3 style).

Algorithm
---------
1. **Layer‑Norm** the 4‑D input (FP32).
2. **Fused projection & gating**:
   * Stack the four weight matrices (left_proj, left_gate,
     right_proj, right_gate) → shape (4·H, dim).
   * One half‑precision `F.linear` produces all four tensors.
   * Split, apply a sigmoid to the two gate tensors and multiply.
3. **Mask** (optional) – an in‑place Triton kernel multiplies the
   `left`/`right` tensors by the binary mask, broadcasting over the
   hidden dimension `H`.
4. **Core O(N³) update** – treat each hidden head `h` as an
   independent matrix and compute the Gram matrix
   ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
   This is performed with a single strided‑batched GEMM
   (`torch.bmm`), which fully exploits Tensor‑Cores on H100.
5. **Output LayerNorm**, gating (`out_gate = sigmoid(out_gate(x))`)
   and final linear projection back to the original `dim` (all FP32).

All heavy arithmetic runs in FP16; only the two LayerNorms and the
final projection use FP32, which gives the best trade‑off between
speed and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    mask_ptr,               # fp16 *
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Multiply left/right [B,N,N,H] by mask [B,N,N] (broadcast over H)."""
    b = tl.program_id(0)         # batch dimension
    i_tile = tl.program_id(1)    # i‑tile
    j_tile = tl.program_id(2)    # j‑tile

    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # ----- load mask (scalar per i,j) -----
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # ----- left tile -----
    left = tl.load(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
        other=0.0,
    )
    # ----- right tile -----
    right = tl.load(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
        other=0.0,
    )

    # ----- apply mask (broadcast over H) -----
    left = left * mask
    right = right * mask

    # ----- store back -----
    tl.store(
        left_ptr
        + b * stride_l_b
        + i[:, None, None] * stride_l_i
        + j[None, :, None] * stride_l_j
        + h[None, None, :] * stride_l_h,
        left,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
    )
    tl.store(
        right_ptr
        + b * stride_r_b
        + i[:, None, None] * stride_r_i
        + j[None, :, None] * stride_r_j
        + h[None, None, :] * stride_r_h,
        right,
        mask=i_valid[:, None, None]
        & j_valid[None, :, None]
        & h_valid[None, None, :],
    )


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place multiplication of ``left`` and ``right`` by ``mask`` using Triton.
    ``mask`` is broadcast over the hidden dimension.
    """
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    B, N, _, H = left.shape
    # Cast mask to half for the kernel (Triton works with fp16 floats)
    mask_h = mask.to(dtype=left.dtype)

    s_l = left.stride()
    s_r = right.stride()
    s_m = mask_h.stride()

    # Tile sizes – can be tuned; 8×8 tiles fit well into shared memory.
    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_H = 64          # H ≤ 128, 64 gives good occupancy; kernel masks excess H.

            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

        left,
        right,
        mask_h,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Load model weights (all stored as float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]        # (H, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]          # (H, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]    # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]        # (H,)
    to_out_weight = weights["to_out.weight"]              # (dim, H)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 3) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 4) Fused projection + gate (half‑precision)
    # --------------------------------------------------------------
    x_fp16 = x_norm.half()  # cast once

    # Concatenate the four (H, dim) weight matrices → (4·H, dim)
    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    ).to(dtype=torch.float16, device=device)

    # Linear produces (B, N, N, 4·H)
    combined = F.linear(x_fp16, combined_weight)  # bias=None

    # Split back into the four tensors
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: [B, N, N, H] (fp16)

    # Apply sigmoid gates
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # --------------------------------------------------------------
    # 5) Optional mask (in‑place via Triton)
    # --------------------------------------------------------------
    if not nomask:
        # ``mask_tensor`` may be bool or float – cast to fp16 inside kernel
        _apply_mask_triton(left, right, mask_tensor)

    # --------------------------------------------------------------
    # 6) Core O(N³) update – batched GEMM (FP16 → FP16)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B·H, N, N) for a strided‑batched GEMM.
    # left: (B, N, N, H) -> (B, H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right: (B, N, N, H) -> (B, H, N, N) and then transpose the last two axes
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Batched matrix multiplication – fully leverages Tensor‑Cores.
    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N)   fp16

    # Reshape back to [B, N, N, H] and promote to FP32 for the following LayerNorm.
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # FP32

    # Output gate (sigmoid) derived from the *normed* input x_norm (FP32)
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))  # [B,N,N,H] FP32
    out = out * out_gate

    # Final linear projection back to original dimensionality.
    out = F.linear(out, to_out_weight)  # [B,N,N,dim] FP32

    return out
```
