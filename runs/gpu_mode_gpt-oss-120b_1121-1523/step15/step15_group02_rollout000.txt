High‑performance outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D pair tensor (float32 → FP16 afterwards).
2️⃣  Optional mask‑multiply (broadcast over the channel dimension) – performed

3️⃣  Fuse the four linear branches (left‑proj, left‑gate, right‑proj, right‑gate)

4️⃣  Core O(N³·H) update – a batched matrix multiplication

5️⃣  Output LayerNorm, per‑position gating, and a final linear back to the
    original dimension (all in FP16).  The result is cast back to float32.

The implementation works for any batch size B∈{1,2}, sequence length N,
input dimension C (dim) and hidden dimension H (hidden_dim).  All tensors are
expected to already reside on the CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# 1️⃣  Triton kernel for in‑place mask multiplication (broadcast over C)
# -----------------------------------------------------------------------
@triton.jit
def _mask_input_kernel(
    x_ptr,                # ptr to input tensor [B,N,N,C]
    mask_ptr,             # ptr to mask tensor [B,N,N]
    B, N, C,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr,
):
    b      = tl.program_id(0)               # batch dimension
    pid_i  = tl.program_id(1)               # tile over i
    pid_j  = tl.program_id(2)               # tile over j

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load mask (scalar) and broadcast over channel dimension C
    # -----------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                     # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # -----------------------------------------------------------------
    # Loop over channel tiles
    # -----------------------------------------------------------------
    off_c = 0
    while off_c < C:
        c = off_c + tl.arange(0, BLOCK_C)
        c_ok = c < C

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + c[None, None, :] * stride_x_c,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
            other=0.0,
        )
        x = x * mask
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + c[None, None, :] * stride_x_c,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & c_ok[None, None, :],
        )
        off_c += BLOCK_C


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """In‑place x *= mask (mask broadcast over the channel dimension)."""
    assert x.is_cuda and mask.is_cuda, "Both tensors must be CUDA tensors"
    B, N, _, C = x.shape
    mask = mask.to(dtype=x.dtype)

    # strides in element units
    s_x = x.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_C = 64

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    _mask_input_kernel[grid](
        x,
        mask,
        B,
        N,
        C,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    # x is modified in‑place


# -----------------------------------------------------------------------
# 2️⃣  Custom Triton‑based forward (outgoing TriMul)
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ---------------------------------
    # Unpack inputs
    # ---------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # C
    hidden = config["hidden_dim"]      # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ---------------------------------
    # Load weights
    # ---------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    left_gate_weight  = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"]  # (H, C)
    out_gate_weight   = weights["out_gate.weight"]    # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (H,)
    to_out_weight      = weights["to_out.weight"]       # (C, H)

    # ---------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ---------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  float32

    # ---------------------------------
    # 2️⃣ Optional mask (in‑place)
    # ---------------------------------
    if not nomask and mask_tensor is not None:
        _apply_mask_triton(x, mask_tensor)   # modifies x in‑place

    # ---------------------------------
    # 3️⃣ Cast to FP16 and fuse four linear branches
    # ---------------------------------
    x_h = x.half()                     # (B, N, N, C)

    # Concatenate four weight matrices (proj + gate) → (4*H, C)
    fused_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                     # (B*N*N, C)

    # Single GEMM produces proj & gate pre‑activations for left/right
    combined = F.linear(x_flat, fused_weight)      # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden)  # (B, N, N, 4*H)

    left_proj, left_gate_pre, right_proj, right_gate_pre = torch.split(
        combined, hidden, dim=-1
    )   # each (B, N, N, H)

    # Apply sigmoid gates (still FP16)
    left_gate  = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()

    left  = left_proj * left_gate
    right = right_proj * right_gate   # (B, N, N, H) FP16

    # ---------------------------------
    # 4️⃣ Core O(N³·H) update – batched GEMM via cuBLAS (torch.bmm)
    # ---------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B*H, i, k)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B*H, j, k)

    # Compute left @ rightᵀ for each head batch
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))                    # (B*H, N, N) FP16

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()      # (B, N, N, H)

    # ---------------------------------
    # 5️⃣ Output LayerNorm, gating, final projection (stay in FP16)
    # ---------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # (B, N, N, H) FP16

    # out_gate = sigmoid(linear(x_h, out_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))            # (B, N, N, H) FP16
    out = out * out_gate

    # Final linear back to original channel dimension C
    out = F.linear(out, to_out_weight.half())                                 # (B, N, N, C) FP16

    # Return as float32 to match reference implementation
    return out.float()
```
