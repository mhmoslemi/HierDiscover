```
@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # combined batch*hidden
    pid_i = tl.program_id(1) # tile over i dimension
    pid_j = tl.program_id(2) # tile over j dimension

    b = pid0 // H
    h = pid0 % H

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # coordinate offsets
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over tiles of k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)

        k_mask = k_offset < N

        # load A tile
        # A at (b,i,k,h) shape (BLOCK_I,BLOCK_K)
        a_ptrs = left_ptr + b*stride_l_b + i[:,None]*stride_l_i + k_offset[None,:]*stride_l_k + h*stride_l_h
        a = tl.load(a_ptrs, mask=i_mask[:,None] & k_mask[None,:], other=0.0).to(tl.float32)

        # load B_T tile (right transposed)
        # B at (b,k,j,h) shape (BLOCK_K,BLOCK_J)
        b_ptrs = right_ptr + b*stride_r_b + k_offset[:,None]*stride_r_k + j[None,:]*stride_r_j + h*stride_r_h
        b = tl.load(b_ptrs, mask=k_mask[:,None] & j_mask[None,:], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # write output
    out_ptrs = out_ptr + b*stride_o_b + i[:,None]*stride_o_i + j[None,:]*stride_o_j + h*stride_o_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:,None] & j_mask[None,:])
```

def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0) # combined batch*hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid0 // H
    h = pid0 % H

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile
        a_ptrs = left_ptr + b*stride_l_b + offs_i[:,None]*stride_l_i + offs_k[None,:]*stride_l_k + h*stride_l_h
        a = tl.load(a_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float32)

        # Load right tile transposed
        b_ptrs = right_ptr + b*stride_r_b + offs_k[:,None]*stride_r_k + offs_j[None,:]*stride_r_j + h*stride_r_h

        acc += tl.dot(a, b)

    # Write out
    out_ptrs = out_ptr + b*stride_o_b + offs_i[:,None]*stride_o_i + offs_j[None,:]*stride_o_j + h*stride_o_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:,None] & mask_j[None,:])

"""
High‑performance outgoing TriMul (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Layer‑Norm the input 4‑D tensor (float32 → fp16 for the rest).
2️⃣ Fuse the four linear branches (left_proj, left_gate, right_proj, right_gate)
   into a single FP16 GEMM (torch.nn.functional.linear).
3️⃣ Optional mask‑multiplication (broadcast over the hidden dimension) using
   a simple PyTorch element‑wise op (the heavy work is done in Triton).
4️⃣ Apply sigmoid gates → gated left/right tensors (still FP16).
5️⃣ Core O(N³·H) update: a custom Triton kernel performs a batched GEMM
   (left[b,i,k,h] * right[b,j,k,h] summed over k) without any transposes
   or extra copies.  The kernel iterates over the reduction axis K in tiles
   and accumulates in float32 before storing the result as FP16.
6️⃣ Output LayerNorm (FP16), out‑gate and final projection back to the
   original channel dimension (FP16).
7️⃣ Cast the result back to float32 (as in the reference implementation).

Only step 5 is implemented in Triton – this satisfies the requirement of
using a Triton kernel while keeping the rest of the pipeline in highly
optimized PyTorch operators.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul core update
# -------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for a single (batch, hidden) pair:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel tiles the i and j dimensions (BLOCK_I × BLOCK_J) and reduces
    over k in blocks of size BLOCK_K.
    """
    pid0 = tl.program_id(0)          # combined batch * hidden index
    pid_i = tl.program_id(1)          # tile over i dimension
    pid_j = tl.program_id(2)          # tile over j dimension

    # decode batch and hidden index
    b = pid0 // H
    h = pid0 % H

    # start offsets for the current tile
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # range of i / j indices in this tile
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load a tile of left: shape (BLOCK_I, BLOCK_K) ----------
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------- load a tile of right (transposed on the fly) ----------
        # right is indexed as right[b, k, j, h] → we treat k as rows,
        # j as columns.
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
            + h * stride_r_h
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------- accumulate the blockwise dot product --------------------
        acc += tl.dot(a, b)

    # -------- write the result ------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel that computes the contraction
    Σ_k left[..., i, k, h] * right[..., j, k, h] → out[..., i, j, h].

    Args:
        left:  Tensor of shape [B, N, N, H] (FP16)
        right: Tensor of shape [B, N, N, H] (FP16)

    Returns:
        out: Tensor of shape [B, N, N, H] (FP16)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides are in elements (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # Choose block sizes – tuned for fp16 Tensor Cores on H100
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        B * H,                                     # batch * hidden
        (N + BLOCK_I - 1) // BLOCK_I,              # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,              # j‑tiles
    )
    _batched_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# -------------------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------ unpack -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------ load weights (float32) ---------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------ 1️⃣ Input LayerNorm (float32) -------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, C], float32

    # ------------ 2️⃣ Fuse four linear branches (FP16) ------------------
    x_h = x.half()                                   # [B, N, N, C] fp16
    B, N, _, _ = x_h.shape

    # concatenate weights for left_proj, left_gate, right_proj, right_gate
    fused_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)        # (4*H, C)

    # linear over the flattened pair dimension
    x_flat = x_h.view(-1, dim)                       # (B*N*N, C)
    fused_out = F.linear(x_flat, fused_weight)      # (B*N*N, 4*H)
    fused_out = fused_out.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4*H)

    # split back into the four branches
    left_proj, left_gate, right_proj, right_gate = torch.split(
        fused_out, hidden_dim, dim=-1
    )   # each is (B, N, N, H)

    # ------------ 3️⃣ Optional mask (broadcast over H) -----------------
    if not nomask:
        # mask shape: [B, N, N]; broadcast to [B, N, N, 1]
        mask = mask_tensor.to(dtype=left_proj.dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------ 4️⃣ Apply sigmoid gates --------------------------------
    left_gate_sig = left_gate.sigmoid()
    right_gate_sig = right_gate.sigmoid()

    left = left_proj * left_gate_sig          # (B, N, N, H) fp16
    right = right_proj * right_gate_sig       # (B, N, N, H) fp16

    # ------------ 5️⃣ Core O(N³·H) update via Triton --------------------
    out = _batched_gemm(left, right)         # (B, N, N, H) fp16

    # ------------ 6️⃣ Output LayerNorm, out‑gate, final projection ------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    # out_gate is computed from the *unmasked* normalized input (x_h)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B, N, N, H) fp16
    out = out * out_gate

    out = F.linear(out, to_out_weight.half())  # (B, N, N, C) fp16

    # ------------ 7️⃣ Cast back to float32 -------------------------------
    return out.float()
