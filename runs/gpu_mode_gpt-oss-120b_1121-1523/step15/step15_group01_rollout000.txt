```
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    # batch/head index
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # tile start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # compute tile ranges
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along K
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k0 = kk * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # Load left tile (M,K)
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i_range[:, None] * stride_l_i
            + k_range[None, :] * stride_l_k
            + h * stride_l_h,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )  # shape (M,K)

        # Load right tile transposed (K,N)
        right_T = tl.load(
            right_ptr
            + b * stride_r_b
            + k_range[:, None] * stride_r_k
            + j_range[None, :] * stride_r_i
            + h * stride_r_h,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )  # shape (K,N)

        if HAS_MASK:
            # Load mask for left (M,K) and right_T (K,N)
            mask_left = tl.load(
                mask_ptr
                + b * stride_m_b
                + i_range[:, None] * stride_m_i
                + k_range[None, :] * stride_m_j,
                mask=i_mask[:, None] & k_mask[None, :],
                other=1.0,
            )
            left = left * mask_left

            mask_right_T = tl.load(
                mask_ptr
                + b * stride_m_b
                + k_range[:, None] * stride_m_j
                + j_range[None, :] * stride_m_i,
                mask=k_mask[:, None] & j_mask[None, :],
                other=1.0,
            )
            right_T = right_T * mask_right_T

        # dot product accumulation (FP32)
        acc += tl.dot(left, right_T)

    # store result
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + b * stride_o_b
        + i_range[:, None] * stride_o_i
        + j_range[None, :] * stride_o_j
        + h * stride_o_h,
        out,
        mask=i_mask[:, None] & j_mask[None, :],
    )
```

Note: The mask values are 0 or 1 (fp16). When mask is missing, we skip loads.

    B, N, _, H = left.shape
    out = torch.empty_like(left) # same dtype and device

    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

"""
Optimised “outgoing” TriMul (AlphaFold‑3) forward pass.
----------------------------------------------------
1️⃣ Layer‑Norm the 4‑D input in FP32.
2️⃣ Fuse the five linear branches (proj‑, gate‑weights) into a single
   half‑precision GEMM.  The result is split into left/right projections,
   left/right gates and the output gate.
3️⃣ Apply the sigmoid gates.
4️⃣ Core O(N³·H) update – a custom Triton kernel computes
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   (optionally multiplied by a binary mask).  The kernel works on
   tiles (BLOCK_M×BLOCK_N) and accumulates with FP32 precision,
   eliminating the costly permute‑+‑contiguous copies required by a
   batched‑gemm implementation.
5️⃣ Layer‑Norm over the hidden dimension, scale by the output gate.
6️⃣ Final half‑precision projection back to the model dimension.
7️⃣ Return FP32 output.

The Triton kernel fuses the mask‑multiplication (if present) with the
dot‑product, removing a separate memory‑bound mask kernel and the
intermediate copies introduced by the previous torch‑bmm approach.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – core O(N³·H) update (mask optional)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # fp16 [B, N, N, H]
    right_ptr,         # fp16 [B, N, N, H]
    out_ptr,           # fp16 [B, N, N, H]
    mask_ptr,          # fp16 [B, N, N] (only used when HAS_MASK=1)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    # --------------------------------------------------------------
    # 0️⃣ Program IDs  (grid = (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N)))
    # --------------------------------------------------------------
    pid = tl.program_id(0)                # merged batch‑head dimension
    b = pid // H                          # batch index
    h = pid % H                           # hidden‑dim index
    pid_i = tl.program_id(1)              # i‑tile
    pid_j = tl.program_id(2)              # j‑tile

    # --------------------------------------------------------------
    # 1️⃣ Tile offsets
    # --------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    i_mask = i_range < N
    j_mask = j_range < N

    # --------------------------------------------------------------
    # 2️⃣ Accumulator (FP32 for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3️⃣ Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        k0 = k_tile * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ----- left tile (i, k) -------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i_range[:, None] * stride_l_i
            + k_range[None, :] * stride_l_k
            + h * stride_l_h,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )  # (BLOCK_M, BLOCK_K)

        # ----- right tile transposed (k, j) ------------------------------------
        right_T = tl.load(
            right_ptr
            + b * stride_r_b
            + k_range[:, None] * stride_r_k
            + j_range[None, :] * stride_r_i
            + h * stride_r_h,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N)

        # ----- optional mask multiplication ------------------------------------
        if HAS_MASK:
            # mask for left (i, k)
            mask_left = tl.load(
                mask_ptr
                + b * stride_m_b
                + i_range[:, None] * stride_m_i
                + k_range[None, :] * stride_m_j,
                mask=i_mask[:, None] & k_mask[None, :],
                other=1.0,
            )
            left = left * mask_left

            # mask for right (k, j) – note the transpose
            mask_right_T = tl.load(
                mask_ptr
                + b * stride_m_b
                + k_range[:, None] * stride_m_j
                + j_range[None, :] * stride_m_i,
                mask=k_mask[:, None] & j_mask[None, :],
                other=1.0,
            )
            right_T = right_T * mask_right_T

        # ----- accumulate -------------------------------------------------------
        acc += tl.dot(left, right_T)   # FP32 accumulation

    # --------------------------------------------------------------
    # 4️⃣ Store result (cast back to fp16)
    # --------------------------------------------------------------
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + b * stride_o_b
        + i_range[:, None] * stride_o_i
        + j_range[None, :] * stride_o_j
        + h * stride_o_h,
        out,
        mask=i_mask[:, None] & j_mask[None, :],
    )


def _trimul_outgoing(left: torch.Tensor,
                     right: torch.Tensor,
                     mask: torch.Tensor | None,
                     nomask: bool) -> torch.Tensor:
    """
    Triton implementation of the TriMul core update.
    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape [B, N, N, H] and dtype torch.float16.
    mask : torch.Tensor | None
        Optional mask of shape [B, N, N] (float16 0/1). Ignored when ``nomask`` is True.
    nomask : bool
        If True the kernel runs without loading the mask (faster).
    Returns
    -------
    out : torch.Tensor
        Tensor [B, N, N, H] (float16) containing the raw update.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # fp16 output

    # Strides in **elements** (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    if nomask:
        # dummy mask – never read because HAS_MASK=0
        mask_ptr = torch.empty((1, 1, 1),
                               dtype=left.dtype,
                               device=left.device)
        s_m = mask_ptr.stride()
        HAS_MASK = 0
    else:
        assert mask is not None, "Mask tensor must be supplied when nomask=False"
        mask_ptr = mask.to(dtype=left.dtype, device=left.device)
        s_m = mask_ptr.stride()
        HAS_MASK = 1

    # Tile sizes – tuned for H100 and hidden dims up to 384
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # merged batch‑head dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N           # j‑tiles
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        mask_ptr,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=HAS_MASK,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Forward‑only outgoing TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Config & hyper‑parameters
    # ------------------------------------------------------------------
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear branches (proj & gates) – half precision
    # ------------------------------------------------------------------
    x_h = x.half()                                          # fp16
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                               # (B·N·N, C)

    # Concatenate the five weight matrices (no bias). Shape: (5·H, C)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(dtype=torch.float16)

    # Single GEMM -> (B·N·N, 5·H)
    combined = F.linear(x_flat, combined_weight)            # fp16
    combined = combined.view(B, N, N, 5 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()
    out_gate = out_gate.sigmoid()                           # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – custom Triton kernel (mask fused)
    # ------------------------------------------------------------------
    out = _trimul_outgoing(left, right, mask_tensor, nomask)   # fp16, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].half(),   # (C, H)
        bias=None,
    )   # (B, N, N, C) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Return FP32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
