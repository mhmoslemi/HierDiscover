```python
    # flatten to (B*N*N, C)
    x_flat = x_h.view(-1, C)

    proj = torch.nn.functional.linear(x_flat, proj_weight)  # (B*N*N, 2*hidden)
    gate = torch.nn.functional.linear(x_flat, gate_weight)  # (B*N*N, 2*hidden)

    proj = proj.view(B, N, N, 2*hidden)
    gate = gate.view(B, N, N, 2*hidden)

    left_proj, right_proj = torch.split(proj, hidden, dim=-1)  # each (B,N,N,hidden)
    left_gate, right_gate = torch.split(gate, hidden, dim=-1)  # each (B,N,N,hidden)

    # 3. optional mask multiply left_proj and right_proj
    if not nomask:
        # mask shape (B,N,N)
        # broadcast mask over hidden dim; we will use Triton kernel
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=left_proj.device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)

    # 4. apply sigmoid gates and elementwise multiply
    left_gate_sig = left_gate.sigmoid()
    right_gate_sig = right_gate.sigmoid()
    left = left_proj * left_gate_sig
    right = right_proj * right_gate_sig

    # 5. core batched matmul using torch.bmm
    # shape to (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    # right transpose on last two dims:
    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))
    # reshape back
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 6. Output normalization
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )

    # 7. out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_weight.half()))  # (B,N,N,hidden)
    out = out * out_gate

    # 8. Final linear
    out = torch.nn.functional.linear(out, to_out_weight.half())  # (B,N,N,dim)

    # Return float32
    return out.float()
```

import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over H)
# -------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    x_ptr, mask_ptr,
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """x[b,i,j,h] *= mask[b,i,j] (mask broadcast over h)."""
    b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # -----------------------------------------------------------------
    # Load mask tile (shape BLOCK_I x BLOCK_J) and broadcast over H
    # -----------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                               # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # -----------------------------------------------------------------
    # Loop over the channel dimension H
    # -----------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j

# Outgoing TriMul – forward‑only implementation
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul (AlphaFold‑3 style).

    The implementation fuses the four linear branches, applies the optional
    mask with a tiny Triton kernel, and performs the costly O(N³·H) update
    with a batched torch.bmm (cuBLAS/Tensor‑Core) call.  All heavy arithmetic
    runs in FP16, only the two LayerNorms are performed in float32 for
    numerical stability.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input channel C
    hidden = config["hidden_dim"]     # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Model weights (stored as float32, cast to FP16 when used)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → stable)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the remaining heavy work
    # -----------------------------------------------------------------
    x_h = x.half()                       # [B, N, N, C] half
    B, N, _, C = x_h.shape

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections + gates (fused per‑branch)
    # -----------------------------------------------------------------
    #   proj: left_proj || right_proj    → shape (B, N, N, 2*H)
    #   gate: left_gate  || right_gate   → shape (B, N, N, 2*H)
    proj_weight = torch.cat([left_proj_weight, right_proj_weight], dim=0).to(
        dtype=torch.float16, device=device
    )  # (2*H, C)
    gate_weight = torch.cat([left_gate_weight, right_gate_weight], dim=0).to(
        dtype=torch.float16, device=device
    )  # (2*H, C)

    x_flat = x_h.view(-1, C)                               # (B·N·N, C)

    proj = F.linear(x_flat, proj_weight)                    # (B·N·N, 2*H)
    gate = F.linear(x_flat, gate_weight)                    # (B·N·N, 2*H)

    proj = proj.view(B, N, N, 2 * hidden)
    gate = gate.view(B, N, N, 2 * hidden)

    left_proj, right_proj = torch.split(proj, hidden, dim=-1)   # each (B,N,N,H)
    left_gate, right_gate = torch.split(gate, hidden, dim=-1)   # each (B,N,N,H)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask – applied to the projected tensors only
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N); broadcast over H
        mask_fp = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp)
        _apply_mask_triton(right_proj, mask_fp)

    # -----------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # -----------------------------------------------------------------
    left = left_proj * left_gate.sigmoid()
    right = right_proj * right_gate.sigmoid()

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update – batched matrix multiplication
    #     left[b, i, k, h] @ right[b, j, k, h]ᵀ   for each head h
    # -----------------------------------------------------------------
    #   reshape to (B*H, N, N) and use torch.bmm (cuBLAS/Tensor‑Core)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)  # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))                # (B·H, N, N) FP16
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()    # (B, N, N, H) FP16

    # -----------------------------------------------------------------
    # 7️⃣ Output LayerNorm (still FP16) and write‑out gating
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )

    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))   # (B,N,N,H) FP16
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight.half())   # (B, N, N, C) FP16

    # -----------------------------------------------------------------
    # Return in float32 to match the reference implementation
    # -----------------------------------------------------------------
    return out.float()
