```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
High‑performance outgoing TriMul (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D pair tensor (float32 → stable).
2️⃣ Fuse the four linear branches (left‑proj, left‑gate, right‑proj, right‑gate)
   into a single half‑precision GEMM.
3️⃣ If a mask is present, multiply *only* the projected tensors (left‑proj &
   right‑proj) by the mask (broadcast over the hidden dimension) using a tiny
   Triton kernel – this respects the original “gate‑then‑mask” order.
4️⃣ Apply sigmoid gates and obtain the gated left/right tensors (half‑precision).
5️⃣ Core O(N³·H) update: treat each hidden slice as an independent matrix
   multiplication and perform all of them with a single batched half‑precision
   cuBLAS gemm (torch.bmm).
6️⃣ Output LayerNorm (half‑precision), element‑wise out‑gate and final
   projection back to the original channel dimension.
7️⃣ Return the result in float32 (as in the reference implementation).

Only the mask‑multiplication step is implemented in Triton; the rest of the
forward pass uses fused PyTorch kernels (half‑precision linear layers,
batched matrix multiplication, etc.) which are already highly optimized on
H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication on a 4‑D tensor
# -------------------------------------------------------------------------
@triton.jit
def _mask_4d_kernel(
    x_ptr,               # fp16/fp32 *
    mask_ptr,            # same dtype as x *
    B, N, H,             # runtime sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """x[b,i,j,h] *= mask[b,i,j]  (mask broadcast over hidden dimension H)"""
    b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load mask (scalar per (i,j)) and broadcast over H
    mask = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask = mask[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    off_h = 0
    while off_h < H:
        h = off_h + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # Load a tile of x
        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # Apply mask (broadcast)
        x = x * mask
        # Store back
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        off_h += BLOCK_H


def _apply_mask_triton_4d(tensor: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place multiplication ``tensor *= mask`` where `mask` is broadcast over the
    hidden dimension.  Both tensors must reside on the same CUDA device and have
    matching batch/sequence dimensions.  The operation is performed by the
    Triton kernel above.
    """
    assert tensor.is_cuda and mask.is_cuda, "Both tensors must be on CUDA"
    B, N, _, H = tensor.shape
    # Ensure mask has same dtype as tensor for the kernel
    mask = mask.to(dtype=tensor.dtype)

    s_x = tensor.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_4d_kernel[grid](
        tensor,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # tensor is modified in‑place


# -------------------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden_dim = config["hidden_dim"]       # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Weights (stored as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear branches (half‑precision)
    # ------------------------------------------------------------------
    x_h = x.half()   # (B, N, N, C) in fp16

    # Concatenate the four weight matrices: [left_proj, left_gate, right_proj, right_gate]
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)  # (4*H, C)

    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                     # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)
    combined = combined.view(B, N, N, 4 * hidden_dim)  # (B, N, N, 4*H)

    # Split the fused result
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )   # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask on the *projected* tensors (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; cast to half for the kernel
        mask_half = mask_tensor.to(dtype=torch.float16)
        _apply_mask_triton_4d(left_proj, mask_half)
        _apply_mask_triton_4d(right_proj, mask_half)

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and obtain gated tensors
    # ------------------------------------------------------------------
    left_gate_sig = left_gate.sigmoid()
    right_gate_sig = right_gate.sigmoid()
    left = left_proj * left_gate_sig   # (B, N, N, H) fp16
    right = right_proj * right_gate_sig

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    # Reshape for batched GEMM: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating, final projection (stay in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )  # fp16

    # out_gate is computed from the *unmasked* normalized input (per the reference)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B, N, N, H) fp16
    out = out * out_gate

    out = F.linear(out, to_out_weight.half())   # (B, N, N, C) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (as in original implementation)
    # ------------------------------------------------------------------
    return out.float()
```
