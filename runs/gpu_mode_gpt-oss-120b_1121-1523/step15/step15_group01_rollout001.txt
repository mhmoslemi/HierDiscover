def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)
    # 1. layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # 2. cast to half
    x_h = x.to(torch.float16)
    B, N, _, _ = x_h.shape
    # flatten for linear
    x_flat = x_h.view(-1, dim)  # (B*N*N, dim)
    # prepare combined weight
    combined_weight = torch.cat([weights["left_proj.weight"],
                                weights["left_gate.weight"],
                                weights["right_proj.weight"],
                                weights["right_gate.weight"],
                                weights["out_gate.weight"]], dim=0).to(dtype=torch.float16, device=device)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, 5*hidden_dim)  # (B,N,N,5H)
    # Allocate output buffers
    left = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    out_gate = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    # Prepare mask
    if nomask:
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        if mask_tensor is None:
            raise ValueError("mask_tensor must be provided when nomask=False")
        mask = mask_tensor.to(dtype=torch.float16, device=device)
    # Launch kernel
    BLOCK_I = 64
    BLOCK_J = 64
    grid = (B, (N + BLOCK_I - 1)//BLOCK_I, (N + BLOCK_J - 1)//BLOCK_J)
    s_comb = combined.stride()
    s_left = left.stride()
    s_right = right.stride()
    s_outgate = out_gate.stride()
    s_mask = mask.stride()
    _trimul_fwd_kernel[grid](
        combined, mask, left, right, out_gate,
        B, N, hidden_dim,
        s_comb[0], s_comb[1], s_comb[2], s_comb[3],
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_outgate[0], s_outgate[1], s_outgate[2], s_outgate[3],
        s_mask[0], s_mask[1], s_mask[2],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        num_warps=8,
    )
    # core matmul
    left_view = left.view(B*hidden_dim, N, N)
    right_view = right.view(B*hidden_dim, N, N)
    out_mat = torch.bmm(left_view, right_view)   # (B*H, N, N) in fp16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)
    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    # multiply by out_gate
    out = out * out_gate
    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())
    # return fp32
    return out.float()

"""
High‑performance outgoing TriMul (AlphaFold‑3 style) – Triton + PyTorch.

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input in FP32.
2️⃣  Cast to FP16 and fuse the five linear projections (proj‑/gate‑weights) into one GEMM.
3️⃣  Fuse mask‑multiplication, sigmoid‑gates and the required transposition of the
    “right” tensor in a custom Triton kernel. The kernel writes three tensors:
        * left  :  [B, H, N, N]   (will be used as‑is for batched GEMM)
        * right :  [B, H, N, N]   (already transposed → (k, j) order)
        * out_gate : [B, N, N, H] (sigmoid(out_gate) for the final scaling)
4️⃣  Batched GEMM (B·H independent matmuls) implements the O(N³·H) core.
5️⃣  Layer‑Norm over the hidden dimension, multiply by out_gate and project back to C.
6️⃣  Return FP32 tensor of shape [B, N, N, C].

Only the mask‑gate‑transpose step is performed in a Triton kernel – all other
operations use highly‑optimised torch kernels (cuBLAS, layer_norm, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, gates and the (i ↔ j) transpose for “right”
# ----------------------------------------------------------------------
@triton.jit
def _trimul_fwd_kernel(
    combined_ptr, mask_ptr,     # input tensors
    left_ptr, right_ptr, out_gate_ptr,  # output tensors
    B, N, H,                    # runtime sizes
    # strides for the combined tensor (B,N,N,5H)
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    # strides for left  tensor (B,H,N,N)
    stride_l_b, stride_l_h, stride_l_i, stride_l_j,
    # strides for right tensor (B,H,N,N) – will hold transposed data
    stride_r_b, stride_r_h, stride_r_i, stride_r_j,
    # strides for out_gate tensor (B,N,N,H)
    stride_og_b, stride_og_i, stride_og_j, stride_og_h,
    # strides for mask tensor (B,N,N)
    stride_m_b, stride_m_i, stride_m_j,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr, NUM_H: tl.constexpr,
):
    """
    Compute:
        left[b,h,i,k]      = (proj_left * sigmoid(gate_left)) * mask[b,i,k]
        right[b,h,k,j]     = (proj_right* sigmoid(gate_right))* mask[b,j,k]
    and
        out_gate[b,i,j,h]  = sigmoid(out_gate_raw)
    The right tensor is written with the (i ↔ j) transpose already applied.
    """
    b = tl.program_id(0)                 # batch index
    ti = tl.program_id(1)                # i‑tile
    tj = tl.program_id(2)                # j‑tile

    # Tile coordinates in the N×N plane
    i = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    j = tj * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N
    ij_mask = i_ok[:, None] & j_ok[None, :]

    # ------------------------------------------------------------------
    # Load mask tile (broadcast later over H)
    # ------------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=ij_mask,
        other=1.0,
    )                                     # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Offsets that are constant for the whole hidden‑loop
    # ------------------------------------------------------------------
    # Base offset for the (i,j) plane inside the combined tensor
    base_comb = (
        b * stride_c_b
        + i[:, None] * stride_c_i
        + j[None, :] * stride_c_j
    )
    # Base offset for the out_gate tensor (no hidden dim yet)
    base_outgate = (
        b * stride_og_b
        + i[:, None] * stride_og_i
        + j[None, :] * stride_og_j
    )

    # ------------------------------------------------------------------
    # Helper constants for group offsets inside the last dimension
    # ------------------------------------------------------------------
    hidden_stride = stride_c_h
    off_lproj = 0
    off_lgate = H
    off_rproj = 2 * H
    off_rgate = 3 * H
    off_outgate = 4 * H

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in blocks of size BLOCK_H (here BLOCK_H=1)
    # ------------------------------------------------------------------
    for hi in range(NUM_H):
        cur_h = hi * BLOCK_H + tl.arange(0, BLOCK_H)   # scalar when BLOCK_H==1
        # ------------------------------------------------------------------
        # Load the five raw slices for this hidden index
        # ------------------------------------------------------------------
        lp = tl.load(
            combined_ptr + base_comb + (off_lproj + cur_h) * hidden_stride,
            mask=ij_mask,
            other=0.0,
        )
        lg = tl.load(
            combined_ptr + base_comb + (off_lgate + cur_h) * hidden_stride,
            mask=ij_mask,
            other=0.0,
        )
        rp = tl.load(
            combined_ptr + base_comb + (off_rproj + cur_h) * hidden_stride,
            mask=ij_mask,
            other=0.0,
        )
        rg = tl.load(
            combined_ptr + base_comb + (off_rgate + cur_h) * hidden_stride,
            mask=ij_mask,
            other=0.0,
        )
        og_raw = tl.load(
            combined_ptr + base_comb + (off_outgate + cur_h) * hidden_stride,
            mask=ij_mask,
            other=0.0,
        )
        # ------------------------------------------------------------------
        # Sigmoid (exact, same as torch.sigmoid)
        # ------------------------------------------------------------------
        sig_lg = 1.0 / (1.0 + tl.exp(-lg))
        sig_rg = 1.0 / (1.0 + tl.exp(-rg))
        sig_og = 1.0 / (1.0 + tl.exp(-og_raw))

        # ------------------------------------------------------------------
        # Apply gates and mask
        # ------------------------------------------------------------------
        left_val  = lp * sig_lg * mask_tile
        right_val = rp * sig_rg * mask_tile

        # ------------------------------------------------------------------
        # Write left  -> (B, H, N, N)  at (b, cur_h, i, j)
        # Write right -> (B, H, N, N)  at (b, cur_h, j, i)   (i↔j transposed)
        # Write out_gate -> (B, N, N, H) at (b, i, j, cur_h)
        # ------------------------------------------------------------------
        tl.store(
            left_ptr
            + b * stride_l_b
            + cur_h * stride_l_h
            + i[:, None] * stride_l_i
            + j[None, :] * stride_l_j,
            left_val,
            mask=ij_mask,
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + cur_h * stride_r_h
            + j[:, None] * stride_r_i     # note swapped i/j
            + i[None, :] * stride_r_j,
            right_val,
            mask=ij_mask,
        )
        tl.store(
            out_gate_ptr
            + base_outgate
            + cur_h * stride_og_h,
            sig_og,
            mask=ij_mask,
        )


# ----------------------------------------------------------------------
# Helper to launch the fused Triton kernel
# ----------------------------------------------------------------------
def _apply_fused_proj_gate_mask(
    combined: torch.Tensor,
    mask: torch.Tensor,
    hidden_dim: int,
    device: torch.device,
):
    """Return left, right (already transposed) and out_gate tensors."""
    B, N, _, _ = combined.shape
    left = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    out_gate = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Tile size – 64×64 works well for N up to 1024 on H100.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 1                     # one hidden channel per inner iteration
    NUM_H = hidden_dim // BLOCK_H    # compile‑time constant

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    s_c = combined.stride()
    s_l = left.stride()
    s_r = right.stride()
    s_og = out_gate.stride()
    s_m = mask.stride()

    _trimul_fwd_kernel[grid](
        combined,
        mask,
        left,
        right,
        out_gate,
        B,
        N,
        hidden_dim,
        s_c[0],
        s_c[1],
        s_c[2],
        s_c[3],
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_og[0],
        s_og[1],
        s_og[2],
        s_og[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        NUM_H=NUM_H,
        num_warps=8,
    )
    return left, right, out_gate


# ----------------------------------------------------------------------
# Forward‑only outgoing TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Config & device
    # ------------------------------------------------------------------
    device = input_tensor.device
    dim = config["dim"]                     # input channel dim C
    hidden_dim = config["hidden_dim"]       # projection dim H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  (fp32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and fuse the five linear branches
    # ------------------------------------------------------------------
    x_h = x.half()
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                 # (B·N·N, C)

    # Concatenate the five weight matrices – shape (5·H, C)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # Single GEMM → (B·N·N, 5·H)
    combined = F.linear(x_flat, combined_weight)          # fp16
    combined = combined.view(B, N, N, 5 * hidden_dim)     # (B, N, N, 5H)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask, gates and transpose with a Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        # all‑ones mask – cheap to allocate
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        if mask_tensor is None:
            raise ValueError("mask_tensor must be provided when nomask=False")
        mask = mask_tensor.to(dtype=torch.float16, device=device)

    left, right, out_gate = _apply_fused_proj_gate_mask(
        combined,
        mask,
        hidden_dim,
        device,
    )
    # --------------------------------------------------------------
    # 4️⃣ Core O(N³·H) update – batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat = left.view(B * hidden_dim, N, N)    # (B·H, N, N)
    right_mat = right.view(B * hidden_dim, N, N)  # (B·H, N, N) – already (k, j)
    out_mat = torch.bmm(left_mat, right_mat)     # fp16, (B·H, N, N)

    # Reshape back to (B, N, N, H)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
    )   # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to the original channel dimension
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].half(),
        bias=None,
    )   # (B, N, N, C)  fp16

    # --------------------------------------------------------------
    # 7️⃣ Return FP32 (API contract)
    # --------------------------------------------------------------
    return out.float()
