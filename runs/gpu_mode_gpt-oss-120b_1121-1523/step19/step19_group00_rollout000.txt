import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_mul_two_kernel(left_ptr, right_ptr, mask_ptr,
                         B, N, H,
                         stride_left_b, stride_left_i, stride_left_j, stride_left_h,
                         stride_right_b, stride_right_i, stride_right_j, stride_right_h,
                         stride_mask_b, stride_mask_i, stride_mask_j,
                         BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    """
    left[b,i,j,h] *= mask[b,i,j] ; right[b,i,j,h] *= mask[b,i,j]
    """
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load mask tile
    mask = tl.load(mask_ptr + b * stride_mask_b
                   + i[:, None] * stride_mask_i
                   + j[None, :] * stride_mask_j,
                   mask=i_ok[:, None] & j_ok[None, :],
                   other=0.0)  # (BLOCK_I, BLOCK_J)
    # broadcast to hidden
    mask = mask[:, :, None]  # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        # left load
        left = tl.load(left_ptr + b * stride_left_b
                       + i[:, None, None] * stride_left_i
                       + j[None, :, None] * stride_left_j
                       + h[None, None, :] * stride_left_h,
                       mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
                       other=0.0)
        # right load
        right = tl.load(right_ptr + b * stride_right_b
                        + i[:, None, None] * stride_right_i
                        + j[None, :, None] * stride_right_j
                        + h[None, None, :] * stride_right_h,
                        mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
                        other=0.0)

        # apply mask
        left = left * mask
        right = right * mask

        # store back
        tl.store(left_ptr + b * stride_left_b
                 + i[:, None, None] * stride_left_i
                 + j[None, :, None] * stride_left_j
                 + h[None, None, :] * stride_left_h,
                 left,
                 mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :])
        tl.store(right_ptr + b * stride_right_b
                 + i[:, None, None] * stride_right_i
                 + j[None, :, None] * stride_right_j
                 + h[None, None, :] * stride_right_h,
                 right,
                 mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :])

        h_start += BLOCK_H

def _apply_mask_two_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place broadcast multiplication of a (B,N,N) mask onto the hidden dimension
    of `left` and `right` tensors (both [B,N,N,H] half).
    """
    B, N, _, H = left.shape
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

        mask,
        B,
        N,
        H,

        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # -----------------------------------------------------------------

    norm_bias = weights["norm.bias"]

    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (C, H)

    # -----------------------------------------------------------------

                     (dim,),

    # -----------------------------------------------------------------

    right_gate = right_gate_pre.sigmoid()

    right_t = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)    # (B,N,N,H) fp16

"""
Outgoing TriMul (AlphaFold‑3) – forward only.
Algorithm
---------
1. Layer‑norm the input (float32) then cast to fp16.
2. Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into one matmul producing a tensor of shape (B·N·N, 4·H).
3. Optionally apply the pair‑wise mask in‑place to the left/right projected
   tensors with a tiny Triton kernel that broadcasts the mask over the hidden
   dimension.
4. Apply sigmoid gates to the projected tensors.
5. Core O(N³·H) update: for each head h compute `left_h @ right_hᵀ`.  This is
   realized as a batched GEMM (`torch.bmm`) on tensors of shape
   (B·H, N, N), which fully exploits the H100 tensor cores.
6. Hidden‑dim layer‑norm, output‑gate (sigmoid) and final projection back to
   the original channel dimension.
7. Return a float‑32 tensor of shape [B, N, N, dim].
Only the mask‐multiplication step is written in Triton; the remaining
operations are performed with highly‑optimized PyTorch kernels in half‑precision.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask over the hidden dimension and multiply
# ----------------------------------------------------------------------
@triton.jit
def _mask_mul_two_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    left[b,i,j,h] *= mask[b,i,j];  right[b,i,j,h] *= mask[b,i,j]
    """
    b = tl.program_id(0)               # batch dimension
    pid_i = tl.program_id(1)           # tile over i
    pid_j = tl.program_id(2)           # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # ------------------------------------------------------------------
    # Load a (BLOCK_I, BLOCK_J) tile of the mask and broadcast
    # ------------------------------------------------------------------
    mask_tile = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i[:, None] * stride_mask_i
        + j[None, :] * stride_mask_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension
    # ------------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        # apply mask
        left = left * mask_tile
        right = right * mask_tile
        # store back
        tl.store(
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + j[None, :, None] * stride_left_j
            + h[None, None, :] * stride_left_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_right_b
            + i[:, None, None] * stride_right_i
            + j[None, :, None] * stride_right_j
            + h[None, None, :] * stride_right_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_two_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    In‑place multiplication of a (B,N,N) mask onto the hidden dimension of
    `left` and `right` tensors (both [B,N,N,H] half).  The mask is broadcasted
    over H.
    """
    B, N, _, H = left.shape
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_mul_two_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # tensors are modified in place; nothing to return
