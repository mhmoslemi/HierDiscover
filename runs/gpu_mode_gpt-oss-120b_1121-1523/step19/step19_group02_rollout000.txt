    B, N, H,

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: broadcast a mask over the hidden dimension and apply it in‑place
# -------------------------------------------------------------------------
@triton.jit
def _mask_mul_two_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """left[b,i,j,h] *= mask[b,i,j];  right[b,i,j,h] *= mask[b,i,j]"""
    b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load a block of the mask (shape BLOCK_I × BLOCK_J)
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]               # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        left = left * mask_tile
        right = right * mask_tile

        tl.store(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + j[None, :, None] * stride_l_j
            + h[None, None, :] * stride_l_h,
            left,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        tl.store(
            right_ptr
            + b * stride_r_b
            + i[:, None, None] * stride_r_i
            + j[None, :, None] * stride_r_j
            + h[None, None, :] * stride_r_h,
            right,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_two_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """In‑place broadcast mask multiplication for left and right tensors."""
    B, N, _, H = left.shape
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _mask_mul_two_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_m[0], s_m[1], s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # left/right are modified in‑place


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM with implicit transpose on the right operand
#    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the right matrix is transposed on‑the‑fly."""
    bh_idx = tl.program_id(0)   # flattened batch * hidden index
    pid_i = tl.program_id(1)    # row tile
    pid_j = tl.program_id(2)    # col tile

    # Recover batch b and hidden head h from the flattened index
    hidden = H
    b = bh_idx // hidden
    h = bh_idx % hidden

    # Base offsets for this (b,h) slice
    off_l = b * stride_l_b + h * stride_l_h
    off_r = b * stride_r_b + h * stride_r_h
    off_o = b * stride_o_b + h * stride_o_h

    # Tile start indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_ok = i < N
    j_ok = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)
        k_ok = cur_k < N

        # left[i, k]
        left = tl.load(
            left_ptr
            + off_l
            + i[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k,
            mask=i_ok[:, None] & k_ok[None, :],
            other=0.0,
        )
        # right[j, k]  --> we need (k, j) layout for the dot product
        right = tl.load(
            right_ptr
            + off_r
            + cur_k[:, None] * stride_r_k
            + j[None, :] * stride_r_i,
            mask=k_ok[:, None] & j_ok[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

        k += BLOCK_K

    # Cast back to FP16 and write out
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + off_o
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j,
        out,
        mask=i_ok[:, None] & j_ok[None, :],
    )


# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be None
            - weights     : dict of model parameters (float32)
            - config      : dict with keys
                * dim          – input channel C
                * hidden_dim   – hidden dimension H
                * nomask       – True ⇒ ignore mask
                * layernorm_eps (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # C
    hidden = config["hidden_dim"]      # H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, C = input_tensor.shape

    # -----------------------------------------------------------------
    # Load weights (already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight   = weights["left_proj.weight"]   # (H, C)
    right_proj_weight  = weights["right_proj.weight"]  # (H, C)

    left_gate_weight   = weights["left_gate.weight"]   # (H, C)
    right_gate_weight  = weights["right_gate.weight"]  # (H, C)
    out_gate_weight    = weights["out_gate.weight"]    # (H, C)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (H,)
    to_out_weight      = weights["to_out.weight"]       # (C, H)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) + cast to FP16
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x_h = x.half()                     # [B, N, N, C]  fp16

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection + gating (single matmul → 4·H outputs)
    # -----------------------------------------------------------------
    # Stack the four projection matrices
    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*H, C)

    # Linear projection (B·N·N, 4·H)
    x_flat = x_h.view(-1, C)                     # (B·N·N, C)
    fused_out = F.linear(x_flat, fused_weight)  # (B·N·N, 4·H)

    # Split back to the four tensors and reshape to (B, N, N, H)
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused_out, hidden, dim=-1
    )
    left_proj = left_proj.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    left_gate_pre = left_gate_pre.view(B, N, N, hidden)
    right_gate_pre = right_gate_pre.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask – in‑place multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 (0/1) and broadcast
        mask_h = mask_tensor.to(dtype=torch.float16, device=device)
        _apply_mask_two_triton(left_proj, right_proj, mask_h)

    # -----------------------------------------------------------------
    # 4️⃣ Apply sigmoid gating
    # -----------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    left = left_proj * left_gate       # (B, N, N, H) fp16
    right = right_proj * right_gate    # (B, N, N, H) fp16

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³·H) update – batched GEMM with implicit transpose
    # -----------------------------------------------------------------
    out = torch.empty_like(left)       # allocate output (B, N, N, H) fp16

    # Strides for the kernel (all in elements, not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # Choose tile sizes – these work well on H100 with fp16 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 prefers 8‑warps per block for fp16 Tensor‑Core work
        num_stages=2,
    )
    # out now holds the result of Σₖ left * right (still fp16)

    # -----------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm, output‑gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.half(),
        bias=to_out_norm_bias.half(),
        eps=eps,
    )
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))  # (B,N,N,H) fp16
    out = out * out_gate
    out = F.linear(out, to_out_weight.half())      # (B, N, N, C) fp16

    # -----------------------------------------------------------------
    # 7️⃣ Return in float32 (compatible with the reference implementation)
    # -----------------------------------------------------------------
    return out.float()
