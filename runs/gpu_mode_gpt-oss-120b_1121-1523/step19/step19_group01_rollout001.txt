"""
TriMul (outgoing) forward pass implemented with fused linear projections,
single mask multiplication kernel, and batched matrix multiply via torch.matmul.
The heavy computation is kept in FP16; layernorms are performed in FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: in-place mask multiplication (broadcast over H)
# --------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    x_ptr,         # pointer to x   (B, N, N, H)
    mask_ptr,      # pointer to mask (B, N, N)
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Perform: x[b,i,j,h] *= mask[b,i,j]
    Mask is broadcast over the hidden dimension.
    """
    b = tl.program_id(0)        # batch dimension
    pid_i = tl.program_id(1)    # tile over i
    pid_j = tl.program_id(2)    # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load mask tile (BLOCK_I, BLOCK_J) and broadcast over H
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )  # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # Loop over hidden dimension
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        x = x * mask_tile
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In-place multiplication x *= mask (broadcast mask over hidden dim).
    Both tensors must be CUDA tensors with dtype float16.
    """
    B, N, _, H = x.shape
    s_x = x.stride()
    s_m = mask.stride()
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64   # enough for all supported hidden sizes (≤384)
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_mul_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        # Some GPUs (e.g., H100) like 8 warps; 4 is safe.
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3) accelerator.
    Steps:
      1. LayerNorm on input (FP32).
      2. Cast to FP16 for heavy compute.
      3. Fuse four linear projections/gates into a single matmul.
      4. Optional mask multiplication (Triton kernel).
      5. Apply sigmoid gates.
      6. Batched GEMM across the hidden dimension using torch.matmul
         on a (B, hidden, N, N) tensor.
      7. LayerNorm on hidden output, gate with out_gate.
      8. Final linear projection back to input dimensionality.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel C
    hidden = config["hidden_dim"]           # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32, stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  dtype = float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for projection / GEMM heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)          # [B, N, N, dim]  dtype = fp16
    B, N, _, C = x_fp16.shape

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear projection + gate (single matmul)
    #    Produce four tensors: left_proj, right_proj, left_gate, right_gate
    # ------------------------------------------------------------------
    # Concatenate all four weight matrices (out_features, in_features)
    # Order: left_proj, right_proj, left_gate, right_gate
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*hidden, dim)

    # Flatten spatial axes for a single linear call
    x_flat = x_fp16.view(-1, C)                # (B*N*N, dim)
    # Linear: (B*N*N, 4*hidden)
    all_out = F.linear(x_flat, proj_gate_weight)  # fp16

    # Split the combined output
    all_out = all_out.view(B, N, N, 4 * hidden)   # (B, N, N, 4*hidden)
    left_proj, right_proj, left_gate, right_gate = torch.split(
        all_out, hidden, dim=-1
    )  # each (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask multiplication (in‑place via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp16)
        _apply_mask_triton(right_proj, mask_fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (elementwise)
    # ------------------------------------------------------------------
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # Release memory we no longer need (optional, may help with peak memory)
    del left_proj, right_proj, left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update via batched GEMM.
    #    left, right : (B, N, N, hidden) --> (B, hidden, N, N)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    # Batched matmul: left @ rightᵀ   (B, hidden, N, N)
    out_hidden = torch.matmul(
        left_perm,
        right_perm.transpose(-1, -2)
    )  # fp16

    # Back to (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out_fp32 = out.float()  # conversion for numerically stable LayerNorm
    out_norm = F.layer_norm(
        out_fp32,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=torch.float32, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=torch.float32, device=device),
        eps=eps,
    )
    # out_gate = sigmoid( x @ out_gate_weight )
    out_gate_fp16 = torch.sigmoid(
        F.linear(
            x_fp16,
            weights["out_gate.weight"].to(dtype=torch.float16, device=device)
        )
    )  # (B, N, N, hidden) fp16
    out_gate = out_gate_fp16.float()

    out = out_norm * out_gate  # still fp32

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype=torch.float32, device=device),
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9️⃣ Return FP32 tensor
    # ------------------------------------------------------------------
    return out

"""
TriMul (outgoing) forward pass – accelerated version.

Algorithm
---------
1️⃣ Layer‑norm the input (float32, stable).
2️⃣ Cast to FP16 for the heavy work.
3️⃣ Fuse the four linear layers (left‑proj, right‑proj, left‑gate, right‑gate)
   into a single matmul; split the result.
4️⃣ Optional mask multiplication is done in‑place with a tiny Triton kernel
   (mask is broadcast over the hidden dimension).
5️⃣ Apply sigmoid gates.
6️⃣ Core O(N³·H) update: for each hidden slice compute
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a single
   batched GEMM via torch.matmul on tensors of shape (B, H, N, N).
7️⃣ Output LayerNorm → out‑gate sigmoid → element‑wise gating.
8️⃣ Final linear projection back to the original channel dimension.
9️⃣ Return a float32 tensor of shape [B, N, N, dim].

Only the mask‑multiply kernel is written in Triton; the rest uses
high‑performance PyTorch primitives on FP16 data.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: in‑place mask multiplication (broadcast over H)
# --------------------------------------------------------------
@triton.jit

    x_ptr,         # pointer to x   (B, N, N, H)
    mask_ptr,      # pointer to mask (B, N, N)
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Perform x[b,i,j,h] *= mask[b,i,j]  (mask broadcast over the hidden dim).
    """
    b = tl.program_id(0)        # batch index
    pid_i = tl.program_id(1)    # tile over i
    pid_j = tl.program_id(2)    # tile over j

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_ok = i < N
    j_ok = j < N

    # Load a (BLOCK_I, BLOCK_J) mask tile and broadcast over H
    mask_tile = tl.load(
        mask_ptr
        + b * stride_m_b
        + i[:, None] * stride_m_i
        + j[None, :] * stride_m_j,
        mask=i_ok[:, None] & j_ok[None, :],
        other=0.0,
    )                               # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]   # (BLOCK_I, BLOCK_J, 1)

    # Iterate over the hidden dimension in blocks
    h_start = 0
    while h_start < H:
        h = h_start + tl.arange(0, BLOCK_H)
        h_ok = h < H

        x = tl.load(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
            other=0.0,
        )
        x = x * mask_tile
        tl.store(
            x_ptr
            + b * stride_x_b
            + i[:, None, None] * stride_x_i
            + j[None, :, None] * stride_x_j
            + h[None, None, :] * stride_x_h,
            x,
            mask=i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :],
        )
        h_start += BLOCK_H


def _apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place multiplication ``x *= mask`` where mask is broadcast over the
    hidden dimension. Both tensors must be on the same CUDA device and be
    of dtype torch.float16.
    """
    B, N, _, H = x.shape
    s_x = x.stride()
    s_m = mask.stride()
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 64                 # H ≤ 384 → at most 6 iterations
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    _mask_mul_kernel[grid](
        x,
        mask,
        B,
        N,
        H,
        s_x[0],
        s_x[1],
        s_x[2],
        s_x[3],
        s_m[0],
        s_m[1],
        s_m[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3) with a Triton mask kernel.
    Returns a float‑32 tensor of shape [B, N, N, dim].
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input channel C
    hidden = config["hidden_dim"]           # projection dimension H
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                          # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy portion
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)               # [B, N, N, dim]  fp16
    B, N, _, C = x_fp16.shape

    # ------------------------------------------------------------------
    # 3️⃣ Fuse four linear layers (proj + gate) into a single matmul
    #    Order after concatenation: left_proj | right_proj | left_gate | right_gate
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)   # (4*hidden, dim)

    x_flat = x_fp16.view(-1, C)                # (B·N·N, dim)
    fused_out = F.linear(x_flat, fused_weight)  # (B·N·N, 4*hidden)  fp16

    fused_out = fused_out.view(B, N, N, 4 * hidden)   # (B, N, N, 4*hidden)
    left_proj, right_proj, left_gate, right_gate = torch.split(
        fused_out, hidden, dim=-1
    )   # each (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask multiplication (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype=left_proj.dtype, device=device)
        _apply_mask_triton(left_proj, mask_fp16)
        _apply_mask_triton(right_proj, mask_fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # Release intermediate tensors early
    del left_proj, right_proj, left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³·H) update: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # Re‑order to (B, hidden, N, N) so torch.matmul can treat it as a batch
    left_perm = left.permute(0, 3, 1, 2)   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2) # (B, hidden, N, N)

    # Batched matrix multiplication: (B, hidden, N, N) × (B, hidden, N, N)ᵀ
    out_hidden = torch.matmul(
        left_perm,
        right_perm.transpose(-1, -2)
    )                                        # (B, hidden, N, N)  fp16

    # Clear left/right to free memory
    del left, right, left_perm, right_perm

    # Restore original layout (B, N, N, hidden)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out_fp32 = out.float()  # LayerNorm in FP32 for numerical stability
    out_norm = F.layer_norm(
        out_fp32,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(dtype=torch.float32, device=device),
        bias=weights["to_out_norm.bias"].to(dtype=torch.float32, device=device),
        eps=eps,
    )                                           # (B, N, N, hidden) fp32

    # out_gate = sigmoid( x_fp16 @ out_gate.weightᵀ )
    out_gate_fp16 = torch.sigmoid(
        F.linear(
            x_fp16,
            weights["out_gate.weight"].to(dtype=torch.float16, device=device),
        )
    )                                           # (B, N, N, hidden) fp16
    out_gate = out_gate_fp16.float()           # to fp32 for the multiplication

    out = out_norm * out_gate                  # (B, N, N, hidden) fp32

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype=torch.float32, device=device),
    )                                           # (B, N, N, dim) fp32

    # ------------------------------------------------------------------
    # 9️⃣ Return
    # ------------------------------------------------------------------
    return out
```
