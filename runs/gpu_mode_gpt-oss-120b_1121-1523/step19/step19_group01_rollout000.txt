@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # batch index (BH)
    pid_m = tl.program_id(1)  # tile over rows
    pid_n = tl.program_id(2)  # tile over cols

    # compute offsets for block
    block_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    block_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ensure bound checks
    mask_i = block_i < M
    mask_j = block_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate in fp32

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_batch * stride_a_batch + block_i[:, None] * stride_a_m + k_offset[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B block: shape (BLOCK_K, BLOCK_N) BUT B is transposed: B[k,j] = original B[j,k]
        # We need B_ptr offsets: batch*stride + k_offset[:, None] * stride_b_k + block_j[None, :] * stride_b_n
        b_ptrs = B_ptr + pid_batch * stride_b_batch + k_offset[:, None] * stride_b_k + block_j[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # a shape (BLOCK_M, BLOCK_K), b (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch + block_i[:, None] * stride_c_m + block_j[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
custom_kernel – Fast outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1. LayerNorm on the input (float32 → more stable).
2. Cast to fp16 for the heavy part.
3. Fuse left/right projections and left/right gates into two linear layers
   (weights are concatenated).  The mask (if present) is applied *before*
   the projection, which is mathematically equivalent to the reference code
   but removes a dedicated Triton mask kernel.
4. Apply sigmoid gates.
5. Core O(N³·H) update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a *single* batched GEMM kernel in Triton.  The kernel
   processes all (batch*head) matrices in one launch, eliminating the
   many‑kernel overhead of torch.bmm.
6. Output LayerNorm → out‑gate sigmoid → element‑wise gating.
7. Final linear projection back to the original channel dimension.
8. Return float32 to match the reference.

The only Triton kernel used is the batched GEMM (step 5), satisfying the
“must‑use‑Triton” requirement while keeping the compute‑bound part highly
optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton batched GEMM kernel (computes left @ rightᵀ for many batches)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    BATCH, M, N, K,                          # dimensions
    stride_a_batch, stride_a_m, stride_a_k,  # A strides
    stride_b_batch, stride_b_k, stride_b_n,  # B strides (B is transposed)
    stride_c_batch, stride_c_m, stride_c_n,  # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of size BATCH.
    A, B, C are in fp16, accumulation is in fp32.
    A layout: (batch, M, K)  – stride_a_m = stride over rows
    B layout: (batch, N, K)  – we treat Bᵀ, so stride_b_k is stride over K (original col)
    C layout: (batch, M, N)
    """
    pid_batch = tl.program_id(0)   # which matrix in the batch
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A block (M x K)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # load Bᵀ block (K x N)  – note the swapped strides
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_b_n   # N dimension (col of Bᵀ)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward‑only outgoing TriMul (AlphaFold‑3) using a fused Triton GEMM.
    Input:
        data = (input_tensor, mask_tensor, weights, config)
    Output:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ----------------------------------------------------------
    # 1️⃣ LayerNorm (float32)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – fp32

    # ----------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ----------------------------------------------------------
    x_h = x.half()                           # [B, N, N, dim], fp16
    x_flat = x_h.view(-1, dim)               # (B·N·N, dim)

    # ----------------------------------------------------------
    # 3️⃣ Fused gate linear (left_gate & right_gate)
    # ----------------------------------------------------------
    gate_weight_cat = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)   # (2*hidden, dim)

    gate_cat = F.linear(x_flat, gate_weight_cat)          # (B·N·N, 2*hidden)
    gate_cat = gate_cat.view(B, N, N, 2 * hidden)        # (B, N, N, 2*hidden)
    left_gate, right_gate = torch.split(gate_cat, hidden, dim=-1)
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    # ----------------------------------------------------------
    # 4️⃣ Apply mask *before* projection (equivalent to masking after)
    # ----------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast over hidden dimension
        mask_exp = mask_tensor.unsqueeze(-1).to(dtype=torch.float16, device=device)
        x_proj_input = x_h * mask_exp
    else:
        x_proj_input = x_h

    # ----------------------------------------------------------
    # 5️⃣ Fused projection linear (left_proj & right_proj)
    # ----------------------------------------------------------
    proj_weight_cat = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(dtype=torch.float16, device=device)   # (2*hidden, dim)

    proj_cat = F.linear(x_proj_input.view(-1, dim), proj_weight_cat)  # (B·N·N, 2*hidden)
    proj_cat = proj_cat.view(B, N, N, 2 * hidden)                    # (B, N, N, 2*hidden)
    left_proj, right_proj = torch.split(proj_cat, hidden, dim=-1)

    # ----------------------------------------------------------
    # 6️⃣ Apply sigmoid gates
    # ----------------------------------------------------------
    left = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # ----------------------------------------------------------
    # 7️⃣ Core O(N³·H) update – batched GEMM via Triton
    # ----------------------------------------------------------
    # Permute to (B, hidden, N, N) → view as (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Output allocation
    out_flat = torch.empty_like(left_flat)

    # Strides (already contiguous)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    # Right is transposed inside the kernel: (K dimension = original col)
    stride_b_k = right_flat.stride(2)   # stride of original column → 1
    stride_b_n = right_flat.stride(1)   # stride of original row  → N

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tile sizes – tuned for H100 fp16 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 8️⃣ Output LayerNorm + out‑gate
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"]
        .to(dtype=out.dtype, device=device),
        bias=weights["to_out_norm.bias"]
        .to(dtype=out.dtype, device=device),
        eps=eps,
    )
    out_gate = torch.sigmoid(
        F.linear(
            x_h,
            weights["out_gate.weight"]
            .to(dtype=torch.float16, device=device),
        )
    )
    out = out * out_gate

    # ----------------------------------------------------------
    # 9️⃣ Final linear projection back to dim
    # ----------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"]
        .to(dtype=torch.float16, device=device),
    )  # (B, N, N, dim), fp16

    # ----------------------------------------------------------
    # 🔟 Return float32 (matches reference)
    # ----------------------------------------------------------
    return out.float()
