"""
Propose a low‑score step function.

Algorithm
---------
1. Build a diverse pool of initial candidates (spectral, Hann, Blackman,
   Gaussian, Kaiser, several random Dirichlet‑like draws and a simple
   alternating‑order assignment).  Keep the best one by the exact fast
   objective `fast_score`.

2. Repeatedly launch short optimisation “runs’’ until the time budget is
   exhausted.  Each run consists of:
   * a warm‑start from the current global best mixed with a fresh random
     vector,
   * a few L‑BFGS steps on a smoothed max‑convolution (log‑sum‑exp),
   * a short Adam run on the true max‑convolution,
   * a local greedy hill‑climb (pairwise mass transfers) and a swap‑based
     simulated annealing to fine‑tune the ordering of the entries.

   The best vector seen across all runs is kept globally.

3. After the budget expires the best vector is returned as a plain
   Python list.  The official `evaluate_sequence` function is called once
   at the end for reporting.

All operations respect the non‑negativity and sum‑to‑one constraints,
and the implementation is fully vectorised with NumPy/ SciPy only.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective (sum=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) + gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Return (f, grad) where f = 2·n·smooth_max(conv) and grad = ∂f/∂y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n-1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'beta':
        a = rng.beta(0.5, 0.5, size=n)
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def alternating_assign(values: np.ndarray) -> np.ndarray:
    """Place sorted values at alternating ends (large–small interleaving)."""
    vals = np.sort(values)[::-1]  # descending
    res = np.empty_like(vals)
    left = 0
    right = len(vals) - 1
    toggle = True
    for v in vals:
        if toggle:
            res[left] = v
            left += 1
        else:
            res[right] = v
            right -= 1
        toggle = not toggle
    return res

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS over the whole λ‑schedule (early‑stop on deadline)."""
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lr0: float = 0.2):
    """Adam on the exact max‑convolution objective (logits = softmax)."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and its max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient wrt a of the max entry (sub‑gradient)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # full gradient of the objective 2·n·max
        grad_a = 2.0 * n * g_a

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = lr0 / np.sqrt(t)      # decaying LR
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 50000,
               delta_frac: float = 0.06):
    """Greedy pairwise mass‑transfer hill‑climb (strictly improving)."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc


def swap_annealing(a: np.ndarray,
                   deadline: float,
                   n: int,
                   fft_len: int,
                   rng: np.random.Generator,
                   init_temp: float = 1e-3,
                   final_temp: float = 1e-7,
                   max_swaps: int = 20000):
    """
    Simple simulated‑annealing on pairwise swaps.
    Accepts worse states with probability exp(-(Δ)/T).
    """
    cur = a.copy()
    cur_sc = fast_score(cur, fft_len, n)
    best = cur.copy()
    best_sc = cur_sc

    swaps = 0
    while time.time() < deadline and swaps < max_swaps:
        # temperature schedule (linear in swap count)
        t_frac = swaps / max_swaps
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            swaps += 1
            continue

        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]    # swap
        sc = fast_score(cand, fft_len, n)

        if sc < cur_sc:
            cur = cand
            cur_sc = sc
            if sc < best_sc:
                best = cand
                best_sc = sc
        else:
            prob = np.exp(-(sc - cur_sc) / max(T, 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = sc
        swaps += 1
    return best, best_sc


def perturb_and_refine(a: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator):
    """
    Random perturbations followed by a cheap Adam run.
    Used as a diversification step inside the global loop.
    """
    best = a.copy()
    best_sc = fast_score(best, fft_len, n)

    while time.time() < deadline:
        # small Gaussian perturbation
        pert = rng.normal(scale=0.01, size=n)
        cand = best + pert
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        # a brief Adam refinement (few seconds max)
        inner_deadline = min(deadline, time.time() + 2.5)
        cand_ref, sc_ref = adam_optimize(cand, inner_deadline, n, fft_len, rng, lr0=0.1)

        if sc_ref < best_sc:
            best = cand_ref
            best_sc = sc_ref
            print(f"[Perturb] new best fast_score={best_sc:.8f}")

    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (power‑of‑two for efficient FFT)
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial pool of candidates
    # ------------------------------------------------------------------
    candidates = []

    # use previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # random families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # a simple alternating ordering of a random Dirichlet draw
    rand_vals = random_initial(n, rng)
    candidates.append(alternating_assign(rand_vals))

    # choose the best initial candidate (exact fast_score)
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Global time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Outer optimisation loop (multiple short runs)
    # ------------------------------------------------------------------
    run_id = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # each run gets roughly 30 s (or the remaining time if smaller)
        run_budget = min(30.0, remaining)
        run_deadline = time.time() + run_budget
        run_id += 1
        print(f"\n--- Run {run_id} (budget {run_budget:.1f}s) ---")

        # --- Warm‑start ------------------------------------------------
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        # --- Phase‑1: short L‑BFGS -----------------------------------
        phase1_deadline = min(run_deadline, time.time() + 8.0)
        lam_schedule = [0.3, 0.15, 0.07, 0.03, 0.015, 0.008, 0.004, 0.002,
                        0.001, 5e-4, 2e-4]
        a_lbfgs, sc_lbfgs = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam=400,
            deadline=phase1_deadline,
            fft_len_conv=fft_len,
            fft_len_g=fft_len,
            n=n)

        if sc_lbfgs < best_sc:
            best_sc = sc_lbfgs
            best_a = a_lbfgs.copy()
            print(f"[Run {run_id}] L‑BFGS improved global best to {best_sc:.8f}")

        # --- Phase‑2: Adam -------------------------------------------
        if time.time() < run_deadline:
            a_adam, sc_adam = adam_optimize(
                a_lbfgs,
                deadline=run_deadline,
                n=n,
                fft_len=fft_len,
                rng=rng,
                lr0=0.15)
            if sc_adam < best_sc:
                best_sc = sc_adam
                best_a = a_adam.copy()
                print(f"[Run {run_id}] Adam improved global best to {best_sc:.8f}")

        # --- Phase‑3: Hill‑climb & swap annealing --------------------
        if time.time() < run_deadline:
            cur_sc = fast_score(best_a, fft_len, n)
            hill_deadline = min(run_deadline, time.time() + 5.0)
            a_hill, sc_hill = hill_climb(best_a,
                                         cur_sc,
                                         fft_len,
                                         n,
                                         hill_deadline,
                                         rng,
                                         max_iter=15000,
                                         delta_frac=0.08)
            if sc_hill < best_sc:
                best_sc = sc_hill
                best_a = a_hill.copy()
                print(f"[Run {run_id}] Hill‑climb improved global best to {best_sc:.8f}")

        if time.time() < run_deadline:
            swap_deadline = min(run_deadline, time.time() + 5.0)
            a_swap, sc_swap = swap_annealing(best_a,
                                             swap_deadline,
                                             n,
                                             fft_len,
                                             rng,
                                             init_temp=5e-4,
                                             final_temp=1e-8,
                                             max_swaps=12000)
            if sc_swap < best_sc:
                best_sc = sc_swap
                best_a = a_swap.copy()
                print(f"[Run {run_id}] Swap‑anneal improved global best to {best_sc:.8f}")

        # --- Phase‑4: Random perturb + short Adam --------------------
        if time.time() < run_deadline:
            pert_deadline = min(run_deadline, time.time() + 4.0)
            a_pert, sc_pert = perturb_and_refine(best_a,
                                                 pert_deadline,
                                                 n,
                                                 fft_len,
                                                 rng)
            if sc_pert < best_sc:
                best_sc = sc_pert
                best_a = a_pert.copy()
                print(f"[Run {run_id}] Perturbation improved global best to {best_sc:.8f}")

        # end of run
        print(f"[Run {run_id}] best fast_score so far = {best_sc:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
