def exp_gradient_step(a, g, eta):
    a = a * np.exp(-eta * g)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full_like(a, 1.0 / a.size)
    return a
```

                         fft_len,
                         rng,
                         initial_eta=0.5,
                         min_eta=1e-7,
                         max_iter=10**9):

    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    eta = initial_eta
    it = 0

        # compute conv and mask
        fa = np.fft.rfft(a, fft_len)

        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        grad = 2.0 * rev_grad[n-1:n-1+n]
        # propose step
        # use exponential update
        a_new = a * np.exp(-eta * grad)
        # renormalize
        s = a_new.sum()
        if s <= 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s
        # evaluate
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_a, best_score = a_new.copy(), new_score
            a = a_new
            # possibly increase learning rate a bit
            eta = min(eta * 1.05, 2.0)
        else:
            # reject, decrease step size
            eta = max(eta * 0.5, min_eta)
            # occasional perturb to avoid stall
            if np.random.rand() < 0.001:
                # random small perturb: add gaussian noise then renorm
                noise = rng.normal(scale=0.001, size=n)
                cand = a + noise
                cand = np.clip(cand, 0, None)
                s2 = cand.sum()
                if s2 > 0:
                    cand /= s2
                    a = cand
        it += 1
    return best_a, best_score

"""
Propose a low‑score step function.

Algorithm outline
-----------------
1. Build a diverse pool of candidate probability vectors (including the
   previously best known sequence if available) and keep the best one
   according to the exact objective `fast_score = 2·n·max(conv)`.

2. Warm‑start with a short L‑BFGS continuation on a log‑sum‑exp
   smoothing of the max‑convolution (the same technique as the previous
   runner, but with a tighter time budget).

3. Refine with a long Adam optimisation that gradually narrows the
   top‑k mask from 20 % to 0.5 % of the convolution entries – this
   flattens the autocorrelation while still following the true
   sub‑gradient of the max.

4. Perform an exact projected gradient descent on the *true* max
   (mask = arg‑max of the convolution).  Each step uses an
   exponential‑gradient (mirror‑descent) update on the simplex and a
   back‑tracking line‑search for the step size.  This stage yields
   fine‑grained improvements that were hard to obtain with the Adam
   phase.

5. Finish with a short simulated‑annealing / hill‑climb polish to
   escape any remaining tiny local minima.

The whole pipeline respects a 1000 s budget and returns the best
sequence found (as a list of non‑negative floats).  The official
`evaluate_sequence` rescales the sequence internally, so we keep the
sum normalised to 1 throughout the search.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (fast_score) – used for bookkeeping
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimizer with top‑k mask
# ----------------------------------------------------------------------


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.20,
                       end_top: float = 0.005):
    """
    Adam optimiser that uses a top‑k mask (rather than the single
    maximal entry) to flatten the whole convolution.
    The top‑k fraction linearly decays from start_top to end_top.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑k fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        # mask of largest entries
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the masked max (linear combination of entries)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # d max/ d a (ignoring the 2·n factor)
        grad_a = 2.0 * n * g_a                         # full objective gradient

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)          # a slightly larger initial step
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to shake off plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact projected gradient descent on the true max (mask = argmax)
# ----------------------------------------------------------------------


def max_gradient_descent(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 0.5,
                         min_eta: float = 1e-7):
    """
    Mirror‑descent on the simplex using the exact sub‑gradient of the max
    convolution (mask = arg‑max).  The step size eta is adapted with a
    simple back‑tracking line search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    eta = eta0
    it = 0

    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # mask of entries attaining (or nearly attaining) the maximum
        eps = max_val * 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max wrt a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        grad = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # --- exponential (mirror) update ---
        a_prop = a * np.exp(-eta * grad)
        s = a_prop.sum()
        if s > 0:
            a_prop /= s
        else:
            a_prop = np.full_like(a, 1.0 / n)

        # evaluate
        prop_score = fast_score(a_prop, fft_len, n)

        if prop_score < best_score:
            best_a, best_score = a_prop.copy(), prop_score
            a = a_prop
            # cautiously increase step size
            eta = min(eta * 1.05, 2.0)
        else:
            # reject – shrink step size
            eta = max(eta * 0.5, min_eta)
            # occasional random jitter to escape flat regions
            if rng.random() < 0.001:
                noise = rng.normal(scale=0.001, size=n)
                cand = a + noise
                cand = np.clip(cand, 0, None)
                cs = cand.sum()
                if cs > 0:
                    cand /= cs
                    a = cand
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing & greedy hill‑climb (final polish)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 200000):
    """
    Simple annealing over the simplex: mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.18
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_score:
            a = cand
            best_score = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 50000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny mass‑transfer moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (a power‑of‑two works well with FFTs)
    # --------------------------------------------------------------
    n = 4096                     # keep the previous size for comparability
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    candidates = []

    # reuse the previously best known construction if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # several random draws for added diversity
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # pick the best-by‑fast_score as the starting point
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – short L‑BFGS continuation (≈20 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1,
        0.05, 0.02, 0.01,
        0.005, 0.002, 0.001,
        5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_deadline:
        # random perturbation of the current best to diversify restarts
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam with a decreasing top‑k mask (≈45 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] starting")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.20,   # start with a fairly large active set
            end_top=0.005     # finish with almost the true max
        )
        print(f"[Adam‑topk] best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Exact max‑gradient descent (≈20 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + 0.20 * budget_s
        print("[max‑grad] start exact max‑gradient descent")
        best_a, best_score = max_gradient_descent(
            best_a,
            phase3_deadline,
            n,
            fft_len,
            rng,
            eta0=0.5,
            min_eta=1e-7
        )
        print(f"[max‑grad] best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated annealing + greedy hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining * 0.65, 120.0)
        print("[Sim‑Anneal] start")
        best_a, best_score = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-8,
            max_moves=300000
        )
        print(f"[Sim‑Anneal] best fast_score={best_score:.9f}")

    if time.time() < deadline:
        hill_deadline = deadline
        cur_score = fast_score(best_a, fft_len, n)
        print("[Hill‑Climb] start")
        best_a, best_score = hill_climb(
            best_a,
            cur_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=80000,
            delta_frac=0.04
        )
        print(f"[Hill‑Climb] best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Return the final sequence (the official scorer rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
