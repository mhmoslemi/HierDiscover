"""
Propose a low‑evaluation step‑function.

Algorithm outline
-----------------
* Choose a moderately large dimension (n≈15k) – larger n gives more
  degrees of freedom while still keeping FFT work cheap.
* Build a diverse pool of initial vectors (uniform, windows, random families,
  and the previous best interpolated to the new size).
* **Phase 1 – fast L‑BFGS cascade** on a log‑sum‑exp surrogate.
  A decreasing λ‑schedule quickly shapes the vector.
* **Phase 2 – smoothed exponentiated‑gradient descent** on the true
  L∞‑objective.  The surrogate uses a softmax over the convolution entries,
  giving a smooth sub‑gradient.  An Adam‑style decay (η∝1/√t) together
  with occasional Gaussian jitter drives the search.
* **Phase 3 – finishing polish** with a very low‑λ L‑BFGS cascade and a short
  simulated‑annealing hill‑climb.
* The best vector found (kept on the probability simplex) is finally
  evaluated with the official ``evaluate_sequence`` and stored in the
  global ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``sum(a)=1``:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp surrogate (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution + gradient.
    Work with log‑variables y so that a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # log‑variables
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam,
                         "gtol": 1e-12,
                         "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score

def exp_gradient_optimize_smooth(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 eta0: float = 30.0,
                                 lam0_factor: float = 0.02,
                                 max_iters: int = 30_000_000):
    """
    Mirror‑SGD with a smooth soft‑max mask.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    last_improve_it = 0
    lam = None

    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        if lam is None:
            lam = max_val * lam0_factor + 1e-12

        # smooth weight vector (soft‑max over conv entries)
        shifted = conv - max_val          # ≤ 0
        exp_shift = np.exp(shifted / lam)  # under‑flow safe for small lam
        w = exp_shift / exp_shift.sum()

        # gradient (2 * conv(mask, reversed a))
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_rev = np.real(np.fft.ifft(fw * fa_rev))
        g = 2.0 * grad_rev[n - 1: n - 1 + n]

        # step size (square‑root decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # occasional evaluation
        if (it & 63) == 0:          # every 64 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve_it = it
                print(f"[exp‑grad‑smooth] it={it} fast_score={best_score:.9f}")

        # jitter if stuck
        if it - last_improve_it > 250_000:
            a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)
            last_improve_it = it

        # slowly tighten the smoothing
        lam = max(lam * 0.9995, 1e-10)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polish)
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Linear temperature schedule.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger to smaller entry
        src, dst = (i, j) if a[i] > a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger vector gives a slightly better bound while
    # still fitting comfortably into memory and into the time budget.
    # ------------------------------------------------------------------
    n = 15000                       # ~15k entries (feel free to tune)
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Interpolate the previous best if present
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Classic window families (if SciPy is present)
    try:
        from scipy.signal import chebwin, kaiser, hann, blackman
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        w = hann(n)
        pool.append(w / w.sum())
        w = blackman(n)
        pool.append(w / w.sum())
    except Exception:
        pass

    # 3) Random families
    for _ in range(16):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.12 * budget_s   # short L‑BFGS warm‑up
    phase2_end = start_time + 0.78 * budget_s   # long smoothed EG descent
    # remainder → final polish

    # ------------------------------------------------------------------
    # Phase 1 – quick L‑BFGS cascade (coarse shaping)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07,
        0.04, 0.022, 0.012, 0.006,
        0.003, 0.0015, 7e-4, 3e-4,
        1.5e-4, 8e-5,
    ]
    maxiter_per_lam = 600

    best_a = None
    best_score = float("inf")

    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter before the L‑BFGS run
        if rng.random() < 0.33:
            a_start = add_noise(
                a_start, sigma=rng.uniform(0.001, 0.014), rng=rng
            )
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase 1] new best fast_score={best_score:.9f}")

    # fallback if nothing succeeded
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long smoothed exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize_smooth(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=35.0,
            lam0_factor=0.02,
            max_iters=40_000_000,
        )
        print(f"[Phase 2] exp‑grad‑smooth best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing (low‑λ L‑BFGS + short anneal)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a low‑λ L‑BFGS refinement
        fine_lam_schedule = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=1400,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase 3‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        # a tiny moving‑average smoothing before annealing
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = np.maximum(smoothed, 0.0)
        smoothed /= smoothed.sum()
        sm_score = fast_score(smoothed, fft_len)
        if sm_score < best_score:
            best_a = smoothed
            best_score = sm_score
            print(f"[Phase 3‑smooth] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[Phase 3‑anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
