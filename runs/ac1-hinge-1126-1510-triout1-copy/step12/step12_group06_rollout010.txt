def lp_refine(a, iterations, t, deadline):
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    for _ in range(iterations):
        if time.time() >= deadline:
            break
        # compute M
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n - 1]
        M = conv.max()
        # LP solve
        a_eps = np.maximum(a, 1e-12)
        A_ub = diags([np.full(n, a_eps[d]) for d in range(n)], offsets=np.arange(n), shape=(2*n-1, n), format='csc')
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_scaled = g / sum_g  # sum=1
        # convex combination
        a = (1 - t) * a + t * g_scaled
        # optionally re-normalize due to rounding error
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum <= 0:
            a = best_a.copy()
        else:
            a /= a_sum
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP] iter {iter_idx} improved fast_score = {best_score:.9f}")
        # optionally update t (decrease)
        t = max(0.1, t * 0.9)
    return best_a, best_score
```

"""
Propose‑candidate optimiser (≈1000 s)

The optimiser proceeds in four stages:

1. **Warm start** – load the previous best sequence if it exists,
   otherwise generate a random probability vector.

2. **Smooth L‑BFGS** – minimise a log‑sum‑exp surrogate of the
   true ‑∞‑objective.  A decreasing λ‑schedule gives a good
   approximation and yields a strong initial point.

3. **Linear‑programming bound reduction** – given the current
   distribution `a`, solve the LP  

        maximise   1ᵀ g
        s.t.       a ∗ g ≤ M          (M = max(a∗a))
                   g ≥ 0

   The matrix of the convolution constraints is a lower‑triangular
   Toeplitz matrix built with `scipy.sparse.diags`.  After solving,
   `g` is renormalised to unit sum and replaces `a`.  Repeating this
   step provably never raises the max‑convolution and quickly drives
   the objective downward.

4. **Exact sub‑gradient descent (mirrored EG)** – a long run of the
   exponentiated‑gradient method on the true max‑convolution refines
   the solution to a (near) stationary point.

5. **Greedy mass‑transfer hill‑climb** – small random transfers of
   mass are accepted only if they strictly improve the exact
   evaluation score.

All stages respect the global deadline; the best sequence seen so
far is stored in the global variable `height_sequence_1` and is
returned as a plain list of floats.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Return the value 2·n·max(conv(a,a)).
    The sequence `a` is assumed to be non‑negative and already normalized
    to have unit sum (the scaling cancels in the final score).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector from several styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density on the lattice."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power, clipped at zero."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass concentrated at the two edges."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained variables; soft‑maxed to the simplex.
    n : int
    fft_len : int
    lam : float
        Smoothing parameter.
    Returns
    -------
    f, grad_y
    """
    # soft‑max to stay on the simplex
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # effective λ (scaled with current max to keep smoothing relative)
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)     # safe: arguments ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                         # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best sequence and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirrored exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored (multiplicative) sub‑gradient descent on the true L∞‑objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (mirrored version)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update with clipping for numerical safety
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # minimal output to avoid clutter
            print(f"[exp‑grad] iter {it} fast_score = {best_score:.9f}")

        it += 1

    return best_a, best_score

def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Simple hill‑climbing: repeatedly transfer a small random fraction of mass
    from one position to another.  Only moves that strictly improve the
    `2·n·max(conv)` score are accepted.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20                # up to 20 % of source mass may be moved

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from the larger entry to the smaller one
        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
                print(f"[local‑refine] new best fast_score = {best_score:.9f}")

    return best_a, best_score

# Linear‑programming bound reduction
# ----------------------------------------------------------------------


def lp_refinement(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 12):
    """
    Iteratively solve the LP
        maximise   1ᵀ g
        s.t.       a ∗ g ≤ M
                   g ≥ 0,
    where M = max(a∗a).  The solution g is renormalised to unit sum
    and becomes the new a.  The process never increases the max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current bound
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # Build the convolution‑matrix A (lower‑triangular Toeplitz)
        # To avoid singular columns we keep a tiny epsilon.
        a_eps = np.maximum(a, 1e-12)

        # Each diagonal d (0 ≤ d < n) holds the constant a_eps[d]
        diag_data = [np.full(n, a_eps[d]) for d in range(n)]
        A_ub = diags(diagonal=diag_data,
                     offsets=np.arange(n),
                     shape=(2 * n - 1, n),
                     format='csc')

        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)                     # maximise sum(g)  ⇔ minimise –sum(g)
        bounds = [(0, None)] * n

        # Solve the LP (HiGHS – fast for this size)
        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        # free the large matrix early
        del A_ub

        if not res.success:
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # Renormalise to unit sum – this automatically lowers the max‑conv.
        a = g / sum_g

        # occasional tiny noise to avoid exact stagnation
        if rng.random() < 0.05:
            a = add_noise(a, sigma=0.001, rng=rng)

        # keep best ever
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP] iter {it} new best fast_score = {best_score:.9f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 4000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Warm start – reuse the previous best if available
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            a = random_initial(n, rng)
    else:
        a = random_initial(n, rng)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed objective
    # --------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    lam_schedule = [
        0.4, 0.25, 0.15, 0.09, 0.06, 0.04,
        0.025, 0.015, 0.009, 0.005, 0.0025,
        0.0012, 6e-4, 3e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6
    ]
    a, _ = lbfgs_cascade(a,
                         lam_schedule,
                         maxiter_per_lam=800,
                         deadline=phase1_end,
                         fft_len=fft_len,
                         n=n)

    # --------------------------------------------------------------
    # Phase 2 – linear‑programming bound reduction
    # --------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    a, _ = lp_refinement(a,
                         deadline=phase2_end,
                         n=n,
                         fft_len=fft_len,
                         rng=rng,
                         max_iters=20)

    # --------------------------------------------------------------
    # Phase 3 – long exponentiated‑gradient descent (exact max)
    # --------------------------------------------------------------
    phase3_end = start_time + 0.85 * budget_s
    a, _ = exp_gradient_optimize(a,
                                 deadline=phase3_end,
                                 n=n,
                                 fft_len=fft_len,
                                 rng=rng,
                                 eta0=18.0)

    # --------------------------------------------------------------
    # Phase 4 – greedy local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        cur_score = fast_score(a, fft_len)
        a, _ = local_refinement(a,
                                cur_score,
                                deadline,
                                n,
                                fft_len,
                                rng)

    # --------------------------------------------------------------
    # Final evaluation with the provided scorer
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
