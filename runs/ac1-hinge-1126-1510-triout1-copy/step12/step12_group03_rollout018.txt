"""
Optimising a non‑negative step function to minimise
2·n·max(self‑convolution) under the normalisation Σa = 1.
The search combines a diverse set of initial families, several
L‑BFGS runs on a smooth log‑sum‑exp surrogate, Adam
optimisation, exponentiated‑gradient, projected sub‑gradient,
simulated‑annealing and a long greedy hill‑climb.  A final
dedicated sub‑gradient descent on the exact max‑convolution
is performed to polish the result further.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (logits → softmax)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density (clipped)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       fft_len_g,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """
    Adam optimiser on the smoothed max‑convolution.
    The smoothing factor λ = lam_factor·max(conv) is kept fixed throughout
    this short phase (very small → close to true max).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution,
    using a mask covering the top‑fraction of entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the smoothed max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())               # index of the peak

        # gradient of conv[idx] w.r.t. a (2 * a[idx - j] where defined)
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # simple line‑search: try current step and half step
        alpha = step0 / np.sqrt(it + 1.0)
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                break
        else:
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 600000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    total_moves = max_moves

    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        # transfer up to 20 % of source mass
        delta = a[i] * rng.random() * 0.20
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.09):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def gradient_refine(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator):
    """
    Dedicated sub‑gradient descent that directly minimises the exact
    max‑convolution.  At each iteration the shift attaining the current
    maximum is identified; the gradient is 2·a shifted by that amount,
    projected to the zero‑sum sub‑space and applied with a decreasing
    step‑size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())

        # build gradient vector g_i = 2 * a_{k_max-i} (where defined)
        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[j[valid]]
        if (k_max % 2) == 0:
            i_center = k_max // 2
            if 0 <= i_center < n:
                g[i_center] = a[i_center]   # diagonal term counted only once

        # projection to keep sum(a) constant
        mean_g = np.mean(g)
        d = -(g - mean_g)          # descent direction (negative gradient)

        # step size schedule
        eta = 0.20 / np.sqrt(it + 1.0)

        a_new = a + eta * d
        # enforce non‑negativity then renormalise
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0.0:
            a_new = a                     # fallback
        else:
            a_new /= s

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new                     # keep the improvement
        else:
            # occasionally accept a non‑improving step to escape plateaus
            if rng.random() < 1e-4:
                a = a_new
        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine follows a multi‑phase pipeline (L‑BFGS, Adam,
    exponentiated‑gradient, sub‑gradient, simulated annealing,
    greedy hill‑climb and a final specialised gradient refinement).
    The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – we use a larger n (up to 8192) without increasing FFT length.
    # --------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build initial candidate pool
    # --------------------------------------------------------------
    candidates = []

    # Interpolate a previously exported solution if available.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Additional families
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(5):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Choose the best start according to the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈30 % time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    # a coarse‑to‑fine λ schedule (few early values, many fine values)
    lam_schedule = [
        0.5, 0.30, 0.15, 0.08, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 2e-05,
        1e-05, 5e-06, 2e-06, 1e-06, 5e-07,
        2e-07, 1e-07, 5e-08, 2e-08, 1e-08,
        5e-09, 2e-09, 1e-09
    ]
    maxiter_per_lam = 1800   # a balance between depth and number of restarts

    while time.time() < phase1_end:
        # blend the current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4          # ≤0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈20 % time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.0008,
                                           lr=0.09,
                                           max_iter=30000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient (≈10 % time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.65 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.04)
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact‑max sub‑gradient descent (≈10 % time)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.75 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase4_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.30)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing (≈10 % time)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.90 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase5_end,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-8,
                                                 max_moves=800000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Long greedy hill‑climb (remaining time before final refine)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 30.0   # keep ~30 s for the final gradient refine
    if time.time() < hill_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        rng,
                                        max_iter=500000,
                                        delta_frac=0.09)
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 7 – Dedicated gradient refinement on exact max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = gradient_refine(best_a,
                                             deadline,
                                             n,
                                             fft_len,
                                             rng)
        print(f"[grad‑refine] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq

"""
Optimising a non‑negative step function to minimise
2·n·max(self‑convolution) under the normalisation Σa = 1.

The search combines many complementary heuristics:
    • A large pool of diverse initial families (spectral, window, random,
      power‑law, etc.).
    • Several L‑BFGS runs on a log‑sum‑exp smoothed surrogate of the max‑conv.
    • Adam optimisation on the same smooth objective.
    • Exponentiated‑gradient descent that focuses on the top‑fraction of
      convolution entries.
    • Projected sub‑gradient descent directly on the exact max‑conv.
    • Simulated annealing with mass‑transfer moves.
    • A long greedy hill‑climb.
    • A final specialised gradient refinement that repeatedly reduces the
      exact max‑convolution by moving mass away from the most critical shift.

The algorithm respects the 1000 s budget and returns the best sequence
found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (logits → softmax)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density (clipped)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       fft_len_g,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """
    Adam optimiser on the smoothed max‑convolution.
    The smoothing factor λ = lam_factor·max(conv) is kept fixed throughout
    this short phase (very small → close to true max).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02):
    """
    Exponentiated‑gradient descent on the exact max‑convolution,
    using a mask covering the top‑fraction of entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the smoothed max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())               # index of the peak

        # gradient of conv[idx] w.r.t. a (2 * a[idx - j] where defined)
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection (maintain sum = 1)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # simple line‑search: try current step and half step
        alpha = step0 / np.sqrt(it + 1.0)
        for factor in (1.0, 0.5, 0.25):
            a_new = a + factor * alpha * d
            a_new = simplex_projection(a_new)
            sc_new = fast_score(a_new, fft_len, n)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                break
        else:
            a = simplex_projection(a + alpha * d)

        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 800000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    total_moves = max_moves

    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        # transfer up to 20 % of source mass
        delta = a[i] * rng.random() * 0.20
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.09):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def gradient_refine(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator):
    """
    Dedicated sub‑gradient descent that reduces the exact max‑convolution.
    At each step the shift attaining the current maximum is located;
    the gradient (2·a shifted) is projected to the zero‑sum subspace and
    applied with a decreasing step size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())

        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[j[valid]]
        if (k_max % 2) == 0:
            i_center = k_max // 2
            if 0 <= i_center < n:
                g[i_center] = a[i_center]   # diagonal term counted once

        # project to keep sum(a) constant
        mean_g = np.mean(g)
        d = -(g - mean_g)          # descent direction

        eta = 0.20 / np.sqrt(it + 1.0)

        a_new = a + eta * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = a

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # occasional random acceptance to escape tiny plateaus
            if rng.random() < 1e-4:
                a = a_new
        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – using a larger n (up to 8192) without extra FFT cost.
    # --------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build initial candidate pool
    # --------------------------------------------------------------
    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(5):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.30, 0.15, 0.08, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 2e-05,
        1e-05, 5e-06, 2e-06, 1e-06, 5e-07,
        2e-07, 1e-07, 5e-08, 2e-08, 1e-08,
        5e-09, 2e-09, 1e-09
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam (≈20 % time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.0008,
                                           lr=0.09,
                                           max_iter=30000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient (≈10 % time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.65 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.04)
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact‑max sub‑gradient (≈10 % time)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.75 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase4_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.30)
        print(f"[Sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing (≈10 % time)
    # ------------------------------------------------------------------
    phase5_end = start_time + 0.90 * budget_s
    if time.time() < phase5_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase5_end,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-8,
                                                 max_moves=800000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Greedy hill‑climb (keep some time for final refine)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 30.0   # reserve ~30 s
    if time.time() < hill_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        rng,
                                        max_iter=500000,
                                        delta_frac=0.09)
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 7 – Final gradient refinement on the exact max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = gradient_refine(best_a,
                                             deadline,
                                             n,
                                             fft_len,
                                             rng)
        print(f"[grad‑refine] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
