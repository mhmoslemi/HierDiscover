"""
Improved step‑function search.

The algorithm works in three phases:

* Phase 1 – generate a diverse pool of “shapes’’ (uniform, windows,
  Beta‑pdfs, Gaussians, exponentials, power‑laws, random vectors) and
  keep the best few (fast_score uses an FFT‑based O(n log n) proxy).
  The best candidate is refined with a cascade of L‑BFGS optimisations
  on a smoothed log‑sum‑exp surrogate of the true L∞‑objective.

* Phase 2 – long mirror‑SGD (exponentiated‑gradient) on the exact
  max‑convolution.  The step‑size decays as 1/√t, occasional Gaussian
  “jitter’’ is added when progress stalls.

* Phase 3 – a very fine L‑BFGS polish with a λ‑schedule that reaches
  10⁻⁸, followed by an optional short simulated‑annealing hill‑climb
  and a final symmetry projection.  The resulting vector is evaluated
  with the official scorer and stored in the global variable
  ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the official score (sum(a)=1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Isotropic Gaussian perturbation followed by simplex projection."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Average with reversed copy (enforces left‑right symmetry)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Shape generators for the initial pool
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density sampled on the mid‑point grid."""
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    pdf = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    pdf = np.maximum(pdf, 0.0)
    pdf /= pdf.sum()
    return pdf


def gaussian_shape(sigma_frac: float, n: int) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the array."""
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    idx = np.arange(n, dtype=np.float64)
    pdf = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    pdf = np.maximum(pdf, 0.0)
    pdf /= pdf.sum()
    return pdf


def exponential_shape(lam: float, n: int) -> np.ndarray:
    """Symmetric two‑sided exponential (Laplace) window."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    pdf = np.exp(-lam * np.abs(idx - centre))
    pdf = np.maximum(pdf, 0.0)
    pdf /= pdf.sum()
    return pdf


def power_shape(p: float, n: int) -> np.ndarray:
    """Shape a[i] ∝ ((i+1)*(n-i))**p, symmetric and peaked in the centre."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    rev = n - np.arange(n, dtype=np.float64)
    base = idx * rev
    pdf = np.power(base, p)
    pdf = np.maximum(pdf, 0.0)
    pdf /= pdf.sum()
    return pdf


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector from several simple families."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a

def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution and its gradient w.r.t. log‑variables y.
    a = softmax(y) is the probability vector.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # log‑space working variable
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter_per_lam,
                    "gtol": 1e-12,
                    "disp": False,
                },
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 30.0,
    max_iters: int = 20_000_000,
):
    """
    Mirror‑SGD (exponentiated‑gradient) on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size (square‑root decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # occasional evaluation
        if (it & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = it
                print(f"[exp‑grad] it={it} fast_score={best_score:.9f}")

        # jitter after stagnation
        if (it - last_improve) > 200_000:
            sigma = rng.uniform(5e-4, 2e-3)
            a = add_noise(a, sigma=sigma, rng=rng)
            last_improve = it

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Linear cooling of temperature.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20  # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.
    The routine respects the 1000 s deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – enlarge a bit compared with the previous run.
    n = 8192                     # larger vectors give a finer discretisation
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 0) reuse the previously discovered construction (if any)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) classic analytic windows (if SciPy signal is available)
    try:
        from scipy.signal import chebwin, kaiser, hann, blackman

        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        w = hann(n)
        pool.append(w / w.sum())
        w = blackman(n)
        pool.append(w / w.sum())
    except Exception:
        pass

    # 2) parametric families
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        pool.append(beta_shape(alpha, n))

    for sigma_frac in (0.05, 0.07, 0.10, 0.13, 0.16):
        pool.append(gaussian_shape(sigma_frac, n))

    for lam in (0.001, 0.002, 0.003, 0.005):
        pool.append(exponential_shape(lam, n))

    for p in (1, 2, 3, 4):
        pool.append(power_shape(p, n))

    # 3) random shapes
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Initial screening – keep the best few candidates
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin
    fast_scores = [(fast_score(a, fft_len), a) for a in pool]
    fast_scores.sort(key=lambda x: x[0])
    top_k = 3
    best_score, best_a = fast_scores[0]
    print(f"[Init] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 1 – quick L‑BFGS cascade on the current best candidate
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    lam_schedule_coarse = [
        0.5,
        0.3,
        0.2,
        0.12,
        0.07,
        0.04,
        0.022,
        0.012,
        0.006,
        0.003,
        0.0015,
        7e-4,
        3e-4,
        1.5e-4,
    ]
    maxiter_per_lam = 500

    a_opt, sc_opt = lbfgs_cascade(best_a, lam_schedule_coarse,
                                 maxiter_per_lam, phase1_end, fft_len, n)
    if sc_opt < best_score:
        best_score, best_a = sc_opt, a_opt.copy()
        print(f"[Phase1] improved fast_score={best_score:.9f}")

    # optional symmetry pass (often beneficial)
    sym_a = symmetrize(best_a)
    sym_score = fast_score(sym_a, fft_len)
    if sym_score < best_score:
        best_a, best_score = sym_a, sym_score
        print(f"[Phase1] symmetry improved fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=40.0, max_iters=30_000_000
        )
        print(f"[Phase2] exp‑grad best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine L‑BFGS polish with a very low λ‑schedule
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [
            5e-5,
            2e-5,
            1e-5,
            5e-6,
            2e-6,
            1e-6,
            5e-7,
            2e-7,
            1e-7,
            5e-8,
            2e-8,
            1e-8,
        ]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_a, best_score = a_refined.copy(), sc_refined
            print(f"[Phase3] refined fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Optional final simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a, best_score, deadline, n, rng
        )
        print(f"[Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
