"""
Improved step‑function optimiser (≤ 1000 s)

The optimiser follows a *coarse‑to‑fine* strategy

1. **Small‑scale genetic search** (n≈2000) – a lightweight GA with
   mutation, crossover and occasional Gaussian jitter explores the
   highly non‑convex landscape very quickly (≈ 45 % of the budget).

2. **Upsampling & fine‑grained optimisation** – the best small‑scale
   candidate is interpolated to the target size (n=4000).  A cascade of
   L‑BFGS runs on a log‑sum‑exp surrogate (decreasing λ‑schedule down
   to 5·10⁻⁸) brings the solution close to a stationary point of the
   true ∞‑norm objective.

3. **Exact exponentiated‑gradient descent** – mirrors the sub‑gradient
   of the true max‑convolution, with a 1/√t step‑size schedule, to
   push the score a little further.

4. **Greedy mass‑transfer hill‑climbing** – random transfers of a
   fraction of mass are accepted only when they improve the exact
   `2·n·max(conv)` score.

All phases respect the global deadline; the best sequence ever seen is
kept in the global variable `height_sequence_1` and finally returned as
a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) – fast version using FFT."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex (non‑negative, sum=1). """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of a few simple random families."""
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'edge', 'bimodal', 'sparse', 'block']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1e-12)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:   # block – a few contiguous blocks with random heights
        nb = rng.integers(2, 6)          # number of blocks
        a = np.zeros(n, dtype=float)
        for _ in range(nb):
            b_len = rng.integers(int(0.02 * n), int(0.15 * n))
            start = rng.integers(0, n - b_len)
            height = rng.random() + 0.1
            a[start:start + b_len] += height
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density on the lattice."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power, clipped at zero."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y ∈ ℝⁿ are unrestricted; a = softmax(y) lives on the simplex.
    """
    # ------- softmax (simplex) -------
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # ------- convolution a*a -------
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)   # safe: arguments ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # ------- gradient wrt conv entries -------
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑prop through a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # size n

    # ------- chain rule through softmax -------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (mirrored)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0):
    """
    Multiplicative sub‑gradient descent on the true L∞‑objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # ---- convolution a*a ----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # sub‑gradient of the max (mask of active indices)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # back‑prop to a (same as in the smoothed case)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size (decaying 1/√t)
        eta = eta0 / np.sqrt(it + 1.0)

        # mirrored (multiplicative) EG update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Greedy local refinement (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Simple hill‑climbing: transfer a random fraction of mass from one
    index to another; only strictly improving moves are kept.
    The allowed transfer fraction shrinks as time runs out.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score

    while time.time() < deadline:
        # adapt transfer magnitude: large at start, smaller later
        progress = (deadline - time.time()) / (deadline - start_time)
        max_frac = 0.25 * progress + 0.02   # 0.27 → 0.02

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
    return best_a, best_score

def genetic_search_small(n: int,
                         pop_size: int,
                         max_time: float,
                         rng: np.random.Generator):
    """
    Very lightweight GA on a reduced problem size (n≈2000).
    Returns the best candidate (as a NumPy array) and its fast_score.
    """
    fft_len = next_pow2(2 * n - 1)

    # ----- initialise population -----
    pool = []
    # a bunch of families – the same as in the large‑scale code
    pool.append(np.full(n, 1.0 / n))                  # uniform
    for _ in range(3):
        pool.append(random_initial(n, rng))
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))

    # random Dirichlet‑type draws (different concentration)
    for _ in range(5):
        alpha = rng.uniform(0.1, 5.0, size=n)
        a = rng.gamma(alpha)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        pool.append(a)

    # ensure we have at least pop_size distinct individuals
    while len(pool) < pop_size:
        pool.append(random_initial(n, rng))

    pop = [np.copy(a) for a in pool[:pop_size]]
    scores = np.array([fast_score(a, fft_len) for a in pop])

    best_idx = np.argmin(scores)
    best_a = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    start = time.time()
    while time.time() - start < max_time:
        # ----- parent selection (tournament) -----
        idx1, idx2 = rng.choice(pop_size, size=2, replace=False)
        parent1 = pop[idx1] if scores[idx1] < scores[idx2] else pop[idx2]

        idx3, idx4 = rng.choice(pop_size, size=2, replace=False)
        parent2 = pop[idx3] if scores[idx3] < scores[idx4] else pop[idx4]

        # ----- crossover (convex combination) -----
        w = rng.random()
        child = w * parent1 + (1.0 - w) * parent2
        child = np.maximum(child, 0.0)
        s = child.sum()
        if s > 0.0:
            child /= s
        else:
            child = np.full(n, 1.0 / n)

        # ----- mutation (mass‑transfer + Gaussian jitter) -----
        if rng.random() < 0.6:
            # mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i != j and child[i] > 0.0:
                frac = rng.random() * 0.4
                delta = child[i] * frac
                child[i] -= delta
                child[j] += delta
        if rng.random() < 0.4:
            child = add_noise(child, sigma=rng.uniform(0.001, 0.01), rng=rng)

        child = np.maximum(child, 0.0)
        child_sum = child.sum()
        if child_sum > 0.0:
            child /= child_sum
        else:
            child = np.full(n, 1.0 / n)

        child_score = fast_score(child, fft_len)

        # ----- replace worst if better -----
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx]:
            pop[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_a = child.copy()

    return best_a, best_score

    maxiter_per_lam = 2000
    print("[L‑BFGS‑fine] start")
    a_fine, sc_fine = lbfgs_cascade(
        a_start=a_large,
        lam_sched=lam_fine,
        maxiter_per_lam=maxiter_per_lam,
        deadline=fine_deadline,
        fft_len=fft_len_large,
        n=n_large
    )
    print(f"[L‑BFGS‑fine] fast_score = {sc_fine:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient descent (exact max)
    # --------------------------------------------------------------
    eg_deadline = start_time + t_ga + t_fine_lbfgs + t_eg
    print("[exp‑grad] start")
    a_eg, sc_eg = exp_gradient_optimize(
        a_start=a_fine,
        deadline=eg_deadline,
        n=n_large,
        fft_len=fft_len_large,
        rng=rng,
        eta0=20.0
    )
    print(f"[exp‑grad] fast_score = {sc_eg:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – greedy local refinement (mass‑transfer hill‑climb)
    # --------------------------------------------------------------
    local_deadline = deadline
    print("[local‑refine] start")
    a_final, sc_final = local_refinement(
        a_start=a_eg,
        start_score=sc_eg,
        deadline=local_deadline,
        n=n_large,
        fft_len=fft_len_large,
        rng=rng
    )
    print(f"[local‑refine] fast_score = {sc_final:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided scorer
    # --------------------------------------------------------------
    final_seq = a_final.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    # store globally for potential warm‑starts in a later run
    globals()['height_sequence_1'] = final_seq
    return final_seq
