def adam_optimize_max(y_start, deadline, n, fft_len, lr_init=0.05):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_y = y.copy()
    best_score = fast_score(np.exp(y - np.max(y)), fft_len)  # compute a from softmax?
    while time.time() < deadline:
        t += 1
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        # compute conv and mask gradient
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(float)
        # gradient wrt a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1:n-1+n]
        # chain rule to y
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # learning rate schedule: maybe decaying
        lr = lr_init / (1 + 0.0001 * t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate
        a_new = np.exp(y - np.max(y))
        a_new /= a_new.sum()
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_y = y.copy()
            best_a = a_new

def beta_initial(n, rng):
    a = rng.uniform(0.2,5.0)
    b = rng.uniform(0.2,5.0)
    x = np.linspace(0,1,n)
    vals = (x**(a-1))*((1-x)**(b-1))
    if vals.sum() == 0:
        vals = np.ones(n)
    vals = np.maximum(vals,0)
    vals /= vals.sum()
    return vals

"""
Optimiser for the step‑function bound (improved).

The search runs several phases within the time budget:
 1. Build a diverse pool of initial probability vectors (windows,
    random families, Beta distributions, etc.).
 2. Run an aggressive L‑BFGS cascade on a smooth log‑sum‑exp
    surrogate of the max‑convolution (decreasing λ‑schedule).
 3. Refine with Adam on the same surrogate.
 4. Optimise the exact max‑convolution using Adam on the
    piece‑wise‑linear objective (sub‑gradient from the active set).
 5. Short simulated‑annealing polishing.
The best vector (normalised to sum = 1) is returned; its score is
also printed via the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def fast_score(a: np.ndarray, fft_len: int) -> float:
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# initial families ---------------------------------------------------------

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a

def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a

def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a

def sinc_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n) - (n - 1) / 2.0
    scale = rng.uniform(0.05, 0.15) * n
    a = np.sinc(x / scale) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def triangular_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    peak_frac = rng.uniform(0.2, 0.6)
    peak = int(peak_frac * n)
    if peak == 0:
        peak = 1
    left = np.linspace(0, 1, peak, endpoint=False)
    middle_len = max(0, n - 2 * peak)
    middle = np.ones(middle_len)
    right = np.linspace(1, 0, peak, endpoint=True)
    a = np.concatenate([left, middle, right])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def blackman_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    try:
        from scipy.signal import blackman
    except Exception:
        return random_initial(n, rng)
    w = blackman(n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def beta_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    a_param = rng.uniform(0.2, 5.0)
    b_param = rng.uniform(0.2, 5.0)
    x = np.linspace(0, 1, n)
    vals = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals

# smoothed objective -------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max
    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

# L‑BFGS cascade -----------------------------------------------------------

def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Adam on smoothed surrogate ----------------------------------------------

def adam_optimize(y_start: np.ndarray,
                  lam_sched,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  lr: float = 0.05):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)
    for lam in lam_sched:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            f, grad_y = objective_and_grad(y, n, fft_len, lam)
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

# Adam on exact max‑convolution -------------------------------------------

def adam_optimize_max(y_start: np.ndarray,
                      deadline: float,
                      n: int,
                      fft_len: int,
                      lr0: float = 0.2):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    a = np.exp(y - np.max(y))
    a /= a.sum()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    while time.time() < deadline:
        t += 1
        a = np.exp(y - np.max(y))
        a /= a.sum()
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)
        # gradient w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # simple decay of learning rate
        lr = lr0 / (1 + 1e-4 * t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # evaluate
        a_curr = np.exp(y - np.max(y))
        a_curr /= a_curr.sum()
        score = fast_score(a_curr, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()
    return best_a, best_score

# Simulated annealing -----------------------------------------------------

def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)
        move_type = rng.choice(['transfer', 'swap', 'local'])
        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # transfer
                j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True
        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score

"""
Optimiser for the step‑function bound (improved).

The search runs inside the given time budget in several phases:
 1. Build a diverse pool of initial probability vectors (windows,
    random families, Beta‑distributions, etc.) and keep the best.
 2. Aggressive L‑BFGS cascade on a smooth log‑sum‑exp surrogate of the
    max‑convolution (decreasing λ‑schedule).
 3. Adam on the same surrogate with a finer λ‑schedule.
 4. Adam directly on the exact max‑convolution (sub‑gradient from the
    active set) – this usually yields the strongest decrease.
 5. A short fine‑L‑BFGS run on the surrogate (tiny λ) for last‑minute
    polishing.
 6. Very short simulated‑annealing polishing.

The final (normalised) vector is returned; its true score is printed via
the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a already sums to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


def sinc_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n) - (n - 1) / 2.0
    scale = rng.uniform(0.05, 0.15) * n
    a = np.sinc(x / scale) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def triangular_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    peak_frac = rng.uniform(0.2, 0.6)
    peak = int(peak_frac * n)
    if peak == 0:
        peak = 1
    left = np.linspace(0, 1, peak, endpoint=False)
    middle_len = max(0, n - 2 * peak)
    middle = np.ones(middle_len)
    right = np.linspace(1, 0, peak, endpoint=True)
    a = np.concatenate([left, middle, right])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def blackman_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    try:
        from scipy.signal import blackman
    except Exception:
        return random_initial(n, rng)
    w = blackman(n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def beta_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    a_param = rng.uniform(0.2, 5.0)
    b_param = rng.uniform(0.2, 5.0)
    x = np.linspace(0, 1, n)
    vals = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y lives in ℝⁿ; a = softmax(y) belongs to the probability simplex.
    """
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)           # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                                 # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]               # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS cascade on the smoothed surrogate
# ----------------------------------------------------------------------


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_sched,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  lr: float = 0.05):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best probability vector encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            f, grad_y = objective_and_grad(y, n, fft_len, lam)
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

def adam_optimize_max(y_start: np.ndarray,
                      deadline: float,
                      n: int,
                      fft_len: int,
                      lr0: float = 0.2):
    """
    Adam optimiser on the true objective max(conv(a,a)).
    Works in the log‑space (soft‑max) to keep the simplex constraints.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    a = np.exp(y - np.max(y))
    a /= a.sum()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        t += 1
        a = np.exp(y - np.max(y))
        a /= a.sum()

        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # gradient w.r.t. a for the (sub‑)gradient of max(conv)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # Adam update on y
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # simple decay of learning rate
        lr = lr0 / (1 + 1e-4 * t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate current solution
        a_curr = np.exp(y - np.max(y))
        a_curr /= a_curr.sum()
        score = fast_score(a_curr, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()
    return best_a, best_score

def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types.
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25            # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # transfer
                j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True
        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score

            pool.append(w)

            pass
        try:
            for alpha in (0.3, 0.5, 0.7):
                w = tukey(n, alpha)
                w = np.maximum(w, 0.0)
                w /= w.sum()
                pool.append(w)
        except Exception:
            pass

    # random families
    for _ in range(10):
        pool.append(random_initial(n, rng))

    # parametric families
    for _ in range(5):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))
        pool.append(sinc_initial(n, rng))
        pool.append(triangular_initial(n, rng))
        pool.append(beta_initial(n, rng))
        pool.append(blackman_initial(n, rng))

    # ------------------------------------------------------------------
    # Keep the best of the initial pool
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in pool:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print("[pool] best initial fast_score =", best_score)

    # ------------------------------------------------------------------
    # Time budget splits
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.01          # safety margin
    phase1_end = start_time + 0.20 * budget_s        # L‑BFGS cascade
    phase2_end = start_time + 0.50 * budget_s        # Adam (smooth)
    phase3_end = start_time + 0.85 * budget_s        # Adam (exact)
    phase4_end = start_time + 0.97 * budget_s        # final polishing

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed surrogate
    # ------------------------------------------------------------------
    lam_schedule_l1 = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.02, 0.015, 0.01, 0.007,
        0.005, 0.003, 0.002, 0.001, 5e-4,
        2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.3:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.006),
                                rng=rng)
        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule_l1,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed surrogate (fine λ‑schedule)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule_adam = [
            5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
            5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7,
            5e-8, 2e-8, 1e-8
        ]
        iters_per_lam = 300
        y_start = np.log(best_a + 1e-12)
        a_adam, sc_adam = adam_optimize(y_start,
                                         lam_schedule_adam,
                                         iters_per_lam,
                                         phase2_end,
                                         fft_len,
                                         n,
                                         lr=0.07)
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam.copy()
            print(f"[Adam‑smooth] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        y_start = np.log(best_a + 1e-12)
        a_exact, sc_exact = adam_optimize_max(y_start,
                                              phase3_end,
                                              n,
                                              fft_len,
                                              lr0=0.2)
        if sc_exact < best_score:
            best_score = sc_exact
            best_a = a_exact.copy()
            print(f"[Adam‑exact] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3.5 – final fine L‑BFGS on the smoothed surrogate (tiny λ)
    # ------------------------------------------------------------------
    if time.time() < phase4_end - 5.0:
        fine_lam_sched = [5e-8, 2e-8, 1e-8, 5e-9, 2e-9, 1e-9]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=800,
                                        deadline=phase4_end - 2.0,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing_improved(
            best_a,
            best_score,
            phase4_end,
            n,
            rng
        )
        print(f"[anneal] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Optional mixing of the top candidates (tiny extra polish)
    # ------------------------------------------------------------------
    candidates = [best_a]
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.shape == best_a.shape:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            candidates.append(prev)
    for i in range(len(candidates)):
        for j in range(i + 1, len(candidates)):
            for _ in range(7):
                w = rng.random()
                mix = w * candidates[i] + (1 - w) * candidates[j]
                mix = np.maximum(mix, 0.0)
                mix /= mix.sum()
                sc = fast_score(mix, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = mix.copy()
                    print(f"[mix] new best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer and expose the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
