"""
Step‑function optimiser – 1000 s budget (enhanced)

This implementation follows the 4‑phase strategy of the reference
solution but adds a systematic *parameter sweep* for classic families
(edge‑power, edge‑exponential, Gaussian, cosine‑power) to obtain a
much better starting point.  After keeping the top candidates we run
a short, aggressive L‑BFGS optimisation on a smoothed log‑sum‑exp
surrogate, a long exact sub‑gradient (mirrored exponentiated‑gradient)
descent, a fine L‑BFGS pass with ultra‑small smoothing, and finally a
gradient‑guided hill‑climb that moves mass from indices with large
sub‑gradient to those with small sub‑gradient.  The best sequence
found is stored in ``height_sequence_1`` and returned.

All steps respect the global deadline (default 1000 s) and use only
NumPy / SciPy.  The code is self‑contained and can be called as
``propose_candidate()``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for ``evaluate_sequence`` when ``sum(a)=1``:
    returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Initial families (deterministic parameter sweep)
# ----------------------------------------------------------------------


def edge_power_initial(n: int, p: float) -> np.ndarray:
    """U‑shaped power law: mass ∝ (dist_to_edge+1)^(-p)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1.0) ** (-p)
    a = a / a.sum()
    return a


def edge_exponential_initial(n: int, rate: float) -> np.ndarray:
    """U‑shaped exponential: mass ∝ exp(-rate·dist_to_edge)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-rate * d)
    a = a / a.sum()
    return a


def gaussian_initial(n: int, sigma_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density; sigma = sigma_frac·n."""
    sigma = sigma_frac * n
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = a / a.sum()
    return a


def cosine_power_initial(n: int, p: float) -> np.ndarray:
    """Cosine window raised to a power p (clipped at zero)."""
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a = a / a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A handful of stochastic families used as a safety net."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    ``y`` is unrestricted; ``a = softmax(y)`` lives on the simplex.
    """
    # softmax (stable)
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)          # safe (≤0)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                # d smooth_max / d conv_i
    # back‑propagation through a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]      # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best distribution found (sum=1) and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in unconstrained log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (mirrored exponentiated gradient)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored (multiplicative) sub‑gradient descent on the true L∞‑objective.
    The update: a ← a * exp(-η·g) followed by simplex projection.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (mirrored version)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update with clipping for safety
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing guided by the sub‑gradient
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Hill‑climb that (i) moves mass from the index with the largest
    sub‑gradient to the one with the smallest, and (ii) occasionally
    tries a completely random transfer.  Only strictly improving
    moves are kept.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20            # up to 20 % of source mass may be moved

    while time.time() < deadline:
        # ---------- compute exact max‑convolution and sub‑gradient ----------
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = np.real(np.fft.ifft(fm * fa_rev))[n - 1: n - 1 + n] * 2.0

        # ---------- deterministic move (gradient guidance) ----------
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if a[src] > 0.0:
            delta = a[src] * rng.random() * max_frac
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            new_score = fast_score(cand, fft_len)
            if new_score < cur_score:
                a = cand
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = cand.copy()
                continue   # go to next iteration

        # ---------- random move (fallback) ----------
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 is a good compromise between quality and speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Global deadline
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # small safety margin

    # --------------------------------------------------------------
    # Build a diverse pool – deterministic sweep + random families
    # --------------------------------------------------------------
    pool = []

    # 0) Bring forward the previous best if present (interpolate)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a_old = np.interp(x_new, x_old, old)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            pool.append(a_old)

    # 1) Systematic families
    # Edge‑power
    for p in np.linspace(0.1, 3.0, 30):
        pool.append(edge_power_initial(n, p))

    # Edge‑exponential
    for rate in np.linspace(0.001, 0.02, 30):
        pool.append(edge_exponential_initial(n, rate))

    # Gaussian
    for sigma_frac in np.linspace(0.05, 0.5, 15):
        pool.append(gaussian_initial(n, sigma_frac, rng))

    # Cosine‑power
    for p in np.linspace(0.5, 5.0, 15):
        pool.append(cosine_power_initial(n, p))

    # 2) Random families (adds stochastic diversity)
    for _ in range(15):
        pool.append(random_initial(n, rng))

    # --------------------------------------------------------------
    # Evaluate all candidates, keep the 20 best (by fast_score)
    # --------------------------------------------------------------
    scored = [(fast_score(a, fft_len), a) for a in pool]
    scored.sort(key=lambda tup: tup[0])
    top_k = 20
    best_candidates = scored[:top_k]

    best_score, best_a = best_candidates[0]       # best among the sweep
    print(f"[init] best fast_score from sweep = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 1 – quick L‑BFGS on a smoothed surrogate
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.25, 0.12, 0.06, 0.03,
        0.015, 0.008, 0.004, 0.002, 0.001,
        5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5
    ]
    phase1_deadline = start_time + 130.0          # allocate ~130 s
    while time.time() < phase1_deadline:
        # Pick a random elite candidate as the seed
        a0 = rng.choice([c[1] for c in best_candidates])
        # Small jitter to avoid stagnation
        if rng.random() < 0.35:
            a0 = add_noise(a0, sigma=rng.uniform(0.001, 0.008), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a0,
            lam_schedule,
            maxiter_per_lam=500,
            deadline=phase1_deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS‑quick] new best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – long exact max‑convolution exponentiated‑gradient descent
    # --------------------------------------------------------------
    phase2_deadline = start_time + 730.0          # about 730 s total for EG
    best_a, best_score = exp_gradient_optimize(
        best_a,
        phase2_deadline,
        n,
        fft_len,
        rng,
        eta0=18.0,
    )
    print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with ultra‑small λ
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        fine_lam_sched = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_fine, sc_fine = lbfgs_cascade(
            best_a,
            fine_lam_sched,
            maxiter_per_lam=1500,
            deadline=deadline - 30.0,
            fft_len=fft_len,
            n=n,
        )
        if sc_fine < best_score:
            best_score, best_a = sc_fine, a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – gradient‑guided greedy local refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = local_refinement(
            best_a,
            best_score,
            deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[local‑refine] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
