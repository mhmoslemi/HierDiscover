"""
Step‑function optimiser (≈ 1000 s budget)

Algorithm
---------
1. Warm‑start – generate a large pool of diverse *non‑symmetric* shapes,
   including monotone decreasing families (exponential, power‑law,
   linear).  All candidates are normalised to a probability vector
   (∑a=1) and scored with a fast FFT‑based proxy
   `fast_score(a)=2·n·max(conv(a,a))`.

2. L‑BFGS cascade – the best warm‑start is refined by a cascade of
   L‑BFGS minimisations on the smooth log‑sum‑exp surrogate of the
   L∞‑objective.  A fine λ‑schedule is used (down to 1e‑07) and a large
   iteration budget per λ.

3. Sub‑gradient descent with exact max‑convolution – a projected
   (additive) sub‑gradient method is run on the *true* max‑convolution
   objective.  The update direction is obtained from the mask of the
   active (max‑attaining) lags, a simple back‑tracking line‑search is
   performed and the iterate is projected back onto the simplex.

4. Final polishing – a short low‑λ L‑BFGS pass, a forced decreasing
   ordering (sorting the vector) and a brief simulated‑annealing
   hill‑climb finish the optimisation.

All stages respect a global 1000 s deadline.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned as
a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.shape[0]

    # sort v in decreasing order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation (including monotone families)
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Create a diverse non‑negative probability vector.
    Families include uniform, edge‑symmetric, and several monotone
    decreasing shapes.
    """
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "linear_decay",
            "exp_decay",
            "power_decay",
            "sparse",
            "mixed",
        ]
    )
    x = np.arange(n, dtype=float)

    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        # symmetric exponential decay from both edges
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        # symmetric power‑law decay from edges
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "linear_decay":
        # strictly decreasing linear ramp (large at left, zero at right)
        a = (n - x) / (n * (n + 1) / 2.0)

    elif mode == "exp_decay":
        # monotone decreasing exponential from the left end
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * x)

    elif mode == "power_decay":
        # monotone decreasing power law from the left end
        p = rng.uniform(0.5, 3.0)
        a = (x + 1) ** (-p)

    elif mode == "sparse":
        # a few random spikes (mostly near the left to encourage monotonicity)
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    else:   # mixed
        # 60 % uniform + 40 % edge‑exponential
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    # avoid exact zeros, normalise to a probability vector
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    max_iters: int = 3000,
):
    """
    Projected additive sub‑gradient descent on the exact L∞–objective.
    Uses a simple back‑tracking line‑search and projection onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- compute convolution and active mask -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- exact sub‑gradient of max(conv) -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]   # length n

        # ----- line‑search on step‑size η -----
        eta = 0.6 / np.sqrt(it + 1.0)   # initial guess
        for _ in range(5):
            cand = a - eta * g
            cand = proj_simplex(cand)
            cand_score = fast_score(cand, fft_len)
            if cand_score < best_score:
                # accept
                a = cand
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                break
            eta *= 0.5   # shrink step if not improving
        # if none of the back‑tracking steps improved we keep the old a
        if it % 200 == 0:
            print(f"[subgrad] iter={it} best fast_score={best_score:.9f}")
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polish)
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score
