"""
Propose a low‑evaluation step‑function.

Algorithm (≈1000 s budget)
---------------------------------
1. Build a rich pool of candidate shapes (windows, random families,
   Gaussian, cosine, power‑law, previous best interpolated to the new length).
2. Phase 1 – short L‑BFGS cascades on a log‑sum‑exp smooth approximation
   of max‑convolution.  Many random restarts, moderate λ‑schedule.
3. Phase 2 – long mirror‑SGD (exponentiated‑gradient) on the *exact* L∞‑objective.
   Uses a momentum term, adaptive step‑size and occasional Gaussian jitter.
4. Phase 3 – final polish:
   * symmetrisation,
   * a very low‑λ L‑BFGS cascade,
   * a short simulated‑annealing hill‑climb (larger mass‑transfer moves).
5. Return the best vector (normalised to sum = 1).  The official scorer is
   called once at the end.

All timings respect a global 1000 s deadline; the function never raises
exceptions and always returns a valid list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the official score when sum(a)=1:
       2·n·max(conv(a,a))  (convolution via FFT)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
            "gaussian",
            "cosine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "two_block":
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    elif mode == "gaussian":
        sigma = rng.uniform(0.1 * n, 0.3 * n)
        x = np.arange(n)
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    elif mode == "cosine":
        # raised cosine (Hann‑like) window
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    else:  # fallback
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b_sum = b.sum()
    if b_sum == 0.0:
        return a.copy()
    return b / b_sum


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. log‑variables y.
    a = softmax(y)   (so that a is a probability vector).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam,
                         "gtol": 1e-12,
                         "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑SGD (exponentiated‑gradient) on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          max_iters: int = 10_000_000):
    """
    Mirror‑SGD on the exact max‑convolution.
    Uses momentum on the sub‑gradient, adaptive step‑size and jitter.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # momentum buffer
    g_avg = np.zeros_like(a)
    beta = 0.9

    it = 0
    last_improve_it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and mask of (almost) maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size (square‑root decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # occasional evaluation
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve_it = it
                print(f"[exp‑grad] it={it} fast_score={best_score:.9f}")

        # jitter if stagnating
        if (it - last_improve_it) > 300_000:
            a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)
            last_improve_it = it

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (simple but effective)
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25        # up to 25 % of src mass may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-14)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderately large vector gives a richer search
    # ------------------------------------------------------------------
    n = 6000                               # a compromise between quality & speed
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Interpolate the previous best, if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            if a0_sum == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0_sum
            pool.append(a0)

    # 2) Classical windows from SciPy (if available)
    try:
        from scipy.signal import chebwin, kaiser, hann, blackman
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        w = hann(n)
        pool.append(w / w.sum())
        w = blackman(n)
        pool.append(w / w.sum())
    except Exception:
        pass

    # 3) Randomly generated families (including the new modes)
    for _ in range(20):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    phase1_end = start_time + 0.12 * budget_s   # quick L‑BFGS warm‑up
    phase2_end = start_time + 0.85 * budget_s   # long EG phase
    # remaining → final polish & anneal

    # ------------------------------------------------------------------
    # Phase 1 – quick L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.022, 0.012, 0.006, 0.003,
        0.0015, 7e-4, 3e-4, 1.5e-4,
        8e-5, 4e-5, 2e-5, 1e-5
    ]
    maxiter_per_lam = 1000

    best_a = None
    best_score = float("inf")
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter on the start vector
        if rng.random() < 0.4:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.012), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best fast_score={best_score:.9f}")

        # occasional symmetrisation trial
        if rng.random() < 0.15:
            sym = symmetrize(a_opt)
            sym_score = fast_score(sym, fft_len)
            if sym_score < best_score:
                best_score = sym_score
                best_a = sym.copy()
                print(f"[Phase1‑sym] improved fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (mirror‑SGD)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=35.0,                # slightly larger initial step
            max_iters=50_000_000,     # generous; loop ends by deadline
        )
        print(f"[Phase2] exp‑grad best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final refinement
    # ------------------------------------------------------------------
    # 3a – enforce symmetry (often beneficial)
    sym_a = symmetrize(best_a)
    sym_score = fast_score(sym_a, fft_len)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        print(f"[Phase3‑sym] symmetry improved fast_score={best_score:.9f}")

    # 3b – very low‑λ L‑BFGS cascade
    if time.time() < deadline:
        fine_lam = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase3‑LBFGS] fast_score={best_score:.9f}")

    # 3c – optional smoothing before annealing
    if time.time() < deadline:
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = np.maximum(smoothed, 0.0)
        sm_tot = smoothed.sum()
        if sm_tot > 0.0:
            smoothed /= sm_tot
            sm_score = fast_score(smoothed, fft_len)
            if sm_score < best_score:
                best_a = smoothed
                best_score = sm_score
                print(f"[Phase3‑smooth] fast_score={best_score:.9f}")

    # 3d – simulated annealing hill‑climb
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
