    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # compute convolution via FFT
        Fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(Fa*Fa, fft_len)[:2*n-1]
        idx = int(np.argmax(conv))

        high = min(idx, n - 1)
        # length = high - low + 1

        g = np.zeros_like(a)
        # positions i = low..high map to a[idx - i]
        # Use integer array

"""
Propose a low‑score step sequence.

The method follows three stages:
1.  Construct a rich pool of candidate shapes (windows, Beta‑densities,
    raised‑cosines, random draws) and keep the best according to the
    exact fast score 2·n·max(conv).
2.  Run a long projected sub‑gradient descent on the true max‑convolution
    (the objective is max_k (a*a)[k]).  The gradient is obtained analytically
    for the active index, a zero‑sum direction is taken, and the iterate is
    projected back onto the probability simplex.
3.  Finish with a few L‑BFGS refinements of a smoothed
    log‑sum‑exp version of the max, using a decreasing smoothing factor.
    The final sequence is returned and also stored in the global variable
    `height_sequence_1` for possible future runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    Implementation follows the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (used for ranking candidates)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum = 1).
    Computed via FFT for O(n log n) time.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral guess – random phase, projected onto simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    # uniform magnitude, random phase
    phase = np.random.uniform(0, 2 * np.pi, L // 2 + 1)
    S = np.exp(1j * phase)
    S[0] = 0.0  # zero‑mean to avoid a huge DC component
    a_complex = np.fft.ifft(S, L)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 -
         0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        step0: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # compute full self‑convolution
        Fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(Fa * Fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())               # index of the current peak

        # gradient of the active convolution entry
        low = max(0, idx - (n - 1))
        high = min(idx, n - 1)
        g = np.zeros_like(a)
        # indices i in [low, high] contribute a[idx-i] to the gradient
        i_range = np.arange(low, high + 1)
        g[low:high + 1] = 2.0 * a[idx - i_range]

        # zero‑sum direction (preserves the simplex sum)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS polishing on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam,
                 deadline,
                 fft_len_conv,
                 fft_len_g,
                 n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result. Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence (list of floats) found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # ------------------------------------------------------------------
    # Problem size – a bit larger than the original 5000 to give more
    # degrees of freedom; the FFT length is a power of two for speed.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an extensive pool of initial candidates.
    # ------------------------------------------------------------------
    candidates = []

    # Use a previously discovered sequence if available.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))
    candidates.append(kaiser_init(n, beta=8.0))

    # Randomised families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # Systematic sweep over Beta and raised‑cosine families
    for _ in range(400):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))

    for _ in range(200):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    # ------------------------------------------------------------------
    # Pick the best candidate by the exact fast_score.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase A – very long projected sub‑gradient descent.
    # Allocate roughly 60 % of the total time.
    # ------------------------------------------------------------------
    phaseA_end = start_time + 0.60 * budget_s
    best_a, best_score = subgradient_descent(best_a,
                                             phaseA_end,
                                             n,
                                             fft_len,
                                             step0=0.45)
    print(f"[sub‑grad] after phase A best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase B – L‑BFGS polishing with a decreasing smoothing factor.
    # Allocate about 30 % of the total time.
    # ------------------------------------------------------------------
    phaseB_end = start_time + 0.90 * budget_s
    if time.time() < phaseB_end:
        lam_schedule = [
            0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
            0.05, 0.03, 0.018, 0.01, 0.006,
            0.003, 0.0015, 0.0008, 0.0004, 0.0002,
            0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
            2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
            5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09
        ]
        best_a, best_score = lbfgs_refine(best_a,
                                          lam_schedule,
                                          maxiter_per_lam=2000,
                                          deadline=phaseB_end,
                                          fft_len_conv=fft_len,
                                          fft_len_g=fft_len,
                                          n=n)
        print(f"[L‑BFGS] after phase B best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase C – final greedy hill‑climb polishing (lightweight).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # simple pairwise mass‑transfer hill‑climb
        cur_score = best_score
        max_iter = 60000
        it = 0
        while time.time() < deadline and it < max_iter:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                it += 1
                continue
            delta = best[i] * rng.random() * 0.07
            if delta <= 0.0:
                it += 1
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = fast_score(cand, fft_len, n)
            if sc < cur_score:
                cur_score = sc
                best_a = cand
                best_score = sc
            it += 1
        print(f"[hill‑climb] final best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (the official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the best sequence for potential future runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
