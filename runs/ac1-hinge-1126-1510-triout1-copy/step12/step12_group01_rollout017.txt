"""
Propose a low‑score step function (n≈4096).

Algorithm
---------
1. Initialise a pool of diverse candidates (spectral, windows, random Dirichlet,
   power‑law, etc.) and keep the best by the *exact* objective
   `fast_score(a) = 2·n·max(a*a)` (a is normalised to sum = 1).

2. Repeatedly launch a short L‑BFGS “continuation’’ from a random blend of the
   current best and a fresh candidate.  The smoothed‑max objective is the
   log‑sum‑exp version (parameter λ) – we drive λ from 0.4 down to 5·10⁻⁶,
   performing several thousand L‑BFGS steps per λ.  Each run is limited by the
   global deadline; the best vector of all runs is retained.

3. Refine the best vector with three first‑order methods:
   * Exponentiated‑gradient (EG) on the exact max‑convolution, updating only
     the top 0.5 % of convolution entries.
   * Adam on the true (non‑smoothed) objective, working in the log‑soft‑max
     space to keep the simplex constraint.
   * A second, longer Adam pass (lower learning‑rate) for fine‑tuning.

4. Local search:
   * Simulated‑annealing with cheap pairwise mass transfers.
   * Deterministic hill‑climb that always accepts a strictly improving move.

5. Return the best sequence found (as a plain list).  The routine obeys the
   1000 s time budget and never exceeds the per‑element limits (≥0, ≤1000).

The whole procedure is deterministic apart from the random generator seed,
so it can be reproduced by re‑running with the same seed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (assumes sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact 2·n·max(a∗a) when a is on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp version)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(y: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         fft_len_grad: int,
                         lam: float):
    """Return (f, ∇_y f) where f = 2·n·smooth_max(conv)."""
    a = softmax(y)                                 # a lives on the simplex

    # ---------- convolution a * a ----------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    # ---------- smoothed max ----------
    c_max = conv.max()
    lam_eff = c_max * lam + 1e-12

    shifted = conv - c_max                         # ≤0
    exp_shift = np.exp(shifted / lam_eff)          # (0,1]
    Z = exp_shift.sum()
    smooth_max = c_max + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient w.r.t. convolution ----------
    w = exp_shift / Z                               # weights, sum to 1
    # back‑prop: d conv / d a = rev(a) * w  (linear convolution)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_grad)
    fw = np.fft.rfft(w, fft_len_grad)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_grad)

    # gradient w.r.t. a (size n, taken from the centre of conv_rev)
    g_a = conv_rev[(n - 1):(n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_a)
    grad_y = a * (g_a - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Initialisers (families of probability vectors)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Take a flat spectrum, sqrt‑phase, inverse FFT, clip negatives."""
    L = next_pow2(2 * n - 1)
    # flat power spectrum (all ones) -> magnitude = 1
    target = np.zeros(L)
    target[:2 * n - 1] = 1.0
    # random phase, symmetrical to obtain a real signal
    rng = np.random.default_rng(0xC0FFEE)
    phase = rng.uniform(0, 2 * np.pi, L // 2 + 1)
    phasor = np.exp(1j * phase)
    # enforce Hermitian symmetry
    spec = np.zeros(L, dtype=np.complex128)
    spec[0] = 1.0
    spec[1:L // 2] = phasor[1:-1]
    spec[L // 2] = phasor[-1]
    spec[L // 2 + 1:] = np.conj(spec[1:L // 2][::-1])
    a_complex = np.fft.ifft(spec).real[:n]
    a_complex[a_complex < 0] = 0.0
    if a_complex.sum() == 0:
        a_complex = np.full(n, 1.0 / n)
    else:
        a_complex /= a_complex.sum()
    return a_complex


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Various random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.03)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.1, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        frac = rng.uniform(0.003, 0.02)
        m = max(1, int(n * frac))
        a = np.full(n, 1e-6)
        heavy = 0.5 / m
        a[:m] = heavy
        a[-m:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.01 * n), int(0.05 * n))
        peak = rng.uniform(0.2, 1.0) / n
        a = np.full(n, 1e-6)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS continuation (smooth max) – returns best found in this run
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_grad: int,
                       n: int):
    """Run L‑BFGS for each λ in `lam_schedule`.  Return best (a,score)."""
    y = np.log(a_start + 1e-12)          # initialise logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=_smooth_obj_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_grad, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.005):
    """EG on the true max‑convolution (only top % of entries are weighted)."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # mask of the largest entries
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient w.r.t. a (via reversed a)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        # step size η ~ 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        # EG update (stay on the simplex automatically)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam on the exact max‑convolution objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lr0: float = 0.3,
                  decay: float = 0.5):
    """Adam on the true objective; works in log‑soft‑max space."""
    y = np.log(a_start + 1e-12)          # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # exact convolution and its (sub‑)gradient at the maximal entry
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a                     # ∂(2·n·max)/∂a

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = lr0 / np.sqrt(t) * decay
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a_curr.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated annealing (pairwise mass transfer)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 4e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200_000):
    """Simple SA that swaps tiny fractions of mass between two indices."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        # linear temperature schedule
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # move a tiny fraction from i to j
        delta = a[i] * rng.random() * 0.12
        if delta <= 0.0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc, best_a = sc_cand, cand.copy()
        move += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Deterministic hill‑climb (tiny pairwise transfers, only improvements)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120_000,
               delta_frac: float = 0.04):
    """Greedy improvement – always keep a move that lowers the exact score."""
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep a power of two for cheap FFTs.
    # --------------------------------------------------------------
    n = 4096                       # length of the sequence (tunable)
    fft_len = next_pow2(2 * n - 1)  # length for the convolution FFTs

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it to current n
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 /= a0.sum()
                candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # Random families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # Choose the best of this crude pool (fast_score assumes sum=1)
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Repeated L‑BFGS continuations (aggressive)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s   # 30 % of total time
    lam_schedule = [
        0.4, 0.3, 0.2, 0.15, 0.1, 0.07,
        0.05, 0.03, 0.02, 0.01, 0.005,
        0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # blend the current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           fft_len,
                                           n)
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[global L‑BFGS] new global best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient (focus on the true max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + 0.15 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                eg_deadline,
                                                n,
                                                fft_len,
                                                top_fraction=0.005)
        print(f"[EG] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + 0.25 * budget_s
        print("[Adam‑1] start")
        best_a, best_sc = adam_optimize(best_a,
                                        adam_deadline,
                                        n,
                                        fft_len,
                                        rng,
                                        lr0=0.3,
                                        decay=0.6)
        print(f"[Adam‑1] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Second Adam pass (slower learning rate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam2_deadline = time.time() + 0.15 * budget_s
        print("[Adam‑2] start")
        best_a, best_sc = adam_optimize(best_a,
                                        adam2_deadline,
                                        n,
                                        fft_len,
                                        rng,
                                        lr0=0.12,
                                        decay=0.5)
        print(f"[Adam‑2] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(12.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=4e-4,
                                              final_temp=2e-7,
                                              max_moves=150_000)
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 20.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=180_000,
                                     delta_frac=0.035)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
