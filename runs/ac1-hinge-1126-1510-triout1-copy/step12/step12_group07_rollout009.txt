"""
Propose a step‑function sequence that minimizes the evaluation score.

Algorithm
---------
1. Initialise with the best known sequence (global ``height_sequence_1``) if
   available; otherwise generate a smooth Gaussian‑shaped density.
2. Normalise to a probability vector (sum = 1).
3. Run a long exponentiated‑gradient (mirror‑descent) optimisation on the
   *exact* objective:
      – compute the self‑convolution via a real FFT,
      – locate the (few) indices attaining the maximal convolution value,
      – obtain a sub‑gradient of the max‑convolution,
      – update the probabilities multiplicatively with a step size ηₜ = η₀/√t,
      – re‑project onto the simplex (by normalisation).
   The update automatically reduces the maximal convolution.
4. Occasionally enforce symmetry, add tiny Gaussian jitter and decay the
   learning‑rate when progress stalls – these cheap perturbations help to
   escape plateaus.
5. After the time budget expires, evaluate the best vector with the official
   ``evaluate_sequence`` scorer and store it globally as ``height_sequence_1``.
   The function returns the final list of heights.

The optimisation runs almost entirely in the exponentiated‑gradient loop,
allowing many thousands of iterations within the 1000 s budget, which is
typically sufficient to improve the previous best bound (≈ 1.5063) towards the
target ≤ 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a already sums to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep the size of the previous best (≈4000) to stay fast.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = old.shape[0]
        a = np.maximum(old, 0.0)
        if a.sum() <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a.sum()
    else:
        # a smooth Gaussian start if no previous solution is present
        n = 4000
        sigma = rng.uniform(0.2 * n, 0.5 * n)
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, initial fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Exponentiated‑gradient descent on the *exact* max‑convolution.
    # ------------------------------------------------------------------
    eta0 = 80.0                # initial learning‑rate multiplier
    it = 0
    last_improve = 0          # iteration index of last improvement
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    while time.time() < deadline:
        # ---- 1. compute self‑convolution and locate the active set ----
        fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        max_val = conv_full.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv_full >= max_val - eps).astype(np.float64)   # 0/1 mask

        # ---- 2. sub‑gradient of the max‑convolution --------------------
        a_rev = a[::-1]
        fm = np.fft.rfft(mask, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(fm * fa_rev, n=fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]   # size n

        # ---- 3. multiplicative EG update --------------------------------
        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * grad, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ---- 4. evaluate current vector ---------------------------------
        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[EG] it={it:<8d} new best fast_score = {best_score:.9f}")

        # ---- 5. occasional stabilising operations -----------------------
        if it % 5000 == 0 and it > 0:
            # enforce (approximate) symmetry – often reduces the peak
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
        if it % 10000 == 0 and it > 0:
            # tiny random jitter to help escape flat regions
            noise = rng.normal(0.0, 0.001, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # ---- 6. learning‑rate decay when stuck -------------------------
        if it - last_improve > 200_000:
            # halve the base step size and reset the stall counter
            eta0 *= 0.5
            last_improve = it
            print(f"[EG] stall detected – decreasing η₀ to {eta0:.3f}")

        it += 1

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] official evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
