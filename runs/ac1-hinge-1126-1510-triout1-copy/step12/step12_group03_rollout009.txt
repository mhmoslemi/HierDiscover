"""
Propose a step function with a low evaluation score.

Algorithm
---------
1. Build a small pool of deterministic candidates (arcsine, spectral,
   Gaussian, power‑law).  Keep the one with the smallest exact score.

2. Iteratively improve the candidate by solving a linear program:
   maximise the L1‑norm of a new vector b subject to the bound
   (f * b) ≤ M, where f is the current sequence and M = max(f * f).
   After the LP we rescale b to have the same L1‑norm as f and
   replace f ← (1‑α)·f + α·b (α≈0.5).  This step is repeated until
   no improvement is possible or the budget runs out.

3. Use a cheap exponentiated‑gradient optimiser on the smoothed max‑conv
   objective to polish the result.

4. Finish with a greedy hill‑climb that performs strictly improving
   mass–transfer moves.

All operations respect the 1000‑second time limit; the best sequence
found before the deadline is returned.  The code works with any length
n (here we use n=5000, the size that gave good results previously).
"""

import time
import numpy as np
import cvxpy as cp

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (sum(a)=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def max_conv(a: np.ndarray, fft_len: int, n: int) -> float:
    """Maximum entry of the self‑convolution of a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(conv.max())


# -------------------------------------------------------------
# Candidate families
# -------------------------------------------------------------


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density (clipped, then normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), 1e-16, None))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """A spectral initial guess based on the square‑root of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def power_law_init(n: int, exp: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


# -------------------------------------------------------------
# Linear‑programming improvement step
# -------------------------------------------------------------


def lp_max_sum(f: np.ndarray, M: float, max_iters: int = 1500):
    """
    Solve  max sum(b)  s.t. conv(f, b) ≤ M,  b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = f.shape[0]
    b = cp.Variable(n, nonneg=True)
    # convolution constraint (full length 2n‑1)
    conv_expr = cp.conv(f, b)          # constant f * variable b
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=max_iters, warm_start=True,
                   verbose=False)
    except Exception:                  # pragma: no cover
        return None
    if b.value is None:
        return None
    b_val = np.maximum(np.array(b.value).reshape(-1), 0.0)
    return b_val


# -------------------------------------------------------------
# Simple exponentiated‑gradient optimiser (smoothed max)
# -------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """Smooth max (log‑sum‑exp) of self‑convolution, with gradient w.r.t. logits."""
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.02,
                          lam_factor: float = 0.001,
                          lr: float = 0.07,
                          max_iter: int = 20000):
    """
    Adam‑style optimisation on the smoothed max‑convolution.
    Stops when the deadline is reached.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f_val, grad = objective_and_grad(y, n, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# -------------------------------------------------------------
# Greedy hill‑climb (strictly improving mass transfers)
# -------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.09):
    """
    Greedy hill‑climb that only accepts strictly better moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# -------------------------------------------------------------
# Main entry‑point
# -------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # Problem size (5000 gave the best historical results)
    # ---------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ---------------------------------------------------------
    # Build a small pool of deterministic starts
    # ---------------------------------------------------------
    candidates = [
        arcsine_init(n),
        spectral_init(n),
        gaussian_init(n),
        power_law_init(n, exp=0.5),
        power_law_init(n, exp=1.0),
    ]

    # If a previous solution is available, interpolate it.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0_sum = a0.sum()
            if a0_sum > 0:
                a0 = a0 / a0_sum
                candidates.append(a0)

    # pick the best start according to the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_score:.8f}")

    # ---------------------------------------------------------
    # Time budget handling
    # ---------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ---------------------------------------------------------
    # Phase 1 – LP‑based improvement loop
    # ---------------------------------------------------------
    lp_deadline = start_time + 0.45 * budget_s   # take ~45 % of the budget
    iteration = 0
    while time.time() < lp_deadline:
        M = max_conv(best_a, fft_len, n)

        # solve the linear programme
        b = lp_max_sum(best_a, M, max_iters=1200)
        if b is None:
            print("[LP] solver failed – exiting LP phase")
            break
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # cannot increase the L1‑norm → no further improvement
            break

        # normalise b to have the same L1‑norm as best_a (which is 1)
        g = b / sum_b

        # try a few blending factors and keep the best improvement
        improved = False
        best_local_score = best_score
        best_candidate = None
        best_alpha = None
        for alpha in (0.2, 0.5, 0.8):
            a_new = (1.0 - alpha) * best_a + alpha * g
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()          # re‑normalise (should stay ≈1)
            sc = fast_score(a_new, fft_len, n)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_candidate = a_new
                best_alpha = alpha
                improved = True

        if improved:
            best_a = best_candidate
            best_score = best_local_score
            iteration += 1
            print(f"[LP {iteration}] improved to {best_score:.8f} (α={best_alpha})")
            # continue with the improved vector
        else:
            # no blending helped – stop the LP loop
            break

    # ---------------------------------------------------------
    # Phase 2 – fine‑tune with exponentiated‑gradient (Adam)
    # ---------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   top_fraction=0.04,
                                                   lam_factor=0.001,
                                                   lr=0.07,
                                                   max_iter=25000)
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ---------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ---------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=200000,
                                        delta_frac=0.10)
        print(f"[Hill‑climb] final fast_score={best_score:.8f}")

    # ---------------------------------------------------------
    # Final official evaluation
    # ---------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official routine
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
