"""
Propose a low‑score non‑negative step‑function.

The optimiser works in five stages:

1.  A dense search over a one‑parameter exponential family
    a_i ∝ r**i (r∈(0,1)) for a moderately large length (default 8000).
    The best exponential seed is kept.

2.  A few random‑probability vectors are added to the candidate pool.

3.  Mirror‑descent (exponentiated‑gradient) using the exact
    L∞‑subgradient of the convolution.  The update is
        a ← a·exp(−η·g) ,   a ← a / sum(a)
    where g_i = 2·∑_{j : i+j=k*} a_j and k* is a
    (possibly non‑unique) index attaining the max convolution.
    The step size η decays as η₀/√t.

4.  A short cascade of L‑BFGS on a smooth log‑sum‑exp surrogate
    (very small smoothing λ) to polish the solution.

5.  Random convex combinations of the best candidate found in this run
    and the previous global solution (if any) – a cheap final “mix”.

The routine respects a total wall‑clock budget (default 1000 s) and
stores the best list in the global variable `height_sequence_1`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a already sums to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y ∈ ℝⁿ is the log‑itself; a = softmax(y) lives on the simplex.
    """
    # softmax → a
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)           # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z                                 # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]               # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Core optimiser components
# ----------------------------------------------------------------------


def exp_gradient_descent(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 30.0):
    """
    Exact L∞‑subgradient exponentiated‑gradient descent.
    Returns the best vector and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a (full)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[EG] it {it:06d} new best = {best_sc:.9f}")

        # occasional jitter to escape flat plateaus
        if rng.random() < 0.005:
            a = add_noise(a, sigma=rng.uniform(0.001, 0.004), rng=rng)

        it += 1

    return best_a, best_sc


def lbfgs_fine_tune(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int):
    """
    Very short cascade of L‑BFGS on the smooth surrogate with tiny λ.
    Returns the best vector seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    lam_sched = [1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]
    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'gtol': 1e-13,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS‑fine] λ={lam:.1e} new best = {best_sc:.9f}")

    return best_a, best_sc


def mix_candidates(candidates, fft_len, deadline):
    """
    Random convex combinations of the given candidates.
    Returns the best vector and its fast_score.
    """
    best_a = candidates[0]
    best_sc = fast_score(best_a, fft_len)
    rng = np.random.default_rng()
    for i in range(len(candidates)):
        for j in range(i + 1, len(candidates)):
            for _ in range(8):
                if time.time() >= deadline:
                    return best_a, best_sc
                w = rng.random()
                mix = w * candidates[i] + (1.0 - w) * candidates[j]
                mix /= mix.sum()
                sc = fast_score(mix, fft_len)
                if sc < best_sc:
                    best_sc = sc
                    best_a = mix.copy()
                    print(f"[mix] new best = {best_sc:.9f}")
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # --------------------------------------------------------------
    # Problem size – a reasonably large n gives a better asymptotic constant.
    # --------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1) Scan a simple exponential family a_i ∝ r**i
    # --------------------------------------------------------------
    best_a = None
    best_sc = np.inf
    r_vals = np.linspace(0.85, 0.9995, 200)   # dense sweep
    for r in r_vals:
        a = np.power(r, np.arange(n, dtype=np.float64))
        a /= a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    print(f"[init‑exp] best score after exponential sweep = {best_sc:.9f}")

    # --------------------------------------------------------------
    # 2) Add a few random starts to diversify the pool
    # --------------------------------------------------------------
    for _ in range(12):
        a = rng.random(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[rand] new best = {best_sc:.9f}")

    # --------------------------------------------------------------
    # 3) Mirror descent (exponentiated‑gradient) on the exact max‑convolution
    # --------------------------------------------------------------
    eg_deadline = deadline - 2.0   # keep a couple of seconds for polishing
    best_a, best_sc = exp_gradient_descent(best_a,
                                            eg_deadline,
                                            n,
                                            fft_len,
                                            rng,
                                            eta0=30.0)
    print(f"[EG] finished, best score = {best_sc:.9f}")

    # --------------------------------------------------------------
    # 4) Short L‑BFGS polishing on a very smooth surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = lbfgs_fine_tune(best_a,
                                          deadline,
                                          n,
                                          fft_len)
        print(f"[L‑BFGS‑fine] finished, best score = {best_sc:.9f}")

    # --------------------------------------------------------------
    # 5) Convex mixing with the previous global solution (if any)
    # --------------------------------------------------------------
    candidates = [best_a]
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.shape == best_a.shape:
            candidates.append(prev / prev.sum())
    if len(candidates) > 1:
        best_a, best_sc = mix_candidates(candidates, fft_len, deadline)
        print(f"[mix] final mix best score = {best_sc:.9f}")

    # --------------------------------------------------------------
    # 6) Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the runner
    print(f"[final] evaluated official score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
