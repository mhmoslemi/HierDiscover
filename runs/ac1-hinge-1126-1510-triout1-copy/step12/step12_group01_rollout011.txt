def lp_refine(a, M, epsilon, t, rng, fft_len_conv):
    # compute indices of active constraints
    conv = np.fft.irfft(np.fft.rfft(a, fft_len_conv)*np.fft.rfft(a, fft_len_conv), fft_len_conv)[:2*n-1]
    active_idxs = np.where(conv >= M - epsilon)[0]
    m = active_idxs.size
    if m == 0:
        return a, False
    # build A_ub
    A = np.zeros((m, len(a)), dtype=np.float64)
    for row_idx, k in enumerate(active_idxs):
        # for each i such that 0 <= i < n and 0 <= k-i < n
        # j = k-i => i = k-j
        # for i from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k - (len(a) - 1))
        i_end = min(k, len(a) - 1)
        # set A[row_idx, i] = a[k-i] for i in range(i_start,i_end+1)
        i_range = np.arange(i_start, i_end+1)
        A[row_idx, i_range] = a[k - i_range]
    # objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(len(a))
    bounds = [(0, None)] * len(a)
    # Solve
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
        sumb = b.sum()
        s = a.sum()
        if sumb <= 0:
            return a, False
        scale = s / sumb
        g = b * scale
        a_new = (1 - t) * a + t * g
        # Ensure nonneg via clip
        a_new = np.clip(a_new, 0, None)
        a_new_sum = a_new.sum()
        if a_new_sum > 0:
            a_new = a_new / a_new_sum * s  # renorm to original sum
        else:
            a_new = a.copy()
        return a_new, True
    else:
        # LP failed/unbounded
        return a, False

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Initial pool** – deterministic (spectral, Hann, Blackman, Gaussian,
   Kaiser) and several random Dirichlet‑like vectors.
2. **Phase 1 – Aggressive L‑BFGS on a smoothed max**  
   Many restarts: blend the current best with a fresh random vector and
   run L‑BFGS on the log‑sum‑exp smoothed version of the max‑convolution.
   A fine λ‑schedule (λ ≈ 0.4 … 5·10⁻⁵) together with many iterations per
   λ drives the solution close to a local optimum.
3. **Phase 2 – Linear‑program “mass‑transfer’’ step**  
   For the current best vector `a` compute the active convolution indices
   (the largest ~0.5 % of the autocorrelation).  Solve the LP  

        maximize   Σ b_j
        subject to  a ∗ b ≤ M   (only on the active indices)
                    b ≥ 0

   (where `M` is the current maximal convolution value).  
   Scale `b` back to the unit‑sum simplex and mix it with `a`
   (`a ← (1‑t)·a + t·b`).  Repeating this step strictly reduces the
   objective.
4. **Phase 3 – Deterministic hill‑climb polish** – tiny pairwise mass
   transfers that improve the exact objective (fast FFT‑based score).
   The loop runs until the time budget is exhausted.
5. **Return** the best list of floats; the official `evaluate_sequence`
   is called once at the end.

All phases respect the global deadline (default 1000 s) and use only
NumPy / SciPy.  The code works on a single thread and stays well within
the allowed resource limits.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact objective for a probability vector (Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y   (y are the logits, a = softmax(y))
    """
    a = softmax(y)

    # a * a via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam)        # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution values (softmax of the exp‑weights)
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient w.r.t. a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random samples on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.031)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS continuation (multiple λ values)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in `lam_schedule`, starting each stage from
    the previous result. Returns best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:   # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑program refinement (mass‑transfer step)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray,
              M: float,
              idxs: np.ndarray,
              t: float,
              n: int,
              fft_len: int):
    """
    Solve   maximize Σ b
    s.t.    conv(a, b)[k] ≤ M   for k in idxs
            b ≥ 0
    Scale b back to the simplex, blend with a using weight t and
    return the new vector.  Returns (a_new, improved_flag).
    """
    m = idxs.size
    if m == 0:
        return a, False

    # Build constraint matrix A (m × n)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        A[row, i_range] = a[k - i_range]

    # Linear program: minimize -Σb  (i.e. maximize Σb)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    rhs = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs')
    if not res.success:
        return a, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a, False

    # Scale b to the unit‑sum simplex (sum_a == 1 throughout)
    g = b / sum_b
    a_new = (1.0 - t) * a + t * g
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new, True


# ----------------------------------------------------------------------
# Hill‑climb (deterministic greedy polishing)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 200000,
               delta_frac: float = 0.06):
    """
    Greedy pairwise mass transfers that strictly improve the exact score.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

        idxs = np.argpartition(conv, -top_k)[-top_k:]

        # LP refinement step
        a_new, improved = lp_refine(best_a, M, idxs, t_mix, n, fft_len)
        if not improved:
            # if LP fails, reduce mixing weight and try again
            t_mix *= 0.5
            if t_mix < 1e-3:
                break
            continue

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"[LP] improved fast_score={best_sc:.8f}")
        else:
            # no improvement – shrink mixing weight to be more conservative
            t_mix *= 0.7

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + remaining
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(
            best_a,
            best_sc,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=300000,
            delta_frac=0.04
        )
        print(f"[Hill‑climb] final fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score (clamps etc.)
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
