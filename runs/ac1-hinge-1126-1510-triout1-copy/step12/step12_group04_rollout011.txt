def lp_optimize_cross(a, n, fft_len, tolerance=0.02, max_constraints=200):
    # compute self-convolution
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    M = conv_full.max()
    # Choose indices where conv >= M * (1 - tolerance)
    idx_candidates = np.where(conv_full >= M * (1 - tolerance))[0]
    if idx_candidates.size > max_constraints:
        # Keep the largest values (sort descending)
        vals = conv_full[idx_candidates]
        order = np.argsort(-vals)
        idx_candidates = idx_candidates[order[:max_constraints]]
    m = idx_candidates.size
    # Build A_ub as sparse
    from scipy.sparse import lil_matrix, csr_matrix
    A_ub = lil_matrix((m, n + 1), dtype=np.float64)
    for row, k in enumerate(idx_candidates):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        j = np.arange(i_min, i_max + 1, dtype=np.int32)
        coeffs = a[k - j]  # a_i where i = k - j
        A_ub[row, j] = coeffs
        A_ub[row, n] = -1.0
    A_ub = A_ub.tocsr()
    # Equality constraint sum(b) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    # Objective: minimize t
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    b_opt = res.x[:n]
    t_opt = res.x[n]
    return b_opt, t_opt

   operations cheap.
2. **Diverse initial pool** – random, window‑function, and interpolated
   previous best vectors, all normalised to sum = 1.
3. **Phase 1 – warm L‑BFGS** on a smoothed log‑sum‑exp surrogate with a
   decreasing λ‑schedule (down to 1e‑5).  The optimisation works in log‑space
   (softmax) and respects the simplex constraints.
4. **Phase 2 – long exponentiated‑gradient (mirror‑SGD)** on the true L∞
   objective, with a square‑root step‑size decay and occasional Gaussian
   jitter.
5. **Phase 3 – LP‑based cross‑convolution refinement**.
   For the current vector `a` we solve a tiny linear programme:
       minimise   t
       s.t.   (a ⊛ b)_k ≤ t   for the k‑indices where a⊛a is close to its peak,
              Σ b = 1,   b ≥ 0 .
   The solution `b` serves as a descent direction; we mix a←(1−τ)a+τb.
   Only a few (≤120) constraints are used, making the LP cheap (highs
   solver, sparse matrix).
6. **Phase 4 – spectral‑magnitude flattening**.
   Repeatedly replace the FFT magnitude of the current vector by a constant,
   keep the phase, inverse‑FFT, then project onto the probability simplex.
   This drives the self‑convolution towards a flatter shape.
7. **Phase 5 – final low‑λ L‑BFGS polish** and a short simulated‑annealing
   hill‑climb.
8. **Evaluation** – the best vector is finally scored with the official
   `evaluate_sequence` function and stored in ``height_sequence_1``.

All phases respect the global 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix, csr_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the official score when sum(a)=1:
    2·n·max(conv(a,a)). Convolution via FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector from several families."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with its reversal)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# L‑BFGS on a smoothed log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution objective and its gradient w.r.t. the log‑variables y.
    a = softmax(y) guarantees a ≥ 0 and Σa = 1.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂ smooth_max / ∂ conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run L‑BFGS over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          max_iters: int = 30_000_000):
    """Mirror‑SGD (multiplicative updates) on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        if (it & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = it
                print(f"[exp‑grad] it={it} fast_score={best_score:.9f}")

        # jitter if stagnating
        if (it - last_improve) > 200_000:
            a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)
            last_improve = it

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming cross‑convolution refinement
# ----------------------------------------------------------------------


def lp_optimize_cross(a: np.ndarray,
                      n: int,
                      fft_len: int,
                      tolerance: float = 0.02,
                      max_constraints: int = 120):
    """
    Solve:
        minimise   t
        s.t.   (a ⊛ b)_k ≤ t   for all k with (a⊛a)_k ≥ (1‑tolerance)·M,
               Σ b = 1,
               b ≥ 0.
    Returns (b_opt, t_opt) or None on failure.
    """
    # full self‑convolution (FFT based)
    fa = np.fft.fft(a, fft_len)
    conv_self = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    M = conv_self.max()
    idx = np.where(conv_self >= M * (1.0 - tolerance))[0]

    if idx.size > max_constraints:
        # keep the largest values only
        vals = conv_self[idx]
        order = np.argsort(-vals)
        idx = idx[order[:max_constraints]]

    m = idx.size
    if m == 0:
        return None

    # Build sparse A_ub matrix: each row corresponds to (a ⊛ b)_k - t ≤ 0
    A_ub = lil_matrix((m, n + 1), dtype=np.float64)
    for row, k in enumerate(idx):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        j = np.arange(i_min, i_max + 1, dtype=np.int32)   # positions of b
        coeffs = a[k - j]                                 # a_{k‑j}
        A_ub[row, j] = coeffs
        A_ub[row, n] = -1.0                               # -t
    A_ub = csr_matrix(A_ub)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: b_i ≥ 0, t ≥ 0
    bounds = [(0, None)] * n + [(0, None)]

    # Objective: minimise t
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    b_opt = res.x[:n]
    t_opt = res.x[n]
    return b_opt, t_opt


# ----------------------------------------------------------------------
# Spectral‑magnitude flattening (alternating projection)
# ----------------------------------------------------------------------


def flatten_fft_phase(a: np.ndarray,
                     fft_len: int,
                     iterations: int,
                     rng: np.random.Generator):
    """
    Iteratively enforce constant FFT magnitude while staying on the probability
    simplex.  This tends to produce a flatter self‑convolution.
    """
    n = a.shape[0]
    for it in range(iterations):
        A = np.fft.fft(a, fft_len)
        # Replace magnitude by its mean (any constant works)
        const_mag = np.mean(np.abs(A))
        A_new = const_mag * np.exp(1j * np.angle(A))
        a = np.real(np.fft.ifft(A_new))
        # projection onto non‑negative simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # occasional jitter to avoid cycling
        if rng.random() < 0.01:
            a = add_noise(a, sigma=rng.uniform(1e-4, 5e-4), rng=rng)
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """Random mass transfers with a linear cooling schedule."""
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        src, dst = (i, j) if a[i] > a[j] else (j, i)
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a moderately large problem size.
    # ------------------------------------------------------------------
    n = 12000                     # larger than the previous 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors.
    # ------------------------------------------------------------------
    pool = []

    # Interpolate previous best if available.
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # Classic window functions (chebyshev, kaiser, hann, blackman)
    try:
        from scipy.signal import chebwin, kaiser, hann, blackman
        for att in (30, 50, 70):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        pool.append(hann(n) / hann(n).sum())
        pool.append(blackman(n) / blackman(n).sum())
    except Exception:
        pass

    # Random families
    for _ in range(15):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Time budget allocation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.18 * budget_s   # quick L‑BFGS warm‑up
    phase2_end = start_time + 0.78 * budget_s   # long EG phase
    # remaining time → LP refinement, spectral flattening, final polish

    # ------------------------------------------------------------------
    # Phase 1 – quick L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 7e-4, 3e-4, 1.5e-4,
        8e-5, 4e-5, 2e-5, 1e-5, 5e-6,
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float("inf")
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        if rng.random() < 0.35:
            a_start = add_noise(
                a_start, sigma=rng.uniform(0.001, 0.015), rng=rng
            )
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # optional symmetry enforcement
    sym_a = symmetrize(best_a)
    sym_score = fast_score(sym_a, fft_len)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        print(f"[Phase1] symmetry improved fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=35.0,
            max_iters=40_000_000,
        )
        print(f"[Phase2] exp‑grad best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP‑based cross‑convolution refinement (several passes)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        lp_res = lp_optimize_cross(best_a, n, fft_len,
                                   tolerance=0.02, max_constraints=100)
        if lp_res is None:
            break
        b_opt, t_opt = lp_res
        # Mix a bit towards the LP solution
        tau = 0.25
        a_new = (1.0 - tau) * best_a + tau * b_opt
        a_new = a_new / a_new.sum()
        sc_new = fast_score(a_new, fft_len)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"[LP‑refine] improved fast_score={best_score:.9f}")
        else:
            # if no improvement, try a smaller step size
            tau = 0.10
            a_new = (1.0 - tau) * best_a + tau * b_opt
            a_new = a_new / a_new.sum()
            sc_new = fast_score(a_new, fft_len)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                print(f"[LP‑refine] (small τ) improved fast_score={best_score:.9f}")
            else:
                # stop early if LP cannot improve further
                break

    # ------------------------------------------------------------------
    # Phase 4 – spectral‑magnitude flattening (alternating projection)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_flat = flatten_fft_phase(best_a, fft_len, iterations=250, rng=rng)
        sc_flat = fast_score(a_flat, fft_len)
        if sc_flat < best_score:
            best_score = sc_flat
            best_a = a_flat.copy()
            print(f"[FFT‑proj] improved fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – final low‑λ L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lambda = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_ref, sc_ref = lbfgs_cascade(
            best_a,
            fine_lambda,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()
            print(f"[Final‑LBFGS] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Optional short annealing if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a, best_score, deadline, n, rng
        )
        print(f"[Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
