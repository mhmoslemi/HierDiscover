"""
Hybrid optimiser for the “max‑convolution” problem.

The programme follows four stages:

1. **Diverse initial pool** – random windows, Gaussians,
   cosine‑powers, edge‑heavy shapes and a *sparse low‑overlap* seed.
   Every candidate is normalised to a probability vector.

2. **Multi‑restart smoothed L‑BFGS** – we minimise a
   log‑sum‑exp surrogate of  max(conv)  with a decreasing λ‑schedule.
   Each restart is a full cascade of L‑BFGS solves; the best
   candidate (lowest fast_score) is kept.

3. **Mirrored exponentiated‑gradient descent** – an exact sub‑gradient
   method on  max(conv)  (the same mask‑based gradient used in the
   original code) with a 1/√t step‑size.

4. **Gradient‑guided hill‑climbing** – after the EG phase we
   repeatedly move a small fraction of mass from positions that
   contribute most to the current maximal convolution entry to
   positions that contribute least.
   Only strictly improving moves are accepted.

All phases respect the global 1000 s deadline; the current best
sequence is stored in the global variable `height_sequence_1` and
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute 2·n·max(conv(a,a))  efficiently (scale‑invariant, n = len(a)).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Initial families

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector from several styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


def sparse_low_overlap_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Pick O(sqrt(n)) random positions and give them equal weight.
    The small support should reduce the number of pairwise sums
    that collide, providing a good starting point for the optimiser.
    """
    s = max(2, int(2.5 * np.sqrt(n)))          # a few‑times √n
    idx = rng.choice(n, size=s, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y are unrestricted variables; a = softmax(y) lies on the simplex.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c           # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)   # safe: arguments ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # Gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # Back‑prop through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]      # size n

    # Chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best candidate found during this run.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution exponentiated‑gradient descent
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 18.0):
    """
    Multiplicative (mirrored) sub‑gradient descent on the exact
    L‑∞‑objective  max(conv).  The step size decays as 1/√t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (mirrored version)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def gradient_guided_refinement(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator):
    """
    Hill‑climbing that explicitly moves mass away from positions that
    contribute most to the current maximal convolution entry.
    Only moves that strictly improve the fast_score are kept.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score

    # parameters for the mass transfer
    max_frac = 0.12          # up to 12 % of source mass in one move
    eps_val = 1e-14

    while time.time() < deadline:
        # current convolution and mask of active max indices
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + eps_val
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the exact max‑convolution (same as in EG)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # contribution of each coordinate to the current max entry:
        # C_i = a_i * g_i / 2  (since g_i = 2 * Σ a_{k‑i})
        contrib = a * g * 0.5

        # choose source proportional to contribution, destination inversely
        prob_src = contrib + 1e-12          # avoid zero
        prob_src /= prob_src.sum()
        src = rng.choice(n, p=prob_src)

        # destination prefers low contribution
        inv = np.max(contrib) - contrib + 1e-12
        prob_dst = inv / inv.sum()
        dst = rng.choice(n, p=prob_dst)

        if src == dst or a[src] <= 0.0:
            continue

        # transfer a small random fraction of source mass
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score - 1e-12:          # strict improvement
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
                print(f"[refine] new best fast_score = {best_score:.9f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) with a low
    evaluation score under the provided `evaluate_sequence`.  The method
    combines a rich initial pool, multi‑restart smoothed L‑BFGS, a long
    exponentiated‑gradient descent on the exact max‑convolution, and a
    gradient‑guided hill‑climbing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # --------------------------------------------------------------
    pool = []

    # 0) Re‑use previous best if present
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) Classic windows (Chebyshev & Kaiser if available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # 2) Random / parametric families
    for _ in range(4):
        pool.append(random_initial(n, rng))
    for _ in range(2):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # 3) Low‑overlap sparse seed (few √n points)
    pool.append(sparse_low_overlap_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    phase1_end = start_time + 0.20 * budget_s        # L‑BFGS restarts
    phase2_end = start_time + 0.70 * budget_s        # long EG descent
    phase3_end = start_time + 0.90 * budget_s        # fine L‑BFGS
    # remaining time → gradient‑guided refinement

    # --------------------------------------------------------------
    # Phase 1 – aggressive multi‑restart L‑BFGS on smoothed objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5,
        1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float('inf')
    restart_counter = 0

    while time.time() < phase1_end:
        restart_counter += 1
        a_start = rng.choice(pool)
        # occasional jitter to diversify neighbouring basins
        if rng.random() < 0.30:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.007),
                                rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[restart {restart_counter}] best fast_score = {best_score:.9f}")

    if best_a is None:           # fallback
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=22.0               # a slightly larger step‑size
        )
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with ultra‑low λ
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        fine_lam_sched = [5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=1800,
                                        deadline=phase3_end,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – gradient‑guided hill‑climbing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = gradient_guided_refinement(best_a,
                                                        best_score,
                                                        deadline,
                                                        n,
                                                        fft_len,
                                                        rng)
        print(f"[final‑refine] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
