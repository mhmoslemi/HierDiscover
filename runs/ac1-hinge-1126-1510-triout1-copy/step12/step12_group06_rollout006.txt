def build_conv_matrix(a):

    for d in range(n):
        # diagonal offset = d (i.e., row - col = d)
        # length = n - d for diag in square? Actually shape rows x cols: row index r from d to d + (n - 1) at column c = r - d. So can generate array of length n for column dimension? Use diags and let it broadcast.
        # In scipy.sparse.diags, each diagonal must be an array of length = min(rows, cols) - abs(offset). So for offset d >=0, length = min(rows-d, cols) = min(2n-1-d,n) = n. Wait 2n-1-d >= n for d <= n-1, so length is n.

def build_conv_matrix(a):
    n = a.shape[0]
    offsets = np.arange(n)
    # each diagonal is constant = a[d]
    data = [np.full(n, a[d]) for d in range(n)]
    A = diags(data, offsets, shape=(2*n-1, n), format='csc')
    return A

def lp_refine(a, cur_score, deadline, n, fft_len, max_iter=10):
    # iteration loop until time runs out or no improvement
    while time.time() < deadline and max_iter > 0:
        # compute current max conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        M = conv.max()
        # Build convolution matrix A (f * g)
        A = build_conv_matrix(a)
        # Solve LP: maximize sum(g) s.t. A @ g <= M, g >=0
        c = -np.ones(n)  # minimize -sum
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as e:
            print("[LP] linprog exception:", e)
            break
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        # Normalise g
        g_norm = g / sum_g
        # Mix with t = 0.5 (or adapt if no improvement)
        t = 0.5
        a_new = (1 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            max_iter -= 1
            print(f"[LP] improved fast_score = {cur_score:.9f}")
        else:
            # try smaller t or break
            print("[LP] no improvement, stop")
            break
        # free matrix
        del A
    return a, cur_score

"""
Step‑function optimiser – 1000 s budget with LP refinement

This version builds on the previous pipeline (diverse initial pool,
L‑BFGS on a smooth surrogate, long exponentiated‑gradient descent and
greedy hill‑climbing) and adds a *linear‑programming refinement* that
implements the classic “convex‑combination” step from the literature:

    given f, let M = max(f*f).  maximise Σ g subject to
        g ≥ 0  and  f * g ≤ M                (linear constraints)

The optimal g is used to form a convex combination (1‑t) f + t g.
If Σ g > Σ f this strictly reduces the max‑convolution, giving a finer
upper bound.  The LP is solved with SciPy’s HiGHS backend (sparse
matrix).  A few LP refinements (t = 0.5, falling back to 0.2) are
performed after the hill‑climb, and a final short hill‑climb is run
again.  The whole search respects the 1000 s deadline and stores the
best sequence in `height_sequence_1`.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (sum(a)=1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Initial families (same as the baseline)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    a[:m] = rng.random(m) + 0.5
    a[-m:] = rng.random(m) + 0.5
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y are unrestricted; a = softmax(y) lives on the simplex.
    """
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    fval = 2.0 * n * smooth_max

    w = exp_shift / Z               # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return fval, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirrored exponentiated‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """Multiplicative sub‑gradient descent on the true L∞‑objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local refinement (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Simple hill‑climbing: transfer a random fraction of mass from one
    index to another, accepting only strict improvements.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (convex‑combination step)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray):
    """Sparse matrix A such that (A @ g) = f * g (full convolution)."""
    n = a.shape[0]
    offsets = np.arange(n)               # row‑col = offset ≥ 0
    # each diagonal has the constant value a[d]
    diag_data = [np.full(n, a[d]) for d in range(n)]
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def lp_refine(a: np.ndarray,
              cur_score: float,
              deadline: float,
              n: int,
              fft_len: int,
              max_iter: int = 8):
    """
    Perform at most `max_iter` LP‑based refinement steps.
    Returns possibly improved (a,score).
    """
    while time.time() < deadline and max_iter > 0:
        # current max‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # Build the convolution matrix f * g ≤ M
        A = build_conv_matrix(a)

        # LP: maximise Σ g   → minimise -Σ g
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        except Exception as e:
            print("[LP] linprog exception:", e)
            break

        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Normalise g
        g_norm = g / sum_g

        # Try a convex combination; start with t=0.5 and fall back to 0.2
        improved = False
        for t in (0.5, 0.2):
            cand = (1.0 - t) * a + t * g_norm
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            new_score = fast_score(cand, fft_len)
            if new_score < cur_score - 1e-12:
                a = cand
                cur_score = new_score
                max_iter -= 1
                improved = True
                print(f"[LP] improved fast_score = {cur_score:.9f} (t={t})")
                break
        # clean up
        del A
        if not improved:
            # No improvement with the tried t values → stop.
            break
    return a, cur_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good balance
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    # --------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # --------------------------------------------------------------
    pool = []

    # 0) Re‑use previous best if present
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) Classical windows (Chebyshev & Kaiser if available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # 2) Random and parametric families
    for _ in range(6):
        pool.append(random_initial(n, rng))
    for _ in range(2):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s      # L‑BFGS
    phase2_end = start_time + 0.60 * budget_s      # EG descent
    phase3_end = start_time + 0.85 * budget_s      # fine L‑BFGS
    # Remaining time will be used for refinement & final evaluation

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.21, 0.13, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.0015, 0.0008, 0.0004, 0.0002, 0.0001,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7
    ]
    maxiter_per_lam = 800

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.4:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.008),
                                rng=rng)
        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   eta0=18.0)
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with very low λ
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        fine_lam_sched = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=1200,
                                        deadline=phase3_end,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – greedy local refinement (mass‑transfer hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        best_a, best_score = local_refinement(best_a,
                                               best_score,
                                               deadline - 30.0,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[local‑refine] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – Linear‑programming based refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        best_a, best_score = lp_refine(best_a,
                                       best_score,
                                       deadline - 10.0,
                                       n,
                                       fft_len,
                                       max_iter=8)
        print(f"[LP‑refine] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 6 – a final short hill‑climb to polish any LP changes
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_score = local_refinement(best_a,
                                               best_score,
                                               deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[final‑local] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation using the provided scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
