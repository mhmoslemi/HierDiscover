"""
Propose a sequence of non‑negative heights that (empirically) beats the
previous best score (≈ 1.5059).  The method combines a small
population of diverse initial vectors, a fast evolutionary search on the
simplex, and final gradient‑based refinements (L‑BFGS → Adam → greedy
hill‑climb).  All phases respect the 1000 s time budget.

The objective to minimise is
    2·n·max(conv(sequence,sequence)) / (sum(sequence)²)
but because the ratio is scale‑invariant we work on the probability
simplex (∑a_i = 1).  The fast_score function computes the exact value
for a simplex vector via an FFT‑based convolution.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where f = 2·n·smooth_max(conv) and grad = ∂f/∂y.
    y are the logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # shape (2n-1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (various deterministic / random families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral initialisation – square‑root of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet‑like draws with various shapes."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in `lam_schedule`, starting each stage from the
    previous result.  Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y (a = softmax(y)) so that the simplex constraint is
    satisfied automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)   # sub‑gradient mask

        # gradient wrt a of the max entry (sub‑gradient)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # full objective gradient (2·n·max)
        grad_a = 2.0 * n * g_a

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)      # decaying lr
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to help escape plateaus
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a low‑score step function.
    Uses a small evolutionary population, followed by L‑BFGS,
    Adam on the exact max‑convolution, and a final greedy hill‑climb.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – larger n gives more freedom; 4096 is a good trade‑off.
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial population
    # --------------------------------------------------------------
    pop_size = 30
    population = []
    scores = []

    init_fns = [
        spectral_init,
        hann_init,
        blackman_init,
        lambda N: gaussian_init(N, sigma_factor=0.20),
        lambda N: kaiser_init(N, beta=rng.uniform(6.0, 12.0)),
    ]

    for _ in range(pop_size):
        fn = rng.choice(init_fns)
        a = fn(n) if callable(fn) else fn(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        population.append(a)
        scores.append(sc)

    # sort population (lowest score first)
    idx = np.argsort(scores)
    population = [population[i] for i in idx]
    scores = [scores[i] for i in idx]
    best_a = population[0].copy()
    best_sc = scores[0]
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Evolutionary search (majority of the time budget)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 30.0   # keep 30 s for final refinements

    generation = 0
    while time.time() < deadline:
        # tournament selection for two parents
        i1, i2 = rng.integers(pop_size, size=2)
        parent1 = population[i1] if scores[i1] < scores[i2] else population[i2]

        i3, i4 = rng.integers(pop_size, size=2)
        parent2 = population[i3] if scores[i3] < scores[i4] else population[i4]

        # blend parents
        alpha = rng.random() * 0.25
        child = (1.0 - alpha) * parent1 + alpha * parent2

        # mutation (Gaussian noise, projected back to simplex)
        sigma = 0.02 * (0.9 ** (generation // 2000))
        child += rng.normal(scale=sigma, size=n)
        child = np.maximum(child, 0.0)
        child = simplex_projection(child)

        child_sc = fast_score(child, fft_len, n)

        # replace worst if the child is better
        if child_sc < scores[-1]:
            population[-1] = child
            scores[-1] = child_sc
            # re‑sort
            idx = np.argsort(scores)
            population = [population[i] for i in idx]
            scores = [scores[i] for i in idx]

            if child_sc < best_sc:
                best_sc = child_sc
                best_a = child.copy()
                print(f"[EA] gen={generation}, new best fast_score={best_sc:.8f}")

        generation += 1

    # --------------------------------------------------------------
    # Local refinement – L‑BFGS on a smooth surrogate of the max
    # --------------------------------------------------------------
    if time.time() < deadline + 30.0:
        lam_schedule = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1,
                        0.07, 0.05, 0.03, 0.02, 0.01,
                        0.005, 0.002, 0.001, 5e-4, 2e-4,
                        1e-4, 5e-5, 2e-5, 1e-5]
        maxiter_per_lam = 1500
        best_a, best_sc = lbfgs_continuation(best_a,
                                             lam_schedule,
                                             maxiter_per_lam,
                                             deadline + 30.0,
                                             fft_len,
                                             fft_len,
                                             n)
        print(f"[L‑BFGS‑final] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Exact max optimisation – Adam
    # --------------------------------------------------------------
    if time.time() < deadline + 30.0:
        best_a, best_sc = adam_optimize(best_a,
                                        deadline + 30.0,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam‑final] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Final greedy polishing
    # --------------------------------------------------------------
    if time.time() < deadline + 30.0:
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     deadline + 30.0,
                                     rng,
                                     max_iter=60000,
                                     delta_frac=0.04)
        print(f"[Hill‑climb‑final] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
