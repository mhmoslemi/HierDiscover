def build_sidon_set(n, target_m=None, max_trials=10):
   best_set = []
   for t in range(max_trials):
        order = rng.permutation(n)  # random ordering
        S = []
        sums = set()
        for x in order:
            ok = True
            for y in S:
                if (x+y) in sums:
                    ok = False
                    break
            if not ok:
                continue
            # also check x+x sum
            if (x+x) in sums:
                continue
            # add
            for y in S:
                sums.add(x+y)
            sums.add(x+x)
            S.append(x)
            if target_m and len(S) >= target_m:
                break
        if len(S) > len(best_set):
            best_set = S
   return best_set

def adam_max_optimize(y_start, iters, deadline, n, fft_len, rng, lr=0.1):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_y = y.copy()
    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)
    t = 0
    for _ in range(iters):
        if time.time() >= deadline:
            break
        t += 1
        # compute subgradient of max conv
        a = np.exp(y - np.max(y))
        a /= a.sum()
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps_max = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_max).astype(np.float64)
        # compute gradient of max conv w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1: n-1+n]  # shape n

        # chain rule through softmax
        # compute grad_y
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate
        a_new = np.exp(y - np.max(y))
        a_new /= a_new.sum()
        sc = fast_score(a_new, fft_len)
        if sc < best_score:
            best_score = sc
            best_y = y.copy()
            best_a = a_new.copy()
    return best_a, best_score

        a /= a.sum()

"""
Propose a low‑bound step‑function using a multi‑stage optimisation.

The algorithm consists of:

1.  A large pool of diverse random / parametric initial vectors, all
    normalised to the probability simplex.
2.  A fast pre‑filter that keeps the best few candidates (by the FFT‑based
    fast_score).
3.  A cascade of L‑BFGS optimisations on a log‑sum‑exp smooth surrogate of
    the max‑convolution, with a decreasing λ‑schedule.
4.  Adam optimisation on the same surrogate for fine‑grained improvement.
5.  Adam optimisation on the *exact* max‑convolution (sub‑gradient) for a
    more faithful descent.
6.  Exponentiated‑gradient (mirror) descent using the true sub‑gradient.
7.  A deterministic hill‑climbing phase (small mass transfers) and a
    short Metropolis annealing polish.
8.  Optional convex mixing of the final candidate with the previous best.

All stages respect a total wall‑clock budget (default 1000 s).  The best
sequence found is stored globally as ``height_sequence_1`` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming a already sums to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (used by L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y lives in ℝⁿ; a = softmax(y) belongs to the probability simplex.
    """
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)           # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                                 # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]               # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS cascade
# ----------------------------------------------------------------------


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    # work in log‑space (softmax will enforce the simplex)
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_sched,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  lr: float = 0.05):
    """
    Adam optimiser (adaptive moment) on the log‑sum‑exp surrogate.
    Returns the best probability vector encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            f, grad_y = objective_and_grad(y, n, fft_len, lam)
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam on the *exact* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def adam_max_optimize(y_start: np.ndarray,
                      max_iters: int,
                      deadline: float,
                      n: int,
                      fft_len: int,
                      rng: np.random.Generator,
                      lr: float = 0.05):
    """
    Adam optimisation on the true max‑convolution (non‑smooth).
    Operates in the log‑space (softmax) so the simplex is respected.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        t += 1

        a = np.exp(y - np.max(y))
        a /= a.sum()

        # exact max‑convolution and its sub‑gradient
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_max = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_max).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]       # size n

        # chain rule through the soft‑max
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate
        a_new = np.exp(y - np.max(y))
        a_new /= a_new.sum()
        sc = fast_score(a_new, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient exponentiated‑gradient descent
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored SGD (exponentiated‑gradient) using the exact L∞‑subgradient.
    Runs until the deadline is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types:
    * mass transfer between two arbitrary positions
    * swapping two entries
    * local neighbour transfer (distance ≤ 5)
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25            # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # 'transfer'
                j = rng.integers(0, n)
            if i == j:
                continue

            # move from larger entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue

            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate‑large size gives a good trade‑off
    # --------------------------------------------------------------
    n = 6000                       # slightly larger than the previous run
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Helper to keep the K best candidates seen so far
    # --------------------------------------------------------------
    K = 10
    best_candidates = []  # list of (score, vector)

    def consider(vec):
        """Normalise vec, compute fast_score and keep top K."""
        a = np.asarray(vec, dtype=float)
        if a.ndim != 1:
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        sc = fast_score(a, fft_len)
        if len(best_candidates) < K:
            best_candidates.append((sc, a))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if sc < best_candidates[-1][0]:
                best_candidates[-1] = (sc, a)
                best_candidates.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # Seed the pool with different families
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # a tiny safety margin

    # previous solution (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
            consider(a0)

    # pure uniform baseline
    consider(np.full(n, 1.0 / n))

    # many random draws from a handful of simple families
    for _ in range(250):
        mode = rng.choice(['uniform', 'random', 'exp', 'power',
                           'edge', 'bimodal', 'sparse'])
        if mode == 'uniform':
            a = np.full(n, 1.0 / n)
        elif mode == 'random':
            a = rng.random(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == 'edge':
            edge_frac = rng.uniform(0.006, 0.03)
            m_edge = max(1, int(n * edge_frac))
            a = np.full(n, 1e-8)
            left = rng.random(m_edge) + 0.5
            right = rng.random(m_edge) + 0.5
            a[:m_edge] = left
            a[-m_edge:] = right
        elif mode == 'bimodal':
            centre = n // 2
            width = rng.integers(int(0.02 * n), int(0.07 * n))
            peak = rng.uniform(0.5, 1.5) / n
            a = np.full(n, 1.0 / n)
            left = max(0, centre - width)
            right = min(n, centre + width)
            a[left:right] = peak
        else:  # sparse
            m = rng.integers(int(0.02 * n), int(0.07 * n))
            idx = rng.choice(n, size=m, replace=False)
            vals = rng.random(m) + 1e-6
            a = np.zeros(n, dtype=float)
            a[idx] = vals
        consider(a)

    # parametric families
    def gaussian():
        sigma = rng.uniform(0.1 * n, 0.5 * n)
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a

    def cosine_power():
        p = rng.uniform(0.5, 3.0)
        centre = (n - 1) / 2.0
        cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
        a = np.maximum(cos_vals, 0) ** p
        a /= a.sum()
        return a

    def sinc_sq():
        x = np.arange(n) - (n - 1) / 2.0
        scale = rng.uniform(0.05, 0.15) * n
        a = np.sinc(x / scale) ** 2
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a

    def triangular():
        peak_frac = rng.uniform(0.2, 0.6)
        peak_len = max(1, int(peak_frac * n))
        a = np.concatenate([
            np.linspace(1, peak_len, peak_len, endpoint=False),
            np.full(n - 2 * peak_len, peak_len),
            np.linspace(peak_len, 1, peak_len, endpoint=True)
        ])
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a

    for _ in range(30):
        consider(gaussian())
        consider(cosine_power())
        consider(sinc_sq())
        consider(triangular())

    # a few sparse “low‑collision” constructions (greedy Sidon‑like)
    def generate_sparse_sidon(target_m):
        """Greedy construction trying to avoid sum collisions."""
        indices = []
        sums = set()
        attempts = 0
        while len(indices) < target_m and attempts < 5 * n:
            cand = rng.integers(0, n)
            ok = True
            for i in indices:
                if (cand + i) in sums:
                    ok = False
                    break
            if not ok:
                attempts += 1
                continue
            if (cand + cand) in sums:
                attempts += 1
                continue
            # accept
            for i in indices:
                sums.add(cand + i)
            sums.add(cand + cand)
            indices.append(cand)
        if not indices:
            return None
        w = np.zeros(n)
        w[indices] = 1.0 / len(indices)
        return w

    for m in (60, 80, 100, 120):
        a = generate_sparse_sidon(m)
        if a is not None:
            consider(a)

    # --------------------------------------------------------------
    # Keep the best few candidates
    # --------------------------------------------------------------
    best_candidates.sort(key=lambda x: x[0])
    best_a = best_candidates[0][1].copy()
    best_score = best_candidates[0][0]
    best_logit = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on the smoothed surrogate
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.02, 0.015, 0.01, 0.007,
        0.005, 0.003, 0.002, 0.001, 5e-4,
        2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7,
        5e-8, 2e-8, 1e-8
    ]
    maxiter_per_lam = 1200

    a_opt, sc_opt = lbfgs_cascade(best_a,
                                  lam_schedule,
                                  maxiter_per_lam,
                                  deadline,
                                  fft_len,
                                  n)
    if sc_opt < best_score:
        best_score, best_a = sc_opt, a_opt.copy()
        best_logit = np.log(best_a + 1e-12)
        print(f"[L‑BFGS] new best = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the smoothed surrogate (more iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        lam_schedule_adam = [
            5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
            5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
        ]
        iters_per_lam = 500
        a_adam, sc_adam = adam_optimize(best_logit,
                                         lam_schedule_adam,
                                         iters_per_lam,
                                         deadline,
                                         fft_len,
                                         n,
                                         lr=0.07)
        if sc_adam < best_score:
            best_score, best_a = sc_adam, a_adam.copy()
            best_logit = np.log(best_a + 1e-12)
            print(f"[Adam‑smooth] new best = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_adammax, sc_adammax = adam_max_optimize(best_logit,
                                                  max_iters=20000,
                                                  deadline=deadline,
                                                  n=n,
                                                  fft_len=fft_len,
                                                  rng=rng,
                                                  lr=0.05)
        if sc_adammax < best_score:
            best_score, best_a = sc_adammax, a_adammax.copy()
            best_logit = np.log(best_a + 1e-12)
            print(f"[Adam‑max] new best = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient descent (exact sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_eg, sc_eg = exp_gradient_optimize(best_a,
                                            deadline,
                                            n,
                                            fft_len,
                                            rng,
                                            eta0=35.0)
        if sc_eg < best_score:
            best_score, best_a = sc_eg, a_eg.copy()
            print(f"[exp‑grad] new best = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – deterministic hill‑climbing (small mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        it = 0
        a = best_a.copy()
        local_best = best_score
        local_best_a = a.copy()
        # limit to a reasonable number of attempts
        while time.time() < deadline and it < 50000:
            it += 1
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move a small fraction from the larger to the smaller entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * 0.03  # up to 3 % of src
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            sc = fast_score(cand, fft_len)
            if sc < local_best:
                local_best = sc
                local_best_a = cand.copy()
                a = cand
                if sc < best_score:
                    best_score, best_a = sc, cand.copy()
                    print(f"[hill‑climb] new best = {best_score:.9f}")
        best_score = local_best
        best_a = local_best_a.copy()

    # --------------------------------------------------------------
    # Phase 6 – short simulated‑annealing polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(best_a,
                                                          best_score,
                                                          deadline,
                                                          n,
                                                          rng)
        print(f"[anneal] best after anneal = {best_score:.9f}")

    # --------------------------------------------------------------
    # Optional mixing of the top candidates (extra polish)
    # --------------------------------------------------------------
    candidates = [best_a]
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.shape == best_a.shape:
            candidates.append(prev / prev.sum())
    for i in range(len(candidates)):
        for j in range(i + 1, len(candidates)):
            for _ in range(5):
                w = rng.random()
                mix = w * candidates[i] + (1 - w) * candidates[j]
                mix /= mix.sum()
                sc = fast_score(mix, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = mix.copy()
                    print(f"[mix] new best = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the platform
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
