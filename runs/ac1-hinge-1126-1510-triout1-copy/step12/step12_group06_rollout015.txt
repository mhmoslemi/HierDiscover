"""
Step‑function optimiser – 1000 s budget (new version)

Key ideas
---------

* Use a larger lattice (n = 8000) – a finer discretisation gives more
  freedom to lower the *continuous* self‑convolution peak, which directly
  improves the discrete score 2·n·max(conv).

* Combine three optimisation stages:
  1. **Smoothed‑max L‑BFGS** – a log‑sum‑exp surrogate with a dense λ‑schedule.
  2. **Projected Adam** on the exact ‑∞‑objective.  The variables are
     the unconstrained logits `y` of a soft‑max, which guarantees
     non‑negativity and unit sum.  The Adam step adapts the learning
     rate per coordinate and converges faster than a plain exponentiated‑
     gradient.
  3. **Greedy mass‑transfer refinement** – hill‑climbing moves that
     strictly improve the true `evaluate_sequence` score.

* Heavy use of FFT (zero‑padded to the next power‑of‑two) for fast
  convolution and gradient computation.  All heavy work therefore runs
  in O(n log n) time.

The function `propose_candidate` follows the global time budget,
stores the best sequence in the global variable `height_sequence_1`,
and returns it as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) – the objective when sum(a)=1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """Smooth surrogate of max‑convolution and its gradient w.r.t. logits y."""
    # soft‑max to stay on simplex
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)  # safe, arguments ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} new best fast_score = {best_score:.9f}")
    return best_a, best_score

def max_conv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max value, sub‑gradient w.r.t. a) of the max‑convolution."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n
    return max_val, g


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lr: float = 0.12):
    """
    Projected Adam on the exact max‑convolution.
    The optimisation variable is the log‑softmax vector y (i.e. a = softmax(y)).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        # ---- compute exact sub‑gradient of max(conv) ----
        a = np.exp(y - np.max(y))
        a /= a.sum()
        max_val, g_a = max_conv_and_subgrad(a, fft_len)

        # chain rule (soft‑max)
        dot_ag = np.dot(a, g_a)
        grad_y = a * (g_a - dot_ag)

        # Adam moment updates
        it += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        # project back onto simplex via soft‑max (implicitly done above)
        a = np.exp(y - np.max(y))
        a_sum = a.sum()
        if a_sum == 0.0:
            a = best_a.copy()
        else:
            a /= a_sum

        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter {it:06d} fast_score = {best_score:.9f}")

        # occasional jitter to shake out of shallow plateaus
        if rng.random() < 0.001:
            y += rng.normal(0.0, 5e-4, size=y.shape)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local refinement (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Simple hill‑climbing: repeatedly transfer a random fraction of mass
    from one index to another.  Only moves that strictly improve the
    `2·n·max(conv)` score are kept.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20                # up to 20 % of source mass may be moved

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
                print(f"[refine] new best fast_score = {best_score:.9f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence().  Returns the best list of floats found before the
    deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 8000 gives more freedom than the previous 4000
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial probability vectors
    # ------------------------------------------------------------------
    pool = []

    # 0) re‑use previous best if present (interpolate to new size)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 1:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) classic windows (Chebyshev & Kaiser) – optional, may be missing
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # 2) random / parametric families
    for _ in range(8):
        pool.append(random_initial(n, rng))
    for _ in range(5):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget splits
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.18 * budget_s       # aggressive L‑BFGS
    phase2_end = start_time + 0.80 * budget_s       # Adam on exact max

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed surrogate
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5,
        1e-5, 5e-6, 2e-6, 1e-6, 5e-7
    ]
    maxiter_per_lam = 800

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        # jitter a bit from time to time
        if rng.random() < 0.35:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.008),
                                rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[phase‑1] new global best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                            phase2_end,
                                            n,
                                            fft_len,
                                            rng,
                                            lr=0.12)
        print(f"[phase‑2 Adam] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with very low λ (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        fine_lam_sched = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=1200,
                                        deadline=deadline - 30.0,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – greedy local refinement (mass‑transfer hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = local_refinement(best_a,
                                               best_score,
                                               deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[local‑refine] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # pre‑imported by the runner
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
