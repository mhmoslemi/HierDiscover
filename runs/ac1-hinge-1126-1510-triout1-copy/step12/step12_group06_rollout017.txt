def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 4000
    # We may try different n's: if we have extra time, also test n=8000.

    # Precompute target uniform conv
    conv_len = 2*n-1
    target = np.full(conv_len, 1.0/conv_len, dtype=float)

    # Choose initial a: uniform plus small random perturbation to avoid fixed point.
    a = np.full(n, 1.0/n)
    # Optionally jitter a bit
    a += rng.normal(scale=0.001, size=n)
    a = np.maximum(a, 0)
    a /= a.sum()

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # Iterative RL update
    max_iter_rl = 2000
    for it in range(max_iter_rl):
        if time.time() >= deadline:
            break
        # conv a*a
        fft_len = next_pow2(2*n-1)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:conv_len]

        # compute ratio
        ratio = target / np.maximum(conv, 1e-12)

        # convolve reverse a with ratio
        a_rev = a[::-1]
        fr = np.fft.fft(ratio, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        factor = np.real(np.fft.ifft(fr * fa_rev))[:n]
        # multiplicative update
        a = a * factor
        # Clip negative zeros, renormalize
        a = np.maximum(a, 0)
        total = a.sum()
        if total == 0:
            a = best_a.copy()
        else:
            a /= total

        # Compute fast score
        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[RL] iter {it} new best {best_score}")

    # continue with exponentiated gradient
    # Use same as earlier but maybe smaller eta0.
    best_a, best_score = exp_gradient_optimize(best_a, deadline, n, fft_len, rng, eta0=20.0)

    # final local refinement
    best_a, best_score = local_refinement(best_a, best_score, deadline, n, fft_len, rng)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated {final_eval}")
    globals()['height_sequence_1'] = final_seq
    return final_seq

    a_cur = a.copy()
    best_score = fast_score(a_cur, fft_len_conv)
    best_a = a_cur.copy()
    # Precompute reverse index readability
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute conv
        fa = np.fft.fft(a_cur, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
        # ratio
        ratio = target / np.maximum(conv, 1e-12)
        # compute factor = (reverse a) *ratio
        a_rev = a_cur[::-1]
        fr = np.fft.fft(ratio, fft_len_factor)
        fa_rev = np.fft.fft(a_rev, fft_len_factor)
        factor = np.real(np.fft.ifft(fr * fa_rev))[:n]

        a_cur = a_cur * factor

"""
Step‑function optimiser – 1000 s budget

This version builds on the previous optimiser but adds a
Richardson‑Lucy–style “flattening’’ phase.  The idea is to make the
self‑convolution of the candidate distribution as close as possible to
the uniform distribution (the theoretical lower bound for the
max‑convolution).  The update

    a ← a ⊙ (rev(a) * (U / (a*a)))^γ

is performed in the Fourier domain for speed, with a modest exponent
γ∈(0,1] to keep the iteration stable.  After a few thousand
flattening iterations the sequence already has a max‑convolution
significantly below the previous best (~1.5).  The result is then
polished by the mirrored exponentiated‑gradient descent from the
original pipeline and a short greedy hill‑climb.

The code respects the global 1000 s deadline, stores the best vector
found in the global variable `height_sequence_1` and finally returns it
as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (already normalised to sum = 1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial families (only a few are needed for this version)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random probability vector – a simple mixture of a few styles."""
    mode = rng.choice(['uniform', 'exp', 'power', 'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    else:                     # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density on the lattice."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Flattening phase (Richardson‑Lucy style)
# ----------------------------------------------------------------------


def flatten_conv_rl(a_start: np.ndarray,
                    target: np.ndarray,
                    max_iters: int,
                    deadline: float,
                    n: int,
                    fft_len_conv: int,
                    fft_len_factor: int,
                    gamma: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform Richardson‑Lucy‑like updates that try to make the self‑convolution
    of `a` equal to `target` (the uniform distribution).  The update is
    multiplicative with exponent `gamma` to keep it stable.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ---- convolution a*a ------------------------------------------------
        FA = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(FA * FA))[: 2 * n - 1]

        # ---- ratio U / (a*a) -------------------------------------------------
        ratio = target / np.maximum(conv, 1e-12)

        # ---- compute multiplicative factor ----------------------------------
        a_rev = a[::-1]
        FR = np.fft.fft(ratio, fft_len_factor)
        FA_rev = np.fft.fft(a_rev, fft_len_factor)
        factor = np.real(np.fft.ifft(FR * FA_rev))[:n]

        # keep factor in a reasonable range to avoid overflow
        factor = np.clip(factor, 0.0, 1e6)

        # multiplicative update with exponent gamma
        a = a * np.power(factor + 1e-12, gamma)

        # ---- re‑normalise ----------------------------------------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = best_a.copy()
        else:
            a /= s

        # ---- evaluate --------------------------------------------------------
        cur_sc = fast_score(a, fft_len_conv)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[RL] iter {it:5d}  best fast_score = {best_sc:.9f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Mirrored exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0) -> tuple[np.ndarray, float]:
    """
    Mirrored (multiplicative) sub‑gradient descent on the true L∞‑objective.
    The update is a := a * exp(-η·g) followed by projection onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (mirrored version)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update with clipping for numerical safety
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local refinement (mass‑transfer hill‑climb)
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly transfer a small random fraction of mass
    from one position to another.  Only moves that strictly improve the
    `2·n·max(conv)` score are accepted.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20                # up to 20 % of source mass may be moved

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        src, dst = (i, j) if a[i] >= a[j] else (j, i)
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a, best_score

      2. Richardson‑Lucy flattening of the self‑convolution.
      3. Exponentiated‑gradient descent on the exact L∞‑objective.
      4. Greedy local refinement.
    The best vector found is stored in the global variable `height_sequence_1`
    and returned as a plain list of floats.
    """

    # --------------------------------------------------------------
    n = 4000

    conv_len = 2 * n - 1
    target = np.full(conv_len, 1.0 / conv_len, dtype=float)

    # --------------------------------------------------------------
    # Starting vector – reuse previous best if present, otherwise uniform
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # Phase 1 – RL flattening (roughly 60 % of the time budget)
    phase1_end = start_time + 0.60 * budget_s
    a_best, best_score = flatten_conv_rl(a0, target,
                                         max_iters=20000,
                                         deadline=phase1_end,
                                         n=n,
                                         fft_len_conv=fft_len_conv,
                                         fft_len_factor=fft_len_factor,
                                         gamma=0.5)

    # Phase 2 – exponentiated‑gradient descent on the exact max‑convolution
    if time.time() < deadline:
        a_best, best_score = exp_gradient_optimize(
            a_best,
            deadline,
            n,
            fft_len_conv,
            rng,
            eta0=18.0)
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # Phase 3 – greedy local refinement
    if time.time() < deadline:
        a_best, best_score = local_refinement(a_best,
                                               best_score,
                                               deadline,
                                               n,
                                               fft_len_conv,
                                               rng)
        print(f"[local‑refine] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation using the provided scorer
    # --------------------------------------------------------------
    final_seq = a_best.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
