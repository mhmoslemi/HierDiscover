"""
step‑function optimiser – 1000 s budget (enhanced, n≈8000)

Algorithm Overview
------------------
1. **Large‑scale warm‑starts** – Sample many shapes from diverse
   parametric families (exponential‑edge, power‑edge, cosine,
   piecewise‑linear, sparse spikes, mixed).  Each shape is
   normalised to a probability vector (∑a=1) and scored with the fast
   FFT proxy 2·n·max(conv(a,a)).  The best `K_warm` candidates are kept.

2. **Focused L‑BFGS refinement** – Each of the top candidates is
   refined by a cascade of L‑BFGS runs on a smoothed log‑sum‑exp
   surrogate of the L∞–objective.  The λ‑schedule proceeds from 0.3
   down to 1e‑7, with a generous iteration budget.  The best refined
   vector becomes the seed for the next stage.

3. **Mirror (exponentiated)‑gradient descent** – A long run of
   multiplicative updates on the exact max‑convolution objective,
   using a momentum‑averaged sub‑gradient and a slowly decaying step
   size.  The tolerance for the max‑mask is raised to 1e‑6·max to
   obtain a smoother descent direction.

4. **Ultra‑fine L‑BFGS polish** – A final cascade with extremely low
   λ values (≤1e‑7) and a tighter convergence tolerance.

5. **Simulated‑annealing hill‑climb** – Random mass transfers with a
   Metropolis acceptance rule finish the search.

All phases respect the global 1000 s deadline; the best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation
# ----------------------------------------------------------------------

        [
            "uniform",
            "exp_edge",
            "power_edge",
            "cosine",
            "piecewise",
            "bimodal",
            "sparse",
            "mixed",
        ]
    )
    x = np.arange(n, dtype=float)

    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "cosine":
        # a_i ∝ 1 - cos(2π i / (n-1))   (zero at edges, max in centre)
        a = 1.0 - np.cos(2.0 * np.pi * x / (n - 1))

    elif mode == "piecewise":
        # linear decay from edges to a flat plateau in the middle
        plateau_frac = rng.uniform(0.2, 0.5)  # fraction of length that stays flat
        plateau_len = int(np.round(plateau_frac * n))
        left_len = (n - plateau_len) // 2
        right_len = n - plateau_len - left_len
        left = np.linspace(1.0, 0.0, left_len, endpoint=False)
        right = np.linspace(0.0, 1.0, right_len, endpoint=False)
        mid = np.zeros(plateau_len)
        a = np.concatenate([left, mid, right])

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        # a few random spikes (5‑15 % of entries)
        m = rng.integers(int(0.05 * n), int(0.15 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    else:  # mixed
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    # avoid exact zeros and normalise to a probability vector
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # ---- softmax ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution (FFT) ----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12  # scale λ to keep smoothing relative

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient contribution for each a_j (size n)
    g_vec = conv_rev[n - 1 : n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
    verbose: bool = False,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(
                    f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}"
                )
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 20.0,
    max_iters: int = 8_000_000,
):
    """
    Mirror‑SGD (exponentiated gradient) on the exact max‑convolution.
    Uses a momentum‑averaged sub‑gradient and a 1/√t step‑size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # momentum for the sub‑gradient
    g_avg = np.zeros_like(a)
    beta = 0.9

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- exact convolution ----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        # mask includes entries very close to the maximum (tolerance ≥ 1e‑6·max)
        tol = max_val * 1e-6 + 1e-12
        mask = (conv >= max_val - tol).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # size n

        # exponential‑moving average of the sub‑gradient
        g_avg = beta * g_avg + (1.0 - beta) * g

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # cheap periodic score check (every 32 iterations)
        if (it & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(
                    f"[exp‑grad] iter={it:,} fast_score={best_score:.9f}"
                )
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature linearly cools from 5 % of the current score.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25  # maximal fraction of source mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more flexibility for flattening
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Timing allocation (approximate)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.30 * budget_s  # warm‑starts
    phase2_end = start_time + 0.60 * budget_s  # L‑BFGS cascade
    phase3_end = start_time + 0.88 * budget_s  # exponentiated‑gradient
    # remaining time will be used for final polish

    # ------------------------------------------------------------------
    # Phase 1 – generate many diverse warm‑starts, keep the best K
    # ------------------------------------------------------------------
    K_warm = 12
    warm_candidates = []  # list of (score, a) sorted ascending
    best_score = float("inf")
    best_a = None

    # reuse previous construction if it exists (interpolated to new n)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            best_a = a0
            best_score = fast_score(best_a, fft_len)
            warm_candidates.append((best_score, best_a.copy()))
            print(f"[init] previous best fast_score={best_score:.9f}")

    while time.time() < phase1_end:
        a = generate_candidate_shape(n, rng)
        score = fast_score(a, fft_len)

        # maintain top‑K list
        if len(warm_candidates) < K_warm:
            warm_candidates.append((score, a.copy()))
            warm_candidates.sort(key=lambda tup: tup[0])
        else:
            if score < warm_candidates[-1][0]:
                warm_candidates[-1] = (score, a.copy())
                warm_candidates.sort(key=lambda tup: tup[0])

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[warm‑start] new best fast_score={best_score:.9f}")

    # ensure we have at least one candidate
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – refined L‑BFGS cascade on the top candidates
    # ------------------------------------------------------------------
    lam_schedule = [
        0.3,
        0.15,
        0.07,
        0.03,
        0.015,
        0.008,
        0.004,
        0.002,
        0.001,
        5e-04,
        2e-04,
        1e-04,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
        2e-06,
        1e-06,
        5e-07,
        2e-07,
        1e-07,
    ]

    # iterate over the warm‑start pool (best first) while time permits
    for rank, (sc, cand) in enumerate(warm_candidates):
        if time.time() >= phase2_end:
            break
        print(f"[L‑BFGS phase2] refining candidate #{rank + 1} (score={sc:.9f})")
        refined_a, refined_sc = lbfgs_cascade(
            cand,
            lam_schedule,
            maxiter_per_lam=1500,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
            verbose=True,
        )
        if refined_sc < best_score:
            best_score = refined_sc
            best_a = refined_a.copy()
            print(f"[L‑BFGS phase2] new overall best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        print("[exp‑grad] starting long mirror‑SGD")
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            eta0=25.0,
            max_iters=12_000_000,
        )
        print(f"[exp‑grad] best fast_score after phase3={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – final ultra‑fine L‑BFGS polish (very low λ)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
            verbose=True,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[final‑LBFGS] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[final‑anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
