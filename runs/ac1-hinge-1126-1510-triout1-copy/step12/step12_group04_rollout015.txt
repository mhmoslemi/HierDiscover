def fast_score(a, fft_len):

    n = a.shape[0]
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[:2*n-1]
    return float(2.0 * n * conv.max())

"""
Propose a low‑evaluation step function.

Strategy
--------

1. **Diverse start pool** – uniform, classic windows, random families
   (the same helpers as the reference implementation).

2. **Phase 1 – log‑sum‑exp L‑BFGS cascade**  
   Minimise a smooth surrogate of the max‑convolution with a decreasing
   λ‑schedule.  Uses `scipy.optimize.minimize` (L‑BFGS‑B).  All FFTs are
   performed with the real‑FFT (`rfft`) for speed.

3. **Phase 2 – Adam‑accelerated exponentiated‑gradient (mirror SGD)**  
   Optimises the *true* max‑convolution.  The sub‑gradient of the
   max‑convolution is obtained by convolving the current mask (the set of
   indices attaining the max) with the reversed vector.  The update is
   performed in log‑space with Adam‑style adaptive moments, which gives
   much faster convergence than the simple 1/√t schedule used before.

4. **Phase 3 – final L‑BFGS polish + short simulated‑annealing**  
   A fine λ‑schedule followed by a tiny moving‑average smoothing and a
   brief Metropolis hill‑climb.

All phases respect the 1000 s global deadline; the best vector found is
stored in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Exact score for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    The convolution is evaluated via a real FFT.
    """
    n = a.shape[0]
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    b /= b.sum()
    return b


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed maximum of the convolution and its gradient w.r.t.
    the log‑variables y (so that a = softmax(y)).
    """
    # ---- softmax ----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution (real FFT) ----
    A = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth_max w.r.t. convolution entries ----
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, fft_len)
    W = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(A_rev * W, fft_len)[: 2 * n - 1]
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # ---- chain rule through softmax ----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter_per_lam,
                    "gtol": 1e-12,
                    "disp": False,
                },
            )
        except Exception:
            continue
        y = res.x
        # back‑to probability simplex
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score

# Adam‑accelerated exponentiated‑gradient on the true L∞‑objective

def adam_exp_gradient_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               eta0: float = 0.5,
                               beta1: float = 0.9,
                               beta2: float = 0.999,
                               eps: float = 1e-8,
                               max_iters: int = 30_000_000):
    """
    Mirror‑SGD with Adam moments for the exact max‑convolution.
    """
    # initialise log‑variables
    L = np.log(a_start + 1e-16)
    m = np.zeros_like(L)
    v = np.zeros_like(L)

    best_a = None
    best_score = float("inf")
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        # ----- probability vector from log‑variables -----
        L_max = np.max(L)
        expL = np.exp(L - L_max)
        a = expL / expL.sum()

        # ----- convolution -----
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12

        # ----- mask of maximal entries -----
        mask = (conv >= max_val - eps_val).astype(np.float64)

        # ----- sub‑gradient of max‑convolution w.r.t. a -----
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, fft_len)
        M_f = np.fft.rfft(mask, fft_len)
        conv_rev = np.fft.irfft(A_rev * M_f, fft_len)[: 2 * n - 1]
        g_raw = 2.0 * conv_rev[n - 1: n - 1 + n]      # size n

        # ----- gradient w.r.t. log‑variables (mirrored SGD) -----
        avg = np.dot(a, g_raw)
        grad_log = a * (g_raw - avg)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        L = L - eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional evaluation -----
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = it
                print(f"[Adam‑exp‑grad] it={it:,} fast_score={best_score:.9f}")

        # ----- jitter if stalled -----
        if (it - last_improve) > 300_000:
            a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)
            L = np.log(a + 1e-16)
            last_improve = it

    if best_a is None:
        # fallback – use the latest a
        best_a = a.copy()
    return best_a, best_score

def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20  # maximal fraction of source mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

    # 3) random families
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.09 * budget_s   # L‑BFGS warm‑up
    phase2_end = start_time + 0.80 * budget_s   # long Adam‑exp‑grad
    # remaining time → final polish + anneal

    # ------------------------------------------------------------------
    # Phase 1 – quick L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.3,
        0.2,
        0.12,
        0.07,
        0.04,
        0.022,
        0.012,
        0.006,
        0.003,
        0.0015,
        7e-4,
        3e-4,
        1.5e-4,
        8e-5,
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float("inf")
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter before L‑BFGS to diversify
        if rng.random() < 0.35:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.015),
                                rng=rng)
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # optional symmetry enforcement (often helpful)
    sym_a = symmetrize(best_a)
    sym_score = fast_score(sym_a, fft_len)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        print(f"[Phase1] symmetry improved fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam‑accelerated exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=0.6,          # slightly larger initial learning rate
            max_iters=40_000_000,
        )
        print(f"[Phase2] Adam‑exp‑grad best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final L‑BFGS polish + short anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # finer λ‑schedule
        fine_lam_schedule = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase3‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        # tiny moving‑average smoothing before annealing
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = np.maximum(smoothed, 0.0)
        smoothed /= smoothed.sum()
        smoothed_score = fast_score(smoothed, fft_len)
        if smoothed_score < best_score:
            best_a = smoothed
            best_score = smoothed_score
            print(f"[Phase3‑smooth] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[Anneal] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
