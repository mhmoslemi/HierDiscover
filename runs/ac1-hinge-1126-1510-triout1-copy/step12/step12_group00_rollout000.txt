"""
Improved search for a low‑score step function.

The algorithm works in several stages, each respecting the overall
time budget (default 1000 s):

1. **Initial pool** – deterministic windows (spectral, Gaussian,
   Hann, Blackman, triangular, Kaiser) plus several random draws.
   If a previous solution is available it is interpolated to the
   current problem size.

2. **Smoothed L‑BFGS** – a continuation method that minimises a
   log‑sum‑exp approximation of the maximal autocorrelation.
   The smoothing parameter λ is decreased aggressively and each λ
   receives many L‑BFGS iterations.

3. **Adam with a top‑k mask** – a custom Adam optimiser that
   back‑propagates the gradient of the sum of the largest k
   convolution entries (k is slowly reduced).  This flattens the
   whole autocorrelation.

4. **Projected sub‑gradient descent on the exact max** – a few
   iterations of a simple sub‑gradient scheme on the true max‑
   objective, with Euclidean projection onto the simplex.

5. **Simulated annealing** – mass‑transfer moves with a geometric
   temperature schedule.

6. **Final greedy hill‑climb** – tiny deterministic moves that
   strictly improve the exact score.

All vectors are kept on the probability simplex (non‑negative,
sum = 1) so that the official ``evaluate_sequence`` does not rescale.
The optimisation is performed for a larger length (n = 8192) than
the reference implementation, which gives the algorithm more
degrees of freedom while still fitting easily in the time budget.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.01):
    """
    Adam optimiser that uses a top‑k mask (rather than the single
    maximal entry) to flatten the whole convolution.
    The top‑k fraction linearly decays from start_top to end_top.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    # time bookkeeping
    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # dynamic top‑k fraction
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        # mask of largest entries
        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of the masked sum (exact sub‑gradient of Σ mask_i * conv_i)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # d Σmask*conv / d a

        grad_a = 2.0 * n * g_a                         # full objective gradient

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule – slower decay than the original
        lr = 0.30 / np.sqrt(t)          # a slightly larger initial step

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def subgradient_max(a: np.ndarray,
                    fft_len: int,
                    n: int,
                    eps: float = 1e-12):
    """
    Compute the sub‑gradient of the exact max‑convolution objective.
    Returns (gradient, current max value).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    # mask of active (max‑tight) entries
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fm = np.fft.rfft(mask, fft_len)
    rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

    grad = 2.0 * rev_grad[(n - 1):(n - 1 + n)]    # derivative of Σ mask_i * conv_i
    return grad, max_val


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_lr: float = 0.2,
                        min_lr: float = 1e-4,
                        max_iter: int = 60000):
    """
    Simple projected sub‑gradient descent on the exact max‑objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    start_time = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        grad, _ = subgradient_max(a, fft_len, n)
        # diminishing step size
        lr = max(min_lr, init_lr / np.sqrt(it + 1))
        a_new = a - lr * grad
        a_new = simplex_projection(a_new)

        a = a_new
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 250000):
    """
    Simple simulated‑annealing over the simplex.  Small mass‑transfer
    moves are proposed; the acceptance probability follows a geometric
    temperature schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.18   # up to ~18% of source mass
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        # accept if better or with Metropolis probability
        if sc < best_score:
            a = cand
            best_score = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_score) / max(T, 1e-18))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 50000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score


def upsample_sequence(seq: np.ndarray, n: int) -> np.ndarray:
    """Interpolate an existing sequence to length n and renormalise."""
    if seq.ndim != 1 or seq.size == 0:
        return np.full(n, 1.0 / n)
    x_old = np.linspace(0, 1, seq.size)
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, seq)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a

    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # random restart around the current best
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.45
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam with top‑k mask (≈40 % of total time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑topk] start exact optimisation")
        best_a, best_score = adam_topk_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            start_top=0.15,
            end_top=0.01
        )
        print(f"[Adam‑topk] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent on exact max (≈10 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + 0.10 * budget_s
        best_a, best_score = subgradient_descent(
            best_a,
            subgrad_deadline,
            n,
            fft_len,
            rng,
            init_lr=0.18,
            min_lr=5e-5,
            max_iter=40000
        )
        print(f"[Sub‑grad] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing (≈15 % of total time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.15 * budget_s)
        best_a, best_score = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-9,
            max_moves=300000
        )
        print(f"[Sim‑Anneal] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=60000,
            delta_frac=0.03
        )
        print(f"[Hill‑climb] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Optional symmetrisation (the optimum is expected to be symmetric)
    # ------------------------------------------------------------------
    best_a = (best_a + best_a[::-1]) / 2.0
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[symmetrised] final fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
