"""
step‑function optimiser – 1000 s budget (enhanced)

Algorithm enhancements
----------------------
* **Richer warm‑starts** – added beta‑distribution family, triangular,
  linear‑decay and power‑decay families.  Small‑scale candidates are
  generated (n≈800‑1200) and interpolated to the target size, providing
  diverse initial shapes that are cheap to evaluate.
* **Multi‑scale initialisation** – a fast optimisation on a small problem
  (n≈1000) is performed, then the solution is up‑sampled and refined on
  the full size.
* **Adam optimisation** – after the L‑BFGS cascade we run an Adam
  optimiser on the log‑sum‑exp (smooth‑max) surrogate, which often
  escapes the plateau that L‑BFGS can get stuck on.
* **More aggressive exponentiated‑gradient** – larger learning‑rate,
  longer run and momentum are used.
* **Extended polishing** – a finer λ‑schedule for the final L‑BFGS
  and a longer simulated‑annealing hill‑climb.

The code respects the 1000 s deadline, stores the best discovered sequence
in the global variable ``height_sequence_1`` and returns it as a list of
floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the official score (assumes sum(a)=1)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def upscale_shape(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 1e-12)
    a_large /= a_large.sum()
    return a_large


# ----------------------------------------------------------------------
# Parametric warm‑start generation
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Create a diverse probability vector using several families.
    The returned vector is already normalised to sum to 1.
    """
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "bimodal",
            "sparse",
            "mixed",
            "beta",
            "triangular",
            "linear_decay",
            "power_decay",
        ]
    )
    x = np.arange(n)
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.001, 0.025)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == "mixed":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    elif mode == "beta":
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        t = (x + 0.5) / n
        a = (t ** (alpha - 1)) * ((1 - t) ** (beta - 1))

    elif mode == "triangular":
        # peak in the middle, zero at the edges
        a = 1.0 - np.abs((2 * x) - (n - 1)) / (n - 1)

    elif mode == "linear_decay":
        # decreasing from left to right
        a = (n - x).astype(float)

    else:  # power_decay
        p = rng.uniform(0.5, 3.0)
        a = (x + 1) ** (-p)

    # avoid exact zeros, then normalise
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]  # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS cascade
# ----------------------------------------------------------------------


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score

def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lr: float = 0.02,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  lam_schedule=None):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    """
    if lam_schedule is None:
        lam_schedule = [
            0.1,
            0.05,
            0.02,
            0.01,
            0.005,
            0.002,
            0.001,
            5e-04,
            2e-04,
            1e-04,
            5e-05,
            2e-05,
            1e-05,
            5e-06,
            2e-06,
            1e-06,
            5e-07,
            2e-07,
            1e-07,
        ]

    start = time.time()
    total_time = deadline - start
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        # determine current lambda by linear schedule over time
        elapsed = time.time() - start
        frac = elapsed / total_time
        idx = min(int(frac * len(lam_schedule)), len(lam_schedule) - 1)
        lam = lam_schedule[idx]

        f, grad = objective_and_grad(y, n, fft_len, lam)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        delta = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - delta

        # periodic score check
        if t % 64 == 0:
            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] iter={t} fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0,
                          max_iters: int = 5_000_000):
    """
    Mirrored SGD (exponentiated‑gradient) on the exact max‑convolution.
    Uses a momentum‑averaged sub‑gradient.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    g_avg = np.zeros_like(a)
    beta = 0.9

    it = 0
    while time.time() < deadline and it < max_iters:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        g_avg = beta * g_avg + (1.0 - beta) * g
        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g_avg, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        if (it & 15) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] iter={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – a moderate size works well for the time budget
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # allocate time to the different phases
    phase1_end = start_time + 0.25 * budget_s
    phase2_end = start_time + 0.45 * budget_s
    phase3_end = start_time + 0.75 * budget_s

    # ------------------------------------------------------------------
    # Phase 1 – diverse warm‑starts (including small‑scale up‑sampling)
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # reuse previous construction if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                a0 = upscale_shape(old, n)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 1e-12)
            a0 /= a0.sum()
            best_a = a0
            best_score = fast_score(best_a, fft_len)
            print(f"[init] previous fast_score={best_score:.9f}")

    while time.time() < phase1_end:
        # 30 % of the time we create a small candidate and upscale it
        if rng.random() < 0.30:
            n_small = rng.integers(800, 1200)
            a_cand = generate_candidate_shape(n_small, rng)
            a_cand = upscale_shape(a_cand, n)
        else:
            a_cand = generate_candidate_shape(n, rng)

        # occasional jitter of a promising candidate
        if best_a is not None and rng.random() < 0.15:
            a_cand = add_noise(a_cand, sigma=rng.uniform(0.001, 0.008), rng=rng)

        score = fast_score(a_cand, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_cand.copy()
            print(f"[warm‑start] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS cascade with a refined λ‑schedule
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [
            0.5,
            0.3,
            0.2,
            0.1,
            0.05,
            0.02,
            0.01,
            5e-03,
            2e-03,
            1e-03,
            5e-04,
            2e-04,
            1e-04,
            5e-05,
            2e-05,
            1e-05,
        ]
        best_a, best_score = lbfgs_cascade(
            best_a,
            lam_schedule,
            maxiter_per_lam=2500,
            deadline=phase2_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS cascade] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            lr=0.03,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )
        print(f"[Adam] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – long exponentiated‑gradient descent (exact L∞)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            eta0=40.0,
            max_iters=12_000_000,
        )
        print(f"[exp‑grad] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final polishing: very low‑λ L‑BFGS + simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07, 5e-08]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=3000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score, best_a = sc_refined, a_refined.copy()
            print(f"[final‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[final‑anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
