The optimizer builds a non‑negative step function `a = (a₀,…,a_{n‑1})` that minimises  

\[
\text{score}(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the discrete convolution of `a` with itself.  
The algorithm is a multi‑phase search:

| Phase | Idea | Details |
|------|------|---------|
| 0 | **Initial pool** | deterministic windows, spectral guess, many random Dirichlet draws, and (if present) the previous best solution; keep the best w.r.t. the exact fast score. |
| 1 | **Smoothed L‑BFGS** | minimise a log‑sum‑exp smooth approximation of `max(a*a)`.  The smoothing parameter λ is decreased aggressively (down to `1e‑6`).  Every λ stage starts from the previous result; a few random restarts keep the search diverse. |
| 2 | **Exact Adam with full‑max mask** | Adam on the *true* objective.  The gradient is the sub‑gradient of `max(a*a)` obtained by back‑propagating a mask that is `1` on all (near) maximal convolution entries. |
| 3 | **Linear‑programming polishing** | With the current vector `f`, compute `M = max(f*f)`.  Solve the LP  

\[
\max\;\sum_j g_j\quad\text{s.t.}\;g_j\ge0,\;(f*g)_k\le M\;(k=0..2n-2),
\]

which moves mass from peaks to valleys without increasing the worst convolution value.  The optimal `g` is renormalised and used as the new candidate. |
| 4 | **Simulated annealing** | Small random mass‑transfer moves accepted with a geometric temperature schedule. |
| 5 | **Greedy hill‑climb** | Strictly improving tiny transfers, finishing the optimisation. |

All phases respect the global 1000 s budget.  
The final sequence is returned as a plain `list[float]`; `evaluate_sequence` rescales it internally.

```python
"""
Multi‑phase optimiser for the step‑function benchmark.

Phases:
0 – diverse deterministic / random initial vectors,
1 – smoothed L‑BFGS (log‑sum‑exp of the convolution),
2 – exact Adam on the true max‑convolution objective,
3 – linear‑programming polishing (maximise total mass under the
    current worst‑convolution bound),
4 – simulated annealing,
5 – greedy hill‑climb.

All phases share a common deadline (default 1000 s).  The best
vector found is returned as a list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (used for bookkeeping only)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto the simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_exact_max_optimize(a_start: np.ndarray,
                            deadline: float,
                            n: int,
                            fft_len: int,
                            rng: np.random.Generator,
                            eps: float = 1e-12):
    """
    Adam optimiser that uses the *exact* max‑convolution gradient
    (sub‑gradient obtained from a mask on all active maximal entries).
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # exact convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        # mask of all entries that are (numerically) maximal
        mask = (conv >= M - eps).astype(np.float64)

        # gradient of the masked max (sum of active gradients)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]      # d max / d a (ignoring the 2·n factor)

        grad_a = 2.0 * n * g_a                         # full objective gradient

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.35 / np.sqrt(t)          # slowly decaying learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def lp_polish(f: np.ndarray,
              M: float,
              deadline: float,
              n: int,
              max_iter: int = 3):
    """
    Linear‑programming improvement:
    maximise Σ g_j subject to (f * g)_k ≤ M  ∀k,  g ≥ 0.
    Returns the best feasible g (renormalised).
    """
    from scipy.sparse import csr_matrix

    best_g = f.copy()
    best_sum = f.sum()
    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        # Build Toeplitz matrix A where (A @ g)[k] = (f * g)[k].
        # Row k receives f[i] whenever i contributes to shift k.
        rows = np.arange(n)[:, None] + np.arange(n)          # shape (n,n)
        rows = rows.ravel()
        cols = np.repeat(np.arange(n), n)
        data = np.repeat(f, n)
        A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        A = A.tocsr()

        # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # LP failed – abort further polishing
            break

        g = np.maximum(res.x, 0.0)
        if g.sum() == 0:
            break
        # Normalise to unit L1 (the objective is homogeneous)
        g = g / g.sum()

        # Verify that the new vector respects the bound (numerical safety)
        fa = np.fft.rfft(f, 2 * n - 1)
        fg = np.fft.rfft(g, 2 * n - 1)
        conv_fg = np.fft.irfft(fa * fg, 2 * n - 1)[: 2 * n - 1]
        if conv_fg.max() > M * 1.000001:          # slight tolerance
            # Scale down to satisfy the constraint
            g = g * (M / conv_fg.max())

        sc = fast_score(g, 2 * n - 1, n)
        if sc < fast_score(best_g, 2 * n - 1, n):
            best_g = g.copy()
            best_sum = g.sum()
    return best_g


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 250000):
    """
    Simple simulated‑annealing over the simplex. Small mass‑transfer moves
    are proposed; the acceptance probability follows a geometric temperature schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.18   # up to ~18 % of source mass
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_score:
            a = cand
            best_score = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (trade‑off between granularity and speed)
    # --------------------------------------------------------------
    n = 4096                         # power‑of‑two size – good FFT behaviour
    fft_len = next_pow2(2 * n - 1)   # length for zero‑padded FFTs

    # --------------------------------------------------------------
    # Build initial candidate pool
    # --------------------------------------------------------------
    candidates = []

    # interpolate previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    for beta in [4.0, 6.0, 9.0, 12.0]:
        candidates.append(kaiser_init(n, beta))

    # randomised families (more diversity)
    for _ in range(10):
        candidates.append(random_initial(n, rng))

    # keep the best deterministic / random start
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Phase 1 – Smoothed L‑BFGS (≈30 % of total time)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # occasional random restart to escape poor basins
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Exact Adam on the true max‑convolution (≈45 % of total time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑exact] start optimisation")
        best_a, best_score = adam_exact_max_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng
        )
        print(f"[Adam‑exact] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – Linear‑programming polishing (≈15 % of total time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + min(0.15 * budget_s, deadline - time.time())
        print("[LP polishing] start")
        # current bound
        conv_cur = np.fft.irfft(np.fft.rfft(best_a, fft_len) ** 2, fft_len)[: 2 * n - 1]
        M_cur = conv_cur.max()
        # run a few polishing steps
        best_a = lp_polish(best_a, M_cur, lp_deadline, n, max_iter=2)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[LP polishing] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated annealing (≈20 % of total time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sa_deadline = time.time() + min(remaining, 0.20 * budget_s)
        best_a, best_score = simulated_annealing(
            best_a,
            sa_deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-8,
            max_moves=200000
        )
        print(f"[Sim‑Anneal] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 5 – Final greedy hill‑climb (≤5 % of total time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = time.time() + min(30.0, deadline - time.time())
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=50000,
            delta_frac=0.04
        )
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
```
