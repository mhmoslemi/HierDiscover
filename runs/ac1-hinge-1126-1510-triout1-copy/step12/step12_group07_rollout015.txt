   cosine‑power, sinc‑squared).  All candidates are normalised to the
   probability simplex.

2. **L‑BFGS cascade on a log‑sum‑exp surrogate** – a short sequence of
   decreasing smoothing parameters (λ) quickly drives the solution
   into a low‑value basin.

3. **Projected‑gradient descent on the exact L∞‑objective** – a fast
   “sub‑gradient” method that directly minimises the maximum of the
   convolution (the true objective).  It repeatedly identifies the
   offending shift, moves mass away from the contributing indices and
   re‑projects onto the simplex.

4. **Metropolis annealing polish** – a few thousand random mass‑swap
   moves with a linearly cooling temperature to squeeze the last few
   digits.

The routine respects a total time budget (default 1000 s).  The best
sequence found is stored globally as `height_sequence_1` and also
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT (returns full length)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * len(a) - 1]
    return conv


def grad_max_convolution(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient (sub‑gradient) of the convolution entry C_k = (a*a)[k]
    w.r.t. a.  For each i the derivative is 2·a_{k‑i} when the index
    k‑i lies inside the vector, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random probability vector from several simple parametric families."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sparse', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        frac = rng.uniform(0.006, 0.03)
        m = max(1, int(n * frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:   # dirichlet
        alpha = rng.uniform(0.05, 0.5)
        a = rng.gamma(alpha, size=n)
        a[a < 1e-14] = 0.0
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def sinc_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n) - (n - 1) / 2.0
    scale = rng.uniform(0.05, 0.15) * n
    a = np.sinc(x / scale) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def subset_uniform_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random subset (sparse support)."""
    m = rng.integers(int(0.02 * n), int(0.2 * n))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """Soft‑max (log‑sum‑exp) surrogate for max‑convolution and its gradient."""
    # soft‑max to stay on the simplex automatically
    max_y = np.max(y)
    e = np.exp(y - max_y)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    max_c = conv.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)      # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z
    # back‑prop through a*a → a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, n=fft_len)
    g_vec = conv_rev[n - 1: n - 1 + n]               # size n

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS sub‑problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)           # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact L∞ objective
# ----------------------------------------------------------------------


def pgd_optimize(a_start: np.ndarray,
                 max_iter: int,
                 deadline: float,
                 fft_len: int,
                 alpha0: float = 0.2):
    """
    Simple projected (Euclidean) gradient descent on max(conv(a,a)).
    Each step moves a in the negative direction of the sub‑gradient
    of the currently maximal convolution entry and projects back onto
    the probability simplex.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    t = 0
    while time.time() < deadline and t < max_iter:
        # full convolution via FFT
        conv = conv_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        g = grad_max_convolution(a, k_max)

        t += 1
        alpha = alpha0 / np.sqrt(t)   # diminishing step‑size
        a_new = a - alpha * g
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing (enhanced)
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """Metropolis annealer with three simple move types."""
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:                       # transfer
                j = rng.integers(0, n)
            if i == j:
                continue
            # move from larger to smaller entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises the evaluation metric.  The best sequence is saved in the
    global variable `height_sequence_1` and also returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                     # a size that balances quality and speed
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diversified pool of starting vectors
    # --------------------------------------------------------------
    pool = []

    # 0) Previous construction, if present
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) Classic families
    for _ in range(8):
        pool.append(random_initial(n, rng))
    for _ in range(3):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(sinc_initial(n, rng))
    for _ in range(4):
        pool.append(subset_uniform_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05     # tiny safety margin

    phase1_end = start_time + 0.15 * budget_s   # brief L‑BFGS warm‑start
    phase2_end = start_time + 0.70 * budget_s   # PGD main work
    phase3_end = start_time + 0.90 * budget_s   # annealing polish

    # --------------------------------------------------------------
    # Phase 1 – quick L‑BFGS warm‑start on smoothed surrogate
    # --------------------------------------------------------------
    lam_sched_l1 = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.02, 0.015, 0.01, 0.007,
        0.005, 0.003, 0.002, 0.001, 5e-4,
        2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7
    ]
    maxiter_per_lam = 400   # a few hundred iterations per λ

    best_a = None
    best_score = float('inf')
    # run a few short L‑BFGS runs until phase‑1 deadline
    while time.time() < phase1_end:
        a0 = rng.choice(pool)
        a_opt, sc_opt = lbfgs_cascade(a0,
                                      lam_sched_l1,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – projected gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    while time.time() < phase2_end:
        # Choose a fresh start: either the current best or a random pool member
        if rng.random() < 0.3:
            a_start = best_a.copy()
        else:
            a_start = rng.choice(pool)

        # Run PGD for a modest number of iterations (adapt to remaining time)
        remaining = phase2_end - time.time()
        max_iters = int(min(5000, remaining * 250))  # ~250 iterations per second
        a_pg, sc_pg = pgd_optimize(a_start,
                                   max_iter=max_iters,
                                   deadline=phase2_end,
                                   fft_len=fft_len,
                                   alpha0=0.25)
        if sc_pg < best_score:
            best_score = sc_pg
            best_a = a_pg.copy()
            print(f"[PGD] new best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – short annealing polish
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing_improved(
            best_a,
            best_score,
            phase3_end,
            n,
            rng,
        )
        print(f"[anneal] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Optional mixing of top candidates (tiny extra polish)
    # --------------------------------------------------------------
    candidates = [best_a]
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.shape == best_a.shape:
            candidates.append(prev / prev.sum())
    for i in range(len(candidates)):
        for j in range(i + 1, len(candidates)):
            for _ in range(3):
                w = rng.random()
                mix = w * candidates[i] + (1 - w) * candidates[j]
                mix /= mix.sum()
                sc = fast_score(mix, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = mix.copy()
                    print(f"[mix] new best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the environment
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
