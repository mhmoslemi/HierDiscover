"""Simple evolutionary optimizer for the step‑function objective.

The optimizer works in three quick phases:

1. **Uniform baseline** – the uniform distribution (all heights equal) gives the
   theoretical lower bound of the objective (score ≈ 2).  This is kept as a
   reference.

2. **Random Dirichlet sampling** – we draw many random probability vectors
   from a Dirichlet distribution with different concentration parameters.
   For every sample the exact score is computed with the fast FFT‑based
   routine.  The best‑scoring sample is retained.

3. **Mass‑transfer hill‑climb** – starting from the best sample we perform
   a greedy local search that repeatedly moves a small fraction of the mass
   from one index to another.  Moves are accepted only if they improve the
   exact score.  This phase refines the solution without any heavy
   optimisation machinery, keeping the implementation lightweight and well
   within the 1000 s budget.

The function returns the best sequence found as a plain list of Python
floats; `evaluate_sequence` will rescale it internally, so we only need to
ensure non‑negativity and that the sum is > 0.01.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a probability vector from Dirichlet(alpha,…,alpha)."""
    # Using Gamma(α,1) trick – works for any α>0
    g = rng.gamma(alpha, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return g / s


def hill_climb(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    time_limit: float,
) -> np.ndarray:
    """Greedy local search: move a tiny amount of mass between two coordinates."""
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    # pre‑allocate arrays for speed
    start_time = time.time()
    while time.time() - start_time < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction (up to 5 %) of the source entry
        delta = a[i] * rng.random() * 0.05
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep it a probability vector (no negative entries)
        if cand[i] < 0:
            continue
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
            a = cand
    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation score.  The routine respects the 1000 s budget and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – choose a moderate n so that FFTs are cheap
    # --------------------------------------------------------------
    n = 4096                           # same size as the reference script
    fft_len = next_pow2(2 * n - 1)     # length for the zero‑padded FFTs

    # --------------------------------------------------------------
    # Phase 1 – uniform baseline (theoretical optimum for a
    #           perfectly balanced distribution)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[baseline] uniform score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – random Dirichlet sampling
    # --------------------------------------------------------------
    phase2_end = time.time() + 0.50 * budget_s   # 50 % of the budget
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0]          # diverse concentration values
    samples_per_alpha = 2000                     # keep the loop cheap

    while time.time() < phase2_end:
        for alpha in alphas:
            a = random_dirichlet(n, alpha, rng)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[sample] new best score = {best_score:.8f} (α={alpha})")
        # small pause to allow the time check to be responsive
        if time.time() >= phase2_end:
            break

    # --------------------------------------------------------------
    # Phase 3 – greedy hill‑climb (fine‑tuning)
    # --------------------------------------------------------------
    remaining = budget_s - (time.time() - (time.time() - budget_s))
    # Give the hill‑climb roughly the remaining time (capped at 30 s)
    hill_time = min(30.0, max(0.0, remaining))
    if hill_time > 0.0:
        print(f"[hill‑climb] starting refinement for {hill_time:.1f}s")
        best_a = hill_climb(best_a, n, fft_len, rng, hill_time)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[hill‑climb] refined score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Return the final sequence (the optimizer works with normalized
    # vectors; evaluate_sequence rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
