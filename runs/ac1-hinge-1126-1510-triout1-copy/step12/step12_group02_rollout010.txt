    n = f.shape[0]
    conv = np.fft.irfft(np.fft.rfft(f, next_pow2(2*n-1)) * np.fft.rfft(f, next_pow2(2*n-1)), next_pow2(2*n-1))[: 2*n-1]
    # Actually compute conv at full length.
    # Determine indices of largest conv values:
    idx_sorted = np.argsort(-conv)   # descending
    top_idx = idx_sorted[:top_k]
    K = len(top_idx)
    
    A = np.zeros((K, n), dtype=np.float64)
    for row_idx, k in enumerate(top_idx):
        # compute slice range
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # indices of g affected: j in [j_start, j_end] inclusive.
        # coefficients are f_{k-j}
        # we can fill:
        # A[row_idx, j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
        idxs = np.arange(j_start, j_end+1)
        A[row_idx, j_start:j_end+1] = f[k - idxs]
    b = np.full(K, M_f)
    # Solve LP: max sum(g) s.t. A @ g <= b, g >= 0.
    # Equivalent to linear programming with objective c = -np.ones(n) (minimize -sum)
    res = linprog(c=-np.ones(n),
                  A_ub=A,
                  b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        # fallback: return f unchanged
        return f
    g0 = res.x
    sum_g = g0.sum()
    if sum_g <= 0:
        return f
    g = g0 / sum_g
    # Mix with some t
    t = 0.2
    f_new = (1-t)*f + t*g
    f_new /= f_new.sum()
    return f_new

"""
Propose a low‑score step function.
---------------------------------
The algorithm is a multi‑phase stochastic‑deterministic search:

1.  Build a very large pool of candidate shapes (classical windows,
    random draws, systematic Beta / raised‑cosine families, mixtures).
2.  Keep the top‑few candidates (by the exact fast_score, i.e. the
    objective for sum‑one sequences).
3.  Phase‑1: many short L‑BFGS refinements from blended starts.
4.  Phase‑2: Adam on the smooth log‑sum‑exp surrogate.
5.  Phase‑3: exact sub‑gradient descent on the true max‑convolution.
6.  Phase‑4: a *linear‑program* based refinement – for the current
    sequence f we solve a tiny LP that maximises the total mass of a
    new sequence g under the constraints (f * g)(k) ≤ max(f * f).
    The result is mixed with f (t≈0.2) and projected onto the simplex.
    Repeating this drives the max‑convolution down.
7.  Phase‑5: simulated‑annealing with many mass‑transfer moves.
8.  Phase‑6: a final greedy hill‑climb with very small steps.
9.  After each improvement the vector is symmetrised
    (a←(a+a[::-1])/2) and the better of the two is kept.
10. The best sequence is returned; the official ``evaluate_sequence``
    is called once at the end for reporting.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Fast exact objective (valid for sum‑one vectors)
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS/Adam)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a

def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a

def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def beta_mix_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two Beta densities with a random weight."""
    a1 = beta_init(n,
                   alpha=rng.exponential(scale=0.8) + 0.1,
                   beta=rng.exponential(scale=0.8) + 0.1)
    a2 = beta_init(n,
                   alpha=rng.exponential(scale=0.8) + 0.1,
                   beta=rng.exponential(scale=0.8) + 0.1)
    w = rng.random()
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 8000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.07):
    """
    Adam optimisation of the smoothed max‑convolution.
    Operates on logits y = log(a).  Adaptive learning rate helps escape
    plateaus that L‑BFGS may get stuck in.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = max(lam * lam_decay, 1e-12)
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score

def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry of the
    self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # compute convolution (FFT based)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())               # index of the peak

        # sub‑gradient of conv[idx] w.r.t. a:
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score

def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1
    return best_a, best_score

def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# ----------------------------------------------------------------------
# Linear‑program based refinement (new phase)
# ----------------------------------------------------------------------
def lp_refine(a_start: np.ndarray,
              current_best_score: float,
              deadline: float,
              fft_len: int,
              n: int,
              top_k: int = 30,
              mix_factor: float = 0.20):
    """
    Solve a tiny LP that maximises the total mass of a new vector g
    under the constraints (a * g)(k) ≤ max(a*a) for the top_k indices.
    Mix the solution with a and iterate while time permits.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = current_best_score

    while time.time() < deadline:
        # current self‑convolution and its maximum
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # indices of the largest entries (including the maximal one)
        idx = np.argpartition(conv, -top_k)[-top_k:]

        # build the constraint matrix A·g ≤ M
        K = len(idx)
        A = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(idx):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            # fill the slice with the appropriate f‑coefficients
            # f_i = a[i]; coefficient for g_j is a_{k-j}
            cols = np.arange(j_start, j_end + 1, dtype=int)
            A[row, j_start:j_end + 1] = a[k - cols]

        b = np.full(K, M)
        # maximise sum(g)  →  minimise -sum(g)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        g = g0 / sum_g

        # convex combination and projection onto the simplex
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        a_new /= a_new.sum()
        sc_new = fast_score(a_new, fft_len, n)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # if no improvement, break to avoid endless loop
            break
    return best_a, best_score

# ----------------------------------------------------------------------
# Symmetrisation helper
# ----------------------------------------------------------------------
def maybe_symmetrize(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Return the better of a and its mirror‑symmetric version."""
    a_sym = (a + a[::-1]) * 0.5
    a_sym /= a_sym.sum()
    sc_a = fast_score(a, fft_len, n)
    sc_sym = fast_score(a_sym, fft_len, n)
    return a_sym if sc_sym < sc_a else a

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # problem size (the classic size used in the reference code)
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a very large pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # previous solution, if any
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # a handful of Kaiser windows with varied betas
    for beta in (4.0, 8.0, 12.0):
        candidates.append(kaiser_init(n, beta))

    # random families (more than before)
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(5):
        candidates.append(pow_decay_init(n, rng))

    # systematic Beta draws (larger sample)
    for _ in range(2000):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))

    # raised‑cosine family
    for _ in range(400):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    # mixtures of two Beta densities
    for _ in range(500):
        candidates.append(beta_mix_init(n, rng))

    # ------------------------------------------------------------------
    # Initial screening – keep the best few candidates
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    scored = []
    for idx, cand in enumerate(candidates):
        sc = fast_score(cand, fft_len, n)
        scored.append((sc, cand))
    scored.sort(key=lambda x: x[0])
    best_score, best_a = scored[0]
    top_candidates = [c for _, c in scored[:20]]
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget split (soft limits, we always check the actual clock)
    # ------------------------------------------------------------------
    total_budget = budget_s
    phase1_end = start_time + 0.30 * total_budget
    phase2_end = start_time + 0.55 * total_budget
    phase3_end = start_time + 0.70 * total_budget
    phase4_end = start_time + 0.85 * total_budget
    phase5_end = start_time + 0.95 * total_budget
    # final phase uses whatever is left

    # ------------------------------------------------------------------
    # Phase‑1 – many short L‑BFGS runs from blended starts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # pick a random top candidate and blend with a fresh random vector
        base = rng.choice(top_candidates)
        rand = random_initial(n, rng)
        alpha = rng.random() * 0.35            # blending weight ≤0.35
        a_start = (1.0 - alpha) * base + alpha * rand
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            best_a = maybe_symmetrize(best_a, fft_len, n)
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase‑2 – Adam optimisation (smooth surrogate)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           max_iters=8000,
                                           lam_init=0.30,
                                           lam_decay=0.9993,
                                           lr=0.07)
        best_a = maybe_symmetrize(best_a, fft_len, n)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase‑3 – exact sub‑gradient descent (focuses on the current peak)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase3_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.30)
        best_a = maybe_symmetrize(best_a, fft_len, n)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase‑4 – LP‑based refinement (new)
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = lp_refine(best_a,
                                       best_score,
                                       phase4_end,
                                       fft_len,
                                       n,
                                       top_k=40,
                                       mix_factor=0.20)
        best_a = maybe_symmetrize(best_a, fft_len, n)
        print(f"[LP‑refine] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase‑5 – Simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < phase5_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 phase5_end,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=1e-7,
                                                 max_moves=300000)
        best_a = maybe_symmetrize(best_a, fft_len, n)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase‑6 – final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=150000,
                                        delta_frac=0.04)
        best_a = maybe_symmetrize(best_a, fft_len, n)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (the official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the best sequence globally for possible warm‑starts in
    # subsequent runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
