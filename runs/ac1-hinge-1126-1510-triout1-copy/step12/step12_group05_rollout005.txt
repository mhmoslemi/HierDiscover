def adam_optimize(a_start, schedule, deadline, n, fft_len, rng):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    best_a = np.exp(y - np.max(y))
    best_a = best_a / best_a.sum()
    best_score = fast_score(best_a, fft_len)
    for lam in schedule: # lam: smoothing factor for loss? Actually we will compute exact max conv maybe fine. maybe schedule reduces step size.
        # inner loop: do many steps while time left
        while time.time() < deadline:
            # compute gradient
            a = np.exp(y - np.max(y))
            a = a / a.sum()
            # compute conv and max
            fa = np.fft.fft(a, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - eps).astype(np.float64)

            # gradient conv
            a_rev = a[::-1]
            fm = np.fft.fft(mask, fft_len)
            fa_rev = np.fft.fft(a_rev, fft_len)
            grad_conv = np.real(np.fft.ifft(fa_rev * fm))
            g = 2.0 * grad_conv[n-1:n-1+n]

            # chain to y
            mean_g = np.dot(a, g)
            grad_y = a * (g - mean_g)

            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.05 / (np.sqrt(v_hat) + epsilon)   # adapt step size per coordinate

            y = y - lr * m_hat

            # evaluate score
            if (t & 31) == 0:
                a = np.exp(y - np.max(y))
                a = a / a.sum()
                score = fast_score(a, fft_len)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[Adam] iter={t} best_score={best_score:.9f}")

            # maybe break out to next lam after some iterations or time
            if time.time() >= deadline:
                break
        # after each schedule maybe reduce something? Not needed.

    return best_a, best_score

"""
step‑function optimiser – 1000 s budget (new version)

Main ideas
----------
1. **Multi‑scale warm‑starts** – run a cheap random‑search for several
   problem sizes (n≈4000, 6000, 8000).  Each size gets a short
   dedicated time slice; the best primitive shape (probability vector)
   is kept as a seed for the deeper optimisation.

2. **L‑BFGS cascade** – exactly as before, but with a slightly larger
   iteration budget per λ and an extended λ‑schedule that goes down
   to 1e‑7.

3. **Adam‑style mirror descent** – optimisation of the *true* L∞
   objective (hard max of the self‑convolution).  The algorithm works
   in the soft‑max (log‑probability) space, uses a momentum‑averaged
   sub‑gradient and an adaptive step‑size (Adam).  It runs for the
   majority of the remaining time and typically squeezes a few 1e‑3
   improvements.

4. **Final polishing** – a short low‑λ L‑BFGS run followed by a
   lightweight simulated‑annealing hill‑climb.

All stages respect the global 1000 s deadline; the best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation (more families)
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse symmetric probability vector using several families."""
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "bimodal",
            "sparse",
            "mixed",
            "arcsine",
            "double_exp",
            "gauss_mix",
        ]
    )
    x = np.arange(n)
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.001, 0.03)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 5.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.2) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        m = rng.integers(int(0.015 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == "mixed":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = 0.6 * np.full(n, 1.0 / n) + 0.4 * np.exp(-rate * d)

    elif mode == "arcsine":
        # a_i ∝ 1/√((i+0.5)*(n-0.5-i))
        a = 1.0 / np.sqrt((x + 0.5) * (n - 0.5 - x))

    elif mode == "double_exp":
        # mixture of two edge exponentials with different rates
        rate1 = rng.uniform(0.001, 0.02)
        rate2 = rng.uniform(0.02, 0.07)
        d = np.minimum(x, n - 1 - x)
        a = 0.5 * np.exp(-rate1 * d) + 0.5 * np.exp(-rate2 * d)

    else:  # gauss_mix
        # up to three symmetric Gaussian bumps
        nb = rng.integers(1, 4)
        a = np.zeros(n, dtype=float)
        for _ in range(nb):
            mu = rng.uniform(0.0, 0.5) * n
            sigma = rng.uniform(0.02, 0.15) * n
            bump = np.exp(-((x - mu) ** 2) / (2.0 * sigma ** 2))
            bump += np.exp(-((x - (n - 1 - mu)) ** 2) / (2.0 * sigma ** 2))
            a += bump

    # avoid exact zeros (log‑softmax would break), then normalise
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the true L∞ objective
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lr0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    check_interval: int = 256,
):
    """
    Mirror‑descent with Adam on the exact max‑convolution objective.
    Operates in the log‑probability (softmax) space.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    t = 0

    while time.time() < deadline:
        # ---- current probability vector ----
        a = np.exp(y - np.max(y))
        a /= a.sum()

        # ---- convolution and mask of maximal entries ----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # ---- sub‑gradient of (hard) max‑convolution ----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # size n

        # ---- chain rule to log‑prob space ----
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (np.sqrt(v_hat) + eps)
        y = y - step * m_hat

        # ---- periodic evaluation ----
        if (t & (check_interval - 1)) == 0:
            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] iter={t} best_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (kept from the previous version)
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.20  # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes we will explore (moderate lengths, still cheap enough)
    # ------------------------------------------------------------------
    candidate_ns = [4000, 6000, 8000]

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – multi‑scale warm‑starts (≈15 % of budget)
    # ------------------------------------------------------------------
    warm_end = start_time + 0.15 * budget_s
    best_global_a = None
    best_global_score = float("inf")
    best_n = None

    for n in candidate_ns:
        fft_len = next_pow2(2 * n - 1)
        phase_end = min(warm_end, start_time + 0.15 * budget_s / len(candidate_ns))
        best_a_n = None
        best_score_n = float("inf")

        # optionally reuse previous construction if it matches the current n
        if "height_sequence_1" in globals():
            old = np.array(globals()["height_sequence_1"], dtype=float)
            if old.ndim == 1 and old.size > 1:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
                a0 = np.maximum(a0, 1e-12)
                a0 /= a0.sum()
                best_a_n = a0
                best_score_n = fast_score(best_a_n, fft_len)
                print(f"[init‑n={n}] re‑interpolated fast_score={best_score_n:.9f}")

        while time.time() < phase_end:
            a = generate_candidate_shape(n, rng)

            # occasional jitter of a promising candidate
            if best_a_n is not None and rng.random() < 0.12:
                a = add_noise(a, sigma=rng.uniform(0.001, 0.006), rng=rng)

            score = fast_score(a, fft_len)
            if score < best_score_n:
                best_score_n = score
                best_a_n = a.copy()
                print(f"[warm‑n={n}] new best fast_score={best_score_n:.9f}")

        # keep the best across all n
        if best_score_n < best_global_score:
            best_global_score = best_score_n
            best_global_a = best_a_n.copy()
            best_n = n

    if best_global_a is None:
        # fallback – uniform distribution
        best_n = 4000
        best_global_a = np.full(best_n, 1.0 / best_n)
        best_global_score = fast_score(best_global_a, next_pow2(2 * best_n - 1))

    print(f"[selection] chosen n={best_n} with fast_score={best_global_score:.9f}")

    n = best_n
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS cascade (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    if time.time() < phase1_end:
        lam_schedule = [
            0.7,
            0.5,
            0.35,
            0.25,
            0.15,
            0.10,
            0.07,
            0.05,
            0.03,
            0.02,
            0.01,
            0.005,
            0.003,
            0.0015,
            0.0008,
            0.0004,
            0.0002,
            0.0001,
            5e-05,
            1e-05,
            5e-06,
            2e-06,
            1e-06,
            5e-07,
            2e-07,
            1e-07,
        ]
        best_global_a, best_global_score = lbfgs_cascade(
            best_global_a,
            lam_schedule,
            maxiter_per_lam=3000,
            deadline=phase1_end,
            fft_len=fft_len,
            n=n,
        )
        print(f"[L‑BFGS cascade] best fast_score={best_global_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style mirror descent on the exact L∞ objective
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    if time.time() < phase2_end:
        best_global_a, best_global_score = adam_optimize(
            best_global_a,
            phase2_end,
            n,
            fft_len,
            rng,
            lr0=0.02,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            check_interval=256,
        )
        print(f"[Adam] best fast_score={best_global_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – short low‑λ L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08, 1e-08]
        a_refined, sc_refined = lbfgs_cascade(
            best_global_a,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_global_score:
            best_global_score = sc_refined
            best_global_a = a_refined.copy()
            print(f"[final‑LBFGS] fast_score={best_global_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – final simulated‑annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_global_a, best_global_score = simulated_annealing_simple(
            best_global_a,
            best_global_score,
            deadline,
            n,
            rng,
        )
        print(f"[final‑anneal] fast_score={best_global_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_global_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # type: ignore  # provided by the environment
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
