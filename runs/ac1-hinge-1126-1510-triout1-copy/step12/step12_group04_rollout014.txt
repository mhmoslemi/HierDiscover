"""
Propose a low‑evaluation step‑function.

The program builds a diverse pool of starting vectors (including an
“arcsine” distribution that is provably optimal for the continuous
problem), runs a short cascade of L‑BFGS optimisations on a smooth
log‑sum‑exp surrogate and then refines the best candidate with a
projected sub‑gradient descent on the true L∞‑objective.
A final symmetrisation and a tiny smoothing step are applied before the
official scorer is called.

The whole search respects the 1000 s deadline; the best sequence found
is stored in the global ``height_sequence_1`` variable and returned
as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score (sum of the vector is 1):
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # zero vector – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    return simplex_projection(noisy)


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce left‑right symmetry (average with reversed copy)."""
    b = (a + a[::-1]) / 2.0
    return simplex_projection(b)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector using several heuristics."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "two_block",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # two_block – two separated uniform blocks
        block_frac = rng.uniform(0.05, 0.15)
        block_len = max(1, int(block_frac * n))
        a = np.zeros(n, dtype=float)
        a[:block_len] = 0.5 / block_len
        a[-block_len:] = 0.5 / block_len
    a /= a.sum()
    return a


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discrete version of the continuous arcsine law:
        f(x) = 1 / (π√(x(1‑x)))  on (0,1).
    The probability mass in bin i is f((i+½)/n) / n,
    then normalised to sum‑1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / (np.pi * np.sqrt(x * (1.0 - x))) / n
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate for L‑BFGS (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------


def surrogate_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. the
    log‑variables y (a = softmax(y)).
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution (FFT) ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)            # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=surrogate_and_grad,
            x0=y,
            args=(n, fft_len, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam,
                     "gtol": 1e-12,
                     "disp": False},
        )
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.2e} new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used for projected GD)
# ----------------------------------------------------------------------


def subgradient_maxconv(a: np.ndarray, fft_len: int):
    """
    Return a sub‑gradient of max(conv(a,a)) w.r.t. a.
    Also returns the current max value (useful for jitter decisions).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n
    return g, max_val, conv


def projected_gradient_optimize(a_start: np.ndarray,
                                deadline: float,
                                n: int,
                                fft_len: int,
                                rng: np.random.Generator,
                                eta0: float = 1.0,
                                max_iters: int = 10_000_000):
    """
    Projected sub‑gradient descent on the true L∞‑objective.
    Uses Euclidean projection onto the simplex after each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        g, max_val, _ = subgradient_maxconv(a, fft_len)

        # step size – simple sqrt decay (tunable via eta0)
        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * g
        a_new = simplex_projection(a_new)

        a = a_new

        if (it & 31) == 0:                     # score check every 32 iters
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = it
                print(f"[PGD] it={it} fast_score={best_score:.9f}")

        # occasional jitter if stuck
        if (it - last_improve) > 300_000:
            a = add_noise(a, sigma=rng.uniform(1e-4, 5e-4), rng=rng)
            last_improve = it

        it += 1

    return best_a, best_score

            a0 = np.interp(x_new, x_old, old)

                                sigma=rng.uniform(0.001, 0.01),
                                rng=rng)

            a_start,

        )

            best_score = sc_opt

            print(f"[Phase1] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # Enforce symmetry – often lowers the max‑convolution
    sym_a = symmetrize(best_a)
    sym_score = fast_score(sym_a, fft_len)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        print(f"[Phase1] symmetry improved fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = projected_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=2.0,                  # a moderate initial step‑size
            max_iters=5_000_000,      # loop stops by deadline
        )
        print(f"[Phase2] projected GD best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final L‑BFGS polish (very low λ) + tiny smoothing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=1200,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase3‑LBFGS] fast_score={best_score:.9f}")

    if time.time() < deadline:
        # very light moving‑average smoothing (helps a little)
        kernel = np.ones(3) / 3.0
        smoothed = np.convolve(best_a, kernel, mode="same")
        smoothed = simplex_projection(smoothed)
        sm_score = fast_score(smoothed, fft_len)
        if sm_score < best_score:
            best_a = smoothed
            best_score = sm_score
            print(f"[Phase3‑smooth] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[Final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
