"""
step‑function optimiser – 1000 s budget (new version)

Goal
----
Improve the bound that was previously 1.50634.  The optimiser
* uses a richer collection of parametric warm‑starts (including a
  greedy “collision‑free” sparse starter that mimics Sidon‑sets);
* runs a *deeper* L‑BFGS cascade with a finer λ‑schedule and up to
  4000 iterations per λ;
* follows with a long exponentiated‑gradient (mirror‑SGD) phase that
  uses Adam‑style moments for a smoother descent;
* finishes with an intensified simulated‑annealing hill‑climb.

All phases respect the global 1000 s deadline and the global variable
``height_sequence_1`` (if present) is used as a warm‑start.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score (the sum of the vector is assumed to be 1):
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric warm‑start generation
# ----------------------------------------------------------------------


def generate_candidate_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a symmetric (or not) probability vector employing a variety
    of families.  One special family attempts to build a Sidon‑like set
    with (almost) unique pairwise sums.
    """
    mode = rng.choice(
        [
            "uniform",
            "exp_edge",
            "power_edge",
            "bimodal",
            "sparse",
            "mixed",
            "sidon_random",
            "tri_piecewise",
        ]
    )
    x = np.arange(n)

    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp_edge":
        rate = rng.uniform(0.0005, 0.02)
        d = np.minimum(x, n - 1 - x)
        a = np.exp(-rate * d)

    elif mode == "power_edge":
        p = rng.uniform(0.2, 4.0)
        d = np.minimum(x, n - 1 - x) + 1.0
        a = d ** (-p)

    elif mode == "bimodal":
        offset = rng.uniform(0.0, 0.25) * n
        mu1 = offset
        mu2 = n - 1 - offset
        sigma = rng.uniform(0.05, 0.18) * n
        a = np.exp(-((x - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((x - mu2) ** 2) / (2.0 * sigma ** 2))

    elif mode == "sparse":
        m = rng.integers(int(0.01 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals

    elif mode == "mixed":
        rate = rng.uniform(0.001, 0.03)
        d = np.minimum(x, n - 1 - x)
        a = 0.5 * np.full(n, 1.0 / n) + 0.5 * np.exp(-rate * d)

    elif mode == "sidon_random":
        # Greedy construction of a set with (almost) unique pairwise sums.
        # The set size is limited to about sqrt(2n) ≈ 90 for n≈4000.
        max_trials = 2000
        sums = set()
        indices = []
        # try to add points one by one, rejecting colliding sums
        while len(indices) < 90 and max_trials > 0:
            cand = rng.integers(0, n)
            # check collisions with existing indices
            collision = False
            for q in indices:
                if (cand + q) in sums:
                    collision = True
                    break
                if (2 * cand) in sums:
                    collision = True
                    break
            if not collision:
                # accept
                for q in indices:
                    sums.add(cand + q)
                sums.add(2 * cand)          # self‑sum
                indices.append(cand)
            max_trials -= 1
        if len(indices) < 5:               # fallback to a cheap dense shape
            a = np.full(n, 1.0 / n)
        else:
            w = np.full(len(indices), 1.0 / len(indices))
            a = np.zeros(n, dtype=float)
            a[indices] = w

    else:  # tri_piecewise – three‑segment linear shape
        left_frac = rng.uniform(0.05, 0.25)
        right_frac = rng.uniform(0.05, 0.25)
        left_len = int(left_frac * n)
        right_len = int(right_frac * n)
        mid_len = n - left_len - right_len
        left_val = rng.uniform(0.5, 2.0)      # relative magnitude
        mid_val = rng.uniform(0.2, 1.0)
        right_val = rng.uniform(0.5, 2.0)
        a = np.concatenate(
            [
                np.linspace(left_val, mid_val, left_len, endpoint=False),
                np.full(mid_len, mid_val),
                np.linspace(mid_val, right_val, right_len, endpoint=True),
            ]
        )
    # avoid exact zeros, normalise to a probability vector
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y,
    where a = softmax(y) is the underlying probability vector.
    """
    # ---- softmax -------------------------------------------------------
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ---- convolution (FFT) --------------------------------------------
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. convolution entries ---------------------------
    w = exp_shift / Z                      # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]    # size n

    # ---- back‑prop through softmax ------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS cascade
# ----------------------------------------------------------------------


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run L‑BFGS for a decreasing sequence of smoothing parameters λ.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        # reconstruct a
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (Adam‑style) on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize_adam(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 30.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 5_000_000,
):
    """
    Mirror‑SGD with Adam moments on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)

        # sub‑gradient (ordered‑pairs) of the hard max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # size n

        # Adam moments
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step‑size schedule (inverse‑sqrt)
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update (mirrored SGD)
        raw = -eta * m_hat / (np.sqrt(v_hat) + eps)
        exp_term = np.exp(np.clip(raw, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # cheap periodic check (every 32 iterations)
        if (t & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad‑adam] iter={t} fast_score={best_score:.9f}")

    return best_a, best_score

# Simulated annealing – more aggressive hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_aggressive(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a temperature that decays
    exponentially.  The move size is allowed to be up to 40 % of the
    source entry to explore more aggressively.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.08, 1e-12)   # warmer start than before
    max_frac = 0.40                         # bigger moves

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * np.exp(-5.0 * frac), 1e-12)   # exponential cooling

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

    globals()["height_sequence_1"] = final_seq
    return final_seq
