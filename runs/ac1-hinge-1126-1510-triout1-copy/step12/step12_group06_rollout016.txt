import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (any scaling)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] == a[-i-1] by averaging with its reverse."""
    return (a + a[::-1]) / 2.0

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector (Dirichlet) and symmetrise."""
    a = rng.gamma(shape=1.0, scale=1.0, size=n)
    a /= a.sum()
    return symmetrize(a)


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density on the lattice."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return symmetrize(a)


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power, clipped at zero."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return symmetrize(a)


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass concentrated at the two edges (symmetric)."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = left_vals  # symmetric
    a /= a.sum()
    return symmetrize(a)


def cheb_kaiser_initial(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Classic Chebyshev and Kaiser windows (already symmetric)."""
    wins = []
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            wins.append(symmetrize(w))
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            wins.append(symmetrize(w))
    except Exception:
        pass
    return wins

# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y ∈ ℝⁿ are unrestricted; a = softmax(y) ∈ simplex.
    """
    # softmax to stay on the simplex
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                   # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)        # safe: arguments ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        # enforce symmetry after each LBFGS step
        a = symmetrize(a)
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0,
                          temperature: float = 1e-8):
    """
    Mirrored (multiplicative) sub‑gradient descent on the true L∞‑objective.
    Uses a soft‑mask generated from the top τ fraction of convolution values.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # soft mask: compute weights proportional to exp((conv - max)/temp)
        # to keep numerical stability we shift by max
        shifted = conv - max_val          # ≤ 0
        # temperature controls how many entries participate
        exp_shift = np.exp(shifted / temperature)
        mask = exp_shift / exp_shift.sum()  # Not a true mask but a probability distribution

        # Back‑propagate mask through convolution:
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # enforce symmetry
        a = symmetrize(a)
        a /= a.sum()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Hill‑climbing with symmetric mass transfers.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20                # up to 20 % of source mass may be moved

    # pre‑compute index pairs respecting symmetry
    half = n // 2
    indices = np.arange(n)

    while time.time() < deadline:
        i = rng.integers(0, n)
        # choose a symmetric partner for i to keep symmetry automatically
        j = rng.integers(0, n)
        if i == j:
            continue

        # decide source/destination based on mass
        if a[i] >= a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # enforce symmetry after the change
        cand = (cand + cand[::-1]) / 2.0
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
                # optional: print progress occasionally
                # print(f"[local‑refine] new best = {best_score:.9f}")

    return best_a, best_score

"""
Optimiser for low L∞ convolution score.

Improvements over the previous version:
- Enforces symmetry of the step function after every update (the problem is
  symmetric, this reduces the effective dimensionality and leads to a smoother
  landscape).
- Uses a larger problem size (n = 8000) which allows a smaller constant
  2·n·max(conv) / (sum a)².
- Refines the smoothed‑max L‑BFGS phase (more λ values, larger iteration
  budget and adaptive early stopping).
- Employs a “soft‑mask’’ exponentiated‑gradient descent that takes several of
  the top convolution entries into account (temperature‑scaled softmax).
- Adds a second short EG‑polish after the greedy hill‑climb.
- Keeps a diversified pool of symmetric starting vectors and repeatedly
  symmetrises candidates.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (any scaling)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] == a[-i-1] by averaging with its reverse."""
    return (a + a[::-1]) / 2.0


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑distributed random vector, then symmetrised."""
    a = rng.gamma(shape=1.0, scale=1.0, size=n)
    a /= a.sum()
    return symmetrize(a)


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density on the lattice."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return symmetrize(a)


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power, clipped at zero."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return symmetrize(a)


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass concentrated at the two edges (symmetric)."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = left_vals  # symmetric copy
    a /= a.sum()
    return symmetrize(a)


def cheb_kaiser_initial(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Classic Chebyshev and Kaiser windows (already symmetric)."""
    wins = []
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            wins.append(symmetrize(w))
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            wins.append(symmetrize(w))
    except Exception:
        pass
    return wins


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y ∈ ℝⁿ are unrestricted; a = softmax(y) ∈ simplex.
    """
    # softmax → simplex
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Cascade of L‑BFGS over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        a = symmetrize(a)
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0,
                          temperature: float = 1e-8):
    """
    Mirrored (multiplicative) sub‑gradient descent on the true L∞‑objective.
    A temperature‑scaled soft‑max over the convolution entries produces a
    smooth estimate of the sub‑gradient of the max‑operator.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        # temperature is relative to the current magnitude of conv
        temp = max(temperature * max_val, 1e-12)

        shifted = conv - max_val                     # ≤ 0
        exp_shift = np.exp(shifted / temp)           # soft‑max numerators
        mask = exp_shift / exp_shift.sum()           # weights sum to 1

        # back‑propagation through convolution a*a → a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        a = symmetrize(a)
        a /= a.sum()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def local_refinement(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     rng: np.random.Generator):
    """
    Simple hill‑climbing where a small random fraction of mass is transferred
    between two positions.  After each move the vector is symmetrised.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    max_frac = 0.20  # up to 20 % of the source mass may be moved

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        if a[i] >= a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = (cand + cand[::-1]) / 2.0          # enforce symmetry
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand /= cand_sum

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a potentially smaller constant.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of symmetric starting vectors.
    # ------------------------------------------------------------------
    pool = []

    # 0) Re‑use the previous best if available (interpolated & symmetrised).
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            a0 = symmetrize(a0)
            pool.append(a0)

    # 1) Classic Chebyshev & Kaiser windows (already symmetric).
    pool.extend(cheb_kaiser_initial(n, rng))

    # 2) Random / parametric families (all symmetrised).
    for _ in range(6):
        pool.append(random_initial(n, rng))
    for _ in range(4):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing budget splits.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.20 * budget_s    # aggressive L‑BFGS
    phase2_end = start_time + 0.80 * budget_s    # long EG descent

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on a smoothed surrogate.
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.24, 0.16, 0.11,
        0.07, 0.045, 0.028, 0.018, 0.011,
        0.007, 0.004, 0.0025, 0.0015, 0.0009,
        0.0005, 0.0003, 0.00015, 8e-5, 4e-5,
        2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 2e-7, 1e-7, 5e-8
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float("inf")

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.3:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.006),
                                rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (exact max).
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=18.0,
            temperature=1e-8
        )
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with ultra‑low λ.
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        fine_lam_sched = [2e-7, 1e-7, 5e-8, 2e-8, 1e-8]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=2000,
                                        deadline=deadline - 30.0,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – greedy local refinement (symmetric mass moves).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = local_refinement(best_a,
                                               best_score,
                                               deadline,
                                               n,
                                               fft_len,
                                               rng)
        print(f"[local‑refine] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – second short EG polish (smaller step size).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            eta0=5.0,
            temperature=1e-8
        )
        print(f"[final‑exp‑grad] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided scorer.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is pre‑imported
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
