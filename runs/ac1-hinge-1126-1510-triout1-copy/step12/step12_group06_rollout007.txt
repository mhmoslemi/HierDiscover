def subgradient_optimize(a_start, deadline):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    while time.time() < deadline:
        max_val, g = max_conv_gradient(a)
        # compute step size: eta = eta0 / sqrt(it+1)
        eta = 10.0 / np.sqrt(it+1.0)
        # perform gradient step: a = a - eta * g
        # However must keep a >=0 and sum to 1.
        # Compute raw update:
        a_new = a - eta * g
        # Clip to zero
        a_new = np.maximum(a_new, 0.0)
        # If sum zero, revert
        s = a_new.sum()
        if s <= 0.0:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

"""
Step‑function optimiser – 1000 s budget (enhanced)

The optimiser follows the same four‑stage outline as the reference code,
but adds a few extra tricks that help shave the last few
0.001 … 0.002 off the score:

* a much longer L‑BFGS cascade with a finer λ‑schedule and higher
  iteration limits;
* a *exact* sub‑gradient descent on the true ‑∞‑objective (the mask
  now contains only the single maximising index, not a band);
* a targeted “gradient‑guided” hill‑climb that repeatedly moves a
  small fraction of mass from the indices with the largest
  sub‑gradient to those with the smallest one – this directly attacks
  the entries that dominate the maximal convolution coefficient;
* occasional random perturbations of the current best solution to
  escape shallow basins.

All phases respect the global deadline; the best candidate seen so far
is stored in the global variable `height_sequence_1` and returned as a
plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a (any scaling)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Initial families (same as reference)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector from several styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian density on the lattice."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power, clipped at zero."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass concentrated at the two edges."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for aggressive L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y ∈ ℝⁿ unrestricted; a = softmax(y) lives on the simplex.
    """
    # softmax to enforce simplex
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # effective λ (scaled with current max to keep smoothing relative)
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                   # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)        # safe: arguments ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.2e}] new best fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient (mask = single max index) – projected gradient descent
# ----------------------------------------------------------------------


def max_conv_and_grad(a: np.ndarray, fft_len: int):
    """
    Compute max(conv) and its exact sub‑gradient (size n).
    Returns (max_val, grad) where grad_i = 2 * a_{t_max - i} when the index is valid,
    otherwise 0.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    t_max = int(conv.argmax())
    max_val = conv[t_max]

    # gradient: for each i we need a_{t_max - i} if that index lies in [0,n)
    grad = np.zeros_like(a)
    i_low = max(0, t_max - (n - 1))
    i_high = min(n - 1, t_max)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    jdx = t_max - idx
    grad[idx] = 2.0 * a[jdx]          # factor 2 accounts for both (i,j) and (j,i)
    return max_val, grad


def exact_subgradient_optimize(a_start: np.ndarray,
                               deadline: float,
                               fft_len: int,
                               eta0: float = 5.0):
    """
    Projected sub‑gradient descent on the true ‑∞‑objective using the exact
    (single‑index) mask.  The update is a := a - η·g followed by simplex projection.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        max_val, g = max_conv_and_grad(a, fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0.0:
            a_new = a.copy()
        else:
            a_new /= s

        a = a_new
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def gradient_guided_refinement(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               fft_len: int,
                               rng: np.random.Generator):
    """
    Hill‑climbing moves: each iteration we look at the exact gradient of the
    current maximal convolution coefficient and move a small fraction of mass
    from the index with the largest gradient to the one with the smallest.
    Only moves that strictly improve the fast_score are accepted.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    n = a.shape[0]

    while time.time() < deadline:
        # exact gradient of the current maximal entry
        _, grad = max_conv_and_grad(a, fft_len)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            # pick a random different destination
            dst = (src + rng.integers(1, n)) % n

        # transfer up to 10 % of the source mass
        delta = a[src] * rng.uniform(0.01, 0.10)
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # re‑normalise (total mass stays 1, but rounding errors may accumulate)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score - 1e-12:   # require a *strict* improvement
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
    return best_a, best_score
