def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine problem size: if global height_sequence_1 present with large size, use that; else default_n = 12000.
    default_n = 12000

    n = default_n

    fft_len = next_pow2(2*n-1)
    fft_len_g = next_pow2(3*n-2)

    # initial best: use arcsine or uniform
    best_a = np.full(n, 1.0/n)  # uniform
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # define lam schedule
    lam_schedule = list(np.geomspace(0.7, 1e-8, num=45))

    # start multi-restart L-BFGS for half time
    phase1_deadline = start_time + 0.55 * budget_s
    modes_list = ["random","exp","power","edge","sidon","binary","bump","harmonic","arcsine","best"]
    iteration=0

    while time.time() < phase1_deadline:
        mode = rng.choice(modes_list)
        if mode=="best":
            y0 = best_y + rng.normal(scale=0.02,size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)
        # L-BFGS
        y_opt = schedule_optimize(y0, n, fft_len, fft_len_g, lam_schedule, maxiter_per_stage=400)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score, best_a, best_y = score_opt, a_opt.copy(), y_opt.copy()
            print(f"[{iteration}] mode={mode} new best fast score {best_score:.6f}")
        # hill climb quick
        hc_dl = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(best_a, best_score, fft_len, n, hc_dl, max_iter=2000)
        best_y = np.log(best_a + 1e-12)
        iteration+=1

    # Subgradient phase with multiplicative updates
    subgrad_deadline = min(deadline - 20, start_time + 0.90*budget_s)
    print("[subgrad] start multiplicative subgradient")
    best_a, best_score = multiplicative_subgradient(best_a, fft_len, n, subgrad_deadline)
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] end best fast score {best_score:.6f}")

    # final fine L-BFGS with low lambda
    if time.time() < deadline - 30:
        fine_sched = list(np.geomspace(1e-4,1e-12,num=30))
        y_fine = schedule_optimize(best_y, n, fft_len, fft_len_g, fine_sched, maxiter_per_stage=800)
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len, n)
        if score_fine < best_score:
            best_score, best_a, best_y = score_fine, a_fine, y_fine
            print("[fine] improved best fast score", best_score)

    # final hill climb
    if time.time() < deadline:
        final_hc_dl = deadline
        best_a, best_score = hill_climb(best_a,best_score,fft_len,n,final_hc_dl,max_iter=50000,max_move=0.25)
        best_y=np.log(best_a+1e-12)

    final_seq = best_a.tolist()
    val= evaluate_sequence(final_seq)
    print("[final] evaluated score", val)
    return final_seq

"""
Hybrid optimisation for the step‑function minimisation problem.

The algorithm works on the probability simplex (∑a = 1) because the
objective 2·n·max(conv)/(∑a)² is scale‑invariant.  The search consists of
four phases:

1.  Multi‑restart continuation L‑BFGS on a smooth log‑sum‑exp
    approximation of the max‑convolution.  The initial points are
    generated by a variety of patterns, importantly a new *arcsine*
    distribution (a_i ∝ 1/√(x·(1‑x))) which is known to flatten the
    self‑convolution.

2.  A short greedy hill‑climb that moves a random amount of mass from
    one index to another while accepting only improving steps.

3.  Projected sub‑gradient descent on the true (non‑smoothed) objective.
    The sub‑gradient of the maximal convolution entry is cheap to
    compute; the iterates are projected onto the simplex by the Euclidean
    projection.

4.  A final fine‑grained L‑BFGS pass with very low smoothing, followed
    by a longer aggressive hill‑climb using any remaining time.

All heavy work (FFT‑based convolutions) is performed with numpy’s fast
FFT, making the search feasible for n≈12 000 within the 1000 s budget.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a (linear)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂c/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
                "ftol": 1e-12,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes include the classic patterns plus a new ``arcsine`` mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Simple greedy Sidon‑set construction
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums or (2 * candidate) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        # Discrete version of the arcsine law: a_i ∝ 1/√(x·(1‑x))
        idx = np.arange(n, dtype=float) + 0.5          # centre of each bin
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

# Local search helpers

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (i.e. minimise max(conv) under Σa = 1, a≥0).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))  # index of maximal entry

        # Sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score
