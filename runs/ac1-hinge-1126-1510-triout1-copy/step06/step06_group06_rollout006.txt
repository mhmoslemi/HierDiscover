"""
Propose a step‑function sequence with a very low evaluation score.

Algorithm (≈100 s total, then fine‑tune until the time budget expires)

1. **Start** – Load the previous best sequence (if any); otherwise use a uniform
   distribution of length ``n = 5000``.  Normalise to the probability simplex.

2. **Global L‑BFGS search** – Several random restarts in the log‑softmax
   parametrisation.  A geometric λ‑schedule drives a smooth‑max surrogate of
   the true objective.  Each restart ends with a L‑BFGS optimisation over a
   few hundred iterations per λ‑level.

3. **Exponentiated‑gradient descent** – Using the exact sub‑gradient of the
   non‑smoothed max‑convolution (the shift that attains the current maximum)
   we perform a multiplicative update
   ``a ← a·exp(−η·grad)``   (projected back to the simplex).  The step size η
   follows a 1/√t schedule and is adaptively increased/decreased.

4. **Simulated‑annealing hill‑climb** – Random mass‑transfer moves are accepted
   if they improve the fast surrogate, otherwise with probability
   ``exp(−Δ/Temp)``.  Temperature decays exponentially from ``1e‑3`` to
   ``1e‑7`` during the remaining time.

5. **Final polishing** – A plain random hill‑climb (strictly improving moves)
   finishes the search.

All steps respect the non‑negative, ≤ 1000 bound and the sum‑≥ 0.01 guard
required by ``evaluate_sequence``.  The best sequence discovered is stored in
the global variable ``height_sequence_1`` for reuse in later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (pure, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1] if np.any(cond) else 0
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence``:
    returns 2·n·max(conv(a,a)) where convolution is computed by FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max (log‑sum‑exp)
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # propagate back to a
    # note: convolution of w with a (reversed) yields ∂max/∂a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """Continuation of L‑BFGS optimisations over a decreasing λ‑schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """Diverse starting points for the log‑soft‑max parameters."""
    mode = rng.choice(["random", "exp", "power", "edge", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    if mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    if mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    if mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        centre = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, centre)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    # best – perturb current best solution
    return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """Exact maximal auto‑convolution value and a sub‑gradient."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map the index so that 0 ≤ k ≤ n‑1 (the relevant half)
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        # only the diagonal entries participate
        grad = 2.0 * a
    else:
        # forward contribution
        grad[: n - k] += a[k:]
        # backward contribution
        grad[k:] += a[: n - k]
    return max_c, grad


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb that only keeps strictly improving moves."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    T0: float = 1e-3,
    T_end: float = 1e-7,
    move_frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the simplex.
    Randomly transfers a fraction of mass from i→j and accepts the move
    with Metropolis probability based on a temperature schedule.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    score_cur = cur_score
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline:
        # exponential cooling
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        T = T0 * (T_end / T0) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)

        if new_score < score_cur - 1e-12:
            cur, score_cur = cand, new_score
            if new_score < best_score:
                best, best_score = cand, new_score
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - score_cur) / max(T, 1e-12))
            if rng.random() < prob:
                cur, score_cur = cand, new_score
    return best, best_score
