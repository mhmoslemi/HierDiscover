"""
Low‑evaluation step sequence search.

The algorithm proceeds in three stages, respecting the overall time budget:

1. **Random‑restart Adam on a smoothed max‑convolution** –
   each restart starts from a random or slightly‑perturbed distribution,
   and a geometric schedule for the smoothing factor (lam) drives the
   objective from a strongly‑smoothed version to the true max.
   Adam (adaptive moment estimation) provides fast, stable descent.

2. **Intensive greedy hill‑climb** –
   after the Adam phase the best sequence is refined by deterministic
   mass‑transfer moves that strictly improve the exact objective
   (2 n · max (a⋆a)).  The hill‑climb runs until the deadline.

3. **Final evaluation** –
   the best distribution (non‑negative and summing to 1) is returned as a
   plain Python list and its exact score is printed.

All heavy operations (FFT‑based convolutions) are performed with NumPy,
and the search respects the 1000 s limit by checking the wall‑clock time
at each iteration.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised array.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray, float]:
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient w.r.t. y.
    Returns (value, grad_y, max_conv) where max_conv is the true max
    before smoothing (used for fast_score without extra FFT).
    """
    a = softmax(y)

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv_full.max()

    # --- smoothed max via log‑sum‑exp ---
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient w.r.t. convolution entries ---
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑prop through convolution (a↔w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y, max_c


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy hill‑climb.
    Transfers a random fraction of mass from one index to another,
    accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score


def adam_optimize(
    a_start: np.ndarray,
    lam_start: float,
    lam_end: float,
    total_iters: int,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution.
    The smoothing factor lam_factor follows a geometric schedule from
    lam_start down to lam_end over total_iters iterations.
    """
    # initialise on the log‑softmax (y) parametrisation
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for t in range(1, total_iters + 1):
        if time.time() >= deadline:
            break

        # geometric schedule for the smoothing factor
        lam_factor = lam_start * (lam_end / lam_start) ** ((t - 1) / (total_iters - 1))

        # compute objective, gradient and the true max‑convolution value
        _, grad_y, max_c = objective_and_grad(
            y, n, fft_len_conv, fft_len_g, lam_factor
        )

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate scaled with the smoothing factor
        lr = 0.12 * (lam_factor / lam_start + 1e-12)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep y numerically stable
        y = np.clip(y, -30.0, 30.0)

        # evaluate exact score (no extra FFT needed – we already have max_c)
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = softmax(y)

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point used by the evaluation harness.
    Returns a list of non‑negative heights with a low evaluation score.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initial sequence
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # initialise from a previously found sequence if available
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # Linear interpolation to required length
            old_n = init_seq.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2    # safety margin

    # allocate phases
    phase_adam_end = start_time + 0.70 * budget_s
    phase_hill_end = deadline

    # --------------------------------------------------------------
    # Phase 1 – many short Adam restarts with geometric lam schedule
    # --------------------------------------------------------------
    lam_start = 0.7
    lam_end = 0.0005
    iter_per_restart = 1200  # moderate length – cheap enough for many restarts

    while time.time() < phase_adam_end:
        # pick start point: either a noisy version of the current best
        # or a fresh random distribution
        if rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        # run Adam on this start
        a_opt, score_opt = adam_optimize(
            a_start,
            lam_start,
            lam_end,
            iter_per_restart,
            fft_len_conv,
            fft_len_g,
            n,
            phase_adam_end,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Adam] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb (deterministic local improvements)
    # --------------------------------------------------------------
    cur_score = fast_score(best_a, fft_len_conv, n)
    best_a, best_score = hill_climb(
        best_a,
        cur_score,
        fft_len_conv,
        n,
        phase_hill_end,
        max_iter=20000,
    )
    print(f"[final hill‑climb] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
