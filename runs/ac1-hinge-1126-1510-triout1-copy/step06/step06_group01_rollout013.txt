def directed_hill_climb(a, deadline):
    n = len(a)
    step_frac = 0.1  # fraction of mass to move initially
    it = 0
    while time.time() < deadline:
        # compute conv, max
        conv = full_conv(a, fft_len)
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(float)
        # compute subgradient as before
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1:n-1+n]

        # identify indices
        i_max = np.argmax(g)   # high gradient -> should reduce
        i_min = np.argmin(g)   # low gradient -> add
        # amount: small fraction of a[i_max] (or overall)
        delta = a[i_max] * step_frac
        if delta == 0:
            break
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # Ensure nonnegativity
        if a_new[i_max] < 0:
            # adjust delta
            delta = a[i_max]
            a_new[i_max] = 0.0
            a_new[i_min] += delta

        # compute new score
        new_score = fast_score(a_new, fft_len, n)
        # accept if improvement or maybe accept with prob
        if new_score < fast_score(a, fft_len, n):
            a = a_new
        # else maybe reduce step_frac gradually
        step_frac *= 0.9999
        it += 1

"""
A multi‑stage optimizer for the step‑function problem.
The search proceeds as follows:

1. **Random‑restart L‑BFGS on a log‑sum‑exp smooth approximation**
   of the max‑convolution, with a continuation of the smoothing factor.
2. **Projected sub‑gradient descent with momentum** (the original implementation).
3. **Exponentiated‑gradient descent** (original implementation).
4. **Fine‑grained L‑BFGS** to polish the current best solution.
5. **Directed descent** – a fast, gradient‑guided transfer of mass between
   the highest‑gradient (bad) coordinate and the lowest‑gradient (good)
   coordinate, with a simple annealing schedule.
   This phase consumes the remaining time budget and is the main novelty;
   it usually squeezes a few extra digits out of the objective.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is normalised (∑a=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # Rare failure – just skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (same as in the original code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the sub‑gradient of
    the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # mirror‑descent on the simplex
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional random jitter to escape plateaux
        if (it + 1) % 5000 == 0:
            if rng.random() < 0.3:
                sigma = rng.uniform(0.001, 0.008)
                a = add_noise(a, sigma, rng)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def directed_descent(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Fast gradient‑guided transfer of mass from the highest‑gradient index
    to the lowest‑gradient index.  A tiny simulated‑annealing component
    (temperature slowly decays) helps escape shallow plateaux.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    cur_score = best_score

    rng = np.random.default_rng()
    step_frac = 0.08          # initial fraction of mass to move
    temperature = 0.06        # initial annealing temperature
    it = 0

    while time.time() < deadline:
        # convolution and mask of the (approximately) active indices
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # indices to move mass between
        i_high = np.argmax(g)   # decreasing a here tends to reduce the max
        i_low = np.argmin(g)    # increasing a here is favourable

        # compute a safe delta
        delta = a[i_high] * step_frac
        if delta <= 0.0:
            # no mass to move - apply a small random jitter and continue
            sigma = rng.uniform(0.001, 0.004)
            a = add_noise(a, sigma, rng)
            continue

        # propose new distribution
        cand = a.copy()
        cand[i_high] -= delta
        cand[i_low] += delta
        if cand[i_high] < 0.0:  # safety correction
            delta = a[i_high]
            cand[i_high] = 0.0
            cand[i_low] = a[i_low] + delta

        cand_score = fast_score(cand, fft_len_conv, n)

        # accept if better, otherwise maybe accept probabilistically
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                a = cand
                cur_score = cand_score

        # decay step size and temperature
        step_frac = max(step_frac * 0.9997, 0.001)   # never go below 0.1 %
        temperature = max(temperature * 0.9995, 1e-6)

        it += 1

        # occasional wider jitter to break out of stagnation
        if it % 4000 == 0:
            if rng.random() < 0.25:
                sigma = rng.uniform(0.002, 0.006)
                a = add_noise(a, sigma, rng)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initial sequence
    # --------------------------------------------------------------
    max_n = 4000
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = max_n
            old_n = init_seq.size
            # linear interpolation to the target length
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # allocate fractions of the total budget
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient + momentum
    phase3_end = start_time + 0.80 * budget_s   # exponentiated gradient
    phase4_end = start_time + 0.90 * budget_s   # short L‑BFGS refinement
    # remainder (≈10 %) for directed descent

    # --------------------------------------------------------------
    # Phase 1 – random‑restart continuation L‑BFGS
    # --------------------------------------------------------------
    lam_schedule = [
        0.7,
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
    ]
    maxiter_per_lam = 1000

    while time.time() < phase1_end:
        # occasional perturbation of the current best to diversify the search
        if best_score < np.inf and rng.random() < 0.12:
            sigma = rng.uniform(0.001, 0.018)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – short L‑BFGS refinement
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        fine_lam_schedule = [0.001, 0.0005, 0.0002, 0.0001]
        fine_maxiter = 500
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            phase4_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS‑fine] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – directed descent (the main refinement)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = directed_descent(
            best_a, deadline, fft_len_conv, fft_len_g, n
        )
        print(f"[directed descent] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final evaluation using the provided external function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
