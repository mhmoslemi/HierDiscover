def propose_candidate(seed=42, budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 2.0
    rng = np.random.default_rng(seed)

    # determine n
    seq0 = globals().get("height_sequence_1")
    if isinstance(seq0, (list, tuple, np.ndarray)):
        arr0 = np.asarray(seq0, dtype=float).ravel()
        if arr0.ndim == 1 and arr0.size >= 3000:
            n = min(arr0.size, 12000)
            init_a = arr0[:n] / np.sum(arr0[:n])  # maybe use first n
        else:
            n = 5000
            init_a = np.full(n, 1.0/n)
    else:
        n = 5000
        init_a = np.full(n, 1.0/n)
    best_a = init_a.copy()
    best_y = np.log(best_a + 1e-12)
    # lengths for FFT
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # compute initial score
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Option: cheap grid search across Beta-like families
    alphas = np.linspace(-0.8, 0.5, 7)
    betas = np.linspace(-0.8, 0.5, 7)
    i_arr = np.arange(n, dtype=float)
    for alpha in alphas:
        for beta in betas:
            prob = (i_arr + 1) ** alpha * (n - i_arr) ** beta
            prob_sum = prob.sum()
            if prob_sum == 0:
                continue
            prob = prob / prob_sum
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] alpha={alpha:.2f} beta={beta:.2f} score={sc:.6f}")

    # Additional heuristic patterns
    # Sidon
    # generate a set of indices with low pairwise sums (similar to earlier)
    # produce some random patterns
    for scale in [0.02, 0.04, 0.08]:
        K = max(2, int(n * scale))
        a = np.zeros(n)
        chosen = rng.choice(n, size=K, replace=False)
        a[chosen] = 1.0 / K
        a = a / a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[rand_sparse] K={K} score={sc:.6f}")

    lam_schedule = list(np.geomspace(0.5, 1e-9, num=60))
    maxiter_per_stage = 1000
    iteration = 0
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon"]

    while time.time() < deadline - 300.0:  # keep ~5 mins for later phases

        sc_opt = fast_score(a_opt, fft_len_conv, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()

                                                eta0=0.15,

        best_y = np.log(best_a + 1e-12)

    # final hill climb aggressive
    if time.time() < deadline:
        best_a, best_score = hill_climb_random(best_a, best_score,
                                               fft_len_conv, n,
                                               deadline,
                                               max_iter=20000)

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)

"""
propose_candidate
=================

A three‑phase stochastic optimiser for the step‑function problem.

1. **Initialisation** –  a handful of parametric families (Beta‑like,
   exponential, power, edge, arcsine, Sidon‑type sparse) are examined
   on the simplex to obtain a decent starting point.

2. **Continuation L‑BFGS** –  minimise a log‑sum‑exp smooth‐max
   approximation of the true objective.  The smoothing factor λ is
   reduced geometrically (≈40 stages).  Each stage runs L‑BFGS for a
   few hundred iterations.  Many random restarts (including a noisy
   perturbation of the incumbent) are performed until a time budget
   reserve is left.

3. **Fine‑grained greedy/annealing phase** –  a directed hill‑climb
   that moves probability mass from indices contributing heavily to the
   current maximal convolution entry towards indices that contribute
   little, with a simulated‑annealing acceptance rule.  After the
   temperature reaches a tiny value, a few rounds of plain random hill‑climb
   and a short projected sub‑gradient descent polish the result.

The routine respects a 1000 s wall‑clock limit, prints progress
information, and returns the best sequence found (a list of non‑negative
floats).  The sequence is already normalised to a probability simplex
(the harness rescales it internally), so the sum of the entries is
irrelevant for the final score.

All helpers are defined at the top level – no closures or lambdas –
and only ``numpy`` and ``scipy`` are used.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective on the simplex:
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max (log‑sum‑exp)
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative wrt conv
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # propagate through convolution (g_vec = ∂conv/∂a)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: a series of minimisations with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a start vector y (pre‑soft‑max) for the given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # a sparse Sidon‑type pattern
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = any(((cand + c) in sums) or ((2 * cand) in sums) for c in chosen)
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = max(1, len(chosen))
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb_random(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb: move a random fraction of mass between
    two positions, accepting only strict improvements.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def directed_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy annealed hill‑climb using the gradient of the *active* convolution
    entry.  At each step we move mass from a coefficient that contributes
    strongly to the current maximum to one that contributes weakly.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # initial temperature – a small fraction of the current score
    temp = max(1e-4, best_score * 0.02)

    while time.time() < deadline:
        # current convolution and its maximiser
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # contribution weights w_i = a_{m-i} (zero if outside range)
        w = np.zeros(n)
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        w[mask] = best[counterpart[mask]]

        # source index (large w) – sample proportionally to w
        if w.sum() > 0:
            probs_src = w / w.sum()
            i = rng.choice(n, p=probs_src)
        else:
            i = rng.integers(0, n)

        # target index (small w) – sample inversely to w
        inv = w.max() - w + 1e-12
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        delta = best[i] * rng.random() * move_frac
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
            temp = max(temp * 0.995, 1e-6)   # cool a bit faster after improvement
        else:
            # accept uphill probabilistically
            prob = np.exp(-(new_score - best_score) / temp)
            if rng.random() < prob:
                best = cand
                best_score = new_score
                temp *= 0.9999   # very slow cooling when accepting uphill

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[sparse] k={k} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (major part of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=40))
    maxiter_per_stage = 800
    iteration = 0
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon"]
    # reserve ~300 s for the fine‑grained phase later
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # brief random hill‑climb to polish the L‑BFGS output
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb_random(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Directed annealed hill‑climb (mass transfer guided by the active convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print("[directed HC] start")
        best_a, best_score = directed_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline - 60.0,
            move_frac=0.18,
        )
        print(f"[directed HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Refine with projected sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.15,
            max_iters=250_000,
        )
        print(f"[subgrad] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive random hill‑climb (any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb_random(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # harness supplies this
    print(f"[final] evaluated score = {final_val:.6f} (fast_score = {best_score:.6f})")
    return final_seq
