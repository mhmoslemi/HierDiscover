"""
Propose a low‑score step function.

Algorithm
---------

* Work in the probability simplex (sum = 1) – the evaluation
  `2·n·max(conv)/(sum²)` is scale‑invariant so we may impose this
  constraint without loss of optimality.

* **Phase 1 – cheap parametric sweep**  
  Test a large collection of Beta‑like families
  a_i ∝ (i+1)^{α}·(n−i)^{β} with random (α,β) in a wide range.
  Keep the best for the next phase.

* **Phase 2 – continuation L‑BFGS**  
  Minimise a smoothed version of the max‑convolution
  `smooth_max = max + λ·log Σ exp(·/λ)`.
  λ is reduced geometrically (≈30 stages) and each stage runs a
  relatively long L‑BFGS optimisation (≈1200 iterations).  Many
  random starts (including perturbations of the current best) are
  performed while respecting the time budget.

* **Phase 3 – Adam‑style projected sub‑gradient descent**  
  Optimise the true nondifferentiable objective using the sub‑gradient
  at the active convolution index.  Adaptive step sizes (Adam) and
  Euclidean projection onto the simplex give a fast descent.

* **Phase 4 – aggressive hill‑climb**  
  Greedy moves of mass between indices (and a simple simulated‑annealing
  acceptance) finish the search.

All heavy work is performed with FFT‑based convolutions, which are
O(n log n) and fast for the chosen problem sizes (up to ≈15000).  The
routine monitors the supplied time budget (default 1000 s) and returns
the best sequence found before the deadline.

The code contains no closures or lambdas and uses only NumPy /
SciPy (both allowed).  The harness supplies `evaluate_sequence`,
which is called only once at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n−1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Return (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-15

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z   # ∂smooth_max/∂conv_i

    # ---- propagate gradient through the convolution ----------------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 1200,
) -> np.ndarray:
    """
    Continuation L‑BFGS: a series of optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting vector y (pre‑soft‑max) for a given heuristic."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "beta_rand":
        # a_i ∝ (i+1)^α (n‑i)^β  with α,β sampled widely
        α = rng.uniform(-2.5, 2.5)
        β = rng.uniform(-2.5, 2.5)
        i = np.arange(n, dtype=float)
        a0 = (i + 1) ** α * (n - i) ** β
        a0 = np.maximum(a0, 0)
        if a0.sum() == 0:
            a0 = np.ones(n)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # deterministic Sidon‑set‑like support (≈√n points)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    temperature: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Greedy / simulated‑annealing hill‑climb.
    Randomly move a fraction of mass from i to j.
    If ``temperature>0`` a worse move can be accepted with probability
    exp(-(Δ)/temperature).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        accept = False
        if new_score < best_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_score - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient descent on the true max‑convolution
    objective.  The sub‑gradient at the active index m is
        g_i = 2·a_{m‑i} (if the index exists).
    """
    a = a_start.copy()
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam updates
        m_t = beta1 * m_t + (1.0 - beta1) * g
        v_t = beta2 * v_t + (1.0 - beta2) * (g * g)

        m_hat = m_t / (1.0 - beta1 ** it)
        v_hat = v_t / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) with a low
    ``evaluate_sequence`` score, respecting the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.2   # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size.  Try to use a longer sequence than before
    # (previous runs used ≈5 k).  12 k is a good compromise.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size * 2), 15000)   # stretch a bit, cap at 15 k
        else:
            n = 12000
    else:
        n = 12000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / np.sum(arr)
        else:
            # truncate / pad uniformly
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1: cheap parametric exploration (Beta‑like families)
    # ------------------------------------------------------------------
    for _ in range(120):   # 120 random (α,β) samples
        α = rng.uniform(-2.5, 2.5)
        β = rng.uniform(-2.5, 2.5)
        i = np.arange(n, dtype=float)
        prob = (i + 1) ** α * (n - i) ** β
        prob = np.maximum(prob, 0)
        if prob.sum() == 0:
            continue
        prob /= prob.sum()
        sc = fast_score(prob, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = prob.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[grid] α={α:.2f} β={β:.2f} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: continuation L‑BFGS with many random starts
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=30))
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine",
             "beta_rand", "sidon", "best"]
    restart_counter = 0

    # Reserve about 300 s for the later Adam + hill‑climb.
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:

        y_opt = schedule_optimize(

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{restart_counter:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # short hill‑climb to polish this candidate

        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------

    final_val = evaluate_sequence(final_seq)   # harness supplies this

    return final_seq
