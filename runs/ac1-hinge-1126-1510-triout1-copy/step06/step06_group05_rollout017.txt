def adam_subgradient(a_start, fft_len, n, deadline, eta=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a_start.copy()
    m = 0
    mt = np.zeros_like(a)
    vt = np.zeros_like(a)
    t_iter = 0
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    while time.time() < deadline:
        t_iter += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        # subgradient g
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # note: need to also account for term when counterpart==idx? Already covered (adds 2*a_i) but correct subgradient of a_i a_i? Actually for i such that 2i=m, the term appears once (a_i*a_i). Derivative w.r.t a_i is 2 a_i (still okay).
        # Update rule (Adam)
        mt = beta1 * mt + (1 - beta1) * g
        vt = beta2 * vt + (1 - beta2) * (g ** 2)

            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Randomised parametric sweep** –  sample a large number of
   Beta‑like families (a_i ∝ (i+1)^α·(n−i)^β) optionally multiplied by a
   mild Gaussian envelope and a uniform offset.  Keep the best candidate.

2. **Continuation L‑BFGS on a smooth approximation** –  minimise the
   log‑sum‑exp (smooth‑max) of the self‑convolution using a decreasing
   smoothing schedule.  Each stage runs many L‑BFGS iterations, and the
   result is polished by a short greedy hill‑climb.

3. **Adam‑style projected sub‑gradient descent** –  use the exact
   sub‑gradient of the true ‖f*f‖_∞ objective and adapt the step size
   with Adam.  Finish with a long greedy hill‑climb.

All helpers are defined at the top level; the routine respects the
supplied time budget (default 1000 s) and returns the best sequence it
found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n-1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max parametrisation (unconstrained optimisation)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where

        f = 2·n·smooth_max(a*a)   (a = softmax(y))

    and ``grad_y`` is the gradient w.r.t. the unconstrained vector ``y``.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                     # max becomes 0
    exp_shift = np.exp(shifted / lam)         # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution ----------------------
    # g_vec = ∂conv/∂a (size n)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting ``y`` (pre‑soft‑max) for a chosen heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Simple hill‑climb (greedy mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i
    to j, accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def adam_subgradient(

) -> tuple[np.ndarray, float]:

        f(a) = 2·n·max_i (a*a)_i .
    The sub‑gradient at the active index m is
        g_i = 2·a_{m‑i}   (if the index exists).
    """
    a = a_start.copy()
    mt = np.zeros_like(a)
    vt = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam update
        mt = beta1 * mt + (1 - beta1) * g
        vt = beta2 * vt + (1 - beta2) * (g ** 2)
        m_hat = mt / (1 - beta1 ** t)
        v_hat = vt / (1 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    ``evaluate_sequence`` score.  The routine respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size.  Re‑use a previous construction if it is long
    # enough, otherwise pick a comfortable size (5000–8000).
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / arr.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 1.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – random parametric sweep (Beta‑like families)
    # ------------------------------------------------------------------
    param_deadline = min(deadline - 250.0, time.time() + 30.0)
    i = np.arange(n, dtype=float)
    while time.time() < param_deadline:
        alpha = rng.uniform(-1.2, 0.8)
        beta = rng.uniform(-1.2, 0.8)
        gamma = rng.uniform(0.0, 0.004)          # Gaussian envelope strength
        eps = rng.uniform(0.0, 0.05)             # uniform mixing term

        with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
            prob = np.power(i + 1.0, alpha) * np.power(n - i, beta)

        prob = np.where(np.isfinite(prob), prob, 0.0)
        if gamma > 0.0:
            prob *= np.exp(-gamma * ((i - (n - 1) / 2.0) ** 2))
        prob += eps
        prob = np.maximum(prob, 0.0)
        s = prob.sum()
        if s == 0.0:
            continue
        prob /= s

        sc = fast_score(prob, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = prob
            print(
                f"[param] α={alpha:.3f} β={beta:.3f} γ={gamma:.5f} ε={eps:.3f} → fast_score = {sc:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 2 – continuation L‑BFGS with aggressive schedule
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=70))
    y_best = np.log(best_a + 1e-12)

    restart = 0
    while time.time() < deadline - 120.0 and restart < 30:
        if restart == 0:
            # perturb the current best to give a fresh start
            y0 = y_best + rng.normal(scale=0.02, size=n)
        else:
            mode = rng.choice(
                ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon"]
            )
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=1200,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            y_best = y_opt
            print(f"[lbfgs {restart:02d}] improved fast_score = {sc_opt:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        y_best = np.log(best_a + 1e-12)
        restart += 1

    # ------------------------------------------------------------------
    # Phase 3 – Adam‑style sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        sub_deadline = deadline - 10.0
        print("[adam] start")
        best_a, best_score = adam_subgradient(
            best_a, fft_len_conv, n, sub_deadline, eta=0.12
        )
        print(f"[adam] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
