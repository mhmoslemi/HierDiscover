"""
Search for a low‑evaluation non‑negative step sequence.

The algorithm mixes three ideas:

1. **LP‑based “max‑convolution improvement”** – for a fixed distribution f we
   maximise the total mass of a second distribution g under the linear
   constraints (f * g) ≤ M where M is the current maximum of f * f.
   After normalising g we obtain a new f with a (strictly) smaller max‑convolution.
   This step is cheap for moderate n (≈2 000) and drives the objective far
   below the uniform value 2.0.

2. **Upscaling** – the best distribution found for the smaller size is
   linearly interpolated to the target length (n = 4000) and re‑normalised.

3. **Fine‑grained gradient polishing** – a continuation of smoothed L‑BFGS,
   projected sub‑gradient descent with momentum, exponentiated‑gradient descent,
   simulated annealing and a final greedy hill‑climb (exactly as in the
   previous best‑known pipeline).  All phases respect the simplex
   (non‑negative, sum = 1) and use FFT‑based convolutions for speed.

The whole search runs for up to the allotted budget (≈1000 s) and returns
the best sequence found.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised (∑a=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives (same as in the previous pipeline)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = opt.minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the sub‑gradient of
    the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional random jitter to escape plateaus
        if (it + 1) % 5000 == 0:
            if rng.random() < 0.3:
                sigma = rng.uniform(0.001, 0.008)
                a = a + rng.normal(0.0, sigma, size=n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def random_anneal(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that moves a random fraction of mass between two
    coordinates.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10  # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: relocate a small random fraction of mass from one
    index to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial_n(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "sin"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sin
        x = np.arange(1, n + 1)
        a = np.sin(np.pi * x / (n + 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

# Linear‑programming “max‑convolution” improvement
# ----------------------------------------------------------------------


_lp_matrix_cache = {}

def lp_matrix_indices(n: int):
    """Return (row_idx, col_idx) for the Toeplitz convolution matrix of size (2n‑1)×n."""
    global _lp_matrix_cache
    if n in _lp_matrix_cache:
        return _lp_matrix_cache[n]
    # each column i contributes f_i to rows i … i+n‑1
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).astype(np.int32).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    _lp_matrix_cache[n] = (row_idx, col_idx)
    return row_idx, col_idx


def lp_improve(
    a: np.ndarray,
    max_it: int,
    deadline: float,
) -> np.ndarray:
    """
    Repeated LP improvement: maximise total mass of g under the linear constraints
    (a * g) ≤ M, where M = max(a*a).  After each LP we renormalise g to sum = 1.
    The process stops when no improvement in the max‑convolution is observed
    or the time budget is exhausted.
    """
    n = a.shape[0]
    row_idx, col_idx = lp_matrix_indices(n)

    while time.time() < deadline and max_it > 0:
        # current bound M
        fft_len = next_pow2(2 * n - 1)
        conv = full_conv(a, fft_len)
        M = conv.max()
        # build A matrix (sparse) for constraint (a * g) ≤ M
        data = np.repeat(a, n)      # shape (n*n,)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
        # linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # if the LP fails, fall back to the current a
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # Normalise to the simplex
        a_new = g / sum_g

        # check improvement
        conv_new = full_conv(a_new, fft_len)
        M_new = conv_new.max()
        if M_new >= M * (1 - 1e-6):
            # no significant improvement
            break
        a = a_new
        max_it -= 1
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Target size (final sequence) – keep the historic size of 4000
    # ------------------------------------------------------------------
    n_big = 4000
    fft_len_big = next_pow2(2 * n_big - 1)
    fft_len_g_big = next_pow2(3 * n_big - 2)

    # ------------------------------------------------------------------
    # Phase 0 – optional warm start from a known construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # interpolate to the target length
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n_big)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n_big, 1.0 / n_big)
        else:
            a0 = np.full(n_big, 1.0 / n_big)
    else:
        a0 = np.full(n_big, 1.0 / n_big)

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based search on a reduced problem size (n≈2000)
    # ------------------------------------------------------------------
    n_small = 2000
    fft_len_small = next_pow2(2 * n_small - 1)

    best_small_a = np.full(n_small, 1.0 / n_small)
    best_small_score = fast_score(best_small_a, fft_len_small, n_small)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # allocate ~30 % of time for the LP phase
    phase1_end = start_time + 0.30 * budget_s

    print("[LP‑phase] start (n_small={}, budget≈{:.1f}s)".format(n_small, phase1_end - start_time))

    while time.time() < phase1_end:
        # generate a diverse random start
        a_rand = random_initial_n(n_small, rng)

        # a few LP improvement steps (max 5)
        a_refined = lp_improve(a_rand, max_it=5, deadline=phase1_end)

        score = fast_score(a_refined, fft_len_small, n_small)
        if score < best_small_score:
            best_small_score = score
            best_small_a = a_refined.copy()
            print("[LP‑phase] new best small score={:.6f}".format(best_small_score))

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to the full size
    # ------------------------------------------------------------------
    print("[up‑sample] best small score={:.6f}".format(best_small_score))
    x_small = np.linspace(0, 1, n_small)
    x_big = np.linspace(0, 1, n_big)
    a_big = np.interp(x_big, x_small, best_small_a)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    best_a = a_big.copy()
    best_score = fast_score(best_a, fft_len_big, n_big)
    print("[up‑sample] initial big score={:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # Phase 3 – gradient‑based polishing (mirrors the previous pipeline)
    # ------------------------------------------------------------------
    # split the remaining time into the same stages as before
    remaining = deadline - time.time()
    if remaining <= 0:
        remaining = 0.0
    phase2_end = time.time() + 0.45 * remaining   # sub‑gradient + momentum
    phase3_end = time.time() + 0.70 * remaining   # exponentiated gradient
    phase4_end = time.time() + 0.80 * remaining   # short L‑BFGS refinement
    phase5_end = time.time() + 0.90 * remaining   # simulated annealing
    # rest for final hill‑climb

    # ----- sub‑gradient with momentum -----
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_big, fft_len_g_big, n_big
        )
        print("[sub‑gradient] score={:.6f}".format(best_score))

    # ----- exponentiated‑gradient descent -----
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_big, fft_len_g_big, n_big
        )
        print("[exp‑gradient] score={:.6f}".format(best_score))

    # ----- short L‑BFGS refinement -----
    if time.time() < phase4_end:
        fine_lam_schedule = [0.001, 0.0005, 0.0002, 0.0001]
        fine_maxiter = 400
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            phase4_end,
            fft_len_big,
            fft_len_g_big,
            n_big,
        )
        if fine_score < best_score:
            best_score = fine_score
            print("[L‑BFGS‑fine] score={:.6f}".format(best_score))

    # ----- simulated annealing -----
    if time.time() < phase5_end:
        best_a, best_score = random_anneal(
            best_a, best_score, fft_len_big, n_big, phase5_end
        )
        print("[anneal] score={:.6f}".format(best_score))

    # ----- final greedy hill‑climb -----
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a, best_score, fft_len_big, n_big, final_deadline, max_iter=12000
    )
    print("[final hill‑climb] score={:.6f}".format(best_score))

    final_seq = best_a.tolist()
    # final evaluation using the provided external function
    final_eval = evaluate_sequence(final_seq)
    print("[final] evaluated score = {:.6f}".format(final_eval))

    return final_seq
