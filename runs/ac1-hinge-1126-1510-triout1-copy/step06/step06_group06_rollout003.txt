    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    for lam in lam_factors:
        for _ in range(max_iter_per_stage):
            t += 1
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            # step size
            step = lr / (np.sqrt(v_hat) + eps)
            y = y - step * m_hat
    return y

   ``2·n·max(conv)`` and keep the best sequence.
5. Refine the best sequence with two simple hill‑climbs:
   * *Targeted* – move mass away from the pair that contributes most
     to the current maximal convolution entry.
   * *Random* – classic mass‑transfer hill‑climb.
6. Finish with a projected sub‑gradient descent on the true (non‑smooth)
   max‑convolution objective.
7. Store the best sequence globally (``height_sequence_1``) and return it.
   All steps respect the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(a,a)) where the convolution is computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (w.r.t. log‑soft‑max params)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # --- convolution a*a (full) ---
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max ---
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # --- gradient of smooth max w.r.t. conv ---
    w = exp_shift / Z

    # --- propagate gradient back to a ---
    # convolution of w with reversed a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the part that corresponds to derivatives w.r.t. a[0..n-1]
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient of f wrt a
    grad_a = 2.0 * n * g_vec

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
# Simple Adam optimiser for the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    max_iter_per_stage: int = 200,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Run a continuation of Adam optimisations over a decreasing λ‑schedule.
    """
    y = y0.copy()
    for lam in lam_factors:
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        t = 0
        for _ in range(max_iter_per_stage):
            t += 1
            loss, grad = objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam_factor=lam
            )
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1**t)
            v_hat = v / (1.0 - beta2**t)
            step = lr / (np.sqrt(v_hat) + eps)
            y = y - step * m_hat
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities (identical to the previous implementation)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Focused hill‑climb that attacks the currently worst convolution shift."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift with maximal overlap

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient of the
    (non‑smoothed) max‑operator.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to the symmetric half so that 0 <= k <= n-1
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad


# ----------------------------------------------------------------------
# Initialisation helpers (produce log‑soft‑max parameters)
# ----------------------------------------------------------------------


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a diverse starting point for the optimiser.
    Various heuristic families are mixed to keep a wide search space.
    """
    mode = rng.choice(
        ["random", "exp_decay", "exp_growth", "power_decay", "power_growth",
         "edge", "triangular", "best"]
    )
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp_decay":
        rate = rng.uniform(0.0005, 0.02)
        a0 = np.exp(-rate * np.arange(n, dtype=float))
    elif mode == "exp_growth":
        rate = rng.uniform(0.0005, 0.02)
        a0 = np.exp(rate * np.arange(n, dtype=float))
    elif mode == "power_decay":
        p = rng.uniform(0.2, 3.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "power_growth":
        p = rng.uniform(0.2, 3.0)
        a0 = ((np.arange(1, n + 1, dtype=float) ** p))
    elif mode == "edge":
        # concentrate a little bit near both ends
        m_edge = max(1, int(n * rng.uniform(0.005, 0.02)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
    elif mode == "triangular":
        # linearly increasing then decreasing
        half = n // 2
        inc = np.linspace(1, half, half, endpoint=False)
        dec = inc[::-1]
        a0 = np.concatenate([inc, dec]) if n % 2 == 0 else np.concatenate([inc, [half], dec])
    else:  # "best"
        # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops short of this limit.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # keep a safety margin so that the harness does not kill us prematurely
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – use a larger n when we have enough time left
    # --------------------------------------------------------------
    if budget_s > 700:
        n = 8000
    else:
        n = 5000

    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) >= n // 2:
        # reuse a truncated/extended version of the previous best
        arr = np.asarray(prev, dtype=float)
        if arr.size > n:
            best_a = arr[:n].copy()
        else:
            # pad with a tiny uniform tail if necessary
            pad = np.full(n - arr.size, 1.0 / (n * 10))
            best_a = np.concatenate([arr, pad])
        best_a = np.maximum(best_a, 0.0)
        if best_a.sum() == 0.0:
            best_a = np.full(n, 1.0 / n)
        else:
            best_a = best_a / best_a.sum()
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len_conv, n)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Global optimisation – Adam with a λ‑schedule and many restarts
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-8, num=40)   # finer temperature schedule
    maxiter_per_stage = 180                         # a bit more work per stage

    iteration = 0
    while time.time() < deadline - 6.0:
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = adam_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_factors,
            max_iter_per_stage=maxiter_per_stage,
            lr=0.07,                # a moderately aggressive learning rate
        )

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] Adam improved fast score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Targeted hill‑climb (short burst)
        # ----------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----------------------------------------------------------
        # Random hill‑climb (short burst)
        # ----------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=4000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Projected sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    step0 = 0.45
    subgrad_iter = 0
    eps = 1e-12

    while time.time() < deadline - 2.0:
        subgrad_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)

        # Simple diminishing step‑size schedule
        step = step0 / np.sqrt(subgrad_iter)

        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)      # gently increase step if we are making progress
        else:
            step0 *= 0.92                       # otherwise shrink it

        if subgrad_iter % 200 == 0:
            print(f"[subgrad] iter={subgrad_iter} best fast score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
