def adam_subgradient(a_start, fft_len, n, deadline, alpha=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t =0
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    while time.time() < deadline:
        t+=1
        conv = conv_fft(a, fft_len, n)
        m_idx = np.argmax(conv)
        # subgradient:
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >=0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2 * a[counterpart[mask]]
        # Adam update:
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = project_onto_simplex(a)
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

    orig_n = len(seq)
    x_orig = np.linspace(0, 1, orig_n)

    seq_up = np.interp(x_new, x_orig, seq)

"""
Propose a low‑score step‑function.

The algorithm mixes several complementary optimisation phases:

1. **Initialisation** – If a previous construction is available it is
   up‑sampled (to a target length) and normalised, otherwise a simple
   arcsine‑shaped distribution is used.

2. **Coarse grid search** – A dense Beta‑family (a_i ∝ (i+1)^α (n‑i)^β)
   is scanned; the best candidate seeds the subsequent optimisation.

3. **Continuation L‑BFGS on a smooth max** – The true objective
   ``2·n·max(conv)`` (with ``sum(a)=1``) is approximated by a
   log‑sum‑exp smoothing.  A geometric schedule drives the smoothing
   factor to 1e‑12 while a fairly long L‑BFGS run (≈1200 iterations)
   per stage refines the solution.  Many random restarts (including a
   perturbation of the current best) are performed until the time budget
   for this phase is exhausted.

4. **Projected Adam on the exact max‑convolution** – After the
   smoothed optimisation, an Adam‑style projected sub‑gradient descent
   directly minimises the true max‑convolution.  The Adam updates
   converge much faster than plain sub‑gradient descent.

5. **Aggressive greedy hill‑climb** – Finally a fast greedy
   hill‑climber repeatedly moves a random fraction of probability mass
   from one index to another, retaining only improving moves.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
so the harness’s denominator ``sum(a)^2`` equals 1 and the score is
exactly ``2·n·max(conv)``.  The routine respects the supplied time
budget (default 1000 s) and prints intermediate scores for monitoring.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation of the objective for probability vectors (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c  # now max = 0
    exp_shift = np.exp(shifted / lam)  # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution ----------------------
    # Compute g_vec = ∂conv/∂a (size n)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)  # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 1200,
) -> np.ndarray:
    """
    Continuation L‑BFGS: a cascade of optimisations with decreasing smoothing.
    Each stage starts from the previous stage's result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j,
    accepting only improvements of the exact (fast) score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_max_conv(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    alpha0: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation on the true max‑convolution objective.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        # sub‑gradient of max(conv) at the active index
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_onto_simplex(a)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a low
    ``evaluate_sequence`` score.  The search respects the time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # -------------------------- time handling ---------------------------------
    deadline = time.time() + budget_s - 0.5          # safety margin

    # -------------------------- problem size ---------------------------------
    # Try to use a longer vector than the historic 5000 if possible.
    # We cap at 10000 to keep memory and runtime reasonable.
    base_seq = globals().get("height_sequence_1", None)
    target_n = 8000
    if isinstance(base_seq, (list, tuple, np.ndarray)):
        base_arr = np.asarray(base_seq, dtype=float).ravel()
        if base_arr.ndim == 1:
            if base_arr.size >= target_n:
                n = min(base_arr.size, 10000)
                best_a = base_arr / base_arr.sum()
                best_y = np.log(best_a + 1e-12)
            else:
                # up‑sample to target_n
                n = target_n
                x_old = np.linspace(0, 1, base_arr.size)
                x_new = np.linspace(0, 1, n)
                up = np.interp(x_new, x_old, base_arr)
                up = up / up.sum()
                best_a = up
                best_y = np.log(best_a + 1e-12)
        else:
            n = target_n
            best_a = np.full(n, 1.0 / n)
            best_y = np.log(best_a + 1e-12)
    else:
        n = target_n
        best_a = np.full(n, 1.0 / n)
        best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # -------------------------- cheap grid search ----------------------------
    alphas = np.linspace(-0.8, 0.5, 11)
    betas = np.linspace(-0.8, 0.5, 11)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            prob = prob / prob.sum()
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # -------------------------- L‑BFGS phase ---------------------------------
    # 50 geometric steps, ending at 1e‑12 smoothing
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=70))
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]
    l_bfgs_end = deadline - 300.0      # keep ~5 min for Adam + final HC

    iteration = 0
    while time.time() < l_bfgs_end:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=1200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # short hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # -------------------------- Adam refinement -------------------------------
    if time.time() < deadline - 120.0:
        adam_deadline = deadline - 60.0
        print("[adam] start")
        best_a, best_score = adam_max_conv(
            best_a,
            fft_len_conv,
            n,
            adam_deadline,
            alpha0=0.2,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[adam] finished → fast_score = {best_score:.6f}")

    # -------------------------- final hill‑climb -----------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # -------------------------- return result --------------------------------
    final_seq = best_a.tolist()
    # The harness will call evaluate_sequence on this list.
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
