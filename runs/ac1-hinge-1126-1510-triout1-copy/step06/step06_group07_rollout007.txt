"""
Hybrid optimizer for the non‑negative step‑function problem.

The algorithm combines three phases under a strict time budget:

1. **Global search** – many restarts of Adam on a smooth‐max
   approximation of the convolution peak.  The temperature τ is
   decreased geometrically (continuation method) so that the
   surrogate converges to the true max.

2. **Fine tuning** – a projected sub‑gradient descent on the exact
   max of the auto‑convolution (the same idea as the previous SOTA
   “targeted hill‑climb” but with Nesterov‑style momentum).

3. **Aggressive random hill‑climb** – short random mass‑transfer
   sweeps that keep the simplex constraint and accept only improving
   moves.

The best sequence found is stored in the global name
`height_sequence_1` for reuse in later runs.  The routine respects
the 1000 s time budget and returns a list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for probability vectors."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_max_and_grad(y: np.ndarray, n: int, fft_len: int, tau: float):
    """
    Compute the smooth approximation of max(a*a) and the gradient
    with respect to the logits y (where a = softmax(y)).
    Returns (objective, grad_y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # smooth‑max via log‑sum‑exp
    shifted = conv - max_c                     # peak at zero
    exp_shift = np.exp(shifted / tau)
    Z = exp_shift.sum()
    smooth_max = max_c + tau * np.log(Z)
    obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # back‑propagation: g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    # Use FFT for the convolution a * w
    fa_rev = np.fft.fft(a[::-1], fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(obj), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray, n: int, fft_len: int,
                  tau_schedule, iters_per_tau: int,
                  rng: np.random.Generator):
    """
    Continuation Adam optimiser.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0  # global iteration counter

    for tau in tau_schedule:
        for _ in range(iters_per_tau):
            t += 1
            obj, grad = smooth_max_and_grad(y, n, fft_len, tau)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.12 / (np.sqrt(v_hat) + eps)   # fixed base learning rate
            y -= step * m_hat

    return y


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, cur_score: float, fft_len: int,
               n: int, deadline: float,
               max_iter: int = 6000,
               max_move_frac: float = 0.25):
    """
    Random pairwise mass transfer that accepts only improvements.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Targeted sub‑gradient descent on the true max (no smoothing)
# ----------------------------------------------------------------------


def subgrad_descent(a: np.ndarray, cur_score: float,
                    n: int, fft_len: int,
                    deadline: float,
                    max_iter: int = 4000,
                    step0: float = 0.3):
    """
    Projected sub‑gradient descent using the exact max of the convolution.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for the active shift k:
        grad = np.zeros_like(best)
        # grad_i = 2 * a_{k-i} when index valid
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range
        grad[i_range] = 2.0 * best[j_range]

        # step size (diminishing)
        step = step0 / np.sqrt(it)

        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-14:
            best = cand
            best_score = cand_score
        else:
            # occasional probabilistic acceptance to escape plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score

    return best, best_score


# ----------------------------------------------------------------------
# Initial shape generators (used for diverse restarts)
# ----------------------------------------------------------------------


def init_random(n, rng):
    return rng.standard_normal(n)


def init_exponential(n, rng):
    rate = rng.uniform(0.001, 0.02)
    a0 = np.exp(-rate * np.arange(n))
    a0 /= a0.sum()
    return np.log(a0 + 1e-12)


def init_powerlaw(n, rng):
    p = rng.uniform(0.3, 2.0)
    a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
    a0 /= a0.sum()
    return np.log(a0 + 1e-12)


def init_edge_heavy(n, rng):
    """Large mass on the two ends, small mass in the middle."""
    m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
    # start from a very simple edge‑heavy shape
    edge_val = np.sqrt(n / 2.0) / n
    centre_val = 1.0 / (np.sqrt(2.0) * n)
    a0 = np.full(n, centre_val)
    a0[:m_edge] = edge_val
    a0[-m_edge:] = edge_val
    a0 /= a0.sum()
    return np.log(a0 + 1e-12)


def init_arcsine_like(n):
    """Symmetric arcsine‑type density: a_i ∝ 1/√(i·(n‑1‑i) + ε)."""
    eps = 1e-6
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a /= a.sum()
    return np.log(a + 1e-12)
