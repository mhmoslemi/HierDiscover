"""
Propose a low‑score step‑function sequence.

This implementation follows a *conditional‑gradient* (Frank–Wolfe) style
scheme: given the current distribution `f` we solve a small linear program
that finds a feasible direction `g` which maximises the total mass
while respecting the current worst autocorrelation bound `M`.  The update
`f ← (1‑t)·f + t·g` strictly reduces the maximal convolution value.
A few cheap hill‑climbs are performed at the end for polishing.

The algorithm differs from the previous L‑BFGS / sub‑gradient approach
and is able to explore a complementary part of the search space.
"""

import time
import numpy as np
from scipy.optimize import linprog


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for a probability vector `a`
    (sum(a) == 1): 2·n·max(a*a).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def build_constraint_matrix(f: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """
    Build a dense constraint matrix A where A[k, j] = f[i] with i + j = shift.
    Each row corresponds to one shift from ``shifts``.
    """
    n = f.size
    m = shifts.size
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(shifts):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        A[row, j] = f[i]
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops a few seconds before the
        deadline to guarantee a return.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse an existing construction if possible,
    # otherwise start from a moderate size.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = int(arr.size)
            f = np.maximum(arr, 0.0)
            if f.sum() == 0.0:
                f = np.full(n, 1.0 / n)
            else:
                f = f / f.sum()
        else:
            n = 2000
            f = np.full(n, 1.0 / n)
    else:
        n = 2000
        f = np.full(n, 1.0 / n)

    # ensure a proper probability vector (the objective is scale‑invariant)
    f = np.maximum(f, 0.0)
    f = f / f.sum()

    best_f = f.copy()
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_f, fft_len, n)
    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – conditional gradient + small hill‑climbs
    # ------------------------------------------------------------------
    # active‑set size (number of convolution constraints fed to the LP)
    active_k = 60
    # step size for the convex combination
    t = 0.30
    iteration = 0

    while time.time() < deadline - 2.0:
        iteration += 1

        # 1) compute the current autocorrelation and the worst value M
        fa = np.fft.fft(f, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # 2) choose a small set of “active’’ shifts where the bound is tight
        #    (always keep the zero shift)
        top_idx = np.argsort(-conv)[: active_k]
        if (n - 1) not in top_idx:
            top_idx = np.append(top_idx, n - 1)
        shifts = np.unique(top_idx)               # ensure no duplicates

        # 3) build the LP constraints A·g ≤ M
        A = build_constraint_matrix(f, shifts)

        # 4) solve: maximise 1ᵀ g  (⇔ minimise –1ᵀ g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        lp_res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(A.shape[0], M, dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0, deadline - time.time() - 0.5)},
        )

        if not lp_res.success:
            # If the LP fails, reduce the active set size and continue.
            active_k = max(5, active_k // 2)
            continue

        g = lp_res.x
        if g.sum() == 0.0:
            # degenerate solution – shrink active set and retry.
            active_k = max(5, active_k // 2)
            continue

        # 5) normalise g to stay on the probability simplex
        g /= g.sum()

        # 6) convex update
        f_new = (1.0 - t) * f + t * g

        # 7) evaluate the fast surrogate (no need to call the costly evaluator)
        new_score = fast_score(f_new, fft_len, n)

        # keep the best version seen so far
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            print(f"[{iteration}] improved fast score = {best_score:.6f}")

        f = f_new

        # occasional cheap random hill‑climb to escape tiny plateaus
        if iteration % 15 == 0:
            # simple mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i != j and best_f[i] > 0.0:
                delta = best_f[i] * rng.random() * 0.15
                cand = best_f.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_score = fast_score(cand, fft_len, n)
                if cand_score < best_score:
                    best_f = cand
                    best_score = cand_score
                    f = cand
                    print(f"[{iteration}] hill‑climb improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with a short exhaustive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        max_iters = 3000
        a = best_f.copy()
        cur_score = best_score
        while time.time() < deadline and max_iters > 0:
            max_iters -= 1
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.20
            a[i] -= delta
            a[j] += delta
            sc = fast_score(a, fft_len, n)
            if sc < cur_score:
                cur_score = sc
                best_f = a.copy()
                best_score = sc
        print(f"[final] polished fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (the evaluation harness will call the true
    # ``evaluate_sequence`` on it).  Store it globally for possible reuse.
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    globals()["height_sequence_1"] = final_seq
    return final_seq
