import cvxpy as cp

def lp_improve(a, M, t=0.3):
    n = len(a)
    a_const = a.astype(float)  # ensure np array
    # define variable g
    g = cp.Variable(n)
    # constraint: conv(a_const, g) <= M
    conv = cp.conv(a_const, g)  # length 2n-1
    constraints = [conv <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # solve using ECOS, maybe parameters for time limit: solver_options={"max_iters":5000}
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    g_val = np.maximum(g.value, 0)  # sometimes small negative due to tolerance
    sum_g = g_val.sum()
    if sum_g <= 0:
        return a
    # normalize to sum 1
    g_norm = g_val / sum_g
    # combine
    new_a = (1 - t) * a + t * g_norm
    # re-normalize to sum 1
    new_a = new_a / new_a.sum()
    return new_a
```

"""
Propose a low‑evaluation step sequence.

Algorithm Overview
------------------
1. **Initialization** – start from the previous best construction (if any)
   or a random distribution on the simplex of length 4000.

2. **Heavy L‑BFGS continuation** – many random‑restart runs on a
   smoothed log‑sum‑exp approximation of the max‑convolution.
   The smoothing factor is decreased in stages (continuation).  The
   current best sequence (and the best score) are kept.

3. **LP‑based flattening** – given the current sequence `a` and its
   convolution peak `M = max(a*a)`, solve the linear program

        maximize   sum(g)
        subject to conv(a,g) ≤ M   (elementwise)
                   g ≥ 0

   (implemented with CVXPY).  The optimiser `g` is a flatter version of
   `a`.  We mix `a ← (1‑t)·a + t·g` (with `t≈0.5`) and repeat a few
   times.  This step is a direct implementation of the “active‑set”
   improvement described in the statement and often yields a noticeable
   drop in the objective.

4. **Projected sub‑gradient descent with momentum** – fine‑tunes the
   sequence on the true max‑convolution using sub‑gradients.

5. **Greedy hill‑climb** – deterministic local moves that strictly
   improve the fast score.

6. **Final evaluation** – the best sequence found is evaluated with
   the provided ``evaluate_sequence`` function and returned.

All operations keep the vector on the probability simplex
(non‑negative, sum = 1), because the evaluation is scale‑invariant.
The code respects the 1000 s time budget by checking the deadline
between phases.
"""

import time
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised (∑a=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: relocate a small random fraction of mass from one
    index to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# LP‑based flattening
# ----------------------------------------------------------------------


def solve_lp_improve(a: np.ndarray, M: float, max_iters: int = 5000) -> np.ndarray:
    """
    Given a current distribution a (sum=1) and a bound M on the convolution,
    solve the LP:
        max   sum(g)
        s.t.  conv(a,g) ≤ M  (elementwise)
              g ≥ 0
    Returns the optimiser g, normalised to sum(g)=1.
    """
    n = a.shape[0]
    a_const = a.astype(float)  # will be treated as a Constant in CVXPY
    g = cp.Variable(n)

    # full convolution expression (length 2n‑1)
    conv_expr = cp.conv(a_const, g)

    constraints = [conv_expr <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is a fast LP/QP solver; we bind a modest iteration budget.
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=max_iters, verbose=False)

    if g.value is None:
        raise RuntimeError("LP did not converge")

    g_val = np.maximum(g.value, 0.0)
    sum_g = g_val.sum()
    if sum_g <= 0.0:
        # Fallback – return the original a (should not happen)
        return a.copy()
    return g_val / sum_g

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP based flattening (active‑set improvement)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        t_mix = 0.5                         # mixing weight for g
        lp_iters = 5                        # how many LP refinements
        for it in range(lp_iters):
            if time.time() >= phase2_end:
                break
            # current max of the convolution
            fa = np.fft.fft(best_a, fft_len_conv)
            conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            M = conv_full.max()

            try:
                g_opt = solve_lp_improve(best_a, M, max_iters=3000)
            except Exception as e:
                print(f"[LP] iteration {it+1} failed: {e}")
                break

            # blend and re‑normalise
            best_a = (1.0 - t_mix) * best_a + t_mix * g_opt
            best_a = best_a / best_a.sum()

            new_score = fast_score(best_a, fft_len_conv, n)
            if new_score < best_score - 1e-9:
                best_score = new_score
                print(f"[LP] iteration {it+1} improved to {best_score:.6f}")
            else:
                print(f"[LP] iteration {it+1} no improvement (score={new_score:.6f})")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=12000,
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
