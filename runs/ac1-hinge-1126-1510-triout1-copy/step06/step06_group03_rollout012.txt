"""
Propose a step‑function (list of non‑negative floats) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (∑ a_i = 1).  The score is homogeneous,
  therefore scaling does not matter.
* Use a moderately large n (≈ 8000–10000).  Larger n gives a finer
  discretisation of the continuous problem where the optimal constant is ≈ 1.
* Initialise several analytically‑motivated candidates:
      – arcsine‑like β(α,α) density,
      – Wigner semicircle,
      – uniform, random Gaussian, sparse binary mask, single Gaussian bump,
      – a “spectral” seed obtained by (approximately) taking the square‑root
        of the Fourier transform of a perfectly flat convolution.
* For every seed run a few L‑BFGS iterations on the **L2** objective  
  L(a)=‖a∗a−c‖₂² with c=1/(2n−1) (the uniform convolution).  This drives the
  auto‑convolution towards flatness while keeping the constraints via a
  soft‑max parametrisation.
* Keep the best candidate (measured with the true fast_score = 2·n·max(conv)).
* Refine the best candidate with projected sub‑gradient descent on the true
  max‑convolution objective.
* Finish with aggressive stochastic hill‑climbing (random and targeted
  moves) under the simplex constraint.
* All steps respect the global 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is length n, fft_len ≥ 2*n‑1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Quick evaluation of the true objective for a probability vector (∑a=1):
    2·n·max(conv).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, ∑x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation helpers (different analytic shapes)
# ----------------------------------------------------------------------


def arcsine_init(n: int, alpha: float = 0.75) -> np.ndarray:
    """Symmetric beta(α,α) density (U‑shaped for α<1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle on {0,…,n‑1}."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)      # ensure positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_a(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector according to the selected mode."""
    if mode == "arcsine":
        return arcsine_init(n, alpha=0.70)
    if mode == "semicircle":
        return semicircle_init(n)
    if mode == "uniform":
        return uniform_init(n)
    if mode == "random":
        return random_gaussian_init(n, rng)
    if mode == "binary":
        return binary_sparse_init(n, rng, density=0.15)
    if mode == "bump":
        return gaussian_bump_init(n, rng)
    # fallback
    return random_gaussian_init(n, rng)


def spectral_init(n: int) -> np.ndarray:
    """
    Approximate a sequence whose auto‑convolution is flat.
    We take the square‑root of the Fourier transform of a perfectly flat target.
    """
    N = next_pow2(2 * n - 1)
    target = np.zeros(N)
    target[: 2 * n - 1] = 1.0 / (2 * n - 1)   # flat target for indices 0..2n-2
    F = np.fft.fft(target)
    sqrt_F = np.sqrt(F)                     # principal complex square‑root
    a_full = np.real(np.fft.ifft(sqrt_F))
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = uniform_init(n)
    else:
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# L2 (squared‑error) objective and its gradient (soft‑max parametrisation)
# ----------------------------------------------------------------------


def l2_objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    avg: float,
) -> tuple[float, np.ndarray]:
    """
    Objective  L(a) = ‖a∗a − avg‖₂²  and its gradient w.r.t. the unconstrained
    vector y (a = softmax(y)).  The constant avg = 1/(2n‑1) is the value of a
    perfectly flat convolution.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    diff = conv - avg
    obj = np.sum(diff * diff)

    # w = ∂L/∂conv = 2·diff
    w = 2.0 * diff

    # Gradient w.r.t. a: conv(a_rev, w)  – extracted central n entries
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return obj, grad_y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb that moves a fraction of mass from i to j."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from indices
    contributing to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # indices that appear in the max entry: i + j = m
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a)=2·n·max(conv)  (i.e. minimise max(conv)
    under the simplex constraint).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # ------------------------------------------------------------------

    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):

        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g    = next_pow2(3 * n - 2)
    avg_conv = 1.0 / (2 * n - 1)   # target flat convolution value

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:

        else:
            best_a = uniform_init(n)
    else:
        best_a = uniform_init(n)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # L2 optimisation phase (≈ 45 % of the budget)
    # ------------------------------------------------------------------
    modes = ["arcsine", "semicircle", "uniform", "random", "binary", "bump", "spectral"]
    mode_cycle = 0
