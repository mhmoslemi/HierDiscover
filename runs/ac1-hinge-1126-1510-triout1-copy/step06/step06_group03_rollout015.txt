def beta_shape(n, alpha):
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1) * np.power(1 - x, alpha - 1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    # avoid zeros: if sum zero, set uniform
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a

"""
Propose a step‑function (list of non‑negative floats) that minimizes
`evaluate_sequence`.  The search works in three phases:

1. **Parameter sweep** – generate a variety of analytically‑motivated
   shapes (beta, cosine‑power, polynomial‑power, mixtures) and keep the
   best few on the probability simplex (∑a_i = 1).

2. **Continuation L‑BFGS** – for each promising start run a smooth
   approximation of the max‑convolution objective with a decreasing
   smoothing factor (λ).  The optimisation is performed on the
   unconstrained log‑softmax parameters; after every L‑BFGS run a short
   greedy hill‑climb is applied.

3. **Projected sub‑gradient descent + final hill‑climb** – refine the
   best candidate with many sub‑gradient steps on the true max‑convolution
   objective, then use aggressive random moves until the time budget
   expires.

All work is done on the simplex, so the scale of the sequence does not
affect the final `evaluate_sequence` score.  The implementation respects
the 1000 s budget, uses only NumPy/SciPy and prints progress messages.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is length n, fft_len ≥ 2*n‑1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a_i = 1):
    2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (log‑softmax parameters).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # shifted so that max becomes 0
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv
    w = exp_shift / Z  # shape (2n‑1,)

    # derivative of conv wrt a (reverse convolution)
    # conv = a ⊛ a   →  ∂conv/∂a = 2 * (a reversed) ⊛ I
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv_i/∂a_i
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # lam is interpreted as lam_factor inside objective_and_grad
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Analytic shape generators
# ----------------------------------------------------------------------


def beta_shape(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) density on (0,1) – for α<1 we obtain U‑shaped curves."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Positive log‑Gaussian random vector, projected onto the simplex."""
    y = rng.standard_normal(n)
    a = np.exp(y)  # ensure positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    """Sparse binary mask (0/1) with given density, then normalized."""
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def cos_power_shape(n: int, p: float) -> np.ndarray:
    """Shape a_i ∝ sin(π·x)^p, where x∈(0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def poly_power_shape(n: int, p: float) -> np.ndarray:
    """Shape a_i ∝ (x·(1‑x))^p, x∈(0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = (x * (1.0 - x)) ** p
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random greedy hill‑climb: move a random fraction of mass from i to j.
    Accept the move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from indices that
    actively contribute to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)

        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv(a,a)).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new
    return best_a, best_score
