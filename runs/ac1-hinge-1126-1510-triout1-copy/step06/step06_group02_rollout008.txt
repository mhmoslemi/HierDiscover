def stochastic_hill_climb(a, init_score, fft_len, n, deadline):
    best = a.copy()
    best_score = init_score
    current = a.copy()
    current_score = init_score
    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        # compute temperature based on remaining time
        elapsed = time.time() - start_time
        total = deadline - start_time
        t = 0.5 * (1 - elapsed / total)  # decreasing from 0.5 to 0
        # choose source and target
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose delta as fraction of source, scaled by temperature
        max_move = 0.2 * t + 0.001   # up to 20% early, down to small later
        delta = current[i] * rng.random() * max_move
        if delta == 0:
            continue
        # propose move
        new = current.copy()
        new[i] -= delta
        new[j] += delta
        new_score = fast_score(new, fft_len, n)
        # accept if improve or with probability
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / (t+1e-12)):
            current = new
            current_score = new_score
            if new_score < best_score:
                best = new
                best_score = new_score
    return best, best_score

    iter = 0
    while time.time() < deadline:
        # conv
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        active = np.where(conv >= max_val - tol)[0]
        # create w
        w = np.zeros_like(conv)

"""
Propose a low‑score step‑function.

Algorithm
---------
1️⃣  Choose a problem size (≈ 8000).  If a previous construction exists,
    reuse its length (capped at 10 000).
2️⃣  Initialise a pool of starting points on the probability simplex
    (uniform, exponentials, power‑law, edge‑heavy, arcsine, Dirichlet,
    random perturbations of the current best).
3️⃣  Run a *continuation* L‑BFGS optimisation on a log‑sum‑exp smooth
    approximation of the max‑convolution.  The smoothing factor is
    decreased geometrically over many stages.
4️⃣  Refine the best vector with an **exponentiated‑gradient** projected
    sub‑gradient method that directly minimises the true max‑convolution
    (exact active‑set gradients, back‑tracking line‑search, monotone
    improvement).  This usually yields a few‑percent further reduction.
5️⃣  Finish with an aggressive hill‑climb that repeatedly moves a random
    fraction of mass between coordinates, accepting only improving moves.
6️⃣  Return the final sequence (the harness will evaluate the exact score).

The whole search respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        # sparse Dirichlet draw (α = 0.1) – many tiny entries
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Simple hill‑climb (only improving moves)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient projected sub‑gradient method (exact max)
# ----------------------------------------------------------------------


def expgrad_subgradient(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective,
    using a multiplicative (exponentiated‑gradient) update that stays on the
    probability simplex.  The method back‑tracks the step‑size to guarantee
    monotone improvement.
    """
    a = a_start.copy()
    best = a.copy()
    # current score = 2 * n * max(conv)
    conv = conv_fft(a, fft_len_conv, n)
    best_score = float(2.0 * n * conv.max())

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # --- convolution of current a ---
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # --- active set (indices where conv is close to the maximum) ---
        # tolerance scales with the spread of the convolution values
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        # --- build weight vector w (uniform on the active indices) ---
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # --- gradient of max(conv) w.r.t. a (averaged over active constraints) ---
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                      # ∂max/∂a

        # --- multiplicative (exponentiated‑gradient) update with back‑tracking ---
        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(8):          # up to 8 halvings of eta
            factor = np.exp(-eta * grad)
            a_new = a * factor
            s = a_new.sum()
            if s == 0.0:           # numerically disastrous – fall back to uniform
                a_new = np.full_like(a, 1.0 / n)
                s = 1.0
            a_new /= s
            new_score = fast_score(a_new, fft_len_conv, n)

            if new_score < best_score - 1e-13:
                # strict improvement: keep it
                best = a_new.copy()
                best_score = new_score
                a = a_new
                improved = True
                break
            elif new_score < cur_score:
                # improve over the current point even if not the best yet
                a = a_new
                improved = True
                break
            else:
                eta *= 0.5   # halve and try again

        if not improved:
            # No improvement even after halving – give up on this iteration
            # (but keep the current a for the next iteration)
            pass

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.

        else:
            n = 8000
    else:

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------

    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the continuation L‑BFGS stage.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=80))
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "dirichlet",
        "best",
    ]

    # Reserve time: 250 s for L‑BFGS, 500 s for exp‑gradient, rest for hill‑climb.
    l_bfgs_deadline = deadline - 600.0
    iteration = 0

    while time.time() < l_bfgs_deadline:
        mode = rng.choice(modes)

            best_y = y_opt.copy()
            print(f"[{iteration:04d}] L‑BFGS mode={mode} → improved fast score = {best_score:.6f}")

        # quick hill‑climb after each L‑BFGS run (≈1 s)

    # Refine with exponentiated‑gradient projected sub‑gradient method.
    # ------------------------------------------------------------------

            best_a,

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
