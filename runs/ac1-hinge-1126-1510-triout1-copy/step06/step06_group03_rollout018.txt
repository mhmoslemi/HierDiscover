"""
Propose a non‑negative step‑function minimising the evaluation score.

Algorithm
---------
1. Choose a problem size n (≈ 5 k–10 k) and work on the probability simplex
   (∑a_i = 1).  The score is homogeneous, so this does not affect the
   objective.
2. Initialise a candidate from several analytic shapes (arcsine, semicircle,
   uniform, random Gaussian, sparse binary mask, Gaussian bump) and, if
   available, from the previously discovered construction ``height_sequence_1``.
   Keep the best of these initialisations.
3. Refine the candidate with a continuation L‑BFGS optimisation of a smooth
   log‑sum‑exp approximation of the max‑convolution objective.  The smoothing
   parameter λ is decreased geometrically in ~30 stages; the number of L‑BFGS
   iterations per stage is modest (500–800) so this phase consumes only a
   fraction of the time budget.
4. Perform a long simulated‑annealing search.  In each iteration a small
   fraction of mass is moved from a source index to a destination index.
   Moves that improve the fast score are always accepted; worsening moves
   are accepted with probability exp(−Δ/T) where the temperature T decays
   exponentially from 0.1·score to 0.001·score.  Random moves and
   “targeted” moves (source taken from the current maximal‑convolution
   contributors) are mixed.
5. Finish with a short greedy hill‑climb and a targeted hill‑climb for final
   polishing.
6. Evaluate the final sequence with ``evaluate_sequence`` and return it.

The whole search respects the 1000 s budget; intermediate progress and scores
are printed.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    a is length n, fft_len >= 2*n-1.
    Returns real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # w = ∂smooth_max/∂conv
    w = exp_shift / Z

    # derivative of conv wrt a (reverse convolution trick)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # Chain rule through the softmax:
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule,
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # reduce iteration budget for very coarse λ, increase for fine λ
        maxit = maxiter_per_stage
        if lam < 1e-3:
            maxit = maxiter_per_stage * 2
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxit, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def arcsine_init(n: int, alpha: float = 0.75) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a log‑softmax start vector for the given mode."""
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)

    y0 = np.log(a0 + 1e-12)
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that only accepts improving moves."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that preferentially moves mass away from current
    maximal‑convolution contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Long simulated‑annealing search.
    Moves a random fraction of mass from a source index to a destination.
    Random and targeted moves are mixed; worsening moves are accepted with
    probability exp(-Δ/T) where T decays exponentially.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule: start at 0.1·score, end at 0.001·score
    T_start = max(score_start * 0.10, 1e-6)
    T_end = max(score_start * 0.001, 1e-9)

    start_sa = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # linear/exponential temperature decay
        elapsed = time.time() - start_sa
        progress = min(elapsed / (deadline - start_sa), 1.0)
        T = T_start * (T_end / T_start) ** progress

        # Choose move type
        if rng.random() < 0.70:
            # random move: source weighted by current mass
            src = rng.choice(n, p=cur_a)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            delta = cur_a[src] * rng.random() * move_frac
            if delta <= 0.0:
                continue
            cur_a[src] -= delta
            cur_a[dst] += delta
            new_score = fast_score(cur_a, fft_len, n)

            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # Metropolis acceptance
                if T > 0 and rng.random() < math.exp(-(new_score - cur_score) / T):
                    accept = True

            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()
            else:
                # revert
                cur_a[src] += delta
                cur_a[dst] -= delta
        else:
            # targeted move: source from the current maximal‑convolution contributors
            conv = conv_fft(cur_a, fft_len, n)
            m = int(np.argmax(conv))
            idx = np.arange(n)
            j = m - idx
            mask = (j >= 0) & (j < n)
            contributors = idx[mask]
            if contributors.size == 0:
                continue
            src = rng.choice(contributors)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            delta = cur_a[src] * rng.random() * move_frac
            if delta <= 0.0:
                continue
            cur_a[src] -= delta
            cur_a[dst] += delta
            new_score = fast_score(cur_a, fft_len, n)

            accept = False
            if new_score < cur_score:
                accept = True
            else:
                if T > 0 and rng.random() < math.exp(-(new_score - cur_score) / T):
                    accept = True

            if accept:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = cur_a.copy()
            else:
                cur_a[src] += delta
                cur_a[dst] -= delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initialise best candidate on the simplex (sum = 1)
    # ------------------------------------------------------------------
    # start from the previous construction if its length matches n
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n and tmp.sum() > 0:
            best_a = tmp / tmp.sum()
        else:
            best_a = uniform_init(n)
    else:
        best_a = uniform_init(n)

    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] initial fast score = {best_score:.6f}")

    # try a handful of analytic shapes and keep the best
    init_modes = ["arcsine", "semicircle", "uniform", "random", "binary", "bump"]
    for mode in init_modes:
        a0 = globals()[f"{mode}_init"](n) if mode in ["arcsine", "semicircle", "uniform"] else \
            generate_initial_y(mode, n, rng)  # reuse generator for the stochastic ones
        # if we used generate_initial_y we get log‑softmax parameters; convert back
        if a0.ndim == 2:  # shouldn't happen, guard
            a0 = a0.ravel()
        s0 = fast_score(a0, fft_len, n)
        if s0 < best_score:
            best_a = a0.copy()
            best_score = s0
            print(f"[init] better shape '{mode}' fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    lbfgs_deadline = start_time + 0.30 * budget_s
    y = np.log(best_a + 1e-12)
    lam_schedule = np.geomspace(0.5, 1e-9, num=30)

    # run L‑BFGS but stop early if we run out of time
    for lam in lam_schedule:
        if time.time() >= lbfgs_deadline:
            break
        maxit = 500
        if lam < 1e-3:
            maxit = 800
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxit, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a_opt = softmax(y)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[lbfgs] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing (major part of the remaining budget)
    # ------------------------------------------------------------------
    sa_deadline = deadline - 30.0  # keep ~30 s for final polishing
    print("[sa] starting simulated annealing")
    best_a, best_score = simulated_annealing(
        best_a,
        best_score,
        fft_len,
        n,
        sa_deadline,
        rng,
        move_frac=0.30,
    )
    print(f"[sa] finished, best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            final_deadline,
            max_iter=15000,
            max_move=0.35,
        )
    if time.time() < deadline:
        print("[final] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=8000,
            max_move=0.25,
        )
    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
