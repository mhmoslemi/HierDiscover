def shape_edge_fraction(n, edge_frac, mass_frac):
    total_edge_len = max(2, int(round(edge_frac * n)))
    if total_edge_len > n:
        total_edge_len = n
    # ensure even for symmetrical edges; split edges equally
    left_len = total_edge_len // 2
    right_len = total_edge_len - left_len
    interior_len = n - total_edge_len
    if interior_len <=0:
        interior_len = 0
    # allocate mass: edge_mass = mass_frac, interior_mass = 1 - mass_frac
    edge_weight = mass_frac / total_edge_len if total_edge_len>0 else 0.0
    interior_weight = (1.0 - mass_frac) / interior_len if interior_len>0 else 0.0
    a = np.full(n, interior_weight, dtype=float)
    a[:left_len] = edge_weight
    a[-right_len:] = edge_weight
    # renormalize to sum=1 in case of rounding errors
    tot = a.sum()
    if tot <= 0:
        a = np.full(n, 1.0/n)
    else:
        a /= tot
    return a

"""
A fast, multi‑stage optimizer for the step‑function minimisation problem.

The algorithm consists of three phases:

1. **Coarse parametric sweep** – generates a few hundred simple families
   (edge‑heavy, exponential, power‑law, symmetric variants) and keeps the
   best one according to the fast surrogate 2·n·max(conv).

2. **Global refinement** – many random restarts of a smooth‑max L‑BFGS
   optimisation (continuation on a λ‑schedule).  The unconstrained
   parameters are the log‑soft‑max variables; the soft‑max guarantees a
   non‑negative simplex vector.

3. **Local polishing** – targeted and random hill‑climbs together with a
   projected sub‑gradient descent on the true max‑convolution objective.
   The whole search stops a short time before the allotted budget.

The final sequence is stored in the global variable ``height_sequence_1``
and returned.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``:
    returns 2·n·max(conv(a,a)) computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # convolution a*a (full) -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max --------------------------------------------------------------
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient wrt conv -------------------------------------------------------
    w = exp_shift / Z

    # propagate through convolution (using FFT) -------------------------------
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient wrt a -----------------------------------------------------------
    # (already multiplied by 2·n)
    grad_a = 2.0 * n * g_vec

    # chain rule through soft‑max ---------------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors,
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ------------------------------------------------------------
# Random initialisations for L‑BFGS
# ------------------------------------------------------------


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point `y` (log‑soft‑max parameters).
    A variety of heuristics are mixed to keep diversity.
    """
    mode = rng.choice(
        ["random", "exp", "power", "edge", "sym_exp", "sym_power", "best"]
    )
    if mode == "random":
        return rng.standard_normal(n)
    if mode == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a0 = np.exp(-rate * np.arange(n, dtype=float))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    if mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    if mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    if mode == "sym_exp":
        rate = rng.uniform(0.0005, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    if mode == "sym_power":
        p = rng.uniform(0.3, 2.0)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
        a0 = d ** (-p)
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    # perturb the current best solution
    return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


# ------------------------------------------------------------
# Simple families of candidate shapes (coarse sweep)
# ------------------------------------------------------------


def shape_edge_fraction(n: int, edge_frac: float, mass_frac: float) -> np.ndarray:
    """
    Two‑sided edge‑heavy shape.
    ``edge_frac``  – proportion of total indices belonging to the two edges.
    ``mass_frac``  – proportion of total mass placed on those edge indices.
    """
    total_edge = max(2, int(round(edge_frac * n)))
    left = total_edge // 2
    right = total_edge - left
    interior = n - total_edge
    if interior <= 0:
        interior = 0
    edge_weight = mass_frac / total_edge if total_edge > 0 else 0.0
    interior_weight = (1.0 - mass_frac) / interior if interior > 0 else 0.0
    a = np.full(n, interior_weight, dtype=float)
    a[:left] = edge_weight
    a[-right:] = edge_weight
    a /= a.sum()
    return a


def shape_exponential_left(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def shape_exponential_sym(n: int, rate: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-rate * d)
    a /= a.sum()
    return a


def shape_power_left(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def shape_power_sym(n: int, p: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
    a = d ** (-p)
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Hill‑climbing utilities
# ------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contributions for this shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # choose a destination with a small partner under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution_subgrad(a: np.ndarray, n: int, fft_len: int):
    """
    Compute the maximal auto‑convolution value and a sub‑gradient of the
    (non‑smoothed) max‑operator.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to shift s ∈ [-(n‑1), … , n‑1]
    if k_max > n - 1:
        s = 2 * n - 2 - k_max
    else:
        s = k_max

    grad = np.zeros_like(a)
    if s == 0:
        grad = 2.0 * a
    else:
        grad[: n - s] += a[s:]
        grad[s:] += a[: n - s]
    return max_c, grad


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation score.
    The function respects a time budget (seconds) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # -----------------------------------------------------------------
    # Problem size
    # -----------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # ensure simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 1️⃣ Coarse parametric sweep
    # -----------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # edge‑heavy families
    for ef in np.linspace(0.02, 0.20, 9):          # edge fraction
        for mf in np.linspace(0.10, 0.60, 11):    # mass on edges
            cand = shape_edge_fraction(n, ef, mf)
            candidates.append(cand)

    # left‑exponential
    for rate in np.linspace(0.0005, 0.020, 15):
        candidates.append(shape_exponential_left(n, rate))

    # symmetric‑exponential
    for rate in np.linspace(0.0005, 0.020, 15):
        candidates.append(shape_exponential_sym(n, rate))

    # left‑power‑law
    for p in np.linspace(0.1, 2.0, 15):
        candidates.append(shape_power_left(n, p))

    # symmetric‑power‑law
    for p in np.linspace(0.1, 2.0, 15):
        candidates.append(shape_power_sym(n, p))

    # evaluate all candidates (fast surrogate)
    for cand in candidates:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score - 1e-12:
            best_a = cand
            best_score = sc

    print(f"[sweep] best fast_score after param sweep = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2️⃣ Global optimisation (L‑BFGS with λ‑schedule)
    # -----------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=40)   # finer schedule
    maxiter_per_stage = 600

    iteration = 0
    while time.time() < deadline - 5.0:
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score - 1e-12:
            best_a = a_opt
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved fast_score = {best_score:.6f}")

        # -------------------------------------------------------------
        # Targeted hill‑climb (short burst)
        # -------------------------------------------------------------
        t_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1200,
            move_frac=0.30,
        )

        # -------------------------------------------------------------
        # Random hill‑climb (short burst)
        # -------------------------------------------------------------
        r_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=4000,
            max_move_frac=0.20,
        )

        iteration += 1

    # -----------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient descent on the true objective
    # -----------------------------------------------------------------
    step0 = 0.45
    sub_iter = 0
    eps = 1e-12
    while time.time() < deadline - 2.0:
        sub_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)

        step = step0 / math.sqrt(sub_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)
        else:
            step0 *= 0.92

        if sub_iter % 250 == 0:
            print(f"[subgrad] iter={sub_iter} best fast_score={best_score:.6f}")

    # -----------------------------------------------------------------
    # Final polishing (unrestricted hill‑climb)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=15000,
            max_move_frac=0.25,
        )

    # -----------------------------------------------------------------
    # Return the final sequence
    # -----------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
