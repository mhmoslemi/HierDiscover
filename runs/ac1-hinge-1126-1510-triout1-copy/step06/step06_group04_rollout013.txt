"""
Propose a step‑function (list of non‑negative floats) with a low
evaluate_sequence() score.

The objective, after the sum‑normalisation ∑a_i = 1, is
    score = 2·n·max(conv(a,a)) .
We minimise the L∞ norm of the self‑convolution.
The algorithm proceeds in three phases

1. **Multi‑restart smoothed L‑BFGS** – the (smooth) log‑sum‑exp
   approximation of max allows a gradient‑based optimisation on the
   unconstrained logits `y` (softmax(y) = a).  A geometric schedule of
   temperatures drives the solution from a very smooth landscape to the
   true L∞ objective.

2. **Simulated‑annealing hill‑climb** – starting from the best vector
   found, random mass‑transfer moves are performed.  Moves that lower
   the true objective are always accepted, uphill moves are accepted
   with a temperature‑dependent probability, allowing the search to
   escape shallow local minima.

3. **Fine L‑BFGS refinement** – a final continuation with a near‑zero
   temperature tightens the solution.

All convolutions are performed with FFTs (O(n log n)).  The final
sequence is returned as a plain list; `evaluate_sequence` will be
called by the harness.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    a : length‑n array, assumed real.
    Returns the first (2·n‑1) real entries.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when ∑a = 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, ∑x = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_grad: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where

        f = 2·n·smooth_max(conv(a,a))

    with a = softmax(y) (so ∑a = 1).  The smooth max is a log‑sum‑exp
    with temperature λ = lam_factor·max(conv).
    """
    a = softmax(y)

    # --- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12          # temperature
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient wrt conv ------------------------------------------------
    w = exp_shift / Z               # ∂smooth_max/∂conv_i
    # derivative of conv[k] w.r.t. a_i is a_{k-i} + a_{i-k}
    # This is a (cross‑) convolution of a with w (with appropriate reversal)
    fw = np.fft.fft(w, fft_len_grad)
    fa_rev = np.fft.fft(a[::-1], fft_len_grad)
    grad_conv = np.real(np.fft.ifft(fa_rev * fw))

    # The middle n entries correspond to ∂conv/∂a_i
    g_vec = grad_conv[(n - 1) : (n - 1 + n)]

    # Chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


def schedule_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_grad: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: start from y0, run a series of minimisations,
    each with a smaller smoothing temperature.
    """
    y = y0.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_grad, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a logits vector y for the given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # heavy mass near the two ends, light in the middle
        w_edge = np.sqrt(n / 2.0) / n
        w_mid = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, w_mid)
        m = max(1, int(n * rng.uniform(0.01, 0.04)))
        a0[:m] = w_edge
        a0[-m:] = w_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy construction of a small Sidon‑like set
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if cand + c in sums:
                    conflict = True
                    break
            if not conflict and (2 * cand not in sums):
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
        k = max(1, len(chosen))
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.25
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def simulated_annealing(

    """
    Simple annealing that repeatedly transfers a small fraction of mass
    from a randomly selected source index to a target index.
    The true objective (2·n·max(conv)) is evaluated after each trial.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    # linear temperature schedule
    total = max(1, deadline - time.time())
    while time.time() < deadline:
        # compute temperature according to remaining fraction of budget
        remain = max(0.0, deadline - time.time())
        t = (remain / total) * (initial_temp - final_temp) + final_temp
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a fraction of source mass
        delta = a[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        cur_score = fast_score(a, fft_len, n)
        delta_score = cur_score - best_score
        accept = False
        if cur_score < best_score:
            best, best_score = a.copy(), cur_score
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-delta_score / max(t, 1e-12)):
                accept = True
        if not accept:
            # revert the move
            a[i] += delta
            a[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluate_sequence() score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (n)
    # --------------------------------------------------------------
    default_n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 9000)

    n = default_n

    # FFT lengths (pad to next power‑of‑two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_grad = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise the current best (probability vector)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp.copy()
            # ensure non‑negative and normalise
            best_a = np.maximum(best_a, 0.0)
            if best_a.sum() < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a /= best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – multi‑restart smoothed L‑BFGS (≈ 45 % of the budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=30))
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "binary",
        "bump",
        "harmonic",
        "best",
    ]
    phase1_deadline = start_time + 0.45 * budget_s
    iter_counter = 0
    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.025, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_grad,
            lam_schedule,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[lbfgs {iter_counter:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # short greedy hill‑climb (deterministic improvement)
        best_a, best_score = simulated_annealing(
            best_a,
            fft_len_conv,
            n,
            deadline=min(deadline, time.time() + 1.5),
            initial_temp=0.0,          # purely greedy for this quick pass
            final_temp=0.0,
            max_move_frac=0.10,
            rng=rng,
        )
        best_y = np.log(best_a + 1e-12)
        iter_counter += 1

    # --------------------------------------------------------------
    # Phase 2 – long annealing (≈ 40 % of the budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.85 * budget_s
    print("[anneal] starting long simulated‑annealing phase")
    best_a, best_score = simulated_annealing(
        best_a,
        fft_len_conv,
        n,
        deadline=phase2_deadline,
        initial_temp=0.04,
        final_temp=1e-5,
        max_move_frac=0.20,
        rng=rng,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[anneal] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final fine L‑BFGS (very low smoothing, if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("[fine] final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=25))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_grad,
            fine_lam_schedule,
            maxiter_per_stage=800,
        )
        a_fine = softmax(y_fine)
        sc_fine = fast_score(a_fine, fft_len_conv, n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine
            best_y = y_fine
            print(f"[fine] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Convert to the format required by evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluate_sequence() returned score = {final_val:.6f}")

    return final_seq
