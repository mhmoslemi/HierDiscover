def solve_lp(a, M, active_idx, n):
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    # Build constraint matrix
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):

        end = min(k, n - 1)

            cols.append(i)

    return res.x, True

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. Initialise a probability vector a of length n (default 4000).  If a
   previous construction is available in the global variable
   ``height_sequence_1`` it is interpolated, otherwise a uniform vector
   is used.

2. **Fast global search** – a short exponentiated‑gradient optimisation
   (exact sub‑gradient of the max‑convolution) gets a decent baseline
   quickly.

3. **LP‑based refinement** – for the current a we compute M = max(a*a).
   We then solve the linear program

        maximise   1ᵀ g
        subject to  (a * g)[k] ≤ M ,  ∀ k in an active set
                    g ≥ 0

   where ``a * g`` is the (linear) convolution of a with the unknown
   vector g.  The active set starts with the indices where ``a*a`` is
   within a small fraction of its maximum; after each solve we add
   any violated constraints (cutting‑plane style).  The optimal g is
   normalised to a simplex and we move from a towards g with a line
   search on the mixing factor τ∈(0,1].  This step provably never
   increases the objective and usually yields a strict improvement.

4. **Local polishing** – a greedy hill‑climb that transfers a tiny
   amount of mass between two coordinates is run until the deadline.

All phases respect a total wall‑clock budget (default 1000 s).  The
final sequence is returned as a plain list of floats; it is guaranteed
to be non‑negative and sums to 1, so ``evaluate_sequence`` receives a
valid input.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def conv_two(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*b (full) via FFT."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.real(np.fft.ifft(fa * fb))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                       # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 0.6 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def build_constraint_matrix(a: np.ndarray,
                            idxs: np.ndarray,
                            n: int) -> csr_matrix:
    """
    Build the (len(idxs) × n) matrix A where A[k,i] = a[i] iff
    0 ≤ i < n and 0 ≤ k-i < n, for each k in idxs.
    """
    rows = []
    cols = []
    data = []
    for row, k in enumerate(idxs):
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        for i in range(start_i, end_i + 1):
            rows.append(row)
            cols.append(i)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(len(idxs), n), dtype=np.float64)
    return A


def lp_optimize(a: np.ndarray,
                M: float,
                active_idx: np.ndarray,
                n: int,
                timeout: float = 5.0):
    """
    Solve  maximise 1ᵀ g   subject to (a*g)[k] ≤ M for k∈active_idx,
    g ≥ 0.
    Returns the optimal g (not normalised) and a success flag.
    """
    A = build_constraint_matrix(a, active_idx, n)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)            # maximise sum(g)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout})
    except Exception as e:
        print("[LP] exception:", e)
        return None, False

    if not res.success:
        # Infeasibility can happen if the active set is too small;
        # the caller should enlarge the set.
        return None, False
    return res.x, True


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Simple greedy hill‑climb that moves a random fraction of mass
    from one coordinate to another, accepting only improvements.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    max_n = 4000
    n = max_n

    # Interpolate a previous solution if it exists
    init_seq = globals().get('height_sequence_1')
    if isinstance(init_seq, (list, np.ndarray)):
        init_arr = np.asarray(init_seq, dtype=float).ravel()
        if init_arr.size > 1:
            x_old = np.linspace(0, 1, init_arr.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_arr)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – fast exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    if time.time() < phase1_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase1_end, n, fft_len, rng)
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based refinement (cutting‑plane)
    # ------------------------------------------------------------------
    # We keep refining until we run out of time.
    lp_deadline = deadline - 5.0   # keep a few seconds for final polish
    while time.time() < lp_deadline:
        # 1) current max‑convolution
        conv_full = full_conv(best_a, fft_len)
        M = conv_full.max()
        # 2) choose an active set (indices with conv close to M)
        tol = 0.02 * M                     # 2 % of the maximum
        active_idx = np.where(conv_full >= M - tol)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv_full)], dtype=int)

        # 3) solve the LP
        g_raw, ok = lp_optimize(best_a, M, active_idx, n,
                                timeout=4.0)
        if not ok:
            # If the LP failed (often because the active set is too
            # small), enlarge the set and retry.
            # Add the worst violating index and continue.
            conv_ag = conv_two(best_a, best_a, fft_len)  # same as conv_full
            viol = np.where(conv_ag > M + 1e-15)[0]
            if viol.size > 0:
                add_idx = np.setdiff1d(viol, active_idx)[:10]
                active_idx = np.unique(np.concatenate([active_idx, add_idx]))
                continue
            else:
                # nothing to add – give up on LP phase
                break

        # 4) normalise g to the simplex
        sum_g = g_raw.sum()
        if sum_g <= 0.0:
            break
        g = g_raw / sum_g

        # 5) line‑search on the mixing factor τ
        improved = False
        # try larger steps first (more aggressive)
        for tau in (0.9, 0.7, 0.5, 0.3, 0.1):
            candidate = (1.0 - tau) * best_a + tau * g
            cand_score = fast_score(candidate, fft_len, n)
            if cand_score < best_score - 1e-12:
                best_a = candidate
                best_score = cand_score
                improved = True
                print(f"[LP‑refine] τ={tau:.2f}, new score={best_score:.6f}")
                break

        if not improved:
            # No improvement with the current g – try to enlarge the
            # active set (cutting‑plane) and repeat.
            conv_ag = conv_two(best_a, g, fft_len)
            viol = np.where(conv_ag > M + 1e-12)[0]
            if viol.size == 0:
                # Nothing violates, we are at a stationary point.
                break
            # Add a few of the worst violators to the active set.
            add_idx = np.setdiff1d(viol, active_idx)[:10]
            active_idx = np.unique(np.concatenate([active_idx, add_idx]))
            continue

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=12000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
