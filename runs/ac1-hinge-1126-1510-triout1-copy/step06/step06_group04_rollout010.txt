def smoothed_descent(a_start, fft_len, n, deadline, lam_factors, eta0=0.2, max_iters=20000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    for lam_factor in lam_factors:
        # inner loop for that factor
        while time.time() < deadline and it < max_iters:
            it += 1
            conv = conv_fft(a, fft_len, n)
            max_c = conv.max()
            lam = max_c * lam_factor + 1e-12
            shift = conv - max_c
            exp_shift = np.exp(shift / lam)
            Z = exp_shift.sum()
            w = exp_shift / Z
            # gradient via reversed convolution (like before)
            fa_rev = np.fft.fft(a[::-1], fft_len)
            fw = np.fft.fft(w, fft_len)
            conv_rev = np.real(np.fft.ifft(fa_rev * fw))
            grad = conv_rev[(n-1):(n-1+n)]
            # step size
            eta = eta0 / np.sqrt(it)
            a = a - eta * grad
            a = project_onto_simplex(a)
            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            # optional occasional random perturb: every 1000 steps.
            if it % 1000 == 0:
                # small random jitter to escape local minima
                jitter = np.random.default_rng().normal(scale=0.001, size=n)
                a = np.maximum(a + jitter, 0)
                a = project_onto_simplex(a)
    return best_a, best_score

"""
Propose a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm Overview
------------------
* **Problem size** – use a fairly large support (≈ 10 000) to gain extra
  freedom, but stay within the time budget.
* **Parameterisation** – work on the probability simplex (∑a=1).  The
  evaluation function is scale‑invariant, so we can safely normalise.
* **Phase 1 – Multi‑restart L‑BFGS** with a log‑sum‑exp (smooth max) objective.
  Random initialisations (Gaussian, exponential, power‑law, edge‑heavy,
  Sidon, binary mask, Gaussian bump, harmonic) and occasional perturbations
  of the current best are tried.
* **Phase 2 – Smoothed gradient descent** (continuous approximation of the
  max‑convolution) with a geometric schedule for the smoothing parameter.
  A simple projected gradient step (with a diminishing step‑size) is applied
  many times; occasional jitter helps to escape shallow plateaus.
* **Phase 3 – Projected sub‑gradient descent** on the true objective
  (max(conv)) for fine tuning.
* **Phase 4 – Aggressive hill‑climb** moving a random fraction of mass from
  one location to another while keeping the simplex constraints.
* The best sequence found is returned; the harness will call
  ``evaluate_sequence`` on it.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = any((candidate + c) in sums for c in chosen) \
                       or (2 * candidate in sums)
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (equivalently minimise max(conv) with sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))                       # index of maximal entry
        # Sub‑gradient of that entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def smoothed_gradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    lam_factors: list[float],
    eta0: float = 0.20,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp smooth approximation of
    max(conv).  ``lam_factors`` is a decreasing sequence; each factor is
    multiplied by the current max to obtain the smoothing λ.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    rng = np.random.default_rng()
    it = 0

    for lam_factor in lam_factors:
        while time.time() < deadline and it < max_iters:
            it += 1
            conv = conv_fft(a, fft_len, n)
            max_c = conv.max()
            lam = max_c * lam_factor + 1e-12
            shift = conv - max_c
            exp_shift = np.exp(shift / lam)
            Z = exp_shift.sum()
            w = exp_shift / Z

            # Gradient of smooth_max w.r.t. a (via reverse convolution)
            fa_rev = np.fft.fft(a[::-1], fft_len)
            fw = np.fft.fft(w, fft_len)
            conv_rev = np.real(np.fft.ifft(fa_rev * fw))
            grad = conv_rev[(n - 1) : (n - 1 + n)]

            # Diminishing step size
            eta = eta0 / np.sqrt(it)
            a = a - eta * grad

            # occasional jitter to escape plateaus
            if it % 2000 == 0:
                a += rng.normal(scale=0.001, size=n)

            a = project_onto_simplex(a)

            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    return best_a, best_score

                       time.time() + 0.20 * (budget_s - (time.time() - start_time)))
    print("[subgrad] start projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len,
        n,
        sub_deadline,
        eta0=0.30,
        max_iters=300_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ---- Phase 4: aggressive hill‑climb -------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=30_000,
            max_move=0.35,
        )
        best_y = np.log(best_a + 1e-12)

    # ---- return the final sequence ------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
