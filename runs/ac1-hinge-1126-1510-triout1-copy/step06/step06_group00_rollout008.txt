"""
Propose a step‑function sequence with a low evaluation score.

The search mixes several ideas that complement the previous state‑of‑the‑art
pipeline:

1. **Parametric seeding** – a small grid of Beta‑distribution shapes (including the
   uniform case) is evaluated first; the best among them is used as a warm‑start.
2. **Continuation L‑BFGS** on a log‑sum‑exp (smooth) approximation of the max
   convolution.  The smoothing parameter schedule is extended down to a very
   small value to get close to the true L∞ objective.
3. **Flatten‑loss gradient descent** – we minimise the variance of the
   self‑convolution (the squared deviation from the average).  This drives the
   convolution towards a flat profile and therefore reduces its supremum.
4. **Exact exponentiated‑gradient descent** on the hard max‑convolution.
5. **Short simulated‑annealing** + **greedy hill‑climb** for final fine‑tuning.

All stages respect the given time budget (default 1000 s) and keep the
sequence non‑negative and normalised to sum‑one (the evaluation function is
scale‑invariant).  The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when a is normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best score={best_score:.6f} (lam={lam:.5f})")
    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (several families mixed)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule
        eta = 0.6 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[exp‑grad] iteration {it} new best score={best_score:.6f}")

        it += 1
    return best_a, best_score


def flatten_variance_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len_conv: int,
                              fft_len_g: int):
    """
    Gradient descent on the squared deviation
        L(a) = Σ_k ( (a*a)[k] - μ )² ,  μ = 1/(2n‑1)
    This encourages the auto‑convolution to become flat,
    which in turn lowers its maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    avg = 1.0 / (2 * n - 1)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        diff = conv - avg

        a_rev = a[::-1]
        fdiff = np.fft.fft(diff, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fdiff))

        grad = 4.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple diminishing step size
        step = 0.2 / np.sqrt(it + 1.0)

        a_candidate = a - step * grad
        a_candidate = simplex_projection(a_candidate)

        score = fast_score(a_candidate, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            print(f"[flatten] iter {it} new best score={best_score:.6f}")

        a = a_candidate
        it += 1
    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing that moves a random fraction of mass between two
    coordinates, accepting occasional uphill moves.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.12
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy hill‑climb that moves a random fraction of the mass from one
    coordinate to another and accepts only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.12
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Helper: parametric Beta family seeding
# ----------------------------------------------------------------------


def beta_family_candidate(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) density on a regular grid, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n  # midpoints in (0,1)
    # avoid exact 0/1 to keep finite values
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    a = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initialisation
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n

    # if a previous construction is present, interpolate it to the target length
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Timing setup
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.3               # tiny safety margin

    phase1_end = start_time + 0.12 * budget_s           # beta‑grid seeding
    phase2_end = start_time + 0.55 * budget_s           # L‑BFGS (heavy)
    phase3_end = start_time + 0.80 * budget_s           # variance flattening
    phase4_end = start_time + 0.92 * budget_s           # exponentiated‑gradient
    phase5_end = start_time + 0.98 * budget_s           # annealing
    phase6_end = deadline                               # final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – parametric Beta‑family exploration
    # --------------------------------------------------------------
    alphas = [0.4, 0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0]
    mixes = np.linspace(0, 1, 11)   # λ for mixture with uniform
    while time.time() < phase1_end:
        for alpha in alphas:
            cand_beta = beta_family_candidate(n, alpha)
            for lam in mixes:
                cand = lam * cand_beta + (1 - lam) * (1.0 / n)
                cand /= cand.sum()
                score = fast_score(cand, fft_len_conv, n)
                if score < best_score:
                    best_score = score
                    best_a = cand.copy()
                    print(f"[beta] α={alpha:.2f} λ={lam:.2f} new best={best_score:.6f}")
        # add a few purely random starts as well
        cand_rand = random_initial(n, rng)
        score = fast_score(cand_rand, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = cand_rand.copy()
            print(f"[rand] new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – continuation L‑BFGS on smooth max‑conv
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001,
                    0.00005, 0.00002]
    maxiter_per_lam = 1500

    while time.time() < phase2_end:
        # Occasionally perturb the current best to escape shallow minima
        if best_score < np.inf and rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase2_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] improved best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – flattening the convolution (variance loss)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = flatten_variance_optimize(best_a,
                                                       phase3_end,
                                                       n,
                                                       fft_len_conv,
                                                       fft_len_g)
        print(f"[flatten] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – exact exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase4_end,
                                                   n,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   rng)
        print(f"[exp‑grad] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – short simulated annealing
    # --------------------------------------------------------------
    if time.time() < phase5_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase5_end)
        print(f"[anneal] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – final greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase6_end:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        phase6_end,
                                        max_iter=20000)
        print(f"[hill‑climb] final score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
