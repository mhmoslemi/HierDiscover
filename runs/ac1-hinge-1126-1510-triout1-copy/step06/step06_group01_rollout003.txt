"""
Search for a low‑evaluation step sequence.

The algorithm proceeds in three stages:

1. **Initialisation** – generate a few random candidates (or reuse the
   provided global `height_sequence_1`) and keep the best one according
   to the fast L‑∞ convolution score.

2. **LP refinement** – given the current sequence f, solve a linear
   program that maximises the total mass of a partner sequence g under
   the constraint   f * g ≤ max(f * f)  (point‑wise).  After normalising g,
   mix it with f (f←(1–t)f+tg) and repeat.  This is the “alternating
   linear‑program” technique described in the problem statement and
   drives the max‑convolution down.

3. **Greedy hill‑climb** – a final local search that repeatedly moves a
   small amount of mass between two coordinates while strictly improving
   the fast score.

All steps respect the non‑negativity and normalisation constraints.
The code is written to run within the 1000 s budget (the remaining time
is checked at each stage).  The returned list is a valid input for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised (∑a=1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)
    return a


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 8000,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb: relocate a small random fraction of mass."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10  # move up to 10 % of mass at i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        # no renormalisation needed – total mass stays the same
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def build_conv_pattern(n: int):
    """
    Pre‑compute the index pattern of the linear convolution matrix.
    For each pair (i,j) with 0≤i,j<n we have a contribution
        row = i + j, column = j, value = f[i].
    Returns three 1‑D integer arrays (row, col, i_idx) of length n**2.
    """
    I = np.arange(n, dtype=np.int32)
    row = (I[:, None] + I[None, :]).astype(np.int32).ravel()
    col = np.tile(I, n).astype(np.int32)
    i_idx = np.repeat(I, n).astype(np.int32)
    return row, col, i_idx


def lp_refine(
    a: np.ndarray,
    row_idx: np.ndarray,
    col_idx: np.ndarray,
    i_idx: np.ndarray,
    deadline: float,
    fft_len_conv: int,
) -> tuple[np.ndarray, float]:
    """
    Iteratively improve ``a`` by solving the LP:

        maximise   sum(g)
        subject to g ≥ 0,
                   f * g ≤ M  (point‑wise)

    where M = max(f * f) and f is the current sequence.
    After each solve, g is normalised to unit sum and mixed with f.
    """
    n = len(a)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    t = 0.3                     # initial mixing weight
    it = 0
    while time.time() < deadline:
        # 1️⃣ current bound M
        fa = np.fft.fft(best_a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # 2️⃣ build the sparse constraint matrix  A_ub @ g ≤ M
        data = best_a[i_idx]                     # f[i] for each (i,j) pair
        A_ub = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)

        # 3️⃣ linear programme (maximise sum(g) ⇒ minimise –sum(g))
        c = -np.ones(n)                           # minimise -∑g_j
        bounds = [(0, None)] * n

        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no increase of mass possible – stop
            print("[LP] sum(g) ≤ 1, terminating refinement.")
            break

        # 4️⃣ normalise g and mix
        g_norm = g / sum_g
        new_a = (1.0 - t) * best_a + t * g_norm
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()

        new_score = fast_score(new_a, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a
            print(f"[LP] iter {it:02d}, t={t:.3f}, score={new_score:.6f}")
        else:
            # mixing did not improve – shrink the step size
            t *= 0.9
            if t < 1e-3:
                print("[LP] mixing weight too small, stopping.")
                break

        it += 1
        if it >= 30:        # safety cap
            break
    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The method combines random initialisation, LP‑based refinement and a
    final greedy hill‑climb.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    n = 4000                     # keep the size that was used in the reference run
    fft_len_conv = next_pow2(2 * n - 1)

    # Start from the supplied construction if it exists
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # Sample a few extra random starts and keep the best one
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for _ in range(5):
        cand = random_initial(n, rng)
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand
    print(f"[init] n={n}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # a tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming refinement (major part of the budget)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 5.0                    # reserve a few seconds for hill‑climb
    row_idx, col_idx, i_idx = build_conv_pattern(n)

    best_a, best_score = lp_refine(
        best_a, row_idx, col_idx, i_idx, lp_deadline, fft_len_conv
    )
    print(f"[LP‑final] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=12000
        )
        print(f"[hill‑climb] final_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the candidate
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluation error:", e)

    return final_seq
