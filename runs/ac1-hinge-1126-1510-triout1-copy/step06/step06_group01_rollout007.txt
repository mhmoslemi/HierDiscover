def shrink_max(a, deadline):
    a = a.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    # compute initial gradient and max
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute subgradient g:
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1:n-1+n]  # size n

        # identify index with highest gradient
        i = np.argmax(g)
        j = np.argmin(g)
        # ensure i != j
        if i == j:
            continue
        # Move small fraction of mass from i to j
        delta = a[i] * 0.01  # 1% of mass
        if delta < 1e-12:
            # if mass too small, break
            break
        a[i] -= delta
        a[j] += delta
        # renormalize to sum=1 but sum unchanged due to transfer
        # compute new score
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        else:
            # revert move if not improving
            a[i] += delta
            a[j] -= delta

    return best_a, best_score

"""
Propose a step sequence with a low evaluate_sequence score.

The algorithm proceeds in several stages within the 1000 s budget:

1. **Heavy L‑BFGS continuation** – many random‑restart L‑BFGS runs on a
   smoothed log‑sum‑exp surrogate of the max‑convolution.  The smoothing
   factor (lam) is decreased geometrically; each stage starts from the
   result of the previous one.

2. **Projected sub‑gradient descent with momentum** – optimises the true
   max‑convolution using the sub‑gradient obtained from the active (max‑tight)
   indices.

3. **Targeted mass‑transfer optimisation** – repeatedly moves a small
   fraction of mass from the index with the largest sub‑gradient to the
   index with the smallest sub‑gradient; only strictly improving moves are kept.

4. **Simulated annealing** – stochastic mass swaps that can accept
   uphill moves with a slowly decreasing temperature.

5. **Greedy hill‑climb** – final deterministic local moves that relocate
   a small random fraction of mass while strictly improving the objective.

All convolutions are performed via FFTs (O(n log n)) and the sequence is kept
on the probability simplex (non‑negative, sum = 1), which makes the objective
scale‑invariant.  The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised (∑a=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    # initialise logits
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # Very rare; skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def targeted_transfer_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Deterministically move a small fraction of mass from the index with the
    largest sub‑gradient to the index with the smallest sub‑gradient.
    Only strictly improving moves are kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # convolution a*a and active mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_max = np.argmax(g)
        i_min = np.argmin(g)

        if i_max == i_min:
            break

        # step size shrinks with the iteration count
        step_frac = 0.015 / np.sqrt(it + 1.0)   # 1‑2 % of the source mass
        delta = a[i_max] * step_frac
        if delta <= 0.0:
            break

        # transfer
        a[i_max] -= delta
        a[i_min] += delta

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert if no improvement
            a[i_max] += delta
            a[i_min] -= delta

        it += 1

    return best_a, best_score


def random_anneal(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that moves a random fraction of mass between two
    coordinates.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.12  # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: relocate a small random fraction of mass from one
    index to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10   # up to 10 % of source mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    max_n = 8000  # a larger size gives more freedom for improvement
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            old_n = init_seq.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, max_n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(max_n, 1.0 / max_n)
        else:
            a0 = np.full(max_n, 1.0 / max_n)
    else:
        a0 = np.full(max_n, 1.0 / max_n)

    fft_len_conv = next_pow2(2 * max_n - 1)
    fft_len_g = next_pow2(3 * max_n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, max_n)
    print(f"[init] n={max_n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # Allocate portions of the total budget
    phase1_end = start_time + 0.35 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.60 * budget_s   # sub‑gradient + momentum
    phase3_end = start_time + 0.75 * budget_s   # targeted transfer
    phase4_end = start_time + 0.90 * budget_s   # simulated annealing
    # remainder for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart continuation L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.02, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0004, 0.0002, 0.0001, 5e-05, 2e-05,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # Occasionally perturb the current best to explore its neighbourhood
        if best_score < np.inf and rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(max_n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            max_n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, max_n
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted mass‑transfer optimisation
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = targeted_transfer_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, max_n
        )
        print(f"[targeted] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = random_anneal(
            best_a, best_score, fft_len_conv, max_n, phase4_end
        )
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a, best_score, fft_len_conv, max_n, final_deadline, max_iter=20000
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final evaluation using the provided external function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
