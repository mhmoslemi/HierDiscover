"""
Propose a low‑score step sequence.

Approach
--------
* Work on a fixed length n = 4000 (the length used by the reference code).
* The objective to minimise is   2·n·max(conv(a,a))  (the sum of a is normalised to 1).
* The search consists of many cheap random restarts followed by
  – a continuation L‑BFGS on a log‑sum‑exp smooth approximation of the max,
  – an exponentiated‑gradient descent on the exact max‑convolution,
  – short hill‑climbs that move a random fraction of mass between two coordinates.
* All stages are time‑boxed; the remaining time is spent on a long
  exponentiated‑gradient refinement (tiny step size) and a final
  L‑BFGS pass with a very small smoothing parameter.
* After each optimisation step we also try the monotone reorderings
  (ascending / descending) of the current vector – by the rearrangement
  inequality this never hurts the max‑convolution and often improves it.
* The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int) -> float:
    """Evaluate 2·n·max(a*a) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation utilities
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max_{k} (a*a)_k where a = softmax(y).
    Returns (objective, gradient) w.r.t. the logits y.
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth max wrt each convolution entry
    w = exp_shift / Z               # shape (2n‑1,)

    # back‑propagation through convolution: dconv/d a = (rev(a) * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (slice the centre part)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    L‑BFGS on the smoothed max objective for a decreasing sequence of lambdas.
    Returns the best a (on the simplex) found during the schedule.
    """
    y = np.log(a_start + 1e-12)               # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exact exponentiated‑gradient descent on max(a*a).
    The update a ← a·exp(−η·g) keeps a on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # decreasing step size
        eta = 0.6 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of mass from i to j.
    Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # starting sequence (use the saved one if present)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.asarray(globals()['height_sequence_1'], dtype=float)
        if init.ndim != 1 or init.size == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            # interpolate to the required length
            x_old = np.linspace(0, 1, init.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv)
    print(f"[init] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # time budget
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # a schedule of lambdas for the continuation L‑BFGS
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002]
    maxiter_per_lam = 800   # a bit more than the reference code

    # --------------------------------------------------------------
    # Phase 1 – many short restarts (L‑BFGS + EG + hill‑climb)
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:   # reserve ~30 s for final refinement
        # start point: either a noisy version of the current best or a fresh random vector
        if rng.random() < 0.25 and best_score < np.inf:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.02), rng=rng)
        else:
            a_start = random_initial(n, rng)

        # occasional monotone ordering – often reduces the peak
        if rng.random() < 0.5:
            a_start = np.sort(a_start)          # ascending
        else:
            a_start = a_start[::-1]             # descending

        # ---- L‑BFGS continuation ----
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              deadline,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # ---- exponentiated‑gradient refinement (short burst) ----
        eg_deadline = min(deadline - 5.0, time.time() + 5.0)   # ~5 s per burst
        a_eg, score_eg = exp_gradient_optimize(a_opt,
                                               eg_deadline,
                                               n,
                                               fft_len_conv,
                                               fft_len_g,
                                               rng)
        if score_eg < best_score:
            best_score, best_a = score_eg, a_eg.copy()
            print(f"[exp‑grad] improved to {best_score:.6f}")

        # ---- short hill‑climb ----
        cur_sc = fast_score(best_a, fft_len_conv)
        a_hc, score_hc = hill_climb(best_a,
                                    cur_sc,
                                    fft_len_conv,
                                    n,
                                    deadline,
                                    max_iter=2000)
        if score_hc < best_score:
            best_score, best_a = score_hc, a_hc.copy()
            print(f"[hill‑climb] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient refinement (tiny step)
    # --------------------------------------------------------------
    print("[final] entering long refinement, current score =", best_score)
    a = best_a.copy()
    it = 0
    while time.time() < deadline - 5.0:   # leave a few seconds for the last L‑BFGS pass
        # convolution and mask for the current peak
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the exact max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # a very small step size (the earlier phases already got close)
        eta = 0.07 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        if (it & 63) == 0:          # evaluate every ~64 iterations
            cur_score = fast_score(a, fft_len_conv)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[refine] new best {best_score:.6f}")
        it += 1

    # --------------------------------------------------------------
    # Phase 3 – final L‑BFGS pass with very small smoothing
    # --------------------------------------------------------------
    logits = np.log(best_a + 1e-12)
    final_lam_sched = [0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for lam in final_lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=logits,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 2000,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        logits = res.x
        a = softmax(logits)
        cur_score = fast_score(a, fft_len_conv)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[final‑lbfgs] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
