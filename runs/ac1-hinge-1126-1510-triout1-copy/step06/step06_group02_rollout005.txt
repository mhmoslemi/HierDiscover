"""
Propose a non‑negative step function with a low `evaluate_sequence` score.

Algorithm overview
------------------
1. Choose a problem size `n` (use the previous best construction if it
   exists, otherwise 8000).  Initialise a probability vector `a` (sum = 1)
   on the simplex.
2. **Exploratory L‑BFGS stage** – many random restarts.  We optimise a
   smooth log‑sum‑exp approximation of the max‑convolution using a
   continuation scheme (geometric reduction of the smoothing factor).
   After each run a fast hill‑climb (mass transfer) is performed.
3. **Projected multi‑active sub‑gradient** – optimise the true
   max‑convolution objective directly.  The gradient is obtained from the
   (almost) active convolution entries and a back‑tracking line‑search
   guarantees monotone improvement.
4. **Perturb‑and‑refine** – repeatedly add a small Gaussian perturbation
   to the logits, run a short L‑BFGS continuation and keep improvements.
5. **Simulated‑annealing refinement** – a temperature‑controlled random
   walk that repeatedly moves a fraction of mass from one coordinate to
   another, accepting worsening moves with a probability that decays.
6. Return the best sequence found (as a plain list).  The external harness
   evaluates the exact `evaluate_sequence` score.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                       # max = 0
    exp_shift = np.exp(shifted / lam)            # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # propagate through convolution (a * w)  – convolution of a_rev with w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        # sparse Dirichlet draw (α = 0.1)
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] <= 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the average of the sub‑gradients of all (almost) active entries,
    and a back‑tracking line search to guarantee monotone improvement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # tolerance for “active” entries
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of (average) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec      # d (a_i a_j) / d a_i = 2 a_j

        # step size
        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            # even reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def perturb_refine(
    a: np.ndarray,
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Repeatedly perturb the logits, run a short L‑BFGS continuation and
    keep any improvement.
    """
    best_a = a.copy()
    best_y = y.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        # perturbation magnitude slowly shrinks with time
        remaining = deadline - time.time()
        scale = max(0.001, 0.2 * remaining / (deadline - start_time))
        y_pert = best_y + rng.normal(scale=scale, size=n)

        y_opt = schedule_optimize(
            y_pert,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[perturb] improved fast score = {best_score:.6f}")

    return best_a, best_y, best_score


def annealing_refine(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that moves a random fraction of mass from one index
    to another.  The temperature decays exponentially.
    """
    rng = np.random.default_rng()
    cur_a = a.copy()
    cur_best = cur_score
    best_a = a.copy()
    best_score = cur_score

    # initial temperature – a fraction of the current gap to the trivial bound
    lower_bound = 2.0 * n / (2 * n - 1)  # = 2n/(2n-1)
    temp = max(1e-4, 0.3 * (cur_score - lower_bound))
    cooling = 0.9995
    max_move_frac = 0.20

    while time.time() < deadline:
        # pick source index proportionally to its mass (more likely to move from a big entry)
        src = rng.choice(n, p=cur_a / cur_a.sum())
        if cur_a[src] <= 0.0:
            continue
        dst = rng.integers(0, n)
        if src == dst:
            continue

        delta = cur_a[src] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_score = fast_score(a_new, fft_len, n)

        # accept if better, otherwise with probability exp(-Δ/temp)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

        temp *= cooling

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low ``evaluate_sequence`` score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ---- problem size -------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ---- initialise best candidate ------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ---- parameters ----------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=60))
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "dirichlet",
        "best",
    ]

    # ---- exploratory L‑BFGS + short hill‑climb -----------------------
    phase1_deadline = deadline - 400.0  # keep ~400 s for later phases
    iteration = 0
    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # brief hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=3000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ---- multi‑active projected sub‑gradient ---------------------------
    subgrad_deadline = deadline - 250.0
    if time.time() < subgrad_deadline:
        print("[multi‑subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            subgrad_deadline,
            eta0=0.30,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[multi‑subgrad] finished → fast score = {best_score:.6f}")

    # ---- perturb‑and‑refine (short L‑BFGS runs) ------------------------
    perturb_deadline = deadline - 120.0
    if time.time() < perturb_deadline:
        print("[perturb‑refine] start")
        best_a, best_y, best_score = perturb_refine(
            best_a,
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            perturb_deadline,
            rng,
        )
        print(f"[perturb‑refine] finished → fast score = {best_score:.6f}")

    # ---- simulated‑annealing refinement --------------------------------
    if time.time() < deadline:
        print("[annealing] start")
        best_a, best_score = annealing_refine(
            best_a, best_score, fft_len_conv, n, deadline
        )
        print(f"[annealing] finished → fast score = {best_score:.6f}")

    # ---- final evaluation ---------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # external harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
