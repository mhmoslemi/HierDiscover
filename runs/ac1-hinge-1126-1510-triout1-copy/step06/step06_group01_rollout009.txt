    best_score = fast_score(a_cur, fft_len, n)
    while time.time() < deadline:
        # compute conv and mask
        fft_len_conv = fft_len
        fa = np.fft.fft(a_cur, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        # subgradient
        a_rev = a_cur[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n-1):(n-1+n)]
        # compute positive and negative indices
        pos_idx = np.where(g > 1e-12)[0]
        neg_idx = np.where(g < -1e-12)[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            break
        # choose max positive gradient and min negative gradient
        i = pos_idx[np.argmax(g[pos_idx])]
        j = neg_idx[np.argmin(g[neg_idx])]

        # no need to renormalize sum, as we transfer.
        new_score = fast_score(a_new, fft_len, n)

"""
Propose a low‑evaluation step sequence.

The search consists of several stages (the whole budget is 1000 s).

1. **Warm‑start** – several different initialisations (uniform,
   random, power‑law, “arcsine” distribution).  
2. **Heavy L‑BFGS continuation** – minimise a smoothed log‑sum‑exp
   approximation of the max‑convolution, gradually decreasing the
   smoothing parameter.  Each stage is restarted from a fresh
   initialisation, keeping the best candidate.
3. **Projected sub‑gradient descent with momentum** – optimisation of the
   true max‑convolution using the active (max‑tight) indices.
4. **Guided mass‑transfer (directed sub‑gradient move)** – a deterministic
   step that moves a small fraction of mass from the index with the
   largest positive sub‑gradient to the index with the largest negative
   sub‑gradient, always accepting improving moves.
5. **Final greedy hill‑climb** – deterministic local moves that
   transfer a small fraction of mass between two coordinates while
   strictly decreasing the score.

All sequences are kept on the probability simplex (non‑negative,
∑a=1).  The score reported by the optimiser is the “fast” score
`2·n·max(conv)`, which equals the official evaluation because the sum
is forced to 1.  The routine stops automatically before the deadline
and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a already on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (linear, length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                # d smooth_max / d conv_i

    # back‑propagation through convolution (a with a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each smoothing factor in lam_schedule,
    starting each stage from the result of the previous stage.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # extremely rare – just skip this smoothing stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (see objective_and_grad)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # decreasing step size ~ 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def directed_mass_transfer(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Deterministic improvement: repeatedly move a small fraction of mass
    from the coordinate with the largest positive sub‑gradient to the
    one with the most negative sub‑gradient, always keeping the best
    candidate seen.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    while time.time() < deadline:
        # current convolution and active mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max entry
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # indices with opposite signs
        pos_idx = np.where(g > 1e-12)[0]
        neg_idx = np.where(g < -1e-12)[0]
        if pos_idx.size == 0 or neg_idx.size == 0:
            break

        i = pos_idx[np.argmax(g[pos_idx])]
        j = neg_idx[np.argmin(g[neg_idx])]

        # transfer a modest amount of mass from i to j
        # we never move more than 5 % of the source coordinate
        transfer = min(a[i] * 0.05, 0.005)
        if transfer <= 0:
            break

        a_new = a.copy()
        a_new[i] -= transfer
        a_new[j] += transfer

        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < best_score:
            a = a_new
            best_a = a_new.copy()
            best_score = new_score
        else:
            # try a smaller transfer; if none improve we stop
            smaller = transfer * 0.2
            a_new = a.copy()
            a_new[i] -= smaller
            a_new[j] += smaller
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score:
                a = a_new
                best_a = a_new.copy()
                best_score = new_score
            else:
                break

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: transfer a random fraction of mass between two
    coordinates, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and renormalise onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "arcsine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # arcsine
        i = np.arange(n, dtype=np.float64)
        eps = 0.5
        a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
        a = np.maximum(a, 1e-12)

    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initialisation
    # --------------------------------------------------------------
    n = 4000                      # length of the candidate sequence
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # linearly interpolate the supplied sequence to the target length
            old_n = init_seq.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # Fast convolution lengths (powers of two for efficient FFT)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time‑budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # tiny safety margin

    # allocate portions of the budget
    phase1_end = start_time + 0.48 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient + momentum
    phase3_end = start_time + 0.85 * budget_s   # directed mass‑transfer
    # remainder for final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – random‑restart continuation L‑BFGS
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.001, 0.0005,
    ]
    maxiter_per_lam = 900

    while time.time() < phase1_end:
        # occasional perturbation of the current best
        if best_score < np.inf and rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic directed mass‑transfer
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = directed_mass_transfer(
            best_a, fft_len_conv, fft_len_g, phase3_end, n
        )
        print(f"[directed] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (no time‑budget split)
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a, best_score, fft_len_conv, n, final_deadline, max_iter=15000
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final external evaluation (the real objective)
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
