"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------

* Initialise a probability vector `a` (sum = 1) – either from the previous
  best construction or a uniform distribution of length `n = 5000`.

* **Phase 1 – Smooth‑max L‑BFGS**  
  Repeated random restarts of a continuation scheme on the smoothed
  maximum `max_λ (a*a)`.  The temperature λ is decreased geometrically;
  each stage runs L‑BFGS for a few hundred iterations.

* **Phase 2 – Exponentiated‑gradient refinement**  
  Starting from the L‑BFGS result we minimise the same smoothed objective
  directly on the simplex using an exponentiated‑gradient update
  `a_i ← a_i·exp(-η·∇_i f)`.  This keeps all entries positive and the
  projection is just a renormalisation.

* **Phase 3 – Targeted hill‑climb** – moves mass away from the pair of
  indices that contributes most to the current maximal convolution entry.

* **Phase 4 – Simulated‑annealing random hill‑climb** – occasional
  uphill moves are accepted with a temperature schedule, allowing the
  search to escape shallow plateaux.

* **Phase 5 – Final polishing** – a few extra random moves.

The whole routine respects the 1000 s budget, constantly checking the
remaining time.  The best sequence found is stored in the global variable
`height_sequence_1` and returned as a plain `list[float]`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``.
    Assumes ``a`` sums to 1.
    Returns 2·n·max(conv(a,a)).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient w.r.t. the coefficients a
# ----------------------------------------------------------------------


def smooth_obj_and_grad_a(
    a: np.ndarray, lam: float, fft_len: int, fft_len_g: int, n: int
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_f) where

        f   = 2 * n * smooth_max(a*a, λ)
        grad_f = gradient of f w.r.t. a (shape n)

    The smooth maximum is defined as
        smooth_max = max_c + λ·log Σ exp((conv - max_c)/λ)
    """
    # full convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    shifted = conv - max_c           # the maximal entry becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    # weight vector for the smoothed max (∂smooth_max/∂conv)
    w = exp_shift / Z                # shape (2*n-1,)

    # gradient of the *unsmoothed* term 2·n·smooth_max:
    #   ∂f/∂a = 2·n * (∂smooth_max/∂a)
    # where ∂smooth_max/∂a = convolution of w with a (both directions)
    # We compute it efficiently via FFT.
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice the part that aligns with a (length n)
    g_vec = conv_rev[n - 1 : n - 1 + n]
    grad_f = 2.0 * n * g_vec
    return float(2.0 * n * smooth_max), grad_f


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------


def exp_gradient_descent(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    fft_len_g: int,
    lam_schedule: np.ndarray,
    iters_per_lam: int,
    eta: float,
) -> np.ndarray:
    """
    Perform exponentiated‑gradient descent on the smoothed‑max objective.
    `lam_schedule` is traversed from large to tiny values.
    """
    a = a_start.copy()
    for lam in lam_schedule:
        for _ in range(iters_per_lam):
            _, grad = smooth_obj_and_grad_a(a, lam, fft_len, fft_len_g, n)
            # multiplicative update; keep positivity automatically
            a = a * np.exp(-eta * grad)
            total = a.sum()
            if total == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / total
    return a


# ----------------------------------------------------------------------
# Hill‑climbing utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb (only accepting improvements)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Move mass away from the pair of indices that contributes most to the
    current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def annealing_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.02,
    cooling: float = 0.999,
    max_iter: int = 20000,
    max_move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing random hill‑climb."""
    best = a.copy()
    best_score = cur_score
    temp = init_temp
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score:
            best_score = new_score
        else:
            # metropolis acceptance
            prob = np.exp(-(new_score - best_score) / max(temp, 1e-12))
            if rng.random() < prob:
                best_score = new_score
            else:
                # revert
                best[i] += delta
                best[j] -= delta

        temp *= cooling
    return best, best_score


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point `y` (log‑soft‑max parameters) for L‑BFGS.
    Several heuristics are mixed to keep diversity across restarts.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    else:  # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smoothed‑max surrogate
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = float(2.0 * n * smooth_max)

    w = exp_shift / Z
    # gradient wrt a (2*n*g_vec) – same as in the smooth objective above
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]
    grad_a = 2.0 * n * g_vec

    # chain rule through soft‑max: ∂a/∂y = diag(a) – a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return f_val, grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops a few seconds early.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size and initial point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on the smoothed max
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=45)   # fine λ schedule
    maxiter_per_stage = 800
    iteration = 0

    while time.time() < deadline - 30.0:          # leave ~30 s for refinements
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved fast score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exponentiated‑gradient refinement (smoothed max)
        # ------------------------------------------------------------------
        # start from the just‑obtained a_opt
        lam_sched_exp = np.geomspace(0.5, 1e-8, num=30)
        a_exp = exp_gradient_descent(
            a_start=a_opt,
            n=n,
            fft_len=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_schedule=lam_sched_exp,
            iters_per_lam=150,
            eta=0.07,
        )
        score_exp = fast_score(a_exp, fft_len_conv, n)
        if score_exp < best_score:
            best_a = a_exp.copy()
            best_score = score_exp
            print(f"[{iteration}] EXP‑grad improved fast score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Targeted hill‑climb (short burst)
        # ------------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 1.2)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=2000,
            move_frac=0.30,
        )

        # ------------------------------------------------------------------
        # Phase 4 – Simulated‑annealing random hill‑climb (short burst)
        # ------------------------------------------------------------------
        an_deadline = min(deadline, time.time() + 1.5)
        best_a, best_score = annealing_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=an_deadline,
            init_temp=0.015,
            cooling=0.998,
            max_iter=8000,
            max_move_frac=0.22,
        )

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 5 – Final polishing with a longer random hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=30000,
            max_move_frac=0.25,
        )
        print(f"[final hill‑climb] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    # The harness will call evaluate_sequence() later; we only log here.
    print(f"[final] fast score (pre‑evaluation) = {best_score:.6f}")
    return final_seq
