def hill_climb_targeted(a, cur_score, fft_len, n, deadline, max_iter=20000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # precompute indices of each a_i (positions).
    # We'll compute conv each iteration (fast_score)
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        p = int(np.argmax(conv))
        # Find all i such that 0 <= i < n and 0 <= p-i < n (valid counterpart)
        # compute contributions a_i*a_{p-i}
        i_vals = np.arange(max(0, p - (n - 1)), min(p, n - 1) + 1)
        j_vals = p - i_vals
        contribs = best[i_vals] * best[j_vals]
        # For i=j case, contribution counted once but okay.
        # Choose index with max contribution (largest a_i*a_j)
        idx_max = np.argmax(contribs)
        i = i_vals[idx_max]
        j = j_vals[idx_max]
        # pick a target index k not equal to i or j
        # maybe random but prefer low a.
        # Choose k among indices where a is small (< threshold)
        low_idx = np.where(best < best.mean())[0]
        if low_idx.size == 0:
            k = rng.integers(n)
        else:
            k = rng.choice(low_idx)
        # Transfer small delta from i (or from j if i==j?) to k.
        # If i!=j, we might reduce both i and j?
        # Simpler: reduce mass from i (or if i=j reduce half) and increase k.
        delta = best[i] * rng.random() * 0.15  # up to 15%
        if delta <= 0:
            continue
        best[i] -= delta
        best[k] += delta
        # Ensure non-negativity
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[k] -= delta
    return best, best_score
```

"""
Propose a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (∑a = 1) – the evaluation is scale invariant.
* Use a moderately large length (≈ 8k–12k) to obtain a rich search space.
* Multi‑restart continuation L‑BFGS on a log‑sum‑exp smoothed version of the
  maximum convolution.  The continuation schedule has many stages and a
  generous per‑stage iteration limit.
* After each L‑BFGS run a short random hill‑climb is performed.
* A long projected sub‑gradient descent on the true objective (2·n·max(conv))
  follows, using a diminishing step‑size and keeping the best iterate.
* Finally an aggressive “max‑focused” hill‑climb that always attacks the
  entries contributing to the current maximal convolution value is run
  until the time budget expires.
* The best sequence found is returned; the harness will call
  ``evaluate_sequence`` on it.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using FFT.
    a must have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    with respect to the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a * a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t conv
    w = exp_shift / Z

    # derivative of conv w.r.t a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # entries corresponding to gradient components (middle n entries)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: a sequence of decreasing smoothing factors.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

# Initialisation helpers

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes: random, exp, power, edge, sidon, binary, bump, harmonic.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb_random(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb: move a random fraction of mass from i to j.
    Keep the move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def hill_climb_targeted(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 10000,
    max_move: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that always attacks the entry of the convolution
    that attains the current maximum.  Mass is moved away from the
    pair (i,j) contributing most to that peak towards a location with
    relatively low current weight.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        conv = conv_fft(best, fft_len, n)
        p = int(np.argmax(conv))

        # indices i that can pair with j = p - i to give sum p
        i_min = max(0, p - (n - 1))
        i_max = min(p, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = p - i_vals

        contribs = best[i_vals] * best[j_vals]
        if contribs.size == 0:
            continue

        # choose the pair with the largest contribution
        idx = int(np.argmax(contribs))
        i = int(i_vals[idx])
        j = int(j_vals[idx])

        # pick a target location with relatively small weight
        low_candidates = np.where(best < best.mean())[0]
        if low_candidates.size == 0:
            k = rng.integers(n)
        else:
            k = rng.choice(low_candidates)

        # move a fraction of mass from i (or j if i==j) to k
        source = i if best[i] >= best[j] else j
        delta = best[source] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[source] -= delta
        best[k] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[source] += delta
            best[k] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (i.e. minimise max(conv) with sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the entry conv[m]
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # Diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score
