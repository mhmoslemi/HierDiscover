"""
Propose a step‑function (list of non‑negative heights) that minimizes

    2·n·max(conv) / (sum(a)**2),

where conv = np.convolve(a, a).  The search is performed entirely on the
probability simplex (∑a = 1) because the objective is scale‑invariant.
The method consists of

1.  A cheap exhaustive grid over a few low‑dimensional families
    (Gaussian, edge‑heavy two‑level, power‑law, symmetric Beta).
    Each candidate is evaluated with a fast FFT based convolution.

2.  If time permits, a few random Dirichlet restarts are examined.

3.  A final greedy hill‑climb that moves a random fraction of mass from
    one index to another while keeping the simplex constraint.

The whole procedure respects the 1000 s budget and never uses closures or
lambda expressions – all helpers are top‑level functions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # keep only the real part (numerical noise may introduce tiny imag.)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)   (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple greedy hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount taken from i (up to max_move of its current mass)
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    default_n = 8000
    # if a previously‑found sequence is supplied we can reuse its size
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 9000)

    n = default_n
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise best candidate (uniform distribution)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # sum == 1
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} uniform score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Helper to evaluate a raw (non‑normalised) array and possibly
    # replace the current best.
    # --------------------------------------------------------------
    def update_if_better(raw_a):
        nonlocal best_a, best_score
        total = raw_a.sum()
        if total <= 0.0:
            return
        a = raw_a / total                     # project onto simplex
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"[update] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Gaussian family  (a_i ∝ exp(-((i-center)^2)/(2*σ^2)))
    # --------------------------------------------------------------
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    for sigma_frac in np.linspace(0.08, 0.45, 20):
        sigma = sigma_frac * n
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        update_if_better(a)
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2) Edge‑heavy two‑level shape.
    #    Parameters: m = number of edge cells on each side,
    #                r = a_edge / a_center  (ratio >0)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # geometric progression gives finer granularity at small values
        m_fracs = np.geomspace(0.001, 0.20, num=30)   # fraction of n
        ratios = np.geomspace(1.0, 200.0, num=30)     # edge‑to‑center ratio
        for mf in m_fracs:
            m = int(round(mf * n))
            # keep at least one edge cell and avoid overlap
            if m < 1 or 2 * m >= n:
                continue
            for r in ratios:
                a_center = 1.0 / (2 * m * r + (n - 2 * m))
                a_edge = r * a_center
                a = np.full(n, a_center)
                a[:m] = a_edge
                a[-m:] = a_edge
                update_if_better(a)
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # 3) Power‑law family  a_i ∝ (i+1)^(-p) + (n-i)^(-p)
    # --------------------------------------------------------------
    if time.time() < deadline:
        for p in np.linspace(0.2, 4.0, 25):
            a = ( (idx + 1) ** (-p) + (n - idx) ** (-p) )
            update_if_better(a)
            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # 4) Symmetric Beta‑shape  a_i ∝ x^(α‑1) * (1‑x)^(α‑1)
    #    (α=0.5 corresponds to the arcsine distribution)
    # --------------------------------------------------------------
    if time.time() < deadline:
        x = (idx + 0.5) / n                         # centre of each bin
        for alpha in np.linspace(0.15, 5.0, 25):
            a = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
            update_if_better(a)
            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # 5) Random Dirichlet restarts (if we still have time)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # Dirichlet with concentration <1 gives sparse vectors,
        # which sometimes produce a very flat convolution.
        conc = rng.uniform(0.05, 0.4)
        a = rng.dirichlet(np.full(n, conc))
        update_if_better(a)

    # --------------------------------------------------------------
    # 6) Final greedy hill‑climb (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=40000,
            max_move=0.30,
        )
        print(f"[final] hill‑climb score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Convert to plain list, evaluate with the supplied harness and
    # return the sequence.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # official evaluation (the harness provides evaluate_sequence)
    final_val = evaluate_sequence(final_seq)   # type: ignore
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
