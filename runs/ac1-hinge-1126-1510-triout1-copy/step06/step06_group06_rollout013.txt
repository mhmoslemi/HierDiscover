"""
Low‑score step function search.

The algorithm consists of three stages:

1.  **Parametric exploration** – a set of simple families (exponential,
    power‑law, linear ramp, heavy‑edge) is scanned.  Each candidate is
    normalised to a probability vector (sum = 1) and evaluated with a fast
    FFT‑based surrogate `fast_score = 2·n·max(conv(a,a))`.  The best
    candidate so far is kept.

2.  **Smooth‑max L‑BFGS refinement** – the current best vector is turned
    into log‑soft‑max parameters `y`.  A continuation of L‑BFGS optimisations
    over a decreasing temperature schedule minimises a smooth approximation
    of the max‑convolution.  The refined vector replaces the best one when
    it improves the score.

3.  **Random hill‑climb** – finally a simple mass‑transfer hill‑climb
    (always keeping only strictly improving moves) is performed until the
    time budget expires.

The whole procedure respects the 1000 s budget, stores the best sequence
in the global variable ``height_sequence_1`` and returns it.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max yielding a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast 2·n·max(conv(a,a)) using FFT (a is assumed to sum to 1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Smoothed max of the auto‑convolution and its gradient w.r.t. `y`."""
    a = softmax(y)

    # convolution a*a (full) via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z

    # propagate back to a (using convolution with reversed a)
    fa_rev = np.fft.rfft(a[::-1], fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient w.r.t. a
    grad_a = 2.0 * n * g_vec

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """Continuation of L‑BFGS over a decreasing λ‑schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple random hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Load previous construction or start from uniform
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalize to a probability vector
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= s

    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1.  Parametric families – cheap global exploration
    # --------------------------------------------------------------

    # ---- Exponential decay -------------------------------------------------
    if time.time() < deadline:
        for r in np.linspace(0.90, 0.9999, 200):
            a = np.power(r, np.arange(n, dtype=float))
            a /= a.sum()
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_a, best_score = a, sc
                print(f"[exp] r={r:.5f} improved fast score = {best_score:.6f}")
            if time.time() > deadline:
                break

    # ---- Power‑law ---------------------------------------------------------
    if time.time() < deadline:
        for p in np.linspace(0.1, 4.0, 200):
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_a, best_score = a, sc
                print(f"[pow] p={p:.3f} improved fast score = {best_score:.6f}")
            if time.time() > deadline:
                break

    # ---- Linear ramp (decreasing to zero) ----------------------------------
    if time.time() < deadline:
        for L_frac in np.linspace(0.5, 1.0, 100):
            L = int(L_frac * n)
            a = np.zeros(n, dtype=float)
            if L > 0:
                a[:L] = (L - np.arange(L, dtype=float)) / L
            a_sum = a.sum()
            if a_sum > 0.0:
                a /= a_sum
                sc = fast_score(a, fft_len, n)
                if sc < best_score:
                    best_a, best_score = a, sc
                    print(f"[ramp] L_frac={L_frac:.3f} improved fast score = {best_score:.6f}")
            if time.time() > deadline:
                break

    # ---- Heavy‑edge two‑level ----------------------------------------------
    if time.time() < deadline:
        for m_frac in (0.01, 0.02, 0.03, 0.05):
            m = int(m_frac * n)
            for ratio in (2, 5, 10, 20):
                a = np.ones(n, dtype=float)
                a[:m] *= ratio
                a[-m:] *= ratio
                a /= a.sum()
                sc = fast_score(a, fft_len, n)
                if sc < best_score:
                    best_a, best_score = a, sc
                    print(f"[edge] m_frac={m_frac:.3f} ratio={ratio} improved fast score = {best_score:.6f}")
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # ---- Random Dirichlet draws (sparse prototypes) -----------------------
    if time.time() < deadline:
        for _ in range(100):
            conc = rng.uniform(0.1, 2.0)
            a = rng.gamma(conc, 1.0, size=n)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a /= a_sum
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_a, best_score = a, sc
                print(f"[dir] conc={conc:.3f} improved fast score = {best_score:.6f}")
            if time.time() > deadline:
                break

    # --------------------------------------------------------------
    # 2.  Smooth‑max L‑BFGS refinement (a few restarts)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        lam_factors = np.geomspace(0.5, 1e-8, num=60)
        maxiter_stage = 1000
        for restart in range(5):
            if time.time() >= deadline - 5.0:
                break
            y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.05, size=n)
            y_opt = schedule_optimize(
                y_start=y0,
                n=n,
                fft_len_conv=fft_len,
                fft_len_g=next_pow2(3 * n - 2),
                lam_factors=lam_factors,
                maxiter_per_stage=maxiter_stage,
            )
            a_opt = softmax(y_opt)
            sc = fast_score(a_opt, fft_len, n)
            if sc < best_score:
                best_a, best_score = a_opt, sc
                print(f"[lbfgs] restart={restart} improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Random hill‑climb (final polishing)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.30,
        )
        print(f"[final hill] best fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
