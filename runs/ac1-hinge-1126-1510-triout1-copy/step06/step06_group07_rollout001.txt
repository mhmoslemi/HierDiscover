    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    lam_factors = np.geomspace(0.5, 1e-14, num=70)
    maxiter_per_stage = 1500
    iteration = 0

    while time.time() < deadline - 2.0: # keep some time for final refinement
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.03)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 3.0)
            a0 = np.arange(1, n+1, dtype=float) ** (-p)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.005, 0.03)))

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm (parallel multi‑restart):
* The problem is to minimise 2·n·max(conv(a,a)) with ∑a = 1, a_i ≥ 0.
* We treat a as a probability vector and work on its (log‑soft‑max) logits.
* A continuation L‑BFGS minimises a smooth approximation of the ℓ∞‑norm.
* After each L‑BFGS run we perform cheap hill‑climbs that move mass away
  from the currently worst convolution shift.
* When the time budget permits we finish with a projected sub‑gradient descent
  on the exact max‑convolution.
* Two independent workers (different random seeds, different support sizes)
  run in parallel; the best sequence from either worker is returned.
* A final lightweight smoothing loop tries to improve the sequence a
  little more before the official evaluation.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
import concurrent.futures

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for a probability vector.
    (Sum of a is assumed to be 1.)
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from:
    Wang, Weiran, and Miguel Á. Carreira‑Perpiñán. “Projection onto the
    probability simplex: An efficient algorithm with a simple proof,
    and an application.” (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed ℓ∞‑norm of the auto‑convolution and its gradient w.r.t.
    the unconstrained logits y.

    Parameters
    ----------
    y : np.ndarray
        Log‑soft‑max parameters.
    n : int
        Length of the underlying sequence.
    fft_len_conv, fft_len_g : int
        Lengths of the FFTs (powers of two) used for the convolution
        and the back‑propagation step.
    lam_factor : float
        Multiplicative factor that determines the smoothing parameter:
        λ = max(conv)·lam_factor + 1e‑12.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # ----- back‑propagation to a -----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (via FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # pick a destination with a tiny partner under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep move
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent w.r.t. the exact max of the auto‑convolution.
    Each iteration identifies the shift achieving the maximum, computes the
    corresponding gradient (2·a shifted), takes a step opposite to it and
    projects back onto the probability simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1
        # compute convolution (FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for shift k:
        grad = np.zeros_like(best)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]

        # step size decreasing like 1/√t
        step = step0 / math.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional stochastic jumps to get out of plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score


# ----------------------------------------------------------------------
# Light smoothing (post‑processing)
# ----------------------------------------------------------------------


def simple_smoothing(seq: np.ndarray, passes: int = 5) -> np.ndarray:
    """
    Apply a few passes of a 3‑point averaging filter and re‑normalise.
    Very cheap and sometimes reduces the ℓ∞‑norm of the convolution.
    """
    a = seq.copy()
    n = a.size
    for _ in range(passes):
        b = a.copy()
        b[1:-1] = (a[:-2] + a[1:-1] + a[2:]) / 3.0
        # ends are left untouched (they have little influence on the max)
        a = b
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Worker that performs an independent search
# ----------------------------------------------------------------------


def _search_worker(seed: int, n: int, time_budget: float):
    """
    Independent optimiser for a single support size.
    Returns (best_sequence_as_list, best_fast_score).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + time_budget

    # initial uniform distribution
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # continuation L‑BFGS hyper‑parameters
    lam_factors = np.geomspace(0.5, 1e-14, num=70)
    maxiter_per_stage = 800

    iteration = 0
    # ------------------------------------------------------------------
    # Global search loop (restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:  # keep a buffer for final refinement
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.025)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 3.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.004, 0.025)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt

        # ----- quick local moves -----
        # targeted hill‑climb
        t_dead = min(deadline, time.time() + 0.6)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=800,
            move_frac=0.28,
        )
        best_y = np.log(best_a + 1e-12)

        # random hill‑climb
        r_dead = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=4000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final refinement on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = subgrad_descent(
            a=best_a,
            cur_score=best_score,
            n=n,
            fft_len=fft_len_conv,
            deadline=deadline,
            max_iter=8000,
            step0=0.32,
        )

    # ------------------------------------------------------------------
    # Optional cheap smoothing passes – they sometimes shave a tiny bit
    # ------------------------------------------------------------------
    final_a = best_a
    final_score = best_score
    for _ in range(3):
        cand = simple_smoothing(final_a, passes=3)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < final_score - 1e-12:
            final_a = cand
            final_score = cand_score
        else:
            break

    return final_a.tolist(), float(final_score)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main function called by the evaluation harness.
    It runs two parallel optimiser instances (different random seeds and
    different support lengths) and returns the best sequence found within
    the allocated time budget.
    """
    # Reserve a few seconds for the final evaluation and printout
    safety_margin = 5.0
    work_budget = max(0.0, budget_s - safety_margin)

    # Two different problem sizes – diversity helps escape local minima
    sizes = [5000, 8000]
    per_worker = work_budget / len(sizes)

    # Start parallel workers
    best_seq = None
    best_val = float("inf")
    begin = time.time()
    with concurrent.futures.ProcessPoolExecutor(max_workers=2) as executor:
        futures = []
        for i, n in enumerate(sizes):
            futures.append(
                executor.submit(_search_worker, seed + i, n, per_worker)
            )
        for fut in concurrent.futures.as_completed(futures):
            seq, fast_sc = fut.result()
            # Compute the *exact* evaluation (the harness will also do this,
            # but we want the true score now to decide which candidate to keep)
            val = evaluate_sequence(seq)
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"[worker] new global best → score = {best_val:.6f}")

    # Store for possible reuse in subsequent runs
    globals()["height_sequence_1"] = best_seq

    elapsed = time.time() - begin
    print(f"[final] evaluated score = {best_val:.6f} (elapsed {elapsed:.1f}s)")
    return best_seq
