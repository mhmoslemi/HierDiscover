def simulated_annealing(a, cur_score, fft_len, n, deadline, total_iters=50000, start_T=0.01, end_T=1e-6):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < total_iters:
        # compute temperature schedule linearly in log
        t = max(end_T, start_T * (1 - it / total_iters))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.2  # max_move_frac
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # accept with probability exp(-(new_score - best_score)/t)
            if rng.random() < np.exp(-(new_score - best_score)/t):
                best_score = new_score
            else:
                # revert
                best[i] += delta
                best[j] -= delta
        it += 1
    return best, best_score
```

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------
1. Initialise from the previous best (if any) or from a uniform vector.
2. Repeatedly perform a multi‑stage L‑BFGS optimisation on a
   smooth‑max surrogate of the true objective.  The smoothing factor
   (lam_factor) follows a geometric schedule from 0.6 down to 1e‑9.
3. After each L‑BFGS run a short targeted hill‑climb and a short random
   hill‑climb are applied to escape tiny plateaus.
4. When the time budget is low, run an Adam optimiser on the final
   log‑soft‑max parameters with a very small smoothing factor to
   fine‑tune the solution.
5. Finish with a simulated‑annealing hill‑climb that occasionally
   accepts worsening moves (temperature slowly decays).  This often
   discovers the last few percent of improvement.
6. The best sequence (normalised to sum = 1) is stored in the global
   variable ``height_sequence_1`` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective related helpers
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) – the exact objective when ∑a=1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max (log‑sum‑exp)
    shifted = conv - max_c                       # peak at zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # propagate gradient back to a
    # note: conv_rev = a reversed convolved with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(y_start: np.ndarray, n: int, fft_len_conv: int,
                      fft_len_g: int, lam_factors, maxiter_per_stage: int = 800):
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-14,
                                "disp": False})
        y = res.x
    return y


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point `y` (log‑soft‑max parameters) for L‑BFGS.
    Heuristics are mixed to keep diversity across restarts.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "best", "sine", "gauss"])
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "sine":
        x = np.arange(n) / (n - 1)
        a0 = np.sin(np.pi * x) ** 2
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "gauss":
        sigma = rng.uniform(0.1, 0.4) * n
        centre = n // 2
        a0 = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    else:  # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


# ----------------------------------------------------------------------
# Simple hill‑climbs
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, cur_score: float, fft_len: int, n: int,
               deadline: float, max_iter: int = 5000,
               max_move_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb on the simplex (strictly improving moves)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(a: np.ndarray, cur_score: float, fft_len: int, n: int,
                        deadline: float, max_iter: int = 500,
                        move_frac: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int,
                  lam_factor: float, epochs: int, step_size: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple Adam optimiser on the smooth‑max surrogate.
    Returns the best (a, score) pair seen during the run.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)
    rng = np.random.default_rng()
    t = 0
    while time.time() < deadline and t < epochs:
        t += 1
        f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= step_size * m_hat / (np.sqrt(v_hat) + eps)

        # occasionally check the true score (every 200 steps)
        if t % 200 == 0:
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    # final evaluation
    final_a = softmax(y)
    final_score = fast_score(final_a, fft_len_conv, n)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(a: np.ndarray, cur_score: float, fft_len: int, n: int,
                        deadline: float, total_iters: int = 20000,
                        start_T: float = 0.02, end_T: float = 1e-6,
                        max_move_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """Annealed random walk on the simplex with occasional uphill acceptance."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < total_iters:
        it += 1
        # linear temperature schedule in log‑space
        T = max(end_T, start_T * (1.0 - it / total_iters))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_score - best_score) / T):
                best_score = new_score
            else:
                best[i] += delta
                best[j] -= delta
    return best, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Uses a combination of L‑BFGS on a smooth surrogate, Adam fine‑tuning,
    and simulated‑annealing hill‑climbing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size and starting point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to simplex (scale‑invariant objective)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS global search (multiple restarts)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.6, 1e-9, num=68)      # finer near zero
    maxiter_per_stage = 1200
    iteration = 0

    # Reserve ~200 s for fine tuning / annealing at the end
    while time.time() < deadline - 200.0:
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = schedule_optimize(y_start=y0,
                                  n=n,
                                  fft_len_conv=fft_len_conv,
                                  fft_len_g=next_pow2(3 * n - 2),
                                  lam_factors=lam_factors,
                                  maxiter_per_stage=maxiter_per_stage)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Light local refinements (fast)
        # ------------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(a=best_a,
                                                 cur_score=best_score,
                                                 fft_len=fft_len_conv,
                                                 n=n,
                                                 deadline=tc_deadline,
                                                 max_iter=800,
                                                 move_frac=0.30)
        best_y = np.log(best_a + 1e-12)

        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(a=best_a,
                                        cur_score=best_score,
                                        fft_len=fft_len_conv,
                                        n=n,
                                        deadline=hc_deadline,
                                        max_iter=4000,
                                        max_move_frac=0.20)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Adam fine‑tuning on the smooth‑max surrogate (very low λ)
    # ------------------------------------------------------------------
    adam_deadline = min(deadline, time.time() + 150.0)  # give it ~150 s
    best_a, best_score = adam_optimize(y0=best_y,
                                       n=n,
                                       fft_len_conv=fft_len_conv,
                                       fft_len_g=next_pow2(3 * n - 2),
                                       lam_factor=1e-9,
                                       epochs=12000,
                                       step_size=0.04,
                                       deadline=adam_deadline)
    best_y = np.log(best_a + 1e-12)
    print(f"[adam] finished with score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing polishing (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(a=best_a,
                                                 cur_score=best_score,
                                                 fft_len=fft_len_conv,
                                                 n=n,
                                                 deadline=deadline,
                                                 total_iters=50000,
                                                 start_T=0.02,
                                                 end_T=1e-7,
                                                 max_move_frac=0.25)
        print(f"[anneal] final polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final random hill‑climb (just in case a tiny budget remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(a=best_a,
                                        cur_score=best_score,
                                        fft_len=fft_len_conv,
                                        n=n,
                                        deadline=deadline,
                                        max_iter=20000,
                                        max_move_frac=0.25)
        print(f"[final hill] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    # official evaluation (the harness will also call evaluate_sequence)
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
