"""
propose_candidate
=================

A refined search for a non‑negative step function (list of floats) that
minimises the supplied ``evaluate_sequence`` score.

The programme proceeds in four stages:

1. **Problem size & initialisation** –  try a variety of simple
   parametric families (beta‑like, power‑law, exponential, edge‑spike,
   cosine) and keep the best as a starting point.  If a previous
   construction ``height_sequence_1`` is available we re‑use it.

2. **Continuation L‑BFGS on a smooth approximation** –  minimise a
   log‑sum‑exp smooth‑max of the autocorrelation.  The smoothing factor
   (λ) is reduced geometrically over 45 stages; each stage runs a
   fairly long L‑BFGS optimisation (≈500 iterations).  Many random
   restarts (including perturbations of the current best) are performed
   until the majority of the time budget is exhausted.

3. **Adam‑style projected sub‑gradient descent** –  after the
   L‑BFGS phase we optimise the true non‑smooth objective using an
   Adam‑style update on the sub‑gradient of the max‑convolution,
   projected onto the probability simplex.

4. **Aggressive hill‑climb** –  a final greedy mass‑transfer hill‑climb
   cleans up any remaining slack.

All helper functions are top‑level, use only ``numpy``/``scipy`` and
respect the total time budget (default 1000 s).  The returned list is
a probability vector; the harness rescales it automatically, so the
sum of the coefficients does not matter.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unrestricted vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution ----------------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1)…(n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)        # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisation stages with
    decreasing smoothing (lam_factor).  Each stage starts from the result
    of the previous stage.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_distribution(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector (sum = 1) for a given heuristic mode."""
    if mode == "random":
        a = rng.random(n)
    elif mode == "uniform":
        a = np.ones(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-rate * d)
    elif mode == "power":
        p = rng.uniform(0.2, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        # a few large spikes at both ends, rest small
        m_edge = max(1, int(n * rng.uniform(0.005, 0.03)))
        a = np.full(n, 1.0)               # baseline
        edge_val = np.sqrt(n / 2.0)       # heuristic strength
        a[:m_edge] = edge_val
        a[-m_edge:] = edge_val
    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    elif mode == "cosine":
        # Hann window – small at edges, large in centre (useful for diversity)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    elif mode == "beta":
        # beta‑like: (i+1)^α (n‑i)^β with random α,β
        α = rng.uniform(-2.0, 0.5)
        β = rng.uniform(-2.0, 0.5)
        i = np.arange(n, dtype=float)
        a = (i + 1) ** α * (n - i) ** β
    else:
        raise ValueError(f"unknown mode {mode}")

    # ensure non‑negative and normalise
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j,
    accepting only improvements of the fast (smoothed) score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.18,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the true objective
        f(a) = max_i (a*a)_i .
    The sub‑gradient is taken at a single maximal index.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    t = 0

    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        # sub‑gradient at the active index
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / arr.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Cheap parametric families (beta‑like, power‑law, arcsine, cosine …)
    # ------------------------------------------------------------------
    # symmetric beta‑like families
    alphas = np.linspace(-2.0, 0.5, 9)
    betas = np.linspace(-2.0, 0.5, 9)
    for α in alphas:
        for β in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** α * (n - i) ** β
            prob = np.maximum(prob, 0.0)
            prob_sum = prob.sum()
            if prob_sum == 0.0:
                continue
            prob = prob / prob_sum
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                print(f"[grid β] α={α:.2f} β={β:.2f} → fast_score = {sc:.6f}")

    # power‑law sweep
    for p in np.linspace(0.2, 3.0, 12):
        prob = (np.arange(1, n + 1, dtype=float) ** (-p))
        prob = prob / prob.sum()
        sc = fast_score(prob, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = prob.copy()
            print(f"[grid pow] p={p:.2f} → fast_score = {sc:.6f}")

    # cosine (Hann) window
    prob = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    prob = prob / prob.sum()
    sc = fast_score(prob, fft_len_conv, n)
    if sc < best_score:
        best_score = sc
        best_a = prob.copy()
        print(f"[grid cos] → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (major part of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(1.0, 1e-11, num=45))
    maxiter_per_stage = 500
    modes = ["random", "exp", "exp_sym", "power", "edge",
             "arcsine", "cosine", "beta", "best"]
    iteration = 0

    # Reserve roughly 180 s for the final Adam + hill‑climb stages
    while time.time() < deadline - 180.0:
        mode = rng.choice(modes)
        if mode == "best":
            # perturb current best
            y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.04, size=n)
        else:
            a0 = generate_initial_distribution(mode, n, rng)
            y0 = np.log(a0 + 1e-12)

        # Continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast_score = {best_score:.6f}")

        # short hill‑climb to polish this result
        hc_deadline = min(deadline - 100.0, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Adam‑style projected sub‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 30.0
        print("[adam] start")
        best_a, best_score = adam_subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.18,
            maxiter_per_stage=0,  # unused, placeholder
        )
        print(f"[adam] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill‑climb] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[final hill‑climb] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
