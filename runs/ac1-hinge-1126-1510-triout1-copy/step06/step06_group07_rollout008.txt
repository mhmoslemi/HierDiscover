"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------
* Work on a probability vector (∑ = 1).  All steps keep the vector on the
  simplex via a soft‑max parametrisation.
* Global search in a reduced‐dimensional space:
    - The sequence is represented by `K` constant blocks of length `L`.
    - A continuation L‑BFGS (smooth‑max surrogate) optimises the block
      weights, which is far cheaper than full‑dimension optimisation.
* The best block‐wise solution is expanded to the full length and
  refined by the original fine‑grained hill‑climb / sub‑gradient routine.
* The whole process respects the 1000 s time budget; the best sequence
  seen is stored in the global name ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` (∑a = 1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full‑dimension smooth‑max objective (used for final polishing)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution (quadratic) and its gradient
    w.r.t. the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a -----
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the full‑dimension smoothed objective.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Segment‑wise smooth‑max objective (low‑dimensional search)
# ----------------------------------------------------------------------


def objective_and_grad_seg(
    y: np.ndarray,
    seg_K: int,
    L: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max objective when the sequence consists of ``seg_K`` equal
    blocks of length ``L``.  ``y`` are the logits of the block probabilities.
    """
    # block probabilities (sum = 1)
    p = softmax(y)

    # full vector (∑ = 1) – each block repeated L times and scaled by 1/L
    a = np.repeat(p, L, axis=0) / L

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * len(a) - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * len(a) * smooth_max

    # ----- gradient w.r.t. a (raw, before the outer 2·n factor) -----
    w = exp_shift / Z
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_raw = conv_rev[(len(a) - 1) : (len(a) - 1 + len(a))]

    # Sum gradient contributions per block
    sum_g = np.add.reduceat(g_raw, np.arange(0, len(a), L))
    mean_sum_g = np.dot(p, sum_g)

    # Gradient w.r.t. logits y
    grad_y = (2.0 * len(a) / L) * p * (sum_g - mean_sum_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize_seg(
    y_start: np.ndarray,
    seg_K: int,
    L: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the block‑wise smoothed objective.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad_seg,
            x0=y,
            args=(seg_K, L, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Focused hill‑climb that reduces the current maximal convolution entry."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

# Projected sub‑gradient descent on the exact max (fine‑grained)
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent for the true max of the auto‑convolution."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    iter_no = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        iter_no += 1
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        grad = np.zeros_like(best)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]

        step = step0 / np.sqrt(iter_no)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score

    return best, best_score

        if mode == "random":
            y0 = rng.standard_normal(seg_K)
        elif mode == "exp":
            rate = rng.uniform(0.005, 0.02)
            a0 = np.exp(-rate * np.arange(seg_K))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(seg_K * rng.uniform(0.02, 0.06)))
            edge_mul = rng.uniform(2.0, 5.0)
            a0 = np.full(seg_K, 1.0)
            a0[:m_edge] = edge_mul
            a0[-m_edge:] = edge_mul
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturb the current best block vector
            y0 = best_y_seg + rng.normal(scale=0.07, size=seg_K)

        # ----- continuation L‑BFGS on block variables -----
        y_opt = schedule_optimize_seg(
            y_start=y0,
            seg_K=seg_K,
            L=L,
            fft_len_conv=seg_fft_len_conv,
            fft_len_g=seg_fft_len_g,
            lam_factors=seg_lam_factors,
            maxiter_per_stage=seg_maxiter_stage,
        )
        p_opt = softmax(y_opt)
        a_full = np.repeat(p_opt, L) / L   # length n, sum = 1
        score_opt = fast_score(a_full, fft_len_conv, n)

        if score_opt < seg_best_score - 1e-12:
            seg_best_score = score_opt
            seg_best_a = a_full.copy()
            best_y_seg = y_opt.copy()
            print(f"[seg] mode={mode} → new best block score = {seg_best_score:.6f}")

        # ----- quick local refinements (full‑resolution) -----
        t_dead = min(deadline, time.time() + 0.7)
        seg_best_a, seg_best_score = targeted_hill_climb(
            a=seg_best_a,
            cur_score=seg_best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y_full = np.log(seg_best_a + 1e-12)

        r_dead = min(deadline, time.time() + 0.7)
        seg_best_a, seg_best_score = hill_climb(
            a=seg_best_a,
            cur_score=seg_best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=8000,
            max_move_frac=0.22,
        )
        best_y_full = np.log(seg_best_a + 1e-12)

    # Adopt the best block‑wise candidate if it beats the previous best
    if seg_best_score < best_score - 1e-12:
        best_a = seg_best_a.copy()
        best_score = seg_best_score
        best_y_full = np.log(best_a + 1e-12)
        print(f"[adopt] block solution improves to {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Fine‑grained polishing with full‑dimension L‑BFGS (few passes)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        fine_lam_factors = np.geomspace(0.5, 1e-12, num=30)
        y_opt_fine = schedule_optimize(
            y_start=best_y_full,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=fine_lam_factors,
            maxiter_per_stage=400,
        )
        a_fine = softmax(y_opt_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score - 1e-12:
            best_a = a_fine.copy()
            best_score = score_fine
            best_y_full = y_opt_fine.copy()
            print(f"[fine] L‑BFGS refinement → {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent on the exact max (robust)
    # --------------------------------------------------------------
    if time.time() < deadline - 3.0:
        sub_dead = deadline - 2.0
        best_a, best_score = subgrad_descent(
            a=best_a,
            cur_score=best_score,
            n=n,
            fft_len=fft_len_conv,
            deadline=sub_dead,
            max_iter=5000,
            step0=0.28,
        )
        best_y_full = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # 4) Final aggressive hill‑climb with any remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y_full = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # 5) Return the best sequence found
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    # Official evaluation (for logging only)
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
