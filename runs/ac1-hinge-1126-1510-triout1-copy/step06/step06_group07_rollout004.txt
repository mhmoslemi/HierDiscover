"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm overview
------------------
* Work with a probability vector `a` (∑a = 1).  All later operations keep the
  vector on the simplex via Euclidean projection.
* Global optimisation:
    - Randomly initialise many “log‑soft‑max” vectors `y`.
    - Perform a continuation of L‑BFGS on a smooth‑max surrogate of the
      auto‑convolution peak.  The surrogate uses a geometric λ‑schedule
      reaching very small λ (≈1e‑14·max(conv)).
    - The smoothing automatically yields a gradient that can be evaluated
      cheaply with FFTs (four transforms per gradient call).
* Local refinement:
    - Aggressive targeted hill‑climb that removes mass from the index that
      contributes most to the current worst convolution entry.
    - Random hill‑climb to escape tiny plateaus.
    - Projected sub‑gradient descent on the exact max (the true objective).
* The whole loop runs until the supplied time budget (≈1000 s) expires.
  The best vector seen (lowest fast‑score = 2·n·max(conv)) is returned.
* The best sequence is stored in the global name `height_sequence_1` for
  possible reuse by subsequent runs.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
    entropy_weight: float = 0.0,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution (quadratic) and its gradient
    w.r.t. the unconstrained log‑soft‑max parameters ``y``.
    An optional entropy regulariser (weight ≥ 0) can be added to encourage
    spreading of the distribution.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a -----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    # ----- optional entropy regularisation -----
    if entropy_weight != 0.0:
        # H(a) = - Σ a_i log a_i  =>  ∂H/∂y_i = a_i (y_i - μ_y)
        mu_y = np.dot(a, y)
        grad_H = a * (y - mu_y)            # gradient of H w.r.t. y
        # Objective: smooth_max - entropy_weight * H
        # => grad = grad_smoothmax - entropy_weight * grad_H
        grad_y -= entropy_weight * grad_H

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
    entropy_weight: float = 0.0,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam, entropy_weight),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (via FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # choose a destination with a tiny partner under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max (no smoothing)
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent w.r.t. the exact max of the auto‑convolution.
    Each iteration identifies the shift achieving the maximum, computes the
    corresponding gradient (2·a shifted), takes a step opposite to it and
    projects back onto the probability simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    iter_no = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        iter_no += 1
        # compute convolution (FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for shift k:
        grad = np.zeros_like(best)
        # grad_i = 2 * a_{k-i} when index valid
        # this is a reversed shifted copy of a
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices i in [lo, hi] have a partner j = k - i in [0, n-1]
        grad[lo : hi + 1] = 2.0 * best[k - np.arange(lo, hi + 1)]

        # decreasing step size
        step = step0 / np.sqrt(iter_no)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional stochastic acceptance to escape plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score

    return best, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative) sequence with a low evaluation
    score.  The function respects the supplied time budget and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)

    # Reserve a small safety margin for final evaluation and housekeeping.
    deadline = time.time() + budget_s - 4.0

    # --------------------------------------------------------------
    # Choose problem size – reuse a previously‑found construction if
    # possible, otherwise start from scratch.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)
        # eager to enlarge the search space: pad to a larger length.
        target_n = max(8000, prev_arr.size * 2)
        n = int(target_n)
        a0 = np.zeros(n, dtype=float)
        start = (n - prev_arr.size) // 2
        a0[start : start + prev_arr.size] = prev_arr
    else:
        n = 8000
        a0 = np.full(n, 1.0 / n)

    # Normalise to probability simplex (∑ = 1) – this is the domain used
    # by the fast surrogate and the optimiser.
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # FFT sizes (constant throughout the run)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Global search – continuation L‑BFGS with many diverse restarts
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-14, num=60)   # finer λ schedule
    maxiter_per_stage = 900                         # balanced work per stage
    entropy_weight = 0.08                           # small entropy push

    iteration = 0
    # Keep attempting new restarts until the budget is almost exhausted.
    while time.time() < deadline - 12.0:
        # ---- pick a diverse starting point ---------------------------------
        mode = rng.choice(
            ["random", "exp", "power", "edge", "best", "mirror", "sine"]
        )
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "best":
            y0 = best_y + rng.normal(scale=0.07, size=n)
        elif mode == "mirror":
            half = n // 2
            left = rng.random(half)
            left /= left.sum()
            a0 = np.concatenate([left, left[::-1]])
            if n % 2 == 1:
                a0 = np.append(a0, 0.0)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # mode == "sine"
            x = np.linspace(0, np.pi, n)
            a0 = (np.sin(x) ** 2) + 0.01
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)

        # ----- continuation L‑BFGS -----------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
            entropy_weight=entropy_weight,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode} → new best fast score = {best_score:.6f}"
            )

        # ----- targeted hill‑climb (quick, aggressive) --------------------
        t_dead = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- random hill‑climb (helps escape very small plateaus) -----
        r_dead = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=8000,
            max_move_frac=0.22,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Refine with projected sub‑gradient descent on the true max
    # --------------------------------------------------------------
    sub_grad_deadline = deadline - 2.0
    best_a, best_score = subgrad_descent(
        a=best_a,
        cur_score=best_score,
        n=n,
        fft_len=fft_len_conv,
        deadline=sub_grad_deadline,
        max_iter=5000,
        step0=0.30,
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Final aggressive random hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.26,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    # The harness will have imported evaluate_sequence already.
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
