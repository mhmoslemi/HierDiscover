def propose_candidate(seed=42, budget_s=1000, **kwargs):

    deadline = time.time() + budget_s - 2
    # Determine n
    init_seq = globals().get('height_sequence_1')
    # We'll set n = length of init_seq if >=4000 else 8000 (but cap 10000)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(tmp.size, 10000)
        else:
            n = 8000
    else:
        n = 8000

    # length set and N.

    # Setup FFT lengths
    fft_len_conv = next_pow2(2*n - 1)
    fft_len_g = next_pow2(3*n - 2)

    # Initialize best_a
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(0, tmp)
            best_a = best_a / best_a.sum()
        else:
            # perhaps differs length, then we discard
            best_a = np.full(n, 1.0/n)
    else:
        best_a = np.full(n, 1.0/n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, score={best_score:.6f}")

    # 1. L-BFGS restarts (pre-optimization)
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=70))  # more fine schedule

    # Determine amount of time to allocate for restarts
    restart_deadline = deadline - 300  # reserve for subsequent phases

    iteration = 0
    modes = ["random","exp","exp_sym","power","edge","arcsine","sidon","dirichlet","best"]
    while time.time() < restart_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # Run schedule_optimize
        y_opt = schedule_optimize(y0, n, fft_len_conv, fft_len_g, lam_schedule,

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} → new best fast score = {best_score:.6f}")

        # Hill climb brief
        hc_deadline = min(deadline, time.time() + 1.2)
        best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n,
                                        hc_deadline, max_iter=4000)

        iteration += 1

    # 2. Multi-active subgradient descent
    if time.time() < deadline - 150:

    # 3. LP Improvement loop
    while time.time() < deadline - 50:
        conv = conv_fft(best_a, fft_len_conv, n)
        M = conv.max()
        # Determine active constraints: within a relative tolerance
        tol = max(1e-12, (M - conv.mean()) * 5e-9)
        active = np.where(conv >= M - tol)[0]

            # fill row
            # initialize zero row
            row_vec = np.zeros(n, dtype=float)
            j_indices = row - np.arange(i_start, i_end+1)
            row_vec[j_indices] = best_a[i_start:i_end+1]
            A_ub[idx] = row_vec

        # Solve LP: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n, dtype=float)  # minimize negative sum
        b_ub = np.full(m, M, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]

        # Use high accuracy method
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 1.0})
        except Exception as e:
            print(f"[LP] failed: {e}")
            break

        if not res.success:
            # maybe LP infeasible; break
            print(f"[LP] no success: {res.message}")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        # Normalize b to sum 1 (simplex)
        b_norm = b_opt / sum_b

        # Combine a and b_norm
        # Choose step size adaptively
        t = 0.3
        improved = False
        while t > 1e-4:
            candidate = (1 - t) * best_a + t * b_norm
            cand_score = fast_score(candidate, fft_len_conv, n)
            if cand_score < best_score - 1e-12:
                best_a = candidate
                best_score = cand_score
                best_y = np.log(best_a + 1e-12)
                improved = True
                print(f"[LP] step t={t:.4f} improved score={best_score:.6f}")
                break
            t *= 0.5
        if not improved:
            # No improvement, break or continue with smaller t?
            # break loop if not improving to avoid endless.
            break

    # 4. Final hill climb
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n,
                                        deadline, max_iter=20000)
        print(f"[final HC] finished, final fast score={best_score:.6f}")

    # final evaluation
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence score={final_val:.6f}")
    return final_seq

"""
Propose a non‑negative step function with a low evaluate_sequence score.

Algorithm
---------
1. Choose a problem size `n` (≈ 5 000–8 000).  If a previous solution
   (`height_sequence_1`) exists we reuse its length, otherwise we use 8000.
2. Initialise a pool of diverse starting points on the probability simplex:
   * random Gaussian logits,
   * exponential / symmetric exponential,
   * power‑law,
   * edge‑heavy,
   * arcsine,
   * sparse Sidon‑like set,
   * Dirichlet(0.1) draw,
   * a **sparse random** pattern (≈ √n non‑zeros),
   * a perturbation of the current best.
3. For each start run a *continuation* L‑BFGS optimisation on the smooth
   log‑sum‑exp surrogate of the max‑convolution.  The smoothing factor λ
   is decreased geometrically over 70 stages; each stage performs up to
   1200 L‑BFGS iterations.  The result is projected back onto the simplex
   via a soft‑max.
4. After each L‑BFGS run a short hill‑climb (≈ 1 s) is performed that moves
   a random fraction of mass between two coordinates and accepts only
   improvements of the fast objective.
5. When the allotted time for the L‑BFGS phase is exhausted we switch to a
   *projected sub‑gradient* method that works on the true max‑convolution.
   At every iteration we average the sub‑gradients of all (almost) active
   convolution entries and use a diminishing step‑size with back‑tracking.
6. A final aggressive hill‑climb uses the remaining time budget.
7. The best probability vector found is returned as a plain list of
   non‑negative floats; the external harness evaluates the exact score.

The method builds on the previous state‑of‑the‑art code but adds:
* a richer set of initialisations (including a sparse random pattern),
* a finer continuation schedule and more L‑BFGS iterations,
* an extended sub‑gradient phase,
* a slightly more aggressive hill‑climb.
These changes give the optimiser more freedom to escape local minima
and usually improve the final bound.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv.real


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    score = 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of the maximum)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where

        f = 2·n·smooth_max(a*a)    with a = softmax(y)

    and grad_y is the gradient w.r.t. the unrestricted vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                # now max = 0
    exp_shift = np.exp(shifted / lam)     # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a aᵀ)·g_vec
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sparse_random":
        # pick O(sqrt(n)) random positions and give them random mass
        k = max(2, int(np.sqrt(n) * rng.uniform(1.5, 3.0)))
        idx = rng.choice(n, size=k, replace=False)
        a0 = np.zeros(n)
        a0[idx] = rng.random(k) + 1e-3
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only moves that strictly improve the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the average of the sub‑gradients of all (almost) active entries,
    and a back‑tracking line search to guarantee monotone improvement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        # build weight vector w (uniform over active indices)
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                         # d (a_i a_j) / d a_i = 2 a_j

        # step size (diminishing) with simple back‑tracking
        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5

        if not improvement:
            # keep moving even if no improvement (helps to escape plateaus)
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise the best candidate on the simplex.
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(0.0, tmp)
            best_a = best_a / best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the continuation L‑BFGS stage.
    # --------------------------------------------------------------
    # a finer geometric schedule and a larger per‑stage iteration budget
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=70))
    iteration = 0
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "dirichlet",

    # reserve ~260 s for the multi‑active sub‑gradient phase
    while time.time() < deadline - 260.0:

        # short hill‑climb after each L‑BFGS run

    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
