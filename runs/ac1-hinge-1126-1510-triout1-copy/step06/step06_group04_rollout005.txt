def annealing_hill_climb(a, cur_score, fft_len, n, deadline, init_temp=0.1, cooling_rate=0.9999, max_iter=50000, max_move=0.3):
    best = a.copy()
    best_score = cur_score
    T = init_temp
    while time.time() < deadline and max_iter>0:
        max_iter-=1
        i,j = rng.integers(0,n,2)
        if i==j: continue
        delta = best[i] * rng.random() * max_move
        if delta <=0: continue
        # propose move
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        # optionally small random jitter around other indices? Not needed.
        new_score = fast_score(new, fft_len, n)
        # accept if better or with probability exp(-(new_score - best_score)/T)
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score)/T):
            best = new
            best_score = new_score
            # optionally update best overall? Keep separate best_original
        T *= cooling_rate
        # optionally reheat if stuck: if iteration%1000=0: T = init_temp
    return best,best_score

"""
Search for a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (∑a = 1); the evaluation is scale‑invariant.
* Phase 1 – many short L‑BFGS runs on a smoothed “log‑sum‑exp” approximation of
  the max‑convolution (continuation on the smoothing parameter).  Each run
  starts from a randomised initial vector (several heuristics are used).
* After each L‑BFGS run a quick greedy hill‑climb is applied.
* Phase 2 – accelerated projected sub‑gradient descent with momentum on the
  true objective f(a)=2·n·max(a∗a).  The step size decays as 1/√t.
* Phase 3 – a long simulated‑annealing hill‑climb that can accept temporary
  uphill moves; this helps escape the remaining local minima.
* Phase 4 – a very fine L‑BFGS refinement with an almost zero smoothing
  parameter, followed by a final aggressive greedy hill‑climb.
* The best sequence found is returned.  All helpers are top‑level functions
  (no closures) and the code respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-16
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # w = d smooth_max / d conv
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # middle n entries correspond to ∂conv/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes:
        random, exp, power, edge, sidon, binary, bump, harmonic.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # simple greedy Sidon set construction
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen) if chosen else 1
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def annealing_hill_climb(

    deadline: float,
    max_iter: int = 100_000,
    init_temp: float = 0.2,
    cooling: float = 0.9999,
    max_move: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that can accept uphill moves.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score_local = cur_score
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = fast_score(new, fft_len, n)
        # accept if better, otherwise with probability exp(-(Δ)/T)
        if new_score < cur_score_local or rng.random() < np.exp(-(new_score - cur_score_local) / max(T, 1e-16)):
            cur = new
            cur_score_local = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
        T *= cooling
        # occasional reheating if stuck for long
        if T < 1e-5:
            T = init_temp
    return best, best_score


def accelerated_subgradient(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    beta: float = 0.9,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective with momentum.
    """
    a = a_start.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # sub‑gradient of the max‑convolution entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # diminishing step size
        eta = eta0 / np.sqrt(it)
        v = beta * v + eta * g
        a_new = a - v
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Problem size (use a reasonably large n but not huge)
    # --------------------------------------------------------------
    default_n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 9000)

    n = default_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / tmp.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS (≈ 35 % of the time budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=25))
    phase1_deadline = start_time + 0.35 * budget_s
    iteration = 0
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "binary",
        "bump",
        "harmonic",
        "best",
    ]

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # quick greedy polishing
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=1500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – accelerated projected sub‑gradient descent
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 30.0, start_time + 0.70 * budget_s)
    print("[subgrad] starting accelerated sub‑gradient descent")
    best_a, best_score = accelerated_subgradient(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.5,
        beta=0.9,
        max_iters=1_000_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final annealing hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        final_hc_deadline = deadline
        print("[final_hc] starting simulated‑annealing hill‑climb")
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_hc_deadline,
            max_iter=200_000,
            init_temp=0.15,
            cooling=0.99995,
            max_move=0.35,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 4 – very fine L‑BFGS refinement (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("[fine] starting final L‑BFGS refinement")
        fine_lam_schedule = list(np.geomspace(1e-4, 1e-12, num=20))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=600,
        )
        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            best_y = y_fine.copy()
            print(f"[fine] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
