def adam_optimize(y0, n, fft_len_conv, fft_len_g, lam, steps=10000, alpha=0.001, beta1=0.9, beta2=0.999, eps=1e-8, deadline=None):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_y = y.copy()
    best_score = fast_score(softmax(y), fft_len_conv, n)
    while (deadline is None or time.time() < deadline) and t < steps:
        t += 1
        f, g = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        y -= alpha * m_hat / (np.sqrt(v_hat) + eps)
        # project to probability simplex after conversion to a
        a = softmax(y)
        # enforce simplex? Softmax already sums to 1
        # optionally can re-normalize: a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_y = y.copy()
    return best_y, best_score

def gradient_based_hill_climb(a, n, fft_len, deadline, max_iter=20000, step_frac=0.1):
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    rng = np.random.default_rng()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        # compute conv and max index
        conv = conv_fft(best_a, fft_len, n)
        m = int(np.argmax(conv))
        # compute subgradient vector
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(best_a)
        g[mask] = 2.0 * best_a[counterpart[mask]]
        # source indices with high g
        # compute probabilities proportional to g * a maybe (higher gradient and more mass)
        prob_source = g * best_a
        if prob_source.sum() == 0:
            # fallback random
            i = rng.integers(n)
        else:
            prob_source = prob_source / prob_source.sum()
            i = rng.choice(n, p=prob_source)
        # target choose low g
        prob_target = (np.max(g) - g) * (1 - best_a) # encourage low gradient and low mass maybe
        if prob_target.sum() == 0:
            j = rng.integers(n)
        else:
            prob_target = prob_target / prob_target.sum()
            j = rng.choice(n, p=prob_target)
        if i == j:
            continue
        delta = step_frac * best_a[i] * rng.random()
        if delta <= 0:
            continue
        best_a[i] -= delta
        best_a[j] += delta
        new_score = fast_score(best_a, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best_a[i] += delta
            best_a[j] -= delta
    return best_a, best_score
```

def simulated_annealing(a, n, fft_len, deadline, init_temp=0.01, decay=0.995, max_iter=100000):
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    cur_a = a.copy()
    cur_score = best_score
    temp = init_temp
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose delta proportion
        delta = cur_a[i] * rng.random() * 0.2
        if delta <= 0:
            continue
        cur_a[i] -= delta
        cur_a[j] += delta
        new_score = fast_score(cur_a, fft_len, n)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / temp):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta
        temp *= decay
        if temp < 1e-8:
            temp = 1e-8
    return best_a, best_score
```

=================

Search for a non‑negative step function (list of floats) that minimises
``evaluate_sequence``.  The objective is proportional to

    2·n·max(conv(a,a))  with  sum(a)=1

so we work on the probability simplex.  The algorithm consists of:

1. **Initialisation** – a cheap grid search over several parametric
   families (beta‑like, exponential, edge‑heavy, arcsine, Sidon) to
   obtain a decent starting distribution.

2. **Continuation L‑BFGS** – minimise a smooth (log‑sum‑exp) surrogate
   of the max‑convolution.  The smoothing factor is reduced
   geometrically (≈60 stages).  Many random restarts (including a small
   perturbation of the current best) are performed while we have budget.

3. **Gradient‑guided hill climbing** – using the true sub‑gradient of the
   max‑convolution we move mass from indices that contribute most to the
   current worst lag to indices that contribute least.  Only improving
   moves are kept.

4. **Simulated annealing** – a temperature‑controlled random walk that
   allows occasional uphill moves, helping to escape shallow local minima.

5. **Final polishing** – a short generic hill‑climb (random pair moves)
   and an evaluation of the true score.

The whole routine respects the time budget (default 1000 s) and prints
progress information.  The returned list is the best distribution found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (output sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution -----
    # Compute g_vec = ∂conv/∂a (size n)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: runs L‑BFGS for each λ in ``lam_schedule``.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators (same as in the reference code)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

# Local optimisation utilities

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j,
    accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def gradient_hill_climb(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
    step_frac: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill climbing based on the true sub‑gradient of
    max(conv).  At each step we move a small amount of mass from a high‑gradient
    index to a low‑gradient one; only improving moves are kept.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of max at index m:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[counterpart[mask]]

        # source indices (large gradient * current mass)
        src_weights = g * best
        src_sum = src_weights.sum()
        if src_sum == 0:
            i = rng.integers(0, n)
        else:
            src_prob = src_weights / src_sum
            i = rng.choice(n, p=src_prob)

        # target indices (small gradient, we also bias toward places with little mass)
        tgt_weights = (np.max(g) - g) * (1.0 - best)
        tgt_sum = tgt_weights.sum()
        if tgt_sum == 0:
            j = rng.integers(0, n)
        else:
            tgt_prob = tgt_weights / tgt_sum
            j = rng.choice(n, p=tgt_prob)

        if i == j:
            continue

        delta = step_frac * best[i] * rng.random()
        if delta <= 0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.015,
    decay: float = 0.997,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk allowing occasional uphill moves.
    """
    cur = a.copy()
    cur_score = fast_score(cur, fft_len, n)
    best = cur.copy()
    best_score = cur_score
    temp = init_temp
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction (up to 15 %) of mass from i to j
        delta = cur[i] * rng.random() * 0.15
        if delta <= 0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        delta_score = new_score - cur_score

        accept = False
        if delta_score < 0:
            accept = True
        else:
            if rng.random() < np.exp(-delta_score / temp):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        temp = max(temp * decay, 1e-9)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    ``evaluate_sequence`` score.  The routine respects the time budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # Prefer to reuse a previously discovered long sequence if available.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size >= 4000:
            n = min(int(arr.size), 10000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate on the simplex
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr / np.sum(arr)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Cheap parametric grid search (Beta‑like families)
    # ------------------------------------------------------------------
    alphas = np.linspace(-0.8, 0.5, 9)
    betas = np.linspace(-0.8, 0.5, 9)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            prob = prob / prob.sum()
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (major part of the budget)
    # ------------------------------------------------------------------
    # A fairly long geometric schedule for the smoothing factor.
    lam_schedule = list(np.geomspace(0.5, 1e-10, num=60))
    iteration = 0
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]

    # Reserve enough time for the later refinement phases.
    deadline = time.time() + budget_s - 1.5
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode:<7} → improved fast_score = {best_score:.6f}")

        # brief greedy hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Gradient‑guided hill climbing (true max‑conv)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        gc_deadline = deadline - 70.0
        print("[grad‑HC] start")
        best_a, best_score = gradient_hill_climb(
            best_a,
            fft_len_conv,
            n,
            gc_deadline,
            max_iter=25000,
            step_frac=0.10,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[grad‑HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 40.0:
        sa_deadline = deadline - 35.0
        print("[SA] start")
        best_a, best_score = simulated_annealing(
            best_a,
            fft_len_conv,
            n,
            sa_deadline,
            init_temp=0.018,
            decay=0.996,
            max_iter=200_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[SA] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final generic hill‑climb (random pair moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30_000,
        )
        print(f"[final HC] finished → fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
