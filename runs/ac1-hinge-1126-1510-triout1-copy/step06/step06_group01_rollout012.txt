def adam_optimize_logits(y, lam, n_iters, fft_len_conv, fft_len_g, n, betas=(0.9,0.999), lr=0.1):
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    eps = 1e-8
    for t in range(1, n_iters+1):
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
        m = betas[0]*m + (1-betas[0])*grad
        v = betas[1]*v + (1-betas[1])*(grad**2)
        m_hat = m / (1 - betas[0]**t)
        v_hat = v / (1 - betas[1]**t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y

def adam_optimize_logits(y0, lam, n_iters, fft_len_conv, fft_len_g, n, lr=0.1):

        v_hat = v/(1-beta2**t)

def beta_distribution(n, alpha):
    # symmetric Beta(alpha, alpha) on discrete grid.
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a = (x * (1 - x) + eps) ** (alpha - 1)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

"""
Propose a low‑evaluation step sequence.

Algorithm outline
-----------------
1. **Initial candidates** – a few handcrafted shapes (uniform,
   symmetric Beta/U‑shaped families, triangular and random).
   The best among them is kept as a starting point.

2. **Heavy L‑BFGS continuation** – many random‑restart runs on a
   log‑sum‑exp smoothing of the max‑convolution.  The smoothing
   factor λ is decreased in stages (continuation) and each stage
   is solved with L‑BFGS‑B.

3. **Adam on the smoothed max** – we switch to first‑order Adam
   optimisation on the same smoothed objective; this gives a
   finer local search while staying on the probability simplex
   (a = softmax(y)).

4. **L₂‑uniform descent** – minimise the squared error to a perfectly
   flat convolution (the uniform vector).  The gradient of the
   L₂‑loss is a simple convolution, so each iteration costs only a
   pair of FFTs.  After every step we project back onto the simplex.

5. **Projected sub‑gradient descent with momentum** – directly
   optimises the true max‑convolution using the sub‑gradient obtained
   from the (almost) active constraints.

6. **Final greedy hill‑climb** – deterministic moves that transfer a
   small fraction of mass between two coordinates while strictly
   improving the objective.

All stages run until the allocated time budget (≈1000 s) expires.
The best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised (∑a=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # Rare numerical issues – just skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: relocate a small random fraction of mass from one
    index to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(alpha, alpha) density sampled on n points.
    For alpha<1 the distribution is U‑shaped (more mass at the ends),
    for alpha>1 it is bell‑shaped.
    """
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n  # centre of each bin
    a = (x * (1.0 - x) + eps) ** (alpha - 1.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def adam_optimize_logits(
    y0: np.ndarray,
    lam: float,
    n_iters: int,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Adam optimisation on the log‑sum‑exp smoothed objective.
    Operates on logits y (a = softmax(y)).
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for t in range(1, n_iters + 1):
        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


def l2_uniform_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Minimises ‖a*a – u‖₂² where u is the uniform vector.
    Simple gradient descent with projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    lr = 0.2
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        diff = conv - target

        # gradient of ½‖diff‖₂²  w.r.t. a : 4 * (diff ⊛ a_rev)
        a_rev = a[::-1]
        fm = np.fft.fft(diff, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad = 4.0 * np.real(np.fft.ifft(fm * fa_rev))[n - 1 : n - 1 + n]

        # simple step with projection
        a_new = simplex_projection(a - lr * grad)

        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            lr = min(lr * 1.03, 1.0)  # slightly enlarge step if improvement
        else:
            lr *= 0.5  # otherwise shrink

        a = a_new
        it += 1
        if it % 500 == 0:
            lr = max(lr * 0.9, 1e-4)  # gradual decay

    return best_a, best_score

    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        a_opt, score_opt = subgradient_optimize_momentum(best_a, phase4_end, fft_len_conv, fft_len_g, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print("[sub‑gradient] new best =", best_score)

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # ------------------------------------------------------------------
    cur_score = fast_score(best_a, fft_len_conv, n)
    best_a, best_score = hill_climb(best_a, cur_score, fft_len_conv, n, deadline, max_iter=12000)
    print("[final hill‑climb] best score =", best_score)

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print("[final] evaluated score =", final_eval)
    return final_seq
