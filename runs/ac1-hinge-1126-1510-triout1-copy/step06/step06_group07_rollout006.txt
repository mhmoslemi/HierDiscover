"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm Overview
------------------
* **Problem size** – Use a relatively large n (8000) to give the optimizer more
  freedom while keeping FFT work cheap (FFT length ≈ 2^15).
* **Parameterisation** – Work on unconstrained log‑softmax variables `y`
  (so the concrete sequence `a = softmax(y)` is always a probability vector:
  non‑negative and sums to 1).  This avoids any explicit projection after each
  gradient step.
* **Global optimisation** – Continuation L‑BFGS on a smooth‑max surrogate of
  the true objective.  The smoothing parameter `λ` follows a geometric schedule
  from a generous value (≈ 0.5·max(conv)) down to ≈ 1e‑14·max(conv).  Each stage
  runs many L‑BFGS iterations; several restarts with diverse initialisations
  are performed until the time budget expires.
* **Local refinements** – After each L‑BFGS run we apply three deterministic
  hill‑climbs:
    - `targeted_hill_climb` focuses on the currently most offending shift.
    - `random_hill_climb` makes random mass‑transfer moves.
    - `adam_optimizer` runs a few Adam steps on the smooth‑max surface.
* **Final polishing** – A pure sub‑gradient descent on the exact max with a
  diminishing step size and a final aggressive random hill‑climb.
* **State reuse** – The best sequence found is stored globally as
  `height_sequence_1` for reuse across runs.

The method is deliberately lightweight (pure NumPy & SciPy) and respects the
1000 s time limit by constantly checking the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:                     # pragma: no cover
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:                # pragma: no cover
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` (sum(a)=1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution (quadratic) and its gradient
    w.r.t. the log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # λ is tiny when max_c is tiny – we keep a tiny floor to avoid div‑by‑0
    lam = max_c * lam_factor + 1e-14

    # ----- smooth max -----
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a -----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    # We compute the cross‑correlation by FFT.
    a_rev = np.fft.fft(a[::-1], fft_len_g)
    w_fft = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(a_rev * w_fft))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 1500,
) -> np.ndarray:
    """
    Continuation L‑BFGS over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ------------------------------------------------------------
# Hill‑climb utilities
# ------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move_frac: float = 0.22,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            # reject
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.28,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that attacks the current worst convolution shift.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # pick a destination index whose partner under shift k is tiny
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand = rng.choice(np.where(partner_weight != np.inf)[0], size=min(7, n), replace=False)
        # avoid self
        cand = cand[cand != i]
        if cand.size == 0:
            continue
        j = int(cand[np.argmin(partner_weight[cand])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ------------------------------------------------------------
# Adam optimisation on smooth‑max (few steps)
# ------------------------------------------------------------


def adam_optimizer(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    max_steps: int = 1200,
    lr: float = 0.07,
) -> np.ndarray:
    """
    Perform a short Adam optimisation on the smooth‑max surrogate.
    Operates directly on the log‑softmax variables `y`.
    """
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    while time.time() < deadline and max_steps > 0:
        max_steps -= 1
        t += 1
        f, g = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.30)
        # Adam update (minimisation)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


# ------------------------------------------------------------
# Projected sub‑gradient descent on the exact max
# ------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 4000,
    step0: float = 0.32,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent w.r.t. the exact max of the auto‑convolution.
    Each iteration identifies the shift achieving the maximum, computes the
    corresponding gradient (2·a shifted), takes a step opposite to it and
    projects back onto the probability simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for shift k:
        grad = np.zeros_like(best)
        # grad_i = 2 * a_{k-i} when index valid
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        # i runs from start..end; j = k-i
        grad[start:end + 1] = 2.0 * best[k - np.arange(start, end + 1)]

        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-13:
            best = cand
            best_score = cand_score
        else:
            # occasional stochastic acceptance to escape plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score


# ------------------------------------------------------------
# Structured initialisers
# ------------------------------------------------------------


def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def init_edge(n: int, rng) -> np.ndarray:
    """High values at both edges, low in the centre."""
    m_edge = max(1, int(n * rng.uniform(0.008, 0.035)))
    edge_val = np.sqrt(n / 2.0) / n          # tuned constant
    centre_val = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, centre_val)
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a /= a.sum()
    return a


def init_double_exp(n: int, rng) -> np.ndarray:
    """Two‑sided exponential decay."""
    rate_left = rng.uniform(0.001, 0.025)
    rate_right = rng.uniform(0.001, 0.025)
    left = np.exp(-rate_left * np.arange(n // 2))
    right = np.exp(-rate_right * np.arange(n // 2)[::-1])
    a = np.concatenate([left, right])
    if a.size > n:               # odd n case
        a = a[:n]
    a /= a.sum()
    return a


def init_triangular(n: int, rng) -> np.ndarray:
    """Linear ramp up then down."""
    peak = rng.uniform(0.5, 2.0) * (1.0 / (n / 2))
    idx = np.arange(n)
    centre = n // 2
    a = np.where(idx <= centre,
                 peak * (idx / centre),
                 peak * ((n - 1 - idx) / (n - 1 - centre)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative floats (the best sequence found before the
    allocated time expires).
    """
    rng = np.random.default_rng(seed)
    # Safety margin for the final call to evaluate_sequence()
    deadline = time.time() + budget_s - 4.0

    # ------------------------------------------------------------------
    # Choose problem size – reuse earlier construction if it looks good
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 3000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 8000
            best_a = init_uniform(n)
    else:
        n = 8000
        best_a = init_uniform(n)

    # Ensure we are on the probability simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = init_uniform(n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute a few deterministic initial vectors and keep the best
    # ------------------------------------------------------------------
    init_modes = ["uniform", "edge", "double_exp", "triangular"]
    for mode in init_modes:
        if mode == "uniform":
            a0 = init_uniform(n)
        elif mode == "edge":
            a0 = init_edge(n, rng)
        elif mode == "double_exp":
            a0 = init_double_exp(n, rng)
        else:  # triangular
            a0 = init_triangular(n, rng)

        score0 = fast_score(a0, fft_len_conv, n)
        if score0 < best_score:
            best_a = a0
            best_y = np.log(best_a + 1e-12)
            best_score = score0
            print(f"[init] mode={mode} → fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS schedule parameters
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-13, num=55)   # finer schedule
    maxiter_per_stage = 1600                         # a bit more work per stage

    iteration = 0
    # ------------------------------- main loop -------------------------------
    while time.time() < deadline - 10.0:
        # ----- pick a diverse starting point -----
        mode = rng.choice(["random", "exp", "power", "edge", "double_exp", "triangular", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.018)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.2)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            a0 = init_edge(n, rng)
            y0 = np.log(a0 + 1e-12)
        elif mode == "double_exp":
            a0 = init_double_exp(n, rng)
            y0 = np.log(a0 + 1e-12)
        elif mode == "triangular":
            a0 = init_triangular(n, rng)
            y0 = np.log(a0 + 1e-12)
        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} → new best fast_score={best_score:.6f}")

        # ----- targeted hill‑climb -----

        best_a, best_score = targeted_hill_climb(

            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=1500,
            move_frac=0.32,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- random hill‑climb -----

        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,

        # ----- short Adam fine‑tuning -----
        a_dead = min(deadline, time.time() + 0.6)

        a_fine = softmax(y_fine)
        score_fine = fast_score(a_fine, fft_len_conv, n)
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine
            best_y = y_fine
            print(f"[{iteration}] Adam fine‑tune → new best fast_score={best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # Projected sub‑gradient descent on the true max
    # --------------------------------------------------------------
    sub_dead = deadline - 2.0
    best_a, best_score = subgrad_descent(
        a=best_a,
        cur_score=best_score,
        n=n,
        fft_len=fft_len_conv,
        deadline=sub_dead,
        max_iter=6000,
        step0=0.30,
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Final aggressive random hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.26,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    # The harness will have imported evaluate_sequence; we call it for logging.
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
