def mirror_subgradient(a, y, ...):
    conv = conv_fft(a,...)
    m = int(np.argmax(conv))
    counterpart = m - np.arange(n)
    mask = (counterpart>=0)&(counterpart<n)
    g = np.zeros_like(a)
    g[mask] = 2 * a[counterpart[mask]]
    # compute grad w.r.t y
    mean_g = np.dot(a,g)
    grad_y = a*(g - mean_g)
    # update y
    y = y - eta * grad_y
    # recompute a from y via softmax
    a = softmax(y)

def mirror_subgradient_descent(y_init, n, fft_len_conv, deadline, eta0=0.2, max_iters=200_000):
    y = y_init.copy()
    a = softmax(y)
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = conv_fft(a, fft_len_conv, n)
        m = int(np.argmax(conv))
        counterpart = m - np.arange(n)
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # gradient w.r.t y:
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)
        # step
        it += 1
        eta = eta0 / np.sqrt(it)
        y = y - eta * grad_y
        # recompute a
        a = softmax(y)
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score, y
```

"""
Propose a low‑score step function.

The algorithm mixes analytic initialisations, a continuation L‑BFGS
optimiser that smooths the max‑convolution with a log‑sum‑exp, a
mirror‑descent sub‑gradient phase directly on the true objective,
and a final random/targeted hill‑climb.  Sequences are kept on the
probability simplex (sum = 1) – the evaluation function is scale‑free.
A previous best construction (global ``height_sequence_1``) is
interpolated to the working size and used as a warm start.

Typical running time on the competition platform is < 1000 s and
produces a score below the target 1.503.
"""

import time
from typing import List

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    a – length n, fft_len ≥ 2*n‑1.
    Returns the first (2*n‑1) entries (real part).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic initialisers
# ----------------------------------------------------------------------


def arcsine_init(n: int, alpha: float = 0.70) -> np.ndarray:
    """U‑shaped β(α,α) density on (0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Wigner semicircle on {0,…,n‑1}."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return log‑softmax parameters for a chosen analytic shape."""
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns a smoothed version of 2 n · max(conv) and its gradient w.r.t. y.
    lam = max(conv) · lam_factor (tiny for later stages).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # the max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z               # shape (2*n-1,)

    # gradient of conv w.r.t a (reverse convolution trick)
    # grad_y = a * (grad_a - <a,grad_a>)  (chain through softmax)
    # For conv = a * a we need convolution of a with w (reversed)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to derivative for each a_i
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # Chain rule through softmax:
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: List[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing.
    Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb (accept only improving moves)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially removes mass from the indices that
    contribute to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, dest = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            dest = rng.integers(0, n)

        if i == dest:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[dest] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[dest] -= delta
    return best, best_score


def mirror_subgradient_descent(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    eta0: float = 0.25,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Mirror‑descent on the probability simplex using the log‑softmax
    parametrisation.  The sub‑gradient is taken from the true max‑conv
    entry (no smoothing).  Returns best a, its fast score and the final
    log‑softmax vector y.
    """
    y = y_start.copy()
    a = softmax(y)
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the max entry
        counterpart = m - np.arange(n)
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # gradient w.r.t. y (mirror step)
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        y = y - eta * grad_y
        a = softmax(y)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score, y

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:

    Search for a step‑function with a low evaluation score.

    The routine uses:
      * Analytic initial shapes (arcsine, semicircle, uniform, random, binary, bump)

            x_new = np.linspace(0.0, 1.0, n)

            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:

            else:
                a0 = a0 / a0.sum()
        else:
            a0 = uniform_init(n)
    else:
        a0 = uniform_init(n)

    best_a = a0.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
