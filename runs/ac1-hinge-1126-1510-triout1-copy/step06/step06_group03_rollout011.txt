"""
A fast, multi‑phase optimizer for the step‑function evaluation problem.

Phase 1 –  Initialise a large distribution (n≈8‑12k) on the probability
simplex.  Several analytic families (Beta, semicircle, uniform, random
etc.) are screened and the one with the smallest fast score
(2·n·max(conv)) is kept.

Phase 2 –  Continuation L‑BFGS on a log‑sum‑exp smoothing of the max‑convolution.
The smoothing parameter λ is decreased geometrically (≈45 stages).  The
variables are unconstrained log‑softmax parameters y (so that a = softmax(y))
automatically respects a≥0 and ∑a=1.

Phase 3 –  Exponentiated‑gradient (mirror descent) on the true objective.
Each iteration picks the currently maximal convolution index k, builds the
sub‑gradient g_i = 2·a_{k‑i} (zero when out of range), and updates
a_i ← a_i·exp(−η·g_i) followed by a simplex projection.  η decays as 1/√t.
This drives the max‑convolution down further while staying in the feasible set.

Phase 4 –  Optional random / targeted hill‑climbing for a final polish.

All phases respect the global 1000 s budget; the final sequence is returned
as a plain list of non‑negative floats (∑a>0).  The code only uses numpy,
scipy and the provided ``evaluate_sequence`` function.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (∑=1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    a is length n, fft_len ≥ 2·n‑1.
    Returns the first 2·n‑1 entries of the real part.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (∑a=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic initialisers
# ----------------------------------------------------------------------


def arcsine_init(n: int, alpha: float = 0.7) -> np.ndarray:
    """β(α,α) density on (0,1) – α≈0.5 gives the classic arcsine law."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)          # positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a log‑softmax start vector for a given mode."""
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max(conv) and its gradient w.r.t. y
    (log‑softmax parametrisation).

    lam = max(conv)·lam_factor  is the smoothing constant.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12          # keep λ>0

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. conv (weights) -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv

    # ----- back‑prop through the convolution -----
    # conv = a * a  →  d conv / d a = 2 * (a convolved with delta)
    # Using FFT: grad_rev = ifft( fft(a[::-1]) * fft(w) )
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]   # derivative for each a_i

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS: start from y_start and run a short L‑BFGS run for each
    λ in ``lam_schedule`` (λ is given as a factor of max(conv)).
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Simple random greedy hill‑climb that moves a fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that preferentially moves mass away from the active max‑entry."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def exponentiated_gradient(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true objective using the (negative) entropy
    geometry (multiplicative updates).  Each iteration uses the exact
    sub‑gradient from the active maximal convolution index.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of conv_k w.r.t. a: g_i = 2·a_{k‑i} (if in range)
        idx = np.arange(n)
        counterpart = k - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Step size schedule
        eta = eta0 / np.sqrt(it)

        # Multiplicative update and projection onto simplex
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # Keep the best seen
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # early exit when we are within a few hundred ms of the deadline
        if time.time() > deadline - 0.3:
            break
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Returns a plain list suitable for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # 1 s safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # If a previous construction is present, respect its length; otherwise use a
    # relatively large size (≈10 k) which provides enough freedom.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 10000                       # default if nothing is provided

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Phase 1 – quick search over a handful of analytic families
    # ------------------------------------------------------------------
    best_a = uniform_init(n)
    best_score = fast_score(best_a, fft_len_conv, n)

    # Scan over a few alphas for the Beta distribution
    for alpha in np.linspace(0.40, 0.90, 11):
        a = arcsine_init(n, alpha=alpha)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a

    # other families
    for mode in ("semicircle", "random", "binary", "bump"):
        if mode == "random":
            a = random_gaussian_init(n, rng)
        elif mode == "binary":
            a = binary_sparse_init(n, rng, density=0.15)
        elif mode == "bump":
            a = gaussian_bump_init(n, rng)
        else:
            a = semicircle_init(n)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a

    print(f"[phase1] initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Continuation L‑BFGS on smoothed max‑convolution
    # ------------------------------------------------------------------
    y_start = np.log(best_a + 1e-12)

    lam_schedule = list(np.geomspace(0.5, 1e-10, num=45))
    # 800 iterations per stage gives ~36 k inner L‑BFGS steps total
    maxiter_per_stage = 800

    # Allow a modest slice of the budget for this phase (~45 %)
    phase2_deadline = start_time + 0.45 * budget_s
    while time.time() < phase2_deadline:
        # occasional perturbation of the current best to escape possible plateaus
        if rng.random() < 0.15:
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = y_start
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[phase2] improved fast score = {best_score:.6f}")

        # short hill‑climb after each L‑BFGS run to polish
        hc_dead = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_dead,
            max_iter=1500,
        )
        best_y = np.log(best_a + 1e-12)
        # refresh starting point for next outer iteration
        y_start = best_y

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient (mirror descent) on the true max
    # ------------------------------------------------------------------
    phase3_deadline = min(deadline - 5.0, start_time + 0.80 * budget_s)
    print("[phase3] starting exponentiated‑gradient")
    best_a, best_score = exponentiated_gradient(
        best_a, fft_len_conv, n, phase3_deadline, eta0=0.5, max_iters=800_000
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[phase3] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final polishing hill‑climbs (random then targeted)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase4] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20_000,
            max_move=0.30,
        )
    if time.time() < deadline:
        print("[phase4] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=10_000,
            max_move=0.20,
        )

    # ------------------------------------------------------------------
    # Final evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
