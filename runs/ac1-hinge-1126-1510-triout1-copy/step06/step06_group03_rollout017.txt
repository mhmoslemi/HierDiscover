def powerlaw_init(n, gamma):
    idx = np.arange(n)
    # symmetric distances from edges
    dist = np.minimum(idx, n - 1 - idx) + 0.5   # shift to avoid zero
    a = (dist) ** (-gamma)
    a = a / a.sum()
    return a

"""
Propose a step‑function (list of non‑negative floats) with a low evaluation score.

Algorithm Overview
------------------
* Choose a problem size n (preferably the length of the supplied
  ``height_sequence_1``; otherwise fallback to 5000).
* Initialise a pool of analytic shapes (beta/U‑shapes, semicircle,
  uniform, random Gaussian, sparse binary, Gaussian bump, power‑law).
  A brief grid‑search over the shape parameters (β‑exponent, power‑law
  exponent) selects the best candidate as a starting point.
* Refine the candidate with a continuation L‑BFGS on a smooth
  log‑sum‑exp approximation of the max‑convolution.  The smoothing
  factor λ is decreased geometrically from 0.5 down to 1e‑12; each stage
  is allowed up to 800 iterations.  Several random restarts are used.
* Perform a long projected sub‑gradient descent on the true objective
  (2·n·max(conv)) with a diminishing step‑size.
* Finish with aggressive hill‑climbing moves:
    – random mass transfers,
    – targeted moves based on the sub‑gradient,
    – a final gradient‑guided exchange.
* Return the best sequence found; its score equals the value returned by
  ``evaluate_sequence`` (the function is supplied by the harness).

The implementation respects the 1000‑second budget, prints progress
messages and uses only top‑level functions (no closures or lambdas).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor  (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # gradient through convolution (a*a)
    # a_rev = a reversed
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def arcsine_init(n: int, alpha: float = 0.75) -> np.ndarray:
    """β(α,α) density on (0,1) – U‑shaped for α<1."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)  # ensure positivity
    a /= a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def powerlaw_init(n: int, gamma: float = 0.5) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (dist_to_edge)^{-γ}."""
    idx = np.arange(n, dtype=float)
    dist = np.minimum(idx, n - 1 - idx) + 0.5
    a = np.power(dist, -gamma)
    a /= a.sum()
    return a

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb moving a fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def target_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from
    indices contributing to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a) = 2·n·max(conv) (sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # Sub‑gradient of the max entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def gradient_targeted_move(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided exchange: move mass from the index with largest
    sub‑gradient to the index with smallest sub‑gradient.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient for max entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[counterpart[mask]]

        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if i == j:
            # fallback to a random exchange
            i, j = rng.integers(0, n, size=2)
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

    candidates.append(binary_sparse_init(n, rng, density=0.15))
    candidates.append(gaussian_bump_init(n, rng))

    # random Gaussian seed (single sample)
    candidates.append(random_gaussian_init(n, rng))

    # evaluate pool
    for a0 in candidates:
        score = fast_score(a0, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a0.copy()

    # ------------------------------------------------------------------
    # Optionally use provided sequence as a seed (if it exists and is long)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a_candidate = tmp / np.sum(tmp)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()

    print(f"[init] start fast score = {best_score:.6f}")

    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # L‑BFGS continuation phase (≈ 60 % of the time budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=70))
    maxiter_per_stage = 800

    main_deadline = start_time + 0.60 * budget_s
    modes = ["arcsine", "semicircle", "uniform", "random", "binary", "bump", "powerlaw"]
    mode_cycle = 0
    iteration = 0

    while time.time() < main_deadline:
        # Choose a start point:
        if iteration % 8 == 0 and iteration > 0:
            # jitter around best found so far
            y0 = best_y + rng.normal(scale=0.015, size=n)
        else:
            mode = modes[mode_cycle % len(modes)]
            mode_cycle += 1
            if mode == "arcsine":
                a0 = arcsine_init(n, alpha=rng.uniform(0.55, 0.90))
            elif mode == "semicircle":
                a0 = semicircle_init(n)
            elif mode == "uniform":
                a0 = uniform_init(n)
            elif mode == "random":
                a0 = random_gaussian_init(n, rng)
            elif mode == "binary":
                a0 = binary_sparse_init(n, rng, density=0.18)
            elif mode == "bump":
                a0 = gaussian_bump_init(n, rng)
            elif mode == "powerlaw":
                a0 = powerlaw_init(n, gamma=rng.uniform(0.2, 1.0))
            else:
                a0 = random_gaussian_init(n, rng)
            y0 = np.log(a0 + 1e-12)

        # Continuation L‑BFGS
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # Light random hill‑climb after each L‑BFGS block
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 5.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a, fft_len_conv, n, subgrad_deadline, eta0=0.35, max_iters=1_200_000
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive final polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=30_000,
            max_move=0.35,
        )
        if time.time() < deadline:
            print("[final] targeted hill‑climb (sub‑gradient based)")
            best_a, best_score = target_hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=12_000,
                max_move=0.30,
            )
        if time.time() < deadline:
            print("[final] gradient‑guided exchange")
            best_a, best_score = gradient_targeted_move(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=10_000,
                max_move=0.30,
            )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
