"""
Hybrid optimizer for the max‑self‑convolution problem.

The algorithm works in three stages (all time‑controlled):
1.  Random restarts of a projected (additive) sub‑gradient descent with an
    adaptive step size.  This method directly minimises the true
    max‑convolution (no smoothing) while staying on the probability simplex.
2.  Exponentiated‑gradient descent (multiplicative updates) starting from the
    best sequence found in stage 1.  The update a ← a·exp(−η·g) automatically
    respects non‑negativity and the simplex constraint.
3.  A long greedy hill‑climb that repeatedly moves a small random fraction of
    mass between two coordinates and accepts only improving moves.

All evaluations of the objective use a fast FFT‑based convolution
(`fast_score`).  The final candidate is checked with the provided
`evaluate_sequence` before being returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast version of the evaluation function (assumes sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random start on the simplex; several families are mixed."""
    mode = rng.choice(['uniform', 'exp', 'gaussian', 'beta', 'linear', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        # symmetric exponential decay from the centre
        rate = rng.uniform(0.001, 0.02)
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-rate * np.abs(x))
    elif mode == 'gaussian':
        sigma = rng.uniform(0.08, 0.3) * n
        x = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-0.5 * (x / sigma) ** 2)
    elif mode == 'beta':
        alpha = rng.uniform(0.2, 3.0)
        beta = rng.uniform(0.2, 3.0)
        t = np.linspace(0, 1, n)
        a = (t ** (alpha - 1)) * ((1 - t) ** (beta - 1) + 1e-12)
    elif mode == 'linear':
        # U‑shaped linear: low in the centre, high at both ends
        x = np.arange(n)
        a = np.minimum(x, n - 1 - x).astype(float) + 1.0
    else:  # edge – a few heavy entries at the boundaries
        a = np.full(n, 1.0 / n)
        frac = rng.uniform(0.008, 0.025)
        m = max(1, int(n * frac))
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Gradient / sub‑gradient utilities
# ----------------------------------------------------------------------


def subgradient_max_conv(a: np.ndarray, fft_len_conv: int, fft_len_g: int):
    """
    Compute a sub‑gradient of max(a*a) with respect to a.
    Returns (g, max_val) where g has the same shape as a.
    """
    n = a.shape[0]
    # forward convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # gradient via correlation of mask with reversed a
    a_rev = a[::-1]
    fw = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    grad_full = np.real(np.fft.ifft(fa_rev * fw))
    # extract the relevant n entries (indices n‑1 … 2n‑2)
    g = 2.0 * grad_full[(n - 1):(n - 1 + n)]
    return g, max_val


def projected_subgradient_optimize(
        a_start: np.ndarray,
        deadline: float,
        n: int,
        fft_len_conv: int,
        fft_len_g: int,
        rng: np.random.Generator,
        init_step: float = 0.5,
        min_step: float = 1e-5):
    """
    Projected sub‑gradient descent on the true L∞ objective.
    Adaptive step‑size: increase on success, decrease on failure.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    eta = init_step
    it = 0
    while time.time() < deadline:
        g, _ = subgradient_max_conv(a, fft_len_conv, fft_len_g)
        a_candidate = simplex_projection(a - eta * g)

        score = fast_score(a_candidate, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate
            # be a little bolder after a win
            eta = min(eta * 1.05, 1.0)
        else:
            # reject – shrink step and occasionally perturb
            eta = max(eta * 0.65, min_step)
            if rng.random() < 0.01:
                a = add_noise(a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        it += 1

        # occasional global perturbation to escape flat regions
        if it % 20000 == 0:
            if rng.random() < 0.3:
                a = add_noise(best_a, sigma=0.02, rng=rng)
                eta = init_step

    return best_a, best_score


def exp_gradient_optimize(
        a_start: np.ndarray,
        deadline: float,
        n: int,
        fft_len_conv: int,
        fft_len_g: int,
        rng: np.random.Generator,
        eta0: float = 0.8):
    """
    Classic exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) automatically stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # compute sub‑gradient of the hard max
        g, _ = subgradient_max_conv(a, fft_len_conv, fft_len_g)

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(
        a_start: np.ndarray,
        init_score: float,
        fft_len_conv: int,
        n: int,
        deadline: float,
        max_iter: int = 200000):
    """
    Greedy hill‑climb: move a random fraction of mass from one coordinate to
    another, accepting only strictly improving moves.
    """
    a = a_start.copy()
    best_score = init_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # propose moving a small fraction of the mass at i to j
        delta = a[i] * rng.random() * 0.08
        if delta <= 0.0:
            it += 1
            continue
        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta
        it += 1
    return a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Starting sequence (use the best known if present, otherwise a Gaussian)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        # Interpolate to the target length if needed
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # Gaussian bump centred in the middle – a solid baseline
        sigma = 0.15 * n
        x = np.arange(n) - (n - 1) / 2.0
        a0 = np.exp(-0.5 * (x / sigma) ** 2)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.4          # tiny safety margin

    phase1_end = start + 0.55 * budget_s      # projected sub‑gradient
    phase2_end = start + 0.80 * budget_s      # exponentiated‑gradient
    # The rest of the budget goes to hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts of projected sub‑gradient descent
    # ------------------------------------------------------------------
    while time.time() < phase1_end:
        # choose a fresh start: either the current best (jittered) or a random family
        if rng.random() < 0.45:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.018), rng=rng)
        else:
            a_start = random_initial(n, rng)

        # give each run a modest slice of the remaining phase‑1 budget
        run_deadline = min(phase1_end, time.time() + rng.uniform(4.0, 12.0))

        cand_a, cand_score = projected_subgradient_optimize(
            a_start, run_deadline, n, fft_len_conv, fft_len_g, rng)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand_a.copy()
            print(f"[proj‑subgrad] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent from the current best
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len_conv, fft_len_g, rng, eta0=0.9)
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (as many iterations as time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=500000)
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
