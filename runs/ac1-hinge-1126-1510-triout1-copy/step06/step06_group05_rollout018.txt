def adam_flatten(a_start, fft_len, n, deadline, lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8, eval_every=500):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    target = 1.0 / (2 * n - 1)
    while time.time() < deadline:
        t += 1
        # compute conv a*a
        conv = conv_fft(a, fft_len, n)
        diff = conv - target
        # compute gradient = 4 * (diff * a_rev) conv slice
        a_rev = a[::-1]
        fd = np.fft.rfft(diff, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fd * fa_rev, fft_len)
        grad = 4.0 * grad_full[(n - 1):(n - 1 + n)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

"""
search for a step‑function that minimises evaluate_sequence

Algorithm
---------
1. Choose a problem size n (reuse an existing long sequence if available,
   otherwise n≈8000).
2. Initialise a probability vector a (sum = 1) – start from the existing
   construction, from a uniform distribution or from a few simple parametric
   families (Beta‑like).  Keep the best by the cheap “fast_score”.
3. Continuation L‑BFGS on a log‑sum‑exp smooth approximation of the
   max‑convolution.  A geometric schedule for the smoothing factor drives
   the optimisation toward the true objective.
4. Projected Adam‑style gradient descent on the squared error
   ‖a∗a − 1/(2n−1)·1‖₂², i.e. we flatten the convolution towards a uniform
   distribution.  The Adam moments accelerate convergence and the Euclidean
   projection enforces the simplex constraints.
5. Optional convex mixing with the uniform distribution.
6. Final greedy hill‑climb (mass moves between two coordinates) for polishing.
7. Return the best sequence (list of floats).  The harness will call
   evaluate_sequence on the result.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv_full)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution ----------------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)        # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j,
    accepting only improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.15  # up to 15 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_flatten(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    lr: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_every: int = 1000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the L2 error
        ½·‖a∗a − 1/(2n−1)·1‖₂²
    which pulls the convolution towards a uniform distribution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    target = 1.0 / (2 * n - 1)

    while time.time() < deadline:
        t += 1
        # convolution a*a
        conv = conv_fft(a, fft_len, n)
        diff = conv - target

        # gradient = 4 * (diff convolved with reversed a)
        a_rev = a[::-1]
        fd = np.fft.rfft(diff, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fd * fa_rev, fft_len)
        grad = 4.0 * grad_full[(n - 1) : (n - 1 + n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = project_onto_simplex(a)

        if t % eval_every == 0:
            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluate_sequence score.  The routine respects the time budget (default
    1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # safety margin for the outer harness
    deadline = time.time() + budget_s - 1.0

    # ------------------------------------------------------------------
    # Choose problem size (reuse a previous construction if possible)
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 12000)          # allow a bit larger n for improvement
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size == n:
            best_a = arr.copy()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_a = best_a / best_a.sum()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Cheap parametric grid search (Beta‑like families)
    # ------------------------------------------------------------------
    alphas = np.linspace(-0.8, 0.5, 7)
    betas = np.linspace(-0.8, 0.5, 7)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=float)
            prob = (i + 1) ** alpha * (n - i) ** beta
            prob = prob / prob.sum()
            sc = fast_score(prob, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = prob.copy()
                print(f"[grid] α={alpha:.2f} β={beta:.2f} → fast_score={sc:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=40))
    maxiter_per_stage = 600
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]

    # Reserve about 300 s for the later refinement stages
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.05, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[lbfgs] mode={mode} → fast_score={sc_opt:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )

    # ------------------------------------------------------------------
    # Adam‑style flattening on the true squared‑error objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        flatten_deadline = deadline - 30.0
        print("[adam] start flattening")
        best_a, best_score = adam_flatten(
            best_a,
            fft_len_conv,
            n,
            flatten_deadline,
            lr=0.2,
            eval_every=800,
        )
        print(f"[adam] finished → fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional mixing with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        uniform = np.full(n, 1.0 / n)
        for alpha in np.linspace(0.0, 1.0, 11):
            cand = (1 - alpha) * best_a + alpha * uniform
            cand = cand / cand.sum()
            sc = fast_score(cand, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[mix] α={alpha:.2f} → fast_score={sc:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=5000,
        )
        print(f"[final HC] finished → fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score={final_val:.6f}")
    return final_seq
