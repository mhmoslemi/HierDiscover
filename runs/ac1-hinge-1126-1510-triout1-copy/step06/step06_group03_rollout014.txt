def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine n
    target_n = 12000
    # If existing height_sequence_1 longer than target_n, cap to its length else target_n.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        orig = np.asarray(init_seq, dtype=float)
        if orig.ndim==1:
            # upsample if needed
            if orig.size >= target_n:
                n = min(orig.size, target_n)
                a0 = orig[:n]
            else:
                # Interpolate to target_n
                n = target_n
                old_x = np.linspace(0,1,orig.size)
                new_x = np.linspace(0,1,n)
                a0 = np.interp(new_x, old_x, orig)
        else:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        n = target_n
        a0 = uniform_init(n)
    else:
        a0 = a0 / a0.sum()
        n = a0.size

    fft_len_conv = next_pow2(2*n-1)
    fft_len_g = next_pow2(3*n-2)
    print(f"[init] n={n}")

    # baseline fast score
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] baseline fast score {best_score:.6f}")

    # scanning arcsine alphas
    alphas = np.linspace(0.35, 0.85, 11)
    for alpha in alphas:
        a = arcsine_init(n, alpha=alpha)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            best_y = np.log(a + 1e-12)
            print(f"[alpha] alpha={alpha:.3f} improved fast score {best_score:.6f}")

    # add random initializations
    for mode in ["random","binary","bump","semicircle","uniform"]:
        y0 = generate_initial_y(mode, n, rng)
        a = softmax(y0)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            best_y = y0.copy()
            print(f"[init] mode={mode} improved fast score {best_score:.6f}")

    # L-BFGS continuation
    # lam schedule
    lam_schedule = list(np.geomspace(0.5, 5e-10, num=30))
    maxiter_per_stage = 400   # moderate
    main_deadline = start + 0.30 * budget_s

    while time.time() < main_deadline:
        # pick random mode or perturbed best
        if rng.random() < 0.2:
            y0 = best_y + rng.normal(scale=0.02, size=n)   # Perturb best
        else:
            mode = rng.choice(["arcsine","semicircle","uniform","random","binary","bump"])
            y0 = generate_initial_y(mode, n, rng)
        y_opt = schedule_optimize(y0, n, fft_len_conv, fft_len_g, lam_schedule,
                                  maxiter_per_stage=maxiter_per_stage)
        a_opt = softmax(y_opt)
        s_opt = fast_score(a_opt, fft_len_conv, n)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[lbfgs] improved fast score {best_score:.6f}")

    # Subgradient descent via exponentiated gradient
    subgrad_deadline = start + 0.80 * budget_s
    a = best_a.copy()
    best_a_sg = a.copy()
    best_score_sg = best_score
    it = 0
    eta0 = 0.5
    while time.time() < subgrad_deadline and it < 500_000:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        # subgradient of max conv entry
        g[mask] = 2.0 * a[counterpart[mask]]
        # exponential update
        eta = eta0 / np.sqrt(it)
        a_new_unnorm = a * np.exp(-eta * g)
        a_new = a_new_unnorm / a_new_unnorm.sum()
        cur_score = fast_score(a_new, fft_len_conv, n)
        if cur_score < best_score_sg:
            best_score_sg = cur_score
            best_a_sg = a_new.copy()
        a = a_new

    best_a = best_a_sg
    best_score = best_score_sg
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] best fast score {best_score:.6f} after {it} iters")

    # Final hill climbing
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n, final_deadline,
                                    max_iter=10000, max_move=0.3)
    best_a, best_score = targeted_hill_climb(best_a, best_score, fft_len_conv, n,
                                            final_deadline, max_iter=5000, max_move=0.25)

    # Evaluate exact
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score {final_val:.6f}")

    return final_seq

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. Choose a problem size (n≈8000–12000).  If a previous construction
   (global ``height_sequence_1``) is available, it is interpolated to the
   target size; otherwise a uniform vector is used.

2. Scan a few analytically‑motivated “U‑shaped’’ initialisations
   (Beta(α,α) with α∈[0.35,0.85] and a few random seeds) and keep the
   best one (lowest fast‐score = 2·n·max(conv) where the sequence is
   normalised to a probability vector).

3. Refine the best candidate with a continuation L‑BFGS on a
   smooth approximation of the max‑convolution:
   f(y)=2·n·smooth_max(conv(softmax(y))).
   The smoothing factor λ is decreased geometrically
   (≈45 stages, λ∈[0.5,10⁻⁹]).

4. Perform a long projected (mirrored) sub‑gradient descent on the
   true objective 2·n·max(conv).  The sub‑gradient is the gradient of the
   maximal convolution entry; the update is multiplicative
   (exponentiated‑gradient) which respects the simplex automatically.

5. Final polishing: a stochastic hill‑climb / simulated‑annealing loop
   that moves a random fraction of mass between two indices.
   Moves are accepted if they improve the fast‑score or with a
   temperature‑dependent probability, allowing the search to escape
   shallow local minima.

6. The resulting sequence is evaluated with ``evaluate_sequence`` and
   returned.  All steps respect a 1000 s wall‑clock budget.

The implementation uses only NumPy and SciPy (for L‑BFGS) and contains
no closures or file I/O, satisfying the competition constraints.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
#  Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) via FFT.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]               # decreasing order
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Initialising families of U‑shaped sequences
# ----------------------------------------------------------------------


def arcsine_init(n: int, alpha: float = 0.70) -> np.ndarray:
    """Beta(α,α) density on (0,1) – U‑shaped for α<1."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)      # ensure positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / np.count_nonzero(mask)
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return log‑softmax parameters (y) for a given mode."""
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
#  Smoothed max‑convolution (L‑BFGS) utilities
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective f(y)=2·n·smooth_max(conv(softmax(y))).

    lam = max(conv)·lam_factor  (the actual smoothing constant).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smooth max
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z                     # shape (2n‑1,)

    # gradient of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]   # length n

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing factor (λ).
    Each stage restarts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
#  Local search helpers (hill‑climb & simulated annealing)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Greedy random hill‑climb that only keeps improving moves."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.005,
    final_temp: float = 1e-6,
    max_iter: int = 20000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Accept uphill moves with a temperature schedule (SA)."""
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # linear cooling
        T = max(final_temp,
                init_temp * (1 - it / max_iter))
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-dE / (T + 1e-12)):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
#  Sub‑gradient descent on the true objective (exponentiated update)
# ----------------------------------------------------------------------


def subgradient_exponential(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.6,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent style (exponentiated) sub‑gradient descent on
    f(a)=2·n·max(conv).  The iterates always stay on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # exponentiated (multiplicative) update
        a_new_unnorm = a * np.exp(-eta * g)
        a_new = a_new_unnorm / a_new_unnorm.sum()

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Returns the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # safety margin – we need a couple of seconds for the final evaluate()
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine problem size (larger n gives more freedom)
    # ------------------------------------------------------------------
    target_n = 12000  # upper bound we are comfortable with
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size > 0:
            # interpolate / upsample to target_n
            if tmp.size >= target_n:
                a0 = tmp[:target_n]
            else:
                old_x = np.linspace(0.0, 1.0, tmp.size)
                new_x = np.linspace(0.0, 1.0, target_n)
                a0 = np.interp(new_x, old_x, tmp)
        else:
            a0 = uniform_init(target_n)
    else:
        a0 = uniform_init(target_n)

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    n = a0.size
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Baseline (uniform) and scan of arcsine alphas
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] baseline fast score = {best_score:.6f}")

    # scan a few alphas for the Beta(α,α) shape
    for alpha in np.linspace(0.35, 0.85, 11):
        a = arcsine_init(n, alpha=alpha)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            best_y = np.log(a + 1e-12)
            print(f"[scan] α={alpha:.3f} improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Random hot‑starts (a few different families)
    # ------------------------------------------------------------------
    for mode in ("random", "binary", "bump", "semicircle", "uniform"):
        y0 = generate_initial_y(mode, n, rng)
        a = softmax(y0)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            best_y = y0.copy()
            print(f"[rand] mode={mode} improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    # geometric λ‑schedule, very fine near zero
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=35))
    maxiter_per_stage = 400
    l_bfgs_deadline = start + 0.30 * budget_s
    iteration = 0
    while time.time() < l_bfgs_deadline:
        # most of the time we perturb the current best, occasionally start fresh
        if rng.random() < 0.2:
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            mode = rng.choice(["arcsine", "semicircle", "uniform",
                               "random", "binary", "bump"])
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        s_opt = fast_score(a_opt, fft_len_conv, n)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[lbfgs] iter={iteration:04d} improved fast score = {best_score:.6f}")
        iteration += 1

    # ------------------------------------------------------------------
    # Exponentiated sub‑gradient descent (≈ 45 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = start + 0.80 * budget_s
    print("[subgrad] starting exponentiated sub‑gradient descent")
    best_a, best_score = subgradient_exponential(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.6,
        max_iters=500_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: hill‑climb + simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] greedy hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
            max_move=0.30,
        )
    if time.time() < deadline:
        print("[final] simulated annealing")
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_temp=0.004,
            final_temp=1e-6,
            max_iter=20000,
            max_move=0.25,
        )

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # Provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
