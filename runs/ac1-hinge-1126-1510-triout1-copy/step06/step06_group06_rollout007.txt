    eps = 1e-8

        m = np.zeros_like(y)
        v = np.zeros_like(y)

            f_val, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm outline
-----------------
* Global optimisation:
  * Multiple random restarts.  For each restart we first run a
    continuation of L‑BFGS on the smooth‑max surrogate (the same
    schedule used in the previous champion) and then polish the
    result with a lightweight Adam optimiser that works directly in the
    log‑soft‑max (probability simplex) space.
* Local refinement:
  * Targeted hill‑climb that moves mass away from the most
    offending pair of indices (the pair that contributes the most to
    the current maximal convolution coefficient).
  * A classic random mass‑transfer hill‑climb (only strictly improving
    moves are kept).
* Escape from plateaux:
  * Simulated‑annealing hill‑climb – occasionally accepts a worse
    move with a temperature that decays over time.
* Final polishing:
  * Projected sub‑gradient descent on the true (unsmoothed) max‑convolution
    objective, followed by a last random hill‑climb.

The code respects the 1000 s time budget, performs no I/O, and leaves the
best sequence in the global variable ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max yielding a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` when ``a`` is already
    normalised to a probability distribution (i.e. sum(a)==1):
        2·n·max(conv(a,a))
    The convolution is performed by FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # propagate gradient back to a (using the adjoint convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # the part that corresponds to the gradient w.r.t. the original a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation of L‑BFGS over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def schedule_optimize_adam(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_stage: int = 120,
) -> np.ndarray:
    """
    Simple Adam optimiser on the same smoothed‑max surrogate.
    Works directly in the log‑soft‑max (unconstrained) space.
    """
    y = y_start.copy()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    for lam in lam_factors:
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        for t in range(steps_per_stage):
            f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (t + 1))
            v_hat = v / (1.0 - beta2 ** (t + 1))
            # learning‑rate loosely scaled with the smoothing factor
            lr = 0.25 * lam
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point ``y`` (log‑soft‑max parameters) for the
    optimiser.  A mixture of heuristics encourages diversity.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "sparse", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "sparse":
        # keep only a few (≈5 %) random positions non‑zero
        mask = rng.random(n) < 0.05
        a0 = np.zeros(n, dtype=float)
        a0[mask] = rng.random(mask.sum())
        if a0.sum() == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    else:  # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that attacks the pair contributing most to the
    current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))  # shift with maximal overlap

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner_idx[mask]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def annealing_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    init_temp: float = 0.02,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing variant of the random hill‑climb.
    Occasionally accepts a worsening move according to a decaying
    temperature schedule.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    temp = init_temp
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            prob = np.exp((cur_score - new_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best = cur.copy()
                best_score = new_score
        else:
            cur[i] += delta
            cur[j] -= delta

        temp = max(temp * decay, 1e-6)
    return best, best_score


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient of the
    (non‑smoothed) max‑operator.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to symmetric half 0 ≤ k ≤ n‑1
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size & initial solution
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the global optimiser
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.6, 1e-9, num=58)      # richer λ‑schedule
    maxiter_per_stage = 600                           # a bit more L‑BFGS work
    steps_per_stage_adam = 100                        # Adam inner steps
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (global + local phases)
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:
        # ---- Randomised start -------------------------------------------------
        y0 = init_y(rng, n, best_a, iteration)

        # ---- L‑BFGS phase -----------------------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )

        # ---- Adam polishing ----------------------------------------------------
        y_opt = schedule_optimize_adam(
            y_start=y_opt,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            steps_per_stage=steps_per_stage_adam,
        )

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] Global opt improved fast score = {best_score:.6f}")

        # ---- Targeted hill‑climb ------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- Random hill‑climb --------------------------------------------------
        rc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=4000,
            max_move_frac=0.18,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- Simulated‑annealing hill‑climb (escape plateaux) ------------------
        sa_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = annealing_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=sa_deadline,
            max_iter=2500,
            init_temp=0.02,
            decay=0.9993,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    step0 = 0.45
    subgrad_iter = 0
    eps = 1e-12
    while time.time() < deadline - 2.0:
        subgrad_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)

        step = step0 / np.sqrt(subgrad_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)
        else:
            step0 *= 0.92

        if subgrad_iter % 250 == 0:
            print(f"[subgrad] iter={subgrad_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with an unrestricted random hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=22000,
            max_move_frac=0.24,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)  # noqa: F821 – supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    print(f"[final] sequence length = {len(final_seq)}")
    return final_seq
