"""
Improved search for a low‑score non‑negative step‑function sequence.

The algorithm proceeds in three phases:

1. **Diverse initialisation** – Build many candidate vectors from
   analytically useful shapes (uniform, exponential, power‑law,
   sinusoidal, Kaiser/DPSS windows, random Dirichlet, edge‑heavy
   ramps).  All candidates are normalised to sum = 1 and the best
   (lowest “fast” score) is kept.

2. **Continuous refinement** – From the current best vector a
   smooth‑max surrogate of the true objective is minimised with a
   multi‑stage L‑BFGS schedule (geometric λ‑schedule).  The
   optimisation works in the log‑soft‑max space, which automatically
   enforces the simplex constraints.

3. **Local hill‑climbing** – Two complementary hill‑climbs are applied:
   *targeted* (moves mass away from the pair of indices that contribute
   most to the maximal convolution entry) and *random* (simple
   mass‑transfer moves).  Both hills are executed as long as time
   permits, finishing with a final polishing run.

The whole search respects the 1000 s budget (a small safety margin is
kept).  The best sequence found is stored in the global variable
``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1] if np.any(cond) else 0
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(a,a))  where the convolution is computed via FFT.
    ``a`` must already be normalised (sum = 1) – the score is scale‑free.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution (via log‑sum‑exp) and its gradient
    w.r.t. the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z

    # back‑prop to a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # pick a target that is “quiet’’ under the same shift
        partner_weight = np.full(n, np.inf)
        in_range = mask
        partner_weight[in_range] = best[partner[in_range]]
        cand_js = rng.integers(0, n, size=8)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point (log‑soft‑max parameters) for L‑BFGS.
    The mode is chosen randomly to keep diversity across restarts.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "sine", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.015, 0.045)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
    elif mode == "sine":
        a0 = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a0[a0 < 0] = 0.0
    else:  # perturb current best
        a0 = best_a.copy()
        a0 += rng.normal(scale=0.04, size=n)
        a0 = np.clip(a0, 0.0, None)
    a0 = np.maximum(a0, 1e-16)
    a0 = a0 / a0.sum()
    return np.log(a0 + 1e-12)

def generate_window_candidates(rng: np.random.Generator, n: int) -> list[np.ndarray]:
    """Return a list of diverse initial sequences (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. exponential families
    for _ in range(5):
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        cand.append(a / a.sum())

    # 3. power‑law families
    for _ in range(5):
        p = rng.uniform(0.3, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        cand.append(a / a.sum())

    # 4. linear ramps (increasing and decreasing)
    a_inc = np.linspace(1e-4, 1.0, n)
    cand.append(a_inc / a_inc.sum())
    a_dec = a_inc[::-1]
    cand.append(a_dec / a_dec.sum())

    # 5. sinusoidal (Hann‑like) window
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.clip(a, 0.0, None)
    cand.append(a / a.sum())

    # 6. Kaiser window (beta sampled)
    try:
        from scipy.signal import windows
        for beta in [2, 4, 6, 8, 10]:
            a = windows.kaiser(n, beta, sym=False)
            a = np.clip(a, 0.0, None)
            cand.append(a / a.sum())
    except Exception:
        pass

    # 7. DPSS (Slepian) windows – if scipy provides them
    try:
        from scipy.signal import windows
        for NW in [2.5, 3.5, 4.5]:
            a = windows.dpss(n, NW, sym=False)
            a = np.abs(a)  # ensure non‑negative
            cand.append(a / a.sum())
    except Exception:
        pass

    # 8. random Dirichlet draws (coarse)
    for _ in range(8):
        alpha = np.ones(n) * rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0)
        cand.append(a / a.sum())

    # 9. Sparse random spikes (useful for exploring extreme shapes)
    for _ in range(5):
        k = rng.integers(3, 12)  # number of non‑zero positions
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k) + 0.1
        a = np.zeros(n)
        a[idx] = vals
        cand.append(a / a.sum())

    # Ensure non‑negativity and numerical cleanliness
    cleaned = []
    for a in cand:
        a = np.maximum(a, 1e-16)
        a = a / a.sum()
        cleaned.append(a)
    return cleaned

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0        # small safety margin

    # ------------------------------------------------------------------
    # Problem size – use the previous construction if available, else 5000
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float).copy()
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise strictly to the probability simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – gather a set of diverse starting points and keep the best
    # ------------------------------------------------------------------
    candidates = generate_window_candidates(rng, n)
    for a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a = a.copy()
            best_score = sc
    print(f"[phase 1] best score after analytic init = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS refinement with many restarts (time‑limited)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.6, 1e-8, num=48)   # fine λ‑schedule
    iteration = 0
    while time.time() < deadline - 5.0:
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=next_pow2(3 * n - 2),
            lam_factors=lam_factors,
            maxiter_per_stage=500,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_a = a_opt.copy()
            best_score = sc_opt
            print(f"[{iteration}] L‑BFGS improved fast score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 2‑a – targeted hill‑climb (quick but powerful)
        # ------------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.33,
        )

        # ------------------------------------------------------------------
        # Phase 2‑b – random hill‑climb (exploration)
        # ------------------------------------------------------------------
        rc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=4000,
            max_move_frac=0.30,
        )

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    step0 = 0.45
    sub_iter = 0
    eps = 1e-12
    while time.time() < deadline - 2.0:
        sub_iter += 1
        # compute max‑convolution and a sub‑gradient (non‑smooth)
        fa = np.fft.fft(best_a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_c = conv.max()
        k_max = int(np.argmax(conv))

        # Map shift index into the (0 … n‑1) range
        if k_max > n - 1:
            k = 2 * n - 2 - k_max
        else:
            k = k_max

        # sub‑gradient of max‑convolution w.r.t. a
        grad = np.zeros_like(best_a)
        if k == 0:
            grad = 2.0 * best_a
        else:
            grad[: n - k] += best_a[k:]
            grad[k:] += best_a[: n - k]

        # Diminishing step size
        step = step0 / np.sqrt(sub_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)   # slightly increase if we make progress
        else:
            step0 *= 0.92                     # otherwise shrink

        if sub_iter % 250 == 0:
            print(f"[subgrad] iter={sub_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – a longer random hill‑climb with any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.35,
        )
        print(f"[final‑hill‑climb] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
