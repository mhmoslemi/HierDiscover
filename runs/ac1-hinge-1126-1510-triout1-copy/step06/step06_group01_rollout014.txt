def solve_lp_for_g(f):
    n = len(f)
    M = np.max(np.convolve(f, f))
    A = build_conv_matrix(f)  # shape (2n-1, n)
    # constraints A @ g <= M
    # variable bounds
    bounds = [(0, None)] * n
    # objective minimize -sum(g) => maximize sum
    c = -np.ones(n)
    # Use high accuracy options
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback
        return None
    g = res.x
    return g

"""
Search for a non‑negative step sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the linear convolution.  The algorithm combines several
heuristics that have proven successful for this problem:

1. **Heavy random‑restart L‑BFGS** on a smooth log‑sum‑exp
   approximation of the max‑convolution.
2. **Monotone LP improvement** (the “alternating optimisation” of the
   description).  For a current distribution f we maximise the total
   mass of a new distribution g under the constraint f∗g ≤ max(f∗f),
   then mix the two distributions.
3. **Short L‑BFGS refinement** (very small smoothing) after the LP.
4. **Projected sub‑gradient descent with momentum**.
5. **Exponentiated‑gradient descent** on the probability simplex.
6. **Final greedy hill‑climb** to polish the result.

All vectors are kept on the simplex (non‑negative, sum = 1) so the
objective reduces to `2·n·max(conv)`.  The code respects the 1000 s
time budget and prints progress messages.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when a is already normalised (∑a=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (full length)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives (same as the previous baseline)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # Very rare; skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the sub‑gradient of
    the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional jitter
        if (it + 1) % 5000 == 0:
            if rng.random() < 0.3:
                sigma = rng.uniform(0.001, 0.008)
                a = a + rng.normal(0.0, sigma, size=n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: relocate a small random fraction of mass from one
    index to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def build_conv_matrix(f: np.ndarray) -> diags:
    """
    Build the (2n‑1)×n non‑negative matrix A such that A @ g = f * g.
    Uses scipy.sparse.diags with one diagonal per offset.
    """
    n = f.shape[0]
    # each column i contributes f[i] on rows i … i+n‑1 → offset = i
    offsets = np.arange(0, n, dtype=int)
    # repeat the vector f for each diagonal
    data = [f.copy() for _ in range(n)]
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def lp_iterative_improvement(
    f: np.ndarray,
    max_iter: int,
    t: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Monotone improvement via the LP described in the problem statement.
    Parameters
    ----------
    f : np.ndarray
        Current distribution (sum = 1).
    max_iter : int
        Maximum number of LP iterations.
    t : float
        Mixing factor 0 < t ≤ 1 used in the update f ← (1‑t)·f + t·ĝ.
    deadline : float
        Absolute time after which the routine must stop.
    Returns
    -------
    best_f, best_score
    """
    n = f.shape[0]
    fft_len = next_pow2(2 * n - 1)

    best_f = f.copy()
    best_score = fast_score(best_f, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current bound M = max(f*f)
        conv_ff = full_conv(best_f, fft_len)
        M = conv_ff.max()

        # build constraint matrix A such that (A @ g) = f * g
        A = build_conv_matrix(best_f)

        # linear program: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None) for _ in range(n)]

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        # free memory as soon as possible
        del A

        if not res.success:
            # LP failed – stop the LP phase
            print("[LP] linprog failed:", res.message)
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break

        # normalise to sum = 1
        g_hat = g / s

        # convex combination
        new_f = (1.0 - t) * best_f + t * g_hat
        # (should already be on simplex, but re‑project to avoid drift)
        new_f = simplex_projection(new_f)

        new_score = fast_score(new_f, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_f = new_f
            print(f"[LP‑iter {it+1}] score={best_score:.6f}")
        else:
            # no improvement – reduce mixing factor to be more conservative
            t = max(t * 0.5, 0.02)

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size and initial sequence
    # --------------------------------------------------------------
    n = 4000  # length of the candidate (kept modest for the LP phase)

    # start from the previously discovered sequence if available,
    # otherwise use a uniform distribution
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size != n:
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            init_seq = np.interp(x_new, x_old, init_seq)
        a0 = init_seq / init_seq.sum()
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # phase boundaries (empirically chosen)
    phase1_end = start_time + 0.45 * budget_s   # heavy L‑BFGS
    phase2_end = start_time + 0.55 * budget_s   # LP improvement
    phase3_end = start_time + 0.70 * budget_s   # short L‑BFGS refinement
    phase4_end = start_time + 0.85 * budget_s   # sub‑gradient + exp‑gradient
    # remainder → final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – random‑restart continuation L‑BFGS
    # --------------------------------------------------------------
    lam_schedule = [
        0.7,
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # occasional perturbation of the current best
        if rng.random() < 0.15:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – monotone LP improvement (alternating optimisation)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = lp_iterative_improvement(
            best_a,
            max_iter=12,          # a dozen LP steps are usually enough
            t=0.2,
            deadline=phase2_end,
        )
        print(f"[LP‑final] best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – short L‑BFGS refinement (very small smoothing)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        fine_lam_schedule = [0.001, 0.0005, 0.0002, 0.0001]
        fine_maxiter = 400
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS‑fine] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – sub‑gradient + exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase4_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] score={best_score:.6f}")
        if time.time() < phase4_end:
            best_a, best_score = exp_gradient_optimize(
                best_a, phase4_end, fft_len_conv, fft_len_g, n
            )
            print(f"[exp‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (uses remaining time)
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a, best_score, fft_len_conv, n, final_deadline, max_iter=20000
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final evaluation using the provided external function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
