def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2  # margin

    # Determine n
    prev = globals().get('height_sequence_1')
    if prev is not None and len(prev) >= 4000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 6000  # or 8000
        best_a = np.full(n, 1.0/n)

    # Ensure sum=1
    best_a = np.maximum(best_a, 0)
    best_a /= best_a.sum()

    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # Determine FFT lengths
    fft_len_conv = next_pow2(2*n - 1)
    fft_len_g = next_pow2(3*n - 2)

    # Phase 0: parameter sweep
    start_phase = time.time()
    if time.time() < deadline - 5:
        # Exponential candidates
        for rate in np.linspace(1e-4, 0.05, 30):
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_a, best_score = a, sc
                best_y = np.log(best_a + 1e-12)

        # Power candidates
        for p in np.linspace(0.1, 4.0, 30):
            a = (np.arange(1, n+1, dtype=float) ** (-p))
            a /= a.sum()
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_a, best_score = a, sc
                best_y = np.log(best_a + 1e-12)

        # Linear decreasing to zero (simple triangle)
        for slope in np.linspace(0, 1, 20):
            a = np.maximum(0, 1 - slope * np.arange(n))
            if a.sum() == 0:
                continue
            a /= a.sum()
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_a, best_score = a, sc
                best_y = np.log(best_a + 1e-12)

    # Phase 1: continuation L-BFGS multi-restart
    lam_factors = np.geomspace(0.5, 1e-12, num=70)
    maxiter_per_stage = 1500

    iteration = 0
    while time.time() < deadline - 5:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(5e-5, 0.07)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.1, 5.0)
            a0 = (np.arange(1, n+1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            # mass concentrated on left side with exponential decay
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # best
            y0 = best_y + rng.normal(scale=0.1, size=n)

        y_opt = schedule_optimize(y0, n, fft_len_conv, fft_len_g,
                                  lam_factors, maxiter_per_stage)
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} new best fast score = {best_score:.6f}")

        # Quick targeted hill climb
        t_dead = min(deadline, time.time() + 0.6)
        best_a, best_score = targeted_hill_climb(best_a, best_score,

        # Random hill climb

        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n, r_dead,
                                        max_iter=5000, max_move_frac=0.25)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # Phase 2: refined subgradient descent
    sub_dead = deadline - 2.0
    if time.time() < sub_dead:
        best_a, best_score = subgrad_descent(best_a, best_score,

                                            deadline=sub_dead,
                                            max_iter=8000,
                                            step0=0.25)

        best_y = np.log(best_a + 1e-12)

    # Phase 3: final simulated annealing
    temp_start = 0.2
    temp_end = 0.001
    total_sa_time = deadline - time.time()
    if total_sa_time > 0.5:
        t0 = time.time()
        iteration = 0
        while time.time() < deadline:
            # temperature schedule linear in time
            elapsed = time.time() - t0
            temp = temp_start + (temp_end - temp_start) * (elapsed / total_sa_time)
            # propose perturbation in log domain
            y_prop = best_y + rng.normal(scale=0.02, size=n)
            a_prop = softmax(y_prop)
            sc_prop = fast_score(a_prop, fft_len_conv, n)
            delta = sc_prop - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):

            iteration += 1

    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

"""
Low‑score step‑function search.

The algorithm consists of several stages, each respecting the total time
budget (default 1000 s):

1. **Parameter sweep** – generate a variety of simple families
   (exponential, power‑law, linear‑decay, mixtures) and keep the best
   candidate according to the fast surrogate
   `2·n·max(conv)`.

2. **Continuation L‑BFGS** – iteratively smooth the max‑operator with a
   decreasing λ‑schedule and run L‑BFGS from many random starts (including
   the current best) while the time budget permits.

3. **Targeted hill‑climb** – repeatedly lower the currently maximal
   convolution entry by moving mass from the index that contributes most
   to it toward an index whose partner under the same shift is small.

4. **Random hill‑climb** – cheap mass transfers between random positions.

5. **Projected sub‑gradient descent** – exact sub‑gradient steps on the
   true max (identifying the worst shift each iteration).

6. **Simulated annealing** – final temperature‑controlled random
   perturbations in the log‑softmax domain.

All intermediate vectors live on the probability simplex (∑ a = 1) so
the final `evaluate_sequence` call is scale‑invariant.
The best sequence found is stored in the global variable
`height_sequence_1` for reuse by future runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # linear convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max = log‑sum‑exp trick
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # back‑propagation to a (conv_rev = a_rev * w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # convolution (full) via FFT
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to shift k
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # choose destination with smallest partner weight under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        # pick a few random candidates and keep the best among them
        cand_js = rng.integers(0, n, size=8)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max (no smoothing)
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent w.r.t. the exact max of the auto‑convolution.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient of conv_k w.r.t. a: g_i = a_{k-i} + a_i (with 0 outside)
        grad = np.zeros_like(best)
        partner = k - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        grad[mask] = best[partner[mask]]
        grad += best  # add a_i term
        # for 2·a_{k-i} when i != k-i, the expression above already accounts for both
        # (the +a_i term handles the i=j case as 2*a_i)

        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional random acceptance to escape plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size – reuse previous construction if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 6000          # a bit larger than the previous 5000
        best_a = np.full(n, 1.0 / n)

    # ensure a proper probability vector
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # FFT sizes (fixed throughout)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 0 – cheap parameter sweep
    # ------------------------------------------------------------------
    sweep_deadline = min(deadline, time.time() + 30.0)   # spend at most 30 s
    # exponential families
    for rate in np.geomspace(0.001, 0.2, num=35):
        if time.time() > sweep_deadline:
            break
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            best_y = np.log(best_a + 1e-12)
    # power‑law families
    for p in np.linspace(0.05, 6.0, num=35):
        if time.time() > sweep_deadline:
            break
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            best_y = np.log(best_a + 1e-12)
    # linear decay (triangular) families
    for beta in np.linspace(0.0, 2.0 / n, num=30):
        if time.time() > sweep_deadline:
            break
        a = np.maximum(0.0, 1.0 - beta * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-12, num=70)   # finer λ schedule
    maxiter_per_stage = 1500

    iteration = 0
    while time.time() < deadline - 10.0:
        # pick a diverse start point
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.0005, 0.12)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.05, 8.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            # a mixture of an exponential tail and a uniform core
            rate = rng.uniform(0.001, 0.05)
            a_exp = np.exp(-rate * np.arange(n))
            a_mix = 0.7 * a_exp + 0.3 * np.ones_like(a_exp) / n
            a_mix /= a_mix.sum()
            y0 = np.log(a_mix + 1e-12)
        else:  # best
            y0 = best_y + rng.normal(scale=0.08, size=n)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score:
            best_a, best_score = a_opt.copy(), sc_opt
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} → new best fast score = {best_score:.6f}")

        # quick targeted hill‑climb
        t_dead = min(deadline, time.time() + 0.7)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=t_dead,
            max_iter=800,
            move_frac=0.33,
        )
        best_y = np.log(best_a + 1e-12)

        # cheap random hill‑climb
        r_dead = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=r_dead,
            max_iter=6000,
            max_move_frac=0.22,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – refined sub‑gradient descent on the exact max
    # ------------------------------------------------------------------
    sub_deadline = deadline - 2.0
    if time.time() < sub_deadline:
        best_a, best_score = subgrad_descent(
            a=best_a,
            cur_score=best_score,
            n=n,
            fft_len=fft_len_conv,
            deadline=sub_deadline,
            max_iter=8000,
            step0=0.28,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing with the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_start = time.time()
        total_sa = deadline - sa_start
        temp_start = 0.2
        temp_end = 0.001
        while time.time() < deadline:
            elapsed = time.time() - sa_start
            temp = temp_start + (temp_end - temp_start) * (elapsed / total_sa)
            # propose a small perturbation in log‑soft‑max space
            y_prop = best_y + rng.normal(scale=0.015, size=n)
            a_prop = softmax(y_prop)
            sc_prop = fast_score(a_prop, fft_len_conv, n)
            delta = sc_prop - best_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
                best_a, best_score, best_y = a_prop, sc_prop, y_prop

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq

    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
