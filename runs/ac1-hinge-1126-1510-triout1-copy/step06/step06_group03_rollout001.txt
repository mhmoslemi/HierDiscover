"""
Propose a step‑function (list of non‑negative floats) with a low
`evaluate_sequence` score.

Algorithm – overview
--------------------
1. **Problem size** – use the length of the provided `height_sequence_1`
   if it exists; otherwise a default of 8000 points.
2. **Initial candidates** – generate several analytically‑motivated
   distributions (arcsine with several α‑values, semicircle, uniform,
   random Gaussian, binary mask, Gaussian bump).  Keep the best
   (lowest fast score = 2·n·max(conv) when the sum is 1).
3. **Symmetrisation** – the optimal solution is expected to be symmetric,
   so after every update we replace the vector by (v+v[::-1])/2.
4. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing factor λ is
   reduced geometrically (≈ 70 stages).  Each stage runs a bounded
   number of L‑BFGS iterations.
5. **Adam on the smoothed objective** – after L‑BFGS we perform a long
   Adam‑type optimisation (adaptive step‑size, projection via the
   soft‑max parametrisation) with a very small λ, keeping the best
   iterate.
6. **Projected sub‑gradient descent** – minimise the true
   `2·n·max(conv)` using Euclidean projection onto the probability simplex.
7. **Greedy hill‑climbs** – two passes (random and targeted) move a small
   fraction of mass between indices while never increasing the fast
   score.
8. **Final evaluation** – the best candidate is returned; its true
   score is obtained by calling `evaluate_sequence`.

All heavy work is done on the probability simplex (sum = 1).  The
`evaluate_sequence` function scales the score by `sum(a)^‑2`, so any
positive scaling of the vector does not affect the ranking – we simply
return the normalised vector.

The implementation respects the 1000 s wall‑time limit and prints a
few progress messages.

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, length 2*n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: (v + v[::-1]) / 2.  Sum is preserved."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective and its gradient w.r.t. the
    unconstrained vector y (log‑softmax parameters).

    lam = max(conv) * lam_factor  (smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # Derivative of smooth_max wrt conv
    w = exp_shift / Z

    # Gradient of conv w.r.t a (reverse convolution)
    # conv = a * a  => d conv / d a = 2 * (a * delta)
    # Use FFT trick for convolution of w with a reversed
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # Chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: decreasing smoothing (λ) schedule.
    Returns the final y vector.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        # enforce symmetry on the underlying probabilities
        a = softmax(y)
        a = symmetrize(a)
        y = np.log(a + 1e-12)
    return y

def arcsine_init(n: int, alpha: float = 0.5) -> np.ndarray:
    """Beta(α,α) density on (0,1); α≈0.5 gives the classic arcsine shape."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)          # ensure positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return log‑softmax parameters for the chosen analytic shape."""
    if mode == "arcsine_0.5":
        a0 = arcsine_init(n, alpha=0.5)
    elif mode == "arcsine_0.6":
        a0 = arcsine_init(n, alpha=0.6)
    elif mode == "arcsine_0.7":
        a0 = arcsine_init(n, alpha=0.7)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)

    a0 = symmetrize(a0)
    y0 = np.log(a0 + 1e-12)
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random greedy hill‑climb with symmetry enforcement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        # enforce symmetry and renormalise (sum is preserved by symmetrisation)
        best = symmetrize(best)
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
            best = symmetrize(best)
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 1500,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from indices that
    contribute to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # indices i that appear in the max entry (0 ≤ j = m‑i < n)
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i = rng.integers(0, n)
        else:
            i = rng.choice(contributors)

        # pick a destination far from the current max region
        j_dest = rng.integers(0, n)
        if i == j_dest:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j_dest] += delta
        best = symmetrize(best)
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j_dest] -= delta
            best = symmetrize(best)
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv)  (sum(a) = 1).
    Symmetry is enforced after every update.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)
        a_new = symmetrize(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    deadline: float,
    step_size: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed log‑sum‑exp objective.
    The iterate lives in the unconstrained log‑softmax space; after each
    update we enforce symmetry on the underlying probability vector.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_a = softmax(y)
    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        y = y - step_size * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry in probability space
        a = softmax(y)
        a = symmetrize(a)
        y = np.log(a + 1e-12)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):

        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

            a = tmp.copy()
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a = uniform_init(n)
            else:
                a = a / a.sum()
            a = symmetrize(a)
            candidates.append(("provided", a))
    # 2. Analytic shapes
    analytic_modes = [
        ("arcsine_0.5", arcsine_init(n, alpha=0.5)),
        ("arcsine_0.6", arcsine_init(n, alpha=0.6)),

    return final_seq
