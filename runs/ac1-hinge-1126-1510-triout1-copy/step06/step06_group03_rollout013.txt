def adam_subgradient(a, fft_len, n, deadline, alpha=0.05, beta1=0.9, beta2=0.999, eps=1e-8, max_iters=1_000_000):

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(conv.argmax())
        # gradient of max entry:
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

"""
A multi‑stage optimizer for the “max self‑convolution” problem.

The algorithm works on the probability simplex (∑ a_i = 1) because the
objective 2·n·max(conv)/(∑ a_i)² is homogeneous.  The optimisation pipeline
is:

1. **Initialisation** – several analytically motivated shapes (arcsine,
   semicircle, uniform, sparse binary, random Gaussian, single bump)
   are prepared on the chosen problem size *n*.

2. **Continuation L‑BFGS** – a smooth log‑sum‑exp surrogate of the max
   function is minimised.  The smoothing factor λ is driven from 0.5 down
   to 1e‑12 in many geometric steps; each step runs L‑BFGS for a few
   hundred iterations.  The best result among all initialisations is kept.

3. **Projected Adam on the true max** – after the L‑BFGS stage a large
   number of Adam‑style sub‑gradient steps are performed on the genuine
   objective f(a)=2·n·max(a*a).  The step is projected onto the simplex
   after each update.

4. **Aggressive hill‑climbing** – random and targeted moves (mass transfer
   between two coordinates) are applied while the fast FFT‑based score
   improves.  A simple simulated‑annealing acceptance rule allows occasional
   uphill steps.

5. **Final evaluation** – the best sequence found is evaluated with the
   supplied `evaluate_sequence` and returned.

All phases respect the overall time budget (default 1000 s).  The routine
uses only NumPy, SciPy and the standard library.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:                     # should never happen
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    a has length n, fft_len ≥ 2*n‑1.
    Returns the *real* part of the convolution (the exact result).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector: 2·n·max(conv)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic initial shapes
# ----------------------------------------------------------------------
def arcsine_init(n: int, alpha: float = 0.70) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)          # enforce positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return soft‑max parameters (log‑probabilities) for the requested mode."""
    if mode == "arcsine":
        a0 = arcsine_init(n, alpha=0.70)
    elif mode == "semicircle":
        a0 = semicircle_init(n)
    elif mode == "uniform":
        a0 = uniform_init(n)
    elif mode == "random":
        a0 = random_gaussian_init(n, rng)
    elif mode == "binary":
        a0 = binary_sparse_init(n, rng, density=0.15)
    elif mode == "bump":
        a0 = gaussian_bump_init(n, rng)
    else:
        a0 = random_gaussian_init(n, rng)

    # log‑softmax parameters (avoid log(0))
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max(conv) and its gradient w.r.t. y.
    The smoothing constant λ = max(conv)·lam_factor  (very small at the end).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # maximum becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights w = ∂smooth_max/∂conv
    w = exp_shift / Z                     # length 2*n-1

    # gradient of conv wrt a (reverse convolution)
    # we need d(conv)/da = 2 * a convolved with a delta‑function,
    # which in the frequency domain is a_rev * w.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to derivative for each a_i
    g_vec = grad_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing factor.
    Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage,
                     "gtol": 1e-12,
                     "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Adam on the real (non‑smoothed) objective
# ----------------------------------------------------------------------
def adam_subgradient(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    alpha: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam to minimise   f(a)=2·n·max(conv)  (equivalently max(conv)).
    Returns the best vector seen during the run.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(conv.argmax())

        # sub‑gradient of the active entry:
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam updates:
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # bias‑correction:
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climbing (random + targeted) with optional simulated annealing
# ----------------------------------------------------------------------
def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
    max_move: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Random greedy hill‑climb: move a random fraction of mass from i to j.
    An uphill move is accepted with a tiny probability that decays with time.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        # trial move
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # simulated‑annealing acceptance (tiny probability)
            # temperature decays linearly with iteration count
            temp = max(0.001, 0.05 * (1.0 - it / max_iter))
            if rng.random() < np.exp(-(new_score - best_score) / temp):
                best_score = new_score
            else:
                # revert
                best[i] += delta
                best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 10000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Move mass away from indices that actively participate in the current
    maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # indices that appear in the max entry:
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i = rng.integers(0, n)
        else:
            i = rng.choice(contributors)

        j = rng.integers(0, n)               # destination can be anywhere
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine respects the 1000 s budget and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # 1 s safety margin

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    # ------------------------------------------------------------------
    # Prefer a reasonably large n but keep the FFT cheap.
    # If a user supplied sequence exists we honour its length (up to 12000).
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initialise the current best candidate (probability simplex)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = uniform_init(n)
    else:
        best_a = uniform_init(n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS (≈ 50 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=70))
    maxiter_per_stage = 800                     # more iterations per λ‑stage
    modes = ["arcsine", "semicircle", "uniform",
             "random", "binary", "bump"]
    mode_cycle = 0
    iteration = 0
    main_phase_deadline = start_time + 0.50 * budget_s

    while time.time() < main_phase_deadline:
        # pick an initialisation
        if iteration % 9 == 0 and iteration > 0:
            mode = "best"
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            mode = modes[mode_cycle % len(modes)]
            mode_cycle += 1
            y0 = generate_initial_y(mode, n, rng)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # short random hill‑climb to polish each L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.75)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Stage 2 – Projected Adam on the true max (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    adam_deadline = min(deadline - 30.0, start_time + 0.85 * budget_s)

    best_a, best_score = adam_subgradient(
