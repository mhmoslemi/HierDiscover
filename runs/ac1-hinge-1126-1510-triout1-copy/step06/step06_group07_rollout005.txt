def pairwise_targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000, move_frac=0.2):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # compute convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        k = int(np.argmax(conv))

        # compute contributions per i for shift k
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i_max = int(np.argmax(contrib))
        j_max = int(k - i_max)

        # choose new indices with low partner values
        # sample candidate positions
        cand_i = rng.integers(0, n, size=5)
        # exclude i_max and j_max
        cand_i = cand_i[(cand_i != i_max) & (cand_i != j_max)]
        if cand_i.size == 0:
            continue
        # pick i_target with minimal partner weight
        partner_weight_i = np.full(n, np.inf)
        partner_idx_i = k - cand_i
        valid_i = (partner_idx_i >= 0) & (partner_idx_i < n)
        partner_weight_i[valid_i] = best[partner_idx_i[valid_i]]
        i_target = int(cand_i[np.argmin(partner_weight_i[cand_i])])

        # for j_target, sample similarly
        cand_j = rng.integers(0, n, size=5)
        cand_j = cand_j[(cand_j != i_max) & (cand_j != j_max) & (cand_j != i_target)]
        if cand_j.size == 0:
            continue
        partner_weight_j = np.full(n, np.inf)
        partner_idx_j = k - cand_j
        valid_j = (partner_idx_j >= 0) & (partner_idx_j < n)
        partner_weight_j[valid_j] = best[partner_idx_j[valid_j]]
        j_target = int(cand_j[np.argmin(partner_weight_j[cand_j])])

        # ensure i_target != j_target; if equal adjust
        if i_target == j_target:
            continue

        # compute delta as fraction of the smaller of a[i_max], a[j_max]
        delta = min(best[i_max], best[j_max]) * move_frac * rng.random()
        if delta <= 0:
            continue
        # move delta from i_max and j_max to i_target and j_target
        best[i_max] -= delta
        best[j_max] -= delta
        best[i_target] += delta
        best[j_target] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_max] += delta
            best[j_max] += delta
            best[i_target] -= delta
            best[j_target] -= delta
    return best, best_score

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm Overview
-------------------
1. Initialise a probability vector `a` (∑a = 1).  If a previous
   construction (`height_sequence_1`) exists we reuse it, otherwise we
   start from the uniform distribution.
2. Global optimisation:
   * Optimise a smooth approximation of the max‑convolution using
     continuation L‑BFGS on unconstrained log‑soft‑max parameters.
   * Several diverse random restarts (uniform, exponential, power‑law,
     edge‑biased, perturbations of the current best) are performed.
3. Local refinements:
   * **Pairwise targeted hill‑climb** – reduces the dominant convolution
     entry by moving mass from the two indices that contribute most to
     that entry to low‑contributing indices.
   * **Random hill‑climb** – simple mass‑transfer moves for extra
     exploration.
   * **Projected sub‑gradient descent** – exact max‑convolution gradient
     with simplex projection.
4. A final aggressive random hill‑climb uses any remaining time.
5. The best sequence seen is stored in the global name
   `height_sequence_1` and returned after the official evaluation.

All operations respect the non‑negativity and sum‑to‑one constraints,
and the whole search is bounded by the supplied time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Utility functions (no closures)
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (∑ = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:                     # fallback – uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# --------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution (quadratic) and its gradient
    w.r.t. the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c               # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a -----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max (diag(a) – a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# --------------------------------------------------------------
# Hill‑climb utilities
# --------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def pairwise_targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb that attacks the dominant convolution entry.
    Moves *both* members of the most damaging pair to low‑impact positions.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (via FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                     # worst shift

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i_max = int(np.argmax(contrib))
        j_max = int(k - i_max)

        # choose low‑impact destinations
        cand_i = rng.integers(0, n, size=5)
        cand_i = cand_i[(cand_i != i_max) & (cand_i != j_max)]
        if cand_i.size == 0:
            continue
        # partner weight for candidates (how badly they would pair)
        weight_i = np.full(n, np.inf)
        p_i = k - cand_i
        valid_i = (p_i >= 0) & (p_i < n)
        weight_i[valid_i] = best[p_i[valid_i]]
        i_target = int(cand_i[np.argmin(weight_i[cand_i])])

        cand_j = rng.integers(0, n, size=5)
        cand_j = cand_j[(cand_j != i_max) & (cand_j != j_max) &
                        (cand_j != i_target)]
        if cand_j.size == 0:
            continue
        weight_j = np.full(n, np.inf)
        p_j = k - cand_j
        valid_j = (p_j >= 0) & (p_j < n)
        weight_j[valid_j] = best[p_j[valid_j]]
        j_target = int(cand_j[np.argmin(weight_j[cand_j])])

        if i_target == j_target:
            continue

        # move a tiny fraction of the *smaller* of the two source masses
        delta = min(best[i_max], best[j_max]) * move_frac * rng.random()
        if delta <= 0.0:
            continue

        best[i_max] -= delta
        best[j_max] -= delta
        best[i_target] += delta
        best[j_target] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_max] += delta
            best[j_max] += delta
            best[i_target] -= delta
            best[j_target] -= delta
    return best, best_score


# --------------------------------------------------------------
# Projected sub‑gradient descent on the true max (no smoothing)
# --------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent w.r.t. the exact max of the auto‑convolution.
    Each iteration identifies the shift achieving the maximum, computes the
    corresponding gradient (2·a shifted), takes a step opposite to it and
    projects back onto the probability simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        it += 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient for the active shift
        grad = np.zeros_like(best)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * best[j]

        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional stochastic acceptance to escape tiny plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score
    return best, best_score
