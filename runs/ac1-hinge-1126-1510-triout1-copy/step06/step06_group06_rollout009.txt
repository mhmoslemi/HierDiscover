def lp_step(f, M):
    n = len(f)
    row_ind = []
    col_ind = []
    data = []
    for k in range(2*n - 1):
        start_i = max(0, k-(n-1))
        end_i = min(n-1, k)
        for i in range(start_i, end_i+1):
            j = k - i
            # f_i * g_j -> coefficient of g_j is f_i
            # Av at row k, column j
            row_ind.append(k)
            col_ind.append(j)
            data.append(f[i])
    A_ub = sp.csr_matrix((data, (row_ind, col_ind)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)

    c = -np.ones(n)  # maximize sum g => minimize -sum g
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        g = res.x
    else:
        # fallback: zero vector
        g = np.zeros(n)
    return g

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------
1. **Problem size** – Use the length of a previously found sequence if it
   exists, otherwise start with `n = 6000`.  The sequence is always kept on
   the probability simplex (sum = 1); scaling does not affect the objective.

2. **Parametric warm‑starts** – Several families of analytically‑motivated
   shapes are generated (power‑law, exponential, triangular, Hann, sine,
   uniform).  A coarse grid over the shape parameters is evaluated with a
   fast FFT‑based surrogate `fast_score`.  The best shape becomes the
   starting point for the subsequent optimisation phases.

3. **Smooth‑max L‑BFGS** – Optimise the smoothed max of the auto‑convolution
   in the unconstrained log‑soft‑max space using a geometric λ‑schedule.
   The gradient is obtained analytically (FFT‑based) and the simplex
   constraint is satisfied automatically by the soft‑max.

4. **Targeted hill‑climb** – Repeatedly identify the shift that attains the
   current maximal convolution entry, locate the pair of indices that
   contributes most to that entry and move a fraction of mass away from it
   towards a “quiet’’ partner index.

5. **Random mass‑transfer hill‑climb** – Classic simplex‑preserving
   hill‑climbing that explores the neighbourhood by moving a random amount
   of probability mass between two coordinates.

6. **Projected sub‑gradient descent** – Directly minimise the true
   objective `max(conv(a,a))` on the simplex by using the sub‑gradient of
   the max‑operator and Euclidean projection onto the simplex.  A simple
   diminishing step‑size schedule is used.

7. **Final polishing** – One last random hill‑climb makes use of any
   remaining time.

The whole search stops before the provided time budget (default 1000 s)
and returns the best sequence found.  The sequence is stored in the global
variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(a,a)) where the convolution is computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # --- convolution a*a (full) ---
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max (log‑sum‑exp) ---
    shifted = conv - max_c            # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # --- gradient of smooth max w.r.t. conv ---
    w = exp_shift / Z

    # propagate gradient back to a
    # compute (a * w) convolved with reversed a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: List[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]
        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]

        # sample a few candidate targets
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient of the
    (non‑smoothed) max‑operator.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to the half‑range 0 ≤ k ≤ n‑1 (symmetry of the full convolution)
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        grad = 2.0 * a
    else:
        grad[: n - k] += a[k:]
        grad[k:] += a[: n - k]
    return max_c, grad

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided evaluation function.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and an initial feasible point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.maximum(np.asarray(prev, dtype=float), 0.0)
        if best_a.sum() == 0.0:
            best_a = np.full(n, 1.0 / n)
        else:
            best_a = best_a / best_a.sum()
    else:
        # a slightly larger size than the previous run (more degrees of freedom)
        n = 6000
        best_a = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Parametric warm‑start search
    # ------------------------------------------------------------------
    def evaluate_candidate(arr: np.ndarray) -> float:
        """Return the fast surrogate score for a candidate array."""
        arr = np.maximum(arr, 0.0)
        if arr.sum() == 0.0:
            return np.inf
        arr = arr / arr.sum()
        return fast_score(arr, fft_len_conv, n)

    best_score = evaluate_candidate(best_a)
    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    # ----- power‑law family : a_i ∝ ((i+1)*(n-i))^p , p<0 ----------------
    p_vals = np.linspace(-2.0, -0.05, 24)
    for p in p_vals:
        idx = np.arange(n, dtype=float)
        shape = ((idx + 1) * (n - idx)) ** p
        score = evaluate_candidate(shape)
        if score < best_score:
            best_score = score
            best_a = shape / shape.sum()
            print(f"[grid] power p={p:.3f} → score {score:.6f}")

    # ----- exponential family : a_i ∝ exp(-c*i)+exp(-c*(n-1-i)) ----------
    c_vals = np.logspace(-4, -0.6, 20)  # roughly 0.0001 … 0.25
    for c in c_vals:
        shape = np.exp(-c * np.arange(n)) + np.exp(-c * np.arange(n - 1, -1, -1))
        score = evaluate_candidate(shape)
        if score < best_score:
            best_score = score
            best_a = shape / shape.sum()
            print(f"[grid] exp c={c:.5f} → score {score:.6f}")

    # ----- triangular (tent) shape --------------------------------------
    tent = np.minimum(np.arange(n) + 1, n - np.arange(n))
    score = evaluate_candidate(tent)
    if score < best_score:
        best_score = score
        best_a = tent / tent.sum()
        print(f"[grid] triangular → score {score:.6f}")

    # ----- Hann window ---------------------------------------------------
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    score = evaluate_candidate(hann)
    if score < best_score:
        best_score = score
        best_a = hann / hann.sum()
        print(f"[grid] Hann window → score {score:.6f}")

    # ----- Sine window ---------------------------------------------------
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    score = evaluate_candidate(sine)
    if score < best_score:
        best_score = score
        best_a = sine / sine.sum()
        print(f"[grid] Sine window → score {score:.6f}")

    # ----- Uniform (baseline) -------------------------------------------
    uniform = np.full(n, 1.0 / n)
    score = evaluate_candidate(uniform)
    if score < best_score:
        best_score = score
        best_a = uniform
        print(f"[grid] uniform → score {score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max L‑BFGS optimisation from the best parametric start
    # ------------------------------------------------------------------
    y_start = np.log(best_a + 1e-12)
    lam_factors = np.geomspace(0.3, 1e-7, num=30)   # decreasing temperature
    y_opt = schedule_optimize(
        y_start=y_start,
        n=n,
        fft_len_conv=fft_len_conv,
        fft_len_g=fft_len_g,
        lam_factors=lam_factors,
        maxiter_per_stage=250,
    )
    best_a = softmax(y_opt)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[lbfgs] after L‑BFGS fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Targeted hill‑climb (tight time budget)
    # ------------------------------------------------------------------
    tc_deadline = min(deadline, time.time() + 0.9)
    best_a, best_score = targeted_hill_climb(
        a=best_a,
        cur_score=best_score,
        fft_len=fft_len_conv,
        n=n,
        deadline=tc_deadline,
        max_iter=1200,
        move_frac=0.30,
    )
    print(f"[t-hc] after targeted hill‑climb score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Random hill‑climb (exploration)
    # ------------------------------------------------------------------
    rc_deadline = min(deadline, time.time() + 0.9)
    best_a, best_score = hill_climb(
        a=best_a,
        cur_score=best_score,
        fft_len=fft_len_conv,
        n=n,
        deadline=rc_deadline,
        max_iter=5000,
        max_move_frac=0.25,
    )
    print(f"[r-hc] after random hill‑climb score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Projected sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    step0 = 0.45
    subgrad_iter = 0
    eps = 1e-12
    while time.time() < deadline - 2.0:
        subgrad_iter += 1
        max_c, grad = max_convolution_subgrad(best_a, n, fft_len_conv)

        # Diminishing step size
        step = step0 / np.sqrt(subgrad_iter)

        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:
            best_a = cand
            best_score = cand_score
            step0 = min(step0 * 1.02, 1.0)      # slowly increase if improving
        else:
            step0 *= 0.93                        # otherwise shrink a bit

        if subgrad_iter % 200 == 0:
            print(f"[subgrad] iter={subgrad_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final polishing with a longer random hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=final_deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        print(f"[final] after polishing score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the result (store in global for reuse)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    # The official evaluation (the driver will also call it)
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)
    return final_seq
