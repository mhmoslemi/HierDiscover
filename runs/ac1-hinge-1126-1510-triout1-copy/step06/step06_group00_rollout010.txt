def additive_subgrad_optimize(a0, deadline, n, fft_len_conv, fft_len_g):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        # compute conv and mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n-1):(n-1+n)]

        # step size schedule
        step = 0.5 / np.sqrt(it + 1.0)  # can also adapt
        # move opposite to gradient
        a_candidate = simplex_projection(a - step * g)

        # compute fast score
        cand_score = fast_score(a_candidate, fft_len_conv, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score

Algorithm outline
-----------------
* Work on a fixed length `n = 4000` (the prior best length).  The search
  proceeds in several *independent cycles* that each consist of
  - a random start (or a perturbed current best),
  - continuation L‑BFGS on a smoothed log‑sum‑exp approximation of the
    max‑convolution,
  - additive projected sub‑gradient descent on the exact max‑convolution,
  - exponentiated‑gradient descent (multiplicative update, stays on the simplex),
  - a short simulated‑annealing phase,
  - a greedy hill‑climb.

* The best sequence over *all* cycles is kept.  The remaining time is spent
  on a final fine‑tune with Adam.

* All vectors are kept on the probability simplex (non‑negative, sum = 1)
  – the evaluation function is invariant under scaling, so the simplex
  normalisation does not affect the final score.

The code follows the same fast FFT utilities as the reference implementation
but explores many more random restarts and adds an additive sub‑gradient
optimiser that often yields a small extra reduction in the max‑convolution.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when `a` lives on the simplex
    (i.e. sum(a) = 1).  This is exactly the objective that `evaluate_sequence`
    would return after the scaling invariance is taken into account.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity (including a Gaussian
    and a sparse Bernoulli‑type family).
    """
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'gaussian', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'gaussian':
        mu = n / 2.0
        sigma = rng.uniform(0.07, 0.20) * n
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    else:  # sparse
        k = rng.integers(2, max(2, int(np.sqrt(n) * 2)))
        a = np.zeros(n)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = rng.random(k) + 0.1
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed approximation of the max‑convolution (log‑sum‑exp).
    Returns (objective, gradient) with respect to the logits `y`
    where `a = softmax(y)`.
    """
    a = softmax(y)

    # a * a (full convolution)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. the convolution entries
    w = exp_shift / Z

    # back‑propagation through the convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """
    Run L‑BFGS for each λ in `lam_schedule`, seeding each stage with the
    result of the previous stage.  Returns the best `a` and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def additive_subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
):
    """
    Projected additive sub‑gradient descent on the exact max‑convolution.
    The update is `a ← Π_S (a - α·g)` where `Π_S` denotes projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # compute full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        # step size schedule (decreasing like 1/√t)
        step = 0.9 / np.sqrt(it + 1.0)

        a_candidate = simplex_projection(a - step * g)

        # occasional jitter to escape plateaus
        if it % 250 == 0 and rng.random() < 0.12:
            a_candidate = add_noise(a_candidate, sigma=rng.uniform(0.001, 0.015), rng=rng)

        cand_score = fast_score(a_candidate, fft_len_conv, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
):
    """
    Exponentiated‑gradient descent (multiplicative update) on the exact
    max‑convolution.  The update keeps `a` on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        eta = 0.6 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def random_anneal(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
):
    """
    Very short simulated annealing that moves a random fraction of mass
    between two coordinates, accepting worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 8000,
):
    """
    Greedy hill‑climb that only accepts strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    lam_factor: float = 0.0015,
    lr: float = 0.12,
):
    """
    Adam optimisation of the smoothed log‑sum‑exp objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation score.
    The function respects the 1000 s time budget and returns the best list of
    floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Fixed problem size (you may change this if you like)
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initial sequence (use the historic best if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            x_old = np.linspace(0, 1, init_seq.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # global best holder
    global_best_a = a0.copy()
    global_best_score = fast_score(global_best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={global_best_score:.6f}")

    # --------------------------------------------------------------
    # Time allocation
    # --------------------------------------------------------------
    start_time = time.time()
    deadline_total = start_time + budget_s - 0.5  # safety margin

    # We run several *cycles*; each cycle gets about 150 s.
    min_cycle_time = 150.0
    cycle = 0

    while time.time() < deadline_total - min_cycle_time:
        cycle += 1
        cycle_deadline = time.time() + min_cycle_time
        print(f"[cycle {cycle}] start (deadline {cycle_deadline - start_time:.1f}s)")

        # ----------- random start (or perturb the global best) ----------
        if global_best_a is not None and rng.random() < 0.35:
            sigma = rng.uniform(0.001, 0.018)
            a_start = add_noise(global_best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        # -------------------- Phase 1: L‑BFGS -------------------------
        lam_schedule = [
            0.5,
            0.35,
            0.25,
            0.18,
            0.12,
            0.08,
            0.05,
            0.03,
            0.018,
            0.01,
            0.006,
            0.003,
            0.001,
            0.0005,
            0.0002,
        ]
        maxiter_per_lam = 1800

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            cycle_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"  [L‑BFGS] score={score_opt:.6f}")

        # -------------------- Phase 2: additive sub‑grad ----------------
        if time.time() < cycle_deadline:
            a_opt, score_opt = additive_subgrad_optimize(
                a_opt,
                cycle_deadline,
                n,
                fft_len_conv,
                fft_len_g,
                rng,
            )
            print(f"  [additive sub‑grad] score={score_opt:.6f}")

        # -------------------- Phase 3: exponentiated gradient ----------
        if time.time() < cycle_deadline:
            a_opt, score_opt = exp_gradient_optimize(
                a_opt,
                cycle_deadline,
                n,
                fft_len_conv,
                fft_len_g,
                rng,
            )
            print(f"  [exp‑grad] score={score_opt:.6f}")

        # -------------------- Phase 4: simulated annealing -------------
        if time.time() < cycle_deadline:
            a_opt, score_opt = random_anneal(
                a_opt,
                score_opt,
                fft_len_conv,
                n,
                cycle_deadline,
            )
            print(f"  [anneal] score={score_opt:.6f}")

        # -------------------- Phase 5: greedy hill‑climb ---------------
        if time.time() < cycle_deadline:
            a_opt, score_opt = hill_climb(
                a_opt,
                score_opt,
                fft_len_conv,
                n,
                cycle_deadline,
                max_iter=10000,
            )
            print(f"  [hill‑climb] score={score_opt:.6f}")

        # keep global best
        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            print(f"[cycle {cycle}] ** new global best fast_score={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tune with Adam on the remaining time budget
    # ------------------------------------------------------------------
    if time.time() < deadline_total and global_best_a is not None:
        logits = np.log(global_best_a + 1e-12)
        global_best_a, global_best_score = adam_optimize(
            logits,
            n,
            fft_len_conv,
            fft_len_g,
            deadline_total,
            lam_factor=0.001,
            lr=0.09,
        )
        print(f"[final Adam] fast_score={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = global_best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
