        if tmp.ndim == 1 and tmp.size >= 4000:

            n = 8000

def reorder_extreme(a):
    # place sorted descending values at alternating ends
    n = a.shape[0]
    indices = np.argsort(-a)  # descending
    b = np.empty_like(a)
    left = 0
    right = n - 1
    toggle = True
    for idx in indices:
        if toggle:
            b[left] = a[idx]
            left += 1
        else:
            b[right] = a[idx]
            right -= 1
        toggle = not toggle
    return b

   only accepts strictly improving moves.
6. Return the final list.  The external harness will call
   ``evaluate_sequence`` on the returned list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                 # max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt conv
    w = exp_shift / Z

    # propagate through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] <= 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the average of the sub‑gradients of all (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active indices: values close to the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of averaged max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        # step size (decreasing) with simple back‑tracking
        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):   # up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            # even reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def reorder_extreme(a: np.ndarray) -> np.ndarray:
    """
    Rearrange the entries of ``a`` by placing the largest values at the
    two ends alternatingly.  This tends to decorrelate a vector from its
    reversal and can lower the max‑convolution.
    """
    n = a.shape[0]
    order = np.argsort(-a)  # descending
    b = np.empty_like(a)
    left = 0
    right = n - 1
    toggle = True
    for idx in order:
        if toggle:
            b[left] = a[idx]
            left += 1
        else:
            b[right] = a[idx]
            right -= 1
        toggle = not toggle
    return b


def simulated_annealing(
    a_start: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that moves a random fraction of mass from one
    coordinate to another.  It always accepts improving moves and
    accepts worsening moves with a temperature‑dependent probability.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    # temperature schedule: exponential cooling from 10 % of the current
    # score down to 1e‑5 of it.
    T0 = cur_score * 0.1
    T_end = cur_score * 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / total_time
        T = T0 * (T_end / T0) ** progress

        # propose a move
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer up to 20 % of source mass
        delta = cur[i] * rng.random() * 0.20
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur.copy()
        else:
            # accept w.p. exp(-(Δ)/T)
            accept_prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < accept_prob:
                cur_score = new_score
                # keep the change
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list) with a low
    ``evaluate_sequence`` score.  The routine respects the time budget,
    uses L‑BFGS on a smoothed surrogate, a long projected sub‑gradient
    phase, simulated annealing and a final hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise the starting point on the simplex.
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a0 = np.maximum(tmp, 0.0)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    if a0.sum() <= 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Continuation L‑BFGS (optional – a quick refinement)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=70))
    maxiter_per_stage = 1000

    # try a few random perturbations of the current best
    while time.time() < deadline - 300.0:  # keep ~5 min for later phases
        y0 = best_y + rng.normal(scale=0.03, size=n)
        y_opt = schedule_optimize(
            y0, n, fft_len, fft_len, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[L‑BFGS] improved fast score = {best_score:.6f}")

        # try a simple permutation that decorrelates the reverse
        perm_a = reorder_extreme(best_a)
        perm_score = fast_score(perm_a, fft_len, n)
        if perm_score < best_score:
            best_score = perm_score
            best_a = perm_a
            best_y = np.log(best_a + 1e-12)
            print(f"[perm] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Long projected sub‑gradient phase
    # ------------------------------------------------------------------
    sub_deadline = deadline - 120.0
    if time.time() < sub_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a, fft_len, fft_len, n, sub_deadline,
            eta0=0.35, max_iters=800_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing (global exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sa_deadline = deadline - 30.0
        print("[SA] start")
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len, n, sa_deadline
        )
        print(f"[SA] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=30000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
