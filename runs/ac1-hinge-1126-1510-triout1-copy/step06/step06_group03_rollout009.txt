"""
Advanced search for a low‑evaluation step‑function.

Main ideas
-----------
* Work on the probability simplex (∑ a_i = 1).  The objective
  `evaluate_sequence` is scale‑invariant, so we can keep the sum = 1.
* Initialise the search with a rich family of analytic shapes:
      – β(p,p) (a generalised arcsine) for several p ∈ (0.5,1)
      – Wigner semicircle
      – Uniform
      – Random Gaussian, sparse binary mask, a single Gaussian bump,
        and a few mixtures of the above.
* For each promising shape run a continuation L‑BFGS optimiser on a
  smoothed “log‑sum‑exp” surrogate of the max‑convolution.
  The smoothing parameter λ is reduced geometrically from a very
  large value (λ ≈ 5·max) down to 1e‑9·max.
* After the L‑BFGS phase a long projected sub‑gradient descent is
  performed on the true objective (max‑convolution).  Step‑size decays
  as 1/√t and each iterate is projected onto the simplex.
* Finally aggressive random / targeted hill‑climbs polish the solution.
* All phases respect a global 1000 s budget; the best sequence found
  is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    a is length n, fft_len >= 2*n-1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the evaluation function for a probability vector
    (∑ a_i = 1): 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c            # makes the max become 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # correlation of a and w (gradient of conv w.r.t. a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # note: lam is passed as lam_factor; inside the objective we compute λ = max_c * lam_factor
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Analytic initialisation families
# ----------------------------------------------------------------------


def beta_init(n: int, p: float) -> np.ndarray:
    """β(p,p) density on (0,1); p≈0.5 gives the classic arcsine."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """
    Discrete Wigner semicircle distribution on {0,…,n‑1}.
    a_i ∝ sqrt(1 - ((i - (n‑1)/2) / ((n‑1)/2))**2 )
    """
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)  # ensure positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised to sum 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s


# ----------------------------------------------------------------------
# Local search helpers (hill‑climbing)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: transfer a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that preferentially moves mass away from the current max‑conv contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Implements the pipeline described in the module
    docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – use a modestly large n for expressive power.
    # ------------------------------------------------------------------
    # If a previous sequence exists and is long enough, reuse its length.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Generate a diverse pool of initial shapes.
    # ------------------------------------------------------------------
    candidates = []

    # 1) Beta(p,p) for several p values (including classical arcsine)
    for p in [0.55, 0.60, 0.65, 0.70, 0.75, 0.80]:
        a = beta_init(n, p)
        candidates.append((f"beta_{p:.2f}", a))

    # 2) Semicircle
    candidates.append(("semicircle", semicircle_init(n)))

    # 3) Uniform
    candidates.append(("uniform", uniform_init(n)))

    # 4) Random Gaussian
    candidates.append(("random", random_gaussian_init(n, rng)))

    # 5) Sparse binary mask
    candidates.append(("binary", binary_sparse_init(n, rng, density=0.15)))

    # 6) Single Gaussian bump
    candidates.append(("bump", gaussian_bump_init(n, rng)))

    # 7) Mixtures of the best beta (p≈0.7) with uniform
    beta_ref = beta_init(n, 0.70)
    for w in [0.2, 0.4, 0.6, 0.8]:
        mixed = mixture_init(beta_ref, uniform_init(n), w)
        candidates.append((f"beta_uniform_mix_{w:.1f}", mixed))

    # 8) If a historic sequence exists, use it (after normalisation)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            candidates.append(("previous", a))

    # ------------------------------------------------------------------
    # Quick scan to keep only the most promising few initialisations.
    # ------------------------------------------------------------------
    scored = []
    base_fft_len = fft_len_conv
    for name, a in candidates:
        sc = fast_score(a, base_fft_len, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:22s} fast score = {sc:.6f}")

    # keep top K candidates (K=4) for full optimisation
    scored.sort()
    top_k = 4
    best_a = None
    best_score = np.inf
    best_y = None

    # ------------------------------------------------------------------
    # Continuation L‑BFGS phase (≈ 50 % of the total budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-9, num=28))
    maxiter_per_stage = 600

    phase_deadline = start_time + 0.50 * budget_s
    print("[phase] L‑BFGS continuation started")

    for rank in range(min(top_k, len(scored))):
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)

        # Run the full continuation schedule
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[l‑bfgs] new best from {name:22s} score = {best_score:.6f}")

        # Occasionally polish the current best with a short random hill‑climb
        if time.time() < phase_deadline:
            bc, bs = hill_climb(
                best_a, best_score, fft_len_conv, n, deadline=phase_deadline, max_iter=2000
            )
            if bs < best_score:
                best_a, best_score = bc, bs
                best_y = np.log(best_a + 1e-12)
                print(f"[l‑bfgs] post‑hill‑climb improved score = {best_score:.6f}")

        if time.time() >= phase_deadline:
            print("[phase] L‑BFGS time exhausted")
            break

    # If nothing succeeded (unlikely), fall back to uniform
    if best_a is None:
        best_a = uniform_init(n)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Random‑restart continuation (more exploratory runs while time permits)
    # ------------------------------------------------------------------
    random_restart_deadline = start_time + 0.70 * budget_s
    restart_count = 0
    while time.time() < random_restart_deadline:
        restart_count += 1
        # Random Gaussian start
        a0 = random_gaussian_init(n, rng)
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule[:12],      # a shorter schedule for quick restarts
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] #{restart_count:03d} improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=500_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive final polishing (random + targeted hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] aggressive random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=15000,
            max_move=0.30,
        )
        if time.time() < deadline:
            print("[final] targeted hill‑climb")
            best_a, best_score = targeted_hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=8000,
                max_move=0.20,
            )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)
        final_val = None

    return final_seq
