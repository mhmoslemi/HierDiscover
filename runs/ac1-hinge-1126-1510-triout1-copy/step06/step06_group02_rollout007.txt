"""
Propose a non‑negative step function with a low `evaluate_sequence` score.

Algorithm
---------
1. Choose a problem size `n`.  Re‑use a previous construction if available,
   otherwise take `n≈8000` (capped at 10000).  All vectors are kept on the
   probability simplex (sum = 1) – scaling does not affect the score.

2. **Fast shape sweep** – evaluate a few parametrised families
   (symmetric exponential, power‑law, edge‑heavy, arcsine) on a coarse
   grid.  Keep the best shape as a warm start.

3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of
   the true max‑convolution.  The smoothing factor λ = max_c·α is reduced
   geometrically over 40 stages; each stage runs a limited‑iteration L‑BFGS.
   Many random restarts (including perturbations of the current best)
   are performed while time permits.

4. **Projected Adam** – a few Adam steps on the same surrogate improve
   the solution further without the overhead of line‑searches.

5. **Multi‑active sub‑gradient descent** – optimise the true objective
   using the average sub‑gradient of all (almost) active convolution
   entries, with a back‑tracking line‑search.

6. **Aggressive hill‑climb** – repeatedly move a random fraction of mass
   from one coordinate to another (up to 20 % of the source mass) and keep
   only improving moves.  The remaining time is spent here.

7. Return the best probability vector found (as a plain list of floats).
   The external harness will perform the exact `evaluate_sequence` call.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained logits y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                     # max = 0
    # Clip exponent to avoid overflow/underflow
    exp_shift = np.exp(np.clip(shifted / lam, -50, 50))
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)              # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of L‑BFGS optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple Adam optimiser for the smoothed surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    steps: int,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lam_factor: float = 0.30,
) -> np.ndarray:
    """
    Adam on the log‑sum‑exp surrogate.  The iterate stays in logit space
    (soft‑max guarantees feasibility).  Returns the final logits.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    for t in range(1, steps + 1):
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


# ----------------------------------------------------------------------
# Initial distribution generators (return logits)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point `y` (pre‑soft‑max) for a given heuristic mode.
    """
    if mode == "random":
        a0 = rng.random(n)
        a0 /= a0.sum()

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        i = np.arange(1, n + 1, dtype=float)
        a0 = i ** (-p)
        a0 /= a0.sum()

    elif mode == "edge":
        m_frac = rng.uniform(0.01, 0.04)
        m = max(1, int(round(n * m_frac)))
        ratio = rng.uniform(2.0, 12.0)       # edge mass / interior mass
        beta = 1.0
        alpha = ratio * beta
        total = 2 * m * alpha + (n - 2 * m) * beta
        a0 = np.full(n, beta / total)
        a0[:m] = alpha / total
        a0[-m:] = alpha / total

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()

    elif mode == "double_exp":
        # mixture of two symmetric exponentials
        r1 = rng.uniform(0.001, 0.02)
        r2 = rng.uniform(0.001, 0.02)
        w = rng.random()
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = w * np.exp(-r1 * d) + (1 - w) * np.exp(-r2 * d)
        a0 /= a0.sum()

    elif mode == "sidon":
        # greedy Sidon‑like set, equal mass on chosen indices
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()

    elif mode == "best":
        # perturb the current best – handled outside via global variable
        raise ValueError("mode 'best' must be treated specially")
    else:
        raise ValueError(f"unknown mode {mode}")

    # Convert to logits for the optimiser
    return np.log(a0 + 1e-12)


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i
    to j (up to 20 % of the source).  Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.20  # up to 20 % per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    The sub‑gradient is the average of the (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec  # d (a_i a_j) / d a_i = 2 a_j

        # step size schedule
        eta = eta0 / np.sqrt(it)

        improved = False
        for _ in range(8):  # back‑tracking
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size selection
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Coarse parametric sweep (cheap)
    # ------------------------------------------------------------------
    def consider_candidate(a_candidate: np.ndarray):
        nonlocal best_a, best_score, best_y
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()
        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[sweep] improved fast score = {best_score:.6f}")

    # symmetric exponential sweep
    for _ in range(40):
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-rate * d)
        consider_candidate(a)

    # power‑law sweep
    for _ in range(40):
        p = rng.uniform(0.3, 2.0)
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        consider_candidate(a)

    # edge‑heavy sweep (grid)
    for m_frac in np.linspace(0.01, 0.06, 9):
        m = max(1, int(round(n * m_frac)))
        for ratio in np.linspace(2.0, 15.0, 7):
            beta = 1.0
            alpha = ratio * beta
            total = 2 * m * alpha + (n - 2 * m) * beta
            a = np.full(n, beta / total)
            a[:m] = alpha / total
            a[-m:] = alpha / total
            consider_candidate(a)

    # arcsine (single candidate – already covered in sweep)
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    consider_candidate(a)

    # ------------------------------------------------------------------
    # 2. Continuation L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=40))
    max_lbgfs_iters = 400
    modes = ["random", "exp_sym", "power", "edge", "arcsine", "double_exp",
             "sidon", "dirichlet", "best"]

    # Reserve ~300 s for L‑BFGS restarts
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=max_lbgfs_iters,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[LBFGS] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb after each L‑BFGS run (≈0.8 s)
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1500
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # 3. Brief Adam fine‑tuning on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[Adam] start fine‑tuning")
        y_adam = adam_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            steps=3000,
            lr=0.06,
            lam_factor=0.20,
        )
        a_adam = softmax(y_adam)
        sc_adam = fast_score(a_adam, fft_len_conv, n)
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam.copy()
            best_y = y_adam.copy()
            print(f"[Adam] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑active projected sub‑gradient descent (core refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 70.0
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.30,
            max_iters=350_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final aggressive hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Return sequence (the harness will compute the exact score)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # Optional: you can see the final exact value here (may be slower)
    # val = evaluate_sequence(final_seq)
    # print(f"[final] exact evaluated score = {val:.6f}")
    return final_seq
