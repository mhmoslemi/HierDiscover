def simulated_annealing(a, score, fft_len, n, deadline):
    cur = a.copy()
    cur_score = score
    best = cur.copy()
    best_score = score
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    start = time.time()
    iteration = 0
    while time.time() < deadline:
        # exponential cooling based on fraction of time elapsed
        t = (time.time() - start) / total_time
        T = (0.1) * (1 - t) + 1e-6
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        move_frac = 0.05  # small moves
        delta = cur[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        if new_score < cur_score - 1e-12 or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        iteration += 1
    return best, best_score
```

def reduce_max_convolution(a, score, fft_len, n, deadline):
    cur = a.copy()
    cur_score = score
    best = cur.copy()
    best_score = score
    rng = np.random.default_rng()
    while time.time() < deadline:
        fa = np.fft.fft(cur, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        k = int(np.argmax(conv))
        # compute contributions for each index i to this k: a_i * a_{k-i}
        contrib = np.zeros(n)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib[mask] = cur[mask] * cur[partner[mask]]
        i = int(np.argmax(contrib))  # index with highest contribution
        # find another index j to shift mass to that reduces component
        # For this shift, the partner for i is p = k - i.
        # If we move mass from i to j such that product a_i*a_p reduces
        # We could move to j where partner weight small.
        p = k - i
        # pick j randomly that has small partner weight
        # compute partner weight for j: cur[k - j] if valid else 0
        # choose j with minimal partner weight
        # sample few candidates
        cand = rng.integers(0, n, size=5)
        # ensure not i
        cand = cand[cand != i]
        if len(cand) == 0:
            continue
        partner_weights = np.full_like(cand, np.inf, dtype=float)
        for idxc, cand_idx in enumerate(cand):
            partner_idx = k - cand_idx
            if 0 <= partner_idx < n:
                partner_weights[idxc] = cur[partner_idx]
        # choose the candidate with smallest partner weight
        j = int(cand[np.argmin(partner_weights)])
        # move small fraction from i to j
        delta = cur[i] * 0.05 * rng.random()
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score

def simulated_annealing_refine(a, score, fft_len, n, deadline):
    cur = a.copy()
    cur_score = score
    best = cur.copy()
    best_score = score
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # compute temperature decaying linearly
        t = (time.time() - start) / total
        T = max(1e-12, 0.05 * (1 - t))  # initial T 0.05
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.1  # move up to 10% of mass at i
        if delta <= 0.0:

        cur[j] += delta

            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------
* Initialise from the previously found best sequence if available,
  otherwise start from the uniform distribution (∑a = 1).
* Refine the distribution using a long simulated‑annealing run.
  - Random mass‑transfer moves keep the simplex constraint.
  - Every other move is *targeted*: the shift with the current maximal
    convolution is identified and a small amount of probability mass is
    moved from the index that contributes most to that shift to a
    candidate with a small partner contribution.
  - Moves that worsen the fast surrogate score are occasionally accepted
    with a temperature that decays linearly to zero, allowing the search
    to escape local minima.
* After the annealing phase, perform a projected sub‑gradient descent
  directly on the exact max‑convolution objective, using a back‑tracking
  line‑search for reliable step‑sizes.
* Return the best sequence found; the harness will call
  ``evaluate_sequence`` to obtain the final score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence``: 2·n·max(conv)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def annealing_refine(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Long simulated‑annealing run.
    Mixes random mass‑transfer moves with targeted moves that aim at the
    currently maximal convolution shift.
    """
    rng = np.random.default_rng()
    cur = a.copy()
    best = cur.copy()
    best_score = cur_score
    cur_score = cur_score

    start = time.time()
    total = max(deadline - start, 1e-12)

    # annealing schedule parameters
    T0 = 0.02                # initial temperature
    move_frac_rand = 0.08    # up‑to 8 % of the source mass for random moves
    move_frac_targ = 0.25    # up‑to 25 % of the source mass for targeted moves
    targ_prob = 0.55         # probability of performing a targeted move

    it = 0
    while time.time() < deadline:
        it += 1
        # temperature decays linearly to almost zero
        elapsed = (time.time() - start) / total
        T = max(1e-12, T0 * (1.0 - elapsed))

        # ----- choose move type -----
        if rng.random() < targ_prob:
            # ----- targeted move -----
            fa = np.fft.fft(cur, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            k_max = int(np.argmax(conv))

            idx = np.arange(n)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)

            contrib = np.zeros_like(cur)
            contrib[mask] = cur[mask] * cur[partner[mask]]

            i = int(np.argmax(contrib))
            if cur[i] <= 0.0:
                continue

            # pick a destination with a small partner under the same shift
            cand = rng.choice(n, size=6, replace=False)
            cand = cand[cand != i]
            if cand.size == 0:
                continue

            partner_weights = np.full(cand.shape, np.inf, dtype=float)
            for ci, cand_idx in enumerate(cand):
                p = k_max - cand_idx
                if 0 <= p < n:
                    partner_weights[ci] = cur[p]
            j = int(cand[np.argmin(partner_weights)])

            delta = cur[i] * move_frac_targ * rng.random()
            if delta <= 0.0:
                continue

            cur[i] -= delta
            cur[j] += delta

        else:
            # ----- random mass‑transfer move -----
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] == 0.0:
                continue
            delta = cur[i] * move_frac_rand * rng.random()
            if delta <= 0.0:
                continue
            cur[i] -= delta
            cur[j] += delta

        # ----- evaluate -----
        new_score = fast_score(cur, fft_len, n)

        # ----- acceptance test -----
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = cur.copy()
            else:
                # revert the move
                cur[i] += delta
                cur[j] -= delta

        # occasional re‑projection to combat drift
        if (it & 1023) == 0:          # every 1024 iterations
            cur = proj_simplex(cur)
            cur_score = fast_score(cur, fft_len, n)

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max (with line‑search)
# ----------------------------------------------------------------------


def subgrad_descent(
    a: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    step0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact max‑convolution objective.
    Uses a simple back‑tracking line‑search to ensure descent.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    iter_no = 0

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        iter_no += 1

        # compute convolution and the worst shift
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # gradient of the active shift (projected onto the simplex hyperplane)
        grad = np.zeros_like(best)
        partner = k - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        grad[valid] = 2.0 * best[partner[valid]]
        # make the gradient tangent to the simplex (sum‑zero)
        grad -= grad.mean()

        # ----- back‑tracking line‑search -----
        step = step0 / np.sqrt(iter_no)
        sigma = 1e-4
        beta = 0.5
        while True:
            cand = proj_simplex(best - step * grad)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score <= best_score - sigma * step * np.linalg.norm(grad) ** 2:
                break
            step *= beta
            if step < 1e-12:
                break

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        else:
            # occasional random acceptance to get out of tiny plateaus
            if rng.random() < 0.001:
                best = cand
                best_score = cand_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin for the final evaluation
    deadline = time.time() + budget_s - 4.0

    # --------------------------------------------------------------
    # Choose problem size – reuse the previous construction if possible
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # normalise onto the simplex (∑ = 1)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Fixed FFT size for the whole run
    # --------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Phase 1 – long simulated annealing refinement
    # --------------------------------------------------------------
    anneal_deadline = deadline * 0.85 + start := time.time() * 0.0  # placeholder to compute later

    # compute a realistic deadline for phase 1 (85 % of budget)
    phase1_deadline = time.time() + (deadline - time.time()) * 0.85
    best_a, best_score = annealing_refine(
        a=best_a,
        cur_score=best_score,
        fft_len=fft_len,
        n=n,
        deadline=phase1_deadline,
    )
    print(f"[phase 1] best fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    phase2_deadline = deadline
    best_a, best_score = subgrad_descent(
        a=best_a,
        cur_score=best_score,
        n=n,
        fft_len=fft_len,
        deadline=phase2_deadline,
        max_iter=6000,
        step0=0.3,
    )
    print(f"[phase 2] final fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
