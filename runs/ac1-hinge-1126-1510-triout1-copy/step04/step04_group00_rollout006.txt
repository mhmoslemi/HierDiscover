    import time, numpy as np, math
    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # margin

    # get starting sequence
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        n = arr.shape[0]
        a = np.maximum(arr, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a.sum()
    else:
        n = 5000
        a = np.full(n, 1.0/n)
    best_a = a.copy()
    best_score = fast_score(best_a, next_pow2(2*n-1), n)
    best_y = np.log(best_a + 1e-12)

    # Precompute FFT sizes
    fft_len_conv = next_pow2(2*n-1)
    fft_len_g = next_pow2(3*n-2)

    # Phase 1: L-BFGS continuation
    lam_factors = np.geomspace(0.5, 1e-5, num=30)
    maxiter_stage = 800  # a bit more iterations

    # We'll attempt up to a certain number of restarts or until time near half.
    restart = 0

        # choose seed mode
        mode = rng.choice(["random","exp","power","edge","best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001,0.02)
            a0 = np.exp(-rate*np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3,2.0)
            a0 = (np.arange(1,n+1,dtype=float) ** (-p))
            a0 /= a0.sum()

        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01,0.04)))

            a_center = 1.0/np.sqrt(2.0) / n

            y0 = np.log(a0 + 1e-12)
        else:

        y_opt = schedule_optimize(y_start=y0, n=n,
                                  fft_len_conv=fft_len_conv,

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()

        # local hill climb quick
        best_a, best_score = hill_climb(best_a, best_score,

                                    max_move_frac=0.20)

    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

"""
Propose a low‑score step‑function sequence.

The algorithm consists of four phases:

1. **Global L‑BFGS search** – many multi‑restart L‑BFGS runs on the
   soft‑max parametrisation.  A geometric continuation of the smoothing
   parameter λ drives the optimisation from a very smooth approximation
   of the max‑convolution to the true max.

2. **Sub‑gradient descent on the true max** – after the global stage the
   exact max of the auto‑convolution is reduced by moving the point on
   the probability simplex opposite to a sub‑gradient (the sum of the
   reversed active‑peak slices).  The step‑size diminishes with √t.

3. **Simulated‑annealing hill‑climb** – random mass‑transfer moves are
   accepted either if they improve the fast score or with a
   temperature‑dependent probability, allowing occasional uphill steps.

4. **Final deterministic hill‑climb** – a greedy random‑mass‑transfer
   hill‑climb finishes the optimisation.

All phases respect the constraints required by ``evaluate_sequence``:
non‑negative entries, sum > 0 (here enforced as a probability vector),
and the final sequence is returned as a plain Python list.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum == 1).

    Returns 2·n·max(a*a), which equals the value returned by
    ``evaluate_sequence`` when ``a`` sums to 1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho_arr[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max coordinates ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                     # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max (w.r.t. conv) ----
    w = exp_shift / Z                          # d smooth_max / d conv_i

    # ---- back‑propagation through convolution ----
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through the soft‑max ----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep the move
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = int(arr.size)
            a = np.maximum(arr, 0.0)
        else:
            n = 5000
            a = np.full(n, 1.0 / n)
    else:
        n = 5000
        a = np.full(n, 1.0 / n)

    # Normalise to the probability simplex
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS with λ continuation
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-5, num=30)   # geometric λ‑schedule
    maxiter_stage = 800                           # more iterations per stage

    restart = 0
    while time.time() < deadline - 400:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.03, size=n)

        # ---- continuation L‑BFGS ----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[LBFGS {restart}] mode={mode} improved fast score = {best_score:.6f}")

        # ---- short greedy hill‑climb to capture local improvements ----
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=min(deadline, time.time() + 0.9),
            max_iter=4000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)
        restart += 1

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    subgrad_iter = 0
    step0 = 0.30
    while time.time() < deadline - 250:
        subgrad_iter += 1

        # compute full convolution (FFT based)
        fa = np.fft.fft(best_a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        M = conv.max()
        # indices where the max is attained (within a tiny tolerance)
        tol = 1e-14
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max (sum of reversed active slices)
        supgrad = np.zeros(n, dtype=float)
        for k in active:
            i0 = max(0, k - (n - 1))
            i1 = min(k, n - 1)
            # a indices that appear in this slice
            idx = np.arange(k - i0, k - i1 - 1, -1)   # descending
            supgrad[i0 : i1 + 1] += 2.0 * best_a[idx]

        step = step0 / math.sqrt(subgrad_iter)
        cand = proj_simplex(best_a - step * supgrad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)

        if subgrad_iter % 200 == 0:
            print(f"[subgrad] iter={subgrad_iter} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    sa_total_iters = 20000
    sa_iter = 0
    temp_start = 1e-3
    temp_end = 1e-7
    while time.time() < deadline - 50 and sa_iter < sa_total_iters:
        # linear temperature schedule
        frac = sa_iter / sa_total_iters
        T = temp_start * (1 - frac) + temp_end * frac

        i, j = rng.integers(0, n, size=2)
        if i != j:
            delta = best_a[i] * rng.random() * 0.30  # up to 30 % of source mass
            if delta > 0.0:
                cand = best_a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_score = fast_score(cand, fft_len_conv, n)

                if cand_score < best_score:
                    best_a, best_score = cand, cand_score
                else:
                    # Metropolis acceptance
                    prob = math.exp(-(cand_score - best_score) / T)
                    if rng.random() < prob:
                        best_a, best_score = cand, cand_score

        sa_iter += 1
        if sa_iter % 2000 == 0:
            print(f"[SA] iter={sa_iter} T={T:.2e} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final deterministic hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    best_a, best_score = hill_climb(
        a=best_a,
        cur_score=best_score,
        fft_len=fft_len_conv,
        n=n,
        deadline=deadline,
        max_iter=12000,
        max_move_frac=0.20,
    )

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
