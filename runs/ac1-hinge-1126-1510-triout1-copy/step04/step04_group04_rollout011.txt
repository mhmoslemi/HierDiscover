"""
Improved search for a step‑function sequence minimizing the evaluation function.
(1) Random support + Dirichlet initialisation.
(2) Active‑set sub‑gradient descent that uses *all* near‑maximal convolution entries.
(3) Fine optimisation by a smooth‑max L‑BFGS cascade.
(4) Final greedy hill‑climb.
All steps respect the constraints of `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns an n‑vector of non‑negative numbers summing to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return e / s

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT (real part only)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming Σa = 1."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("project_to_simplex works on 1‑D arrays")
    # Standard O(n) algorithm (Michelot)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a random fraction of mass from i to j."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # max shifted to 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z     # derivative of smooth_max w.r.t. conv_i

    # gradient wrt a : g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def active_subgradient_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_g: int,
    deadline: float,
    init_step: float = 1.0,
    min_step: float = 1e-7,
    eps_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that uses **all** convolution entries whose
    value is within ``eps_frac`` of the current maximum.
    Projection on the simplex guarantees feasibility.
    """
    a = a0.copy()
    cur_score = fast_score(a, next_pow2(2 * n - 1), n)
    step = init_step
    rng = np.random.default_rng()
    while time.time() < deadline:
        # exact linear convolution (FFT is faster for n≈5000)
        conv = conv_fft(a, next_pow2(2 * n - 1), n)
        M = conv.max()
        eps = max(eps_frac * M, 1e-12)
        mask = (conv >= M - eps).astype(float)          # shape (2n‑1,)

        # gradient = 2 * (a_rev ⊛ mask)[n‑1 : n‑1+n]
        fa_rev = np.fft.fft(a[::-1], fft_len_g)
        fw = np.fft.fft(mask, fft_len_g)
        grad_full = np.real(np.fft.ifft(fa_rev * fw))
        grad = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # simple projected gradient step (descent direction = +grad, we step opposite)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, next_pow2(2 * n - 1), n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            step = min(step * 1.1, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return a, cur_score

def generate_random_support_vec(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Create a probability vector whose support has size k."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.random(k)
    w /= w.sum()
    a = np.zeros(n)
    a[idx] = w
    return a

"""
Improved search for a step‑function sequence minimizing the evaluation function.

Steps
-----
1️⃣ Random initialisations (support‑sized vectors, Dirichlet draws, small
   perturbations of the current best).

2️⃣ Active‑set sub‑gradient descent:
   * All convolution entries whose value is within a tiny fraction of the
     current maximum are treated as “active”.
   * A projected gradient step (using the exact gradient of the max‑convolution)
     reduces the peak.
   * Projection onto the probability simplex guarantees feasibility.

3️⃣ Fine polishing with a smooth‑max L‑BFGS cascade (decreasing smoothing).

4️⃣ Final greedy hill‑climb that moves a small amount of mass between random
   positions as long as the fast score improves.

All helpers are top‑level, no closures or lambdas, and the code respects the
1000 s time budget. The returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming Σa = 1."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb moving a random fraction of mass from i to j."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05  # up to 5 % of the mass at i
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z  # d smooth_max / d conv_i

    # gradient w.r.t. a: g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Active‑set sub‑gradient optimisation
# ----------------------------------------------------------------------


def active_subgradient_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_g: int,
    deadline: float,
    init_step: float = 1.0,
    min_step: float = 1e-7,
    eps_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using **all** convolution entries whose value is within
    ``eps_frac`` of the current maximum.  After each step the vector is projected
    onto the probability simplex.
    """
    a = a0.copy()
    cur_score = fast_score(a, next_pow2(2 * n - 1), n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        # exact convolution (FFT version)
        conv = conv_fft(a, next_pow2(2 * n - 1), n)
        M = conv.max()
        eps = max(eps_frac * M, 1e-12)
        mask = (conv >= M - eps).astype(float)   # shape (2n‑1,)

        # gradient = 2 * (a_rev ⊛ mask)[n‑1 : n‑1+n]
        fa_rev = np.fft.fft(a[::-1], fft_len_g)
        fw = np.fft.fft(mask, fft_len_g)
        grad_full = np.real(np.fft.ifft(fa_rev * fw))
        grad = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # projected gradient descent step
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, next_pow2(2 * n - 1), n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            step = min(step * 1.1, 2.0)   # modest acceleration
        else:
            step *= 0.5
            if step < min_step:
                break
    return a, cur_score


def generate_random_support_vec(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector of length n whose support has exactly k entries."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.random(k)
    w /= w.sum()
    a = np.zeros(n)
    a[idx] = w
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine mixes random support generation,
    an active‑set sub‑gradient optimiser, a smooth‑max L‑BFGS polishing
    and a final greedy hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – reuse a previously discovered construction if any
    # --------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1000:
            if tmp.size > n:
                a0 = tmp[:n]
            else:
                a0 = np.concatenate([tmp, np.zeros(n - tmp.size, dtype=float)])
            a0 = np.clip(a0, 0, None)
            s = a0.sum()
            a0 = np.full(n, 1.0 / n) if s < 1e-12 else a0 / s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # FFT lengths (pre‑computed)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Phase 1 – random restarts + active‑set sub‑gradient descent
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 30.0:   # reserve ≈30 s for polishing
        mode = rng.choice(["support", "dirichlet", "best"])
        if mode == "support":
            # support size roughly between 0.8·√n and 2.2·√n
            k = int(rng.integers(int(np.sqrt(n) * 0.8), int(np.sqrt(n) * 2.2) + 1))
            a0 = generate_random_support_vec(n, k, rng)
        elif mode == "dirichlet":
            alpha = rng.uniform(0.1, 5.0)
            a0 = rng.dirichlet(np.ones(n) * alpha)
        else:  # small perturbation of the current best
            a0 = best_a + rng.normal(scale=0.02, size=n)
            a0 = np.clip(a0, 0, None)
            a0 = a0 / a0.sum()

        sub_deadline = min(deadline - 30.0, time.time() + 5.0)  # ~5 s per trial
        a_opt, score_opt = active_subgradient_optimize(
            a0, n, fft_len_g, sub_deadline, init_step=0.8, eps_frac=0.001
        )
        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            print(f"[iter {iteration}] mode={mode} new best fast score = {best_score:.6f}")
        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – smooth‑max L‑BFGS cascade (fine polishing)
    # --------------------------------------------------------------
    y = np.log(best_a + 1e-12)
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=30))
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 300, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        best_a = softmax(y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )
        print(f"[final hill] fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
