def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

import scipy.sparse as sp

def solve_lp_max_sum(f, M):
    n = len(f)
    rows = 2*n - 1
    # Build data for sparse matrix
    data = []
    row_indices = []
    col_indices = []
    for k in range(rows):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            j = k - i
            data.append(f[i])
            row_indices.append(k)
            col_indices.append(j)
    A = sp.csr_matrix((data, (row_indices, col_indices)), shape=(rows, n))
    c = -np.ones(n)
    b = M * np.ones(rows)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
A multi‑restart optimiser that adds a linear‑programming direction step.
The LP step solves:
    maximise   sum(g_j)
    subject to   (f * g)(k) ≤ M   for all k
                g_j ≥ 0
where M = max_k (f * f)(k).
The solution g points to a direction that can increase the total mass
without raising the maximal auto‑correlation.  By blending the current
vector with the normalised LP‐solution we decrease the objective
2·n·max(conv)/(sum²).  The optimiser performs many random restarts,
a quick projected sub‑gradient phase, a few LP‑updates, and a final
greedy hill‑climb.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a sequence that is already on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (quick descent)
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Very light projected sub‑gradient descent with momentum.
    Used only as a fast pre‑processor before the LP stage.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # linear convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n - 1: 2 * n - 1]

        # momentum update
        v = beta * v + (1.0 - beta) * g
        step = 0.5 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def lp_max_sum_direction(f: np.ndarray, M: float):
    """
    Solve
        maximise    sum(g)
        subject to  (f * g)(k) ≤ M   for all shifts k
                    g ≥ 0
    Returns the raw solution g (may be un‑normalised).  If the LP
    is unbounded or fails, ``None`` is returned.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build a sparse Toeplitz‑like matrix A where
    # A[k, j] = f[k - j]  (if 0 ≤ k‑j < n, else 0)
    # This can be expressed as n constant diagonals.
    offsets = -np.arange(n)                     # 0, -1, … , -(n‑1)
    data = [np.full(n, f[d]) for d in range(n)]  # each diagonal has length n
    A = sp.diags(data, offsets, shape=(rows, n), format='csr')

    # Objective: maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # linprog with the HiGHS interior‑point solver handles sparse matrices efficiently.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30.0})

    if not res.success:
        return None
    return res.x


def lp_improve(f: np.ndarray,
               max_iter: int,
               deadline: float,
               fft_len_conv: int,
               n: int,
               blend_factor: float = 0.25):
    """
    Repeatedly apply the LP direction step and blend the result back into ``f``.
    ``blend_factor`` controls the convex combination weight for each LP update.
    """
    cur = f.copy()
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        # current maximal autocorrelation
        M = full_conv(cur, fft_len_conv).max()
        # LP direction
        g_raw = lp_max_sum_direction(cur, M)
        if g_raw is None:
            break
        # normalise the direction to the simplex (sum = 1)
        if g_raw.sum() == 0:
            break
        g = g_raw / g_raw.sum()
        # convex blend and re‑project onto the simplex
        cur = simplex_projection((1.0 - blend_factor) * cur + blend_factor * g)
    return cur


# ----------------------------------------------------------------------
# Greedy hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000):
    """
    Simple greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The method combines random restarts, a fast sub‑gradient phase,
    a linear‑programming direction step, and final greedy hill‑climb.
    The best list found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    max_n = 3000                     # upper bound on the sequence length
    # if a previous construction exists we start from it, otherwise from uniform
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        n = min(init_seq.size, max_n)
        a0 = init_seq[:n].copy()
        a0[a0 < 0] = 0.0
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n // 2               # moderate default size
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Phase A – many short random restarts with sub‑gradient descent
    # ------------------------------------------------------------------
    phase_a_end = start_time + 0.30 * budget_s
    while time.time() < phase_a_end:
        a_start = random_initial(n, rng)
        # give each restart a tiny budget for the cheap sub‑gradient phase
        sg_deadline = min(phase_a_end, time.time() + 0.03)
        a_opt, score_opt = subgradient_optimize_momentum(a_start,
                                                         sg_deadline,
                                                         fft_len_conv,
                                                         fft_len_g,
                                                         n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[sub‑grad restart] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – LP‑guided improvement on the current best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = deadline - 0.3
        # a handful of LP updates (each cheap because n is moderate)
        best_a = lp_improve(best_a,
                            max_iter=8,
                            deadline=lp_deadline,
                            fft_len_conv=fft_len_conv,
                            n=n,
                            blend_factor=0.2)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[LP improve] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=6000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
