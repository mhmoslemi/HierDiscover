"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works in three phases:

1️⃣  **Exploratory stage on a smaller problem (n≈2000).**
    Many random restarts are performed; each restart runs a cascade of
    L‑BFGS optimisations on a smooth‑max approximation of the convolution
    peak, followed by a short hill‑climb and a sub‑gradient descent.
    The best distribution found for the small size is kept.

2️⃣  **Upsampling & fine‑tuning on the target size (n≈5000).**
    The small‑size solution is interpolated to the full length,
    re‑normalised to sum = 1 and then refined with a second L‑BFGS cascade,
    an aggressive hill‑climb, a sub‑gradient descent and a final L‑BFGS
    run with a very small smoothing parameter.

3️⃣  **Polishing.**  A last hill‑climb runs until the time budget expires.
    The resulting vector is returned as a plain list of non‑negative floats.

All operations respect the constraints checked by ``evaluate_sequence``:
non‑negative entries, finite values and a positive total mass.
The code is deliberately written with all helpers at module level
(no closures, no lambdas) and uses only NumPy / SciPy.
"""

import time
import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part only)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the convolution peak and its gradient w.r.t. y.
    ``lam`` is the smoothing parameter (larger → smoother).
    """
    a = softmax(y)

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # avoid division by zero
    lam_eff = max_c * lam + 1e-12
    shifted = conv - max_c                      # now max = 0
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth max ---
    w = exp_shift / Z                      # d smooth_max / d conv_i
    # compute w * a (convolution of a with w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # the relevant part (n elements) corresponds to indices n‑1 … 2n‑2
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # chain rule through softmax
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS calls with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Random initialisers (pre‑soft‑max)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point `y` (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # heavy mass near the two ends, lighter interior
        edge_frac = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Local search primitives
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a random fraction of mass from i to j.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    After each step the vector is projected back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        # gradient of the active convolution entry
        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # tiny jitter to avoid getting stuck on flat plateaus
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score

def upsample_distribution(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """
    Linear interpolation of a distribution from size `len(a_small)`
    to `n_big`.  The result is re‑normalised to sum = 1.
    """
    m_small = a_small.size
    x_small = np.linspace(0.0, 1.0, m_small)
    x_big = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_big, x_small, a_small)

    total = a_big.sum()
    if total <= 0.0:
        a_big = np.full(n_big, 1.0 / n_big)
    else:
        a_big /= total
    return a_big

        sub_deadline = min(deadline, time.time() + 0.5)
        a_opt, score_opt = subgradient_optimize(
            a_opt,
            n_big,
            fft_len_big,
            sub_deadline,
            init_step=0.5,
        )

        if score_opt < best_score_big:
            best_score_big = score_opt
            best_a_big = a_opt
            best_y_big = np.log(best_a_big + 1e-12)
            print(
                f"[big] iter={iteration_big} mode={mode} improved score={best_score_big:.6f}"
            )
        iteration_big += 1

    # --------------------------------------------------------------
    # 3️⃣  Final fine L‑BFGS with a very small smoothing factor
    # --------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.2, 2e-6, num=25))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y_big,
            args=(n_big, fft_len_big, fft_len_g_big, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 600, "gtol": 1e-12, "disp": False},
        )
        best_y_big = res.x
        best_a_big = softmax(best_y_big)
        best_score_big = fast_score(best_a_big, fft_len_big, n_big)
        print(f"[fine L‑BFGS] lam={lam:.3e} fast score={best_score_big:.6f}")

    # --------------------------------------------------------------
    # Final greedy polishing (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a_big, best_score_big = hill_climb(
            best_a_big,
            best_score_big,
            fft_len_big,
            n_big,
            deadline,
            max_iter=8000,
        )
        print(f"[final hill‑climb] fast score={best_score_big:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a_big.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
