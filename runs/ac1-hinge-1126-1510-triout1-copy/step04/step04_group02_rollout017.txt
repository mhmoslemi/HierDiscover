"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm Overview
------------------
We treat the problem as minimising   max_k (a * a)[k]   with a_i ≥ 0, Σ a_i = 1.
For a probability vector the evaluation becomes 2·n·max(conv).

The search consists of three stages (running on two CPU cores in parallel):
1. **Stochastic L‑BFGS optimisation**
   * A soft‑max parametrisation y → a = softmax(y) guarantees the simplex.
   * The nonsmooth max is approximated by a log‑sum‑exp with a decreasing
     smoothing parameter lam.  A cascade of short L‑BFGS runs (continuation)
     refines the solution from several random starts.
2. **Projected sub‑gradient descent**
   * Directly minimises the true objective (the maximum convolution entry).
   * Uses the exact sub‑gradient of the active convolution index and Euclidean
     projection onto the probability simplex.
3. **Aggressive greedy polishing**
   * A long hill‑climb that moves a random fraction of mass between two
     coordinates; later iterations accept *any* improvement, early iterations
     also allow small uphill steps (simulated‑annealing style).

Both processes split the total time budget (default 1000 s) equally between two
workers; the best candidate from either worker is returned.

The implementation avoids closures, uses only ``numpy``/``scipy`` and respects
the 2‑CPU limit.
"""

import time
import numpy as np
from scipy.optimize import minimize
from concurrent.futures import ProcessPoolExecutor, as_completed

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n-1) using FFT of length ``fft_len``."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns smoothed objective f = 2·n·smooth_max(conv) and its gradient
    w.r.t. soft‑max parameters y.
    """
    a = softmax(y)

    # ----- convolution of a with itself -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12           # smoothing scale
    shifted = conv - max_c                      # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. conv -----
    w = exp_shift / Z          # ∂ smooth_max / ∂ conv_i   (size 2n‑1)

    # ----- back‑propagation to y (softmax) -----
    # g_vec = Σ_j a_j * w_{i+j}   for i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing parameters.  Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial‑point generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for the given ``mode``.
    Modes:
        random  – standard normal noise
        exp     – exponential decay
        power   – power‑law decay
        edge    – heavy mass on the two ends
        sidon   – greedy Sidon‑set support (≈√n)
        best    – small Gaussian perturbation of the current best
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_center = 1.0 / np.sqrt(2.0) / n
        a_edge = np.sqrt(n / 2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate) not in sums:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Local search heuristics
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.10,
    temperature: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Greedy (or simulated‑annealing) hill‑climb.
    Moves a random fraction of mass from i to j.
    If ``temperature`` > 0, uphill moves may be accepted with probability exp(-Δ/T).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        improve = new_score < best_score
        accept = improve
        if not improve and temperature > 0.0:
            # simulated‑annealing acceptance
            accept = np.exp(-(new_score - best_score) / temperature) > rng.random()
        if accept:
            if new_score < best_score:
                best_score = new_score
            # keep the move
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    (max convolution entry).  Returns the best iterate encountered.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # Sub‑gradient at the active shift
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Core search routine (run in each worker process)
# ----------------------------------------------------------------------


def _run_worker(seed: int, n: int, init_seq: np.ndarray, time_budget: float) -> tuple[list, float]:
    """
    Independent optimisation thread.
    Returns (best_sequence_as_list, best_score).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + time_budget - 0.5  # tiny safety margin

    # ----- FFT lengths (powers of two for speed) -----
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ----- initialise best candidate (probability simplex) -----
    if init_seq is not None and init_seq.size == n:
        best_a = init_seq / np.sum(init_seq)
    else:
        best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[worker {seed}] start n={n} init score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Stochastic L‑BFGS phase (≈ 60 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=30))
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    while time.time() < deadline - 150.0:   # reserve time for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[worker {seed}] mode={mode} improved L‑BFGS score={best_score:.6f}")

        # Quick hill‑climb to polish this L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
            max_move=0.08,
            temperature=0.0,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent (≈ 30 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 20.0
        print(f"[worker {seed}] starting sub‑gradient descent")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.20,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[worker {seed}] sub‑grad finished best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final aggressive hill‑climb (simulated‑annealing style)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        # temperature schedule: start slightly warm, cool to zero
        start_T = 0.03
        end_T = 0.0
        total_hc_iters = 20000
        cur_T = start_T
        t0 = time.time()
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=total_hc_iters,
            max_move=0.12,
            temperature=cur_T,
        )
        best_y = np.log(best_a + 1e-12)

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by harness
    print(f"[worker {seed}] final evaluated score = {final_val:.6f}")
    return final_seq, final_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    The routine runs two parallel workers and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we target a moderately large n (≈ 8000)
    # ------------------------------------------------------------------
    n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
    init_arr = np.asarray(init_seq, dtype=float) if isinstance(init_seq, (list, tuple, np.ndarray)) else None
    if init_arr is not None and init_arr.size != n:
        init_arr = None  # fallback to uniform if size mismatches

    # ------------------------------------------------------------------
    # Split time budget between two workers
    # ------------------------------------------------------------------
    workers = 2
    per_worker_budget = budget_s / workers

    results = []
    with ProcessPoolExecutor(max_workers=workers) as executor:
        futures = [
            executor.submit(_run_worker, rng.integers(1_000_000), n, init_arr, per_worker_budget)
            for _ in range(workers)
        ]
        for fut in as_completed(futures):
            seq, score = fut.result()
            results.append((seq, score))

    # pick the best result
    best_seq, best_score = min(results, key=lambda x: x[1])
    print(f"[master] best score after parallel search = {best_score:.6f}")
    return best_seq
