"""Search for a low‑score step‑function via multi‑start continuation L‑BFGS,
projected sub‑gradient descent and directed gradient‑balancing moves.

The algorithm proceeds in phases:
1. initialise a probability vector a (sum = 1) either from a supplied
   global `height_sequence_1` or from a diverse set of random
   parametrisations (uniform, exponential, power‑law, edge‑heavy,
   Sidon‑set, Gaussian mixtures).
2. For the majority of the time budget run many continuation L‑BFGS
   optimisations of a soft‑max parametrisation y, using a decreasing
   log‑sum‑exp smoothing of the max‑convolution.
   After each L‑BFGS run a short greedy hill‑climb refines the candidate.
3. A projected sub‑gradient descent on the exact max‑convolution
   objective further reduces the value.
4. A specialised “gradient‑balance” local search repeatedly moves a
   fraction of probability mass from indices that contribute most to
   the current maximal convolution entry to indices that contribute
   the least.
   Several random perturbations of the best solution are tried.
5. Final aggressive hill‑climb polishing.

The best vector found is returned as a plain list; the harness then
evaluates it with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        # fallback to uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. soft‑max params
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp approximation of the max‑convolution objective
    and its gradient with respect to the unconstrained vector y
    (soft‑max parametrisation).

    The smoothness parameter λ is set to lam_factor * max(conv) + ε.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c            # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                # ∂smooth_max / ∂conv_i

    # gradient of smooth_max w.r.t. a:
    # (a * (conv_rev - mean_g)) works as derived in the reference code
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a series of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # coarse edge‑heavy shape
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge_rand":
        # random edge size and weight ratio
        m_edge = max(1, int(n * rng.uniform(0.005, 0.08)))
        ratio = rng.uniform(1.0, 12.0)          # edge weight = ratio * interior weight
        interior_weight = 1.0
        edge_weight = ratio * interior_weight
        total = (n - 2 * m_edge) * interior_weight + 2 * m_edge * edge_weight
        a0 = np.full(n, interior_weight / total)
        a0[:m_edge] = edge_weight / total
        a0[-m_edge:] = edge_weight / total
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "gauss":
        # mixture of two Gaussians
        mu1 = rng.integers(int(0.2 * n), int(0.8 * n))
        mu2 = rng.integers(int(0.2 * n), int(0.8 * n))
        sigma1 = rng.uniform(0.03 * n, 0.15 * n)
        sigma2 = rng.uniform(0.03 * n, 0.15 * n)
        w = rng.random()
        x = np.arange(n, dtype=float)
        a0 = w * np.exp(-0.5 * ((x - mu1) / sigma1) ** 2)
        a0 += (1.0 - w) * np.exp(-0.5 * ((x - mu2) / sigma2) ** 2)
        if a0.sum() == 0:
            a0 = np.ones(n)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "best":
        raise ValueError("mode 'best' should be handled outside")
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j
    (keeping the sum = 1). Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] <= 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective f(a)=max(conv).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # compute sub‑gradient g_i = 2 a_{m-i}
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def gradient_balance(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 30_000,
    start_move_frac: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Directed local search that repeatedly moves probability mass from
    indices that contribute most to the current maximal convolution entry
    to indices that contribute the least.
    """
    cur_a = a_start.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute convolution and index of maximum
        conv = conv_fft(cur_a, fft_len, n)
        m = int(np.argmax(conv))

        # compute counterpart values b_i = a_{m-i} for valid i
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        b = np.zeros_like(cur_a)
        b[mask] = cur_a[counterpart[mask]]

        # contributions to the maximal entry
        contrib = cur_a * b

        # source: index with largest contribution (and non‑zero mass)
        # if all contributions equal (unlikely) we skip
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src_idx = src_candidates[np.argmax(contrib[src_candidates])]

        # target: index with smallest counterpart value (prefer zero)
        # break ties by choosing the one with smallest a (so mass moves outward)
        tgt_idx = np.argmin(b + (np.arange(n) == src_idx) * 1e6)

        if src_idx == tgt_idx:
            # choose a random alternative target to avoid stagnation
            tgt_idx = rng.integers(0, n)
            if tgt_idx == src_idx:
                continue

        # move a fraction of source mass toward target
        move_frac = start_move_frac * (0.995 ** it)  # decay slowly
        move_frac = max(move_frac, 0.001)           # lower bound
        delta = cur_a[src_idx] * move_frac
        if delta <= 0.0:
            continue
        cur_a[src_idx] -= delta
        cur_a[tgt_idx] += delta

        new_score = fast_score(cur_a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[src_idx] += delta
            cur_a[tgt_idx] -= delta
    return best_a, best_score

"""Search for a low‑score step‑function via multi‑start L‑BFGS,
projected sub‑gradient descent and directed gradient‑balancing moves.

The algorithm runs in four phases:
1. Initialise a probability vector (sum=1) from a supplied construction
   or from diverse random parametrisations (uniform, exponential,
   power‑law, edge‑heavy, Sidon, Gaussian mixtures).
2. Perform many continuation L‑BFGS optimisations on a smoothed
   log‑sum‑exp surrogate of the max‑convolution, interleaved with a
   short greedy hill‑climb.
3. Refine the best vector with projected sub‑gradient descent on the
   true max‑convolution objective.
4. Apply a specialised “gradient‑balance” local search that repeatedly
   moves mass from indices that contribute most to the current maximal
   convolution entry to indices that contribute least, with several
   random restarts.  A final aggressive hill‑climb polish is applied
   before returning the result.

All helper functions are defined at the top level; the harness will
call ``evaluate_sequence`` on the returned list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. soft‑max params
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp approximation of the max‑convolution objective
    and its gradient w.r.t. the unconstrained vector y (soft‑max
    parametrisation).

    The smoothness parameter λ = lam_factor * max(conv) + ε.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                      # ∂smooth_max / ∂conv_i

    # gradient of smooth_max w.r.t. a (via convolution with w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:

    Run a series of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.

    y = y_start.copy()

        mu1 = rng.integers(int(0.2 * n), int(0.8 * n))
        mu2 = rng.integers(int(0.2 * n), int(0.8 * n))
        sigma1 = rng.uniform(0.03 * n, 0.15 * n)
        sigma2 = rng.uniform(0.03 * n, 0.15 * n)
        w = rng.random()
        x = np.arange(n, dtype=float)
        a0 = w * np.exp(-0.5 * ((x - mu1) / sigma1) ** 2)
        a0 += (1.0 - w) * np.exp(-0.5 * ((x - mu2) / sigma2) ** 2)
        if a0.sum() == 0:
            a0 = np.ones(n)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

# Local search primitives
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j while
    keeping the sum = 1.  Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] <= 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(conv) with sum(a)=1.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def gradient_balance(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 30_000,
    start_move_frac: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Directed local search that repeatedly moves probability mass from
    indices that contribute most to the current maximal convolution entry
    to indices that contribute the least.
    """
    cur_a = a_start.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(cur_a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        b = np.zeros_like(cur_a)
        b[mask] = cur_a[counterpart[mask]]

        contrib = cur_a * b
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src_idx = src_candidates[np.argmax(contrib[src_candidates])]
        tgt_idx = np.argmin(b + (np.arange(n) == src_idx) * 1e6)
        if src_idx == tgt_idx:
            tgt_idx = rng.integers(0, n)
            if tgt_idx == src_idx:
                continue

        move_frac = start_move_frac * (0.995 ** it)
        move_frac = max(move_frac, 0.001)
        delta = cur_a[src_idx] * move_frac
        if delta <= 0.0:
            continue
        cur_a[src_idx] -= delta
        cur_a[tgt_idx] += delta

        new_score = fast_score(cur_a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            cur_a[src_idx] += delta
            cur_a[tgt_idx] -= delta
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns a plain list of non‑negative floats (sum > 0).  The harness
    will call ``evaluate_sequence`` on the result.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderately large n gives a richer search space
    # ------------------------------------------------------------------
    n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 12000)

    # FFT lengths – powers of two for speed
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate on the probability simplex (sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – stochastic continuation L‑BFGS + quick hill‑climb
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=24))
    iteration = 0
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "edge_rand",
        "sidon",
        "gauss",
        "best",
    ]
    while time.time() < deadline - 220.0:   # reserve ~3 min for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}"
            )

        # short hill‑climb polishing
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
            max_move=0.12,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 80.0
        print("[subgrad] starting projected sub‑gradient descent")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.25,
            max_iters=600_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – directed gradient‑balance search with random restarts
    # ------------------------------------------------------------------
    bal_deadline = deadline - 30.0
    if time.time() < bal_deadline:
        print("[balance] start directed gradient‑balance search")
        best_a, best_score = gradient_balance(
            best_a,
            fft_len_conv,
            n,
            bal_deadline,
            max_iter=40_000,
            start_move_frac=0.14,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[balance] after deterministic run fast score = {best_score:.6f}")

        while time.time() < bal_deadline - 5.0:
            pert = best_a + rng.normal(scale=0.004, size=n)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum() + 1e-30
            req_deadline = min(bal_deadline, time.time() + 12.0)
            bal_a, bal_score = gradient_balance(
                pert,
                fft_len_conv,
                n,
                req_deadline,
                max_iter=20_000,
                start_move_frac=0.11,
            )
            if bal_score < best_score:
                best_score = bal_score
                best_a = bal_a.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[balance] improved via restart fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final aggressive hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_hc_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_hc_deadline,
            max_iter=20_000,
            max_move=0.18,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Return the candidate
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
