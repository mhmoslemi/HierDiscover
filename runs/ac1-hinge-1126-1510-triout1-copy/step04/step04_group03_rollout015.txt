"""
Improved step‑function search using a linear‑programming exchange step.

The algorithm:
1.  Initialise a non‑negative sequence (use the previous best if available,
    otherwise a uniform distribution) and project onto the probability simplex.
2.  Repeatedly:
        – compute the current maximal self‑convolution M,
        – solve the LP  max 1·g   s.t.  g≥0  and  f∗g ≤ M   (f fixed),
        – normalise the LP solution to sum=1,
        – update f ← (1‑τ)·f + τ·g  (τ decays during the run).
    This “exchange” step strictly reduces the peak of the convolution.
3.  A short continuation L‑BFGS on a smoothed max (log‑sum‑exp) refines the
    result.
4.  A final greedy hill‑climb polishing step.
All steps respect the simplex constraint, use FFT‑based convolutions and stay
well within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a already normalised (sum = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, ∑x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (for L‑BFGS refinement)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (same as the previous code)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Continuation L‑BFGS on the smoothed max."""
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
        # short hill‑climb after each λ
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """Greedy mass‑transfer hill‑climb (keeps sum = 1)."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# LP‑exchange optimisation
# ----------------------------------------------------------------------
def build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1) × n matrix A where (A·g)_k = (f∗g)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def lp_exchange_optimize(a0: np.ndarray,
                         deadline: float,
                         max_iter: int = 30,
                         tau_start: float = 0.4,
                         tau_decay: float = 0.9):
    """
    Exchange iteration:
        maximise ∑g   s.t.   g≥0,   f∗g ≤ M
    where M = max(f∗f).  Update f ← (1‑τ)f + τg.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    tau = tau_start

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # current peak
        conv = full_conv(a, fft_len)
        M = conv.max()

        # build linear constraints f∗g ≤ M
        A = build_constraint_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # linear programme: maximise ∑g  →  minimise −∑g
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        # solve with HiGHS (fast interior‑point)
        res = linprog(c=c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if res.status != 0 or res.x is None:
            # LP failed – abort exchange loop
            print("[lp] iteration", it, "failed (status", res.status, ")")
            break

        g_raw = res.x
        s = g_raw.sum()
        if s <= 0:
            print("[lp] zero‑sum solution, stop")
            break
        g = g_raw / s                     # normalise to sum = 1

        # convex combination
        a = (1.0 - tau) * a + tau * g
        # (a stays on the simplex automatically)

        # evaluate
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[lp] iter {it:02d} τ={tau:.3f} score={best_score:.6f}")

        # decay step size
        tau *= tau_decay
        if tau < 0.01:
            tau = 0.01

    return best_a, best_score
