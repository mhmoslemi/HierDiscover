"""
Propose a step‑function (non‑negative vector) with a low score for
evaluate_sequence.  The algorithm is a hybrid of:

1.  A broad parametric sweep (uniform, linear, exponential, power‑law,
    sinusoidal, random Dirichlet, mixtures) to collect promising seeds.
2.  Continuation L‑BFGS on a log‑sum‑exp smooth approximation of the
    max‑convolution.  The λ‑schedule is extended down to 1e‑4 and many
    random restarts are performed.
3.  A long projected sub‑gradient descent with momentum that uses the
    exact max‑convolution (mask of active indices) as a sub‑gradient.
4.  Simulated‑annealing / random perturbations to escape tiny basins.
5.  A final greedy hill‑climb (mass transfer) for polishing.

All phases respect the probability simplex (a_i≥0, Σa_i=1) and use
FFT‑based convolutions (O(n log n)).  The time budget (~1000 s) is split
automatically; the best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast proxy for evaluate_sequence (a is already on the simplex)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def smooth_objective_and_grad(y: np.ndarray, n: int,
                              fft_len_conv: int, fft_len_g: int,
                              lam: float):
    """
    Log‑sum‑exp smooth approximation of max(conv) and its gradient.
    Returns (objective, gradient) where the objective is
    2·n·smooth_max (the factor 2·n is constant for the optimizer).
    """
    a = softmax(y)                     # a lives on the simplex

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i
    # convolution of a_rev with w (linear term of chain rule)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # same length as a

    # chain rule through soft‑max (see e.g. Amari 1997)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected (momentum) sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        deadline: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int,
                        init_step: float = 0.5):
    """
    Projected sub‑gradient descent with simple momentum.
    Optimises the true objective 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = fast_score(a, fft_len_conv, n)

    # momentum parameters
    vel = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (as in the paper)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple Nesterov‑type momentum
        step = init_step / np.sqrt(it + 1.0)
        vel = 0.9 * vel - step * g
        a_cand = simplex_projection(a + vel)

        cand_val = fast_score(a_cand, fft_len_conv, n)
        if cand_val < best_val:
            best_val = cand_val
            best_a = a_cand.copy()
        a = a_cand
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer) – strictly improving moves only
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """Mass‑transfer hill‑climb that preserves the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07  # up to 7 % of mass from i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS on the smooth max
# ----------------------------------------------------------------------
def lbfgs_continuation(start_a: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous one.
    """
    y = np.log(start_a + 1e-12)          # initialise logits
    best_a = start_a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=smooth_objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # occasional numerical failures – just continue
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # short hill‑climb after each λ to sharpen
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)
    return best_a, best_score


# ----------------------------------------------------------------------
# Random annealing / perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  cur_score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """Accept occasional worsening moves with a slowly cooling temperature."""
    cur_a = a.copy()
    cur_score = cur_score
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T = 0.025
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # up to 12 % of a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T = max(T * 0.90, 1e-5)

    return best_a, best_score

    # ------------------------------------------------------------------
