"""
Optimiser for the step‑function evaluation problem.

Main ideas
----------
*   A fast parametric sweep produces a set of promising seeds.
*   Each seed is refined by a custom Adam optimiser that minimises a
    smoothed version of the max‑convolution (log‑sum‑exp smoothing).
    The smoothing factor λ is decreased according to a schedule, which
    gradually turns the objective into the true max‑convolution.
*   The Adam optimiser works on the soft‑max (log‑probability) parameters,
    thus keeping the sequence non‑negative and automatically normalised.
*   After the stochastic optimisation we run a long projected sub‑gradient
    descent (exact max, not smoothed) to push the peak down further.
*   Finally a greedy hill‑climb polishes the solution.
*   The whole search respects the 1000 s budget; the best sequence found
    before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max: ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score for a simplex‑normalised sequence a: 2·n·max(a*a)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (objective, gradient) for the smoothed max objective.
    The objective equals 2·n·smooth_max(conv_a), where conv_a = a*a
    and a = softmax(y).
    """
    a = softmax(y)

    # a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smoothed max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                # smoothing scale
    shifted = conv_full - max_c                     # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                                # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # obtain the part that aligns with a (indices n‑1 … 2n‑2)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max objective.
    Works in the log‑probability space (soft‑max), therefore the simplex
    constraints are satisfied automatically.
    """
    # initialise logits
    eps = 1e-12
    y = np.log(a_start + eps)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    lr0 = 0.15

    # schedule for the smoothing factor λ
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.06, 0.03,
                    0.015, 0.008, 0.004, 0.002, 0.001,
                    5e-4, 2e-4, 1e-4]

    steps_per_lam = 300            # internal Adam steps for each λ
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score

            t += 1
            f_val, grad = objective_and_grad(y, n,
                                             fft_len_conv,
                                             fft_len_g,
                                             lam)

            # Adam moments update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)

            # bias‑corrected estimates
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # learning‑rate schedule (decays like 1/√t)
            lr = lr0 / np.sqrt(t)

            # parameter update
            y = y - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # evaluate current a
            a = softmax(y)
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Direct minimisation of 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line‑search and keeps the iterate on the
    probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution and mask of the active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max (exact)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size (decreasing)
        step = 0.6 / np.sqrt(it + 1.0)

        # back‑tracking line‑search
        improved = False
        for _ in range(6):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    repeatedly move a small random fraction of mass from one coordinate to
    another, accepting only moves that strictly improve the score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # random fraction of the source mass
        delta = best[i] * rng.random() * 0.06
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score

    # Phase 0 – cheap parametric sweep to collect promising seeds
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # 1) Uniform
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_candidate(lin)

    # 3) Exponential families
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # 4) Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # 6) Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # 7) Random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Keep the best few seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin
    phase1_end = start_time + 0.70 * budget_s       # majority for Adam
    phase2_end = start_time + 0.95 * budget_s       # sub‑gradient
    # phase3 is the final polishing hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation with many restarts
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < phase1_end and restarts < 60:
        # Pick a good seed and add a tiny jitter for diversity
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 2e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        # Allocate a per‑restart budget (dynamic, but at most 25 s)
        remaining = phase1_end - time.time()
        per_restart_deadline = time.time() + min(25.0, remaining * 0.6)

        a_opt, score_opt = adam_optimize(a_start,
                                         per_restart_deadline,
                                         fft_len_conv,
                                         fft_len_g,
                                         n)

        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[Adam] new best score = {best_score:.6f} (restart {restarts})")
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.1)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑grad] score after sub‑gradient = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb polishing
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    # ------------------------------------------------------------------
    # Return the final sequence
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
