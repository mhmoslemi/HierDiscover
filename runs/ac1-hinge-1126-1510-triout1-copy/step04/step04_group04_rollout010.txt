"""
Propose a step‑function sequence with a low evaluation score.

The search mixes several complementary ingredients:

* **Low‑dimensional random starts** – many simple parametric shapes
  (exponential, power‑law, edge‑heavy, sinusoidal, “sqrt‑bell”) are
  turned into a probability vector via a soft‑max.
* **Smooth‑max L‑BFGS** – a smooth approximation of the peak of the
  (non‑linear) self‑convolution is minimised with a cascade of
  decreasing smoothing parameters.  This quickly finds a good basin.
* **Targeted hill‑climb** – after each L‑BFGS run we greedily move a
  random fraction of mass between two coordinates if the exact
  max‑convolution improves.
* **Linear‑programming refinement** – solve the LP  
      maximise  Σ b_j   s.t.   b≥0 ,   (a∗b)[k] ≤ M     
  where M is the current max‑convolution.  The optimiser b is scaled to
  a probability vector and mixed back into a.  Repeating this step
  aggressively pushes the peak down.
* **Projected sub‑gradient descent** – a final short sub‑gradient loop
  on the exact objective cleans up any remaining spikes.
* **Polishing hill‑climb** – a final greedy scan uses any remaining
  time.

All steps respect the non‑negativity and simplex constraints required
by ``evaluate_sequence``.  The routine runs for the full time budget
(≈ 1000 s) and returns the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (smooth‑max) and its gradient w.r.t. the soft‑max parameters y
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and gradient wrt y.

    The smooth max is
        max_c + lam * log( Σ exp((c_i - max_c)/lam) )
    where lam = lam_factor * max_c (+tiny epsilon).
    """
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient ----------
    w = exp_shift / Z               # d smooth_max / d conv_i
    # convolution of a with w (reversed) gives gradient wrt a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Simple hill‑climb using the exact max‑convolution
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep the move only when the exact max‑convolution improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.15  # up to 15 % of mass at i
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximising total mass of b)
# ----------------------------------------------------------------------


def lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    mix_factor: float = 0.2,
    max_passes: int = 6,
) -> np.ndarray:
    """
    Repeated LP refinement.

    For the current a we solve
        maximise   Σ b_j
        s.t.      b ≥ 0,
                  (a ∗ b)[k] ≤ M     (M = max (a∗a))
    The optimal b is normalised to sum = 1 and mixed back in.
    """
    rows = 2 * n - 1
    while time.time() < deadline and max_passes > 0:
        max_passes -= 1
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # Build convolution matrix A where (A·b)[k] = (a∗b)[k]
        # Each column j contributes a to rows j … j+n‑1
        A = sp.lil_matrix((rows, n))
        for col in range(n):
            A[col : col + n, col] = a
        A = A.tocsr()

        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(rows),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b_opt / sum_b
        # simple line‑search for the best mixing proportion
        best_alpha = 0.0
        best_M = M
        for alpha in np.linspace(0.0, 1.0, 11):
            a_new = (1.0 - alpha) * a + alpha * g
            m_new = np.convolve(a_new, a_new).max()
            if m_new < best_M:
                best_M = m_new
                best_alpha = alpha
        if best_alpha == 0.0:
            break
        a = (1.0 - best_alpha) * a + best_alpha * g
        print(f"[lp] reduced M to {best_M:.6f}")

    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact objective (max‑convolution)
# ----------------------------------------------------------------------


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the genuine max‑convolution.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        # gradient for the active sum index
        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]
        # diagonal terms appear only once
        diag = (i[mask] == j[mask])
        grad[i[mask][diag]] -= best[i[mask][diag]]

        # small jitter to escape flat plateaus
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score

    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sin":
        # sinusoidal envelope, shifted to be non‑negative
        t = np.linspace(0, np.pi, n)
        a0 = np.sin(t) + 0.1
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sqrt":
        # bell shape built from sqrt(i)*(sqrt(n‑i))
        i = np.arange(n)
        a0 = np.sqrt(i + 1) * np.sqrt(n - i)
        a0 += 1e-12
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # very sparse Sidon‑like support
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if conflict:
                continue
            chosen.append(candidate)
            for c in chosen:
                sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Problem size – try to reuse an existing construction if present
    # --------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 1000:
            n = max(n, int(tmp.size))
            best_a = np.clip(tmp, 0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Pre‑computations
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the L‑BFGS cascade
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 5e-5, num=30))
    maxiter_stage = 600            # more iterations per stage than before

    modes = ["random", "exp", "power", "edge", "sin", "sqrt", "sidon", "best"]
    iteration = 0

    # ------------------------------------------------------------------
    # Main stochastic optimisation loop (random restarts + L‑BFGS)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # L‑BFGS with gradually decreasing smoothing
        y_opt = y0.copy()
        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=y_opt,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_stage, "gtol": 1e-12, "disp": False},
            )
            y_opt = res.x

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # local greedy hill‑climb (tiny time slice)
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=4000
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Post‑processing refinements
    # ------------------------------------------------------------------

    # 1. Sub‑gradient descent on the exact objective
    sub_deadline = min(deadline, time.time() + 30.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.5
    )
    best_y = np.log(best_a + 1e-12)

    # 2. Linear‑programming refinement (only for moderate n to keep it cheap)
    if n <= 3000 and time.time() + 10.0 < deadline:
        lp_deadline = deadline - 5.0
        best_a = lp_refine(best_a, n, lp_deadline, mix_factor=0.2, max_passes=8)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)
        print(f"[lp] post‑refine fast score = {best_score:.6f}")

    # 3. Final fine L‑BFGS pass with a very small smoothing factor
    fine_lam_schedule = list(np.geomspace(0.3, 2e-5, num=25))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.3g} fast score = {best_score:.6f}")

    # 4. Final aggressive hill‑climb with any remaining time
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Return the sequence (list of floats) for the external evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # compute the real evaluation (the harness will also do it, but we
    # print it for our own log)
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)

    return final_seq
