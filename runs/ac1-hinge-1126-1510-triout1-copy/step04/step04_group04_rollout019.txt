def adam_optimize(y, n, fft_len_conv, fft_len_g, lam, lr=0.05, max_iter=1000):
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps=1e-8
    t = 0
    for it in range(max_iter):
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
        t += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Optimisation of a non‑negative step‑function sequence for a custom evaluation
metric.

Approach
--------
1️⃣ Initialise a probability vector (sum = 1).  If a previous construction
   (`height_sequence_1`) exists it is reused, otherwise a uniform vector is
   taken.

2️⃣ Re‑parameterise the vector as a soft‑max of unconstrained variables
   `y`.  Optimisation is performed on `y` to avoid explicit simplex handling.

3️⃣ A smooth approximation of the max‑convolution is minimised with Adam.
   A geometric schedule of smoothing factors (`lam_factor`) drives the
   optimisation from a very smooth landscape (large `lam_factor`) to the
   true L∞‑objective (tiny `lam_factor`).  Several random restarts are used.

4️⃣ The best soft‑max parameters are expanded to a full length‑`n` vector and
   refined with a fast greedy hill‑climb that moves a small fraction of mass
   from indices contributing most to the maximal convolution entry to indices
   contributing least.  Random moves are mixed in to escape plateaux.

5️⃣ Finally the sequence is returned.  The evaluation harness will call
   ``evaluate_sequence`` on the returned list.

The whole routine respects a user‑specified time budget (default 1000 s) and
returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (used by Adam)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of 2·n·max(a*a) and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained parameters (soft‑max will be applied).
    n : int
        Length of the step function.
    fft_len_conv, fft_len_g : int
        FFT lengths for the forward and gradient convolutions.
    lam_factor : float
        Smoothing factor – the actual λ used is lam = max(conv) * lam_factor.

    Returns
    -------
    f : float
        Smoothed objective value.
    grad_y : np.ndarray
        Gradient w.r.t. ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a (in frequency domain) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12               # smoothing magnitude
    shifted = conv - max_c                         # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of the smooth‑max w.r.t. the convolution entries ----
    w = exp_shift / Z                               # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    lr: float = 0.05,
    max_iter: int = 200,
) -> np.ndarray:
    """
    Run Adam on ``objective_and_grad`` for a fixed smoothing factor.

    Parameters
    ----------
    y_start : np.ndarray
        Initial parameters.
    lam_factor : float
        Smoothing factor for the current stage.
    lr : float
        Learning rate.
    max_iter : int
        Number of Adam updates.

    Returns
    -------
    np.ndarray
        Optimised parameters (still in the pre‑soft‑max space).
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for _ in range(max_iter):
        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ----------------------------------------------------------------------
# Greedy hill‑climb (directed + random moves)
# ----------------------------------------------------------------------


def random_hill_step(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    rng: np.random.Generator,
    max_delta_frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass between two random positions."""
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, cur_score
    delta = a[i] * rng.random() * max_delta_frac
    if delta <= 0.0:
        return a, cur_score

    a[i] -= delta
    a[j] += delta
    new_score = fast_score(a, fft_len, n)
    if new_score < cur_score:
        return a, new_score
    # revert
    a[i] += delta
    a[j] -= delta
    return a, cur_score


def directed_hill_step(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    rng: np.random.Generator,
    max_delta_frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Move mass from an index that contributes most to the current maximal
    convolution entry to an index that contributes the least.
    """
    # compute full convolution once
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    s = int(conv.argmax())

    idx = np.arange(n)
    mask = (s - idx >= 0) & (s - idx < n)
    if not mask.any():
        return a, cur_score

    # contributions a_{s-i}
    contrib = np.zeros(n)
    contrib[mask] = a[s - idx[mask]]

    # candidates for removal: need a[i] > 0
    cand_i = idx[mask & (a > 1e-12)]
    if cand_i.size == 0:
        return a, cur_score
    i_max = cand_i[np.argmax(contrib[cand_i])]

    # candidates for addition (including also indices with zero contribution)
    cand_j = idx[mask]
    # avoid moving to the same place
    cand_j = cand_j[cand_j != i_max]
    if cand_j.size == 0:
        return a, cur_score
    j_min = cand_j[np.argmin(contrib[cand_j])]

    delta = a[i_max] * max_delta_frac
    if delta <= 0.0:
        return a, cur_score

    a[i_max] -= delta
    a[j_min] += delta

    new_score = fast_score(a, fft_len, n)
    if new_score < cur_score:
        return a, new_score
    # revert
    a[i_max] += delta
    a[j_min] -= delta
    return a, cur_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_no_improve: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Mixed hill‑climb: alternating directed and random steps until the time
    limit is hit or a large number of consecutive non‑improving attempts
    occurs.
    """
    rng = np.random.default_rng()
    no_improve = 0
    while time.time() < deadline:
        if rng.random() < 0.6:
            a, new_score = directed_hill_step(a, cur_score, fft_len, n, rng)
        else:
            a, new_score = random_hill_step(a, cur_score, fft_len, n, rng)
        if new_score < cur_score:
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= max_no_improve:
            break
    return a, cur_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # problem size and initial vector
    # --------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1:
            best_a = np.clip(tmp, 0, None)
            total = best_a.sum()
            if total < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / total
                if best_a.size != n:
                    # reshape / pad to required size
                    if best_a.size < n:
                        pad = np.full(n - best_a.size, 0.0)
                        best_a = np.concatenate([best_a, pad])
                    else:
                        best_a = best_a[:n]
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # FFT lengths
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Adam optimisation with a decreasing smoothing schedule
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.4, 5e-6, num=28))
    restarts = 8
    iters_per_lam = 180           # ~180 Adam updates per lam value
    for rs in range(restarts):
        if time.time() >= deadline:
            break
        # random perturbation of the starting point
        y0 = best_y + rng.normal(scale=0.02, size=n)
        y_opt = y0.copy()
        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            y_opt = adam_optimize(
                y_opt,
                n,
                fft_len_conv,
                fft_len_g,
                lam_factor=lam,
                lr=0.07,
                max_iter=iters_per_lam,
            )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[restart {rs}] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Extensive hill‑climb (directed + random) on the best vector
    # --------------------------------------------------------------
    hc_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        hc_deadline,
        max_no_improve=8000,
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
