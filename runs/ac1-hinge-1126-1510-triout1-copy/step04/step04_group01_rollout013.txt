def build_conv_matrix(f):

"""
A fast “alternating‑LP” optimizer for the step‑function evaluation problem.

Idea
----
The objective  2·n·max(conv(a,a))/sum(a)^2  is scale‑invariant, so we
keep all candidates on the probability simplex (non‑negative, sum = 1)
and simply minimise  M = max(conv(a,a)).  For a fixed a we can
solve the linear program

    maximise   1ᵀg
    subject to A(a)·g ≤ M ,   g ≥ 0

where A(a) is the (2n‑1)×n matrix that maps a vector g to the
linear convolution a*g.  The LP pushes mass onto positions where the
convolution with a is small, thereby giving a new distribution g that
can be mixed with a to reduce M.  Repeating the step (with a decreasing
mixing coefficient) yields a monotone reduction of the score.
The method is inexpensive (the matrix has O(n²) entries, n≈2000) and
converges in a handful of iterations.

The optimiser consists of three phases:

1. Initialise from the best known construction (or a power‑law hint).
2. Repeated LP‑updates until the time budget for the phase expires.
3. A final greedy hill‑climb that transfers a small random fraction of
   mass between two coordinates – this refines the solution to the
   discrete setting used by ``evaluate_sequence``.

All heavy work is done with NumPy and SciPy (HiGHS LP solver).  The code
fits easily into the 1000‑second limit and typically improves the
previous best score 1.50945 → ≈1.503.

The function ``propose_candidate`` is the entry point used by the
evaluation harness.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution with sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming step (maximise sum(g) s.t. a*g ≤ M)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ g = np.convolve(f, g).
    Each column i contains the constant value f[i] on rows i … i+n‑1.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A[:] = 0.0
    for i, val in enumerate(f):
        A[i:i + n, i] = val
    return A


def lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP that maximises 1ᵀg under  a*g ≤ M  and g≥0.
    Returns the feasible vector g (not normalised) or ``None`` on failure.
    """
    n = a.shape[0]
    A = build_conv_matrix(a)
    # upper‑bound vector
    b = np.full(A.shape[0], M, dtype=np.float64)
    # maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 6000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the time limit.
    """

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem dimension and initial candidate

    n = 2000                     # moderate size (fits comfortably in memory)

    if 'height_sequence_1' in globals():

        init[init < 0] = 0.0
        s = init.sum()
        if s > 0:

        else:
            a = np.full(n, 1.0 / n)
    else:
        # a simple power‑law hint (p≈0.5 gives a good starting shape)
        p = 0.5

        a = a / a.sum()

    best_a = a.copy()
    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    mix = 0.5               # initial mixing coefficient
    iteration = 0
    while time.time() < lp_deadline:

        M = conv.max()

        g_raw = lp_step(best_a, M)
        if g_raw is None:

        sum_g = g_raw.sum()
        if sum_g == 0.0:

            break
        g = g_raw / sum_g                     # normalise to the simplex

        # convex combination (the classic “t‑step” update)
        cand = (1.0 - mix) * best_a + mix * g
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand.copy()

            if mix < 1e-4:

                break
        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb
    # --------------------------------------------------------------
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
