   best_score = 2*n * b.max()
   best_a = a.copy()
   rng = np.random.default_rng()
   while time.time() < deadline and max_iter > 0:

        hi = min(k, n-1)

"""
Propose a low‑score step‑function sequence.

This implementation builds on the previous best construction (if available)
and then performs a fast “max‑convolution descent”.  At each iteration we
identify the index k where the auto‑convolution a*a attains its maximum,
compute the exact sub‑gradient of that max (which is 2·a_{k‑j} for each
coordinate j), and move a small amount of mass from the most harmful
coordinate to the least harmful one.  The convolution is updated
incrementally with O(n) vector operations, so millions of moves can be
tried within the 1000 s budget.  Occasionally a random move is tried to
escape plateaus.  The best vector seen is returned and finally evaluated
by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def initial_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT.  Returns the 2·n‑1 length array.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return conv


def incremental_update(
    a_old: np.ndarray,
    b: np.ndarray,
    i: int,
    j: int,
    delta: float,
) -> None:
    """
    Apply the effect of moving ``delta`` mass from index ``i`` to ``j``.
    ``a_old`` is the array *before* the move (required for the linear part).
    The convolution vector ``b`` (length 2·n‑1) is modified in‑place.
    """
    n = a_old.shape[0]
    # linear part  – 2·δ·a_{k‑i}  and  +2·δ·a_{k‑j}
    b[i : i + n] -= 2.0 * delta * a_old
    b[j : j + n] += 2.0 * delta * a_old
    # quadratic correction
    b[2 * i] += delta * delta
    b[2 * j] += delta * delta
    b[i + j] -= 2.0 * delta * delta


# ----------------------------------------------------------------------
# Main entry point for the evaluator harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Load a good starting point (previous best if present)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        a = np.asarray(prev, dtype=float)
        a = np.maximum(a, 0.0)          # safety
    else:
        # a uniform start of size 5000
        a = np.full(5000, 1.0 / 5000.0)

    # Normalise to a probability vector (sum = 1)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(a.shape[0], 1.0 / a.shape[0])
    else:
        a /= a_sum

    n = a.shape[0]
    # --------------------------------------------------------------
    # 2️⃣  Compute the initial full convolution
    # --------------------------------------------------------------
    b = initial_convolution(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    print(f"[init] n={n} start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Max‑convolution descent (many cheap iterations)
    # --------------------------------------------------------------
    max_move_frac = 0.25            # start with up‑to‑25 % of the source mass
    prob_random = 0.12              # probability of performing a pure random move
    iter_cnt = 0
    last_log = time.time()

    while time.time() < deadline:
        iter_cnt += 1

        # periodic logging (every ~5 s)
        if time.time() - last_log > 5.0:
            print(f"[{iter_cnt}] best score = {best_score:.6f}")
            last_log = time.time()

        # ---- decide which move to try ---------------------------------
        if rng.random() < prob_random:
            # pure random transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
        else:
            # deterministic move guided by the current maximum of the convolution
            k_max = int(b.argmax())
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            idxs = np.arange(lo, hi + 1, dtype=int)          # all j that affect b[k_max]
            # sub‑gradient g_j = 2·a_{k_max‑j}
            grad_vals = 2.0 * a[k_max - idxs]
            # most harmful (largest gradient) and most innocent (smallest)
            i = idxs[int(np.argmax(grad_vals))]
            j = idxs[int(np.argmin(grad_vals))]
            if i == j or np.isclose(grad_vals.max(), grad_vals.min()):
                # fall back to a tiny random perturbation
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue

        # ---- how much mass to move? -----------------------------------
        if a[i] <= 0.0:
            continue
        # move a random fraction (≤ max_move_frac) of the source mass
        frac = rng.random() * max_move_frac
        delta = a[i] * frac
        if delta <= 1e-15:
            continue

        # ---- snapshot current state (cheap copies) --------------------
        a_old = a.copy()
        b_old = b.copy()

        # ---- apply the move -------------------------------------------
        a[i] -= delta
        a[j] += delta
        incremental_update(a_old, b, i, j, delta)

        # ---- evaluate improvement --------------------------------------
        new_score = 2.0 * n * b.max()
        if new_score < best_score - 1e-13:
            # accept
            best_score = new_score
            best_a = a.copy()
            # gradually shrink the step size for fine‑grained tuning
            max_move_frac = max(0.01, max_move_frac * 0.9995)
        else:
            # reject – revert to the previous state
            a[:] = a_old
            b[:] = b_old

    # --------------------------------------------------------------
    # 4️⃣  Final polishing with a tiny random hill‑climb (optional)
    # --------------------------------------------------------------
    # If a little time remains, perform a few classic hill‑climb steps
    # using the fast FFT based score to possibly squeeze a bit more.
    if time.time() < deadline:
        def fast_score(a_vec):
            return float(2.0 * n * np.max(np.real(np.fft.ifft(
                np.fft.fft(a_vec, next_pow2(2 * n - 1)) ** 2))[:2 * n - 1]))

        # tiny local search
        for _ in range(20000):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = best_a[i] * rng.random() * 0.08
            if delta <= 0.0:
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = fast_score(cand)
            if cand_score < best_score - 1e-13:
                best_a = cand
                best_score = cand_score

    # --------------------------------------------------------------
    # 5️⃣  Return the best sequence found (evaluate with the official function)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
