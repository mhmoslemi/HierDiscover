"""
Improved multi‑phase optimizer for the max‑convolution bound.

The search proceeds in four stages:

1️⃣  Random restarts of a smooth‑max L‑BFGS optimisation (many
    iterations, a dense smoothing schedule, aggressive hill‑climbs).
2️⃣  A long exact‑max sub‑gradient descent (projected onto the simplex)
    that directly reduces the true convolution peak.
3️⃣  Optional linear‑programming refinement for moderate problem sizes
    (n ≤ 3000) – it maximises the total mass under the current peak bound.
4️⃣  A final fine‑grained L‑BFGS run with a very small smoothing factor,
    followed by a long hill‑climb polish.

All steps respect the non‑negative, sum‑positive constraints required by
``evaluate_sequence``.  The routine adapts to the remaining time budget and
returns the best sequence found before the deadline.
"""

import time
from typing import List

import numpy as np
from scipy import sparse as sp
from scipy.optimize import linprog, minimize


# ----------------------------------------------------------------------
# Helper utilities – pure functions, no closures or lambdas
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    The smooth max uses a temperature ``lam``.
    """
    a = softmax(y)

    # ---- linear convolution a*a (full length) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # smooth approximation of the max
    shifted = conv - max_c  # max now 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max (w.r.t. the convolution vector) ----
    w = exp_shift / Z  # dw/dconv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # back‑propagate through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given initialization mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy construction of a (rough) Sidon set
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = any(candidate + c in sums for c in chosen) or (candidate * 2 in sums)
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    The move is kept only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.12  # up to 12 % of the source mass
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: List[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """Run a cascade of L‑BFGS optimisations with decreasing smoothing."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution.
    After each step the vector is projected onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        k_star = int(np.argmax(conv))

        i_idx = np.arange(n)
        j_idx = k_star - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        grad = np.zeros_like(best)
        grad[i_idx[mask]] = 2.0 * best[j_idx[mask]]

        # tiny jitter to avoid stagnation
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.3, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    mix_factor: float = 0.15,
    max_iter: int = 7,
) -> np.ndarray:
    """
    Linear‑programming refinement.
    For the current a we solve
        maximise   Σ b_j
        s.t.      b ≥ 0,
                  (a * b)[s] ≤ M   (M = max (a*a))
    The optimal b is scaled to unit sum and mixed back into a.
    """
    rows = 2 * n - 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # Build convolution matrix A where (A·b)[s] = (a*b)[s]
        A = sp.lil_matrix((rows, n))
        for col in range(n):
            A[col : col + n, col] = a
        A = A.tocsr()

        # Solve max Σ b   ⇔   min (−Σ b)
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(rows),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b_opt / sum_b
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        new_M = np.convolve(a_new, a_new).max()
        if new_M < M:
            a = a_new
            print(f"[lp] reduced M from {M:.6f} to {new_M:.6f}")
        else:
            break
    return a
