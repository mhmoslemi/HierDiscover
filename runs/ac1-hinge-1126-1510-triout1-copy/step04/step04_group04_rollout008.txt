"""
Propose a step‑function sequence with a low evaluation score.

Strategy
--------
1.  Initialise a probability vector of length n (default 5000).
    * Uniform distribution (baseline).
    * Simple symmetric “inverse‑power” families a_i ∝ (i+1)^(-p)+(n-i)^(-p)
      with an extra mixing with the uniform vector.
    The best of these cheap candidates is kept.

2.  Refine the best candidate with a series of **smooth‑max L‑BFGS**
    optimisations (softmax parametrisation).  Several random restarts
    are performed while the time budget permits.

3.  Perform aggressive **hill‑climbing** – move a fraction of the mass
    from indices that contribute most to the current peak towards
    indices that contribute little.  The move is kept if the fast
    (FFT‑based) score improves.

4.  Finish with an **exact sub‑gradient descent** on the true max‑convolution
    (projected onto the probability simplex).  The step‑size is
    adapted dynamically.

5.  Return the final sequence (list of non‑negative floats).  The
    sequence is normalised to unit sum; the evaluation function is
    scale‑invariant, so this does not affect the score.

All heavy steps are guarded by a deadline, ensuring the function
returns before the 1000 s time limit.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. y.
    The parameter `lam_factor` controls the smoothing strength:
        lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z               # d smooth_max / d conv_i
    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb (greedy local search)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially moves mass away from
    indices contributing most to the current peak.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # pre‑allocate arrays for speed
    idx = np.arange(n)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # compute the current peak position and a cheap “contribution”
        # measure for each index (how much it participates in the max sum)
        conv = conv_fft(best, fft_len, n)
        s_star = int(conv.argmax())

        # contribution = a_i * a_{s_star-i}
        j = s_star - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[j[mask]]

        # pick a source index weighted by its contribution
        total = contrib.sum()
        if total == 0.0:
            # fallback to random pick
            i = rng.integers(0, n)
        else:
            prob = contrib / total
            i = rng.choice(n, p=prob)

        # pick a destination uniformly (can be the same – then we skip)
        j_dest = rng.integers(0, n)
        if i == j_dest:
            continue

        # move a fraction of the source mass
        delta = best[i] * rng.random() * 0.15  # up to 15 % of the source mass
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j_dest] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j_dest] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    After each step the vector is projected back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    idx = np.arange(n)

    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(conv.argmax())

        # gradient of the max‑convolution (exact)
        j = s_star - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[mask] = 2.0 * best[j[mask]]

        # small jitter to escape flat regions
        grad += rng.normal(scale=1e-5, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.3, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "power":
        p = rng.uniform(0.2, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # modest edge weighting (similar to the original code)
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon‑set construction (few points, just for diversity)
        chosen = []
        sums = set()
        for cand in range(n):
            good = True
            for c in chosen:
                if (cand + c) in sums or (cand * 2) in sums:
                    good = False
                    break
            if good:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 5000
    # If a previous solution is present, reuse its length (it may be larger)
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1000:
            n = max(n, int(tmp.size))
            best_a = tmp.copy()
            best_a = np.clip(best_a, 0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # FFT lengths
    # ------------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) cheap parametric grid search (inverse‑power family)
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.05, 2.0, 20)          # exponent
    mix_vals = np.linspace(0.0, 0.85, 9)         # how much of the edge‑shape to mix in
    i = np.arange(n, dtype=float)

    for p in p_vals:
        # symmetric inverse‑power shape
        edge = ( (i + 1) ** (-p) + (n - i) ** (-p) )
        edge /= edge.sum()
        for mix in mix_vals:
            candidate = (1.0 - mix) * (np.full(n, 1.0 / n)) + mix * edge
            candidate = np.clip(candidate, 0, None)
            candidate /= candidate.sum()
            s = fast_score(candidate, fft_len, n)
            if s < best_score:
                best_a = candidate
                best_score = s
                best_y = np.log(best_a + 1e-12)
                print(f"[grid] p={p:.3f} mix={mix:.2f} improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) L‑BFGS refinement with several random restarts
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 0.001, num=15))
    maxiter_per_stage = 250
    modes = ["random", "power", "edge", "sidon", "grid"]

    while time.time() < deadline - 120.0:   # keep ~2 min for later phases
        mode = rng.choice(modes)
        if mode == "grid":
            # start from the current best but add a small perturbation
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0, n, fft_len, fft_len, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[L‑BFGS] mode={mode} improved fast score = {best_score:.6f}")

        # short aggressive hill‑climb to capture any immediate improvement
        hc_deadline = min(deadline - 60.0, time.time() + 1.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len, n, hc_deadline, max_iter=3000
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # 3) Extended hill‑climb (targeted moves)
    # ------------------------------------------------------------------
    hc_deadline = min(deadline - 30.0, time.time() + 15.0)
    best_a, best_score = hill_climb(
        best_a, best_score, fft_len, n, hc_deadline, max_iter=20_000
    )
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient descent (as much time as remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 5.0
        best_a, best_score = subgradient_optimize(
            best_a, n, fft_len, sub_deadline, init_step=0.5
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # 5) Final polishing: a last round of hill‑climb (very short)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len, n, deadline, max_iter=5_000
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Prepare final sequence, enforce non‑negativity and renormalise.
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_a, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.full_like(final_seq, 1.0 / n)
    else:
        final_seq = final_seq / total

    # ------------------------------------------------------------------
    # Official evaluation (the harness supplies evaluate_sequence)
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(final_seq.tolist())  # type: ignore
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq.tolist()
