"""
Gaussian‐based construction with local hill‑climb refinement.

The objective is 2·n·max(a*a)/∑a² (the denominator is 1 after L1‑normalisation).
A symmetric “truncated Gaussian’’ shape makes the autocorrelation almost flat,
drastically reducing the peak.  We scan a range of standard deviations σ,
pick the best Gaussian, then perform a cheap random mass‑transfer hill‑climb
while respecting the simplex constraints.  The final sequence is returned as
a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the score for a probability vector (∑a = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    deadline: float,
    max_move_frac: float = 0.20,
    max_iter: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    n = best.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        # transfer a random fraction of the source mass
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len)
        if new_score < best_score - 1e-12:  # strict improvement
            best_score = new_score
            # keep move
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    The algorithm:
      1. Choose a length n (5000 by default or the length of a previously
         found sequence if available).
      2. Scan a set of Gaussian standard deviations σ and keep the one
         giving the smallest fast_score.
      3. Refine the candidate with a random mass‑transfer hill‑climb while
         staying within the remaining time budget.
      4. Return the final sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 500:
            n = int(arr.size)
        else:
            n = 5000
    else:
        n = 5000

    # ------------------------------------------------------------------
    # Search over Gaussian shapes
    # ------------------------------------------------------------------
    mu = (n - 1) / 2.0
    fft_len = next_pow2(2 * n - 1)

    # range of σ to explore – from 0.25·n up to 0.5·n (the latter gives the
    # flattest admissible Gaussian on the interval)
    sigma_vals = np.linspace(0.25 * n, 0.5 * n, 31)

    best_a = np.full(n, 1.0 / n)  # fallback
    best_score = fast_score(best_a, fft_len)
    best_sigma = None

    for sigma in sigma_vals:
        # Gaussian values (truncated by the interval [0,n-1])
        diffs = np.arange(n) - mu
        a = np.exp(-(diffs * diffs) / (2.0 * sigma * sigma))
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a
            best_sigma = sigma

    print(f"[gauss] best sigma = {best_sigma:.2f}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional fine‑grained refinement of σ (tiny local search)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        # simple ternary‑search‑like refinement around the best σ
        lo = max(best_sigma * 0.8, 0.20 * n)
        hi = min(best_sigma * 1.2, 0.5 * n)
        for _ in range(5):
            s1 = lo + (hi - lo) / 3.0
            s2 = hi - (hi - lo) / 3.0
            a1 = np.exp(-((np.arange(n) - mu) ** 2) / (2.0 * s1 * s1))
            a1 /= a1.sum()
            a2 = np.exp(-((np.arange(n) - mu) ** 2) / (2.0 * s2 * s2))
            a2 /= a2.sum()
            score1 = fast_score(a1, fft_len)
            score2 = fast_score(a2, fft_len)
            if score1 < score2:
                hi = s2
                if score1 < best_score:
                    best_score, best_a, best_sigma = score1, a1, s1
            else:
                lo = s1
                if score2 < best_score:
                    best_score, best_a, best_sigma = score2, a2, s2
        print(f"[refine] refined sigma = {best_sigma:.2f}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive hill‑climb (random mass transfers) – final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            deadline=deadline,
            max_move_frac=0.20,
            max_iter=30_000,
        )
        print(f"[hill] after hill‑climb fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
