"""
Optimisation of a step‑function to minimise
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective is scale‑invariant; we therefore work with probability vectors
(sum(a)=1).  The algorithm proceeds in three phases:

1.  Random “shape” initialisation.
    Several families (power‑law, exponential, two‑sided power‑law, beta,
    triangular, sine and a uniform baseline) are sampled with various
    lengths.  The best few candidates according to a fast FFT‑based
    score are retained.

2.  Smoothed‑max Adam optimisation.
    The true objective  max(conv) is approximated by the smooth
    log‑sum‑exp:
        f(a) = (1/α)·log Σ_k exp(α·c_k) ,   c = a * a .
    Its gradient can be written in closed form using FFTs:
        ∇_a f = (2 / Σ w_k)·correlation(w, a)
    where w_k = exp(α·(c_k - max(c))).  An Adam optimiser on the
    soft‑max parametrisation (logits → probability simplex) minimises f.
    The smoothing parameter α is increased from a small value to a larger
    one during the run, allowing smooth early updates and a sharp final
    focus on the true maximum.

3.  Final refinement and verification.
    The best sequence found is optionally refined for a few seconds,
    then passed to the supplied ``evaluate_sequence`` function and
    returned as a plain Python list.

All computation respects the 1000 s time budget and uses only NumPy.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a probability vector (sum≈1)."""
    n = seq.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta_: float) -> np.ndarray:
    """Discretised Beta‑pdf: (i+1)^(α-1)·(n-i)^(β-1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta_ - 1.0))


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangular roof."""
    i = np.arange(n, dtype=np.float64)
    return np.minimum(i + 1, n - i)


def generate_sine(n: int) -> np.ndarray:
    """Half‑period sine, positive on (0,π)."""
    i = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (i + 0.5) / n)


def generate_uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def generate_random_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet (α=1) – uniform on the simplex."""
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (softmax parametrisation)
# ----------------------------------------------------------------------


def adam_optimize_smooth(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.9,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 0.2,
    alpha_end: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max parametrisation to minimise the smoothed max
    of the self‑convolution.

    Parameters
    ----------
    a0 : np.ndarray
        Initial probability vector (non‑negative, not necessarily normalised).
    time_limit : float
        Seconds allocated for this run.
    start_time : float
        ``time.time()`` when the optimisation started (used to enforce the limit).
    lr0, beta1, beta2, eps : Adam hyper‑parameters.
    alpha_start, alpha_end : smoothing schedule for the log‑sum‑exp.

    Returns
    -------
    best_seq : np.ndarray
        The best probability vector encountered.
    best_val : float
        Corresponding evaluation score (2·n·max(conv)).
    """
    n = a0.shape[0]
    # initialise logits (log‑softmax variables)
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # bookkeeping
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    fft_len = next_pow2(2 * n)

    while True:
        now = time.time()
        if now > start_time + time_limit:
            break

        it += 1
        # ---------- soft‑max ----------
        y_shift = y - np.max(y)               # stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()                # lies on simplex, sum≈1

        # ---------- convolution ----------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())

        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---------- smoothing factor ----------
        elapsed = now - start_time
        total = time_limit
        # increase α exponentially (smooth → sharp)
        if total > 0:
            ratio = min(1.0, elapsed / total)
        else:
            ratio = 1.0
        alpha = alpha_start * (alpha_end / alpha_start) ** ratio

        # ---------- gradient of smooth max ----------
        # w_k = exp(alpha * (c_k - M))  (max at 1)
        w = np.exp(alpha * (conv - M))
        sum_w = w.sum()
        if sum_w == 0.0:
            sum_w = 1.0
        # FFT of the weight vector
        fw = np.fft.rfft(w, n=fft_len)
        # grad_raw = 2 * correlation(w, a)   (via IFFT of conj(w_fft) * a_fft)
        grad_raw = np.fft.irfft(np.conj(fw) * fa, n=fft_len)[:n]
        grad_a = (2.0 * grad_raw) / sum_w       # normalized gradient

        # ---------- gradient w.r.t. logits ----------
        ag = np.dot(a, grad_a)                  # scalar ⟨a, grad⟩
        grad_y = a * (grad_a - ag)              # Jacobian of softmax

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid stagnation
        if it % 3000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    # Base deterministic candidates

    # aim for ~200 random shapes (fast)
    while len(init_candidates) < 250 and time.time() < deadline:
        n = random.choice(base_lengths)
        fam = random.choice(families)

        if fam == "power":
            p = random.uniform(0.05, 2.0)
            seq = generate_power_law(n, p)
        elif fam == "exp":
            lam = random.uniform(1e-5, 5e-2)
            seq = generate_exponential(n, lam)
        elif fam == "twosided":
            pl = random.uniform(0.05, 2.0)
            pr = random.uniform(0.05, 2.0)
            wl = random.uniform(0.2, 3.0)
            wr = random.uniform(0.2, 3.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        elif fam == "beta":
            a = random.uniform(0.05, 3.0)
            b = random.uniform(0.05, 3.0)
            seq = generate_beta_shape(n, a, b)
        else:  # random Dirichlet
            seq = generate_random_dirichlet(n)

        # normalise to a probability vector
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        init_candidates.append(seq)

    # --------------------------------------------------------------
    # 2) Keep the most promising seeds (based on fast_score)
    # --------------------------------------------------------------
    # evaluate all candidates quickly
    scored = []
    for idx, cand in enumerate(init_candidates):
        val = fast_score(cand)
        scored.append((val, cand))
    scored.sort(key=lambda x: x[0])
    top_k = 5                # number of seeds we will run a full optimisation on
    top_candidates = [cand for _, cand in scored[:top_k]]

    # --------------------------------------------------------------
    # 3) Optimise each top candidate (time split)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left after candidate generation")

    # triangular time allocation: earlier candidates get more budget
    weights = np.arange(top_k, 0, -1, dtype=float)
    weights /= weights.sum()
    allocated = weights * remaining

    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, (cand, t_alloc) in enumerate(zip(top_candidates, allocated)):
        if time.time() > deadline:
            break
        print(f"[optim] start candidate #{idx+1}/{top_k} (len={cand.size}) — time {t_alloc:.1f}s")
        seq, val = adam_optimize_smooth(cand, t_alloc, time.time())
        print(f"[optim] candidate #{idx+1} best score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # --------------------------------------------------------------
    # 4) Optional fine‑tuning of the global best (if time permits)
    # --------------------------------------------------------------
    if best_overall_seq is not None and (deadline - time.time()) > 5.0:
        extra = (deadline - time.time()) * 0.9
        print(f"[fine‑tune] refining best sequence for {extra:.1f}s")
        seq, val = adam_optimize_smooth(best_overall_seq, extra, time.time())
        if val < best_overall_val:

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")
    return final_list
