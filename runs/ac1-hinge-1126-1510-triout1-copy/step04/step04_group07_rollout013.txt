def generate_power_law(n,p): return (np.arange(1,n+1, dtype=np.float64) ** (-p))

def generate_exponential(n, lam): return np.exp(-lam * np.arange(n, dtype=np.float64))

def generate_two_sided_power_law(n,p_left, p_right, w_left=1.0, w_right=1.0):
    i = np.arange(1,n+1, dtype=np.float64)
    left = w_left * i**(-p_left)
    right = w_right * (n - i + 1)**(-p_right)
    return left+right

def generate_beta_shape(n, alpha, beta): i = np.arange(1,n+1); return (i**(alpha-1))*((n - i + 1)**(beta-1))

def generate_mix_power_beta(n,p,alpha,beta,w):
    return w * generate_power_law(n,p) + (1-w) * generate_beta_shape(n,alpha,beta)

def propose_candidate(seed=42, budget_s=1000):

    start = time.time()

    top_k = 5
    top_candidates = []  # sorted by low score

            seq = np.maximum(prev, 0.0)
            s = seq.sum()
            if s < 1e-12:
                # fallback uniform
                n = len(seq)
                seq = np.full(n, 1.0/n)
            else:
                seq = seq / s
            val = fast_score(seq)
            # store as custom candidate
            top_candidates.append({"family":"custom","n":seq.size,"shape_func":None,"params":None,
                                   "bounds_lo":None,"bounds_hi":None,"seq":seq,"score":val})
    except NameError:
        pass

    # random search loop
    families = ["power","exp","twosided","beta","mix_power_beta"]
    length_choices = [5000,8000,10000,12000]

    while time.time() < deadline - 30:   # keep at least 30s for refinement
        fam = random.choice(families)
        n = random.choice(length_choices)

        # generate random candidate
        if fam == "power":
            p = random.uniform(0.05, 2.0)
            seq = generate_power_law(n, p)
            params = np.array([p])
            lo = np.array([0.05])
            hi = np.array([2.0])

        elif fam == "exp":

            params = np.array([pl, pr, wl, wr])
            lo = np.array([0.05, 0.05, 0.2, 0.2])
            hi = np.array([2.0, 2.0, 5.0, 5.0])
            shape_func = lambda n, pl, pr, wl, wr: generate_two_sided_power_law(n, pl, pr, wl, wr)

"""
Hybrid search for a low‑score step function.

The search consists of three stages:

1. **Random sampling** – generate many candidates from a handful of
   parametric families (power‑law, exponential, two‑sided power‑law,
   Beta‑pdf and a simple mixture).  The best K candidates are kept.

2. **Local refinement** – each of the K survivors is improved by a
   simulated‑annealing style multiplicative perturbation of its
   parameters (the step size decays linearly).  The convolution bound
   (the objective) is evaluated with a fast FFT based routine.

3. **Fine‑tuning** – the current overall best vector is handed to a
   short Adam optimisation on the full log‑softmax parametrisation,
   again using the exact sub‑gradient of the max‑convolution.

The whole routine respects the 1000 s budget and returns the best
sequence found as a plain Python list.  All numbers are non‑negative,
the sequence is normalised to sum = 1 (the objective is scale‑invariant)
and the final value is verified with the supplied ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Imports and utilities
# ----------------------------------------------------------------------
import time
import random
import numpy as np

# --------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) where ``seq`` is assumed to have
    a unit L1‑norm (the ratio required by ``evaluate_sequence``).
    Computed via an FFT for speed.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # Normalise – the objective is scale invariant, but keep the
    # convention that the sum equals 1.
    seq = seq / s
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Shape generators (parametric families)
# ----------------------------------------------------------------------
def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^{alpha-1}·(n-i)^{beta-1}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_mix_power_beta(
    n: int, p: float, alpha: float, beta: float, w: float
) -> np.ndarray:
    """
    Simple convex combination of a power‑law and a Beta shape.
    """
    return w * generate_power_law(n, p) + (1.0 - w) * generate_beta_shape(
        n, alpha, beta
    )


# ----------------------------------------------------------------------
# Adam optimiser on the full vector (soft‑max parametrisation)
# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on the log‑soft‑max variables of ``a0``.
    Returns the best sequence seen and its exact (fast) score.
    """
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)                # logits

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ---------- soft‑max ----------
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # ---------- convolution ----------
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---------- sub‑gradient ----------
        eps_act = 1e-12
        active = np.where(conv >= M - eps_act)[0]
        g = np.zeros_like(a)

        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if it % 2000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

# ----------------------------------------------------------------------
