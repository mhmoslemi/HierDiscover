"""
A multi‑strategy optimizer for the step‑function attack.

The search combines

* a continuation‑L‑BFGS phase on a smooth approximation of the L∞‑norm
  of the auto‑convolution (soft‑max logits parametrisation);
* aggressive targeted hill‑climbing that moves mass from the largest
  contributors of the current peak to the smallest ones;
* a linear‑programming refinement step (the “LP‑step” from the
  original proof) which, given a current function f, finds a
  direction g that maximises its ℓ₁‑norm while keeping the
  convolution f∗g below the current peak M.  Mixing f←(1‑τ)f+τg
  strictly reduces the peak;
* occasional random hill‑climb polishing.

Multiple random restarts with diverse initialisations (uniform,
exponential, power‑law, Gaussian, edge‑biased) are performed until the
time budget (≈ 1000 s) is exhausted.  The best probability vector (sum = 1)
found is returned as a plain list of floats – it is evaluated with
``evaluate_sequence`` at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces probabilities that sum to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the first 2*n‑1 entries (the full linear convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector ``a``.
    This matches ``evaluate_sequence`` when ``sum(a)=1``:
        score = 2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max + gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(a*a) and its gradient w.r.t. logits ``y``.
    ``lam_factor`` is the multiplicative factor that determines the smoothing
    width (λ = max(c)·lam_factor).  This function is called by L‑BFGS.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: a sequence of L‑BFGS runs with decreasing
    smoothing factor ``lam_factor``.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def random_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb on the simplex.
    A random fraction of mass is moved from one index to another and
    the move is kept only if the true fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that acts on the *current* maximal convolution peak.
    Mass is transferred from the coordinate that contributes most to that
    peak to the one that contributes least.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions of each index to the peak k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random move
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * frac * rng.random()
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step” of the reference paper)
# ----------------------------------------------------------------------


def lp_refinement(
    a: np.ndarray,
    t: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Given a probability vector ``a`` (∑a=1) compute the LP:

        maximise   ∑ b_j
        subject to  (a * b)_k ≤ M   for every k
                    b_j ≥ 0

    where M = max(a*a).  The optimal b is renormalised to sum‑1,
    then mixed with ``a`` using mixing weight ``t``.
    The operation strictly reduces the maximal convolution value.
    """
    n = a.size
    # current peak
    fft_len = next_pow2(2 * n - 1)
    conv = conv_fft(a, fft_len, n)
    M = conv.max()

    m = 2 * n - 1
    # Build the constraint matrix A (size m×n) : (a * b)_k = Σ_j a_{k-j}·b_j
    # For each column j the coefficients a appear shifted by j positions.
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = a

    b_ub = np.full(m, M, dtype=np.float64)
    c = -np.ones(n)                     # maximise ∑b  → minimise −∑b
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # fall back to the original vector
        return a, fast_score(a, fft_len, n)

    b = res.x
    sum_b = b.sum()
    if sum_b <= 1e-12:
        return a, fast_score(a, fft_len, n)

    g = b / sum_b                      # renormalise to sum‑1
    a_new = (1 - t) * a + t * g
    new_score = fast_score(a_new, fft_len, n)
    return a_new, new_score


# ----------------------------------------------------------------------
# Diverse initialisers for random restarts
# ----------------------------------------------------------------------


def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def init_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.3, 2.0)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = rng.uniform(0.2 * n, 0.8 * n)
    sigma = rng.uniform(0.08 * n, 0.25 * n)
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def init_edge_biased(n: int, rng: np.random.Generator) -> np.ndarray:
    # a few heavy values at both ends, a small flat centre
    m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
    edge_val = np.sqrt(n / 2.0) / n
    centre_val = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, centre_val)
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a /= a.sum()
    return a


def init_mixture_two_spikes(n: int, rng: np.random.Generator) -> np.ndarray:
    # two narrow spikes placed at random positions
    a = np.zeros(n, dtype=float)
    pos1 = rng.integers(0, n // 2)
    pos2 = rng.integers(n // 2, n)
    width = max(1, int(0.02 * n))
    for p in (pos1, pos2):
        start = max(0, p - width // 2)
        end = min(n, start + width)
        a[start:end] = 1.0
    a /= a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm runs a handful of
    random restarts, each consisting of a continuation L‑BFGS phase,
    aggressive targeted hill‑climbing, a few LP‑refinement steps and a
    final polishing round.  The best sequence found within the time
    budget is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size – keep it moderate to allow LP steps
    # --------------------------------------------------------------
    n = 2000
    # try to reuse the previously best sequence if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= n:
            a0 = arr[:n].copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 1e-12:
                a0 /= s
            else:
                a0 = init_uniform(n)
        else:
            a0 = init_uniform(n)
    else:
        a0 = init_uniform(n)

    best_a = a0
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # continuation‑L‑BFGS parameters
    lam_factors = np.geomspace(0.5, 5e-8, num=40)   # more stages & finer final λ
    maxiter_stage = 500

    init_functions = [
        init_uniform,
        init_exponential,
        init_powerlaw,
        init_gaussian,
        init_edge_biased,
        init_mixture_two_spikes,
    ]

    iteration = 0

    # --------------------------------------------------------------
    # Main multi‑restart loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick an initialisation strategy
        init_func = rng.choice(init_functions)
        a_start = init_func(n, rng) if init_func.__code__.co_argcount == 2 else init_func(n)
        y_start = np.log(a_start + 1e-12)

        # 1) continuation L‑BFGS
        y_opt = schedule_optimize(
            y_start=y_start,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] L‑BFGS ({init_func.__name__}) improved score={best_score:.6f}")

        # 2) aggressive targeted hill‑climb
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=4000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # 3) occasional LP refinement (every 3rd iteration)
        if iteration % 3 == 0:
            best_a, best_score = lp_refinement(best_a, t=0.25)
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] LP‑refinement score={best_score:.6f}")

        # 4) random polishing if time permits
        if time.time() < deadline - 2.0:
            best_a, best_score = random_hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=3000,
            )
            best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final deep refinement
    # --------------------------------------------------------------
    print("[final] starting deep L‑BFGS refinement")
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"  [final‑L‑BFGS] lam={lam:.2e} fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing passes
    # --------------------------------------------------------------
    final_deadline = deadline
    if time.time() < final_deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_moves=8000,
            frac=0.05,
        )
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=5000,
        )

    # --------------------------------------------------------------
    # Return result (as a plain list of floats)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence = {final_val:.6f}")
    return final_seq
