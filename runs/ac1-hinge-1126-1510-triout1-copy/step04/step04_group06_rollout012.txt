"""
propose_candidate: a multi‑restart continuation L‑BFGS + Adam sub‑gradient
+ enhanced simulated annealing to minimise 2·n·max(conv(a,a)).
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

def softmax(y):
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def next_pow2(x):
    return 1 << ((x-1).bit_length())

def full_conv(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2*len(a)-1]

def fast_score(a, fft_len, n):
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v):
    """Euclidean projection onto simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    """Smooth max objective using log‑sum‑exp; returns value and gradient w.r.t. y."""
    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2*n-1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                # d(smooth_max)/d conv
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n-1):(n-1+n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def adam_subgradient(a0, deadline, fft_len_conv, fft_len_g, n,
                     lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam‑style sub‑gradient descent on the exact max (non‑smooth)."""
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        # convolution and max
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
        max_val = conv.max()
        # active mask
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n-1):(n-1+n)]

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection onto simplex
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
    return best, best_score

def hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000, rng=None):
    """Small random mass transfers that strictly improve the score."""
    if rng is None:
        rng = np.random.default_rng()
    best = a.copy()
    best_score = cur_score
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a fraction of i to j
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def simulated_annealing(a_start, score_start, fft_len, n, deadline, rng):
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0
    while time.time() < deadline:
        # pick source with noticeable mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue
        # target distinct
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_delta = cur_a[i] * 0.3
        delta = rng.random() * max_delta
        if delta == 0:
            continue
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                cur_a[i] += delta
                cur_a[j] -= delta
                # temperature update
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        it += 1
        T *= decay
        # occasional reheating
        if it % 5000 == 0:
            T = max(best_score * 0.02, 1e-8)
    return best_a, best_score

def generate_initial(mode, n, rng):

        a2 = np.exp(-rate2 * np.arange(n)[::-1])

        a = a[:n]
        a = a / a.sum()
    else:  # random Dirichlet
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    return a

"""
propose_candidate – improved multi‑phase optimisation.

The algorithm builds on the previous L‑BFGS + sub‑gradient pipeline but adds
* a continuation scheme (gradually decreasing the log‑sum‑exp smoothing)
* Adam‑style projected sub‑gradient descent on the true max‑convolution
* a richer set of random initialisers (including a discretised Beta shape)
* a slightly more aggressive simulated‑annealing schedule
* a final greedy hill‑climb.

All phases respect the 1000 s budget.  The best vector is stored in the
global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max onto the probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * len(a) - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) for a already normalised (∑a=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed‑max objective based on log‑sum‑exp and its gradient w.r.t. y.
    Returns (value, gradient_y).  The smoothing parameter is
    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ---- smooth approximation of max(conv) ----
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient optimisation (exact max)
# ----------------------------------------------------------------------


def adam_subgradient(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    lr: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the true max‑convolution.
    Keeps the iterate on the simplex via Euclidean projection.
    """
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active‑set mask (indices attaining the max)
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

    return best, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """Local search that repeatedly moves a tiny fraction of mass between two
    positions if it improves the (fast) score."""
    if rng is None:
        rng = np.random.default_rng()
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score

def simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that relocates a random fraction of mass.
    Improves exploration after the gradient phases.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:

        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.3
        delta = rng.random() * max_delta
        if delta == 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

            best_score = cur_score
            best_a = cur_a.copy()

        it += 1
        T *= decay
        if it % 5000 == 0:
            T = max(best_score * 0.02, 1e-8)

def generate_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n according to the requested mode.
    Modes:
        uniform – plain uniform.
        exp     – decreasing exponential from the left.
        power   – power‑law decay.
        beta    – discretised Beta distribution (flexible left/right shape).
        mix     – mixture of a left‑hand and a right‑hand exponential.
        random  – Dirichlet(0.5) random.
    """
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
        a = a / a.sum()
    elif mode == "power":
        p = rng.uniform(0.2, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = a / a.sum()
    elif mode == "beta":
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        i = np.arange(1, n + 1, dtype=float)
        a = (i ** (alpha - 1)) * ((n + 1 - i) ** (beta - 1))
        a = a / a.sum()
    elif mode == "mix":
        # two exponentials, one anchored at the left, one at the right
        rate1 = rng.uniform(0.0005, 0.02)
        rate2 = rng.uniform(0.0005, 0.02)
        left = np.exp(-rate1 * np.arange(n))
        right = np.exp(-rate2 * np.arange(n)[::-1])
        alpha = rng.random()
        a = alpha * left + (1 - alpha) * right
        a = a / a.sum()
    else:  # random
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    return a

        prev = np.maximum(prev, 0.0)
        max_n = 2500
        if prev.size > max_n:
            prev = prev[:max_n]
        if prev.sum() > 0:
            a0 = prev / prev.sum()
            n = a0.size
        else:
            n = 2500
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2500
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # FFT lengths (powers of two for speed)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # auxiliary convolutions

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with decreasing smoothing
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.3,
        0.15,
        0.08,
        0.04,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
    ]
    phase1_deadline = start_time + 0.30 * budget_s
    restarts = 12

    for r in range(restarts):
        if time.time() >= phase1_deadline:
            break
        mode = rng.choice(["uniform", "exp", "power", "beta", "mix", "random"])
        a_start = generate_initial(mode, n, rng)
        y = np.log(a_start + 1e-12)

        for lam in lam_schedule:
            if time.time() >= phase1_deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 600, "gtol": 1e-12, "disp": False},
            )
            y = res.x
            a_candidate = softmax(y)
            score_candidate = fast_score(a_candidate, fft_len_conv, n)
            if score_candidate < best_score:
                best_score = score_candidate
                best_a = a_candidate.copy()
                elapsed = time.time() - start_time
                print(
                    f"[L‑BFGS restart={r} lam={lam:.4f}] score={best_score:.6f} time={elapsed:.1f}s"
                )

        # short hill‑climb after each continuation block
        hill_deadline = min(phase1_deadline, time.time() + 0.10)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hill_deadline,
            max_iter=800,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style projected sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = adam_subgradient(
            best_a,
            phase2_deadline,
            fft_len_conv,
            fft_len_g,
            n,
            lr=0.2,
        )
        print(f"[Adam sub‑grad] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for non‑local moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[SA] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_iter=3000,
        rng=rng,
    )

    # ------------------------------------------------------------------
    # Optional reporting via the supplied evaluator
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float("nan")
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # store globally for possible reuse in a later call
    globals()["height_sequence_1"] = final_a.tolist()

    return final_a.tolist()
