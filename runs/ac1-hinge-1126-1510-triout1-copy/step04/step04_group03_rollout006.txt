    a = a.copy()

        conv = np.convolve(a, a)
        M = conv.max()

        # Build as csc matrix with diagonals offset i: values a[i]
        data = [np.full(n, a[i]) for i in range(n)]

        b = cp.Variable(n)

        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

"""
Optimisation of a non‑negative step function for the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)²)

The objective is scale‑invariant, therefore we keep the sequence on the
probability simplex (sum = 1).  The algorithm consists of:

1.   A cheap parametric sweep (uniform, linear, exponential, power‑law,
     edge‑heavy, sinusoid, Dirichlet draws) to obtain a few promising seeds.
2.   Continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
     max‑convolution.  A fine λ‑schedule and many restarts are used.
3.   An LP “balancing’’ step (the classic iteration of the reference paper):
     maximise ∑b subject to b ≥ 0 and (a * b) ≤ M (where M is the current
     peak).  Mixing a small portion of the optimiser b back into a
     provably reduces the peak.  A handful of such LP steps are performed.
4.   Projected sub‑gradient descent (exact max‑convolution) for a quick
     final polish.
5.   A greedy hill‑climb that repeatedly transfers a small random mass
     between two coordinates while preserving the simplex.

All steps respect the simplex constraint and use FFT‑based convolutions
(O(n log n)).  The total budget (default 1000 s) is split automatically.
The best sequence found before the deadline is returned as a plain list.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy.sparse import diags, csc_matrix

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:      # safeguard – L‑BFGS occasionally blows up
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # short hill‑climb after each λ
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent
    with a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# LP balancing step (the classic “M‑balancing” iteration)
# ----------------------------------------------------------------------
def lp_improve(a: np.ndarray,
               n: int,
               deadline: float,
               t_init: float = 0.1,
               max_iters: int = 12):
    """
    Perform a few iterations of the linear‑programming balancing step:

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k    (M = max(a*a))

    After each LP we mix a small fraction of the (scaled) solution back into
    the current vector a.  The mixing factor t is halved whenever a step does
    not improve the objective.
    """
    try:
        import cvxpy as cp
    except Exception as e:
        print("[lp] cvxpy not available:", e)
        return a                     # fall back to the original vector

    a = a.copy()
    a = a / a.sum()                   # ensure we are on the simplex

    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build the convolution matrix T (size (2n‑1)×n)  –  each diagonal i
        # contains the constant a[i].
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear programme
        b = cp.Variable(n)
        constraints = [T @ b <= M, b >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=400, abstol=1e-9, reltol=1e-9)
        except Exception as e:
            print("[lp] solve exception:", e)
            break
        if b.value is None:
            break
        b_val = np.maximum(np.array(b.value).ravel(), 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        # Scale to keep the simplex sum = 1
        b_scaled = b_val / s

        # Mix and project back onto the simplex (the mixture already has sum = 1)
        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a, n, next_pow2(2 * n - 1))

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init               # reset mixing factor after a successful step
            it += 1
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step sequence.
    Returns a plain Python list of floats (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size and initial distribution
    # ------------------------------------------------------------------
    max_n = 2500                     # a few thousand entries – safe for LP
    # use a previously discovered construction if it exists
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0[a0 < 0] = 0.0
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (collect seeds)
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len_conv)
        candidates.append((sc, arr))

    # uniform
    add_candidate(np.full(n, 1.0 / n))

    # linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # exponential families
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # random Dirichlet (Gamma) draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # keep the best few seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:6]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin
    phase1_end = start + 0.50 * budget_s      # L‑BFGS
    phase2_end = start + 0.80 * budget_s      # LP + sub‑gradient
    # remaining time will be used for a final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.8, 0.6, 0.4, 0.2, 0.1, 0.08, 0.05,
                    0.03, 0.02, 0.01, 0.008, 0.005,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 350

    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – LP balancing + projected sub‑gradient
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # LP stage (takes a few seconds)
        lp_deadline = min(phase2_end, deadline - 0.3)
        print("[LP] starting balancing iterations")
        best_a = lp_improve(best_a, n, lp_deadline, t_init=0.12, max_iters=12)
        best_score = fast_score(best_a, n, fft_len_conv)
        print(f"[LP] post‑LP score={best_score:.6f}")

        # Sub‑gradient descent (short)
        if time.time() < phase2_end:
            sg_deadline = min(phase2_end, deadline - 0.2)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (as much time as left)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    # ------------------------------------------------------------------
    # Return the sequence (plain list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)    # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
