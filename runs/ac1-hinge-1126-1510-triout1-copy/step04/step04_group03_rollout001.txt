def smooth_max_and_grad(a, lam_factor):
    # compute conv = a*a full length.
    conv = full_conv(a, fft_len_conv)
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z
    # compute gradient wrt a: 2 * convolution reversed a with w
    a_rev = a[::-1]
    fm = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    grad_a = 2.0 * conv_g[(n - 1):(n - 1 + n)]
    # Could also compute final gradient of smooth_max directly, but this is same.
    return max_c, lam, w, grad_a

def mirror_descent(a_start, lam_schedule, max_iters_per_lam, deadline, fft_len_conv, fft_len_g, n):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it_total = 0
    for lam_factor in lam_schedule:
        if time.time() >= deadline:
            break
        # set max iterations for this lambda
        max_it = max_iters_per_lam
        for it in range(max_it):
            if time.time() >= deadline:
                break
            # compute smooth max and gradient
            # lam_factor is factor of max, we compute current lam inside
            # compute convolution
            conv_full = full_conv(a, fft_len_conv)
            max_c = conv_full.max()
            lam = max_c * lam_factor + 1e-12
            shifted = conv_full - max_c
            exp_shift = np.exp(shifted / lam)
            Z = exp_shift.sum()
            w = exp_shift / Z
            # gradient wrt a:
            a_rev = a[::-1]
            fw = np.fft.fft(w, fft_len_g)
            fa_rev = np.fft.fft(a_rev, fft_len_g)
            conv_grad = np.real(np.fft.ifft(fa_rev * fw))
            grad_a = 2.0 * conv_grad[(n - 1):(n - 1 + n)]

            # step size schedule
            step = 1.0 / np.sqrt(it_total + 1.0) * 0.5   # tuned
            # update using exponentiated update
            # to ensure positivity we use multiplicative update
            a_new = a * np.exp(-step * grad_a)
            # normalize
            a_new_sum = a_new.sum()
            if a_new_sum == 0:
                a_new = np.full_like(a, 1.0 / n)
            else:
                a_new /= a_new_sum

            # evaluate score
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
            # accept if improves else keep old (or keep new? Could accept occasional).
            if new_score <= fast_score(a, fft_len_conv, n):
                a = a_new
            # else keep a unchanged

            it_total += 1
            # maybe early stop if improvement small?

    return best_a, best_score

"""
Improved step‑function optimizer for the evaluation function.

The algorithm works in several stages and respects a 1000 s time budget.

1. **Parameter sweep** – generate a diverse set of seed distributions
   (uniform, linear, exponential, power‑law, edge‑heavy, sinusoidal,
   block‑wise, sparse Dirichlet, and the previously best known sequence).

2. **Continuation L‑BFGS** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) while gradually reducing the smoothing
   factor.  Multiple random restarts are performed from the best seeds.

3. **Mirror‑descent on the smoothed max** – exponentiated‑gradient updates
   on the probability simplex (multiplicative updates followed by
   renormalisation).  The smoothing factor is again decreased in a
   continuation schedule.

4. **Projected sub‑gradient descent** – exact max‑convolution sub‑gradient
   with a simple back‑tracking line‑search.

5. **Random annealing** – occasional downhill moves with a temperature
   schedule to escape tiny local minima.

6. **Greedy hill‑climb** – final polishing by moving small random
   fractions of mass between coordinates.

All convolutions are performed with FFTs (O(n log n)).  The final
sequence is returned as a Python list and its score is printed using the
provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1 (the true score is 2·n·max(a*a) / sum(a)², but the ratio is
    scale‑invariant, so we only need max(a*a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def smooth_max_and_grad(a: np.ndarray,
                        lam_factor: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int):
    """
    Compute a smoothed version of max(a*a) using log‑sum‑exp
    and also return the gradient w.r.t. a (exact for the smooth max).
    """
    # convolution a*a
    conv_full = full_conv(a, fft_len_conv)          # length 2n-1
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z                               # softmax weights

    # gradient of smooth max w.r.t. a:
    # ∂smooth/∂a = 2 * (a_rev * w)  (cross‑correlation)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_grad = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = 2.0 * conv_grad[(n - 1):(n - 1 + n)]

    # smooth max value (not needed directly here, but convenient)
    smooth_val = max_c + lam * np.log(Z)

    return smooth_val, max_c, lam, w, grad_a


# ----------------------------------------------------------------------
# Continuation L‑BFGS on the smoothed max
# ----------------------------------------------------------------------
def obj_and_grad_y(y: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_g: int,
                   lam_factor: float):
    """
    L‑BFGS objective: smooth max (log‑sum‑exp) of the convolution.
    The optimisation variable is y such that a = softmax(y) lives on the
    probability simplex.
    Returns (objective, gradient w.r.t. y).
    """
    # softmax to obtain a on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # smooth max and its gradient w.r.t. a
    smooth_val, max_c, lam, w, grad_a = smooth_max_and_grad(
        a, lam_factor, fft_len_conv, fft_len_g, n)

    # chain rule: ∂obj/∂y_i = a_i * (∂obj/∂a_i - Σ_k a_k ∂obj/∂a_k)
    mean_grad = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_grad)

    obj = 2.0 * n * smooth_val
    return obj, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in `lam_schedule`, using the result of the
    previous λ as a warm start.
    """
    # initialise y = log(a) (adding a small epsilon to avoid log(0))
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=obj_and_grad_y,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Occasionally L‑BFGS may fail – ignore and continue.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

        # short hill‑climb after each λ to sharpen the result
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len_conv,
                                     n,
                                     hc_deadline,
                                     max_iter=1500)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Mirror descent on the smoothed max (multiplicative updates)
# ----------------------------------------------------------------------
def mirror_descent(a0: np.ndarray,
                   lam_schedule,
                   iters_per_lam: int,
                   deadline,
                   fft_len_conv,
                   fft_len_g,
                   n):
    """
    Exponentiated‑gradient (mirror) descent on the smoothed max.
    Uses multiplicative updates a <- a * exp(-η * grad) and re‑normalises.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)
    it_total = 0

    for lam_factor in lam_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break

            # compute smoothed max and its gradient wrt a
            smooth_val, max_c, lam, w, grad_a = smooth_max_and_grad(
                a, lam_factor, fft_len_conv, fft_len_g, n)

            # step size (simple schedule)
            step = 0.5 / np.sqrt(it_total + 1.0)

            # multiplicative update and projection onto simplex (renormalise)
            a_new = a * np.exp(-step * grad_a)
            sum_new = a_new.sum()
            if sum_new == 0.0:
                a_new = np.full_like(a, 1.0 / n)
            else:
                a_new /= sum_new

            # evaluate
            sc_new = fast_score(a_new, fft_len_conv, n)
            if sc_new < best_sc:
                best_sc, best_a = sc_new, a_new.copy()

            # accept if improvement, otherwise keep old a
            if sc_new <= fast_score(a, fft_len_conv, n):
                a = a_new

            it_total += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline,
                         fft_len_conv,
                         fft_len_g,
                         n):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line‑search and Euclidean projection onto
    the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # compute convolution and the active mask (indices attaining the max)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max (see comment in the analysis)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            cand = simplex_projection(a - step * g)
            sc_cand = fast_score(cand, fft_len_conv, n)
            if sc_cand < best_sc:
                best_sc, best_a = sc_cand, cand.copy()
                a = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_sc = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_sc = fast_score(best, fft_len, n)
        if new_sc < best_sc:
            best_sc = new_sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Random annealing phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Accept worsening moves with a decreasing temperature to escape
    shallow local minima.
    """
    cur_a = a.copy()
    cur_sc = score
    best_a = a.copy()
    best_sc = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_sc = fast_score(cand, fft_len, n)
        if new_sc < cur_sc:
            cur_a, cur_sc = cand, new_sc
            if new_sc < best_sc:
                best_a, best_sc = cand.copy(), new_sc
        else:
            prob = np.exp(-(new_sc - cur_sc) / T)
            if rng.random() < prob:
                cur_a, cur_sc = cand, new_sc

        it += 1
        if it % 5000 == 0:
            T = max(T * 0.92, 1e-5)

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (few thousand entries – still fast with FFT)
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)    # for a*a
    fft_len_g = fft_len_conv               # same length is sufficient

    # --------------------------------------------------------------
    # Helper to store candidate seeds
    # --------------------------------------------------------------
    candidates = []

    def add_candidate(arr: np.ndarray):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0.0:
            return
        arr = np.maximum(arr, 0.0)
        arr /= arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # --------------------------------------------------------------
    # 0. Sweep a variety of parametric families (including previous best)
    # --------------------------------------------------------------
    # Uniform
    add_candidate(np.full(n, 1.0 / n))

    # Linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_candidate(lin)

    # Exponential families
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # Edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # Block‑wise constant (B blocks)
    for B in [5, 10, 20]:
        block_weights = rng.dirichlet(alpha=np.ones(B))
        block_len = n // B
        arr = np.repeat(block_weights, block_len)[:n]
        add_candidate(arr)

    # Sparse Dirichlet (few non‑zeros)
    for k in [10, 20, 40]:
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.gamma(1.0, 1.0, size=k)
        arr = np.zeros(n, dtype=float)
        arr[idx] = vals
        add_candidate(arr)

    # Random Dirichlet draws (various concentrations)
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Previous best known sequence (if available)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size >= n:
                prev = prev[:n]
            else:
                # pad with tiny uniform mass to reach length n
                pad = np.full(n - prev.size, 1e-12)
                prev = np.concatenate([prev, pad])
            add_candidate(prev)

    # Keep the top seeds (sorted by quick fast_score)
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _sc, arr in candidates[:8]]
    print(f"[sweep] best fast_score from sweep = {candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    phase1_end = start_time + 0.60 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.80 * budget_s   # Mirror descent
    phase3_end = start_time + 0.95 * budget_s   # Sub‑gradient + anneal
    final_deadline = deadline

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule_lbfgs = [0.5, 0.3, 0.2, 0.1, 0.07, 0.04,
                          0.02, 0.01, 0.005, 0.002, 0.001]
    maxiter_per_lam = 350
    best_a = top_seeds[0].copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    restarts = 0
    while time.time() < phase1_end and restarts < 25:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 5e-4
        a0 = np.maximum(seed_vec + jitter, 0.0)
        a0 /= a0.sum()

        a_opt, sc_opt = lbfgs_continuation(a0,

                                           phase1_end,

        if sc_opt < best_sc:

        restarts += 1

    # --------------------------------------------------------------
    # Phase 2 – mirror descent on smoothed max (exponentiated updates)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule_mirror = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        iters_per_lam = 300
        best_a, best_sc = mirror_descent(best_a,
                                         lam_schedule_mirror,
                                         iters_per_lam,
                                         phase2_end,
                                         fft_len_conv,
                                         fft_len_g,

        print(f"[mirror] post‑mirror fast_score = {best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        sg_deadline = min(phase3_end, final_deadline - 0.3)
        best_a, best_sc = subgradient_optimize(best_a,
                                               sg_deadline,
                                               fft_len_conv,
                                               fft_len_g,
                                               n)
        print(f"[sub‑gradient] after exact sub‑gradient fast_score = {best_sc:.6f}")

        # Random annealing to shake out tiny local minima
        anneal_deadline = min(phase3_end, final_deadline - 0.2)
        best_a, best_sc = random_anneal(best_a,
                                        best_sc,
                                        fft_len_conv,
                                        n,
                                        anneal_deadline)
        print(f"[anneal] after annealing fast_score = {best_sc:.6f}")

    # --------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # --------------------------------------------------------------
    best_a, best_sc = hill_climb(best_a,
                                 best_sc,
                                 fft_len_conv,
                                 n,
                                 final_deadline,
                                 max_iter=10000)

    # --------------------------------------------------------------
    # Return the sequence and print the final evaluated score
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
