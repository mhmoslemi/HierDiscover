"""
Propose a step‑function that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

where `conv` is the linear convolution.
The code

* explores a few one‑parameter families (exponential, symmetric exponential,
  power‑law) by direct scalar optimisation,
* refines the best candidate with a short L‑BFGS run on a smoothed max,
* finishes with a momentum sub‑gradient descent and a final greedy hill‑climb.

All phases respect the 1000 s time budget.  The final sequence is stored in the
global `height_sequence_1` for possible reuse in later calls.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, minimize_scalar

# ------------------------------------------------------------
# Basic linear‑algebra utilities (FFT based convolution)
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a already normalised (sum=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Smoothed‑max objective used by L‑BFGS
# ------------------------------------------------------------

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective based on log‑sum‑exp.
    Returns (f, grad_y) where f = 2·n·smooth_max(conv) and grad_y is
    the gradient w.r.t. the unconstrained vector y (soft‑max is applied inside).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ------------------------------------------------------------
# Momentum sub‑gradient descent (true max)
# ------------------------------------------------------------

def subgradient_momentum_optimize(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """Projected sub‑gradient descent with momentum on the true max‑conv."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9                    # momentum coefficient
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active set mask (indices attaining the max)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (order‑0 derivative of max)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step ~ 1/√t
        step = 1.0 / np.sqrt(it + 1.0)
        a_new = a - step * v
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ------------------------------------------------------------

def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """Local improvement by moving a tiny fraction of mass between two indices."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ------------------------------------------------------------
# Parameter‑family search helpers
# ------------------------------------------------------------

def _exponential_vec(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-alpha * i)."""
    a = np.exp(-alpha * np.arange(n))
    return a / a.sum()


def _sym_exponential_vec(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-alpha * min(i, n‑1‑i)).  Symmetric about the centre."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d)
    return a / a.sum()


def _powerlaw_vec(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
    return a / a.sum()


def optimise_family(fun, bounds, deadline, max_evals=80):
    """
    Simple bounded scalar optimisation using Brent's method.
    Stops early if the global deadline is reached.
    """
    best_x = None
    best_score = math.inf

    def wrapped(x):
        nonlocal best_x, best_score
        if time.time() > deadline:
            raise RuntimeError("time")
        val = fun(x)
        if val < best_score:
            best_score = val
            best_x = x
        return val

    try:
        minimize_scalar(wrapped,
                        bounds=bounds,
                        method='bounded',
                        options={'xatol': 1e-8, 'maxiter': max_evals})
    except RuntimeError:
        pass
    return best_x, best_score
