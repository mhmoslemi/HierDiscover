Improved step‑function optimizer.

The optimizer works in several phases within the 1000 s budget:

1.  **Parametric seed sweep** – builds a diverse collection of
    probability vectors (uniform, linear, exponential, power‑law,
    sinusoid, random Dirichlet with many concentrations and
    mixtures).  The best few are kept as seeds.

2.  **Continuation L‑BFGS** – each seed is refined with a
    log‑sum‑exp (smooth‑max) surrogate of the max‑convolution.
    A decreasing λ‑schedule (0.5 → 0.001) and a larger iteration budget
    are used.  The soft‑max parametrisation guarantees the simplex
    constraints automatically.

3.  **Projected sub‑gradient descent** – the exact non‑smooth objective
    is attacked directly.  An adaptive step‑size (∝1/√t) together with
    a back‑tracking line‑search is employed.  Momentum is added to
    accelerate convergence.

4.  **Simulated‑annealing** – random mass transfers between two
    coordinates are performed; worsening moves are accepted with a
    temperature that decays slowly (geometric schedule).

5.  **Intensive hill‑climbing** – a greedy pairwise mass‑transfer
    routine (up to 20 000 successful moves) polishing the final
    candidate.

All steps respect the probability simplex (non‑negative, sum = 1) and
use FFT‑based convolution (O(n log n)).  The best sequence found
before the deadline is returned as a plain Python list.

# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a).
    The sequence is assumed to be normalised to sum = 1,
    therefore the true score equals 2·n·max(conv).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Smoothed max (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------
def smooth_objective_and_grad(y: np.ndarray, n: int,
                              fft_len_conv: int, fft_len_g: int,
                              lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)                     # automatically satisfies sum=1

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth max -----
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12   # smoothing scale
    shifted = conv_full - max_c        # improve numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                   # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Projected sub‑gradient descent (with simple momentum)

def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int,
                         momentum: float = 0.9):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a back‑tracking line‑search and a small momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)              # momentum buffer
    it = 0

    while time.time() < deadline:
        # ---- current max‑convolution and active mask ----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ---- sub‑gradient of the max‑convolution ----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)

            best_a = a_cand.copy()
            a = a_cand
        else:
            # revert momentum for the next iteration (to stay stable)
            v = np.zeros_like(v)

        it += 1

    return best_a, best_score

# Greedy hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb that moves a small random fraction of mass from a
    donor index to a receiver index.  Only strictly improving moves are
    kept.  The loop stops when the deadline is reached or when the
    maximum number of attempts is exhausted.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move only a modest fraction to keep the search stable
        delta = best[i] * rng.random() * 0.03
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best (λ={lam:.5f}) score={best_score:.6f}")

                    fft_len: int,
                    n: int,

    Accept worsening moves with a decreasing temperature.

    cur_score = score
    best_a = a.copy()

            if new_score < best_score:

        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (collect promising seeds)
    # ------------------------------------------------------------------
    seeds = []

    def add_seed(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.size == 0 or arr.sum() <= 0.0:
            return
        arr = np.maximum(arr, 0.0)
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        seeds.append((sc, arr))

    # 1) Uniform (baseline)
    add_seed(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_seed(lin)

    # 3) Exponential families
    for rate in np.logspace(-4, -2, 5):
        add_seed(np.exp(-rate * np.arange(n)))

    # 4) Power‑law families
    for p in np.linspace(0.2, 3.0, 7):
        add_seed((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_seed(arr)

    # 6) Sinusoidal‑squared (always non‑negative)
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_seed(sin_arr + 1e-6)

    # 7) Random Dirichlet draws (various concentrations)
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(6):
            add_seed(rng.gamma(alpha, 1.0, size=n))

    # 8) Mixture of two Dirichlets (biased towards a structured shape)
    for a1, a2 in [(0.1, 5.0), (5.0, 0.1)]:
        for _ in range(4):
            d1 = rng.gamma(a1, 1.0, size=n)
            d2 = rng.gamma(a2, 1.0, size=n)
            mix = 0.5 * d1 + 0.5 * d2
            add_seed(mix)

    # retain the best few seeds
    seeds.sort(key=lambda t: t[0])
    top_seeds = [arr for _sc, arr in seeds[:12]]
    print(f"[sweep] best sweep score = {seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5         # safety margin

    phase1_end = start_time + 0.55 * budget_s      # L‑BFGS stage
    phase2_end = start_time + 0.80 * budget_s      # sub‑gradient stage
    phase3_end = deadline                         # annealing stage

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.010,
                    0.006, 0.003, 0.0015, 0.001]
    maxiter_per_lam = 350
    restarts = 0

    while time.time() < phase1_end and restarts < 30:
        # pick a seed and inject tiny jitter for diversity
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 5e-4
        a_start = seed_vec + jitter
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (with momentum)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.3)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n,
                                                  momentum=0.85)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing / random perturbations
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_anneal(best_a,
                                              best_score,
                                              fft_len_conv,
                                              n,
                                              phase3_end)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=20000)
    print(f"[final hill‑climb] evaluated score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence as a plain Python list
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # final sanity check – will be used by the outer harness
    try:
        final_eval = evaluate_sequence(final_seq)   # type: ignore
        print(f"[final] evaluate_sequence = {final_eval:.6f}")
    except Exception as exc:
        # if something went wrong, still return the raw sequence
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
