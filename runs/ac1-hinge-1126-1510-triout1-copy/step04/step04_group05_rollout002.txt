    g = res.x
    return g

def simulated_annealing(a, cur_score, fft_len, n, deadline):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # exponential cooling
    T0 = 0.01  # initial temperature relative to cur_score
    while time.time() < deadline:
        # compute remaining fraction of time to schedule temperature
        fraction = (deadline - time.time()) / (deadline - start_time)
        T = T0 * fraction
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05  # move up to 5% of mass at i
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        delta_score = new_score - best_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / T):
            # accept move
            if new_score < best_score:
                best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Hybrid optimizer for the step‑function attack.

The algorithm builds on the previous “continuation L‑BFGS + hill‑climb”
pipeline and adds three new ingredients:

1. **Linear‑programming refinement** – given the current sequence `f`,
   we solve a linear program for a witness `g` that satisfies
   `f * g ≤ max(f*f)`.  Mixing `f` and `g` (convex combination) provably
   does not increase the worst‑case convolution and often reduces it.

2. **Aggressive targeted hill‑climb** – the same greedy mass‑transfer
   that directly attacks the dominant convolution peak, but with a
   larger move budget and a slightly smaller transfer fraction.

3. **Simulated annealing polishing** – a temperature‑controlled random
   walk on the simplex that can escape small local minima left by the
   deterministic steps.

All phases respect the 1000 s budget, use only NumPy / SciPy,
and return a non‑negative list whose evaluation is printed for reference.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.linalg import toeplitz


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns the first ``2*n‑1`` entries (full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector ``a``.
    This matches the value returned by ``evaluate_sequence`` (the denominator
    ``sum(a)**2`` equals 1 after normalisation).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution and its gradient w.r.t.
    the unconstrained logits ``y`` (the log‑softmax coordinates).

    Parameters
    ----------
    y : ndarray
        Unconstrained parameters; a = softmax(y).
    n : int
        Length of the probability vector.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for the convolution and the auxiliary
        convolution needed for the gradient.
    lam_factor : float
        Multiplicative factor for the smoothing width λ.

    Returns
    -------
    f : float
        Smoothed version of ``2·n·max(a*a)``.
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1   (cross‑correlation)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list that gradually reduces the smoothing.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Generic random hill‑climb on the simplex.
    Randomly transfers a fraction of mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb.
    At each step the index contributing most to the current peak
    (k* = argmax conv) donates a small fraction of its mass to the
    index with the smallest contribution.  This directly reduces the
    true ``max(a*a)``.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        # current convolution and peak index
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random move
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


def combine_sequences(f: np.ndarray, g: np.ndarray, t: float) -> np.ndarray:
    """
    Convex combination (1‑t)·f + t·g followed by normalisation to sum‑1.
    """
    new = (1.0 - t) * f + t * g
    s = new.sum()
    if s > 1e-12:
        new = new / s
    else:
        new = np.full_like(new, 1.0 / len(new))
    return new


def solve_lp_for_g(a: np.ndarray, max_conv: float) -> np.ndarray:
    """
    Linear‑programming step: maximise Σ g_j subject to
        (a * g)_k ≤ max_conv    for all k,
        g_j ≥ 0.
    Returns the optimal g (may be all zeros if LP fails).
    """
    n = len(a)
    # Build Toeplitz matrix that implements the convolution a * g.
    c = np.concatenate([a, np.zeros(n - 1)])
    r = np.concatenate([a[:1], np.zeros(n - 1)])
    C = toeplitz(c, r)                     # shape (2n‑1, n)

    # Linear programme: minimize -Σg_j  (i.e. maximise Σg_j)
    obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, max_conv)
    bounds = [(0.0, None)] * n

    res = linprog(
        c=obj,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    else:
        # If the LP fails we fall back to a zero vector (which is feasible).
        return np.zeros_like(a)


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk on the simplex.
    Moves that worsen the objective are occasionally accepted,
    with a linearly cooling schedule.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.02 * best_score   # initial temperature (relative to the score)

    while time.time() < deadline:
        # linear cooling factor based on remaining fraction of the budget
        fraction = max(0.0, (deadline - time.time()) / (deadline - start))
        T = T0 * fraction + 1e-12

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        dE = new_score - best_score

        if dE < 0 or rng.random() < np.exp(-dE / T):
            # accept
            if new_score < best_score:
                best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (its sum will be 1 after the
        internal normalisation).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – keep a manageable length (2000) but reuse a previous
    # construction if one exists (height_sequence_1).
    # --------------------------------------------------------------
    default_n = 2000
    if "height_sequence_1" in globals():
        prev = globals()["height_sequence_1"]
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            a0 = arr[:default_n].copy()
        else:
            a0 = np.full(default_n, 1.0 / default_n)
    else:
        a0 = np.full(default_n, 1.0 / default_n)

    n = default_n
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 > 1e-12:
        a0 = a0 / s0
    else:
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Pre‑compute FFT lengths for the convolutions.
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best solution.
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Continuation L‑BFGS hyper‑parameters.
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 5e-6, num=30)   # smoothing schedule
    maxiter_stage = 400

    iteration = 0
    # --------------------------------------------------------------
    # Multi‑restart optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a = a / a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a = a / a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a = a / a.sum()
            y0 = np.log(a + 1e-12)
        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ----- Continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- Aggressive targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=4000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Linear‑programming refinement.
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        conv_cur = conv_fft(best_a, fft_len_conv, n)
        M_cur = conv_cur.max()
        print(f"[LP] current max convolution = {M_cur:.8f}")

        g = solve_lp_for_g(best_a, M_cur)

        # Try a few mixing ratios and keep the best.
        mix_best_score = best_score
        mix_best_a = best_a.copy()
        for t in [0.15, 0.30, 0.45, 0.60, 0.75]:
            a_mix = combine_sequences(best_a, g, t)
            sc = fast_score(a_mix, fft_len_conv, n)
            if sc < mix_best_score:
                mix_best_score = sc
                mix_best_a = a_mix

        if mix_best_score < best_score:
            best_score = mix_best_score
            best_a = mix_best_a
            best_y = np.log(best_a + 1e-12)
            print(f"[LP] mixing improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Post‑LP targeted hill‑climb.
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        hc_deadline = min(deadline, time.time() + 1.2)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=6000,
            frac=0.06,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Simulated annealing polishing (uses any remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        sa_deadline = deadline
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            sa_deadline,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[SA] final fast score after annealing = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final deep L‑BFGS pass (fine‑tuning with the smallest λ values).
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        for lam in lam_factors[-10:]:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 500, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len_conv, n)
            print(f"[final L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the final sequence (global variable for possible reuse).
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
