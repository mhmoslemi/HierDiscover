def propose_candidate(seed=42, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.3 # margin
    np.random.seed(seed)
    random.seed(seed)

    # sequences length: could be a parameter, maybe 12000.
    n = 12000

    # initialize starting seq: try previous height_sequence_1 or generate.
    init_seq = None
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # global variable
        if prev.ndim == 1 and prev.size >0:
            init_seq = prev.copy()
    except Exception:
        pass

    if init_seq is None or init_seq.size != n:
        # generate random shape (power law)
        p = random.uniform(0.2, 1.0)
        init_seq = np.arange(1, n+1, dtype=np.float64) ** (-p)

    # normalize to simplex
    init_seq = np.maximum(init_seq, 0.0)
    init_seq_sum = init_seq.sum()
    if init_seq_sum < 1e-12:
        init_seq = np.full(n, 1.0/n)
    else:
        init_seq = init_seq / init_seq_sum

    # Run optimizer
    best_seq, best_score = adam_optimize_simplex(init_seq, deadline - time.time())

    # final verification using evaluate_sequence
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final evaluate_sequence score {final_score}")
    return final_seq

"""
Optimise non‑negative step‑function heights to minimise the evaluation
metric 2·n·max(conv(a,a))/sum(a)^2.  The optimizer works directly on the
soft‑max parametrisation (log‑its) of the coefficients and performs an
Adam update on the sub‑gradient of the ℓ∞‑norm of the self‑convolution.
A single long optimisation run (or a few restarts) is performed within the
time budget, starting from the previously discovered sequence if it is
available.
"""

import time
import random
import numpy as np

def next_pow2(x: int) -> int:
    p = 1
    while p < x:
        p <<= 1
    return p

def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_shift = y - np.max(y)
    exp_y = np.exp(y_shift)
    return exp_y / exp_y.sum()

def subgradient_max_conv(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv)`` w.r.t. ``a``.
    ``conv`` is the full linear convolution a*a.
    """
    n = a.shape[0]
    M = conv.max()
    active = np.where(conv >= M - eps)[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        g[i_range] += 2.0 * a[j_range]
    return g

def adam_optimize_softmax(a0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max (log‑its) parametrisation.
    Returns the best sequence found (normalised to sum==1) and its
    associated objective value 2·n·max(conv).
    """
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)                # log‑its
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a0.copy()
    best_val = float("inf")
    start = time.time()
    it = 0
    # pre‑compute FFT length
    fft_len = next_pow2(2 * n)
    while True:
        it += 1
        now = time.time()
        if now - start >= time_limit:
            break

        # ---- soft‑max ----
        a = softmax(y)

        # ---- convolution ----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---- sub‑gradient of max(conv) ----
        eps_act = max(1e-12, 1e-9 * M)          # a tiny relative tolerance
        g = subgradient_max_conv(a, conv, eps_act)

        # ---- gradient w.r.t. logits ----
        ag = np.dot(a, g)                 # scalar ⟨a,g⟩
        grad_y = a * (g - ag)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / (np.sqrt(it) + 1.0)     # decay similar to the reference code
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter to escape plateaus ----
        if it % 3000 == 0:
            sigma = 0.02 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

def generate_two_sided_power_law(n: int, p_left: float, p_right: float,
                                 w_left: float = 1.0, w_right: float = 1.0) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right

convolution attains its maximum) is identified and a sub‑gradient of
the max‑norm is computed.  An Adam update is applied to the logits,
followed by a small Gaussian jitter every few thousand steps to escape
plateaus.

The routine first tries to reuse the previously discovered sequence
(`height_sequence_1`).  If that is unavailable it falls back to a
two‑sided power‑law of length ≈ 12000.  The bulk of the time budget is
spent on a long optimisation run from this start; if time permits a
short secondary run from a fresh random start is performed as a guard
against local minima.  The best sequence found is finally checked with
the official ``evaluate_sequence`` function and returned as a plain
Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_shift = y - np.max(y)
    exp_y = np.exp(y_shift)
    return exp_y / exp_y.sum()


def subgradient_max_conv(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv)`` w.r.t. ``a``.
    ``conv`` is the full linear convolution a*a (length 2n‑1).
    """
    n = a.shape[0]
    M = conv.max()
    active = np.where(conv >= M - eps)[0]
    g = np.zeros_like(a)

    # each active index contributes 2*a_{k-i} to the gradient (except on the
    # diagonal where the factor is 1, but the 2× formulation is a valid
    # sub‑gradient for all cases).
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        g[i_range] += 2.0 * a[j_range]
    return g


def adam_optimize_softmax(a0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max (log‑its) parametrisation.
    Returns the best sequence (normalised to sum = 1) and its objective
    value ``2·n·max(conv)``.
    """
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a0.copy()
    best_val = float("inf")
    start = time.time()
    it = 0
    fft_len = next_pow2(2 * n)

    while True:
        it += 1
        if time.time() - start > time_limit:
            break

        # ---------- soft‑max ----------
        a = softmax(y)

        # ---------- convolution ----------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---------- sub‑gradient of the max ----------
        # a tiny relative tolerance helps stabilise the active set
        eps_act = max(1e-12, 1e-9 * M)
        g = subgradient_max_conv(a, conv, eps_act)

        # ---------- gradient w.r.t. logits ----------
        ag = np.dot(a, g)                     # ⟨a,g⟩
        grad_y = a * (g - ag)                 # Jᵀ·g for soft‑max Jacobian

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / (np.sqrt(it) + 1.0)        # decaying step‑size
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- jitter ----------
        if it % 3000 == 0:                    # occasional random perturbation
            sigma = 0.02 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple shape generators (used for a secondary random start)
# ----------------------------------------------------------------------


def generate_two_sided_power_law(n: int,
                                 p_left: float,
                                 p_right: float,
                                 w_left: float = 1.0,
                                 w_right: float = 1.0) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a low
    ``evaluate_sequence`` value.  The routine respects the 1000 s time
    budget, prints progress information and returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # 1) Choose an initial shape.
    # --------------------------------------------------------------
    try:
        # Re‑use the previously discovered sequence if it exists.
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim != 1 or prev.size == 0:
            raise ValueError
        n = prev.size
        a0 = np.maximum(prev, 0.0)
        s = a0.sum()
        a0 = a0 / s if s >= 1e-12 else np.full(n, 1.0 / n)
        print(f"[info] Starting from previous sequence of length {n}.")
    except Exception:
        # Fallback: a symmetric two‑sided power‑law of moderate length.
        n = 12000
        a0 = generate_two_sided_power_law(n, 0.55, 0.55, 1.0, 1.0)
        a0 = a0 / a0.sum()
        print(f"[info] No previous sequence – using synthetic length {n}.")

    # --------------------------------------------------------------
    # 2) Main optimisation run (≈80 % of the remaining budget).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left <= 0:
        raise RuntimeError("Time exhausted before optimisation could start")
    main_time = max(0.0, 0.8 * time_left)
    print(f"[optim] Main run – length {n}, time budget {main_time:.1f}s")
    best_seq, best_val = adam_optimize_softmax(a0, main_time)
    print(f"[optim] Main run best value {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Optional secondary run from a fresh random start (if time permits).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        # random power‑law start
        p = random.uniform(0.2, 1.2)
        a1 = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        a1 = a1 / a1.sum()
        print(f"[optim] Secondary run – time budget {time_left:.1f}s")
        seq2, val2 = adam_optimize_softmax(a1, time_left)
        print(f"[optim] Secondary run best value {val2:.6f}")
        if val2 < best_val:
            best_val = val2
            best_seq = seq2
            print("[optim] New global best discovered in secondary run")

    # --------------------------------------------------------------
    # 4) Final verification with the official evaluator.
    # --------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # type: ignore[name-defined]
    print(f"[final] evaluate_sequence reports {final_score:.6f}")

    return final_list
