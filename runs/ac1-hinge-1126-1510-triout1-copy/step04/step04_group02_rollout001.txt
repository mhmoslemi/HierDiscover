"""
Hybrid search for a low‑score step‑function.
-------------------------------------------------
1. Choose a moderate length n (≈10 k) and work on the probability
   simplex (sum = 1) using a soft‑max parametrisation.
2. Multiple random restarts:
   * Initialise a pre‑soft‑max vector y (several heuristics).
   * Run a continuation L‑BFGS on a smoothed log‑sum‑exp
     approximation of the max‑convolution.  The smoothing factor
     (lam_factor) decays geometrically.
   * Refine the result with a few iterations of Adam (still on the
     smoothed objective) and keep the best candidate.
   * A short greedy hill‑climb cleans the current best.
3. Projected sub‑gradient descent on the *true* objective
   (max‑convolution) – this reduces the bound further.
4. Final polishing:
   * Simulated‑annealing hill‑climb that occasionally accepts
     worsening moves, allowing the algorithm to escape local minima.
   * Optional sparsification (zeroing tiny entries) and a tiny
     smoothing pass.
5. Return the best sequence (list of non‑negative floats).  The harness
   will evaluate it with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed to have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam_factor behaves like a smoothing parameter: lam = max(conv) *
    lam_factor + 1e‑12.
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z   # ∂ smooth_max / ∂ conv_i

    # Gradient wrt a:
    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)   # chain rule through soft‑max

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 200,
) -> np.ndarray:
    """
    Run a series of L‑BFGS optimisations with decreasing
    smoothing (lam_factor). Each stage starts from the previous
    result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    max_iters: int = 200,
    lr: float = 0.1,
) -> np.ndarray:
    """
    Adam optimiser on the smoothed objective.  The smoothing factor
    lam_factor is kept constant (usually the smallest value from the
    schedule).  Returns the final y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, max_iters + 1):
        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given heuristic mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon‑set (size ≈ √n)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Accept a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def hill_climb_sa(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.15,
    T_start: float = 0.02,
    T_end: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.  Worsening moves may be accepted
    with probability exp(-(Δ)/T), where T decreases linearly in time.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-9)

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        elapsed = time.time() - start_time
        # linear temperature schedule
        T = T_start + (T_end - T_start) * (elapsed / total_time)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)

        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    (2·n·max(conv) with sum = 1).  Keeps the best iterate seen.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def sparsify_and_normalize(a: np.ndarray, thresh: float = 1e-8) -> np.ndarray:
    """Zero out very small entries and renormalise to sum = 1."""
    a2 = a.copy()
    a2[a2 < thresh] = 0.0
    s = a2.sum()
    if s == 0.0:
        return a
    return a2 / s


def smooth_distribution(a: np.ndarray, window: int = 5) -> np.ndarray:
    """Simple moving‑average smoothing followed by re‑normalisation."""
    if window < 2:
        return a
    kernel = np.ones(window) / window
    a_s = np.convolve(a, kernel, mode="same")
    a_s = np.maximum(a_s, 0.0)
    s = a_s.sum()
    if s == 0.0:
        return a
    return a_s / s

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a few ten thousand entries give a rich search space.
    # --------------------------------------------------------------
    n = 10000  # default length; can be increased if time permits
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (uniform distribution on the simplex)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many stochastic L‑BFGS + Adam restarts
    # --------------------------------------------------------------
    phase1_deadline = deadline - 250.0   # reserve time for later phases
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=14))
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    iter_counter = 0

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=180,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iter_counter:04d}] mode={mode} L‑BFGS improved = {best_score:.6f}")

        # short Adam refinement using the smallest smoothing factor
        y_ref = adam_optimize(
            y_opt,
            n,
            fft_len_conv,
            fft_len_g,
            lam_factor=lam_schedule[-1],
            max_iters=150,
            lr=0.07,
        )
        a_ref = softmax(y_ref)
        score_ref = fast_score(a_ref, fft_len_conv, n)
        if score_ref < best_score:
            best_score = score_ref
            best_a = a_ref
            best_y = y_ref
            print(f"[{iter_counter:04d}] mode={mode} Adam refined = {best_score:.6f}")

        # quick greedy hill‑climb
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iter_counter += 1

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true objective
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 80.0
        print("[subgrad] starting projected sub‑gradient descent")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.18,
            max_iters=450_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final aggressive polishing (simulated‑annealing hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb_sa(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=15000,
            max_move=0.20,
            T_start=0.02,
            T_end=1e-5,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Optional sparsification & tiny smoothing
    # --------------------------------------------------------------
    best_a = sparsify_and_normalize(best_a, thresh=1e-7)
    best_a = smooth_distribution(best_a, window=3)
    best_a = sparsify_and_normalize(best_a, thresh=1e-9)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
