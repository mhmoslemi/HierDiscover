"""
Search for a low‑score non‑negative step sequence.

The algorithm combines three stages:
1️⃣ Multiple L‑BFGS restarts on a smoothed max‑convolution objective
   (log‑sum‑exp with a decreasing temperature schedule).
2️⃣ A fast incremental simulated‑annealing / hill‑climb that updates the
   convolution by hand (O(n) slice operations) – this allows millions of
   cheap moves and aggressively reduces the peak of the self‑convolution.
3️⃣ A short final greedy hill‑climb for polishing.

All vectors stay on the simplex (non‑negative, sum = 1) and the routine
keeps the best sequence found before the 1000 s deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int) -> float:
    """Score = 2·n·max(a*a) assuming sum(a)=1."""
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (smooth_max_objective, gradient) w.r.t. logits y.
    The objective approximates 2·n·max(a*a) where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # back‑prop through convolution:
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]  # ∂smooth_max/∂a_i

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS with continuation on the smoothing parameter
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from
    the result of the previous stage. Returns best sequence found.
    """
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:           # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc

# Random initialiser (more families for diversity)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'spikes'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spikes
        k = rng.integers(2, max(3, int(0.15 * n)))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = rng.random(k) + 0.1
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Fast incremental simulated‑annealing / hill‑climb
# ----------------------------------------------------------------------


def fast_local_search(a_start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_move_frac: float = 0.2):
    """
    Incremental SA that updates the convolution by hand.
    Moves a small fraction of mass from a source index to a target index.
    Accepts worsening moves with a temperature that decays over time.
    Returns the best sequence encountered.
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    conv = np.convolve(a, a)                     # full convolution (2n‑1)
    cur_max = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * cur_max

    T = 0.05                                     # initial temperature
    it = 0
    # schedule parameters
    dec_every = 4000
    while time.time() < deadline and it < 5_000_000:
        # pick a source proportionally to its mass (more likely to move large mass)
        i = rng.choice(n, p=a)
        j = rng.integers(0, n)
        if i == j:
            it += 1
            continue
        # choose a fraction of the source mass to move
        alpha = rng.random() * max_move_frac
        delta = a[i] * alpha
        if delta <= 0.0:
            it += 1
            continue

        # ----- compute candidate convolution -----
        cand = conv.copy()
        # linear part  -2*delta*a shifted by i , +2*delta*a shifted by j
        cand[i:i + n] -= 2.0 * delta * a
        cand[j:j + n] += 2.0 * delta * a
        # quadratic correction terms
        cand[2 * i] += delta * delta
        cand[2 * j] += delta * delta
        cand[i + j] -= 2.0 * delta * delta

        new_max = cand.max()
        # ----- acceptance test -----
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            conv = cand
            cur_max = new_max
            sc = 2.0 * n * cur_max
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
        # temperature schedule
        if it % dec_every == 0 and it > 0:
            T = max(T * 0.92, 1e-5)
            max_move_frac = max_move_frac * 0.97  # slowly shrink step size
        it += 1
    return best_a, best_sc


# ----------------------------------------------------------------------
# Final greedy hill‑climb (tiny moves, strict improvement)
# ----------------------------------------------------------------------


def final_hill_climb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iter: int = 3000):
    """Very small random transfers that are kept only if they improve the score."""
    rng = np.random.default_rng()
    a_cur = a.copy()
    conv = np.convolve(a_cur, a_cur)
    cur_max = conv.max()
    cur_sc = 2.0 * n * cur_max
    best_a, best_sc = a_cur.copy(), cur_sc

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move at most 2 % of the source mass
        delta = a_cur[i] * rng.random() * 0.02
        if delta <= 0.0:
            it += 1
            continue

        # incremental update
        conv[i:i + n] -= 2.0 * delta * a_cur
        conv[j:j + n] += 2.0 * delta * a_cur
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        new_max = conv.max()
        new_sc = 2.0 * n * new_max

        if new_sc < cur_sc:        # accept strictly improving moves
            a_cur[i] -= delta
            a_cur[j] += delta
            cur_max, cur_sc = new_max, new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_cur.copy()
        else:
            # revert the slice updates (undo the move)
            conv[i:i + n] += 2.0 * delta * a_cur
            conv[j:j + n] -= 2.0 * delta * a_cur
            conv[2 * i] -= delta * delta
            conv[2 * j] -= delta * delta
            conv[i + j] += 2.0 * delta * delta
        it += 1
    return best_a, best_sc

    a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)

    # pre‑compute FFT lengths
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_sc = fast_score(best_a, n)
    print(f"[init] n={n}, fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # Phase split (empirical)
    phase1_end = start_time + 0.40 * budget_s   # L‑BFGS restarts
    phase2_end = start_time + 0.70 * budget_s   # fast SA/local search
    # remaining time for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with a fine lam schedule
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.010,
                    0.006, 0.003, 0.0015, 0.0008]
    maxiter_per_lam = 700

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len_conv,
                                           fft_len_g,
                                           n)
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – fast incremental simulated annealing / hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sa_deadline = phase2_end - 0.3
        best_a, best_sc = fast_local_search(best_a, n, sa_deadline, max_move_frac=0.15)
        print(f"[fast SA] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb for polishing
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_sc = final_hill_climb(best_a, n, final_deadline, max_iter=8000)
    print(f"[final hill‑climb] best fast_score={best_sc:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
