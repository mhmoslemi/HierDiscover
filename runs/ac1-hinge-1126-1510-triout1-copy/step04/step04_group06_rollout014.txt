"""
Propose a near‑optimal non‑negative step sequence.
The method consists of:

  1. Several random restarts of an L‑BFGS optimiser on a smooth log‑sum‑exp
     surrogate of the max‑convolution (the “smoothed max”).  The smoothing
     parameter is decreased gradually; each run is warm‑started from a
     different simple distribution (uniform, edge‑heavy, exponential,
     power‑law or a perturbed previous best).

  2. A long projected sub‑gradient descent (with momentum) on the true
     max‑convolution.  This phase directly reduces the `max(conv)` term.

  3. An aggressive simulated‑annealing phase that makes non‑local moves
     (shifting a random fraction of the mass between two indices).

  4. A final greedy hill‑climb that swaps tiny amounts of mass between
     pairs of positions if it improves the fast (FFT‑based) score.

All phases respect the overall time budget (default 1000 s).  The best
sequence found is stored globally as ``height_sequence_1`` for possible
reuse and returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # keep only the needed part
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv) for a probability vector (sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max (log‑sum‑exp) objective and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained variables; a = softmax(y) lies on the simplex.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths for the convolution and the gradient convolution.
    lam_factor : float
        Controls smoothing; smaller → closer to true max.

    Returns
    -------
    f : float                — 2·n·smooth_max(conv)
    grad_y : np.ndarray      — gradient w.r.t. y
    """
    a = softmax(y)                     # map to simplex

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # --- smooth max via log‑sum‑exp ---
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12   # smoothing scale
    shifted = conv_full - max_c         # for numeric stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth max ---
    w = exp_shift / Z                    # ∂smooth_max/∂conv_k
    # compute g_i = Σ_k a_k * w_{i+k}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the central n entries (indices n‑1 … 2n‑2)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Sub‑gradient descent with momentum (true max)
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """
    Projected sub‑gradient descent (with momentum) on the true max‑convolution.
    Returns the best feasible vector found before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # current full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active‑set mask (indices attaining the max, tolerance for numerical noise)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a:
        # ∂max/∂a_i = 2 * Σ_j mask_{i+j} * a_j
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size ~ 1/√t
        step = 1.0 / np.sqrt(it + 1.0)
        a_new = a - step * v
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass swaps)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Local “swap” optimisation on the simplex.
    At each step a small fraction of mass is moved from i to j.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 4 % of source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local exploration)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Classic simulated‑annealing that relocates a random fraction of mass.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature: a few percent of the current score
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9994
    it = 0

    while time.time() < deadline:
        # pick a source index with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving up to 25 % of the source mass
        max_delta = cur_a[i] * 0.25
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta
        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert move
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 4000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)

        it += 1

    return best_a, best_score

# Helper to generate various initial distributions

def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n.
    Modes:
        'uniform' – plain uniform.
        'edge'    – heavier mass near the two ends.
        'exp'     – decaying exponential.
        'power'   – power‑law decay.
        'random'  – Dirichlet(0.5) random.
    """
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.002)          # small rate → slow decay
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.0)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)

    return a

    # Use the previous construction if it exists, otherwise start with length 2000.

        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        max_n = 4000
        if prev.size > max_n:
            prev = prev[:max_n]
        if prev.sum() > 0:
            a0 = prev / prev.sum()
            n = a0.size
        else:
            n = 2000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()                     # safety projection
    fft_len_conv = next_pow2(2 * n - 1)    # for a*a
    fft_len_g = next_pow2(3 * n - 2)       # for auxiliary convolutions

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts on the smoothed max.
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.006, 0.003, 0.001]
    lbfgs_counter = 0

    while time.time() < phase1_end:
        mode = rng.choice(['uniform', 'edge', 'exp', 'power', 'random'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        else:
            a_start = build_initial(mode, n, rng)
            y0 = np.log(a_start + 1e-12)

        lam = lam_schedule[lbfgs_counter % len(lam_schedule)]

        # L‑BFGS on the smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1500,
                                'gtol': 1e-12,
                                'disp': False})

        a_candidate = softmax(res.x)
        cur_score = fast_score(a_candidate, fft_len_conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            print(f"[L‑BFGS {lbfgs_counter}] mode={mode} lam={lam:.4f} score={best_score:.6f}")

        # A quick hill‑climb to polish the L‑BFGS output
        hill_deadline = min(phase1_end, time.time() + 0.12)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hill_deadline,
                                        max_iter=800)

        lbfgs_counter += 1

    # ------------------------------------------------------------------
    # Phase 2 – extended projected sub‑gradient descent on the true max.
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_momentum_optimize(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing (non‑local moves).
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (fine polishing).
    # ------------------------------------------------------------------
    final_deadline = deadline
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      final_deadline,
                                      max_iter=5000)

    # ------------------------------------------------------------------
    # Final evaluation (optional, just for logging)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # store globally for possible reuse in a later call
    globals()['height_sequence_1'] = final_a.tolist()

    return final_a.tolist()
