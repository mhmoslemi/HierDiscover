"""
Propose a low‑score non‑negative step sequence.

The search works in several stages (all respecting the simplex ∑a = 1):
1.  Lots of random restarts of a continuation L‑BFGS optimisation on a
    smoothed version of the max‑convolution (log‑sum‑exp).  The smoothing
    parameter is driven down to 5e‑4 so the final iterates see almost the
    true max.
2.  Projected sub‑gradient descent with momentum on the exact max‑convolution.
3.  Adam optimisation on the same smoothed objective (fine‑tuning).
4.  A directed hill‑climb that repeatedly moves a small fraction of mass
    from indices with the largest sub‑gradient to those with the smallest.
5.  A short simulated‑annealing pass to escape possible local minima.

All phases share a common fast convolution implementation based on FFTs,
so the algorithm easily handles n up to 4000 within the 1000 s budget.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) assuming ∑a = 1 (the denominator of evaluate_sequence)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient
# ----------------------------------------------------------------------


def smooth_obj_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max–convolution (log‑sum‑exp) and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑prop through convolution: dconv/da = (a_rev * mask) in frequency domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each smoothing λ in lam_schedule, starting each stage from
    the previous result.
    Returns the best (a, score) found during the continuation.
    """
    y = np.log(a_start + 1e-12)               # logits, always feasible
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=smooth_obj_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Occasionally numerical issues happen – just ignore this restart.
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum (exact max)
# ----------------------------------------------------------------------


def subgradient_momentum(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Momentum helps to smooth the noisy sub‑gradients.
    """
    a = a0.copy()
    best = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.85
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # binary mask of active constraints

        # sub‑gradient of the max (2·conv_g on the support side)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum
        v = beta * v + (1.0 - beta) * g

        # step size 1/√t
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        sc_candidate = fast_score(a_candidate, fft_len_conv, n)
        if sc_candidate < best_sc:
            best_sc = sc_candidate
            best = a_candidate.copy()

        a = a_candidate
        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective (fine‑tuning)
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.001,
                  lr: float = 0.13) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = smooth_obj_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Deterministic hill‑climb using sub‑gradient information
# ----------------------------------------------------------------------


def directed_hillclimb(a0: np.ndarray,
                       start_score: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int,
                       deadline: float,
                       max_iter: int = 12000) -> tuple[np.ndarray, float]:
    """
    Move mass from the index with largest sub‑gradient to that with smallest.
    Accept only improving moves; step size decays over time.
    """
    a = a0.copy()
    best = a.copy()
    best_sc = start_score

    it = 0
    while time.time() < deadline and it < max_iter:
        # exact max convolution and sub‑gradient
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        i_up = np.argmax(g)        # decreasing a_i reduces the max
        i_down = np.argmin(g)      # increasing a_j is safest

        if g[i_up] <= g[i_down] + 1e-14:
            # gradient already nearly flat – stop
            break

        # transfer a small fraction of mass from i_up to i_down
        frac = 0.08 * (0.99 ** (it // 400))   # decays slowly
        delta = min(frac * a[i_up], a[i_up] * 0.5)

        if delta <= 0:
            it += 1
            continue

        a[i_up] -= delta
        a[i_down] += delta

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        else:
            # revert – we only keep improving moves
            a[i_up] += delta
            a[i_down] -= delta

        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Simulated annealing (lightweight)
# ----------------------------------------------------------------------


def anneal(a0: np.ndarray,
           start_score: float,
           fft_len_conv: int,
           n: int,
           deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic annealing: random mass transfer between two coordinates,
    occasional uphill moves accepted with a decreasing temperature.
    """
    cur = a0.copy()
    cur_sc = start_score
    best = cur.copy()
    best_sc = start_score

    rng = np.random.default_rng()
    T = 0.03
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.12
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len_conv, n)
        if sc < cur_sc:
            cur, cur_sc = cand, sc
            if sc < best_sc:
                best, best_sc = cand.copy(), sc
        else:
            # metropolis criterion
            if rng.random() < np.exp(-(sc - cur_sc) / T):
                cur, cur_sc = cand, sc
        it += 1
        if it % 3000 == 0:
            T *= 0.88
            T = max(T, 1e-5)
    return best, best_sc

# Random initialiser (various families)

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'spike'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.015)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.1, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.005, 0.025)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.08 * n))
        peak = rng.uniform(0.6, 1.8) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:                # spike – a few big coordinates, rest tiny
        a = np.full(n, 1e-6)
        k = rng.integers(2, 7)          # number of spikes
        idx = rng.choice(n, k, replace=False)
        a[idx] = rng.random(k) * 0.4 + 0.3
        a /= a.sum()

    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

        if prev.ndim == 1 and prev.size > 0:
            # keep the sign‑preserved non‑negative part, pad with tiny mass if needed

                a0 = prev[:max_n].copy()
            else:
                a0 = np.concatenate([prev, np.full(max_n - prev.size, 1e-6)])
        else:
            a0 = np.full(max_n, 1.0 / max_n)
    else:
        a0 = np.full(max_n, 1.0 / max_n)

    # Normalise to the simplex
    a0 /= a0.sum()

    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a
    fft_len_g = next_pow2(3 * n - 2)      # length for gradient‑related FFTs

    best_a = a0.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    # --------------------------------------------------------------

    start_time = time.time()

    # Phase splits (empirically chosen)
