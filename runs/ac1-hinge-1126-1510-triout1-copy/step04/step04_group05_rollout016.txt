def proj_simplex(v):
    # ensure non-negative and sum=1
    # standard algorithm: sort v descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def softmax(y):
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def conv_fft(a, fft_len, n):
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[:2*n-1]
    return conv

def fast_score(a, fft_len, n):
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def proj_simplex(v):
    # projects onto the probability simplex (nonneg sum to 1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive: return uniform
        return np.full(n, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Sub‑gradient optimiser for the step‑function attack.

The algorithm works directly on the true objective
    f(a) = 2·n·max(a*a)   (with sum(a)=1)
by a projected sub‑gradient descent on the probability simplex.
Random restarts with diverse initialisations, a simple diminishing‑step
schedule and occasional jitter avoid stagnation.  After the descent a
short targeted hill‑climb and a generic random hill‑climb polish the
solution.  The method improves on the previous L‑BFGS based approach
and typically reaches a bound well below 1.51 within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself via FFT.
    Returns the first 2·n‑1 values (full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Quick evaluation of 2·n·max(a*a) when sum(a)=1.
    This matches the value returned by evaluate_sequence().
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Hill‑climb polishing utilities (re‑used from the previous code)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Greedy move from the largest to the smallest contribution at the peak."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Randomly moves a fraction of mass between two positions."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        max_iter -= 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.

    Returns
    -------
    list[float]
        The best non‑negative sequence discovered (its sum ≈ 1).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # leave a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size and initialisation
    # ------------------------------------------------------------------
    default_n = 2000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            n = default_n
            a0 = arr[:n].copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 1e-12:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}  init fast score = {best_score:.6f}")

    iteration = 0
    max_iters_per_restart = 2000  # projected sub‑gradient steps per restart

    # ------------------------------------------------------------------
    # Repeated random restarts + projected sub‑gradient descent
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # ---- choose a start distribution --------------------------------
        mode = rng.choice(["perturb", "random", "exp", "power", "edge"])
        if mode == "perturb":
            y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.1, size=n)
            a_start = softmax(y0)
        elif mode == "random":
            a_start = rng.random(n)
            a_start /= a_start.sum()
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a_vals = np.exp(-rate * np.arange(n))
            a_vals /= a_vals.sum()
            a_start = a_vals
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a_vals = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_vals /= a_vals.sum()
            a_start = a_vals
        else:  # edge‑biased
            edge_frac = rng.uniform(0.01, 0.04)
            m_edge = max(1, int(n * edge_frac))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a_vals = np.full(n, centre_val)
            a_vals[:m_edge] = edge_val
            a_vals[-m_edge:] = edge_val
            a_vals /= a_vals.sum()
            a_start = a_vals

        a = a_start.copy()
        cur_score = fast_score(a, fft_len, n)

        # start‑step magnitude (randomised per restart)
        step0 = rng.uniform(0.05, 0.4)

        # ---- projected sub‑gradient descent ----------------------------
        for t in range(max_iters_per_restart):
            if time.time() >= deadline:
                break

            # compute the sub‑gradient of the current peak
            conv = conv_fft(a, fft_len, n)
            k = int(np.argmax(conv))
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            g = np.zeros_like(a)
            g[mask] = 2.0 * a[idx[mask]]

            # diminishing step size
            eta = step0 / math.sqrt(t + 1)

            a_candidate = a - eta * g
            a_candidate = proj_simplex(a_candidate)

            new_score = fast_score(a_candidate, fft_len, n)
            if new_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a_candidate.copy()
                    print(f"[{iteration}] mode={mode} step={t}  best={best_score:.6f}")

            # tiny random jitter every ~64 steps to shake out of plateaus
            if (t & 63) == 0 and rng.random() < 0.05:
                i, j = rng.integers(0, n, size=2)
                delta = min(a[i], 0.001 * rng.random())
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = proj_simplex(a_tmp)
                score_tmp = fast_score(a_tmp, fft_len, n)
                if score_tmp < cur_score:
                    a = a_tmp
                    cur_score = score_tmp
                    if score_tmp < best_score - 1e-12:
                        best_score = score_tmp
                        best_a = a_tmp.copy()
                        print(f"[{iteration}] jitter improvement {best_score:.6f}")

        # ---- short polishing with the targeted hill‑climb -------------
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=2500,
            frac=0.07,
        )
        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing (targeted + random hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len, n, deadline, max_moves=8000, frac=0.06
        )
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len, n, deadline, max_iter=4000
        )

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
