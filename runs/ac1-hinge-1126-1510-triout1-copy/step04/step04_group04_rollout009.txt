def subgradient_optimize(a, n, fft_len, deadline, init_step=0.5, min_step=1e-8):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)  # length 2n-1
        s_star = int(np.argmax(conv))
        # compute gradient w.r.t max (subgradient)
        grad = np.zeros_like(best)
        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad[i[mask]] = 2 * best[j[mask]]
        # jitter
        grad += rng.normal(scale=1e-5, size=n)
        a_new = best - step * grad
        a_new = project_to_simplex(a_new)
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score

"""
Search for a non‑negative step‑function (vector) that minimizes
evaluate_sequence.  The algorithm combines

* many random initialisations and a smooth‑max L‑BFGS optimisation,
* aggressive projected sub‑gradient descent on the exact max‑convolution,
* a peak‑aware greedy hill‑climb,
* a final short L‑BFGS polishing stage.

All vectors are kept on the probability simplex (sum = 1) so the
scale‑invariance of the objective is respected.  The code adapts the
FFT length automatically and respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        # avoid division by zero – fall back to uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # ----- linear convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z   # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    # we need a reversed vector for cross‑correlation
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    A move is kept only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        else:
            a0[:] = 1.0 / n
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("project_to_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    Steps are projected back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)
        s_star = int(np.argmax(conv))

        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # add a tiny random jitter to avoid stalling on flat plateaus
        grad += rng.normal(scale=1e-5, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def peak_aware_greedy(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    move_fraction: float = 0.05,
    min_improve: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly target the current maximal convolution entry.
    Move a fraction of mass from the index that contributes most to the peak
    to the index that contributes least.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # gradient of the (active) max entry
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        donor = int(np.argmax(grad))
        recipient = int(np.argmin(grad))

        if grad[donor] <= 0:
            # no contributor – give up
            break

        delta = best[donor] * move_fraction
        if delta <= 0:
            break

        a_new = best.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta

        # enforce non‑negativity (should already hold)
        if a_new[donor] < 0:
            a_new[recipient] += a_new[donor]  # compensate
            a_new[donor] = 0.0

        new_score = fast_score(a_new, fft_len, n)
        if new_score + min_improve < best_score:
            best = a_new
            best_score = new_score
        else:
            # try a smaller move next round
            move_fraction *= 0.5
            if move_fraction < 1e-4:
                break

    return best, best_score

        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        else:
            a0[:] = 1.0 / n
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("project_to_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    Steps are projected back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)
        s_star = int(np.argmax(conv))

        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        grad += rng.normal(scale=1e-5, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def peak_aware_greedy(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    move_fraction: float = 0.05,
    min_improve: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Greedy moves that directly target the current maximal convolution entry.
    Move a fraction of mass from the index that contributes most to the peak
    to the index that contributes least.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        donor = int(np.argmax(grad))
        recipient = int(np.argmin(grad))

        if grad[donor] <= 0:
            break

        delta = best[donor] * move_fraction
        if delta <= 0:
            break

        a_new = best.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta

        if a_new[donor] < 0:
            a_new[recipient] += a_new[donor]
            a_new[donor] = 0.0

        new_score = fast_score(a_new, fft_len, n)
        if new_score + min_improve < best_score:
            best = a_new
            best_score = new_score
        else:
            move_fraction *= 0.5
            if move_fraction < 1e-4:
                break

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0) ready for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – reuse a previously discovered construction if possible
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1_000:
            n = min(max(5_000, int(tmp.size)), 8_000)   # cap to keep runtime reasonable
            best_a = tmp[:n].copy()
            best_a = np.clip(best_a, 0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            n = 6_000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 6_000
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.6, 5e-5, num=28))
    maxiter_stage = 500
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]

    # --------------------------------------------------------------
    # Main stochastic L‑BFGS + hill‑climb loop
    # --------------------------------------------------------------
    while time.time() < deadline - 60:   # keep about a minute for refinements
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast_score = {best_score:.6f}")

        # short local hill‑climb
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Extended sub‑gradient descent (uses exact max via FFT)
    # --------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 180.0)   # ~3 min
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.6
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Peak‑aware greedy polishing
    # --------------------------------------------------------------
    peak_deadline = min(deadline, time.time() + 30.0)
    best_a, best_score = peak_aware_greedy(
        best_a, n, fft_len_conv, peak_deadline, move_fraction=0.07
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Final fine L‑BFGS with very small smoothing
    # --------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 2e-5, num=20))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 600, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.4g} fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final greedy polishing (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
