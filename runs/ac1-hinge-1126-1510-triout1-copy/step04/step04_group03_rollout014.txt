def upscale_vector(v_small, n_big):
    n_small = v_small.shape[0]
    x_small = np.arange(n_small)
    x_big = np.linspace(0, n_small-1, n_big)
    v_big = np.interp(x_big, x_small, v_small)
    v_big = v_big / v_big.sum()
    return v_big

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    # Determine n
    max_n = 2000 # or maybe 2000 for speed
    n = max_n
    # Compute FFT lengths
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    # Initialize candidate list
    candidates = []
    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))
    # Parameteric seeds
    # Uniform
    add_candidate(np.full(n, 1.0 / n))
    # Linear decreasing
    add_candidate(np.arange(n, 0, -1))
    # Exponential
    for rate in [0.001, 0.003, 0.005, 0.01, 0.02]:
        add_candidate(np.exp(-rate * np.arange(n)))
    # Power law
    for p in [0.5, 0.8, 1.2, 2.0]:
        add_candidate((np.arange(1, n+1) ** (-p)))
    # Edge heavy (mass at ends)
    for edge_frac in [0.02, 0.05, 0.08, 0.10]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)
    # Edge profile: 1/(i+1) + 1/(n-i)
    for p in [0.5, 1.0]:
        i = np.arange(n)
        arr = (1.0/((i+1)**p) + 1.0/((n-i)**p))
        add_candidate(arr)
    # Sinusoidal squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-4)  # offset
    # Random Dirichlet with various concentrations
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(8):
            add_candidate(rng.gamma(alpha, 1.0, size=n))
    # Keep top seeds
    candidates.sort(key=lambda t:t[0])
    top_seeds = [arr for _,arr in candidates[:12]]

                                               fft_len_conv, fft_len_g,

"""
Improved step‑function optimizer.

The search works on a probability vector a (non‑negative, sum = 1).
The evaluation to minimise is 2·n·max(a*a).  The algorithm
combines:

  1.  A large parametric sweep (including many random Dirichlet draws)
      to obtain a short list of promising seeds.
  2.  Continuation L‑BFGS on a smoothed max (log‑sum‑exp) with a fine
      λ‑schedule and many restarts.
  3.  An extended projected sub‑gradient descent (exact max‑convolution
      as sub‑gradient) with a simple back‑tracking line‑search.
  4.  A short Adam‑optimisation (still on the smoothed max) to polish
      the result.
  5.  Randomised simulated annealing followed by a long greedy hill‑climb.

All steps keep the vector on the probability simplex (Euclidean projection).
The time budget (default 1000 s) is split automatically; the best sequence
found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1 (the true score is 2·n·max(a*a) / sum(a)², but the ratio is
    scale‑invariant, so we only need max(a*a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max (w.r.t. convolution entries)
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing λ schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best vector found."""
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Occasionally L‑BFGS may fail – ignore and continue.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # short hill‑climb after each λ can sharpen the result
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with back‑tracking line search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int,
                         step0: float = 0.5):
    """
    Directly minimise 2·n·max(a*a) via exact sub‑gradient descent.
    Uses a simple back‑tracking line‑search and Euclidean simplex projection.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set (indices attaining the max)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = step0 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.05,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8):
    """
    Adam optimisation in log‑it space, using the smoothed‑max objective.
    Keeps the best vector seen (according to the exact max‑convolution).
    """
    y = np.log(a0 + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        obj, grad = objective_and_grad(y, n,
                                       fft_len_conv, fft_len_g,
                                       lam_factor)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 10 % of the source mass
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing + random perturbations
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.05
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 20 % of the source mass
        delta = cur_a[i] * rng.random() * 0.20
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T = max(T * 0.90, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a Python list of floats that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we keep it moderate (2000) to obtain many iterations.
    # ------------------------------------------------------------------
    max_n = 2000
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)    # for a*a
    fft_len_g = next_pow2(3 * n - 2)       # for gradient / mask convolutions

    # ------------------------------------------------------------------
    # Collect a diverse set of seed vectors
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0.0:
            return
        arr = np.clip(arr, 0.0, None)
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # 1) Uniform (baseline)
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # 3) Exponential families
    for rate in [5e-4, 1e-3, 2e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # 4) Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Edge‑heavy (mass at both ends)
    for edge_frac in [0.01, 0.02, 0.04, 0.07]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # 6) Heavy‑edges with inverse‑polynomial decay
    for p in [0.5, 1.0]:
        i = np.arange(n)
        arr = 1.0 / ((i + 1) ** p) + 1.0 / ((n - i) ** p)
        add_candidate(arr)

    # 7) Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-4)

    # 8) Random Dirichlet draws (various concentrations)
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(6):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # If a global construction is present, add it (resized if needed)
    if 'height_sequence_1' in globals():
        try:
            old = np.array(globals()['height_sequence_1'], dtype=float)
            if old.ndim == 1 and old.size > 0:
                if old.size != n:
                    # linear interpolation to current size
                    x_old = np.linspace(0, old.size - 1, old.size)
                    x_new = np.linspace(0, old.size - 1, n)
                    resized = np.interp(x_new, x_old, old)
                    add_candidate(resized)
                else:
                    add_candidate(old)
        except Exception:
            pass

    # Keep a handful of the most promising seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _, arr in candidates[:12]]
    print(f"[sweep] best initial fast_score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    phase1_end = start_time + 0.48 * budget_s      # L‑BFGS stage (~48 %)
    phase2_end = start_time + 0.80 * budget_s      # longer sub‑gradient / Adam
    # remaining time will be used for annealing + hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (many restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.9, 0.7, 0.5, 0.4, 0.3, 0.2,
        0.15, 0.1, 0.07, 0.05, 0.03,
        0.02, 0.01, 0.005, 0.002, 0.001
    ]
    maxiter_per_lam = 500
    best_a = top_seeds[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        # pick a seed and add a tiny jitter
        seed_vec = top_seeds[rng.integers(0, len(top_seeds))]
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()
        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len_conv, fft_len_g, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f} (restart {restarts+1})")
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – extended sub‑gradient descent + Adam fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # sub‑gradient (exact max)
        sg_deadline = phase2_end - 0.2 * (budget_s - (phase2_end - start_time))
        best_a, best_score = subgradient_optimize(
            best_a, sg_deadline, fft_len_conv, fft_len_g, n, step0=0.4
        )
        print(f"[sub‑gradient] score after refinement = {best_score:.6f}")

        # Adam optimisation on a very small smoothing factor
        if time.time() < phase2_end:
            adam_deadline = phase2_end
            best_a, best_score = adam_optimize(
                best_a, adam_deadline, fft_len_conv, fft_len_g, n,
                lam_factor=0.001, lr=0.04
            )
            print(f"[Adam] score after fine‑tune = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing + final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_anneal(
            best_a, best_score, fft_len_conv, n, deadline
        )
        print(f"[anneal] score after anneal = {best_score:.6f}")

    # final hill‑climb (as much as time permits)
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a, best_score, fft_len_conv, n, final_deadline, max_iter=25000
    )

    # ------------------------------------------------------------------
    # Return the final sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
