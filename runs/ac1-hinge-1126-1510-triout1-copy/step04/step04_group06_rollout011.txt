sequence is always kept on the probability simplex, i.e. non‑negative
and summing to one – exactly the domain of ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a already normalised (sum = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective based on log‑sum‑exp and its gradient.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained variables; a = softmax(y) lies on the simplex.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths for the convolution and the gradient convolution.
    lam_factor : float
        Controls the smoothing; smaller → closer to the true max.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    # compute g_i = Σ_k a_k * w_{i+k}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (with momentum)
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """Momentum sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.85                    # slightly less damping than before
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active‑set mask (indices attaining the max, numerically tolerant)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient computation (see derivation in the original code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update and diminishing step size
        v = beta * v + (1.0 - beta) * g
        step = 0.8 / np.sqrt(it + 1.0)          # a bit larger than before
        a = simplex_projection(a - step * v)

        # keep the best ever seen
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000):
    """Local optimisation that repeatedly moves a tiny fraction of mass."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """Standard SA that relocates a random fraction of mass."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score

    # Initial temperature chosen as a few percent of the current score
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur[i] < 1e-16:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move up to 30 % of the source mass (larger jumps for SA)
        delta = cur[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert move
                cur[i] += delta
                cur[j] -= delta
                # temperature decay and continue
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # decay temperature
        T *= decay
        if it % 4000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP step from the paper)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray,
              fft_len: int,
              n: int,
              deadline: float,
              max_iters: int = 5):
    """
    Solve the LP:
        maximise Σ b_j
        subject to b_j ≥ 0
                  (a * b)_k ≤ M   for all k,
    where M = max(a*a).  Return the best refined vector found.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        # current bound
        conv_full = np.convolve(best, best)          # dense, length 2n‑1
        M = conv_full.max()

        # Build the constraint matrix A (size (2n‑1) × n)
        # Row k contains a_{k‑j} for each variable b_j, zero if out of range.
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = best                       # slice assignment is fast

        # Linear program (highs works well for modest sizes)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, (deadline - time.time()))})

        if not res.success:
            # If LP fails (rare), abort further refinements.
            break

        b = res.x
        # Normalise to sum = 1
        if b.sum() == 0:
            break
        b_norm = b / b.sum()

        # Evaluate the candidate
        cand_score = fast_score(b_norm, fft_len, n)
        if cand_score < best_score:
            best_score = cand_score
            best = b_norm.copy()
            # Print progress for the user
            print(f"[LP {it}] improved score to {best_score:.6f}")

        # Mix current best with the LP solution (convex combination)
        t = rng.random()
        mixed = (1 - t) * best + t * b_norm
        mixed = mixed / mixed.sum()
        mixed_score = fast_score(mixed, fft_len, n)
        if mixed_score < best_score:
            best_score = mixed_score
            best = mixed.copy()
            print(f"[LP {it}] mixed improvement to {best_score:.6f}")

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Construction of various initial probability vectors
# ----------------------------------------------------------------------


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector of length n according to the requested mode."""
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)          # fraction of indices at each edge
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a /= a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising ``evaluate_sequence``.
    The function respects the time budget and returns the best list of floats.
    """
    rng = np.random.default_rng(seed + 987654)

    # --------------------------------------------------------------
    # Choose problem size n and initialise a starting point.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        max_n = 3500
        if prev.size > max_n:
            prev = prev[:max_n]
        if prev.sum() > 0:
            a0 = prev / prev.sum()
            n = a0.size
        else:
            n = 2500
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2500
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()                       # safety normalisation

    # --------------------------------------------------------------
    # FFT lengths (powers of two for speed)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS on a smooth log‑sum‑exp surrogate.
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.006, 0.003, 0.001]
    lbfgs_iter = 0

    while time.time() < phase1_end:
        mode = rng.choice(['uniform', 'edge', 'exp', 'power', 'random'])
        a_start = build_initial(mode, n, rng)
        y0 = np.log(a_start + 1e-12)

        lam = lam_schedule[lbfgs_iter % len(lam_schedule)]

        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'gtol': 1e-12,
                                'disp': False})

        a_candidate = softmax(res.x)
        cand_score = fast_score(a_candidate, fft_len_conv, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[L‑BFGS {lbfgs_iter}] mode={mode} lam={lam:.4f} score={best_score:.6f}")

        # brief hill‑climb to polish this candidate
        hill_deadline = min(phase1_end, time.time() + 0.12)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hill_deadline,
                                        max_iter=800)

        lbfgs_iter += 1

    # --------------------------------------------------------------
    # Phase 2 – long momentum sub‑gradient descent on the true max.
    # --------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_momentum_optimize(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – linear‑programming refinement (LP step).
    # --------------------------------------------------------------
    phase3_end = start_time + 0.85 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = lp_refine(best_a,
                                       fft_len_conv,
                                       n,
                                       phase3_end,
                                       max_iters=6)
        print(f"[LP refinement] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing for non‑local exploration.
    # --------------------------------------------------------------
    phase4_end = deadline
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase4_end,
                                                 rng)
        print(f"[SA] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb polish (more iterations).
    # --------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=6000)

    # --------------------------------------------------------------
    # Final evaluation (optional, just for reporting).
    # --------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # store globally for later runs
    globals()['height_sequence_1'] = final_a.tolist()

    return final_a.tolist()
