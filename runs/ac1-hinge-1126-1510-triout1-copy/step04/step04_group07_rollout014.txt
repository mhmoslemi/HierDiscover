def smoothed_adam_optimize(a0, time_limit, start_time, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_seq = a0.copy()
    best_val = fast_score(best_seq)
    it = 0
    while True:
        now = time.time()
        if now > start_time + time_limit:
            break
        it += 1
        # softmax
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()
        # convolution a*a
        fft_len = next_pow2(2*n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
        M = conv.max()
        # compute smoothed weights
        tau = min(200.0, 0.5 * it)  # increase tau over time
        # avoid overflow: subtract M
        w = np.exp(tau * (conv - M))
        w_sum = w.sum()
        w /= w_sum
        # compute gradient of smooth max: g = 2 * (a_rev convolve w)
        a_rev = a[::-1]
        # convolution a_rev * w (full)
        L = next_pow2(len(a_rev) + len(w) - 1)
        f_rev = np.fft.rfft(a_rev, n=L)
        fw = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(f_rev * fw, n=L)[:len(a_rev)+len(w)-1]
        # gradient entries for i in 0..n-1
        # Grad_i = 2 * grad_full[i + (len(w) - 1)//? Actually w length = 2n-1.
        # w length = 2n-1. So offset = (len(w)-1) // 2 = n-1.
        offset = n - 1
        g = 2.0 * grad_full[offset: offset + n]
        # gradient wrt logits
        ag = np.dot(a, g)
        grad_y = a * (g - ag)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_y
        v = beta2 * v + (1-beta2) * (grad_y**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # track best using true score
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # occasional jitter
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)
    return best_seq, best_val

"""
Improved step‑function search.

The evaluation metric is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where *n* is the sequence length.  Since the score is scale‑invariant we
optimise over the probability simplex (∑aᵢ = 1, aᵢ ≥ 0).

The core optimiser uses a *smoothed* version of the non‑smooth max:
    Lτ(a) = (1/τ)·log ∑ₖ exp(τ·convₖ(a))
which converges to the true max as τ→∞.  The gradient of Lτ is the
weighted sum of the gradients of each convolution entry,
g = 2·(a_rev ∗ w) where
    wₖ ∝ exp(τ·(convₖ – max(conv)).
Thanks to the convolution structure both the convolution and its
gradient are obtained with two FFTs per iteration, making the method
fast enough to run millions of updates in the 1000 s budget.

A two‑stage scheme is employed:
  * a few diverse initial shapes are short‑optimised (≈30 s each);
  * the best candidate is then refined for the rest of the time
    with a gradually increasing τ and an Adam‑style step‑size schedule.

The final sequence is returned as a plain Python list.  All numbers are
forced to be non‑negative by the soft‑max parametrisation.
"""

import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Faster version of evaluate_sequence (no safety checks).
    Computes 2 * n * max(conv(seq, seq)) / (sum(seq)**2).
    seq must be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    L = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (used for random starts)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------


def smoothed_adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex with a smoothed max objective.
    Returns the best sequence seen (as a probability vector) and its
    fast_score.
    """
    n = a0.shape[0]

    # logits for the soft‑max parametrisation
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ----- soft‑max -----
        y_shift = y - np.max(y)          # stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()          # always lies on the simplex

        # ----- convolution a * a -----
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())

        # ----- smoothed weighting -----
        # τ grows slowly so that early iterations see a smooth landscape
        tau = min(200.0, 0.5 * it)       # can be tuned
        w = np.exp(tau * (conv - M))    # max → 1, others ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:                # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth objective -----
        a_rev = a[::-1]
        # convolution a_rev * w  (full length = n + (2n‑1) – 1 = 3n‑2)
        L = next_pow2(len(a_rev) + len(w) - 1)
        f_rev = np.fft.rfft(a_rev, n=L)
        f_w = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(f_rev * f_w, n=L)[: len(a_rev) + len(w) - 1]

        # offset = n‑1 aligns the result with the original index order
        offset = n - 1
        g = 2.0 * grad_full[offset : offset + n]   # shape (n,)

        # ----- back‑prop through the soft‑max -----
        ag = np.dot(a, g)            # scalar ⟨a,g⟩
        grad_y = a * (g - ag)        # Jacobian of soft‑max applied

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep best (measured with the true max) -----
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional jitter to escape tiny plateaus
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the 1000 s time budget, prints progress, and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 0) initialise a small pool of plausible start points
    # --------------------------------------------------------------
    init_candidates: list[np.ndarray] = []

    # Use the previously discovered sequence if it exists

        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() < 1e-12:
                prev = np.full(prev.shape, 1.0 / prev.shape[0])
            else:
                prev = prev / prev.sum()
            init_candidates.append(prev.copy())
            print("[info] previous sequence incorporated as start")
    except Exception:
        pass

    # Default deterministic start – a symmetric two‑sided power law
    default_len = 12000
    default_seq = generate_two_sided_power_law(default_len, 0.6, 0.6, 1.0, 1.0)
    default_seq = np.maximum(default_seq, 0.0)
    default_seq = default_seq / default_seq.sum()
    init_candidates.append(default_seq.copy())

    # A few random families (to increase diversity)
    families = ["power", "exp", "twosided", "beta", "u_shape"]
    desired_pool = 6

    while len(init_candidates) < desired_pool and time.time() < deadline:
        n = random.choice([8000, 10000, 13000, 15000])
        fam = random.choice(families)

        if fam == "power":
            p = random.uniform(0.05, 1.2)
            seq = generate_power_law(n, p)
        elif fam == "exp":
            lam = random.uniform(1e-4, 4e-2)
            seq = generate_exponential(n, lam)
        elif fam == "twosided":
            pl = random.uniform(0.1, 1.4)
            pr = random.uniform(0.1, 1.4)
            wl = random.uniform(0.5, 3.0)
            wr = random.uniform(0.5, 3.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        elif fam == "beta":
            a = random.uniform(0.1, 2.0)
            b = random.uniform(0.1, 2.0)
            seq = generate_beta_shape(n, a, b)
        else:  # u_shape – very high mass at the edges, tiny in the middle
            p_edge = random.uniform(1.5, 3.0)
            centre = np.full(n, 0.0)
            # assign a power‑law on each side, but very steep
            seq = generate_two_sided_power_law(n, p_edge, p_edge, 1.0, 1.0)
            # reduce the central part a bit more
            mask = np.logical_and(np.arange(n) > n // 4, np.arange(n) < 3 * n // 4)
            seq[mask] *= 0.2

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        init_candidates.append(seq)

    print(f"[info] {len(init_candidates)} initial candidates generated")

    # --------------------------------------------------------------
    # 1) short optimisation of all candidates (≈30 s each)
    # --------------------------------------------------------------
    short_budget = min(30.0, (deadline - time.time()) / max(len(init_candidates), 1))
    best_overall_seq: np.ndarray | None = None
    best_overall_val: float = float("inf")

    for idx, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        t_limit = min(short_budget, remaining)
        if t_limit <= 0.0:
            break
        print(f"[phase‑1] optimise candidate #{idx+1}/{len(init_candidates)} "
              f"(len={a0.size}) – budget {t_limit:.1f}s")
        seq, val = smoothed_adam_optimize(a0, t_limit, time.time())
        print(f"[phase‑1] candidate #{idx+1} best fast_score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # --------------------------------------------------------------
    # 2) long refinement of the current best
    # --------------------------------------------------------------
    if best_overall_seq is None:
        # fallback: uniform distribution
        n_fallback = 10000
        best_overall_seq = np.full(n_fallback, 1.0 / n_fallback)

    remaining = deadline - time.time()
    if remaining > 20.0:
        print(f"[phase‑2] long refinement on best candidate – budget {remaining:.1f}s")
        seq, val = smoothed_adam_optimize(best_overall_seq, remaining, time.time())
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"[phase‑2] improved best fast_score to {best_overall_val:.6f}")

    # --------------------------------------------------------------
    # 3) final verification with the official evaluator
    # --------------------------------------------------------------
    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"[final] evaluate_sequence reports {final_score:.6f}")

    return final_list
