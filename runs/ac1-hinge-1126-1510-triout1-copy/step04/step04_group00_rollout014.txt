def max_conv_gradient(a):
    # compute convolution c = a * a
    n = a.shape[0]
    fft_len = next_pow2(2*n-1)
    Fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(Fa * Fa))[:2*n-1]
    max_c = conv.max()
    # local tolerance: choose epsilon = max_c * 1e-12
    eps = max_c * 1e-12
    mask = (conv >= max_c - eps).astype(float)   # may have multiple entries
    # compute gradient g = 2 * conv(a, mask_rev)
    fft_len_g = next_pow2(3*n-2)    # length >= n + (2n-1) -1 = 3n-2
    Fa_rev = np.fft.fft(a[::-1], fft_len_g)
    Fw = np.fft.fft(mask, fft_len_g)
    conv_rev = np.real(np.fft.ifft(Fa_rev * Fw))
    g = 2.0 * conv_rev[(n-1):(2*n-1)]
    return max_c, g

   sub‑gradient of the nonsmooth objective
        f(a) = 2·n·max( a*a )
   (the factor 2·n is the constant scaling used by ``evaluate_sequence``.
   The sub‑gradient is obtained by a second FFT‑based convolution.
   The iterate is projected back onto the simplex (non‑negative, sum = 1).

   A diminishing step size  ηₜ = η₀ / sqrt(t)  with a normalisation by the
   gradient norm gives stable progress.  The mask that defines the
   sub‑gradient is gradually refined: we start with a relatively large
   tolerance (top few percent of the convolution) and shrink it as the
   optimisation proceeds.

3. **Aggressive hill‑climb** – a short greedy random‑mass‑transfer hill‑climb
   (the same routine used by the previous state‑of‑the‑art) is applied
   after each descent phase to capture any remaining discrete improvement.

4. **Multiple restarts** – the whole descent+hill‑climb loop is executed
   repeatedly until the time budget is exhausted, each time from a fresh
   random start (or from the current best).  The best sequence seen so far
   is returned.

The method spends most of the allotted time on cheap O(n log n) FFT
operations and avoids the heavyweight L‑BFGS schedule of the reference
implementation, enabling many more gradient steps and typically pushing
the bound below the previous record (≈ 1.509) within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (∑a=1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy random‑mass‑transfer hill‑climb on the simplex.
    Strictly improves the fast score (2·n·max) and stops when time runs out.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgrad_max_convolution(
    a: np.ndarray,
    mask_frac: float,
    fft_len_conv: int,
    fft_len_grad: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximum of the auto‑convolution and its exact sub‑gradient.
    ``mask_frac`` determines the tolerance for the active set:
    all indices with value ≥ (1‑mask_frac)·max are treated as active.
    Returns (max_c, grad) where grad is a vector of length n.
    """
    n = a.shape[0]

    # ---- convolution a*a ------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()

    # ---- active mask (indices close to the maximum) --------------------
    if mask_frac <= 0.0:
        # exact max only
        mask = (conv == max_c).astype(float)
    else:
        thresh = max_c * (1.0 - mask_frac)
        mask = (conv >= thresh).astype(float)

    # ---- gradient of the (smooth) max -----------------------------------
    # g_i = 2 * sum_{k} a_k * mask_{i+k}
    fa_rev = np.fft.fft(a[::-1], fft_len_grad)
    fw = np.fft.fft(mask, fft_len_grad)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad = 2.0 * conv_rev[(n - 1) : (2 * n - 1)]

    return float(max_c), grad.astype(np.float64)


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Seconds available for the search (the function tries to stop a bit
        before the deadline).

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size – use a previously discovered construction if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 500:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # ensure a genuine probability vector
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    # initial score (exact value 2·n·max)
    fft_len_conv = next_pow2(2 * n - 1)
    max_c = np.real(np.fft.ifft(np.fft.fft(best_a, fft_len_conv) ** 2))[
        : 2 * n - 1
    ].max()
    best_score = 2.0 * n * max_c

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (used throughout)
    # ------------------------------------------------------------------
    fft_len_grad = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Main optimisation loop – many cheap projected sub‑gradient steps
    # ------------------------------------------------------------------
    restart_counter = 0
    global_iter = 0

    while time.time() < deadline - 5.0:
        restart_counter += 1

        # ----- initialise a new start point --------------------------------
        if restart_counter % 5 == 0:
            # fresh random start (Dirichlet like)
            a = rng.random(n)
            a = proj_simplex(a)
        else:
            # continue from the best found so far
            a = best_a.copy()

        # ----- sub‑gradient descent ----------------------------------------
        local_iters = 0
        # parameters for the schedule
        eta0 = 2.0
        while time.time() < deadline - 4.0 and local_iters < 8000:
            local_iters += 1
            global_iter += 1

            # gradually shrink the active mask
            mask_frac = max(0.02 / np.sqrt(local_iters), 0.0005)

            max_c, grad = subgrad_max_convolution(
                a, mask_frac, fft_len_conv, fft_len_grad
            )
            cur_score = 2.0 * n * max_c

            # keep the best ever seen
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[iter {global_iter}] new best score = {best_score:.6f}")

            # normalized step
            step = eta0 / np.sqrt(global_iter) / (np.linalg.norm(grad) + 1e-12)
            a = proj_simplex(a - step * grad)

        # ----- aggressive hill‑climb --------------------------------------
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=6000,
            max_move_frac=0.20,
        )
        print(f"[post‑hc {restart_counter}] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb with any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=15000,
            max_move_frac=0.20,
        )
        print(f"[final‑hc] polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
