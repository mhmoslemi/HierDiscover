"""
Propose a low‑score step‑function sequence.

The algorithm works in three stages:

1. **Global stochastic L‑BFGS** – a few multi‑restart L‑BFGS runs on the
   soft‑max parametrisation of the sequence.  The smooth‑max
   approximation of the convolution peak is gradually sharpened by a
   decreasing λ‑schedule.

2. **Exact sub‑gradient descent on the true max** – after the global
   stage we minimize the true objective
   `max_k (a*a)[k]` on the probability simplex using the exact
   sub‑gradient (the sum of the shifts that realise the current peak).
   This removes the bias introduced by the smooth approximation.

3. **Aggressive hill‑climb** – a final random‑mass‑transfer hill‑climb
   finishes the optimisation.

All operations are performed with FFTs for speed; the whole routine
stops before the time budget expires and returns the best sequence it
found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum == 1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (log‑soft‑max coordinates).
    """
    a = softmax(y)

    # ----- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max ----------
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max (gradient w.r.t. y)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        # L‑BFGS on the smooth objective
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:   # strict improvement
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_optimize(
    a_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    Uses the exact sub‑gradient (sum of the shifted vectors that achieve
    the current maximum) and projects onto the probability simplex.
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    step0 = 0.6
    it = 0
    no_improve = 0

    while time.time() < deadline - 0.5:
        it += 1
        # ----- convolution for the current a -----
        fa = np.fft.fft(a, fft_len_conv)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv_full.max()

        # indices where the maximum is attained (allow tiny tolerance)
        eps = max_val * 1e-12 + 1e-15
        max_idx = np.where(conv_full >= max_val - eps)[0]

        # limit the number of active indices to keep the gradient cheap
        if max_idx.size > 80:
            max_idx = rng.choice(max_idx, size=80, replace=False)

        # indicator vector for the active (max) positions
        indicator = np.zeros(2 * n - 1, dtype=float)
        indicator[max_idx] = 1.0

        # ----- exact sub‑gradient of max(conv) -----
        fa_rev = np.fft.fft(a[::-1], fft_len_g)
        fw = np.fft.fft(indicator, fft_len_g)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        grad_raw = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * grad_raw   # derivative of max(conv) w.r.t a

        # ----- projected step -----
        step = step0 / np.sqrt(it)
        cand = proj_simplex(a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - 1e-12:
            best_a = cand.copy()
            best_score = cand_score
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasional random perturbation to escape plateaus
            if no_improve % 200 == 0:
                a = proj_simplex(best_a + 0.02 * rng.normal(size=n))

    return best_a, best_score
