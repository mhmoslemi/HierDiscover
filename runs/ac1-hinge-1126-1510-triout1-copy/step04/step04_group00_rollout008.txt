def subgradient_descent(a, n, fft_len, deadline, max_iter=2000):

    best_score = fast_score(best_a, fft_len, n)

        conv = np.real(np.fft.ifft(np.fft.fft(best_a, fft_len) * np.fft.fft(best_a, fft_len)))[: 2 * n - 1]

def annealing_move(a, cur_score, fft_len, n, start_time, deadline, init_temp=0.001, decay=0.999, max_iter=20000):
    best = a.copy()
    best_score = cur_score
    T = init_temp
    while time.time() < deadline and max_iter>0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.2  # max 20% of source mass
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep
        else:
            # accept w prob
            if rng.random() < np.exp(-(new_score - best_score) / T):
                best_score = new_score  # accept worse
            else:
                # revert
                best[i] += delta
                best[j] -= delta
        T *= decay
        max_iter -= 1
    return best, best_score

"""
Propose a low‑score step‐function sequence.

The optimizer works in several layers:

1. **Initial guess** – an arcsine‐like shape (high mass at the ends,
   low in the centre) or a random perturbation thereof.
2. **Continuation L‑BFGS** – minimise a smooth approximation of the
   infinity‑norm of the auto‑convolution using a geometric λ‑schedule.
3. **Exact sub‑gradient descent** – after the smooth phase we
   minimise the true max‑convolution using the exact sub‑gradient
   (the gradient of the active convolution entries) together with a
   Euclidean projection onto the probability simplex.
4. **Aggressive hill‑climb / simulated annealing** – random mass
   transfers (up to 20 % of the source) are accepted if they improve
   the score or, with a slowly cooling temperature, if they are
   slightly worse.  This helps to escape shallow local minima.
5. **Restart loop** – the whole pipeline is executed several times
   (different seeds, different initial families) until the time budget
   is exhausted.  The best sequence found is returned.

The whole routine respects the ``budget_s`` argument (default 1000 s)
and never exceeds it.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of   2·n·max(a*a)   for a probability vector (Σa=1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution a*a and its gradient w.r.t. y
    (the unconstrained log‑soft‑max coordinates).
    """
    a = softmax(y)

    # --- a*a -----------------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of the smooth max ------------------------------------
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max (gradient w.r.t. y)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int,
    deadline: float,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    The function aborts early if the deadline is reached.
    """
    y = y_start.copy()
    for lam in lam_factors:
        if time.time() > deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def arcsine_shape(n: int) -> np.ndarray:
    """Arcsine‑like (Beta(½,½)) shape, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = np.clip(a, 1e-12, None)
    a /= a.sum()
    return a


def grad_max_exact(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of   max_k (a*a)_k   w.r.t. a.
    If several k attain the maximum, their gradients are averaged.
    """
    max_c = conv.max()
    idx = np.where(conv >= max_c - eps)[0]   # possibly multiple argmaxes
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        # vector of a[k - i] for i = i_start … i_end
        g[i_start : i_end + 1] += a[k - np.arange(i_start, i_end + 1)]
    return 2.0 * g


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector encountered and its fast score.
    """
    a = a0.copy()
    best_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_iter:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        # exact sub‑gradient
        g = grad_max_exact(a, conv, eps=1e-12)
        # diminishing step size
        step = 0.8 / np.sqrt(it + 1)
        cand = proj_simplex(a - step * g)
        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
        else:
            # still move a a little to escape plateaus
            a = cand
        it += 1
    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
    rng: np.random.Generator = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that randomly transfers a fraction of the mass from
    one index to another and keeps any strict improvement.
    """
    if rng is None:
        rng = np.random.default_rng()
    best = a.copy()
    best_score = cur_score
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep the move
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_temp: float = 5e-4,
    decay: float = 0.9995,
    max_iter: int = 15000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with a Metropolis acceptance rule.
    The temperature decays exponentially.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score_local = cur_score
    T = init_temp
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        accept = False
        if new_score < cur_score_local:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score_local) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score_local = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert the move
            cur[i] += delta
            cur[j] -= delta

        T *= decay
        it += 1
    return best, best_score

# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    # Larger n gives more freedom but slows each iteration.
    # Adapt n to the amount of time we seem to have.
    remaining = deadline - time.time()
    if remaining > 600:
        n = 8000
    else:
        n = 5000

    # ------------------------------------------------------------------
    # Initialise from a previous run if available, otherwise use the
    # arcsine shape (high at the boundaries, low in the middle).
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= n:
            best_a = arr[:n].copy()
        else:
            best_a = arcsine_shape(n)
    else:
        best_a = arcsine_shape(n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # L‑BFGS continuation schedule (global search)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=35)   # finer schedule
    maxiter_stage = 800                           # more work per stage

    iteration = 0
    while time.time() < deadline - 10.0:
        # ------------------------------------------------------------------
        # Choose a starting point for this restart.
        # ------------------------------------------------------------------
        mode = rng.choice(["arcsine", "exp", "power", "random", "perturb"])
        if mode == "arcsine":
            a0 = arcsine_shape(n)
            y0 = np.log(a0 + 1e-12)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.03)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.5)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "random":
            y0 = rng.standard_normal(n)
        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
            deadline=deadline - 5.0,   # a little slack for later phases
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}"
            )

        # ----- exact sub‑gradient descent (max‑convolution) -----
        sub_deadline = min(deadline - 2.0, time.time() + 2.5)
        best_a, best_score = subgradient_descent(
            best_a,
            n=n,
            fft_len=fft_len_conv,
            deadline=sub_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- aggressive hill‑climb -----
        hc_deadline = min(deadline - 1.0, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=6000,
            max_move_frac=0.20,
            rng=rng,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- simulated annealing polishing -----
        anneal_deadline = deadline - 0.5
        if time.time() < anneal_deadline:
            best_a, best_score = simulated_annealing(
                a=best_a,
                cur_score=best_score,
                fft_len=fft_len_conv,
                n=n,
                deadline=anneal_deadline,
                rng=rng,
                init_temp=6e-4,
                decay=0.9993,
                max_iter=12000,
                max_move_frac=0.20,
            )
            best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.15,
            rng=rng,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}  (fast score = {best_score:.6f})")
    # expose the best sequence globally for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
