"""
Improved step‑function search for the min–max convolution problem.

Strategy
--------
1.  Construct a diverse set of seed distributions (uniform, linear,
    exponential, power‑law, edge‑heavy, sinusoidal, random Dirichlet,
    sparse binary “sets”).  Keep the few best.
2.  For each seed run a continuation L‑BFGS on the log‑sum‑exp
    smooth‑max of the convolution.  The λ‑schedule is longer and goes
    down to 1e‑4·max_c, and each stage is allowed more iterations.
    The variables are the soft‑max parameters y, guaranteeing a
    probability simplex (∑a=1, a_i≥0) at all times.
3.  Once the smoothed problem is exhausted, switch to an exact
    projected sub‑gradient descent with a back‑tracking line‑search.
    This phase is run for the bulk of the remaining time.
4.  A long simulated‑annealing phase (random mass transfers with
    Metropolis acceptance) tries to escape any remaining local minima.
5.  Finally a massive greedy hill‑climb (mass‑transfer moves that are
    accepted only if they improve the true fast‑score) polishes the
    result.

All steps use FFT‑based convolutions (O(n log n)).  The total time budget
is split dynamically; the best sequence found before the deadline is
returned.  The returned list respects the constraints of
evaluate_sequence() (non‑negative, sum>0.01, each entry ≤1000).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a distribution with ∑a=1 (scale‑invariant)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) where the objective is
    2·n·smooth_max(conv) using log‑sum‑exp smoothing.
    """
    a = softmax(y)                     # a ∈ simplex
    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth max -----
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. a -----
    w = exp_shift / Z                                 # ∂smooth_max/∂conv_s
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # Only the n entries that correspond to linear convolution indices
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(a*a) by exact sub‑gradient descent with a simple
    back‑tracking line search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # exact max convolution and active mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max (same as in the original code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step‑size with back‑tracking
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a small random fraction of mass from
    one coordinate to another and keeps only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05  # up to 5 % of mass at i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS with a fine λ‑schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, using the result of the
    previous stage as the starting point.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()

        # short hill‑climb after each λ to sharpen
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

    return best_a, best_score

# Simulated annealing / random perturbations

def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.025
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # up to 12 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T = max(T * 0.93, 1e-5)

    return best_a, best_score

        add_candidate(np.exp(-rate * np.arange(n)))

    # 4) Power‑law families
    for p in [0.2, 0.5, 0.8, 1.2, 1.7, 2.2]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Edge‑heavy (mass at both ends)
    for edge_frac in [0.008, 0.015, 0.025]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # 6) Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # 7) Random Dirichlet draws
    for alpha in [0.4, 0.8, 1.5, 3.0]:
        for _ in range(3):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # 8) Sparse binary “set” constructions
    for K in [300, 500, 800]:
        mask = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=K, replace=False)
        mask[idx] = 1.0
        add_candidate(mask)   # equal weight on the chosen positions

    # keep the top seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    phase1_end = start_time + 0.55 * budget_s   # ~55 % for L‑BFGS
    phase2_end = start_time + 0.80 * budget_s   # ~25 % for sub‑gradient
    phase3_end = deadline                        # remainder for annealing

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (many restarts, finer λ schedule)
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.011, 0.007,
                    0.004, 0.0025, 0.0015, 0.001, 0.0005,
                    0.0002, 0.0001]          # down to 1e‑4·max_c
    maxiter_per_lam = 350                  # a bit longer per λ
    restarts = 0

    while time.time() < phase1_end and restarts < 15:
        # pick a good seed and jitter
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 2e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS restart {restarts}] new best score={best_score:.6f}")
        restarts += 1

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.4)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing + greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        # annealing
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase3_end)
        print(f"[anneal] score={best_score:.6f}")

        # final long hill‑climb (use as much remaining time as possible)
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=20000)

    # --------------------------------------------------------------
    # Return the sequence (as a plain Python list)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external provided function
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
