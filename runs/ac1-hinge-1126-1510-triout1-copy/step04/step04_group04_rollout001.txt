"""
Propose a low‑score step‑function using a multi‑stage optimizer:

1️⃣  Several random restarts on the probability simplex.  Each restart
    runs a cascade of L‑BFGS optimisations on a smooth log‑sum‑exp
    approximation of the max‑convolution (the “lam” parameter is
    decreased geometrically).

2️⃣  A few passes of Adam (first‑order momentum) on the same smooth
    objective, again with a decreasing lam schedule.

3️⃣  Exact sub‑gradient descent on the true max‑convolution, with
    projection onto the simplex and an adaptive step‑size.

4️⃣  A refined hill‑climb that moves a very small amount of mass from
    the currently largest coefficient to a random location, accepting
    only moves that lower the true objective (fast evaluation via FFT).

The whole pipeline respects the constraints of ``evaluate_sequence``:
all entries are non‑negative, the sum is positive, and the runtime is
kept below the 1000 s budget by splitting the time budget among the
stages.  The best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Smooth max‑convolution objective (log‑sum‑exp) and gradient
# --------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_objective, gradient) w.r.t. y.
    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z          # d smooth_max / d conv_i

    # convolution of a with w (restricted to length n)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)   # back‑prop through softmax

    return f, grad_y


# --------------------------------------------------------------
# Adam optimiser on the smooth objective
# --------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    step_size: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter_per_lam: int = 120,
) -> np.ndarray:
    """First‑order Adam on the smooth max‑convolution objective."""
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    for lam in lam_schedule:
        for _ in range(max_iter_per_lam):
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y -= step_size * m_hat / (np.sqrt(v_hat) + eps)
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a pre‑soft‑max vector y for the chosen mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy construction of a small Sidon‑like set
        chosen = []
        sums = set()
        for cand in range(n):
            ok = True
            for c in chosen:
                if (cand + c) in sums:
                    ok = False
                    break
            if ok and (cand * 2) not in sums:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
        k = max(1, len(chosen))
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# --------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# --------------------------------------------------------------


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        k_star = int(np.argmax(conv))

        i = np.arange(n)
        j = k_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # small jitter to escape plateaus
        grad += rng.normal(scale=1e-5, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.3, 2.0)
        else:
            step *= 0.6
            if step < min_step:
                break

    return best, best_score


# --------------------------------------------------------------
# Very fine hill‑climb (tiny mass moves, accept‑only‑improve)
# --------------------------------------------------------------


def fine_hill_climb(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a tiny fraction of the mass.
    Only improving moves are kept.
    """
    rng = np.random.default_rng()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    # initial epsilon relative to the largest coefficient
    epsilon = max(best) * 0.01

    moves = 0
    while time.time() < deadline and moves < max_moves:
        moves += 1
        # pick source i proportionally to its mass (more likely to be large)
        i = rng.choice(n, p=best)
        if best[i] < epsilon:
            continue
        # pick a distinct target uniformly
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = min(epsilon, best[i] * rng.random() * 0.5)
        if delta <= 0.0:
            continue

        # propose move
        a_new = best.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score:
            best = a_new
            best_score = new_score
            # slowly tighten epsilon
            epsilon = max(epsilon * 0.995, 1e-9)
        # else reject

    return best, best_score

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse known construction if present, otherwise 5000.
    # ------------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1000:
            n = max(n, int(tmp.size))
            best_a = np.clip(tmp, 0.0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Stage 1 – L‑BFGS + Adam on a smooth approximation
    # ------------------------------------------------------------------
    stage1_deadline = start_time + 0.55 * budget_s
    lam_schedule_lbfgs = list(np.geomspace(0.4, 0.001, num=22))
    lam_schedule_adam = list(np.geomspace(0.05, 5e-5, num=20))

    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    while time.time() < stage1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ---- L‑BFGS cascade -------------------------------------------------
        y_opt = y0.copy()
        for lam in lam_schedule_lbfgs:
            res = minimize(
                fun=objective_and_grad,
                x0=y_opt,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 250, "gtol": 1e-12, "disp": False},
            )
            y_opt = res.x

        # ---- Adam refinement ------------------------------------------------
        y_opt = adam_optimize(
            y_opt,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_adam,
            step_size=0.05,
            max_iter_per_lam=120,
        )

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent
    # ------------------------------------------------------------------
    stage2_deadline = start_time + 0.85 * budget_s
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, stage2_deadline, init_step=0.5
    )
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Stage 3 – fine greedy hill‑climb with tiny moves
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = fine_hill_climb(
            best_a, n, fft_len_conv, deadline, max_moves=20000
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
