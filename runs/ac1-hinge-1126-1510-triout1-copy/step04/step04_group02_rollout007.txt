"""
Search for a non‑negative step‑function (vector) that minimizes
the evaluation metric 2 n·max(conv)/sum(a)².
The algorithm mixes random restarts of a smoothed L‑BFGS optimiser,
coarse then fine smoothing, projected sub‑gradient descent,
and a simulated‑annealing hill‑climber.  It runs for the whole
budget (≈1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a must have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns f(y) = 2·n·smooth_max(conv) and its gradient w.r.t. y,
    where a = softmax(y) and smooth_max uses a log‑sum‑exp with
    smoothing width lam = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max shifted to 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z   # shape (2*n-1,)

    # compute gradient ∂conv/∂a (the cross‑correlation of a and w)
    # Using FFTs: (a_rev * w) gives vector g_i = Σ_j a_j w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # gradient w.r.t. the soft‑max parameters y
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_schedule: list[float],
                      maxiter_per_stage: int = 250):
    """
    Run a cascade of L‑BFGS optimisation stages with decreasing
    smoothing factor (lam_factor).  Each stage starts from the result
    of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Supported modes:
        random – standard normal
        exp    – decaying exponential
        power  – power‑law decay
        edge   – heavy edges, light centre
        sidon  – greedy Sidon‑set with equal mass
        perm_* – apply a random permutation to the *_ distribution
        best   – small Gaussian perturbation of the current best (handled outside)
    """
    if mode == "random":
        a0 = rng.standard_normal(n)
        y0 = np.log(np.abs(a0) + 1e-12)  # sign ignored; turn into positive probabilities
        # add random permutation to avoid any ordering bias
        perm = rng.permutation(n)
        y0 = y0[perm]
        return y0

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
        return y0

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
        return y0

    elif mode == "edge":
        # mass concentrated at the two ends, remainder uniform tiny value
        m_edge = max(1, int(n * rng.uniform(0.005, 0.02)))
        edge_val = np.sqrt(n / 2.0) / n
        centre_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, centre_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
        return y0

    elif mode == "sidon":
        # Greedy Sidon set (≈√n elements).  Equal mass on chosen indices.
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)  # zeros will become ~-inf; safe because they stay zero after softmax
        return y0

    elif mode.startswith("perm_"):
        # Build the base distribution, then permute it randomly.
        base = mode.split("_", 1)[1]
        y_base = generate_initial_y(base, n, rng)
        # y_base is already log‑probabilities; convert to probabilities,
        # permute, then go back to log‑space.
        a_base = softmax(y_base)
        perm = rng.permutation(n)
        a_perm = a_base[perm]
        y_perm = np.log(a_perm + 1e-12)
        return y_perm

    else:
        raise ValueError(f"unknown mode {mode}")

def hill_climb_sa(a: np.ndarray,
                  cur_score: float,
                  fft_len: int,
                  n: int,
                  deadline: float,
                  max_move: float = 0.20,
                  temp_initial: float = 0.02,
                  temp_final: float = 1e-6,
                  max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.
    At each step a random fraction of mass is moved from i to j.
    Moves that improve the fast score are always accepted;
    otherwise they are accepted with probability exp(-Δ/temperature).
    """
    cur = a.copy()
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # linear temperature schedule
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)
        temp = temp_initial * (1 - fraction) + temp_final * fraction

        i = rng.integers(n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        # propose
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_score - cur_score) / (temp + 1e-12)):
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = cur.copy()
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta

    return best, best_score


def subgradient_descent(a_start: np.ndarray,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        eta0: float = 0.20,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a)).
    Uses diminishing step size η_t = η0 / sqrt(t).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

"""
Search for a non‑negative step‑function (vector) that minimizes

    score = 2 n·max(conv(a,a)) / (sum(a)²)

The algorithm mixes random restarts of a smoothed L‑BFGS optimiser,
coarse‑then‑fine smoothing, projected sub‑gradient descent on the true
max‑convolution objective, and a simulated‑annealing hill‑climber.
It runs for the whole time budget (≈ 1000 s) and returns the best
sequence found.  All numbers are non‑negative; the vector is kept on
the probability simplex (sum = 1) because the score is scale‑invariant.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a must have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns f(y) = 2·n·smooth_max(conv) and its gradient w.r.t. y.
    a = softmax(y).  smooth_max uses a log‑sum‑exp with width
    lam = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c  # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z  # shape (2*n-1,)

    # gradient ∂conv/∂a (cross‑correlation of a and w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient w.r.t. the soft‑max parameters y
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial‑point generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Supported modes:
        random, exp, power, edge, sidon,
        perm_<base>  (random permutation of a base distribution),
        best (handled outside).
    """
    if mode == "random":
        # Gaussian noise turned positive by absolute value then logged
        a0 = np.abs(rng.standard_normal(n))
        y0 = np.log(a0 + 1e-12)
        # random permutation to avoid positional bias
        perm = rng.permutation(n)
        return y0[perm]

    if mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    if mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    if mode == "edge":
        # heavier mass at both ends, light in the centre
        m_edge = max(1, int(n * rng.uniform(0.005, 0.02)))
        edge_val = np.sqrt(n / 2.0) / n
        centre_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, centre_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    if mode == "sidon":
        # Greedy Sidon set (≈√n elements) with equal weight
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        return np.log(a0 + 1e-12)

    if mode.startswith("perm_"):
        base = mode.split("_", 1)[1]
        y_base = generate_initial_y(base, n, rng)
        a_base = softmax(y_base)
        perm = rng.permutation(n)
        a_perm = a_base[perm]
        return np.log(a_perm + 1e-12)

    raise ValueError(f"unknown mode {mode}")


# ----------------------------------------------------------------------
# Local search helpers
# ----------------------------------------------------------------------


def hill_climb_sa(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_move: float = 0.20,
    temp_initial: float = 0.02,
    temp_final: float = 1e-6,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.
    Randomly move a fraction of mass from i to j.
    Improvements are always accepted; otherwise moves are accepted
    with probability exp(-Δ/temperature) where temperature decreases linearly.
    """
    cur = a.copy()
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # linear temperature schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)
        temp = temp_initial * (1 - frac) + temp_final * frac

        i = rng.integers(n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        # propose move
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_score - cur_score) / (temp + 1e-12)):
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = cur.copy()
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    f(a) = 2·n·max(conv(a,a)).
    Diminishing step size η_t = η0 / √t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score
