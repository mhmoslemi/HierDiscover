    iter = 0
    step = 0.1
    best = a
    best_score = fast_score(a, fft_len, n)
    while time.time() < deadline and iter < max_iter:
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        m = conv.argmax()
        # gradient: g[i] = 2 * a[m-i] if in range else 0
        lo = max(0, m-(n-1))

def weighted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter, max_move_frac):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    prob = best / best.sum()
    while time.time() < deadline and max_iter>0:
        max_iter-=1
        i = rng.choice(n, p=prob)  # picks high weight index
        j = rng.integers(0, n)  # destination
        if i == j: continue
        delta = best[i] * rng.random() * max_move_frac
        if delta<=0: continue
        best[i] -= delta
        best[j] += delta
        # update prob? Might keep original.
        new_score = fast_score(best, fft_len, n)

        else:
            best[i] += delta

    return best, best_score

    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. Initialise from a known construction (if any) or uniform.
2. Randomised parametric sweep (inverse‑power, exponential,
   cosine‑bias, edge‑plateau) to get a decent starting shape.
3. Global search – multi‑restart L‑BFGS on a smooth approximation of
   the max‑convolution, using a geometric λ‑schedule that drives the
   smooth max to the true max.
4. Weighted hill‑climb: mass is transferred from large‑weight indices
   (chosen proportionally to their current value) to random targets.
   A large move‑fraction is used first, then reduced.
5. Refined projected‑gradient descent on the smooth max with a
   decreasing smoothing factor λ and a diminishing step size.
6. Simulated‑annealing hill climbing (small random moves with a
   temperature schedule) to escape any remaining local minima.
7. Return the best sequence found (non‑negative, sum‑normalised).
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum == 1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (log‑soft‑max coordinates).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # convolution of a with w (only the part that matters for the gradient)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Gradient of the *exact* max (sub‑gradient) via smooth approximation
# ----------------------------------------------------------------------


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns a vector of length n (no 2·n scaling).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # derivative of smooth max w.r.t a is 2 * g
    return 2.0 * g

# Hill‑climbing helpers
# ----------------------------------------------------------------------


def weighted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially removes mass from large‑weight
    indices (chosen with probability proportional to a_i) and adds it
    to a random target.  The move is kept only if it strictly improves
    the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # initialise sampling probabilities
    prob = best / best.sum()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i = rng.choice(n, p=prob)          # source biased by current mass
        j = rng.integers(0, n)            # destination uniform
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            prob = best / best.sum()       # update probabilities after success
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Classic random hill‑climb (kept for compatibility).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_conv, n)

        if sc_opt < best_score - 1e-9:
            best_score = sc_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved to {best_score:.6f}")

        # Aggressive weighted hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = weighted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
            max_move_frac=0.15,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – refined projected‑gradient descent (smooth max)
    # ------------------------------------------------------------------
    lam_factor_pg = 5e-6
    step0 = 0.4
    max_pg_iter = 20000
    pg_iter = 0
    while time.time() < deadline - 4.0 and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)
        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)

        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            # tighten λ after a successful step
            lam_factor_pg = max(lam_factor_pg * 0.95, 1e-8)
        else:
            # occasionally relax λ to allow movement
            if pg_iter % 250 == 0:
                lam_factor_pg = max(lam_factor_pg * 0.5, 1e-8)

    print(f"[pg] iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb (fine polishing)
    # ------------------------------------------------------------------
    temperature = 0.015
    cooling = 0.9995
    sa_iter = 0
    while time.time() < deadline - 1.0 and temperature > 1e-6:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best_a[i] * rng.random() * 0.05
        if delta == 0.0:
            continue
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len_conv, n)
        diff = new_score - best_score
        if diff < 0 or rng.random() < np.exp(-diff / temperature):
            best_a = a_new
            best_score = new_score
            best_y = np.log(best_a + 1e-12)
        temperature *= cooling
        sa_iter += 1

    print(f"[sa] iterations={sa_iter} final fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
