"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm Overview
------------------
* Work on the probability simplex (sum = 1) – the evaluation is
  ``2·n·max(conv)`` for such sequences.
* Use a smooth log‑sum‑exp surrogate for the max‑convolution and minimise
  it with a continuation L‑BFGS scheme.  The smoothing parameter is a
  *factor* of the current maximum (lam = max(conv)·factor) – the original
  code mistakenly ignored the schedule; we fix that.
* After each L‑BFGS run perform a brief greedy hill‑climb that moves a
  random fraction of mass between two coordinates.
* Run many random restarts (different initial shapes) while time permits.
* Refine the best solution with a projected sub‑gradient descent on the
  true objective (max‑convolution) and a final aggressive hill‑climb.
* The whole process respects the supplied ``budget_s`` (default 1000 s).
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    The smoothing temperature is ``lam = max(conv) * lam_factor``.
    """
    a = softmax(y)

    # --- convolution a * a -------------------------------------------------
    Fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(Fa * Fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max -------------------------------------------------------
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient w.r.t. a -------------------------------------------------
    w = exp_shift / Z               # ∂ smooth_max / ∂ conv_i
    a_rev = a[::-1]
    Fa_rev = np.fft.fft(a_rev, fft_len)
    Fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(Fa_rev * Fw))[: 2 * n - 1]
    g_a = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

    # --- chain rule through softmax ---------------------------------------
    dot = np.dot(a, g_a)
    grad_y = a * (g_a - dot)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 150,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations with decreasing
    smoothing factors.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam_factor in lam_schedule:
        # L‑BFGS on the smoothed objective
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len, lam_factor),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j
    (keeping the sum = 1).  Keep a move only if the fast_score improves.
    The transferred fraction may be up to 10 % of source mass per move.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up‑to‑10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(a*a)  (equivalently max(conv) with sum(a)=1).

    Each iteration:
        * compute the convolution,
        * locate its maximal entry,
        * compute the exact sub‑gradient of that entry,
        * step with η_t = η0 / sqrt(t),
        * project back onto the probability simplex.
    The best iterate encountered is returned.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal convolution entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score
