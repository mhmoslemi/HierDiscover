def subgradient_backtrack(a0, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    step = 1.0
    while time.time() < deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # mask of max
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute subgradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1:n-1+n]  # shape n

        # project gradient direction? we move opposite to gradient.
        descent_dir = -g

        # line search
        found = False
        step_local = step
        while step_local > 1e-8:
            a_candidate = simplex_projection(a + step_local * descent_dir)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score - 1e-12:  # improvement
                found = True
                break
            step_local *= 0.5
        if not found:
            # no improvement, maybe reduce step size further or break
            step *= 0.5
            if step < 1e-9:
                break
        else:
            a = a_candidate
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            # possibly adapt step size
            step = step_local * 1.1  # slightly increase
        it += 1
    return best_a, best_score
```

"""
Propose a low‑score non‑negative step sequence.

The optimizer consists of three stages:
1.  Many fast random restarts of a continuation L‑BFGS that minimises a
    log‑sum‑exp smooth approximation of the max‑convolution.
2.  A projected sub‑gradient method with an adaptive back‑tracking line‑search.
    The sub‑gradient is exact for the (non‑smooth) max‑operator.
3.  A final greedy hill‑climb that moves a small fraction of mass between two
    coordinates only when the move strictly improves the score.

All iterates live on the probability simplex (non‑negative, sum = 1) and all
convolutions are performed by FFT for O(n log n) speed.  The time budget
(≈1000 s) is split automatically; the best sequence found before the deadline
is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a).  Assumes Σa = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'gaussian',
         'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'gaussian':
        mu = rng.uniform(0.3, 0.7) * n
        sigma = rng.uniform(0.05, 0.15) * n
        xs = np.arange(n)
        a = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    else:                     # sparse – a few equal spikes
        k = rng.integers(max(2, n // 200), max(10, n // 20))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k

    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (many restarts)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each smoothing parameter in lam_schedule,
    starting each stage from the previous result.
    Returns (best_a, best_score).
    """
    y = np.log(a_start + 1e-12)                 # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray,
                        fft_len_conv: int,
                        fft_len_g: int):
    """
    Returns (max_val, conv, subgrad) where subgrad = ∇_a max_k (a*a)[k].
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    subgrad = 2.0 * conv_g[(n - 1):(n - 1 + n)]
    return max_val, conv, subgrad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_backtrack(a0: np.ndarray,
                          deadline: float,
                          fft_len_conv: int,
                          fft_len_g: int,
                          n: int):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses an adaptive back‑tracking line search to guarantee descent.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    # initial step size – a heuristic
    step_size = 0.5
    it = 0
    while time.time() < deadline:
        max_val, conv, g = compute_subgradient(a,
                                                fft_len_conv,
                                                fft_len_g)
        # descent direction (negative sub‑gradient)
        d = -g

        # back‑track
        succeeded = False
        step = step_size
        while step > 1e-12:
            cand = simplex_projection(a + step * d)
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score - 1e-13:
                succeeded = True
                break
            step *= 0.5

        if not succeeded:
            # could not improve – shrink aggressive step size and continue
            step_size *= 0.5
            if step_size < 1e-12:
                break
            continue

        # accept the move
        a = cand
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            # allow step to grow a little after a success
            step_size = min(step * 1.2, 5.0)

        it += 1
        # optional: log every few thousand iterations
        if it % 2000 == 0:
            print(f"[sub‑grad] iter {it}, score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient descent (multiplicative updates)
# ----------------------------------------------------------------------


def exp_gradient_descent(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Multiplicative (exponentiated) gradient descent on the simplex.
    Uses the exact sub‑gradient of max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    eta = 0.8                     # initial learning rate
    it = 0
    while time.time() < deadline:
        _, _, g = compute_subgradient(a, fft_len_conv, fft_len_g)

        # multiplicative update
        a_new = a * np.exp(-eta * g)
        # avoid all zeroes (unlikely but safe)
        if a_new.sum() == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / a_new.sum()

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            # increase step size modestly when we improve
            eta = min(eta * 1.05, 5.0)
        else:
            # decrease step size on failure
            eta *= 0.7

        a = a_new
        it += 1
        if it % 2000 == 0:
            print(f"[exp‑grad] iter {it}, score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of mass from i → j.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 4 % of the mass at i
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

                                                  fft_len_g,
                                                  n)
        print(f"[exp‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)
    print(f"[final hill‑climb] best score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
