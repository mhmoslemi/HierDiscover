    A_ub = diags(data, offsets, shape=(2*n-1, n), format='csr')

    bounds = [(0, None)] * n
    # Solve LP using linprog highss
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b == 0:
        return f
    g = b_opt / sum_b
    return g

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy
    from scipy.optimize import minimize, linprog
    from scipy.sparse import diags
    # define utilities functions (same as above)

    np.random.seed(seed)  # for reproducibility
    rng = np.random.default_rng(seed)

    max_n = 4000
    n = max_n
    # possible use shorter n if memory insufficient
    if n <= 0: ...
    # init with uniform
    a0 = np.full(n, 1.0 / n)

    # compute fft lengths
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # parametric sweep
    candidates = []
    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))
    # Uniform always added
    add_candidate(np.full(n, 1.0/n))
    # Linear decreasing
    add_candidate(np.arange(n, 0, -1))
    # Exponential families
    for rate in [0.001, 0.0025, 0.005, 0.01,0.02]:
        add_candidate(np.exp(-rate * np.arange(n)))
    # Power law families
    for p in [0.3,0.6,1.0,1.5,2.0]:
        add_candidate((np.arange(1,n+1)**(-p)))
    # Edge heavy
    for edge_frac in [0.001,0.002,0.005,0.01]:
        m = max(1, int(n*edge_frac))
        arr = np.zeros(n)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)
    # Edge heavy with inverse distance
    dists = np.minimum(np.arange(n), np.arange(n)[::-1]) + 1
    add_candidate(1.0/dists)
    # Sinusoidal squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)
    # Random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))
    # Keep top seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _, arr in candidates[:10]]
    print(f"[sweep] best score {candidates[0][0]:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Phase 1: L-BFGS for each seed with many iterations
    lam_schedule = [0.5,0.3,0.2,0.1,0.06,0.03,0.015,0.008,0.004,0.002,0.001,5e-4,2e-4,1e-4]
    maxiter_per_lam = 500
    restarts = 0
    while time.time() < deadline * 0.55 and restarts < 30:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-4
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start = a_start / a_start.sum()
        a_opt, score_opt = lbfgs_continuation(a_start, lam_schedule, maxiter_per_lam,
                                              deadline * 0.55, fft_len_conv, fft_len_g, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart {restarts}] new best {best_score:.6f}")
        restarts += 1

    # Phase 2: LP improvement
    def lp_step(a):
        # compute M
        conv = full_conv(a, fft_len_conv)
        M = conv.max()
        # build A_ub
        # using diags
        data = []
        offsets = []
        for d in range(n):
            length = (2*n-1) - d
            data.append(np.full(length, a[d]))
            offsets.append(d)
        A_ub = diags(data, offsets, shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return a
        b_opt = res.x
        sb = b_opt.sum()
        if sb <= 0:
            return a
        g = b_opt / sb
        return g

    lp_deadline = start_time + budget_s * 0.75
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < 5:
        g = lp_step(best_a)
        t = 0.2  # mixing
        new_a = (1 - t) * best_a + t * g
        new_a = simplex_projection(new_a)  # ensure sum=1
        new_score = fast_score(new_a, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            print(f"[LP iteration {lp_iter}] improved to {best_score:.6f}")
        else:
            print(f"[LP iteration {lp_iter}] no improvement")
        lp_iter += 1

    # Phase 3: Subgradient descent
    if time.time() < deadline * 0.9:
        sg_deadline = min(deadline * 0.9, deadline - 0.5)
        best_a, best_score = subgradient_optimize(best_a, sg_deadline,
                                                  fft_len_conv, fft_len_g, n)
        print(f"[subgradient] score={best_score:.6f}")

    # Phase 4: Random annealing
    if time.time() < deadline * 0.98:
        best_a, best_score = random_anneal(best_a, best_score,
                                           fft_len_conv, n,
                                           deadline * 0.98)
        print(f"[anneal] score={best_score:.6f}")

    # Final hill climb
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n,

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq

"""
Propose a low‑score step function.

The search consists of:
1.  A cheap parametric sweep (uniform, linear, exponential, power‑law,
    edge‑heavy, inverse‑distance, sin², random Dirichlet) producing a few
    promising seeds.
2.  Continuation L‑BFGS on the smoothed max‑convolution (log‑sum‑exp) with
    a decreasing smoothing schedule and many random restarts.
3.  A small number of exact LP‑updates (maximise the ℓ₁‑norm under the
    current max‑convolution bound) to push mass towards the edges.
4.  Projected sub‑gradient descent (exact max‑convolution as sub‑gradient)
    for further refinement.
5.  Simulated‑annealing + greedy hill‑climb for polishing.
All steps keep the vector on the probability simplex (non‑negative,
sum = 1).  The routine adapts its internal time‑budget to stay within the
overall limit (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c            # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice to obtain the part that corresponds to a (length n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple λ values)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best={best_score:.6f} (lam={lam})")
        # a quick hill‑climb to sharpen
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Directly minimise 2·n·max(a*a) via sub‑gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """Randomly move a fraction of mass between two coordinates."""
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 5000 == 0:
            T = max(T * 0.92, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step (maximise ℓ₁‑norm under current peak)
# ----------------------------------------------------------------------
def lp_improve(f: np.ndarray, fft_len_conv: int, n: int):
    """
    Solve  max Σ b_j   s.t.  f * b ≤ M , b ≥ 0
    where M = max(f*f).  Return the normalised vector g = b / Σb .
    If LP fails, return the original f.
    """
    # current max‑convolution
    conv = full_conv(f, fft_len_conv)
    M = conv.max()

    # Build Toeplitz‑like constraint matrix A (shape (2n‑1, n))
    # A_{k,j} = f_{k-j} if 0 ≤ k-j < n, else 0.
    # Use scipy.sparse.diags to create it efficiently.
    data = []
    offsets = []
    for d in range(n):
        length = (2 * n - 1) - d          # number of entries on diagonal d
        data.append(np.full(length, f[d], dtype=np.float64))
        offsets.append(d)
    A_ub = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)    # minimise -Σb  → maximise Σb
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print("[lp] linprog exception:", e)
        return f

    if not res.success:
        # No improvement possible; keep the current distribution.
        return f

    b_opt = res.x
    sb = b_opt.sum()
    if sb <= 0:
        return f
    g = b_opt / sb
    return g

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a Python list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for gradient / mask convolutions

    # --------------------------------------------------------------
    # Initial uniform distribution (sum = 1)
    # --------------------------------------------------------------
    a0 = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep to collect promising seeds
    # --------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=np.float64)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # 1) Uniform
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=np.float64))

    # 3) Exponential families
    for rate in [0.001, 0.0025, 0.005, 0.01, 0.02]:
        add_candidate(np.exp(-rate * np.arange(n, dtype=np.float64)))

    # 4) Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=np.float64) ** (-p)))

    # 5) Edge‑heavy (mass at both ends)
    for edge_frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=np.float64)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # 6) Inverse‑distance to the nearest edge (strong edge bias)
    dists = np.minimum(np.arange(n), np.arange(n)[::-1]) + 1.0
    add_candidate(1.0 / dists)

    # 7) Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # 8) Random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Keep the top few seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin
    phase1_end = start_time + 0.45 * budget_s    # ~45 % for L‑BFGS
    phase2_end = start_time + 0.70 * budget_s    # next ~25 % for LP updates
    phase3_end = start_time + 0.85 * budget_s    # sub‑gradient
    phase4_end = start_time + 0.95 * budget_s    # annealing
    # final polishing uses whatever remains

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with many random restarts
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.06, 0.03,
                    0.015, 0.008, 0.004, 0.002, 0.001]
    maxiter_per_lam = 200   # a bit more than the original 250, still cheap
    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-4
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f} after {restarts+1} restarts")
        restarts += 1

    # --------------------------------------------------------------
    # Phase 2 – exact LP improvement (few iterations, heavy edges)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        lp_iter = 0
        while time.time() < phase2_end and lp_iter < 4:
            g = lp_improve(best_a, fft_len_conv, n)
            # mix the new vector slowly to keep stability
            t = 0.25
            new_a = (1.0 - t) * best_a + t * g
            new_a = simplex_projection(new_a)   # re‑project onto simplex
            new_score = fast_score(new_a, fft_len_conv, n)
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                print(f"[LP {lp_iter}] improved to {best_score:.6f}")
            else:
                print(f"[LP {lp_iter}] no improvement")
            lp_iter += 1

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (exact max‑conv)
    # --------------------------------------------------------------
    if time.time() < phase3_end:

        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing + greedy perturbations
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = random_anneal(best_a, best_score,
                                           fft_len_conv, n,
                                           phase4_end)
        print(f"[anneal] best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing hill‑climb (as much time as remains)
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    print(f"[final] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the sequence (plain Python list) and report the true score.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function supplied by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
