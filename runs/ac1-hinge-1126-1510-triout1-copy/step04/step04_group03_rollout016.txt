"""
Hybrid step‑function optimiser.

The search consists of:
1.  A fast parametric sweep to collect a few promising seeds.
2.  Continuation L‑BFGS on a log‑sum‑exp smooth approximation of the max‑convolution.
3.  An alternating linear‑program refinement (the “Frank–Wolfe‑style” update described
    in the statement).  The LP maximises the ℓ₁‑norm of a candidate while respecting
    the current maximal convolution value M, then the new candidate is mixed in.
4.  Projected sub‑gradient descent (exact max‑convolution used as a sub‑gradient).
5.  Simulated‑annealing + greedy hill‑climb for polishing.
All phases obey the probability simplex (non‑negative, sum = 1) – the evaluation
function is scale‑invariant, therefore the final list can be returned as‑is.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities (same as the original script)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution with ∑a=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max based on log‑sum‑exp (returns objective and gradient)."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation with simple back‑tracking line‑search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Directly minimise 2·n·max(a*a) via sub‑gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """Greedy mass‑transfer hill‑climb (accept only strictly improving moves)."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing λ schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Continuation L‑BFGS (smooth max) with a short hill‑climb after each λ."""
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # short hill‑climb to sharpen
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score

# Simulated‑annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """Randomly move a fraction of mass between two coordinates."""
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 5000 == 0:
            T = max(T * 0.92, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement (Frank–Wolfe style)
# ----------------------------------------------------------------------
def build_constraint_matrix(f: np.ndarray):
    """Construct A (shape (2n‑1, n)) s.t. A·b = f * b (linear convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            coeff = f[k - i]          # f_{k-i}
            if coeff != 0.0:
                data.append(coeff)
                row_ind.append(k)
                col_ind.append(i)
    return sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))


def lp_refine(a0: np.ndarray,
              n: int,
              fft_len_conv: int,
              deadline: float):
    """
    Alternate: maximise ℓ₁‑norm of a candidate b under (a₀ * b) ≤ M,
    then mix the normalised b back into a.  Repeats until the time runs out.
    Returns the best sequence found (still on the simplex) and its max‑conv.
    """
    a = a0.copy()
    best_a = a.copy()
    best_M = np.max(full_conv(a, fft_len_conv))
    t = 0.5               # mixing weight
    it = 0

    while time.time() < deadline:
        # current bound M
        conv = full_conv(a, fft_len_conv)
        M = conv.max()

        # build the linear‑program constraints
        A = build_constraint_matrix(a)
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)                 # maximise sum(b)  →  minimise –sum(b)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print("[lp] linprog failed, breaking")
            break
        b = res.x
        b_sum = b.sum()
        if b_sum == 0.0:
            break

        # normalise to keep the simplex condition
        g = b / b_sum

        # blend
        a_new = (1 - t) * a + t * g
        M_new = np.max(full_conv(a_new, fft_len_conv))

        # keep best
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            print(f"[lp] iter {it:3d} t={t:.3f} M={M_new:.6f}")

        # accept the update if it reduces the peak, otherwise shrink t
        if M_new < M:
            a = a_new
        else:
            t *= 0.5
            if t < 1e-4:
                break
        it += 1

    return best_a, best_M
