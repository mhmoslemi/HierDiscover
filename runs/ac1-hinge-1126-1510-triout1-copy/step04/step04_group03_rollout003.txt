def exp_grad_descent(a, deadline):
    # a sum=1
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it=0
    while time.time() < deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # subgradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n-1):(n-1 + n)]
        # step size

"""
Improved step‑function search for the min–max convolution problem.

Algorithm
---------
1.  **Parametric seeding** – generate a variety of analytic shapes
    (uniform, linear, triangular, quadratic, sinusoidal, exponential,
    beta‑like) and a few random Dirichlet draws.  Keep the best few as
    warm‑starts.
2.  **Continuation L‑BFGS** – minimise a smooth approximation of the
    max‑convolution (log‑sum‑exp) with a schedule of decreasing smoothing
    parameters λ.  Each λ‑stage runs L‑BFGS‑B for a few hundred iterations.
    After each stage a short greedy hill‑climb is performed.
3.  **Exponentiated‑gradient sub‑gradient descent** – using the exact
    sub‑gradient of the (non‑smoothed) max‑convolution, update the
    probability vector with a multiplicative (mirror‑descent) step and
    renormalise.  This respects the simplex and often pushes the peak
    further down than plain additive updates.
4.  **Greedy hill‑climbing** – fine‑tune the final solution by moving a
    small random fraction of mass between two coordinates and accepting
    only strict improvements.
5.  Return the best sequence found before the overall deadline.

All operations respect the constraints (non‑negative entries, sum = 1).
The code is fully self‑contained and only calls the provided
``evaluate_sequence`` function at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) assuming a sums to 1.
    The true evaluate_sequence score equals this value.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)                     # a lies on the simplex

    # convolution a*a (full length 2·n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12    # smoothing parameter
    shifted = conv_full - max_c        # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                  # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best vector seen.
    """
    y = np.log(a_start + 1e-12)               # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] λ={lam:.6f} new best={best_score:.6f}")

        # short hill‑climb after each λ to sharpen
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def exp_gradient_descent(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Mirror‑descent on the simplex using the exact sub‑gradient of
    max_i (a*a)_i.  Update: a ← a * exp(-η·g) then renormalise.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current auto‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (non‑smoothed) max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size schedule
        eta = 0.5 / np.sqrt(it + 1.0)

        # mirror‑descent update (multiplicative)
        a_new = a * np.exp(-eta * g)
        sum_new = a_new.sum()
        if sum_new == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= sum_new

        cand_score = fast_score(a_new, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_new.copy()
            # keep the improvement
            a = a_new
        else:
            # still move (the exponential update never leaves the simplex)
            a = a_new

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move at most 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert move
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score

        if arr.sum() <= 0.0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # uniform
    add_candidate(np.full(n, 1.0 / n))

    # linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_candidate(lin)

    # linear increasing (mirror of decreasing)
    add_candidate(np.arange(1, n + 1, dtype=float))

    # triangular (tent)
    tri = np.concatenate([np.arange(1, n // 2 + 1),
                          np.arange(n // 2 + (n % 2), 0, -1)])
    add_candidate(tri)

    # quadratic (beta(2,2)‑like)
    quad = (np.arange(1, n + 1, dtype=float) *
            (n - np.arange(0, n, dtype=float))) ** 0.5
    add_candidate(quad)

    # sinusoidal‑squared
    sin_sq = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_sq + 1e-6)

    # arcsine‑like (beta(0.5,0.5))
    eps = 1e-6
    arcs = 1.0 / np.sqrt(
        (np.arange(1, n + 1, dtype=float) + eps) *
        (n - np.arange(0, n, dtype=float) + eps))
    add_candidate(arcs)

    # exponential families
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # random Dirichlet draws (Gamma)
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # include the previous solution if available
    if 'height_sequence_1' in globals():
        add_candidate(init_seq[:n])

    # keep top seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:8]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # Time budget splitting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    phase1_end = start_time + 0.60 * budget_s       # L‑BFGS
    phase2_end = start_time + 0.85 * budget_s       # exp‑gradient descent
    # remaining time for final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.06, 0.03,
                    0.015, 0.008, 0.004, 0.002,
                    0.001, 0.0005, 0.0002, 0.0001, 5e-05]
    maxiter_per_lam = 300

    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        # pick a seed and add jitter
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] restart {restarts} new best={best_score:.6f}")

        restarts += 1

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        exp_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_score = exp_gradient_descent(best_a,
                                                  exp_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[exp‑grad] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (as much as time permits)
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)

    # --------------------------------------------------------------
    # Return the result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
