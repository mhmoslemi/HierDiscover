def correlation(a, p):
    n = a.shape[0]
    a_rev = a[::-1]
    # length for convolution full L = n + len(p) - 1 = n + (2n-1) -1 = 3n-2.

"""
Propose a low‑score step‑function.

Algorithm
---------

* **Initial pool** – generate a handful of normalized candidate
  sequences (uniform, power‑law, exponential, two‑sided power‑law,
  beta‑pdf) of moderate length (8k‑18k).  If a previous best
  ``height_sequence_1`` exists it is also used.

* **Smooth‑max (ℓₚ) optimisation** – the coefficients are represented via
  a soft‑max of unconstrained logits `y`.  For a given `a = softmax(y)`
  we compute the full a*a convolution by an FFT.  Instead of the true
  max we minimise the ℓₚ‑norm of the convolution (`p≈10`), whose
  gradient is a weighted correlation of `a` with the weight vector.
  The gradient w.r.t. the logits is obtained by the soft‑max Jacobian.
  An Adam optimiser with a decaying step‑size updates the logits.
  Every few thousand steps a tiny Gaussian jitter is added to escape
  plateaus.

* **Time‑budget split** – the total 1000 s budget is divided among a
  few independent restarts; each restart receives an equal share of
  the remaining time.  The best sequence seen overall is retained.

* **Local permutation refinement** – with whatever time is left after
  the gradient runs a cheap random‑swap local search is performed on
  the best sequence; a swap is kept if it strictly improves the true
  evaluation score.

* **Final verification** – the best sequence found is returned as a
  plain Python list and its score is printed using the supplied
  ``evaluate_sequence`` function.

The implementation uses only NumPy, respects the 2‑CPU limit and
performs no I/O.  It is self‑contained and can be called directly as
``propose_candidate(seed=…, budget_s=1000)``.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Return the exact ``evaluate_sequence`` value for a non‑negative
    array (sum is not assumed to be 1).  Implemented with an FFT for
    speed.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    L = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def full_convolution(a: np.ndarray, L: int = None) -> np.ndarray:
    """Full a*a convolution (length 2n‑1) using real FFT."""
    n = a.shape[0]
    if L is None:
        L = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g = 2 * (a ⋆ w) where (a ⋆ w)[k] = Σ_i a_i w_{i+k},
    for k = 0 … n‑1.  Implemented via FFT.
    """
    n = a.shape[0]
    # reverse a for convolution
    a_rev = a[::-1]
    L = next_pow2(3 * n - 2)          # length needed for full conv of a_rev (n) and w (2n‑1)
    fa = np.fft.rfft(a_rev, n=L)
    fw = np.fft.rfft(w, n=L)
    full = np.fft.irfft(fa * fw, n=L)   # length L, but only first 3n‑2 entries are non‑zero
    start = n - 1                       # slice that yields the correlation length n
    g = 2.0 * full[start:start + n]
    return g


# ----------------------------------------------------------------------
# Shape generators (used for random restarts)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# ℓₚ‑norm based Adam optimiser
# ----------------------------------------------------------------------


def lp_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    p: float = 10.0,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex (via soft‑max) to minimise
    the ℓₚ‑norm of the self‑convolution.  Returns the best sequence
    seen and its *fast_score*.
    """
    n = a0.shape[0]
    # logits for soft‑max
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # bookkeeping
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ----- soft‑max (keeps us on the probability simplex) -----
        y_shift = y - np.max(y)                # numeric stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()                # a ≥ 0, Σ a = 1

        # ----- convolution -----
        L_conv = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(fa * fa, n=L_conv)[: 2 * n - 1]

        # current true score (max‑based) – keep the best ever seen
        cur_score = fast_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.copy()

        # ----- ℓₚ‑norm and its gradient -----
        conv_nonneg = np.maximum(conv, 0.0)      # avoid tiny negatives from FFT noise
        # ℓₚ‑norm of the convolution (not directly needed, only the weights)
        conv_pow = conv_nonneg ** p
        total_pow = conv_pow.sum()
        if total_pow == 0.0:                     # degenerate case – skip this step
            w_norm = np.full_like(conv, 1.0 / conv.size)
        else:
            # weight = conv_i^{p‑1} / (∑ conv_j^p)^{(p‑1)/p}
            weight = conv_nonneg ** (p - 1.0)
            norm_factor = total_pow ** ((p - 1.0) / p)
            w_norm = weight / norm_factor

        # gradient w.r.t. a (size n) :  g = 2 * (a ⋆ w_norm)
        g = correlation(a, w_norm)

        # gradient w.r.t. logits, using Jacobian of soft‑max
        ag = np.dot(a, g)                       # scalar ⟨a,g⟩
        grad_y = a * (g - ag)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter to avoid stagnation -----
        if it % 2000 == 0:
            sigma = 0.02 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple random‑swap local refinement
# ----------------------------------------------------------------------


def local_swap_refine(seq: np.ndarray, time_limit: float, start_time: float) -> np.ndarray:
    """
    Perform a cheap random‑swap hill‑climb for the remaining time.
    The swap is kept only if it strictly improves the true score.
    """
    best = seq.copy()
    best_val = fast_score(best)
    n = best.shape[0]

    attempts = 0
    while time.time() < start_time + time_limit:
        i, j = random.sample(range(n), 2)
        # perform swap
        best[i], best[j] = best[j], best[i]
        cur_val = fast_score(best)
        attempts += 1
        if cur_val < best_val:
            best_val = cur_val
            # keep the improvement
        else:
            # revert
            best[i], best[j] = best[j], best[i]

    if attempts:
        print(f"[refine] performed {attempts} swaps, best score {best_val:.6f}")
    return best
