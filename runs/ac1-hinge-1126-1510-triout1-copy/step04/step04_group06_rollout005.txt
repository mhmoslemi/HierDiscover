"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm consists of four stages:

1. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  A schedule of decreasing
   smoothing factors is used so that the solution gradually approaches
   the true nonsmooth problem.

2. **Adam‑style projected sub‑gradient descent** – directly minimises
   the true max‑convolution using the sub‑gradient given by the active
   set (indices attaining the current maximum).  Adam’s adaptive step‑
   sizes and Euclidean projection onto the probability simplex give a
   fast, stable refinement.

3. **Simulated annealing** – non‑local random mass moves are accepted
   with a Metropolis criterion.  This helps to escape possible local
   minima after the gradient phases.

4. **Final greedy hill‑climb** – tiny deterministic mass transfers
   between two positions are performed while the score improves.

The routine re‑uses a previously found construction (`height_sequence_1`)
if it exists, otherwise starts from the uniform distribution.
All steps respect the supplied time budget (default 1000 s) and the
best sequence found is stored globally for later calls.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # fallback – should not happen
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) for a already on the simplex."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective (log‑sum‑exp) and its gradient w.r.t. y.
    a = softmax(y) is a probability vector.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy hill‑climb: repeatedly move a tiny fraction of mass between
    two positions if the (fast) score improves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

            best_score = score_candidate
            best_a = a_candidate.copy()
            print(f"[L‑BFGS] lam={lam:.5f} score={best_score:.6f}")

        # short hill‑climb polish after each L‑BFGS run (few iterations)
        hill_deadline = min(deadline * 0.55,
                            time.time() + 0.12)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hill_deadline,
                                        max_iter=800)

    # ------------------------------------------------------------------
    # Phase 1 – Adam‑style projected sub‑gradient descent on the true max.
    # ------------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)   # first‑moment estimate
    v = np.zeros_like(a)   # second‑moment estimate
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    it = 0

    while time.time() < deadline * 0.80:
        # convolution of a with itself
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active (max‑attaining) indices – tiny tolerance for
        # numerical noise.
        eps = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - eps).astype(float)

        # sub‑gradient of the max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = 0.6 * m_hat / (np.sqrt(v_hat) + eps_adam)   # learning‑rate ≈0.6
        a_new = a - step
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter={it} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing (global exploration).
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = best_score
    best_sa = cur_a.copy()
    best_sa_score = cur_score
    T = max(cur_score * 0.02, 1e-9)   # initial temperature
    decay = 0.998
    sa_iter = 0

    while time.time() < deadline:
        # choose a source index that actually carries mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-15:
            continue

        # distinct target index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 30 %) of the source mass
        delta = cur_a[i] * rng.random() * 0.30
        if delta <= 0:
            continue
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len_conv, n)
        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert move
                cur_a[i] += delta
                cur_a[j] -= delta
                # temperature update and continue
                T *= decay
                continue

        if cur_score < best_sa_score:
            best_sa_score = cur_score
            best_sa = cur_a.copy()
            # optional reheating when a new best is found
            T = max(cur_score * 0.02, 1e-9)
            print(f"[SA] iter={sa_iter} score={best_sa_score:.6f}")

        T *= decay
        sa_iter += 1

    best_a = best_sa
    best_score = best_sa_score

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb polish.
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=5000)

    # ------------------------------------------------------------------
    # Report the score from the supplied evaluator (optional).
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # Store globally for possible reuse in later runs.
    globals()['height_sequence_1'] = final_a.tolist()

    return final_a.tolist()
