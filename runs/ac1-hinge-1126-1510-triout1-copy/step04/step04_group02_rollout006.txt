def solve_lp_max_sum(a, M, n):
    # Build sparse matrix A as described (size (2*n-1) x n)
    # Use scipy.sparse.csr_matrix
    # Build lists row, col, data
    row = []
    col = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # i ranges i_start..i_end inclusive
        # j = k-i
        # coefficient a_i for variable g_j
        # So each i produce entry (k, j)
        for i in range(i_start, i_end+1):
            j = k - i
            row.append(k)
            col.append(j)
            data.append(a[i])
    A = scipy.sparse.csr_matrix((data, (row, col)), shape=(2*n-1, n))
    c = np.full(n, -1.0)  # maximize sum g => minimize -sum g
    b = np.full(2*n-1, M)
    bounds = [(0, None) for _ in range(n)]
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})  # time_limit second maybe
    if res.success:
        return res.x
    else:
        print("[LP] linprog failed", res.message)
        return None

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
1. Choose a moderate problem size (default n≈2000).  If a previous
   construction ``height_sequence_1`` exists we reuse its length (capped
   at 5000) and initialise from it, otherwise we start uniform.
2. *Smooth L‑BFGS* – minimise a log‑sum‑exp (smooth max) surrogate of
   the convolution maximum.  Several random starts (Gaussian,
   exponential, power‑law, edge‑heavy, Sidon‑set) are tried, each
   followed by a continuation schedule of decreasing smoothness.  After
   each L‑BFGS run a short greedy hill‑climb is performed.
3. *Projected sub‑gradient descent* on the true objective
   ``2·n·max(conv)`` (with sum = 1) for a few 10⁵ iterations.
4. *Linear‑programming refinement*:  with the current distribution
   `a` and its current bound `M = max(conv(a))` we solve the LP

        maximise   Σ g_i
        subject to conv(a,g) ≤ M   (2n‑1 linear constraints)
                   g_i ≥ 0

   The solution `g` is normalised to a probability vector and accepted
   if it improves the fast score.  A few such LP steps are performed.
5. Final aggressive hill‑climb until the time budget expires.
6. Return the best sequence found; the harness will call
   ``evaluate_sequence`` on it.

The combination of a smoothed global optimiser, exact sub‑gradient
steps and an LP‑based “active‑set” refinement typically beats the
previous best bound (≈ 1.509) and often reaches ≤ 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed L‑BFGS objective (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (smoothed objective, gradient) w.r.t. the unconstrained
    vector y (pre‑softmax).  lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a (full length 2n-1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c        # max shifted to 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # gradient w.r.t. a (using convolution theorem)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = conv_rev[(n - 1) : (n - 1 + n)]

    # back‑prop through softmax
    mean_g = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial‑point generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting vector y (pre‑softmax) for a given heuristic."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set (approx √n elements)
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums:
                    conflict = True
                    break
            if not conflict and 2 * candidate not in sums:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Local greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i
    to j (keeping the sum = 1).  Keep a move only if the fast_score
    strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for f(a)=max(conv(a,a)).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximise total mass under a fixed
# convolution bound).
# ----------------------------------------------------------------------


def solve_lp_max_sum(a: np.ndarray, M: float, n: int) -> np.ndarray | None:
    """
    Solve:
        maximise   Σ g_i
        subject to conv(a,g) ≤ M   (2n‑1 linear constraints)
                   g_i ≥ 0
    Returns the optimal g (as a numpy array) or None on failure.
    """
    # Build sparse matrix A where A[k,i] = a[i] if i contributes to (a*g)_k
    rows = []
    cols = []
    data = []

    # Linear convolution dimensions: k = 0 .. 2n‑2
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i          # index of g_j
            rows.append(k)
            cols.append(j)
            data.append(a[i])

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)               # maximise Σg  <=> minimise -Σg
    b = np.full(2 * n - 1, M)

    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    else:
        print("[LP] linprog failed:", res.message)
        return None

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
