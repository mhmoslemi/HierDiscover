"""
Propose a low‑score step‑function.

Algorithm
---------
* **Random starts** – generate a diverse pool of normalized shapes
  (power‑law, exponential, two‑sided power‑law, Beta‑pdf) of moderate
  length (≈8k–15k).  Include the previously discovered sequence
  `height_sequence_1` if it exists.

* **Smoothed max‑convolution optimisation** – the objective
  `2·n·max(conv(a,a))/sum(a)^2` is invariant to scaling, so we keep the
  coefficients on the probability simplex (soft‑max parametrisation).
  The non‑smooth “max” is approximated by a log‑sum‑exp with a
  temperature β that is increased during optimisation.  The gradient
  of the smoothed objective can be written in closed form using a
  convolution; FFT‑based convolutions make each iteration fast
  (O(N log N)).  Adam with a decaying step‑size and occasional jitter
  drives the iterates to a (local) minimum.

* **Time‑budget handling** – the total 1000 s budget is split among a
  number of independent restarts; each restart receives an equal share
  of the remaining time.  The best sequence seen overall is refined
  for any leftover time.

The implementation respects the 2‑CPU limit, uses only NumPy and the
provided `evaluate_sequence`, and prints progress information.
"""

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)**2) using FFT.
    The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:                     # avoid division by zero
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta_: float) -> np.ndarray:

    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(α-1) * (n-i)^(β-1)

    i = np.arange(1, n + 1, dtype=np.float64)

def smoothed_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta0: float = 5.0,
    beta_max: float = 200.0,
    beta_inc_factor: float = 1.15,
    beta_inc_every: int = 2000,
    jitter_every: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector `a` (sum = 1) to minimise the
    evaluation score.  The non‑smooth max is replaced by a smooth
    log‑sum‑exp with temperature β that is increased during optimisation.
    Returns the best sequence encountered and its score.
    """
    n = a0.shape[0]
    eps = 1e-16
    # initialise logits for the soft‑max parametrisation
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999

    best_seq = a0.copy()
    # because a0 is already normalised, fast_score = 2·n·max(conv)
    best_val = fast_score(best_seq)

    it = 0
    beta = beta0

    while True:
        now = time.time()
        if now > start_time + time_limit:
            break
        it += 1

        # ---------- soft‑max ----------
        y_shift = y - np.max(y)          # numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()          # stays on the simplex

        # ---------- convolution a*a ----------
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        max_conv = float(conv.max())

        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---------- smooth max via log‑sum‑exp ----------
        # w_k = softmax(β·conv_k)
        beta_conv = beta * conv
        max_beta_conv = np.max(beta_conv)
        w = np.exp(beta_conv - max_beta_conv)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---------- gradient of the smoothed objective ----------
        # g = 2 * (w * a_rev)_{(n‑1)+j}
        a_rev = a[::-1]
        len_w = w.size
        len_rev = a_rev.size
        fft_len_g = next_pow2(len_w + len_rev - 1)
        fw = np.fft.rfft(w, n=fft_len_g)
        fra = np.fft.rfft(a_rev, n=fft_len_g)
        h = np.fft.irfft(fw * fra, n=fft_len_g)
        start_idx = len_rev - 1
        g = 2.0 * h[start_idx : start_idx + n]

        # ---------- gradient w.r.t. logits ----------
        ag = np.dot(a, g)               # scalar ⟨a,g⟩
        grad_y = a * (g - ag)           # Jacobian of soft‑max

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape flat regions
        if it % jitter_every == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

        # increase the temperature β gradually
        if it % beta_inc_every == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_inc_factor)

    return best_seq, best_val

    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_overall_seq is not None and time_left > 5.0:
        print(f"[refine] final refinement for {time_left:.1f}s")
        seq, val = smoothed_optimize(
            best_overall_seq,
            time_limit=time_left,
            start_time=time.time(),
            lr0=0.3,
            beta0=10.0,
            beta_max=300.0,
            beta_inc_factor=1.2,
            beta_inc_every=1500,
            jitter_every=2500,
        )
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"[refine] improved final score to {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback and final verification
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # never happened, but keep a safe default
        n = 10000
        best_overall_seq = np.full(n, 1.0 / n)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")
    return final_list
