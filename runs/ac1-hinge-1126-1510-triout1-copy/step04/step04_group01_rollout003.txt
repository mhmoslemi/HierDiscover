"""
Improved multi‑scale optimisation for the step‑function “max‑self‑convolution”
attack.

Key ideas
---------
* Search over several problem sizes (n = 1000 … 4000).  Larger n gives a
  larger search space and often yields a lower value of
  2·n·max(a*a) while keeping the normalisation ∑a = 1.
* For each size run a very aggressive L‑BFGS continuation with a finer
  lam‑schedule (down to 1e‑4).  The objective is a log‑sum‑exp smooth
  approximation of the max‑convolution.
* Follow up with a long projected sub‑gradient descent (momentum) that works
  directly on the true max (no smoothing) and uses a decreasing step size.
* Fine‑tune with Adam on the logit parametrisation.
* A short simulated‑annealing phase and an extensive greedy hill‑climb finish
  the search.
* All phases respect the probability simplex (non‑negative, sum = 1) by using
  a soft‑max or Euclidean projection.

The routine is deliberately time‑aware: the total budget (default 1000 s) is
split among the phases, and every loop checks the deadline.  The best sequence
found over all examined sizes is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from numpy.fft import fft, ifft

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already normalised to sum=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carpenter (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev ⋆ w)
    a_rev = a[::-1]
    fa_rev = fft(a_rev, fft_len_g)
    fw = fft(w, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, warm‑starting each stage.
    Returns the best sequence (as a plain ndarray) and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # occasional numerical issues – just skip this stage
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (true max, momentum)
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with momentum on the genuine max‑convolution.
    The step size follows 0.6 / sqrt(it+1) and the mask that selects the
    active constraints shrinks over time.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution
        conv = full_conv(a, fft_len_conv)

        max_val = conv.max()
        # tolerance – start generous and tighten as we iterate
        eps = max_val * max(0.01, 1.0 / np.sqrt(it + 1.0))
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size schedule
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the logit parametrisation (smoothed max)
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.13):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n,
                                     fft_len_conv, fft_len_g,
                                     lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (light) – random mass transfers
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Very light annealing: move a random fraction (≤10 %) from one index
    to another, accept worsening moves with a slowly decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.025
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T = max(T * 0.90, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (strict improvement)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 12000,
               max_fraction: float = 0.10):
    """
    Deterministic hill‑climb: repeatedly pick two coordinates and move a
    random fraction (up to max_fraction) of mass from one to the other.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_fraction
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Random initialiser (diverse families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:                     # sparse – Dirichlet with small concentration
        a = rng.gamma(0.3, 1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes to explore – the larger the size, the richer the search space
    # ------------------------------------------------------------------
    size_options = [1200, 1800, 2400, 3000, 3600, 4000]

    # Keep the global best across all sizes
    global_best_a = None
    global_best_score = float('inf')
    global_best_n = None

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # 0.5 s safety margin

    for n in size_options:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising for n = {n} ===")
        fft_len_conv = next_pow2(2 * n - 1)
        fft_len_g = next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Initialise: use previous global best truncated (if any) or a fresh random start
        # ------------------------------------------------------------------
        if global_best_a is not None:
            a0 = np.pad(global_best_a, (0, max(0, n - global_best_a.size)), mode='constant')
            a0 = a0[:n]
            a0[a0 < 0] = 0.0
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)

        best_a = a0.copy()
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[init] fast_score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Time budget split for this size (proportional to n)
        # ------------------------------------------------------------------
        now = time.time()
        size_deadline = min(deadline,
                            now + (budget_s * 0.6) * (n / max(size_options)))
        phase1_end = now + 0.42 * (size_deadline - now)   # L‑BFGS
        phase2_end = now + 0.70 * (size_deadline - now)   # sub‑gradient + Adam
        # remaining time → anneal + hill‑climb

        # ------------------------------------------------------------------
        # Phase 1 – L‑BFGS continuation with a fine lam schedule
        # ------------------------------------------------------------------
        lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                        0.07, 0.045, 0.03, 0.018, 0.01,
                        0.006, 0.003, 0.0015, 0.001, 0.0005,
                        0.0003, 0.0001]
        maxiter_per_lam = 800

        while time.time() < phase1_end:
            a_start = random_initial(n, rng)
            a_opt, score_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                phase1_end,
                fft_len_conv,
                fft_len_g,
                n
            )
            if score_opt < best_score:
                best_score, best_a = score_opt, a_opt.copy()
                print(f"[L‑BFGS] new best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – sub‑gradient with momentum (true max) and then Adam
        # ------------------------------------------------------------------
        if time.time() < phase2_end:
            # sub‑gradient
            sg_deadline = phase2_end - 0.5
            best_a, best_score = subgradient_optimize_momentum(
                best_a,
                sg_deadline,
                fft_len_conv,
                fft_len_g,
                n
            )
            print(f"[sub‑gradient] score = {best_score:.6f}")

        if time.time() < deadline:
            # Adam fine‑tuning (smoothed max, very low lam)
            adam_deadline = deadline - 0.7
            logits = np.log(best_a + 1e-12)
            best_a, best_score = adam_optimize(
                logits,
                n,
                fft_len_conv,
                fft_len_g,
                adam_deadline,
                lam_factor=0.001,
                lr=0.12
            )
            print(f"[Adam] score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – short annealing
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = deadline - 0.4
            best_a, best_score = random_anneal(
                best_a,
                best_score,
                fft_len_conv,
                n,
                anneal_deadline
            )
            print(f"[anneal] score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – final greedy hill‑climb (more iterations & larger moves)
        # ------------------------------------------------------------------
        final_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=18000,
            max_fraction=0.08
        )
        print(f"[final hill‑climb] score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Keep the best across all sizes
        # ------------------------------------------------------------------
        if best_score < global_best_score:
            global_best_score = best_score
            global_best_a = best_a.copy()
            global_best_n = n
            print(f"**** New global best (n={n}) = {global_best_score:.6f}")

        # update remaining budget for next size
        remaining = deadline - time.time()
        if remaining < 5.0:   # very little time left – stop early
            break

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = global_best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"\n[final] n={global_best_n}, evaluated score = {final_eval:.6f}")

    return final_seq
