"""
Propose a better step‑function sequence.

This implementation follows the multi‑phase scheme of the reference
solution but adds several improvements:

* A fast parametric sweep over simple families (exponential,
  power‑law, quadratic) to obtain a strong deterministic start.
* More aggressive L‑BFGS runs with a finer smoothing schedule.
* An Adam‑style momentum sub‑gradient optimiser (instead of plain
  momentum) – typically converges faster.
* A slightly more explorative simulated‑annealing phase.
* Two passes of greedy hill‑climbing (large then fine moves).

All phases respect the global time budget (default 1000 s) and the
vector is always projected onto the probability simplex, which makes
the objective invariant under scaling.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initial sequence generators
# ----------------------------------------------------------------------


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a probability vector of length n.

    Modes:
        'uniform' – flat.
        'edge'    – heavy edges, lighter interiors.
        'exp'     – exponential decay from the left (mirrored).
        'power'   – power‑law decay (mirrored).
        'quad'    – quadratic bump (higher in the centre).
        'random'  – Dirichlet(0.5) sample.
    """
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)          # proportion of indices at each edge
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'quad':
        # quadratic bump centred in the middle, zero at the ends
        i = np.arange(n, dtype=float)
        centre = (n - 1) / 2.0
        x = 1.0 - ((i - centre) / centre) ** 2          # parabola ranging [0,1]
        a = np.maximum(x, 0)
        a /= a.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed max objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed‑max (log‑sum‑exp) objective and its gradient w.r.t. y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient optimiser on the true max
# ----------------------------------------------------------------------


def subgradient_adam_optimize(a0: np.ndarray,
                              deadline: float,
                              fft_len_conv: int,
                              fft_len_g: int,
                              n: int):
    """
    Adam optimiser for min 2·n·max(a*a).  Operates on the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = 0.05                     # base learning rate

    while time.time() < deadline:
        t += 1
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active constraints
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the true max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local moves)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """Standard SA with occasional big relocations."""
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        # pick a source with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        # target index (different)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction (up to 30 %) of the source mass
        delta = rng.random() * cur_a[i] * 0.30
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            accept_prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < accept_prob:
                cur_score = new_score
            else:
                # revert move
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        # occasional reheating if temperature drops too low
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing (local polishing)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000,
               delta_frac: float = 0.04):
    """
    Repeatedly transfer a tiny fraction of mass between two positions.
    delta_frac controls the maximal fraction of the source mass moved.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Deterministic parametric sweep (cheap but often powerful)
# ----------------------------------------------------------------------


def parametric_sweep(n: int,
                     fft_len: int,
                     rng: np.random.Generator,
                     deadline: float):
    """
    Scan a few low‑dimensional families (exponential, power‑law, quadratic)
    and keep the best vector found before the deadline is reached.
    """
    best_a = None
    best_score = float('inf')

    # Exponential decay from the edges (symmetric)
    for _ in range(40):
        if time.time() >= deadline:
            break
        beta = rng.uniform(0.0005, 0.025)
        i = np.arange(n, dtype=float)
        d = np.minimum(i, n - 1 - i)
        a = np.exp(-beta * d)
        a = a / a.sum()
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # Power‑law decay from the edges (symmetric)
    for _ in range(40):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.1, 3.0)
        i = np.arange(n, dtype=float)
        d = np.minimum(i, n - 1 - i) + 1.0
        a = d ** (-p)
        a = a / a.sum()
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # Quadratic bump centred in the middle (smooth hill)
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = np.ones(n, dtype=float)
        centre = (n - 1) / 2.0
        i = np.arange(n, dtype=float)
        a = 1.0 - ((i - centre) / centre) ** 2
        a = np.maximum(a, 0.0)
        # add a tiny random perturbation to avoid symmetry lock‑in
        a += rng.normal(scale=0.001, size=n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising `evaluate_sequence`.

    The strategy:
        1. Initialise from a previous run if available; otherwise use a
           deterministic parametric sweep to obtain a strong start.
        2. Multiple stochastic L‑BFGS refinements on a smooth
           log‑sum‑exp surrogate of the max‑convolution.
        3. Adam‑style sub‑gradient descent on the true max.
        4. Simulated annealing for non‑local exploration.
        5. Two passes of greedy hill‑climbing (large then fine moves).

    The best vector found before the time limit is returned.
    """
    rng = np.random.default_rng(seed + 123456)

    # ------------------------------------------------------------------
    # Choose problem size (n) and starting vector.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        max_n = 4000
        if prev.size > max_n:
            prev = prev[:max_n]
        if prev.sum() > 0:
            a0 = prev / prev.sum()
            n = a0.size
        else:
            n = 2500
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2500
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # FFT lengths (powers of two for speed)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)     # for a*a
    fft_len_g = next_pow2(3 * n - 2)        # for auxiliary convolutions

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – quick deterministic sweep (takes ~5–10 s)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 0.06 * budget_s
    sweep_a, sweep_score = parametric_sweep(n, fft_len_conv, rng, sweep_deadline)
    if sweep_a is not None and sweep_score < best_score:
        best_a, best_score = sweep_a, sweep_score
        print(f"[sweep] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS on a smoothed surrogate.
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_factors = [0.5, 0.3, 0.15, 0.08, 0.04,
                   0.02, 0.01, 0.006, 0.003, 0.001, 0.0005]
    lbfgs_iter = 0
    while time.time() < phase1_end:
        mode = rng.choice(['uniform', 'edge', 'exp', 'power', 'quad', 'random'])
        a_start = build_initial(mode, n, rng)
        y0 = np.log(a_start + 1e-12)

        lam = lam_factors[lbfgs_iter % len(lam_factors)]

        # L‑BFGS on smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'gtol': 1e-12,
                                'disp': False})

        a_candidate = softmax(res.x)
        score_candidate = fast_score(a_candidate, fft_len_conv, n)

        if score_candidate < best_score:
            best_score = score_candidate
            best_a = a_candidate.copy()
            print(f"[L‑BFGS {lbfgs_iter}] mode={mode} lam={lam:.4f} score={best_score:.6f}")

        # short hill‑climb after each L‑BFGS run for quick polishing
        hill_deadline = min(phase1_end, time.time() + 0.07)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hill_deadline,
                                        max_iter=800,
                                        delta_frac=0.04)

        lbfgs_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – Adam sub‑gradient descent on the true max.
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam_optimize(best_a,
                                                      phase2_end,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
        print(f"[Adam sub‑grad] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing.
    # ------------------------------------------------------------------
    phase3_end = deadline
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (coarse then fine).
    # ------------------------------------------------------------------
    # Coarse moves (up to 20 % of source mass)
    coarse_deadline = min(deadline, time.time() + 0.03 * budget_s)
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    coarse_deadline,
                                    max_iter=2000,
                                    delta_frac=0.20)

    # Fine moves (up to 4 % of source mass)
    final_deadline = deadline
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      final_deadline,
                                      max_iter=4000,
                                      delta_frac=0.04)

    # ------------------------------------------------------------------
    # Optional evaluation with the official evaluator (for logging only)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # Store globally for possible reuse in a later call
    globals()['height_sequence_1'] = final_a.tolist()

    return final_a.tolist()
