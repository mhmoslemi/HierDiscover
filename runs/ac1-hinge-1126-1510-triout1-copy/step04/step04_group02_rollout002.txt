    rng = np.random.default_rng()

"""
Proposes a non‑negative step‑function (a list of heights) that aims to
minimise the metric

    2·n·max(convolution(a,a)) / (sum(a)^2)

used by the evaluation harness.  The algorithm consists of three
phases, all respecting the 1000 s time budget.

1. **Smooth L‑BFGS explorations** – a log‑sum‑exp smoothing of the
   max‑convolution is minimised using several random restarts.  This
   quickly finds good basins of attraction.
2. **Targeted sub‑gradient / gradient‑guided hill climbing** – after each
   L‑BFGS run we repeatedly move probability mass from indices that
   contribute most to the current maximal convolution entry to indices
   that contribute the least, i.e. from large to small components of the
   exact sub‑gradient.  This phase runs for the bulk of the remaining
   time and is capable of fine‑tuning the solution.
3. **Random greedy polishing** – finally a simple random move hill‑climb
   with a larger allowable transfer fraction is performed until the
   budget expires.

All vectors are kept on the probability simplex (sum = 1, non‑negative)
so the metric equals `2·n·max(conv)`.  The code prints progress and
returns the best sequence it found as a plain list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that maps ℝⁿ to the probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using real FFT.
    a must have length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max(conv) and its gradient w.r.t.
    the unconstrained vector y (soft‑max parametrisation).

    The smoothing parameter is lam = max(conv)·lam_factor.
    """
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z               # ∂ smooth_max / ∂ conv_i

    # g_i = Σ_j a_j * w_{i+j}  (linear convolution a * w)
    fa_rev = np.fft.rfft(a[::-1], fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # relevant slice correspond to indices (n‑1 … n‑1+n‑1)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 150,
) -> np.ndarray:
    """
    Run a series of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous stage.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Allowed modes:
        random  – Gaussian vector
        exp     – exponential decay
        power   – power‑law decay
        edge    – heavy edges, light centre
        sidon   – sparse Sidon‑set pattern
        sparse  – random sparse support of size ≈ √n
        best    – perturbation of the current best (handled elsewhere)
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * cand not in sums):
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "sparse":
        # Randomly select ≈ √n positions
        k = max(2, int(np.sqrt(n) * rng.uniform(1.0, 2.5)))
        idx = rng.choice(n, size=k, replace=False)
        a0 = np.zeros(n)
        a0[idx] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def random_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a random fraction (≤max_move) of mass from
    i → j and keeps the move only if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
    step_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb.  In each iteration we:
        * locate the current maximal convolution entry m,
        * compute the exact sub‑gradient D_i = 2·a_{m‑i} (zero outside support),
        * move a fraction of probability mass from the index with largest D
          (and non‑zero mass) to the index with smallest D.
    The move is kept only if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # Build sub‑gradient D
        D = np.zeros_like(best)
        start = max(0, m - (n - 1))
        end = min(n - 1, m)
        idx_range = np.arange(start, end + 1, dtype=int)
        counterpart = m - idx_range
        D[idx_range] = 2.0 * best[counterpart]

        # indices with positive mass
        pos_idx = np.where(best > 0.0)[0]
        if pos_idx.size == 0:
            break
        i_max = pos_idx[np.argmax(D[pos_idx])]
        i_min = int(np.argmin(D))

        if i_max == i_min:
            continue

        delta = best[i_max] * rng.random() * step_frac
        if delta <= 0.0:
            continue

        best[i_max] -= delta
        best[i_min] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_max] += delta
            best[i_min] -= delta

    return best, best_score


def projected_subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.35,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on the true objective
    f(a) = max(conv(a,a)).  The step size is η_t = η0 / √t.
    The best iterate encountered is returned.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # Sub‑gradient of the active entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    # Default size – big enough for a rich search, but not too big.
    n = 5000
    # Re‑use a previously discovered construction if available.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)

    # FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate on the simplex
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – several L‑BFGS restarts with decreasing smoothing
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=24))
    modes = ["random", "exp", "power", "edge", "sidon", "sparse", "best"]
    iteration = 0

    while time.time() < deadline - 150.0:   # keep ~2.5 min for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=150,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # quick random hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – long gradient‑guided hill‑climb (targeted moves)
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print("[targeted] starting gradient‑guided hill‑climb")
        target_deadline = deadline - 50.0
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            target_deadline,
            max_iter=20000,
            step_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[targeted] fast score after targeting = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (fine tuning)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 20.0
        best_a, best_score = projected_subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.35,
            max_iters=600_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] fast score after sub‑grad = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing – random hill‑climb with larger moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] polishing with random hill‑climb")
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # The harness will evaluate the exact score; we display the fast estimate
    # for information.
    final_fast = fast_score(best_a, fft_len_conv, n)
    print(f"[final] fast estimate = {final_fast:.6f}")
    return final_seq
