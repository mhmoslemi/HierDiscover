"""
Improved step‑function optimizer.

The algorithm consists of three phases:

1. **Global non‑convex search** – many multi‑restart L‑BFGS runs on the
   soft‑max parametrisation.  A geometric continuation schedule drives the
   smooth‑max approximation (λ) from a moderate value down to 1e‑6, which
   closely approximates the true ℓ∞‑norm of the auto‑convolution.

2. **Exact sub‑gradient descent** – starting from the best L‑BFGS solution
   we minimise the genuine objective  
       J(a) = 2·n·max (a*a)   (∑a = 1, a≥0)  
   by repeatedly (a) locating the active indices where the convolution attains
   its maximum, (b) computing the exact sub‑gradient of the max‑operator,
   (c) taking a projected step on the probability simplex with a
   1/√t step‑size schedule.  This removes the smoothing bias of phase 1
   and yields a finer optimum.

3. **Aggressive hill‑climb & symmetrisation** – a final greedy random‑mass‑
   transfer hill‑climb (up to 20 % per move) polishes the solution.
   Because the problem is symmetric, we also test the mirror of the best
   sequence and keep it if it improves the score.

The whole search respects the 1000 s budget and returns the best
non‑negative sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max yielding a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (∑a = 1).
    Identical to ``evaluate_sequence`` for a normalised sequence.
    """
    Fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(Fa * Fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (used by L‑BFGS global search)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (log‑soft‑max coordinates).
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    Fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(Fa * Fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max -----------------------------------------
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    Fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(Fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Greedy hill‑climb (phase 3)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true ℓ∞‑objective (phase 2)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    max_iter: int = 20000,
    step0: float = 0.5,
    tol_active: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on J(a) = 2·n·max(a*a) (∑a=1, a≥0).

    Returns the best vector encountered and its fast_score.
    """
    cur = a0.copy()
    cur_score = fast_score(cur, fft_len_conv, n)
    best = cur.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # auto‑convolution via FFT
        Fa = np.fft.fft(cur, fft_len_conv)
        conv = np.real(np.fft.ifft(Fa * Fa))[: 2 * n - 1]

        max_c = conv.max()
        # active set (indices attaining the maximum, within small tolerance)
        active = np.where(conv >= max_c - tol_active)[0]

        # exact sub‑gradient of max_c w.r.t. a
        g = np.zeros_like(cur)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            g[idx] += 2.0 * cur[k - idx]

        # projected step (step size ~ 1/√t)
        step = step0 / np.sqrt(it)
        cand = proj_simplex(cur - step * g)
        cand_score = fast_score(cand, fft_len_conv, n)

        # monotone acceptance (helps stability)
        if cand_score < cur_score - 1e-14:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
        # else keep current point (step size will shrink automatically)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise onto the probability simplex (sum = 1)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 1 – Global L‑BFGS search (multiple restarts)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-6, num=40)   # finer continuation
    maxiter_stage = 800                              # more internal iterations

    iteration = 0
    # reserve ~200 s for the global search, the rest for refinement
    while time.time() < deadline - 200.0:
        mode = rng.choice(
            ["random", "exp", "power", "edge", "sym", "best"]
        )

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
        elif mode == "sym":
            # symmetric random start
            a0 = rng.random(n)
            a0 = (a0 + a0[::-1]) / 2.0
        else:   # perturb the current best
            a0 = best_a.copy()
            a0 += rng.normal(scale=0.03, size=n)
            a0 = np.maximum(a0, 0.0)

        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()

        y0 = np.log(a0 + 1e-12)

        # ---- continuation L‑BFGS ----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}"
            )

        # ---- brief hill‑climb to exploit local gains ----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=3000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent on the true objective
    # ------------------------------------------------------------------
    sg_deadline = deadline - 30.0  # keep 30 s for final polishing
    best_a, best_score = subgradient_descent(
        a0=best_a,
        n=n,
        fft_len_conv=fft_len_conv,
        deadline=sg_deadline,
        max_iter=20000,
        step0=0.5,
        tol_active=1e-12,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[sg] sub‑gradient best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Optional symmetrisation – the problem is symmetric, try mirroring
    # ------------------------------------------------------------------
    sym_a = (best_a + best_a[::-1]) / 2.0
    sym_score = fast_score(sym_a, fft_len_conv, n)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        print(f"[sym] symmetric version improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence; official evaluation performed by the harness
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # assumed to be pre‑imported
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
