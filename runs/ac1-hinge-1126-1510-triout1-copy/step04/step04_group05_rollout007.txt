def simulated_annealing(a, cur_score, fft_len, n, deadline, init_temp=1e-4, cooling=0.9995):
   best = a.copy()
   best_score = cur_score
   cur_a = a.copy()
   cur_score = cur_score
   T = init_temp
   rng = np.random.default_rng()
   # how many iterations? as many as time allows
   while time.time() < deadline:
       i, j = rng.integers(0, n, size=2)
       if i==j: continue
       # delta as fraction of a[i] maybe uniform in [0, max_move_frac]
       max_move_frac = 0.2
       delta = cur_a[i] * rng.random() * max_move_frac
       if delta<=0: continue
       cur_a[i] -= delta
       cur_a[j] += delta
       new_score = fast_score(cur_a, fft_len, n)
       # accept if better or random
       dE = new_score - cur_score
       if dE < 0 or rng.random() < np.exp(-dE / T):
           cur_score = new_score
           # Update best if better
           if new_score < best_score:
               best_score = new_score
               best = cur_a.copy()
       else:
           # revert
           cur_a[i] += delta
           cur_a[j] -= delta
       # cooling
       T *= cooling
   return best, best_score

"""
Hybrid optimizer for the step‑function “max‑convolution” attack.

The search consists of three phases:

1. **Continuation L‑BFGS** – a smooth approximation of the max‑convolution
   (log‑sum‑exp) is minimised on the probability simplex using a soft‑max
   parametrisation.  A geometric schedule of smoothing parameters (λ) is
   applied, giving a fast descent into a good basin.

2. **Deterministic greedy refinement** – a targeted hill‑climb repeatedly
   transfers mass from the largest contributor to the smallest one for the
   currently maximal convolution index.  This directly reduces the true
   ⟨max a*a⟩.

3. **Simulated‑annealing polishing** – random mass transfers are accepted
   with a Metropolis criterion that slowly cools.  Occasional uphill moves
   allow the search to escape shallow local minima that the greedy step
   cannot leave.

The routine runs many random restarts (including several hand‑crafted
initial shapes), keeps the best‑ever sequence and returns it after the
budget (≈ 1000 s) expires.  The final result is passed through
``evaluate_sequence`` for the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first 2·n‑1 entries (full linear convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector ``a``.
    This equals the value returned by ``evaluate_sequence`` when
    ``sum(a) == 1`` (the denominator then equals 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(a*a) and its gradient w.r.t. the
    unconstrained logits ``y`` (where a = softmax(y)).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: repeatedly minimise the smoothed objective while
    decreasing the smoothing width λ (given by ``lam_factors``).
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Greedy (targeted) hill climbing – the “sub‑gradient” step
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy refinement that reduces the true max‑convolution.
    At each iteration the index(s) that contribute most to the current
    peak are identified and a small fraction of their mass is transferred
    to the index(s) with the smallest contribution.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback – random pair
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing – escapes shallow local minima
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 1e-4,
    cooling: float = 0.9995,
    max_move_frac: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers accepted with a Metropolis criterion.
    A slowly decaying temperature allows occasional uphill moves.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    T = init_temp
    rng = np.random.default_rng()

    # Adapt temperature to the scale of the objective
    if T <= 0:
        T = 1e-8

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject – revert
            cur[i] += delta
            cur[j] -= delta

        T *= cooling
        # keep temperature from under‑flow
        if T < 1e-12:
            T = 1e-12

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0            # safety margin

    # --------------------------------------------------------------
    # Problem size (length of the sequence)
    # --------------------------------------------------------------
    default_n = 2000
    # Try to reuse the previously discovered construction if it exists.
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            n = default_n
            a0 = arr[:n].copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 1e-12:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    # explicit normalisation (sum = 1)
    a0 = a0 / np.maximum(a0.sum(), 1e-12)

    best_a = a0.copy()
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)                    # logits for L‑BFGS
    print(f"[init] n={n}   fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS continuation schedule
    # --------------------------------------------------------------
    fft_len_conv = fft_len
    fft_len_g = next_pow2(3 * n - 2)

    lam_factors = np.geomspace(0.5, 5e-8, num=45)     # finer final schedule
    maxiter_stage = 500                              # a bit more per stage

    # --------------------------------------------------------------
    # Helper to pick an initial shape (seed) for a restart
    # --------------------------------------------------------------
    def init_shape(mode: str) -> np.ndarray:
        if mode == "random":
            y = rng.standard_normal(n)
            return softmax(y)
        elif mode == "exp":
            rate = rng.uniform(0.002, 0.015)
            a = np.exp(-rate * np.arange(n))
        elif mode == "power":
            p = rng.uniform(0.4, 2.2)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == "edge":
            # exponential decay from the edges toward the centre
            d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
            decay = rng.uniform(0.003, 0.02)
            a = np.exp(-decay * d)
        elif mode == "double_exp":
            # symmetric double exponential (large at both ends, tiny centre)
            d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
            decay = rng.uniform(0.01, 0.04)
            a = np.exp(-decay * d)
        elif mode == "best":
            return best_a
        else:
            a = np.full(n, 1.0 / n)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)
        # return logits for softmax
        return a

    # --------------------------------------------------------------
    # Multi‑restart optimisation loop (L‑BFGS + targeted hill‑climb)
    # --------------------------------------------------------------
    iteration = 0
    modes = ["random", "exp", "power", "edge", "double_exp", "best"]
    while time.time() < deadline - 5.0:
        mode = rng.choice(modes)

        a_start = init_shape(mode)
        y_start = np.log(a_start + 1e-12)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y_start=y_start,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # one quick targeted hill‑climb (≈0.6 s)
        t_hc_deadline = min(deadline, time.time() + 0.65)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            t_hc_deadline,
            max_moves=2500,
            frac=0.08,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Deep refinement of the current best (more L‑BFGS steps)
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[refine‑L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final aggressive polishing – targeted hill‑climb + annealing
    # --------------------------------------------------------------
    # targeted hill‑climb (a few seconds)
    if time.time() < deadline:
        t_hc_deadline = min(deadline, time.time() + 2.5)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            t_hc_deadline,
            max_moves=8000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

    # simulated annealing for the remaining time
    if time.time() < deadline:
        anneal_deadline = deadline
        # temperature proportional to the current objective magnitude
        init_T = max(1e-6, best_score * 5e-5)
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            anneal_deadline,
            init_temp=init_T,
            cooling=0.9992,
            max_move_frac=0.12,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Normalise (just in case) and produce the final list
    # --------------------------------------------------------------
    final_seq = best_a / max(best_a.sum(), 1e-12)   # enforce sum ≈ 1
    final_seq = final_seq.tolist()

    # Official evaluation (the harness will also call evaluate_sequence,
    # but we log the score here for debugging)
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence globally so that subsequent runs can use it
    globals()["height_sequence_1"] = final_seq

    return final_seq
