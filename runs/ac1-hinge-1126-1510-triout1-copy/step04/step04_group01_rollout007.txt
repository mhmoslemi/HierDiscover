"""
Improved 1000‑second optimizer for the step‑function score.

The algorithm is a parallel multi‑restart optimiser that works on several
sequence lengths (2000 and 4000).  For each length we perform:

1. Many random restarts of a continuation L‑BFGS optimiser on a
   log‑sum‑exp (smooth‑max) surrogate.  The smoothing parameter is
   driven down to 5·10⁻⁴ for a very sharp approximation.
2. A projected sub‑gradient descent with momentum.
3. An Adam optimiser on the same smooth surrogate.
4. A short simulated‑annealing phase.
5. A greedy hill‑climb that moves up to 20 % of the mass between two
   coordinates and accepts only improving moves.

All phases keep the vector on the simplex (non‑negative, sum = 1).  The
best sequence obtained by any worker process is returned.  The code uses
only NumPy, SciPy and the standard library, fits inside the 1000 s budget
and respects the “≤ 2 CPUs” rule by running at most two processes in
parallel.
"""

import time
import numpy as np
from scipy.optimize import minimize
import multiprocessing as mp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (result length = 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a distribution already
    normalised to sum = 1 (i.e. the value returned by evaluate_sequence).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best distribution found.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Very unlikely, but stay robust.
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Simple projected sub‑gradient descent with Nesterov‑style momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        v = beta * v + (1.0 - beta) * g

        step = 0.4 / np.sqrt(it + 1.0)          # slightly larger than before
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the logit parameters
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.006,
                  lr: float = 0.18):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy hill‑climb: repeatedly move a random fraction (up to 20 %) of
    the mass from one coordinate to another, accepting only strictly
    improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.20    # up to 20 %
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing phase
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simple annealer: move a random fraction (up to 10 %) of mass between
    two entries while accepting worsening moves with a decreasing
    temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.025
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 5000 == 0:
            T *= 0.90
            T = max(T, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialiser
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'mirror'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.002, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.008, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.015 * n), int(0.06 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # mirror – a symmetric mirror of a random vector
        half = rng.random(n // 2 + 1)
        a = np.concatenate([half, half[-2::-1]]) if n % 2 else np.concatenate([half, half[::-1]])
    a /= a.sum()
    return a

# Core optimisation routine (run in a worker process)
# ----------------------------------------------------------------------


def run_optimize(seed: int, n: int, time_budget: float):
    """
    Perform a full optimisation for a given length `n` and a time budget.
    Returns the best sequence (as a NumPy array) and its score.
    """
    rng = np.random.default_rng(seed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # start from a uniform distribution – it gives a deterministic baseline
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[worker seed={seed} n={n}] initial fast_score={best_score:.6f}")

    start = time.time()
    deadline = start + time_budget - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with a longer smoothing schedule
    # ------------------------------------------------------------------
    phase1_end = start + 0.45 * time_budget
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.0015, 0.0008]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[worker seed={seed}] L‑BFGS new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – momentum sub‑gradient descent
    # ------------------------------------------------------------------
    phase2_end = start + 0.80 * time_budget
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                            phase2_end,
                                                            fft_len_conv,
                                                            fft_len_g,
                                                            n)
        print(f"[worker seed={seed}] sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = deadline - 0.5
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           adam_deadline,
                                           lam_factor=0.003,
                                           lr=0.15)
        print(f"[worker seed={seed}] Adam best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.3
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[worker seed={seed}] anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)
    print(f"[worker seed={seed}] final hill‑climb best={best_score:.6f}")

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Problem sizes we will try – 2000 (used in the reference) and 4000
    # (the maximum that still fits comfortably in memory).
    sizes = [2000, 4000]

    # Split the total budget roughly evenly between the two workers.
    # If only one size fits in the remaining time we just run that.
    total_start = time.time()
    total_deadline = total_start + budget_s - 0.5   # safety

    # Decide actual time allocation (a bit generous to the first worker).
    time_per_worker = (budget_s - 2.0) / len(sizes)

    # ------------------------------------------------------------------
    # Parallel execution (max 2 processes)
    # ------------------------------------------------------------------
    manager = mp.Manager()
    return_dict = manager.dict()

    def worker_task(idx, n, t_budget):
        a, sc = run_optimize(seed + idx * 1000, n, t_budget)
        return_dict[idx] = (a, sc)

    processes = []
    for i, n in enumerate(sizes):
        p = mp.Process(target=worker_task, args=(i, n, time_per_worker))
        p.start()
        processes.append(p)

    # Wait for all processes (or for the global deadline)
    for p in processes:
        while p.is_alive() and time.time() < total_deadline:
            p.join(timeout=0.1)
        if p.is_alive():
            p.terminate()
            p.join()

    # ------------------------------------------------------------------
    # Gather the best result
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for idx in range(len(sizes)):
        if idx not in return_dict:
            continue
        a, sc = return_dict[idx]
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[main] new global best from worker {idx}: {best_score:.6f}")

    # If for some reason both workers failed (unlikely), fall back to uniform.
    if best_seq is None:
        n = sizes[0]
        best_seq = np.full(n, 1.0 / n)

    final_seq = best_seq.tolist()
    # Evaluate finally (the runner will also evaluate, we just log it)
    try:
        final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception:
        final_eval = None
        print("[final] evaluation failed (function not present)")

    return final_seq
