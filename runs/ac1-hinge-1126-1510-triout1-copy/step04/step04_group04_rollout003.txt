"""
Step‑function optimizer for the “max‑convolution” evaluation metric.

Main ideas
----------
* **Spectral seed** – build a distribution whose *cyclic* self‑convolution is
  (approximately) constant by solving (FFT(a))² = FFT(constant).  After
  clipping to non‑negative values this gives a very flat starting point.
* **Multi‑restart L‑BFGS** – minimise a smooth log‑sum‑exp approximation of
  the max‑convolution.  The schedule of smoothing parameters runs from coarse
  to very fine, repeatedly from many random seeds (including the spectral
  one).  The parametrisation a = softmax(y) guarantees feasibility.
* **Greedy hill‑climbing** – local moves that transfer a small fraction of
  mass between two positions are accepted only if the fast (FFT‑based) score
  improves.
* **Mirror‑descent (exponentiated gradient)** – a run of exponentiated‑gradient
  descent on the *exact* max‑convolution (sub‑gradient of the L∞ norm) while
  staying on the probability simplex.
* **Final polishing** – a short sub‑gradient phase and a final L‑BFGS pass
  with an ultra‑small smoothing factor.

The whole pipeline respects the constraints of ``evaluate_sequence``:
all entries are non‑negative, the sum is > 0, and no element exceeds 1000.
The code runs for at most the allotted time (≈ 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy import sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1. Spectral initialisation (constant target convolution)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """
    Build a distribution whose *cyclic* self‑convolution is (approximately)
    constant.  The result is projected onto the simplex.
    """
    L = next_pow2(2 * n - 1)
    # target constant convolution (average value)
    const_val = 1.0 / (2 * n - 1)
    c = np.zeros(L, dtype=float)
    c[: 2 * n - 1] = const_val

    C_fft = np.fft.fft(c)
    # square‑root in the complex plane
    A_fft = np.sqrt(C_fft)
    a_full = np.fft.ifft(A_fft).real

    # keep only the first n entries (the rest are artefacts of the cyclic model)
    a = a_full[:n]
    a = np.maximum(a, 0.0)                # enforce non‑negativity
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= total                         # normalise to a probability vector
    return a


# ----------------------------------------------------------------------
# 2. Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (log‑sum‑exp) approximation of the L∞ norm
    of the self‑convolution and its gradient w.r.t. the pre‑soft‑max vector ``y``.
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z   # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# 3. Random initial points (various heuristics)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point *y* (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums:
                    conflict = True
                    break
                if (2 * candidate) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "spectral":
        a0 = spectral_init(n)
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# 4. Optimisation primitives
# ----------------------------------------------------------------------


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """Cascade of L‑BFGS runs with decreasing smoothing (λ)."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    The move is kept only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    After each step the vector is projected onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # tiny random jitter to escape plateaus
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def mirror_descent_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    eta0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the exact max‑convolution.
    Works directly on the simplex (updates a ← a·exp(‑η·∇)).
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]
        # jitter
        grad += rng.normal(scale=1e-4, size=n)

        eta = eta0 / np.sqrt(t + 1)
        # exponentiated update; keep numbers in a reasonable range
        update = np.exp(-eta * grad)
        best = best * update
        best = np.maximum(best, 1e-15)       # avoid exact zeros
        best /= best.sum()

        cur_score = fast_score(best, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
        t += 1

    return best, best_score


# ----------------------------------------------------------------------
# 5. Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    The method combines:
      • a spectral starting point (almost constant self‑convolution);
      • many random L‑BFGS restarts on a smooth max‑approximation;
      • aggressive hill‑climbing;
      • mirror‑descent and sub‑gradient refinement;
      • a final fine L‑BFGS pass.
    """
    rng = np.random.default_rng(seed)
    # leave a short safety margin for the final evaluate_sequence call
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # problem size – a moderate size that still allows many iterations
    # ------------------------------------------------------------------
    n = 5000

    # use a previously stored construction if it exists and is large enough
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1_000:
            n = max(n, int(tmp.size))
            best_a = np.clip(tmp, 0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    # guarantee a valid probability vector
    best_a = project_to_simplex(best_a)
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # FFT lengths needed by the optimiser
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # smoothing schedule for the log‑sum‑exp surrogate
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=30))

    # ------------------------------------------------------------------
    # multi‑restart L‑BFGS + hill‑climb loop
    # ------------------------------------------------------------------
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "spectral", "best"]
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=250,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # a quick local hill‑climb (short time slice)
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Mirror‑descent refinement (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        md_deadline = min(deadline, time.time() + 40.0)
        best_a, best_score = mirror_descent_optimize(
            best_a, n, fft_len_conv, md_deadline, eta0=0.5
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[mirror] refined fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient descent (fine‑grained descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sg_deadline = min(deadline, time.time() + 30.0)
        best_a, best_score = subgradient_optimize(
            best_a, n, fft_len_conv, sg_deadline, init_step=0.5
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] refined fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final ultra‑fine L‑BFGS pass (λ → 0)
    # ------------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-6, num=25))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] λ={lam:.2e} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final greedy polishing (use the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Return the candidate (the harness will call evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
