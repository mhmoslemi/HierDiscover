def alternating_projection(a, fft_len, max_iter, rng, deadline):
    best_a = a
    best_score = fast_score(a, fft_len, len(a))
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # FFT forward
        Fa = np.fft.fft(a, fft_len)
        # Power (sum of squares)
        power = np.mean(np.abs(Fa) ** 2)   # = sum_i a_i^2
        # Set all magnitudes to sqrt(power)
        Fa_new = np.sqrt(power) * np.exp(1j * np.angle(Fa))
        # Inverse FFT
        a_new = np.real(np.fft.ifft(Fa_new))[:len(a)]
        # Project onto non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            # reinitialize random
            a_new = rng.random(len(a))
            a_new /= a_new.sum()
        else:
            a_new /= s
        a = a_new
        # Periodically evaluate score
        if it % 10 == 0:
            sc = fast_score(a, fft_len, len(a))
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score
```

def subgradient_descent(a, fft_len_conv, max_iter, rng, deadline):
    n = len(a)
    best_a = a
    best_score = fast_score(a, fft_len_conv, n)
    eta0 = 0.5
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # Compute gradient of max conv
        Fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(Fa * Fa))[: 2*n - 1]
        k_max = np.argmax(conv)
        delta = np.zeros(fft_len_conv)
        delta[k_max] = 1.0
        grad = 2.0 * np.real(np.fft.ifft(Fa * np.fft.fft(delta, fft_len_conv)))[:n]
        # Step size schedule
        eta = eta0 / np.sqrt(it)
        a_candidate = proj_simplex(a - eta * grad)
        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()

"""
Propose a low‑score step‑function using three complementary heuristics:

1. **Alternating projection (AP)** – enforce a flat Fourier magnitude
   (which would give a constant auto‑convolution) and project back onto the
   probability simplex.  Repeating the two projections quickly drives the
   convolution towards its average value.

2. **Sub‑gradient descent on the exact max‑convolution** – locate the index
   where the current auto‑convolution is maximal, compute the exact gradient
   of that entry (via FFT) and take a projected‑gradient step on the simplex.

3. **Aggressive hill‑climbing** – random mass‑transfer moves that keep any
   strict improvement.  This polishes the solution found by the two
   smoother stages.

The routine performs many short restarts of (1) + (2) and a final hill‑climb
until the time budget (default 1000 s) is exhausted.  All numbers are
non‑negative and sum to one, so the returned score is exactly the
``evaluate_sequence`` value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1).

    Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Heuristic 1 – Alternating projection (flatten spectrum)
# ----------------------------------------------------------------------


def alternating_projection(a0: np.ndarray, fft_len: int, max_iter: int,
                           rng: np.random.Generator,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a few alternating steps:
        * set all FFT magnitudes to the same value (constant power)
        * project the resulting time‑domain signal onto the non‑negative simplex
    Returns the best vector seen during the run and its fast_score.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # forward FFT
        Fa = np.fft.fft(a, fft_len)

        # average power = Σ a_i^2 = mean(|F|^2)
        power = np.mean(np.abs(Fa) ** 2)

        # enforce constant magnitude (keep the phases)
        Fa = np.sqrt(power) * np.exp(1j * np.angle(Fa))

        # inverse FFT and keep first n samples
        a = np.real(np.fft.ifft(Fa))[:n]

        # project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:               # re‑initialise if we lost all mass
            a = rng.random(n)
            a /= a.sum()
        else:
            a /= s

        if (it & 7) == 0:          # evaluate every few iterations
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Heuristic 2 – Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray, fft_len: int, max_iter: int,
                        rng: np.random.Generator,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of the linear auto‑convolution using sub‑gradient steps.
    The gradient of the maximal entry is obtained analytically via FFT.
    """
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    eta0 = 0.5

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # compute linear convolution via FFT
        Fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(Fa * Fa))[: 2 * n - 1]
        k_max = int(conv.argmax())

        # gradient of conv[k_max] w.r.t. a:
        delta = np.zeros(fft_len)
        delta[k_max] = 1.0
        grad = 2.0 * np.real(np.fft.ifft(Fa * np.fft.fft(delta, fft_len)))[:n]

        # projected step on the simplex
        eta = eta0 / np.sqrt(it)
        a_new = proj_simplex(a - eta * grad)

        sc = fast_score(a_new, fft_len, n)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new               # accept improvement
        # otherwise keep the old a (standard sub‑gradient behaviour)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Heuristic 3 – Simple hill‑climbing on the simplex
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, cur_score: float, fft_len: int, n: int,
               deadline: float, max_iter: int = 5000,
               max_move_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of the mass from one coordinate to another.
    Keep any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:      # strict improvement
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    n = 2000

        # ------------------------------------------------------------------
        sg_deadline = min(deadline, time.time() + 0.9)
