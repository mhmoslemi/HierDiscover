def optimize_sequence(a0, time_limit, start_time=None):

    y = np.log(a0 + 1e-12)
    # Adam states
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_seq = a0.copy()
    best_val = 2 * n * max(conv) maybe compute via fast_score (fast_score returns objective 2*n*maxconv because sum a =1). Actually fast_score returns 2 n maxconv / sum(a)^2; with sum=1 returns correct objective.

    iteration = 0
    eps = 1e-12 (Adam epsilon). 
    # epsilon smoothing for softmax over conv
    epsilon_factor = 0.1  # initial relative
    epsilon_min = 1e-8
    M_current = None

    While time remaining:
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / np.sum(exp_y)

        # Compute conv a*a:
        fft_len = next_pow2(2*n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n - 1]
        M = np.max(conv)
        # compute objective:
        val = 2 * n * M  # sum=1
        if val < best_val:
            best_val = val
            best_seq = a.copy()

        # compute weights p
        eps_temp = max(epsilon_factor * M, epsilon_min)
        # Softmax of conv divided by eps_temp
        max_conv = np.max(conv)  # ensure stability
        scaled = (conv - max_conv) / eps_temp  # relative diff
        exp_scaled = np.exp(scaled)
        p = exp_scaled / np.sum(exp_scaled)  # length 2n-1

        # Compute gradient g = conv(p, a) + conv(p, rev(a)) offset
        L = next_pow2(3*n - 2)
        # conv1 = p * a
        fp = np.fft.rfft(p, n=L)
        fa = np.fft.rfft(a, n=L)
        conv_pa = np.fft.irfft(fp * fa, n=L)[: 3*n - 2]  # length

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=L)
        conv_p_arev = np.fft.irfft(fp * fa_rev, n=L)[: 3*n - 2]

        g = conv_pa[0:n] + conv_p_arev[n-1 : n-1 + n]

        # gradient wrt y:
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        # learning rate schedule (maybe constant):
        lr = lr0 / (np.sqrt(iteration) + 1)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Decay epsilon factor
        epsilon_factor *= 0.9995  # gradually shrink to zero

        # jitter every maybe 5000 iterations
        if iteration % 5000 == 0:
           sigma = 0.01 / np.sqrt(n)
           y += sigma * np.random.randn(n)

    return best_seq, best_val

"""
Improved step‑function search.

The algorithm proceeds in three stages:

1. **Randomised shape pool** – generate a modest set of candidate
   sequences using a few analytic families (power‑law, exponential,
   two‑sided power‑law, Beta) with various lengths (≈8k‑20k).  The global
   best known sequence (``height_sequence_1``) is also inserted if
   available.

2. **Soft‑max + smoothed max‑convolution optimisation** – each candidate
   is represented by log‑softmax variables ``y`` so that the actual
   coefficients ``a`` lie on the probability simplex (∑aᵢ=1,
   aᵢ≥0).  The objective to minimise (for a simplex‑normalised
   sequence) is

       f(a) = 2·n·max_k (a∗a)_k .

   Instead of using only the exact active set of the maximum, we use a
   smooth approximation
       M_ε = ε·log∑_k exp((a∗a)_k/ε)
   with a slowly shrinking temperature ``ε``.
   The gradient of ``M_ε`` w.r.t. ``a`` is a weighted sum of
   the convolution derivatives and can be computed efficiently by two
   FFT‑based convolutions.

   An Adam optimiser (β₁=0.9, β₂=0.999) with a decaying step size and a
   small Gaussian jitter every few thousand steps drives the parameters
   toward lower values of ``f``.  The best sequence seen during the run
   is kept.

3. **Time‑budget allocation** – after a brief initialisation phase the
   majority of the 1000 s budget is spent refining the single most
   promising candidate; the remaining top candidates receive a much
   smaller share.  The final sequence is verified with the supplied
   ``evaluate_sequence`` function.

All helper functions are top‑level; no I/O or external solvers are used.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the attack metric (2·n·max(conv))/sum^2.
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Shape generators (used for random starts)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1)·(n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Optimiser – Adam on the smoothed max‑convolution objective
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the quantity 2·n·max(conv(a,a)) under the
    simplex constraint (∑a=1, a≥0).  ``a0`` must already be normalised.
    The method uses a temperature‑smoothed log‑sum‑exp approximation of
    the max, whose gradient is obtained via two FFT‑based convolutions.
    Returns the best sequence encountered and its metric value.
    """
    n = a0.shape[0]
    # initialise logits (log‑softmax parameters)
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # pre‑compute FFT lengths
    fft_len_a = next_pow2(2 * n)                # for a*a
    fft_len_p = next_pow2(3 * n - 2)            # for p*a and p*a_rev

    # bookkeeping of the best sequence
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    # smoothing schedule parameters
    eps_factor = 0.5          # initial ε = eps_factor * M
    eps_decay = 0.9995
    eps_min_factor = 1e-12    # absolute floor for ε

    it = 0
    while True:
        now = time.time()
        if now > start_time + time_limit:
            break

        it += 1

        # ----- soft‑max to obtain a (lies on the simplex) -----
        y_shift = y - np.max(y)            # numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # ----- convolution a*a (full) -----
        fa_full = np.fft.rfft(a, n=fft_len_a)
        conv = np.fft.irfft(fa_full * fa_full, n=fft_len_a)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M               # sum(a)=1 ⇒ denominator = 1

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- smoothed max (log‑sum‑exp) -----
        eps = max(eps_factor * M, eps_min_factor)
        # stable softmax over conv / eps
        conv_shifted = conv - M
        exp_scaled = np.exp(conv_shifted / eps)
        p = exp_scaled / exp_scaled.sum()   # weights, length 2n‑1

        # ----- gradient of M_ε w.r.t. a (via convolutions) -----
        # p * a  (full convolution, length 3n‑2)
        fp = np.fft.rfft(p, n=fft_len_p)
        fa_p = np.fft.rfft(a, n=fft_len_p)
        conv_pa = np.fft.irfft(fp * fa_p, n=fft_len_p)[: 3 * n - 2]

        # p * a_rev
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, n=fft_len_p)
        conv_par = np.fft.irfft(fp * fa_rev, n=fft_len_p)[: 3 * n - 2]

        # gradient g_i = Σ_k p_k (a_{k-i}+a_{i-k})
        g = conv_pa[0:n] + conv_par[n - 1 : n - 1 + n]

        # ----- map gradient to logits (soft‑max Jacobian) -----
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ----- temperature decay -----
        eps_factor *= eps_decay

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

        # optional cheap progress report
        if it % 5000 == 0:
            print(f"[optim] it={it}, best={best_val:.6f}, M={M:.6e}, eps={eps:.2e}")

    return best_seq, best_val
