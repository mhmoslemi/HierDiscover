def propose_candidate(...):

    rng = np.random.default_rng(seed)

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        # maybe currently sum already normalized? We'll renorm.

    else:
        n = 2000

    n = a.size
    best_a = a.copy()
    best_M = np.max(np.convolve(best_a, best_a))
    best_score = 2 * n * best_M  # for sum=1

    start = time.time()
    deadline = start + budget_s - 0.5

    # Phase: LP-based improvement
    # We'll start with t=0.5 (mix ratio)
    t = 0.5
    # Use small epsilon for LP constraints maybe 1e-12

    while time.time() < deadline:
        # compute current M
        conv = np.convolve(best_a, best_a)
        M = conv.max()
        # Build LP matrix
        # Precompute row idx, col idx, data once per iteration building from best_a
        f = best_a
        # Build H matrix
        # Using sparse
        n = len(f)
        # Compute data, row_idx again
        data = np.tile(f, n)
        row_idx = np.concatenate([np.arange(j, j+n) for j in range(n)])
        col_idx = np.repeat(np.arange(n), n)
        H = csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        # Solve LP
        c_obj = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=H, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            # cannot improve; break
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0 or np.isnan(sum_b):
            break
        g = b / sum_b
        # Mix
        new_a = (1 - t) * f + t * g
        # Guarantee non-neg and normalized (should sum to 1)
        new_a = np.maximum(new_a, 0)

        # Evaluate
        new_conv = np.convolve(new_a, new_a)
        new_M = new_conv.max()
        new_score = 2 * n * new_M

            best_M = new_M
            best_a = new_a.copy()
            print(f"[LP] new_score {best_score:.6f} iteration time {time.time()-start:.1f}s")
            # maybe increase step size t towards 0.5 else adapt
            # keep t stable
        else:

            t *= 0.5

                break

def run_lp_update(f, max_iters, deadline):
    best_f = f.copy()
    n = len(f)
    best_score = 2*n*np.max(np.convolve(best_f,best_f))
    t = 0.5
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute M
        conv = np.convolve(f,f)
        M = conv.max()
        # build H
        H = build_conv_matrix(f) # sparse CSR
        # solve LP
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0,None)]*n
        res = linprog(c_obj, A_ub=H, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            break
        g = b / sum_b
        new_f = (1 - t) * f + t * g
        # re-normalize (should already sum to 1)
        new_f = np.maximum(new_f, 0)
        new_f = new_f / new_f.sum()
        # evaluate
        new_score = 2*n*np.max(np.convolve(new_f,new_f))
        if new_score < best_score - 1e-12:
            best_f = new_f.copy()
            best_score = new_score
            f = new_f
            # keep t same
            print(f"[LP iter {it}] improved score {best_score:.6f}")
        else:
            # reduce step size
            t *= 0.5
            if t < 1e-4:
                break
    return best_f, best_score

"""
Optimization of a non‑negative step sequence for the bound
    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is based on the classic alternating LP scheme
described in the problem statement:

1. Start from a feasible vector a (non‑negative, sum = 1).
2. Compute M = max(conv(a,a)).
3. Solve the linear program
        maximise   1ᵀb
        subject to  a * b ≤ M   (component‑wise convolution)
                    b ≥ 0
   The feasible set is a polyhedron; the optimum pushes the mass of b
   away from the locations where a is large, thus decreasing the next
   convolution peak.
4. Rescale b to have unit sum (g = b / 1ᵀb) and replace a by a convex
   combination a ← (1‑t)·a + t·g (t≈0.5).  The new vector has a strictly
   smaller maximal convolution value unless a is already optimal.
5. Repeat steps 2‑4 for a few iterations (time‑limited).
6. Polish the best vector with a cheap greedy hill‑climb on the original
   objective.

All work is performed on the probability simplex, therefore the score
returned by ``evaluate_sequence`` equals ``2·n·max(conv)``.  The routine
runs for at most the supplied time budget (default 1000 s) and returns
the best list of heights found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a probability vector (sum=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a probability vector of length n.
    Modes:
        'uniform' – all equal,
        'edge'    – larger values at the two ends,
        'exp'     – exponential decay,
        'power'   – power‑law decay,
        'random'  – Dirichlet(0.5) sample.
    """
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        # a few large entries at the two ends, tiny interior
        frac = rng.uniform(0.004, 0.02)
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n          # larger than uniform
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Very cheap local optimisation on the simplex.
    Moves a tiny fraction of mass between two random positions if it improves
    the (fast) score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Convolution matrix for the linear program
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> diags:
    """
    Returns a sparse matrix H (shape (2n-1, n)) such that
          H @ b = conv(f, b)   (full linear convolution).
    The i‑th diagonal of H (offset i) contains the constant f[i].
    """
    n = f.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal consists of the same coefficient f[i]
    data = np.outer(f, np.ones(n))             # shape (n, n) – row i = f[i] repeated
    H = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return H


# ----------------------------------------------------------------------
# LP‑based improvement step
# ----------------------------------------------------------------------


def lp_improve(a0: np.ndarray,
               deadline: float,
               max_iter: int = 30,
               t_init: float = 0.5):
    """
    Perform a few alternating LP updates starting from a0.
    Returns the best vector found and its fast_score.
    """
    f = a0.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_score = fast_score(best_f, next_pow2(2 * n - 1), n)
    t = t_init

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current maximal convolution value
        conv = np.convolve(f, f)
        M = conv.max()

        # linear program: maximise sum(b)  s.t.  conv(f,b) ≤ M , b ≥ 0
        H = build_conv_matrix(f)                     # (2n‑1) × n
        c = -np.ones(n)                              # maximise → minimise -∑b
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=H, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – abort the alternating process
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        g = b / sum_b                               # unit‑sum version of b
        new_f = (1.0 - t) * f + t * g
        new_f = np.maximum(new_f, 0.0)
        new_f /= new_f.sum()                         # keep on the simplex

        new_score = fast_score(new_f, next_pow2(2 * n - 1), n)

        if new_score < best_score - 1e-12:
            best_f = new_f.copy()
            best_score = new_score
            f = new_f
            # keep step size, continue improving
            print(f"[LP iter {it}] improved score {best_score:.6f}")
        else:
            # no improvement – shrink the mixing factor
            t *= 0.5
            if t < 1e-4:
                break

    return best_f, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising ``evaluate_sequence``.
    Uses the alternating linear‑program scheme together with a final greedy
    hill‑climb.  The function respects the time budget and returns the best
    list of floats found.
    """
    rng = np.random.default_rng(seed + 123456)

    # ------------------------------------------------------------------
    # Problem size & initial point
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        n = prev.size if prev.size > 0 else 2000
        a0 = prev.copy()
        if a0.sum() > 0.0:
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Multiple short LP‑based runs with different random starts
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:   # keep a few seconds for polishing
        mode = rng.choice(['edge', 'exp', 'power', 'random'])
        a_start = build_initial(mode, n, rng)
        a_opt, score_opt = lp_improve(a_start, deadline,
                                      max_iter=30, t_init=0.5)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[restart] mode={mode} new best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy hill‑climb polish (fast_score uses FFT)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fft_len = next_pow2(2 * n - 1)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, n,
                                        deadline,
                                        max_iter=5000)

    # ------------------------------------------------------------------
    # Final evaluation for reporting
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(best_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # store for possible reuse by a later call
    globals()['height_sequence_1'] = best_a.tolist()
    return best_a.tolist()
