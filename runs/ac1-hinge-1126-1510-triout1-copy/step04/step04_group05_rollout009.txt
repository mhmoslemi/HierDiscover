"""
Propose candidate for the step‑function minimisation.

The search combines:
* a soft‑max parametrisation of the vector,
* a continuation L‑BFGS optimiser on a smooth approximation of the
  max‑convolution (geometric schedule of the smoothing parameter),
* aggressive targeted hill‑climbing that moves mass from indices that
  contribute most to the current peak to indices that contribute least,
* a fast random hill‑climber for polishing,
* a set of diverse initialisations (random, exponential, power‑law,
  left‑biased, right‑biased and a copy of the previous best sequence).

All operations are performed on the probability simplex; the final vector
is returned unchanged – the evaluation function is scale‑invariant.
The routine respects the 1000 s budget and prints intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returns length 2*n‑1 array."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a vector that sums to 1."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max‑convolution and its gradient w.r.t. the logits `y`.

    The smooth approximation uses a log‑sum‑exp with a smoothing
    parameter λ = lam_factor * max(conv).  The gradient is obtained
    analytically via the chain‑rule through the soft‑max.
    """
    a = softmax(y)

    # a*a convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c            # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j} for i = 0..n‑1
    # (equivalent to a * w reversed convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def schedule_optimize(y_start: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_g: int,
                      lam_factors: list[float],
                      maxiter_per_stage: int = 400):
    """
    Continuation of L‑BFGS optimisations.
    Each stage reduces the smoothing factor `lam_factor`.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_stage,
                                'gtol': 1e-14,
                                'disp': False})
        y = res.x
    return y

def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Simple random hill‑climb on the simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_moves: int = 8000,
                        frac: float = 0.07):
    """
    Greedy move that diminishes the current highest convolution peak.
    At each step the index that contributes most to the peak donates a
    small fraction of its mass to the index that contributes least.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to the peak k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback – random move
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[j] -= delta  # bug? Should be i_dst.
            # Correction: revert to original.
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score

def init_logits(mode: str, n: int, rng: np.random.Generator,
                prev_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce an initial logit vector for the soft‑max parametrisation.
    """
    if mode == "random":
        y = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        y = np.log(a + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.1, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        y = np.log(a + 1e-12)
    elif mode == "left":
        # mass concentrated at the left side, monotone decreasing afterwards
        frac = rng.uniform(0.01, 0.08)
        m = max(1, int(n * frac))
        a = np.concatenate([np.linspace(1.0, 0.1, m),
                            np.full(n - m, 0.0)])
        a = np.clip(a, 0, None)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        y = np.log(a + 1e-12)
    elif mode == "right":
        frac = rng.uniform(0.01, 0.08)
        m = max(1, int(n * frac))
        a = np.concatenate([np.full(n - m, 0.0),
                            np.linspace(0.1, 1.0, m)[::-1]])
        a = np.clip(a, 0, None)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        y = np.log(a + 1e-12)
    elif mode == "prev" and prev_seq is not None:
        arr = prev_seq.copy()
        if arr.size < n:
            # pad with zeros at the end
            arr = np.r_[arr, np.zeros(n - arr.size)]
        else:
            arr = arr[:n]
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s > 1e-12:
            arr /= s
        else:
            arr = np.full(n, 1.0 / n)
        y = np.log(arr + 1e-12)
    else:
        y = rng.standard_normal(n)
    return y

"""
Propose candidate for the step‑function minimisation.

The search combines:

* a soft‑max parametrisation of the vector,
* a continuation L‑BFGS optimiser on a smooth approximation of the
  max‑convolution (geometric schedule of the smoothing parameter),
* aggressive targeted hill‑climbing that moves mass from the indices
  that contribute most to the current peak to the indices that
  contribute least,
* a random hill‑climber for final polishing,
* a collection of diverse initialisations (random, exponential,
  power‑law, left‑biased, right‑biased and a copy of the previous best
  sequence).

All manipulations keep the vector on the probability simplex; the
evaluation function is scale‑invariant, so the final vector can be
returned unchanged.  The routine respects the 1000 s budget and prints
intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns the first ``2*n‑1`` entries (the full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector ``a``.
    This matches the value returned by ``evaluate_sequence`` (the
    denominator ``sum(a)**2`` equals 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max‑convolution and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution and its gradient w.r.t.
    the unconstrained logits ``y`` (the log‑softmax coordinates).

    ``lam_factor`` controls the smoothing width λ = lam_factor * max(conv).
    """
    a = softmax(y)

    # ----------- convolution a * a ------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max (log‑sum‑exp) of the convolution values
    shifted = conv - max_c                # now max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----------- gradient of smooth max ------------
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list that gradually reduces the
    smoothing.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climbers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Generic random hill‑climb on the simplex.
    Randomly transfers a fraction of mass from one index to another and
    keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 8000,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb.
    At each step the index contributing most to the present peak
    (k* = argmax conv) donates a small fraction of its mass to the index
    that contributes the least – a move that directly reduces the true
    ``max(a*a)``.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random move if the contribution vector is flat
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert the move
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def init_logits(
    mode: str,
    n: int,
    rng: np.random.Generator,
    prev_seq: np.ndarray | None,
) -> np.ndarray:
    """
    Produce a starting logit vector for the soft‑max parametrisation.
    The function supports several heuristic patterns.
    """
    if mode == "random":
        y = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        y = np.log(a + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.1, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        y = np.log(a + 1e-12)

    elif mode == "left":
        # mass concentrated on the left side with a gentle decay
        frac = rng.uniform(0.01, 0.08)
        m = max(1, int(n * frac))
        # decreasing values followed by zeros
        a = np.concatenate([np.linspace(1.0, 0.1, m), np.zeros(n - m)])
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        y = np.log(a + 1e-12)

    elif mode == "right":
        # mirror of the left‑biased pattern
        frac = rng.uniform(0.01, 0.08)
        m = max(1, int(n * frac))
        a = np.concatenate([np.zeros(n - m), np.linspace(0.1, 1.0, m)[::-1]])
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        y = np.log(a + 1e-12)

    elif mode == "prev" and prev_seq is not None:
        arr = prev_seq.copy()
        if arr.size < n:
            arr = np.r_[arr, np.zeros(n - arr.size)]
        else:
            arr = arr[:n]
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s > 1e-12:
            arr /= s
        else:
            arr = np.full(n, 1.0 / n)
        y = np.log(arr + 1e-12)

    else:
        y = rng.standard_normal(n)

    return y
