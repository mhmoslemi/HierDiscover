def subgradient_optimize_enhanced(a, n, fft_len, deadline, init_step=0.5, min_step=1e-9):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = np.convolve(best, best)
        max_val = conv.max()
        max_idxs = np.where(conv >= max_val - 1e-12)[0]  # tolerance
        # build weight vector w of length 2n-1 where w[s]=1/len(max_idxs) for s in max_idxs
        w = np.zeros_like(conv)
        w[max_idxs] = 1.0 / len(max_idxs)
        # compute gradient via convolution with reversed a
        rev_a = best[::-1]
        grad_full = np.convolve(w, rev_a)
        grad = 2.0 * grad_full[n-1: n-1 + n]
        # add jitter
        grad += rng.normal(scale=1e-5, size=n)
        a_new = best - step * grad
        a_new = project_to_simplex(a_new)
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score
```

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

The algorithm is a multi‑stage optimiser:

1️⃣  Initialise a population of candidates on the probability simplex.
    Besides the standard “random”, “exp”, “power”, “edge” and “sidon”
    seeds we also use an *arcsine* distribution – the discretised
    β(½,½) law – which is a good analytical starting point for flat
    auto‑convolution.

2️⃣  Refine each seed with a cascade of L‑BFGS runs on a smooth‑max
    surrogate of the convolution peak.  The smoothing parameter
    (called *lam*) is decreased geometrically so the optimiser sees
    progressively the true max.

3️⃣  A greedy hill‑climb (mass transfer between two indices) is applied
    after every L‑BFGS pass – this quickly exploits local improvements
    that the gradient step may miss.

4️⃣  An enhanced sub‑gradient descent works directly on the exact
    max‑convolution.  Instead of a single active shift we build a
    convex combination of all shifts that are (almost) maximal and
    project the update onto the simplex.  Momentum‑like step‑size
    adaptation accelerates convergence.

5️⃣  Finally we run a very low‑lam L‑BFGS fine‑tune and a last hill‑climb.

All stages respect the constraints enforced by ``evaluate_sequence``.
The routine respects a 1000‑second wall‑clock budget and returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z   # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict:
                if candidate * 2 in sums:
                    conflict = True
                else:
                    chosen.append(candidate)
                    for c in chosen:
                        sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "arcsine":
        # discretised β(½,½) distribution
        i = np.arange(n, dtype=float)
        vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
        vals = np.clip(vals, 0, None)
        vals /= vals.sum()
        y0 = np.log(vals + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: transfer a random fraction of mass from i to j.
    Accept only strict improvements of the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Enhanced sub‑gradient descent on the exact max‑convolution.
    The sub‑gradient is averaged over all (almost) maximal shifts.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    eps = 1e-12

    while time.time() < deadline:
        conv = np.convolve(best, best)
        max_val = conv.max()
        # indices whose value is within eps of the maximum
        max_idx = np.where(conv >= max_val - eps)[0]
        # build a sparse weight vector over shifts
        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / max_idx.size

        # gradient = 2 * (w * a_rev) on the relevant window
        rev = best[::-1]
        grad_full = np.convolve(w, rev)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # tiny jitter to escape plateaus
        grad += rng.normal(scale=1e-5, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.3, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3.0   # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    # A modest size keeps each iteration cheap while still offering
    # enough degrees of freedom to beat the uniform bound.
    n = 5000
    # if a previous construction exists, reuse its length as a lower bound
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1000:
            n = max(n, int(tmp.size))
            best_a = tmp.copy()
            best_a = np.clip(best_a, 0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_a = np.clip(best_a, 0, None)
    best_a = best_a / best_a.sum()
    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=24))
    maxiter_stage = 350
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "arcsine", "best"]

    # ------------------------------------------------------------------
    # main stochastic L‑BFGS + hill‑climb loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # aggressive local hill‑climb
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=6000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # refined sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 45.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.6
    )
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # final fine L‑BFGS (extremely low smoothing)
    # ------------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(1e-5, 1e-7, num=12))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.1e} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final greedy polishing (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    # store the best sequence globally for possible reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
