"""
Propose a low‑score step‑function.

Algorithm
---------

* Build a small pool of diverse initial candidates (uniform, power‑law,
  exponential, two‑sided power‑law and Beta shapes) with lengths between
  12 k and 30 k.  If a previous solution `height_sequence_1` exists it is
  added to the pool.

* For each candidate run a custom Adam optimiser that works directly on
  the probability simplex (projection onto the simplex after every update).
  The objective uses a *smooth* approximation of the max‑convolution:
      M_α(a) = (1/α)·log(∑_k exp(α·(a∗a)_k))
  whose gradient can be written in closed‑form using a single FFT‑based
  correlation.  The smoothing parameter α is increased during the run
  (5 → 15 → 50 → 200) to gradually turn the smooth surrogate into the
  true max.

* The optimiser records the best `(a,score)` encountered (score is the
  original `evaluate_sequence` value).  A few jitter steps are added
  periodically to escape plateaus.

* The total 1000 s time budget is split among the candidates; the best
  overall sequence is finally verified with the provided `evaluate_sequence`
  and returned as a plain list.

The implementation uses only NumPy and respects the 2‑CPU, no‑I/O
constraints.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (no clipping, same as evaluate_sequence)."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (used for random starts)

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))

# Core operations: convolution and gradient of smoothed max
# ----------------------------------------------------------------------


def linear_conv(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a using FFT."""
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def grad_smooth_max(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of M_α = (1/α)·log Σ exp(α·conv) w.r.t. a.
    Returns a vector of shape a.shape (length n).
    """
    n = a.shape[0]
    # compute soft‑max weights in a numerically stable way
    conv_shift = conv - conv.max()
    w = np.exp(alpha * conv_shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation g_j = 2 * Σ_i a_i * w_{i+j}
    # we compute it via FFT:
    # reverse w and convolve it with a, then extract the required slice.
    w_rev = w[::-1]                     # length 2n‑1
    L = next_pow2(n + w_rev.size - 1)   # ≥ 3n‑2
    Fa = np.fft.rfft(a, n=L)
    Fwr = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(Fa * Fwr, n=L)      # length L
    # valid slice: indices n‑1 … 2n‑2 (inclusive) → length n
    start = n - 1
    slice_ = conv_rev[start : start + n]       # forward order
    g = 2.0 * slice_[::-1]                     # reverse to obtain g_j
    return g

# Optimiser (Adam on the simplex, smooth max surrogate)
# ----------------------------------------------------------------------


def adam_optimize_smooth(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.4,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a probability vector a (sum=1, a_i≥0) to minimise
    the original objective 2·n·max(a∗a)/(sum a)².
    Uses a smoothed max (log‑sum‑exp) during optimisation.
    Returns the best a encountered and its true (unsmoothed) score.
    """
    n = a0.shape[0]
    a = a0.copy()
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        now = time.time()
        if now > start_time + time_limit:
            break
        it += 1

        # full convolution and true score
        conv_full = linear_conv(a)
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- smoothing schedule for α -----
        if it < 2000:
            alpha = 5.0
        elif it < 5000:
            alpha = 15.0
        elif it < 10000:
            alpha = 50.0
        else:
            alpha = 200.0

        # ----- gradient of smoothed max -----
        g = grad_smooth_max(a, conv_full, alpha)

        # ----- Adam update on the unconstrained vector -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        a_tmp = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection back onto the simplex -----
        a = project_simplex(a_tmp)

        # ----- occasional jitter to avoid stagnation -----
        if it % 1500 == 0:
            sigma = 0.02 / np.sqrt(n)
            a = a + sigma * np.random.randn(n)
            a = project_simplex(a)

    return best_seq, best_val

    # ---------------------------------------------------------------
    length_choices = [12000, 15000, 20000, 25000, 30000]
    families = ["power", "exp", "twosided", "beta"]
    desired = 6  # total pool size (including the two above)

    while len(init_candidates) < desired and time.time() < deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.1, 2.0)
            seq = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(1e-4, 1e-2)
            seq = generate_exponential(n, lam)
        elif family == "twosided":
            pl = random.uniform(0.1, 2.0)
            pr = random.uniform(0.1, 2.0)
            wl = random.uniform(0.5, 3.0)
            wr = random.uniform(0.5, 3.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        else:  # beta
            a = random.uniform(0.1, 2.5)
            b = random.uniform(0.1, 2.5)
            seq = generate_beta_shape(n, a, b)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        init_candidates.append(seq.copy())
        print(f"[info] generated {family} candidate (len={n})")

    # ------------------------------------------------------------------
    # 2) Optimise each candidate (time budget split evenly)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left after initialisation")

    per_candidate = remaining / len(init_candidates)
    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        print(
            f"[optim] start candidate #{idx+1}/{len(init_candidates)} "
            f"(len={a0.size}) – budget {per_candidate:.1f}s"
        )
        seq, val = adam_optimize_smooth(a0, per_candidate, time.time())
        print(f"[optim] candidate #{idx+1} best true score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – very simple uniform distribution
        n = 15000
        best_overall_seq = np.full(n, 1.0 / n)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")
    return final_list
