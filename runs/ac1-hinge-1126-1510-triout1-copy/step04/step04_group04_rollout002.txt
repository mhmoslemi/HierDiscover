"""
Propose a step‑function sequence with a low evaluation score.

Approach
--------
1. Initialise the search from the best known construction (if any) or from a
   collection of diverse heuristic seeds (random, exponential, power‑law,
   edge‑focused, Sidon‑set, Gaussian).

2. Re‑optimise the normalised vector using a cascade of L‑BFGS runs on a
   smooth approximation of the max‑convolution.  The schedule progressively
   reduces the smoothing parameter so that the objective converges to the
   true max‑convolution.

3. Perform an extended projected sub‑gradient descent on the exact
   max‑convolution (the true objective) with an adaptive step‑size.

4. Apply a simulated‑annealing phase that repeatedly moves a random
   fraction of probability mass between two positions, accepting worsening
   moves with a temperature that cools to zero.  This helps escape the
   plateaux that sub‑gradient descent may get trapped in.

5. Finish with a short polishing L‑BFGS run at very low smoothing and a
   final greedy hill‑climb.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
so the evaluation score `2·n·max(conv)` computed by ``fast_score`` equals
the value returned by ``evaluate_sequence``.  The routine respects the
1000 s time budget and prints progress messages.

The final sequence (list of floats) is returned ready for the harness.
"""

import time
import numpy as np
from scipy.optimize import minimize
from numpy.fft import fft, ifft

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Quick evaluation of the objective for a probability vector ``a``.
    Returns 2·n·max(a*a) (identical to ``evaluate_sequence`` when sum(a)=1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth‑max version of the objective and its gradient w.r.t. ``y``.
    ``lam_factor`` controls the amount of smoothing (larger → smoother).
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = fft(a, fft_len_conv)
    conv = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of the smooth max --------------------------------------
    w = exp_shift / Z                           # ∂ smooth_max / ∂ conv_i

    # convolution of a with w (the Jacobian of the smooth max)
    fa_rev = fft(a[::-1], fft_len_g)
    fw = fft(w, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    # entries (n‑1 … 2·n‑2) correspond to the n components we need
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule,
    maxiter_per_stage: int = 300,
):
    """
    Run a cascade of L‑BFGS calls with decreasing smoothing.
    ``lam_schedule`` is an iterable of smoothing factors.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Helper for generating diverse starting points
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point (pre‑soft‑max) according to ``mode``."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # put a little extra mass near the two ends
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon‑set construction (distinct pairwise sums)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (cand * 2 not in sums):
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = max(1, len(chosen))
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "gauss":
        # discrete Gaussian centred in the middle
        sigma = rng.uniform(0.08, 0.20) * n
        pos = np.arange(n)
        centre = (n - 1) / 2.0
        a0 = np.exp(-0.5 * ((pos - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (exact max via FFT)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
):
    """
    Greedy hill‑climb: relocate a random fraction of mass from i→j.
    Accept only strict improvements.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.6,
    min_step: float = 1e-9,
):
    """
    Projected sub‑gradient descent on the true objective.
    The step size decays adaptively; a small Gaussian jitter avoids flat stalls.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        # sub‑gradient of max‑convolution at s_star
        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # tiny random jitter to break ties
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best, best_score = a_new, new_score
            step = min(step * 1.2, 2.0)   # accelerate when successful
        else:
            step *= 0.5
            if step < min_step:
                break

    print(f"[subgrad] finished after {iteration} iterations, score={best_score:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing for fine‑grained polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    start_temp: float = 0.01,
    end_temp: float = 1e-6,
):
    """
    Simple annealing scheme that repeatedly moves a random fraction of mass.
    Accepts worsening moves with probability exp(-Δ/T).  Temperature linearly
    decays to ``end_temp``.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # linear schedule based on remaining fraction of time
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = end_temp + (start_temp - end_temp) * (1.0 - frac)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] <= 0.0:
            continue
        # move up to 50 % of the mass at i (helps explore)
        delta = best[i] * rng.random() * 0.5
        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = fast_score(new_a, fft_len, n)
        d = new_score - best_score
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            best, best_score = new_a, new_score

    print(f"[anneal] final annealed score = {best_score:.6f}")
    return best, best_score

        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] n={n}, initial fast score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Stochastic L‑BFGS runs with diverse starts
    # --------------------------------------------------------------
    lam_schedule = np.geomspace(0.4, 0.001, num=18)          # decreasing smoothing
    maxiter_stage = 350                                    # a bit more per stage
    modes = ["random", "exp", "power", "edge", "sidon", "gauss", "best"]
    iteration = 0

    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.025, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage=maxiter_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a, best_score, best_y = a_opt, score_opt, y_opt
            print(f"[{iteration}] mode={mode} improved fast score={best_score:.6f}")

        # short aggressive hill‑climb after each L‑BFGS batch
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=3000
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # 3.  Lengthened projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 120.0)   # give it up to two minutes
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.6
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # 4.  Simulated annealing for fine‑grained escape from local minima
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, anneal_deadline
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # 5.  Final polishing L‑BFGS with very low smoothing
    # --------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_sched = np.geomspace(0.3, 3e-5, num=22)
        for lam in fine_lam_sched:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 500, "gtol": 1e-12, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len_conv, n)
            print(f"[fine L‑BFGS] lam={lam:.2e}, fast score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Last greedy hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # 7.  Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
