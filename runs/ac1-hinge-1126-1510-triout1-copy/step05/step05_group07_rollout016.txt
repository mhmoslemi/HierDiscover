def subgrad_multi_descent(a0, n, fft_len, max_iters, deadline, eta0=0.3, epsilon=1e-8):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    i_arr = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        max_c = conv.max()
        # indices near max
        active = np.where(conv >= max_c - epsilon)[0]
        # compute gradient sum over active indices
        grad = np.zeros_like(a)
        for k in active:
            j = k - i_arr
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]

        # average
        if len(active) > 1:
            grad /= len(active)

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # occasional perturb
        if (t + 1) % 500 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(scale=0.005, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score

def lp_refine(a, n, max_iters, deadline, t=0.2):
    """
    Perform a few iterates of the LP-based refinement described
    in the reference algorithm.
    """
    a_cur = a.copy()
    sum_target = a_cur.sum()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute current max bound
        fft_len = next_pow2(2 * n - 1)
        conv = conv_fft(a_cur, fft_len, n)
        M = conv.max()
        # build convolution matrix for a_cur
        f = a_cur
        offsets = np.arange(n)
        # data for each diagonal
        data = [np.full(2 * n - 1 - off, f[off], dtype=float) for off in offsets]
        A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
        # solve LP: maximize sum(g) => min -1^T g
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        # scale g to have same sum as a_cur
        g_scaled = g * (sum_target / sum_g)
        # convex combination
        a_cur = (1 - t) * a_cur + t * g_scaled
        # projection onto simplex (to correct numeric errors)
        a_cur = project_simplex(a_cur)
    return a_cur

"""
Hybrid optimizer for the step‑function “attack”.

The routine searches for a non‑negative sequence a_0,…,a_{n‑1}
(minimum sum is enforced by normalisation to 1) that minimises

    2·n·max(a∗a) / (∑a)^2

where * denotes the linear convolution.  The score is invariant to a
global scaling, therefore we keep the vector on the probability simplex
throughout the search.

Algorithmic ingredients
-----------------------
1.  Randomised restarts with diverse initialisations (uniform,
    exponential, power‑law, edge‑biased, perturbations of the current
    best).
2.  Continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution.
3.  Projected sub‑gradient descent on the *exact* max‑convolution.
    The sub‑gradient is averaged over all active (near‑maximal) shift
    indices for a more stable descent direction.
4.  Two cheap hill‑climbs:
        – a greedy “targeted” move that shifts mass from the largest
          contributor to the smallest contributor of the current peak,
        – a random mass‑transfer hill‑climb.
5.  A few LP‑based refinements (the linear programme described in the
    problem statement) which, for a fixed f, finds a vector g with the
    largest possible ℓ₁‑norm under the same convolution bound.
    Mixing f and g with a small step size further reduces the bound.
6.  A final low‑λ L‑BFGS polishing phase.

All steps respect the simplex constraint by Euclidean projection.
The code runs for at most the supplied budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i ≥ 0)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_multi_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
    eps_rel: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).

    Uses an averaged sub‑gradient over all (near‑)active convolution
    indices to obtain a more stable descent direction.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        max_c = conv.max()
        # indices whose value is within eps_rel of the maximum
        eps = max(eps_rel * max_c, 1e-12)
        active = np.where(conv >= max_c - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            j = k - i
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        if len(active) > 0:
            grad /= len(active)

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        # keep the best solution seen so far
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional perturbation to escape plateaus
        if (t + 1) % 500 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random and targeted hill‑climbs (same logic as the reference code)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2500,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the largest contributor to the
    smallest contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming based refinement (the “LP‑step” described in the
# reference attack).
# ----------------------------------------------------------------------


def lp_refine(
    a: np.ndarray,
    n: int,
    max_iters: int,
    deadline: float,
    t: float = 0.2,
) -> np.ndarray:
    """
    Perform a few iterations of the LP‑based refinement:

        maximise   Σ_j g_j
        subject to  (a * g)_k ≤ M   for all k,
                    g_j ≥ 0,
        where M = max(a * a).

    The feasible g is then rescaled to have the same ℓ₁‑norm as a and mixed
    with a using a convex combination with weight t.
    """
    a_cur = a.copy()
    sum_target = a_cur.sum()
    fft_len = next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a_cur, fft_len, n)
        M = conv.max()

        # Build the (2n‑1)×n convolution matrix for a_cur.
        # Column j contains a_cur shifted down by j positions.
        offsets = np.arange(n)
        data = [np.full(2 * n - 1 - off, a_cur[off], dtype=float) for off in offsets]
        A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

        # Solve: maximise Σ g_j  ⇔ minimise -Σ g_j
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=(0, None),
            method="highs",
        )
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Rescale to keep the ℓ₁‑norm constant and convex‑combine.
        g_scaled = g * (sum_target / sum_g)
        a_cur = (1 - t) * a_cur + t * g_scaled
        a_cur = project_simplex(a_cur)  # numerical safety

    return a_cur


# ----------------------------------------------------------------------
# Optimisation driver for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length *n* within the allocated time slice.
    Returns the best probability vector (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ---- initialise ----------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # smoothing schedule for L‑BFGS (more stages → smoother continuation)
    lam_factors = np.geomspace(0.5, 5e-6, num=30)

    iteration = 0

    while time.time() < deadline - 5.0:
        # ---- random initialisation mode --------------------------------
        mode = rng.choice(["uniform", "exp", "power", "edge", "best"])
        if mode == "uniform":
            y0 = np.log(np.full(n, 1.0 / n) + 1e-12)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ---- 1) continuation L‑BFGS -----------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=250,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS score = {best_score:.6f}")

        # ---- 2) multi‑active projected sub‑gradient descent ------------
        sub_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = subgrad_multi_descent(
            best_a,
            n,
            fft_len,
            max_iters=2500,
            deadline=sub_deadline,
            eta0=0.36,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- 3) targeted hill‑climb ------------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=4000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # ---- 4) random hill‑climb (polish) ----------------------------
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ---- final low‑λ L‑BFGS polishing ---------------------------------
    while time.time() < deadline - 1.5:
        for lam in lam_factors:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 500, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.4g} score = {best_score:.6f}")

    # ---- LP‑based refinement (few iterations) -------------------------
    lp_deadline = min(deadline, time.time() + 2.0)
    best_a = lp_refine(best_a, n, max_iters=4, deadline=lp_deadline, t=0.25)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[LP‑refine] final score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low evaluation
    score.  The function returns the best sequence (as a list of floats)
    found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # 2 s safety margin

    # Use the longest reasonable size – larger n usually gives a better
    # constant factor in the score.
    n = 4000  # feel free to bump to 5000/6000 if time permits
    remaining = total_deadline - time.time()
    if remaining < 5.0:
        raise RuntimeError("Insufficient time budget.")

    prev = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev, dtype=float) if prev is not None else None

    print(f"[main] Optimising length n={n} with budget ≈{remaining:.1f}s")
    best_a, best_score = optimise_one_size(
        n=n,
        budget_s=remaining,
        seed=seed,
        prev_seq=prev_arr,
    )
    print(f"[main] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence for a possible warm‑start in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
