"""
Propose a low‑score step‑function sequence.

The algorithm is a heavier‑weight variant of the “hybrid optimizer”
used previously.  It concentrates the whole 1000 s budget on a single
length (n = 4000) and performs many restarts of a smooth log‑sum‑exp
approximation of the max‑convolution, followed by a long projected
sub‑gradient descent, targeted and random hill‑climbs, and a final
low‑λ L‑BFGS refinement.  The schedule for the smooth max (λ) is
more gradual than before, and each optimisation stage is allowed more
iterations.  All updates keep the vector on the probability simplex,
so the evaluation function is invariant to scaling.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using real FFT, trimmed to length 2n‑1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
):
    """
    Return smoothed objective 2·n·smooth_max(a*a) and its gradient
    with respect to logits y, where a = softmax(y).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # max shifted to zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv (length 2n‑1)

    # g_i = Σ_j a_j * w_{i+j}   (i = 0..n‑1)
    # implement via convolution of a (reversed) with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.38,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).

    Returns the best vector encountered and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)

        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if (t + 1) % 300 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climbs
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2500,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.09

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3500,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Greedy move from the strongest contributor to the weakest
    of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Main optimisation driver (single problem size)
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length n within the given time slice.
    Returns the best probability vector and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ----------- initial vector ---------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # smoother schedule – more stages, finer final λ
    lam_factors = np.geomspace(1.0, 5e-8, num=45)

    iteration = 0

    while time.time() < deadline - 6.0:
        # ------- random initialisation of logits --------------------
        mode = rng.choice(["uniform", "random", "exp", "power", "edge", "best"])
        if mode == "uniform":
            y0 = np.log(np.full(n, 1.0 / n) + 1e-12)
        elif mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.018)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.008, 0.035)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.06, size=n)

        # 1) continuation L‑BFGS on smooth max‑approximation
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_score:.6f}")

        # 2) exact sub‑gradient descent (more iterations)
        sub_deadline = min(deadline, time.time() + 1.2)  # ~1 s per sub‑run
        best_a, best_score = subgrad_descent(
            best_a,
            n,
            fft_len,
            max_iters=4500,
            deadline=sub_deadline,
            eta0=0.38,
        )
        best_y = np.log(best_a + 1e-12)

        # 3) targeted hill‑climb
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=3500,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # 4) random hill‑climb for polishing
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final low‑λ L‑BFGS refinement (use remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.2:
        for lam in lam_factors:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 650, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.3g} fast score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with minimal
    evaluate_sequence() score.  The entire time‑budget is spent on a
    single problem size (n = 4000) with many restarts and a
    fine‑grained continuation L‑BFGS schedule, followed by
    projected sub‑gradient descent, targeted and random hill‑climbs,
    and a final low‑λ L‑BFGS polishing phase.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    total_deadline = start + budget_s - 2.0  # safety margin

    # problem size – the larger the better, but 4000 is a good compromise
    n = 4000
    remaining = total_deadline - time.time()
    if remaining < 5.0:
        # too little time – fall back to a uniform sequence
        seq = [1.0 / n] * n
        return seq

    # warm‑start with any previously stored sequence
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)

    print(f"[main] Optimising length n={n} with budget ≈{remaining:.1f}s")
    best_a, best_score = optimise_one_size(
        n=n,
        budget_s=remaining,
        seed=seed,
        prev_seq=prev_arr,
    )
    print(f"[main] best fast score after optimisation = {best_score:.6f}")

    # final evaluation with the official function
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returns {final_val:.6f}")

    # expose for potential reuse in subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
