def targeted_hill_climb(a, current_score, fft_len, n, deadline, max_iter=5000):

    best = a.copy()

    best_score = current_score

    rng = np.random.default_rng()

    it = 0

    while time.time() < deadline and it < max_iter:

        # compute conv, max index, active range
        conv = full_conv(best, fft_len)

        k = conv.argmax()

        max_val = conv[k]

        # active range
        lo = max(0, k - (n - 1))

        hi = min(k, n - 1)

        i_range = np.arange(lo, hi+1)

        j_range = k - i_range

        # compute contributions (full impact on conv[k]):

        contrib = best[i_range] * best[j_range]  # product each pair (i,j)

        # For i!=j, the product appears twice in conv k, but we just want to find large ones

        # find index i_star with max contrib

        idx = np.argmax(contrib)

        i_star = i_range[idx]

        # choose an index outside the active range with minimal value

        # outside indices:

        outside_mask = np.ones(n, dtype=bool)

        outside_mask[lo:hi+1] = False

        # If all indices are inside (possible when n small? not possible for interior k), but we cover.

        if outside_mask.any():

            # pick index with smallest a, maybe far

            cand_out = np.where(outside_mask)[0]

            # choose index with smallest a (maybe to avoid increasing other conv peaks)

            j_star = cand_out[np.argmin(best[cand_out])]

        else:

            # fallback random index distinct from i_star

            j_star = (i_star + rng.integers(1, n)) % n

        # determine delta to move from i_star to j_star

        # delta limited to some fraction of a[i_star] (max 5% or 0.02)

        max_delta = min(best[i_star], 0.1 * best[i_star])

        if max_delta <= 0:

            it += 1; continue

        delta = max_delta * rng.random()

        # apply move

        trial = best.copy()

        trial[i_star] -= delta

        trial[j_star] += delta

        # evaluate

        new_score = fast_score(trial, fft_len, n)

        if new_score < best_score:

            best = trial

            best_score = new_score

        # else ignore

        it += 1

    return best, best_score

"""
Propose a low‑score non‑negative step sequence.

The search proceeds in several stages:
  1. A quick random sampling to obtain a good starting shape.
  2. Continuation L‑BFGS on a smooth log‑sum‑exp (soft‑max) surrogate.
  3. Projected sub‑gradient descent on the exact max‑convolution.
  4. A “targeted” hill‑climb that moves mass away from the
     most‑active convolution index.
  5. A short simulated‑annealing polish.

All stages respect the simplex constraint (non‑negative, sum = 1) and
use fast FFT‑based convolution.  The routine runs for roughly 1000 s
and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution
    already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective and its gradient
# -------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # derivative w.r.t. a

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# -------------------------------------------------------------
# L‑BFGS continuation on the smooth surrogate
# -------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)               # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:   # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# -------------------------------------------------------------


def projected_subgrad_descent(a_start: np.ndarray,
                             deadline: float,
                             fft_len: int,
                             n: int):
    """
    Sub‑gradient descent on the exact max‑convolution
    with Euclidean projection onto the simplex.
    Returns the best sequence observed.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta0 = 1.0                     # initial step‑size factor

    while time.time() < deadline:
        conv = full_conv(a, fft_len)
        k = int(conv.argmax())                     # index of the current max

        # sub‑gradient of max(conv) w.r.t. a
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_idx = np.arange(lo, hi + 1)
        j_idx = k - i_idx
        g = np.zeros_like(a)
        g[i_idx] = 2.0 * a[j_idx]                  # exact sub‑gradient

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # gradient step + simplex projection
        a = a - eta * g
        a = simplex_projection(a)

        # evaluation
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# -------------------------------------------------------------
# Targeted hill‑climb that moves mass away from the active index
# -------------------------------------------------------------


def targeted_hill_climb(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 5000):
    """
    Greedy hill‑climb that explicitly reduces the contribution of the
    currently maximal convolution entry.  Only strictly improving moves
    are kept.
    """
    best = a_start.copy()
    best_score = start_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = full_conv(best, fft_len)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        i_idx = np.arange(lo, hi + 1)
        j_idx = k - i_idx
        # contribution of each i to conv[k] (single product)
        contrib = best[i_idx] * best[j_idx]

        # pick the index with largest contribution
        i_star = i_idx[np.argmax(contrib)]

        # choose a destination outside the active interval with smallest mass
        outside = np.ones(n, dtype=bool)
        outside[lo:hi + 1] = False
        if outside.any():
            cand = np.where(outside)[0]
            j_star = cand[np.argmin(best[cand])]
        else:
            # fallback: any different index
            j_star = (i_star + rng.integers(1, n)) % n

        # move a modest fraction of the mass
        max_move = min(best[i_star], 0.05 * best[i_star])
        if max_move <= 0.0:
            it += 1
            continue
        delta = max_move * rng.random()

        trial = best.copy()
        trial[i_star] -= delta
        trial[j_star] += delta

        new_score = fast_score(trial, fft_len, n)
        if new_score < best_score:
            best = trial
            best_score = new_score
        it += 1
    return best, best_score


# -------------------------------------------------------------
# Simple simulated annealing (used as a final polish)
# -------------------------------------------------------------


def random_anneal(a_start: np.ndarray,
                  start_score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a_start.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = start_score
    rng = np.random.default_rng()
    T = 0.02  # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T *= 0.92
            T = max(T, 1e-5)
    return best_a, best_score


# -------------------------------------------------------------
# Random initialiser (a mixture of several families)
# -------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sine', 'halfsine'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    else:  # halfsine
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

    """

    # -------------------- problem size --------------------
    max_n = 4000                      # upper bound on length we are willing to try

    best_a = a0.copy()

    return final_seq
