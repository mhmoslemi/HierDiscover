def adam_optimize(y_init, n, fft_len_conv, fft_len_g, lam_factor_start, lam_factor_end, steps):
    y = y_init.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    for t in range(1, steps+1):
        lam_factor = lam_factor_start * ((lam_factor_end/lam_factor_start) ** ((t-1)/(steps-1))) # geometric schedule
        loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        y = y - 0.1 * m_hat/(np.sqrt(v_hat)+eps)
    return y
```

"""
Algorithm: multi‑restart Adam + geometric λ‑schedule + aggressive hill‑climbs.
---------------------------------------------------------------------------
* Initialise several log‑soft‑max vectors `y` (different shapes: random,
  exponential, power‑law, perturbations of the current best).
* For each start run Adam on a smoothed version of the max‑convolution
  (smooth‑max λ = lam_factor·max + 1e‑12) with a geometric continuation
  Schedule `lam_factor` from 0.5 down to ~1e‑9.  The gradient w.r.t. `y`
  (including the soft‑max Jacobian) is supplied by `objective_and_grad`.
* Keep the best probability vector `a = softmax(y)`.
* After each Adam phase perform a short targeted hill‑climb that moves mass
  away from the indices contributing most to the current maximal
  convolution entry, followed by a classic random hill‑climb.
* When the time budget is almost exhausted, run a few iterations of
  projected‑gradient descent on the (almost) true max‑convolution
  (λ → 0) and a final unrestricted hill‑climb.
* The sequence is stored globally as ``height_sequence_1`` for reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of the official evaluation (∑a = 1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (gradient w.r.t. log‑soft‑max parameters y)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Loss = 2·n·smooth_max(conv) and its gradient w.r.t. y."""
    a = softmax(y)

    # convolution a*a (full, linear) via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max via log‑sum‑exp
    shifted = conv - max_c                     # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    loss = 2.0 * n * smooth_max

    # d(smooth_max)/d conv
    w = exp_shift / Z

    # back‑prop to a (g_vec = ∂smooth_max/∂a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ∂loss/∂a = 2·n·g_vec
    grad_a = g_vec * (2.0 * n)

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean)

    return float(loss), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser with a λ‑continuation schedule
# ----------------------------------------------------------------------


def adam_schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    iters_per_lambda: int = 150,
) -> np.ndarray:
    """Run Adam on the smoothed objective for every λ in the schedule."""
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    lr = 0.10  # base learning rate for the log‑soft‑max parameters

    for lam in lam_factors:
        for _ in range(iters_per_lambda):
            loss, grad = objective_and_grad(
                y, n, fft_len_conv, fft_len_g, lam_factor=lam
            )
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


# ----------------------------------------------------------------------
# Hill‑climbs
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random walk on the simplex, keep only strict improvements."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """Reduce the current maximal convolution entry by moving mass away from its
    biggest contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift with maximal value

        # contributions of each index to that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]
        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift (used to pick a harmless target)
        partner_weight = np.full(n, np.inf)
        in_range = mask
        partner_weight[in_range] = best[partner[in_range]]

        # choose a random target where the partner weight is small
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Gradient of the (almost) true max‑convolution (λ → 0)
# ----------------------------------------------------------------------


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of smooth‑max convolution w.r.t. a (without the outer 2·n factor).
    Used in the final projected‑gradient polishing stage.
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]
    return 2.0 * g  # derivative of 2·n·smooth_max w.r.t. a (outer 2·n is applied later)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence (non‑negative, sum normalised) with a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, budget_s - 2.0)

    # ------------------------------------------------------------------
    # Choose a problem size – reuse a previous construction if present
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 5000                     # a reasonable trade‑off between quality & time
        best_a = np.full(n, 1.0 / n)

    # Normalise to a probability vector
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths used throughout the run
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Main optimisation loop (random restarts → Adam → hill‑climbs)
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0:
        # ----- initialise a new log‑soft‑max vector y0 -----
        mode = rng.choice(["random", "exp", "power", "prev"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.0005, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 3.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb current best
            y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)

        # ----- Adam optimisation with a λ‑continuation schedule -----
        lam_factors = np.geomspace(0.5, 1e-9, num=70)   # finer schedule than before
        y_opt = adam_schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            iters_per_lambda=150,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- targeted hill‑climb (focus on the current peak) -----
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1500,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- random hill‑climb (helps escape tiny plateaus) -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=5000,
            max_move_frac=0.15,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing – projected gradient descent on the (almost) true max
    # ------------------------------------------------------------------
    lam_factor_pg = 1e-6
    pg_iter = 0
    while time.time() < deadline - 2.0 and pg_iter < 2500:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)   # length‑n gradient
        step = 0.30 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            lam_factor_pg = max(lam_factor_pg * 0.9, 1e-9)

    print(f"[pg] iterations={pg_iter}, best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final unrestricted hill‑climb with any remaining time budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.22,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
