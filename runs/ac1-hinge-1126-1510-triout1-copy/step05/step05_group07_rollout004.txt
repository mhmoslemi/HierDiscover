def exp_grad_descent(a0, n, L, deadline, max_iters_per_restart, η0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, L, n)
    for t in range(max_iters_per_restart):
        if time.time() > deadline:
            break
        # compute conv
        conv = conv_fft(a, L, n)
        k = int(np.argmax(conv))
        # compute gradient components
        # generate index array i=0..n-1
        i = np.arange(n)
        idx = k - i
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2 * a[idx[mask]]
        # step size
        η = η0 / np.sqrt(t + 1)
        # exponentiated update: a = a * np.exp(-η * grad)
        # to avoid overflow/underflow, shift exponent by max
        # but we can simply compute exp(-η * grad)
        a = a * np.exp(-η * grad)
        # renormalize
        sum_a = np.sum(a)
        if sum_a <= 0:
            # fallback to uniform
            a = np.full(n, 1.0 / n)
        else:
            a /= sum_a
        # compute score
        score = fast_score(a, L, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def exp_gradient_descent(a_init, n, L, time_remaining, max_iters=50000, eta0=0.5):

   a = a_init.copy()
   best_a = a.copy()
   best_score = fast_score(a, L, n)
   i = np.arange(n)
   start = time.time()
   t = 0
   while t < max_iters and time.time() - start < time_remaining:
       conv = conv_fft(a, L, n)
       k = int(np.argmax(conv))
       idx = k - i
       mask = (idx >=0) & (idx < n)
       grad = np.zeros_like(a)

           grad[mask] = 2 * a[idx[mask]]

       eta = eta0 / np.sqrt(t + 1)
       # exponentiated update
       a = a * np.exp(-eta * grad)

       sum_a = a.sum()
       a = a / sum_a

"""
Hybrid exponential‑gradient + random hill‑climb optimizer for the
step‑function attack.

The algorithm works exclusively on the simplex (∑a_i = 1, a_i ≥ 0) because
the evaluation score is invariant under a global scaling of the
coefficients.  For a given length ``n`` we repeatedly

1. initialise a probability vector with a diverse family of seeds,
2. minimise the true objective  f(a)=2·n·max(a*a) by mirror descent
   (exponential‑gradient updates on the log‑domain) using the exact
   max‑convolution obtained by an FFT,
3. polish the result with a cheap random mass‑transfer hill‑climb.

All steps are timed so that the total wall‑clock budget (default 1000 s)
is respected; the best sequence found over the allocated time is finally
passed to ``evaluate_sequence`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·n‑1) using a real FFT.
    ``fft_len`` must be a power of two ≥ 2·n‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact value of 2·n·max(a*a) for a probability vector ``a``.
    This is the same quantity that ``evaluate_sequence`` returns
    when the sum of ``a`` equals 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Mirror‑descent (exponential‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_descent(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    eta0: float = 0.5,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the exact sub‑gradient of the
    non‑smooth objective f(a)=2·n·max(a*a).

    Parameters
    ----------
    a_start :
        Initial probability vector (sum = 1).
    n :
        Length of the vector.
    fft_len :
        FFT length (power of two ≥ 2·n‑1).
    deadline :
        Unix time after which the routine must stop.
    eta0 :
        Initial step‑size scaling factor (the schedule is η_t = η0/√t).
    max_iters :
        Upper bound on the number of inner iterations.

    Returns
    -------
    best_a, best_score
        The best vector observed and its fast_score.
    """
    a = a_start.copy()
    # work in the log‑domain for numerical stability
    log_a = np.log(a + 1e-30)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)                     # constant index array

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # current convolution and the active shift
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a (2·a_{k‑i} for feasible i)
        idx = k - i
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        if mask.any():
            grad[mask] = 2.0 * a[idx[mask]]

        # step‑size schedule
        eta = eta0 / np.sqrt(t + 1.0)

        # mirror‑descent update on the log‑variables
        log_a -= eta * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            # numerical safety – fall back to uniform
            a = np.full(n, 1.0 / n)
            log_a = np.log(a)

        # keep the best solution seen so far
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def hill_climb_random(
    a: np.ndarray,
    n: int,
    fft_len: int,
    max_moves: int = 3000,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small fraction of mass from one entry to another."""
    if rng is None:
        rng = np.random.default_rng()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    max_move_frac = 0.10

    for _ in range(max_moves):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Optimisation driver for a single length ``n``
# ----------------------------------------------------------------------


def optimise_one_size(n: int, total_deadline: float, seed: int) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length ``n`` within the remaining
    time budget (``total_deadline``).  Returns the best probability vector
    (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    fft_len = next_pow2(2 * n - 1)

    # start from the uniform distribution
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    restart_counter = 0
    while time.time() < total_deadline - 0.5:   # keep a small safety margin
        # ----- choose a diverse initialisation ---------------------------------
        mode = rng.choice(
            ["uniform", "dirichlet", "exp", "power", "edge", "perturb"]
        )
        if mode == "uniform":
            a0 = np.full(n, 1.0 / n)
        elif mode == "dirichlet":
            alpha = rng.uniform(0.1, 2.0)
            a0 = rng.dirichlet(np.full(n, alpha))
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
        elif mode == "power":
            p = rng.uniform(0.5, 2.5)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
        elif mode == "edge":
            # concentrate a modest amount at the two edges
            edge_len = max(1, int(rng.integers(1, max(2, int(0.03 * n)))))
            edge_frac = rng.uniform(0.1, 0.3)   # total mass placed on edges
            interior_len = n - 2 * edge_len
            a0 = np.empty(n, dtype=float)
            a0[:edge_len] = (edge_frac / 2) / edge_len
            a0[-edge_len:] = (edge_frac / 2) / edge_len
            if interior_len > 0:
                a0[edge_len:-edge_len] = (1.0 - edge_frac) / interior_len
        else:  # "perturb"
            log_best = np.log(best_a + 1e-12)
            log_pert = log_best + rng.normal(scale=0.1, size=n)
            a0 = np.exp(log_pert)
            a0 /= a0.sum()

        # ----- allocate a slice of the remaining budget for this restart --------
        now = time.time()
        remaining = total_deadline - now
        # give the descent at most 5 s (or half the remaining time, whichever is smaller)
        descent_budget = min(5.0, remaining * 0.5)

        # ----- exponential‑gradient descent on the exact objective ---------------
        a_opt, score_opt = exp_gradient_descent(
            a_start=a0,
            n=n,
            fft_len=fft_len,
            deadline=now + descent_budget,
            eta0=0.5,
            max_iters=200_000,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(
                f"[{restart_counter}] init={mode}  exp‑grad best_score={best_score:.6f}"
            )

        # ----- short random hill‑climb for polishing ---------------------------
        # allocate the leftover fraction of the slice for polishing
        polishing_time = total_deadline - time.time() - 0.3
        if polishing_time > 0.2:
            moves = int(polishing_time * 400)  # rough moves per second
            best_a, best_score = hill_climb_random(
                best_a, n, fft_len, max_moves=moves, rng=rng
            )
            print(
                f"[{restart_counter}] after hill‑climb best_score={best_score:.6f}"
            )

        restart_counter += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
