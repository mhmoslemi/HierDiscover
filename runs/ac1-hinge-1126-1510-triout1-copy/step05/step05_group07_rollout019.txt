def lp_refine(a, M, t=0.5):
    n = len(a)
    # build A matrix
    rows = np.concatenate([np.arange(j, j+n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return original a
        return a
    g = res.x
    # Normalize to sum 1
    s = g.sum()
    if s <= 0:
        return a
    g_norm = g / s
    # convex combo
    a_new = (1 - t) * a + t * g_norm
    # ensure nonneg and sum=1
    a_new = project_simplex(a_new)
    return a_new
```

"""
Hybrid step‑function optimiser.

The algorithm combines three complementary ingredients:

1. **Projected sub‑gradient descent** on the true non‑smooth objective
   `F(a)=2·n·max(a*a)` (the “fast” score).  This quickly drives the
   maximal convolution downward while keeping the vector on the simplex.

2. **Linear‑programming refinement** – given the current step‑function `f`,
   we solve
        maximise   Σ g_j
        subject to (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,
   where `M = max(f*f)`.  The LP pushes mass into indices where `f`
   is small, thereby reducing the future maximal convolution.  The
   solution `g` is normalised to a probability vector and mixed with `f`
   (convex combination).  Repeating this a few times refines the shape
   far more aggressively than random hill‑climbs alone.

3. **Targeted and random hill‑climbs** for a final polishing phase.

All updates stay on the probability simplex (`Σ a_i = 1, a_i ≥ 0`);
the final sequence is passed directly to the supplied `evaluate_sequence`
function.  The routine automatically respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σ x = 1} (Wang‑Carreira‑Perpiñán, 2013)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max at index k
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional perturbation to escape plateaus
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement step
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray, M: float, t: float = 0.5) -> np.ndarray:
    """
    Given a probability vector a and the current bound M = max(a*a),
    solve the LP maximise Σ g_j  s.t. (a * g)_k ≤ M and g ≥ 0,
    then mix the (normalised) solution back into a.
    """
    n = len(a)
    # Build the convolution matrix A where (A @ g)_k = (a * g)_k
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)                       # a_i appears on the i‑th diagonal
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                            # maximise Σ g  <=> minimise -Σ g
    bounds = [(0, None)] * n

    # solve – the “highs” solver is fast for the size we use (n≈4000)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # fallback – return the original vector unchanged
        return a

    g = res.x
    s = g.sum()
    if s <= 0:
        return a
    g_norm = g / s                              # bring back to a probability vector

    # convex combination (t controls how aggressively we move towards g)
    a_new = (1.0 - t) * a + t * g_norm
    a_new = project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climb helpers (exact same logic as the reference code)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Greedy climb that re‑balances the biggest contributors of the current peak."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to random move
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score

# Main optimisation driver (single size)
# ----------------------------------------------------------------------

    """

        if s > 1e-12:

        else:
            a = np.full(n, 1.0 / n)
    else:

    fft_len = next_pow2(2 * n - 1)
    best_a = a.copy()

    iteration = 0
    while time.time() < deadline - 2.5:
        # --- diverse initialisations -------------------------------------------------
        mode = rng.choice(["uniform", "exp", "power", "edge", "random"])
        if mode == "uniform":

        elif mode == "exp":

        elif mode == "power":

        else:  # random Dirichlet‑like

        a = a0.copy()

        # --- 1) sub‑gradient descent (strong non‑smooth optimiser) ------------------
