"""
Propose a low‑evaluation non‑negative step function.

The algorithm follows several complementary phases within the 1000 s limit:

1.  **Parametric seed sweep** – generate a few hundred simple families
    (uniform, linear, exponential, power‑law, edge‑heavy, sinusoid,
    Gaussian, Beta‑shaped, random Dirichlet) and keep the best few seeds.

2.  **Continuation L‑BFGS** – optimise a smooth log‑sum‑exp approximation
    of the max‑convolution.  A decreasing λ‑schedule is used; after each
    λ stage a short hill‑climb is performed.

3.  **Projected sub‑gradient descent** – directly minimise the true
    objective 2·n·max(a∗a) with a back‑tracking line‑search.

4.  **Simulated annealing** – a stochastic mass‑transfer walk that
    occasionally accepts uphill moves, allowing the search to escape
    shallow local minima.

5.  **Final greedy hill‑climb** – fine polish using tiny random
    mass transfers.

All phases respect the simplex constraint (non‑negative entries,
sum = 1) and use FFT‑based convolutions (O(n log n)).  The best sequence
found before the deadline is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a∗a) for a already normalised to sum = 1."""
    # linear convolution via real FFT
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a (full length 2·n‑1) via FFT
    fa = rfft(a, fft_len_conv)
    conv_full = irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c          # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i

    # compute gradient w.r.t. a (using convolution with reversed a)
    a_rev = a[::-1]
    fw = rfft(w, fft_len_g)
    fa_rev = rfft(a_rev, fft_len_g)
    conv_rev = irfft(fa_rev * fw, fft_len_g)
    # central n entries of the reversed convolution give ∂conv/∂a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple λ levels)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous one.  Returns the best vector found.
    """
    y = np.log(a_start + 1e-12)          # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # occasional numerical difficulties – ignore this λ
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # very short hill‑climb after each λ (quick refinement)
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=800)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent
    with a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set (indices attaining the max)
        fa = rfft(a, fft_len_conv)
        conv = irfft(fa * fa, fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (exact)
        a_rev = a[::-1]
        fm = rfft(mask, fft_len_g)
        fa_rev = rfft(a_rev, fft_len_g)
        conv_g = irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line‑search on the simplex
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # move at most 5 % of the source mass
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta
        # may introduce tiny negative due to rounding – correct
        if best[i1] < 0.0:
            best[i1] = 0.0
            best[i2] = best.sum()   # re‑normalise (rare case)
            best /= best.sum()
        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta
        i += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer walk with uphill acceptance)
# ----------------------------------------------------------------------
def simulated_anneal(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     fft_len: int,
                     n: int):
    """
    Stochastic annealing on the exact objective.
    Moves a small amount of mass between two random coordinates,
    accepting uphill changes with probability exp(-Δ/T).
    """
    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    it = 0

    while time.time() < deadline:
        # Linear cooling schedule
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time + 1e-12)
        T = 0.001 * (1.0 - frac) + 1e-9   # temperature decays to ~0

        # Choose a source with non‑negligible mass
        i = rng.integers(0, n)
        if cur[i] <= 1e-12:
            it += 1
            continue
        j = rng.integers(0, n)
        if i == j:
            it += 1
            continue

        # Fraction of source mass to move – larger early, smaller later
        max_frac = max(0.02, 0.2 * (1.0 - frac))
        delta = cur[i] * rng.random() * max_frac
        if delta <= 0.0:
            it += 1
            continue

        # Propose move
        cur[i] -= delta
        cur[j] += delta

        # Occasionally re‑normalise to combat drift
        if it % 5000 == 0:
            cur /= cur.sum()

        new_score = fast_score(cur, n, fft_len)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill acceptance
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation metric.  Returns a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------------
    # Problem size
    # ---------------------------------------------------------------
    max_n = 2500                     # safe upper bound for LP‑free run
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0[a0 < 0] = 0.0
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    # ---------------------------------------------------------------
    # Initial cheap parametric sweep (collect seeds)
    # ---------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len_conv)
        candidates.append((sc, arr))

    # Uniform
    add_candidate(np.full(n, 1.0 / n))

    # Linear decreasing / increasing
    add_candidate(np.arange(n, 0, -1, dtype=float))
    add_candidate(np.arange(1, n + 1, dtype=float))

    # Exponential families (decay & growth)
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))          # decay
        add_candidate(np.exp(-rate * np.arange(n)[::-1]))   # growth

    # Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))
        add_candidate((np.arange(1, n + 1, dtype=float) ** (p)))

    # Edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # Gaussian‑like (centered)
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((np.arange(n) - (n - 1) / 2) / sigma) ** 2)
    add_candidate(gauss + 1e-4)

    # Beta‑shaped families
    xs = (np.arange(n) + 0.5) / n
    for a_param, b_param in [(0.5, 0.5), (0.7, 0.7), (1.0, 2.0), (2.0, 1.0),
                             (1.5, 1.5), (0.8, 1.2), (1.2, 0.8)]:
        beta_arr = (xs ** (a_param - 1)) * ((1 - xs) ** (b_param - 1))
        add_candidate(beta_arr + 1e-4)

    # Random Dirichlet / Gamma draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(6):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Keep a few best seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:8]]
    best_a = top_seeds[0].copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    print(f"[init] n={n}, initial fast_score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Timing budget split
    # ---------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    phase0_end = start_time + 0.07 * budget_s   # sweep (already done)
    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.75 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.95 * budget_s   # simulated annealing
    # remaining time for final hill‑climb

    # ---------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ---------------------------------------------------------------
    lam_schedule = [0.8, 0.6, 0.4, 0.2, 0.1, 0.08, 0.05,
                    0.03, 0.02, 0.01, 0.008, 0.005,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 400
    restarts = 0
    max_restarts = 30

    while time.time() < phase1_end and restarts < max_restarts:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

        restarts += 1

    # ---------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # ---------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.1)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase 3 – simulated annealing
    # ---------------------------------------------------------------
    if time.time() < phase3_end:
        sa_deadline = min(phase3_end, deadline - 0.1)
        best_a, best_score = simulated_anneal(best_a,
                                              best_score,
                                              sa_deadline,
                                              fft_len_conv,
                                              n)
        print(f"[anneal] score={best_score:.6f}")

    # ---------------------------------------------------------------
    # Final hill‑climb polishing (as much time as left)
    # ---------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=20000)

    # ---------------------------------------------------------------
    # Return the sequence (plain list)
    # ---------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
