def build_constraint_matrix(f, active):
    n = len(f)
    m = len(active)
    A = np.zeros((m, n), dtype=np.float64)
    for i, idx in enumerate(active):
        k_start = max(0, idx - (n - 1))
        k_end = min(n - 1, idx)
        ks = np.arange(k_start, k_end + 1, dtype=int)
        js = idx - ks
        A[i, ks] = f[js]
    return A

def lp_maximize_sum_g(f, M, active):
    A = build_constraint_matrix(f, active)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(len(f), dtype=np.float64)
    bounds = [(0, None)] * len(f)
    # Use the "highs" method; allow tolerance
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None, None
    g = res.x
    # Ensure small negative entries are zero
    g = np.maximum(g, 0.0)
    sum_g = g.sum()
    return g, sum_g

"""
Propose a non‑negative step‑function (list of floats) with a low evaluation
score.

Algorithm
---------
* Choose a problem size n ≈ 5 000 (large enough to give a good bound,
  still cheap for many FFTs).
* Initialise a probability vector a (sum = 1).  Use the previously found
  construction if available, otherwise start from uniform or a random
  soft‑max initialisation.
* **Phase 1 – stochastic L‑BFGS:**  multi‑restart continuation L‑BFGS on a
  smooth log‑sum‑exp approximation of the max‑convolution.  This quickly
  produces a decent shape.
* **Phase 2 – active‑set LP refinement:**  for the current a compute
  M = max(conv(a,a)).  Build the linear programme
        maximise  ∑ g_j   subject to  (a * g)_i ≤ M  for all i where the bound
  is tight (active set).  Solve the LP with SciPy’s HiGHS solver, obtain
  a vector g with larger ℓ₁‑norm, rescale it to sum = 1 and mix
        a ← (1‑t)·a + t·g   with a small step t.  Repeat until no
  improvement is possible or time runs out.
* **Phase 3 – projected sub‑gradient descent:**  a standard sub‑gradient
  method on the exact objective 2·n·max(conv), projected onto the simplex.
* **Phase 4 – aggressive hill‑climb:**  simple greedy moves that transfer
  a random fraction of mass from one index to another while improving the
  fast score.
* Return the final sequence (list of non‑negative floats).  The evaluation
  function is scale‑invariant, so the sum of the returned numbers is
  irrelevant (the harness will normalise it internally).

The combination of a global L‑BFGS search, an exact LP step that exploits
the linear structure of the convolution constraints, and final local
search gives a better bound than the previous baseline.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes:
        random – normal distribution
        exp    – exponential decay
        power  – power‑law decay
        edge   – mass at the two edges
        sidon  – Sidon‑set with equal mass
        binary – random 0/1 mask with uniform mass on the 1's
        bump   – a single Gaussian bump
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums or 2 * candidate in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (equivalently minimise max(conv) with sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))                       # index of maximal entry
        # Sub‑gradient of that entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def build_constraint_matrix(f: np.ndarray, active: list[int]) -> np.ndarray:
    """
    Build a dense matrix A where A[i,j] is the coefficient of g_j in the
    convolution (f * g) at index active[i].
    """
    n = f.shape[0]
    m = len(active)
    A = np.zeros((m, n), dtype=np.float64)
    for row, idx in enumerate(active):
        # k runs over all positions of g that contribute to conv[idx]
        k_start = max(0, idx - (n - 1))
        k_end = min(n - 1, idx)
        ks = np.arange(k_start, k_end + 1, dtype=int)
        js = idx - ks
        A[row, ks] = f[js]
    return A


def lp_maximize_sum_g(f: np.ndarray, M: float, active: list[int]):
    """
    Solve   maximise 1ᵀg   s.t.   (f * g)_i ≤ M  for i∈active ,   g ≥ 0.
    Returns the optimal g and its ℓ₁‑norm.
    """
    A = build_constraint_matrix(f, active)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)  # minimise -∑g = maximise ∑g
    bounds = [(0, None)] * f.shape[0]

    # HiGHS is fast for moderate sized LPs
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        return None, None
    g = np.maximum(res.x, 0.0)   # clip tiny negatives
    return g, g.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we finish before the harness timeout
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS (≈ 30 % of time budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=30))
    modes = ["random", "exp", "power", "edge", "sidon", "binary", "bump", "best"]
    phase1_deadline = start_time + 0.30 * budget_s
    iteration = 0

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage=300
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[L‑BFGS {iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – active‑set LP refinement (≈ 45 % of time budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * budget_s
    print("[LP] starting active‑set linear‑programming refinement")
    while time.time() < phase2_deadline:
        conv_a = np.convolve(best_a, best_a)
        M = conv_a.max()
        # indices where the bound is (almost) tight
        tol = 1e-12
        active = np.where(conv_a >= M - tol)[0].tolist()

        g, sum_g = lp_maximize_sum_g(best_a, M, active)
        if g is None or sum_g is None:
            print("[LP] LP failed – exiting LP phase")
            break
        # If the LP cannot increase the ℓ₁‑norm, we have reached a fixed point
        if sum_g <= 1.0 + 1e-12:
            print("[LP] no increase in sum(g); terminating LP phase")
            break

        # Normalise g to keep the total mass = 1
        g_scaled = g / sum_g

        # Try a few mixing coefficients
        improved = False
        for t in (0.3, 0.2, 0.1, 0.05):
            a_new = (1 - t) * best_a + t * g_scaled
            a_new = project_onto_simplex(a_new)   # safe‑guard against numerical drift
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-9:
                best_a = a_new
                best_score = new_score
                best_y = np.log(best_a + 1e-12)
                print(f"[LP] mixed with t={t:.2f}, new fast score = {best_score:.6f}")
                improved = True
                break
        if not improved:
            # LP gave a better direction but mixing does not improve – stop
            print("[LP] mixing did not improve – exiting LP phase")
            break

    # --------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent (≈ 15 % of time budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 5.0, start_time + 0.90 * budget_s)
    print("[subgrad] final projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a, fft_len_conv, n, subgrad_deadline, eta0=0.50, max_iters=500_000
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 4 – aggressive hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=30_000, max_move=0.30
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
