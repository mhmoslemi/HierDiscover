def entropic_descent(a_start, fft_len, n, deadline, eta0=1.0):

    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    it=0

    while time.time()<deadline:
        it+=1
        # compute smoothed gradient at current a? Or true subgradient of max conv? Use subgradient (g vector computed for argmax conv index).
        conv = conv_fft(a, fft_len, n)
        m = np.argmax(conv)
        counterpart = m - np.arange(n)
        mask = (counterpart >=0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2*a[counterpart[mask]]
        eta = eta0 / np.sqrt(it)   # or linear step.
        # multiplicative update
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum>0:
            a /= a_sum
        else:
            a = best # fallback
        # evaluate
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
    return best, best_score

def hill_climb_anneal(a, cur_score, fft_len, n, deadline, max_iter=20000, max_move=0.2, init_temp=1e-3, final_temp=1e-8):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose delta
        delta = best[i] * rng.random() * max_move
        if delta <= 0:
            continue
        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score:
            best = new_a
            best_score = new_score
        else:
            # compute temperature
            elapsed = time.time() - start
            total = deadline - start
            if total <= 0:
                T = 0.0
            else:
                # linear cooling
                T = init_temp + (final_temp - init_temp) * (elapsed / total)
                # chance to accept worse
                prob = np.exp((best_score - new_score) / max(T, 1e-12))
                if rng.random() < prob:
                    best = new_a
                    best_score = new_score
        # optionally adapt max_move?
    return best, best_score
```

"""
Propose a low‑score step‑function.

The algorithm works on the probability simplex (sum = 1) – the actual
scale cancels out in the evaluation.  It consists of three phases:

1.  Multi‑restart continuation L‑BFGS on a log‑sum‑exp smoothed
    approximation of the max‑convolution.  The smoothing factor is
    driven down to ≈10⁻⁹, so the final stage works on the true
    non‑smooth objective.

2.  A projected *exponentiated‑gradient* (multiplicative) descent that
    follows the exact sub‑gradient of max(conv).  This often yields a
    small but systematic improvement beyond L‑BFGS.

3.  A long aggressive hill‑climb (greedy transfer of mass) that
    finishes the remaining time budget.  After the hill‑climb the
    sequence is symmetrised (a ← (a+rev(a))/2) – symmetry never hurts
    the max‑convolution and often lowers it.

The search runs for at most the supplied `budget_s` seconds and returns
the best sequence found.  The code is deliberately self‑contained;
only `numpy`, `scipy.optimize.minimize` and the provided
`evaluate_sequence` are required.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation for a probability vector (sum = 1):
          2·n·max(conv)    (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor  (plus 1e‑12 for safety).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sym":
        # symmetric distribution: generate half and mirror
        half_len = n // 2
        half = rng.random(half_len) + 1e-3
        half /= half.sum()
        if n % 2 == 0:
            a0 = np.concatenate([half, half[::-1]])
        else:
            middle = rng.random() + 1e-3
            a0 = np.concatenate([half, [middle], half[::-1]])
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

# Local search helpers

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Accepted only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def entropic_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.60,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected *exponentiated‑gradient* descent on the true objective
    f(a) = 2·n·max(conv).  It uses the exact sub‑gradient of the
    maximal convolution entry.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))          # index of the maximal entry

        # sub‑gradient of that entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # multiplicative update (exponentiated gradient)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = best.copy()   # fallback – should not happen

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
    return best, best_score


def hill_climb_anneal(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 50000,
    max_move: float = 0.25,
    init_temp: float = 1e-3,
    final_temp: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb with a tiny simulated‑annealing acceptance rule.
    """
    current = a.copy()
    current_score = cur_score
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = current[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        new = current.copy()
        new[i] -= delta
        new[j] += delta
        new_score = fast_score(new, fft_len, n)

        if new_score < current_score:
            current = new
            current_score = new_score
            if new_score < best_score:
                best = new
                best_score = new_score
        else:
            # temperature linearly decays
            elapsed = time.time() - start
            total = deadline - start
            if total <= 0.0:
                T = 0.0
            else:
                T = init_temp + (final_temp - init_temp) * (elapsed / total)
            # Metropolis acceptance
            if T > 0.0:
                prob = np.exp((current_score - new_score) / T)
                if rng.random() < prob:
                    current = new
                    current_score = new_score
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # Prefer to reuse a previously found construction if it exists.
    init_seq = globals().get("height_sequence_1", None)

        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)
        else:
            n = 6000
    else:
        n = 6000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart continuation L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=38))
    maxiter_per_stage = 500
    phase1_deadline = start_time + 0.45 * budget_s
    modes = ["random", "exp", "power", "edge", "sidon",
             "binary", "bump", "sym", "best"]
    iteration = 0

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)

        # continuation L‑BFGS

            best_y = y_opt.copy()

        iteration += 1

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a_sym = (best_a + best_a[::-1]) / 2.0

    return final_seq
