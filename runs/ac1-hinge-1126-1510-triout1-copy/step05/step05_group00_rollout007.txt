def lp_refine(f, S, deadline, max_iter=20):
    n = f.size
    fft_len = next_pow2(2*n-1)
    best_f = f.copy()
    best_M = compute_maxconv(f, fft_len)
    it=0
    while time.time() < deadline and it < max_iter:
        M = best_M
        # build A
        A = diags(np.tile(best_f, (n, 1)), offsets=np.arange(n), shape=(2*n-1, n), format='csc')
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[lp] linprog failed at iter", it, res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break
        # scale to S
        g_scaled = g * (S / sum_g)
        # line search for t
        t = 0.5
        improved = False
        while t > 1e-4:
            new_f = (1-t)*best_f + t*g_scaled
            # ensure nonnegative (should be)
            # compute M_new
            M_new = compute_maxconv(new_f, fft_len)
            if M_new < best_M - 1e-8:
                best_f = new_f
                best_M = M_new
                improved = True
                break
            t *= 0.5
        if not improved:
            # cannot improve further
            break
        it += 1
        print(f"[lp refine] iter {it}, M={best_M:.6f}")
    return best_f, best_M

"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm works in three stages:

1. **Continuous optimisation** – Several random‑restart L‑BFGS
   optimisations on a smooth “log‑sum‑exp” approximation of the
   max‑convolution, together with a short projected sub‑gradient/Adam
   phase.  All vectors are kept on the probability simplex (∑a_i = 1;
   the final objective is invariant under scaling).

2. **Linear‑program refinement** – Given the current distribution f,
   we solve a linear program that maximises the total mass of a second
   distribution g under the hard constraint `conv(f,g) ≤ M`
   (the current worst‑case convolution value).  After rescaling g to
   sum 1 we mix `f ← (1‑t)·f + t·g` with a line‑search on t, which
   strictly reduces the max‑convolution.  Repeating this step a few
   times typically pushes the score well below the previous best
   (≈1.508).

3. **Final greedy hill‑climb** – A fast local search that moves a
   small fraction of mass between two indices while the objective
   improves.

The whole procedure respects the 1000 s time budget and returns the
best sequence found before the deadline.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic FFT helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the self‑convolution of a."""
    return float(full_conv(a, fft_len).max())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv(a,a)) – the exact evaluation when Σa = 1."""
    return 2.0 * n * conv_max(a, fft_len)


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z    # d smooth_max / d conv_i

    # back‑prop through convolution: d conv / d a = (a_rev * mask) in freq domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives used in the first stage
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Continuation L‑BFGS over a decreasing list of smoothing parameters.
    Returns the best distribution (∑=1) found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with momentum, staying on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max‑convolution objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing that moves a random fraction of mass between two
    indices, accepting worsening moves under a decaying temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10            # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another.  Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse family of random points on the probability simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑program refinement (stage 2 of the algorithm)
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray,
              n: int,
              fft_len: int,
              deadline: float,
              max_iter: int = 15,
              tol: float = 1e-8):
    """
    LP‑based refinement:
    * f – current distribution (∑f = 1)
    * Solve   max Σ g_i   s.t.  conv(f,g) ≤ M,  g ≥ 0
    * Rescale g to sum = 1 and mix f ← (1‑t)f + t g with a line‑search.
    Repeats until time runs out or no improvement.
    Returns the refined distribution and the corresponding fast_score.
    """
    best_f = f.copy()
    best_M = conv_max(best_f, fft_len)          # maximum entry of f*f
    best_score = fast_score(best_f, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        M = best_M

        # ----- build the convolution matrix A (size (2n‑1) × n) -----
        # each column j contains the constant f_i on diagonal offset i
        offsets = np.arange(n, dtype=int)
        # each diagonal is a constant vector equal to the corresponding f[i]
        data = np.tile(best_f, (n, 1))          # shape (n, n), column j = f
        A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # ----- linear program: maximise sum(g) under conv(f,g) ≤ M -----
        c = -np.ones(n)                         # minimising –sum(g) ⇔ maximise sum(g)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        # clean up the matrix to free memory
        del A

        if not res.success:
            print(f"[lp] Linprog failed at iter {it}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break

        # ----- rescale g to have sum = 1 -----
        g = g / sum_g

        # ----- line‑search on mixing weight t -----
        t = 0.5
        improved = False
        while t > 1e-4 and time.time() < deadline:
            new_f = (1.0 - t) * best_f + t * g
            # sum stays exactly 1 by construction
            M_new = conv_max(new_f, fft_len)
            if M_new < best_M - tol:
                best_f = new_f
                best_M = M_new
                best_score = fast_score(best_f, fft_len, n)
                improved = True
                print(f"[lp] iter {it+1}, t={t:.4f}, M={best_M:.6f}")
                break
            t *= 0.5

        if not improved:
            # no progress – stop the refinement loop
            break

        it += 1

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial distribution
    # ------------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = max_n
            # linear interpolation to length n
            old_n = init_seq.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS heavy phase
    phase2_end = start_time + 0.78 * budget_s   # sub‑gradient + Adam
    phase3_end = start_time + 0.92 * budget_s   # LP refinement
    # remainder for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – Random‑restart continuation L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        # occasional perturbation of the current best
        if best_score < np.inf and rng.random() < 0.12:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len,
                                              fft_len,
                                              n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Momentum sub‑gradient + Adam fine‑tune
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                          phase2_end,
                                                          fft_len,
                                                          fft_len,
                                                          n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    if time.time() < phase2_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len,
                                           fft_len,
                                           phase2_end,
                                           lam_factor=0.002,
                                           lr=0.14)
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Linear‑program refinement (the new ingredient)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = lp_refine(best_a,
                                       n,
                                       fft_len,
                                       phase3_end,
                                       max_iter=12)
        print(f"[LP refine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Short simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.25
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len,
                                    n,
                                    final_deadline,
                                    max_iter=8000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Scale to the required sum Σa = √(2 n) and final evaluation
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_a * target_sum).tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
