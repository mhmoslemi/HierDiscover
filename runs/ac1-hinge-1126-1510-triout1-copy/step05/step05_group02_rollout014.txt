def generate_shape(n, rng):
    mode = rng.choice([...])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp_left':
        rate = rng.uniform(0.001, 0.2) * (1.0 / n) # maybe scale
        a = np.exp(-rate * np.arange(n))
    elif mode == 'exp_right':
        rate = rng.uniform(0.001, 0.2) * (1.0 / n)
        a = np.exp(-rate * np.arange(n-1, -1, -1))
    elif mode == 'beta':
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        x = np.linspace(0, 1, n)
        a = np.power(x, alpha-1) * np.power(1-x, beta-1)
    elif mode == 'beta_skew':
        # same as beta (alpha,beta) can be skewed
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        x = np.linspace(0, 1, n)
        a = np.power(x, alpha-1) * np.power(1-x, beta-1)
    elif mode == 'gaussian':
        mean = rng.uniform(0, n-1)
        sigma = rng.uniform(n/20, n/2)
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - mean) / sigma) ** 2)
    elif mode == 'bimodal':
        # two Gaussians
        mean1 = rng.uniform(0, n/2)
        mean2 = rng.uniform(n/2, n-1)
        sigma = rng.uniform(n/20, n/5)
        i = np.arange(n)
        a = np.exp(-0.5 * ((i-mean1)/sigma)**2) + np.exp(-0.5 * ((i-mean2)/sigma)**2)
    elif mode == 'powerlaw':
        p = rng.uniform(0.5, 3.0)
        a = 1.0 / (np.arange(1, n+1) ** p)
    elif mode == 'random_noise':
        a = rng.random(n)
    else:
        ...
    # ensure non-negative
    a[a < 0] = 0
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a
```

"""
search for a non‑negative step sequence minimizing
    2·n·max(conv(a,a)) / (sum(a)**2)

The program works in several phases:

1. **Random shape search** – generate many parametrised families
   (exponential, Gaussian, Beta, power‑law, bimodal, etc.) on a
   moderately large n (≈4000).  Keep the top‑k candidates according
   to the fast (FFT based) score.

2. **L‑BFGS continuation** – starting from the best shapes run a
   continuation of the smooth‑max objective (log‑sum‑exp) with a
   decreasing smoothing parameter.  The optimisation is performed on
   the soft‑max logits, guaranteeing the simplex constraints.

3. **Exponentiated‑gradient mirror descent** – a sub‑gradient method
   targeting the exact max‑convolution.  It works directly on the
   simplex and quickly reduces the maximal autocorrelation.

4. **Adam fine‑tuning** – solves the same smooth objective with an
   adaptive step‑size to polish the result.

5. **Greedy hill‑climb** – small mass transfers between coordinates
   while preserving the sum; accepts only improving moves.

All phases respect the overall 1000 s budget and the best sequence
found is returned.  The algorithm is deliberately stochastic; a
different seed yields a different search trajectory.
"""

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Utility functions
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation 2·n·max(a*a) assuming sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# -------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective and its gradient
# -------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# -------------------------------------------------------------
# L‑BFGS continuation on the smoothed max
# -------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)    # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            # keep going with the current y
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------
# Exponentiated‑gradient mirror descent (exact max)
# -------------------------------------------------------------


def exp_mirror_descent(a0: np.ndarray,
                       deadline: float,
                       fft_len_conv: int,
                       n: int):
    """
    Projected exponentiated‑gradient descent on the exact max‑convolution.
    Operates directly on the simplex. Returns best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    eta0 = 0.7  # initial learning‑rate (larger than 1.0 can be unstable)

    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())

        # sub‑gradient of the max (exact, not smoothed)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]   # d max / d a_i

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# -------------------------------------------------------------
# Adam optimisation on the smoothed objective
# -------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.005,
                  lr: float = 0.10):
    """
    Adam optimisation of the smoothed max objective starting from logits y0.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# -------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# -------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy mass‑transfer hill‑climb.
    Pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# -------------------------------------------------------------
# Random shape families (for the initial stochastic phase)
# -------------------------------------------------------------


def random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex using a variety of
    parametric families (exponential, Gaussian, beta, power‑law, bimodal,
    etc.).  The returned array is normalised to sum = 1.
    """
    mode = rng.choice(['uniform', 'exp_left', 'exp_right',
                       'beta', 'gaussian', 'bimodal',
                       'powerlaw', 'random', 'half_sine',
                       'sin', 'mixed_exp'])

    if mode == 'uniform':
        a = np.full(n, 1.0 / n)

    elif mode == 'exp_left':
        # rate range comfortably includes 1/n
        rate = rng.uniform(0.00005, 0.005)
        a = np.exp(-rate * np.arange(n))

    elif mode == 'exp_right':
        rate = rng.uniform(0.00005, 0.005)
        a = np.exp(-rate * np.arange(n-1, -1, -1))

    elif mode == 'beta':
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        x = np.linspace(0.0, 1.0, n)
        a = np.power(x, alpha - 1) * np.power(1.0 - x, beta - 1)

    elif mode == 'gaussian':
        mean = rng.uniform(0.3, 0.7) * (n - 1)
        sigma = rng.uniform(n/40.0, n/4.0)
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - mean) / sigma) ** 2)

    elif mode == 'bimodal':
        # two Gaussians with possibly different heights
        mean1 = rng.uniform(0.1, 0.4) * (n - 1)
        mean2 = rng.uniform(0.6, 0.9) * (n - 1)
        sigma = rng.uniform(n/50.0, n/6.0)
        i = np.arange(n)
        a = (rng.uniform(0.4, 0.7) *
             np.exp(-0.5 * ((i - mean1) / sigma) ** 2) +
             rng.uniform(0.3, 0.6) *
             np.exp(-0.5 * ((i - mean2) / sigma) ** 2))

    elif mode == 'powerlaw':
        p = rng.uniform(0.5, 3.0)
        a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)

    elif mode == 'random':
        a = rng.random(n)

    elif mode == 'half_sine':
        a = np.sin(np.pi * np.arange(n) / (n - 1))

    elif mode == 'sin':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)

    elif mode == 'mixed_exp':
        # mixture of left and right exponentials
        rate1 = rng.uniform(0.00005, 0.005)
        rate2 = rng.uniform(0.00005, 0.005)
        weight = rng.random()
        a = (weight * np.exp(-rate1 * np.arange(n)) +
             (1 - weight) * np.exp(-rate2 * np.arange(n-1, -1, -1)))

    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n // 2
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n // 2
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)      # length for a*a
    fft_len_g = next_pow2(3 * n - 2)         # length for gradient‑related conv

    # initialise best with the (possibly warm) start
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # allocate phases (percentages of the whole budget)
    phase1_end = start_time + 0.20 * budget_s   # random shape search
    phase2_end = start_time + 0.55 * budget_s   # L‑BFGS continuation
    phase3_end = start_time + 0.80 * budget_s   # mirror‑descent
    phase4_end = start_time + 0.95 * budget_s   # Adam fine‑tuning
    # remaining time → hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random shape exploration
    # ------------------------------------------------------------------
    top_k = 12
    top_scores = np.full(top_k, np.inf)
    top_arrays = [None] * top_k

    while time.time() < phase1_end:
        a_cand = random_shape(n, rng)
        sc = fast_score(a_cand, fft_len_conv, n)
        # keep the best `top_k` candidates
        worst_idx = int(np.argmax(top_scores))
        if sc < top_scores[worst_idx]:
            top_scores[worst_idx] = sc
            top_arrays[worst_idx] = a_cand.copy()
        # also keep a global best
        if sc < best_score:
            best_score, best_a = sc, a_cand.copy()
            print(f"[shape] new global best score={best_score:.6f}")

    # sort the top candidates for later use
    sorted_idx = np.argsort(top_scores)
    top_scores = top_scores[sorted_idx]
    top_arrays = [top_arrays[i] for i in sorted_idx if top_arrays[i] is not None]

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS continuation on the best shapes
    # ------------------------------------------------------------------
    lam_schedule = [0.6, 0.45, 0.33, 0.25, 0.18,
                    0.13, 0.09, 0.06, 0.04, 0.025,
                    0.015, 0.009, 0.005, 0.003, 0.001]
    maxiter_per_lam = 2000

    # try each top candidate (up to a small number)
    for a_start in top_arrays[:5]:
        if time.time() >= phase2_end:
            break
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase2_end,
                                           fft_len_conv,
                                           fft_len_g,
                                           n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient mirror descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        md_deadline = phase3_end - 0.2
        best_a, best_score = exp_mirror_descent(best_a,
                                                md_deadline,
                                                fft_len_conv,
                                                n)
        print(f"[mirror] after mirror descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam fine‑tuning on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        adam_deadline = phase4_end - 0.2
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           adam_deadline,
                                           lam_factor=0.007,
                                           lr=0.12)
        print(f"[Adam] after Adam score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (until the very end)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=25000)
    print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert to list and evaluate with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
