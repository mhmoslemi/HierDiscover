```
def projected_subgradient(a, max_iter, deadline):
    best_a = a
    best_score = fast_score(a)
    for t in range(max_iter):
        if time.time() >= deadline: break
        conv = full_conv(a, fft_len)
        k = conv.argmax()
        # compute gradient for max
        g = np.zeros_like(a)
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] = 2 * a[j]   # as above
        # step size decreasing
        step = 1.0/(np.sqrt(t+1) * (np.linalg.norm(g) + 1e-12))
        a_new = a - step * g
        # project back to simplex
        a_new = simplex_projection(a_new)
        # evaluate
        score = fast_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

```
def random_spectral_initial(n, rng):
    N = next_pow2(2*n)
    phases = rng.uniform(0, 2*np.pi, N)
```

"""
Propose a low‑score non‑negative step sequence.

Algorithm outline
-----------------
1. Choose a problem size `n` (default 2000).  All vectors are kept on the
   probability simplex (sum = 1) because the evaluation function is
   scale‑invariant.
2. Run many *continuation* L‑BFGS optimisations on a smoothed max‑convolution
   objective (log‑sum‑exp smoothing).  The optimisation is restarted from a
   diverse set of initialisations (uniform, exponential, power‑law,
   sinusoidal and a “flat‑spectrum” random start).
3. Refine the best candidate with a long exponentiated‑gradient (mirror) descent
   on the exact non‑smooth maximum of the convolution.
4. Perform a few extra steps of a projected sub‑gradient method (Euclidean
   projection onto the simplex) to push the peak down further.
5. Finish with a short simulated‑annealing stage and a greedy hill‑climb that
   moves a tiny amount of mass between two coordinates only when the move
   strictly improves the fast score.
6. Return the best sequence found before the global deadline.

All phases respect the time budget (`budget_s`, default 1000 s) and print
intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize

# -----------------------------------------------------------------------
# Utility functions (no closures, top‑level)
# -----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the objective when `a` is already normalised
    to sum = 1:  `2·n·max(a*a)`.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -----------------------------------------------------------------------
# Random initialiser – a mixture of families for diversity
# -----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector on `{0,…,n‑1}`."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sine",
            "halfsine",
            "flat_fft",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sine":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    elif mode == "halfsine":
        a = np.sin(np.pi * np.arange(n) / (n - 1))
    else:  # flat_fft – random phases, flat magnitude → roughly white spectrum
        N = next_pow2(2 * n)
        phases = rng.uniform(0, 2 * np.pi, N)
        fft_vec = np.exp(1j * phases)
        seq = np.fft.ifft(fft_vec).real
        a = seq[:n]
        a -= a.min()  # shift to non‑negative
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# -----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) and its gradient
# -----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Returns the smoothed objective value and its gradient w.r.t. logits `y`.
    The smoothing uses a log‑sum‑exp approximation of the max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through the convolution:
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the middle n entries correspond to derivatives w.r.t. a_i
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# -----------------------------------------------------------------------
# Continuation L‑BFGS (calls the smoothed objective)
# -----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each `lam` in `lam_schedule`, starting each stage from the
    result of the previous stage.  Returns the best sequence encountered.
    """
    # initialise logits from the start point
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            # continue with current y even if L‑BFGS failed
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# -----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# -----------------------------------------------------------------------


def exp_mirror_descent(a0: np.ndarray,
                       deadline: float,
                       fft_len_conv: int,
                       n: int):
    """
    Exponentiated‑gradient (mirror) descent on the true max‑convolution.
    Operates directly on the simplex and keeps the best solution seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv)

    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())

        # sub‑gradient of the max (ordered pairs)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update & renormalisation
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# -----------------------------------------------------------------------
# Projected sub‑gradient descent (Euclidean projection onto simplex)
# -----------------------------------------------------------------------


def projected_subgradient(a0: np.ndarray,
                          deadline: float,
                          fft_len_conv: int,
                          n: int,
                          max_iter: int = 10_000):
    """
    Simple Euclidean projected sub‑gradient descent on the exact max.
    Accepts only improving moves (no acceptance of worsening steps).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv)

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())

        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        # step size ∝ 1/√t  (scaled by a normalising factor)
        step = 0.5 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # keep the old a if no improvement
            pass
        it += 1
    return best_a, best_score


# -----------------------------------------------------------------------
# Adam optimisation on the smoothed objective (fine tuning)
# -----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective, starting from logits `y0`.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# -----------------------------------------------------------------------
# Simulated annealing (small random mass shuffles)
# -----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accepts worsening moves with a temperature schedule.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T = max(T * 0.92, 1e-5)
    return best_a, best_score


# -----------------------------------------------------------------------
# Greedy hill‑climb (mass transfers)
# -----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy hill‑climb: repeatedly move a small fraction of mass from one
    coordinate to another; accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Main entry point used by the evaluation harness
# -----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # -------------------------------------------------------------------
    # Timing set‑up
    # -------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # -------------------------------------------------------------------
    # Problem size (may be warm‑started from a previous solution)
    # -------------------------------------------------------------------
    max_n = 3000                                 # we stay comfortably below 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n // 2
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n // 2
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a convolution
    fft_len_g = next_pow2(3 * n - 2)      # auxiliary length for gradient passes

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # -------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts (≈60 % of the budget)
    # -------------------------------------------------------------------
    phase1_end = start_time + 0.60 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.15, 0.10,
        0.07, 0.045, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001,
    ]
    maxiter_per_lam = 500

    while time.time() < phase1_end:

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:

            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # -------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient mirror descent (≈20 % of budget)
    # -------------------------------------------------------------------
    if time.time() < deadline:

            md_deadline,
            fft_len_conv,

    # -------------------------------------------------------------------

            fft_len_conv,
            n,
            max_iter=5000,
        )

    # -------------------------------------------------------------------

            logits,
            n,
            fft_len_conv,
            fft_len_g,
            adam_deadline,
            lam_factor=0.002,
            lr=0.15,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # -------------------------------------------------------------------
    # Phase 5 – simulated annealing (≈3 % of budget)
    # -------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = start_time + 0.95 * budget_s
        best_a, best_score = random_anneal(
            best_a,
            best_score,
            fft_len_conv,
            n,
            anneal_deadline,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # -------------------------------------------------------------------
    # Phase 6 – final greedy hill‑climb (remainder of time)
    # -------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=10000,
    )

    # -------------------------------------------------------------------
    # Produce final output
    # -------------------------------------------------------------------
    final_seq = best_a.tolist()

        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as exc:  # safety – in case the environment does not expose it yet
        print("[final] could not evaluate inside the search function:", exc)

    return final_seq
